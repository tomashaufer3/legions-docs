How it works

The merged volume in library/education/ is a 279-page image scan with no text layer, so every later reference to it — a transcription row, a check, a page cite in EDUCATION_LADDER.md — has to address a PDF page, while the source itself is numbered in two independent sequences: roman for the Einleitung and arabic for the tabellarischer Teil. This maps one onto the other and names what is on each page, so nothing downstream has to re-derive the offset — which is not constant: it shifts at each of the two gaps in the scan.

Two pages of the source were never downloaded and are recorded as rows with an empty pdf_page rather than silently skipped: Einleitung X-XI and table page 15 (which breaks Table 6).

The section and table anchors are read off the printed Inhalt des 1. Heftes (PDF page 2), not guessed.

def main() -> int:
    data = list(rows())
    OUT.parent.mkdir(parents=True, exist_ok=True)
    with open(OUT, "w", newline="", encoding="utf-8") as fh:
        writer = csv.DictWriter(fh, fieldnames=FIELDS)
        writer.writeheader()
        writer.writerows(data)

    mapped = [r["pdf_page"] for r in data if r["pdf_page"] != ""]
    gaps = [r["folio"] for r in data if r["pdf_page"] == ""]
    # Every PDF page is claimed exactly once, in order: the offset arithmetic is the only
    # thing this file exists to get right, so it is asserted rather than eyeballed.
    assert mapped == list(range(1, N_PAGES + 1)), "pdf_page must be 1..279, no gap or repeat"

    print(f"wrote {OUT.relative_to(ROOT)}")
    print(f"  rows             : {len(data)}")
    print(f"  mapped pdf pages : {len(mapped)} (1..{max(mapped)})")
    print(f"  gaps recorded    : {len(gaps)} (folios {', '.join(gaps)})")
    return 0

def _anchor(anchors, n):
    """The last anchor at or before ``n``; anchors must be sorted ascending."""
    hit = None
    for entry in anchors:
        if entry[0] <= n:
            hit = entry[1:]
        else:
            break
    return hit
