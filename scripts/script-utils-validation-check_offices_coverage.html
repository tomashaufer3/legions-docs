How it works

The route cuts a page into sections at its headings: Ortsbestandteile / Části obcí (parts, read for settlement types) and Ämter, Anstalten usw. / Úřady, ústavy (offices, read for the institution pictograms). On some pages where a court runs on from the page before, the cutter labels an offices section, or the offices half of a section, parts. The parts readers are asked for settlement and industry symbols only (lexikon_1910_vision.SECTION_CATEGORIES), so every pictogram under that heading is lost. No sealed, certification or gold page has the layout, so no score saw it. Found 2026-11-05 by comparing the two Böhmen editions.

Three signals per parts section of a full run, none of which reads an image:

zero_reads no reader read anything in the section, while DINOv3 found glyphs in it; dino_inst DINOv3, which runs on every section whatever its label, found glyphs of an offices-only category in it (OFFICES_ONLY) — a lower bound in the German volumes, where it leaves most glyphs unclassed (D-115, D-191); blind_munis the municipalities on that page of a court with only parts sections there — at least BLOCK_MIN of them — carry no pictogram at all in register_institutions.

A section with any signal is a candidate; the list is confirmed by eye before notebook 17 re-reads anything (D-192). For Böhmen it also measures the two editions against each other over the municipalities the crosswalk pairs (D-150): those blind on one side, and per symbol the agreement on the pairs both editions read.

Validation only; reads the full runs' ledgers and section file names and data/final/register, writes output/validation/offices_coverage.{json,txt}, offices_coverage_candidates.csv and offices_coverage_regions.csv (each offices region of a flagged section, by its heading: what notebook 17 re-reads).

def ledger_counts(edition: str) -> tuple[Counter, dict[str, Counter]]:
    """``(reader reads per section, DINOv3 classes per section)`` off the run's mark ledger."""
    reads, dino = Counter(), defaultdict(Counter)
    path = FROZEN / f"{edition}_full" / "ledger" / f"ledger_{edition}_full.csv"
    with open(path, encoding="utf-8", newline="") as fh:
        for r in csv.DictReader(fh):
            if ".png#" not in r["mark"]:
                continue                      # an entry-level token, not a boxed mark
            image = r["mark"].split(".png#")[0]
            if r["step"] == "read" and r["read"] not in ("", "—"):
                reads[image] += 1
            elif r["step"] == "dinov3":
                dino[image][r["read"]] += 1
    return reads, dino

def blind_blocks(edition: str, places: pd.DataFrame, pictogram_keys: set) -> pd.DataFrame:
    """A court's municipalities on one page (by the page their row starts on), with how many
    carry any pictogram."""
    mun = places[(places.edition == edition) & (places.place_level == "municipality")].copy()
    mun["page"] = mun.pages.astype(str).str.split(";").str[0]

def candidates(edition: str, places: pd.DataFrame, pictogram_keys: set) -> tuple[pd.DataFrame, dict]:
    """Every parts section with a signal, and the edition's summary."""
    sec = sections(edition)
    reads, dino = ledger_counts(edition)
    kinds = sec.groupby(["page", "slug"]).kind.agg(set)
    parts_only_pages = {p for (p, _), k in kinds.items() if k == {"parts"}}
    blocks = blind_blocks(edition, places, pictogram_keys)
    blind = blocks[(blocks.with_pictogram == 0) & (blocks.munis >= BLOCK_MIN)]
    blind_on = blind[blind.page.isin(parts_only_pages)].groupby("page").munis.sum()
    parts = sec[sec.kind == "parts"].copy()
    parts["reads"] = parts.section.map(lambda s: reads[s])
    parts["dino_glyphs"] = parts.section.map(lambda s: sum(dino[s].values()))
    parts["dino_inst"] = parts.section.map(lambda s: sum(n for t, n in dino[s].items() if t in OFFICES_ONLY))
    parts["dino_inst_top"] = parts.section.map(lambda s: ", ".join(
        f"{t} {n}" for t, n in dino[s].most_common() if t in OFFICES_ONLY)[:80])
    parts["parts_only_page"] = [kinds[(p, s)] == {"parts"} for p, s in zip(parts.page, parts.slug)]
    parts["blind_munis"] = [int(blind_on.get(p, 0)) if po else 0
                            for p, po in zip(parts.page, parts.parts_only_page)]
    parts["zero_reads"] = (parts.reads == 0) & (parts.dino_glyphs > 0)
    signal = parts.zero_reads | (parts.dino_inst >= DINO_INST_MIN) | (parts.blind_munis > 0)
    cand = parts[signal].assign(edition=edition)
    mun = places[(places.edition == edition) & (places.place_level == "municipality")]
    summary = {"municipalities": len(mun),
               "no_pictogram": int(sum((edition, k, m) not in pictogram_keys
                                       for k, m in zip(mun.court_key, mun.muni_seq))),
               "parts_sections": len(parts), "offices_sections": int((sec.kind == "offices").sum()),
               "blind_blocks": len(blind), "blind_block_munis": int(blind.munis.sum()),
               "blind_blocks_on_parts_only_pages": int(blind.page.isin(parts_only_pages).sum()),
               "blind_blocks_elsewhere": blind[~blind.page.isin(parts_only_pages)][
                   ["court", "page", "munis"]].to_dict("records"),
               "candidates": len(cand), "zero_reads": int(cand.zero_reads.sum()),
               "dino_inst": int((cand.dino_inst >= DINO_INST_MIN).sum()),
               "blind_munis": int((cand.blind_munis > 0).sum())}
    return cand, summary

def bohemia_editions(places: pd.DataFrame, inst: pd.DataFrame, coops: pd.DataFrame) -> dict:
    """The two Böhmen editions over the municipalities the crosswalk pairs one to one."""
    de = places[(places.edition == "de-bohemia") & (places.place_level == "municipality")
                & places.cz_muni_seq.notna()]
    pairs = pd.DataFrame({"court_key": de.court_key, "de": de.muni_seq, "cz": de.cz_muni_seq.astype(int)})
    one_to_one = ~pairs.duplicated(["court_key", "cz"], keep=False) & ~pairs.duplicated(["court_key", "de"], keep=False)
    pairs = pairs[one_to_one]
    pic = inst[inst.source == "pictogram"]
    marks = {ed: pic[pic.edition == ed].groupby(["court_key", "muni_seq"]).mark.agg(set) for ed in ("cz", "de-bohemia")}
    de_coop = coops[coops.edition == "de-bohemia"].groupby(["court_key", "muni_seq"]).size()
    none = frozenset()
    cz = [marks["cz"].get((k, m), none) for k, m in zip(pairs.court_key, pairs.cz)]
    dm = [marks["de-bohemia"].get((k, m), none) for k, m in zip(pairs.court_key, pairs.de)]
    blind_de = [bool(c) and not d for c, d in zip(cz, dm)]
    blind_cz = [bool(d) and not c for c, d in zip(cz, dm)]
    both = [not (a or b) for a, b in zip(blind_de, blind_cz)]
    rows = {}
    for mark in COMPARED:
        c = [mark in s for s, ok in zip(cz, both) if ok]
        d = [mark in s for s, ok in zip(dm, both) if ok]
        n_both = sum(a and b for a, b in zip(c, d))
        n_cz, n_de = sum(a and not b for a, b in zip(c, d)), sum(b and not a for a, b in zip(c, d))
        rows[mark] = {"both": n_both, "cz_only": n_cz, "de_only": n_de,
                      "agree": round(n_both / max(1, n_both + n_cz + n_de), 3)}
    c = [("[cooperative]" in s) for s, ok in zip(cz, both) if ok]
    d = [(k, m) in de_coop.index for k, m, ok in zip(pairs.court_key, pairs.de, both) if ok]
    n_both = sum(a and b for a, b in zip(c, d))
    rows["[cooperative] / German letters"] = {
        "both": n_both, "cz_only": sum(a and not b for a, b in zip(c, d)),
        "de_only": sum(b and not a for a, b in zip(c, d)),
        "agree": round(n_both / max(1, sum(a or b for a, b in zip(c, d))), 3)}
    pair = {"[castle]", "[poorhouse]"}
    differ = [(c & pair) != (d & pair) for c, d, ok in zip(cz, dm, both) if ok]
    swapped = [("[poorhouse]" in d and "[poorhouse]" not in c and "[castle]" in c)
               for c, d, ok in zip(cz, dm, both) if ok]
    return {"pairs": len(pairs), "blind_in_german": sum(blind_de),
            "their_czech_pictograms": sum(len(s) for s, b in zip(cz, blind_de) if b),
            "blind_in_czech": sum(blind_cz), "both_read": sum(both), "by_symbol": rows,
            "castle_poorhouse_differ": sum(differ), "german_poorhouse_czech_castle": sum(swapped)}

def locate_headings(edition: str, section: str, tpl: dict) -> list[tuple[int, str, float]]:
    """``[(y, kind, score)]``, top to bottom: each line matching a heading template, as the kind
    it matches better."""
    import numpy as np
    img, found = _image(edition, section), []
    for kind in ("parts", "offices"):
        rows = _ncc(img, tpl[(TEMPLATE_OF[edition], kind)]).max(axis=1)
        for y in np.argsort(-rows):
            if rows[y] < MATCH_MIN:
                break
            if all(abs(y * SCALE - f[0]) > SAME_LINE or f[1] != kind for f in found):
                found.append((int(y) * SCALE, kind, round(float(rows[y]), 3)))
    keep = []
    for f in sorted(found):
        if keep and abs(f[0] - keep[-1][0]) < SAME_LINE:
            keep[-1] = max(keep[-1], f, key=lambda g: g[2])
        else:
            keep.append(f)
    return keep

def ruled_headings(located: list, ruled: list) -> list[tuple[int, str, float | None, str]]:
    """``[(y, kind, score, source)]``: the located headings with the eye's rulings applied. A ruling
    on the same line sets the kind (``none`` drops the match); one the locator missed is added."""
    out = []
    for y, kind, score in located:
        ruling = next((k for ry, k in ruled if abs(ry - y) <= SAME_LINE), None)
        if ruling != "none":
            out.append((y, ruling or kind, score, "eye" if ruling else "template"))
    for ry, k in ruled:
        if k != "none" and not any(abs(ry - y) <= SAME_LINE for y, *_ in out):
            out.append((ry, k, None, "eye"))
    return sorted(out)

def offices_regions(edition: str, flagged: set, heads: dict) -> list[dict]:
    """The offices regions inside the flagged parts sections. A court's sections are walked in
    reading order (page, then the cutter's index); an unflagged section keeps its label, a flagged
    one switches at each heading in it, and the text before its first heading continues the
    section before it — unless that heading opens a parts list within :data:`MARGIN` of the top."""
    sec = sections(edition)
    sec["index"] = sec.section.str.rsplit("_", n=1).str[1].astype(int)
    by_court = defaultdict(list)
    for s in sec.sort_values(["page", "index"]).itertuples():
        by_court[s.slug].append((s.section, s.kind))
    out = []
    for seq in by_court.values():
        state = None
        for section, kind in seq:
            if section not in flagged:
                state = kind
                continue
            cur, y_from, score, source = state, 0, None, ""
            for y, k, sc, src in heads.get(section, []):
                margin = y_from == 0 and k == "parts" and y <= MARGIN
                if cur == "offices" and y > y_from + SAME_LINE and not margin:
                    out.append(_region(edition, section, y_from, y, score, source))
                cur, y_from, score, source = k, y, sc, src
            if cur == "offices":
                out.append(_region(edition, section, y_from, None, score, source))
            state = cur
    return out

def eye_recall(located: dict, ruled: dict) -> dict:
    """The locator against the rulings: headings found, missed, matches the eye rejected, and
    matches no ruling covers (the ones a look must settle)."""
    hit, missed, rejected = 0, [], 0
    for section, rows in ruled.items():
        for y, kind in rows:
            match = any(abs(y - ly) <= SAME_LINE and (kind == lk or kind == "none")
                        for ly, lk, _ in located.get(section, []))
            if kind == "none":
                rejected += match
            elif match:
                hit += 1
            else:
                missed.append(f"{section} {kind} y={y}")
    unruled = [f"{s} {k} y={y} ({sc})" for s, found in located.items() for y, k, sc in found
               if not any(abs(y - ry) <= SAME_LINE for ry, _ in ruled.get(s, []))]
    return {"ruled": sum(len(r) for r in ruled.values()), "found": hit, "missed": missed,
            "rejected_matches": rejected, "unruled_matches": unruled}

    inst = pd.read_parquet(REGISTER / "register_institutions.parquet")
    coops = pd.read_parquet(REGISTER / "register_cooperatives.parquet")
    pic = inst[inst.source == "pictogram"]
    keys = set(zip(pic.edition, pic.court_key, pic.muni_seq))
    frames, summaries, located, heads, regions = [], {}, {}, {}, []
    tpl, ruled = templates(), eye_rulings()
    for ed in EDITIONS:
        cand, summaries[ed] = candidates(ed, places, keys)
        frames.append(cand)
        for section in cand.section:
            located[section] = locate_headings(ed, section, tpl)
            heads[section] = ruled_headings(located[section], ruled.get(section, []))
        regions += offices_regions(ed, set(cand.section), heads)
    reg = pd.DataFrame(regions, columns=["edition", "section", "y_from", "y_to", "how",
                                         "heading_score", "heading_source"])
    unconfirmed = (reg.heading_source == "template") & (reg.heading_score < CONFIRM_BELOW)
    rec = {"check": __version__, "run": time.strftime("%Y-%m-%d %H:%M:%S"),
           "offices_only_tokens": len(OFFICES_ONLY), "editions": summaries,
           "bohemia": bohemia_editions(places, inst, coops),
           "regions": {"regions": len(reg), "sections": int(reg.section.nunique()),
                       "by_edition": reg.groupby("edition").size().to_dict(),
                       "to_confirm": int(unconfirmed.sum())},
           "eye_recall": eye_recall(located, ruled)}
    OUT.mkdir(parents=True, exist_ok=True)
    reg.to_csv(OUT / "offices_coverage_regions.csv", index=False, encoding="utf-8")
    cols = ["edition", "section", "page", "slug", "reads", "dino_glyphs", "dino_inst", "dino_inst_top",
            "parts_only_page", "blind_munis", "zero_reads"]
    pd.concat(frames)[cols].to_csv(OUT / "offices_coverage_candidates.csv", index=False, encoding="utf-8")
    (OUT / "offices_coverage.json").write_text(json.dumps(rec, indent=1, ensure_ascii=False, default=str),
                                               encoding="utf-8")
    text = report(rec)
    (OUT / "offices_coverage.txt").write_text(text, encoding="utf-8")
    print(text)
    print("\n  written: output/validation/offices_coverage.{json,txt}, offices_coverage_candidates.csv, "
          "offices_coverage_regions.csv")
    return 0

"""Offices sections no reader read: a check on the section cutter, every full run (D-192).

The route cuts a page into sections at its headings: *Ortsbestandteile* / *Části obcí* (parts,
read for settlement types) and *Ämter, Anstalten usw.* / *Úřady, ústavy* (offices, read for the
institution pictograms). On some pages where a court runs on from the page before, the cutter
labels an offices section, or the offices half of a section, ``parts``. The parts readers are
asked for settlement and industry symbols only (``lexikon_1910_vision.SECTION_CATEGORIES``), so
every pictogram under that heading is lost. No sealed, certification or gold page has the
layout, so no score saw it. Found 2026-11-05 by comparing the two Böhmen editions.

Three signals per parts section of a full run, none of which reads an image:

``zero_reads``
    no reader read anything in the section, while DINOv3 found glyphs in it;
``dino_inst``
    DINOv3, which runs on every section whatever its label, found glyphs of an offices-only
    category in it (:data:`OFFICES_ONLY`) — a lower bound in the German volumes, where it
    leaves most glyphs unclassed (D-115, D-191);
``blind_munis``
    the municipalities on that page of a court with only parts sections there — at least
    :data:`BLOCK_MIN` of them — carry no pictogram at all in ``register_institutions``.

A section with any signal is a **candidate**; the list is confirmed by eye before notebook 17
re-reads anything (D-192). For Böhmen it also measures the two editions against each other over
the municipalities the crosswalk pairs (D-150): those blind on one side, and per symbol the
agreement on the pairs both editions read.

Validation only; reads the full runs' ledgers and section file names and ``data/final/register``,
writes ``output/validation/offices_coverage.{json,txt}``, ``offices_coverage_candidates.csv`` and
``offices_coverage_regions.csv`` (each offices region of a flagged section, by its heading: what
notebook 17 re-reads).

    python utils/validation/check_offices_coverage.py
"""
from __future__ import annotations

__version__ = "0.2.0"   # 2026-11-05: the headings located by template, the offices regions

import csv
import json
import re
import sys
import time
from collections import Counter, defaultdict
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from utils.lexikon_1910 import lexikon_1910_vision as vision  # noqa: E402

FROZEN = ROOT / "output" / "frozen"
REGISTER = ROOT / "data" / "final" / "register"
ABBREVIATIONS = ROOT / "reference" / "places" / "institution_abbreviations.csv"
OUT = ROOT / "output" / "validation"
EDITIONS = ("cz", "de-bohemia", "de-moravia", "de-silesia")
#: A court's municipalities on one page, fewer than this, can all lack a pictogram by chance.
BLOCK_MIN = 3
#: DINOv3 institution glyphs in a parts section before it counts as a signal.
DINO_INST_MIN = 3
#: Symbols compared across the two Böhmen editions (the cooperative separately: D-139).
COMPARED = ("[fire-brigade]", "[school]", "[post]", "[gendarme]", "[telegraph]", "[telephone]",
            "[rail-station]", "[revenue-guard]", "[customs]", "[castle]", "[poorhouse]")
_SECTION_RE = re.compile(r"(?P<page>\d{4})_(?P<slug>.+)_(?P<kind>parts|offices)_\d+$")


def offices_only() -> set[str]:
    """Pictogram tokens of a category only an offices section is asked for."""
    shared = set(vision.SECTION_CATEGORIES["parts"])
    rows = pd.read_csv(ABBREVIATIONS, dtype=str, keep_default_na=False)
    rows = rows[rows.kind == vision.PICTOGRAM_KIND]
    return set(rows[rows.category.isin(set(vision.SECTION_CATEGORIES["offices"]) - shared)].code)


OFFICES_ONLY = offices_only()


def sections(edition: str) -> pd.DataFrame:
    """One row per cut section image: page, district slug, kind."""
    rows = []
    for f in sorted((FROZEN / f"{edition}_full" / "sections").glob("*.png")):
        m = _SECTION_RE.match(f.stem)
        if m:
            rows.append({"section": f.stem, **m.groupdict()})
    return pd.DataFrame(rows)


def ledger_counts(edition: str) -> tuple[Counter, dict[str, Counter]]:
    """``(reader reads per section, DINOv3 classes per section)`` off the run's mark ledger."""
    reads, dino = Counter(), defaultdict(Counter)
    path = FROZEN / f"{edition}_full" / "ledger" / f"ledger_{edition}_full.csv"
    with open(path, encoding="utf-8", newline="") as fh:
        for r in csv.DictReader(fh):
            if ".png#" not in r["mark"]:
                continue                      # an entry-level token, not a boxed mark
            image = r["mark"].split(".png#")[0]
            if r["step"] == "read" and r["read"] not in ("", "—"):
                reads[image] += 1
            elif r["step"] == "dinov3":
                dino[image][r["read"]] += 1
    return reads, dino


def blind_blocks(edition: str, places: pd.DataFrame, pictogram_keys: set) -> pd.DataFrame:
    """A court's municipalities on one page (by the page their row starts on), with how many
    carry any pictogram."""
    mun = places[(places.edition == edition) & (places.place_level == "municipality")].copy()
    mun["page"] = mun.pages.astype(str).str.split(";").str[0]
    mun["pictogram"] = [(edition, k, m) in pictogram_keys for k, m in zip(mun.court_key, mun.muni_seq)]
    return mun.groupby(["court_key", "page"], dropna=False).agg(
        court=("court_cs", "first"), munis=("pictogram", "size"),
        with_pictogram=("pictogram", "sum")).reset_index()


def candidates(edition: str, places: pd.DataFrame, pictogram_keys: set) -> tuple[pd.DataFrame, dict]:
    """Every parts section with a signal, and the edition's summary."""
    sec = sections(edition)
    reads, dino = ledger_counts(edition)
    kinds = sec.groupby(["page", "slug"]).kind.agg(set)
    parts_only_pages = {p for (p, _), k in kinds.items() if k == {"parts"}}
    blocks = blind_blocks(edition, places, pictogram_keys)
    blind = blocks[(blocks.with_pictogram == 0) & (blocks.munis >= BLOCK_MIN)]
    blind_on = blind[blind.page.isin(parts_only_pages)].groupby("page").munis.sum()
    parts = sec[sec.kind == "parts"].copy()
    parts["reads"] = parts.section.map(lambda s: reads[s])
    parts["dino_glyphs"] = parts.section.map(lambda s: sum(dino[s].values()))
    parts["dino_inst"] = parts.section.map(lambda s: sum(n for t, n in dino[s].items() if t in OFFICES_ONLY))
    parts["dino_inst_top"] = parts.section.map(lambda s: ", ".join(
        f"{t} {n}" for t, n in dino[s].most_common() if t in OFFICES_ONLY)[:80])
    parts["parts_only_page"] = [kinds[(p, s)] == {"parts"} for p, s in zip(parts.page, parts.slug)]
    parts["blind_munis"] = [int(blind_on.get(p, 0)) if po else 0
                            for p, po in zip(parts.page, parts.parts_only_page)]
    parts["zero_reads"] = (parts.reads == 0) & (parts.dino_glyphs > 0)
    signal = parts.zero_reads | (parts.dino_inst >= DINO_INST_MIN) | (parts.blind_munis > 0)
    cand = parts[signal].assign(edition=edition)
    mun = places[(places.edition == edition) & (places.place_level == "municipality")]
    summary = {"municipalities": len(mun),
               "no_pictogram": int(sum((edition, k, m) not in pictogram_keys
                                       for k, m in zip(mun.court_key, mun.muni_seq))),
               "parts_sections": len(parts), "offices_sections": int((sec.kind == "offices").sum()),
               "blind_blocks": len(blind), "blind_block_munis": int(blind.munis.sum()),
               "blind_blocks_on_parts_only_pages": int(blind.page.isin(parts_only_pages).sum()),
               "blind_blocks_elsewhere": blind[~blind.page.isin(parts_only_pages)][
                   ["court", "page", "munis"]].to_dict("records"),
               "candidates": len(cand), "zero_reads": int(cand.zero_reads.sum()),
               "dino_inst": int((cand.dino_inst >= DINO_INST_MIN).sum()),
               "blind_munis": int((cand.blind_munis > 0).sum())}
    return cand, summary


def bohemia_editions(places: pd.DataFrame, inst: pd.DataFrame, coops: pd.DataFrame) -> dict:
    """The two Böhmen editions over the municipalities the crosswalk pairs one to one."""
    de = places[(places.edition == "de-bohemia") & (places.place_level == "municipality")
                & places.cz_muni_seq.notna()]
    pairs = pd.DataFrame({"court_key": de.court_key, "de": de.muni_seq, "cz": de.cz_muni_seq.astype(int)})
    one_to_one = ~pairs.duplicated(["court_key", "cz"], keep=False) & ~pairs.duplicated(["court_key", "de"], keep=False)
    pairs = pairs[one_to_one]
    pic = inst[inst.source == "pictogram"]
    marks = {ed: pic[pic.edition == ed].groupby(["court_key", "muni_seq"]).mark.agg(set) for ed in ("cz", "de-bohemia")}
    de_coop = coops[coops.edition == "de-bohemia"].groupby(["court_key", "muni_seq"]).size()
    none = frozenset()
    cz = [marks["cz"].get((k, m), none) for k, m in zip(pairs.court_key, pairs.cz)]
    dm = [marks["de-bohemia"].get((k, m), none) for k, m in zip(pairs.court_key, pairs.de)]
    blind_de = [bool(c) and not d for c, d in zip(cz, dm)]
    blind_cz = [bool(d) and not c for c, d in zip(cz, dm)]
    both = [not (a or b) for a, b in zip(blind_de, blind_cz)]
    rows = {}
    for mark in COMPARED:
        c = [mark in s for s, ok in zip(cz, both) if ok]
        d = [mark in s for s, ok in zip(dm, both) if ok]
        n_both = sum(a and b for a, b in zip(c, d))
        n_cz, n_de = sum(a and not b for a, b in zip(c, d)), sum(b and not a for a, b in zip(c, d))
        rows[mark] = {"both": n_both, "cz_only": n_cz, "de_only": n_de,
                      "agree": round(n_both / max(1, n_both + n_cz + n_de), 3)}
    c = [("[cooperative]" in s) for s, ok in zip(cz, both) if ok]
    d = [(k, m) in de_coop.index for k, m, ok in zip(pairs.court_key, pairs.de, both) if ok]
    n_both = sum(a and b for a, b in zip(c, d))
    rows["[cooperative] / German letters"] = {
        "both": n_both, "cz_only": sum(a and not b for a, b in zip(c, d)),
        "de_only": sum(b and not a for a, b in zip(c, d)),
        "agree": round(n_both / max(1, sum(a or b for a, b in zip(c, d))), 3)}
    pair = {"[castle]", "[poorhouse]"}
    differ = [(c & pair) != (d & pair) for c, d, ok in zip(cz, dm, both) if ok]
    swapped = [("[poorhouse]" in d and "[poorhouse]" not in c and "[castle]" in c)
               for c, d, ok in zip(cz, dm, both) if ok]
    return {"pairs": len(pairs), "blind_in_german": sum(blind_de),
            "their_czech_pictograms": sum(len(s) for s, b in zip(cz, blind_de) if b),
            "blind_in_czech": sum(blind_cz), "both_read": sum(both), "by_symbol": rows,
            "castle_poorhouse_differ": sum(differ), "german_poorhouse_czech_castle": sum(swapped)}


# ── where the offices start: the headings, located by template ─────────────────────────────
#: Each heading's own words, cut from one heading read by eye per edition
#: (:data:`EYE`): ``(section, y0, y1, x0, x1)`` at full scale. The cut stops before the words
#: the two headings share (*im Gerichtsbezirk*, *v soudním okresu*), which otherwise outweigh the
#: words that tell them apart. Schlesien has no heading of its own on the list and uses Mähren's.
TEMPLATES = {
    ("de-bohemia", "offices"): ("0152_novebenatky_parts_0", 1249, 1311, 344, 1041),   # Ämter, Anstalten usw.
    ("de-bohemia", "parts"): ("0152_novebenatky_parts_0", 221, 276, 457, 1132),       # Ortsbestandteile
    ("de-moravia", "offices"): ("0054_velkemezirici_parts_1", 213, 275, 305, 1170),
    ("de-moravia", "parts"): ("0054_velkemezirici_parts_0", 224, 281, 395, 1068),
    ("cz", "offices"): ("0317_vodnany_parts_3", 670, 730, 420, 1223),                 # Úřady, ústavy a t. d.
    ("cz", "parts"): ("0083_dvurkralovenadlabem_parts_0", 184, 244, 344, 744),        # Části osad
}
TEMPLATE_OF = {"cz": "cz", "de-bohemia": "de-bohemia", "de-moravia": "de-moravia", "de-silesia": "de-moravia"}
#: The headings read by eye on the candidate sections (2026-11-05): the locator's yardstick.
EYE = ROOT / "reference" / "places" / "offices_headings_by_eye.csv"
#: On the headings read by eye a true heading scores 0.465 or more and an entry line at most
#: 0.36; a match under :data:`CONFIRM_BELOW` is listed for a look before notebook 17 uses it.
MATCH_MIN, CONFIRM_BELOW = 0.42, 0.50
SCALE = 2          # matched at half scale
SAME_LINE = 40     # pixels: two matches closer than this are one line


def _image(edition: str, section: str):
    from PIL import Image
    import numpy as np
    im = Image.open(FROZEN / f"{edition}_full" / "sections" / f"{section}.png").convert("L")
    return np.asarray(im.resize((im.width // SCALE, im.height // SCALE)), dtype=float)


def _ncc(img, tpl):
    """Normalised cross-correlation of ``tpl`` over ``img`` (valid positions only)."""
    import numpy as np
    from scipy.signal import fftconvolve
    if img.shape[0] < tpl.shape[0] or img.shape[1] < tpl.shape[1]:
        return np.zeros((1, 1))
    t = tpl - tpl.mean()
    ones = np.ones_like(tpl)
    num = fftconvolve(img, t[::-1, ::-1], mode="valid")
    s, s2 = fftconvolve(img, ones, mode="valid"), fftconvolve(img ** 2, ones, mode="valid")
    return num / (np.sqrt(np.maximum(s2 - s ** 2 / tpl.size, 1e-6)) * np.sqrt((t ** 2).sum()))


def templates() -> dict:
    out = {}
    for key, (section, y0, y1, x0, x1) in TEMPLATES.items():
        out[key] = _image(key[0], section)[(y0 - 4) // SCALE:(y1 + 4) // SCALE, x0 // SCALE:x1 // SCALE]
    return out


def locate_headings(edition: str, section: str, tpl: dict) -> list[tuple[int, str, float]]:
    """``[(y, kind, score)]``, top to bottom: each line matching a heading template, as the kind
    it matches better."""
    import numpy as np
    img, found = _image(edition, section), []
    for kind in ("parts", "offices"):
        rows = _ncc(img, tpl[(TEMPLATE_OF[edition], kind)]).max(axis=1)
        for y in np.argsort(-rows):
            if rows[y] < MATCH_MIN:
                break
            if all(abs(y * SCALE - f[0]) > SAME_LINE or f[1] != kind for f in found):
                found.append((int(y) * SCALE, kind, round(float(rows[y]), 3)))
    keep = []
    for f in sorted(found):
        if keep and abs(f[0] - keep[-1][0]) < SAME_LINE:
            keep[-1] = max(keep[-1], f, key=lambda g: g[2])
        else:
            keep.append(f)
    return keep


#: The strip above a section's opening parts heading is page margin when the heading sits this
#: close to the section's top (the widest such margin on the 2026-11-05 candidates: 320 px).
MARGIN = 400


def eye_rulings() -> dict[str, list[tuple[int, str]]]:
    """``{section: [(y, kind)]}`` read by eye; kind ``none`` marks a match that is an entry line."""
    out = defaultdict(list)
    for r in pd.read_csv(EYE, dtype={"section": str}).itertuples():
        out[r.section].append((int(r.y), r.kind))
    return out


def ruled_headings(located: list, ruled: list) -> list[tuple[int, str, float | None, str]]:
    """``[(y, kind, score, source)]``: the located headings with the eye's rulings applied. A ruling
    on the same line sets the kind (``none`` drops the match); one the locator missed is added."""
    out = []
    for y, kind, score in located:
        ruling = next((k for ry, k in ruled if abs(ry - y) <= SAME_LINE), None)
        if ruling != "none":
            out.append((y, ruling or kind, score, "eye" if ruling else "template"))
    for ry, k in ruled:
        if k != "none" and not any(abs(ry - y) <= SAME_LINE for y, *_ in out):
            out.append((ry, k, None, "eye"))
    return sorted(out)


def offices_regions(edition: str, flagged: set, heads: dict) -> list[dict]:
    """The offices regions inside the flagged parts sections. A court's sections are walked in
    reading order (page, then the cutter's index); an unflagged section keeps its label, a flagged
    one switches at each heading in it, and the text before its first heading continues the
    section before it — unless that heading opens a parts list within :data:`MARGIN` of the top."""
    sec = sections(edition)
    sec["index"] = sec.section.str.rsplit("_", n=1).str[1].astype(int)
    by_court = defaultdict(list)
    for s in sec.sort_values(["page", "index"]).itertuples():
        by_court[s.slug].append((s.section, s.kind))
    out = []
    for seq in by_court.values():
        state = None
        for section, kind in seq:
            if section not in flagged:
                state = kind
                continue
            cur, y_from, score, source = state, 0, None, ""
            for y, k, sc, src in heads.get(section, []):
                margin = y_from == 0 and k == "parts" and y <= MARGIN
                if cur == "offices" and y > y_from + SAME_LINE and not margin:
                    out.append(_region(edition, section, y_from, y, score, source))
                cur, y_from, score, source = k, y, sc, src
            if cur == "offices":
                out.append(_region(edition, section, y_from, None, score, source))
            state = cur
    return out


def _region(edition, section, y_from, y_to, score, source) -> dict:
    return {"edition": edition, "section": section, "y_from": y_from, "y_to": y_to,
            "how": "continuation" if y_from == 0 else "after heading",
            "heading_score": score, "heading_source": source}


def eye_recall(located: dict, ruled: dict) -> dict:
    """The locator against the rulings: headings found, missed, matches the eye rejected, and
    matches no ruling covers (the ones a look must settle)."""
    hit, missed, rejected = 0, [], 0
    for section, rows in ruled.items():
        for y, kind in rows:
            match = any(abs(y - ly) <= SAME_LINE and (kind == lk or kind == "none")
                        for ly, lk, _ in located.get(section, []))
            if kind == "none":
                rejected += match
            elif match:
                hit += 1
            else:
                missed.append(f"{section} {kind} y={y}")
    unruled = [f"{s} {k} y={y} ({sc})" for s, found in located.items() for y, k, sc in found
               if not any(abs(y - ry) <= SAME_LINE for ry, _ in ruled.get(s, []))]
    return {"ruled": sum(len(r) for r in ruled.values()), "found": hit, "missed": missed,
            "rejected_matches": rejected, "unruled_matches": unruled}


def report(rec: dict) -> str:
    lines = [f"Offices coverage — sections the readers never read (check_offices_coverage "
             f"{__version__}, D-192), {rec['run']}", ""]
    for ed, s in rec["editions"].items():
        lines += [f"{ed}: {s['municipalities']} municipalities, {s['no_pictogram']} with no pictogram; "
                  f"{s['parts_sections']} parts / {s['offices_sections']} offices sections",
                  f"  blind blocks (≥{BLOCK_MIN} municipalities of a court on a page, no pictogram): "
                  f"{s['blind_blocks']} holding {s['blind_block_munis']}, "
                  f"{s['blind_blocks_on_parts_only_pages']} on a page cut into parts sections only",
                  f"  candidate parts sections: {s['candidates']} (zero reads {s['zero_reads']}, "
                  f"DINOv3 institution glyphs ≥{DINO_INST_MIN} {s['dino_inst']}, on a blind block "
                  f"{s['blind_munis']})"]
        for b in s["blind_blocks_elsewhere"]:
            lines.append(f"  blind block with an offices section on its page (another cause): "
                         f"{b['court']} p. {b['page']}, {b['munis']} municipalities")
    b = rec["bohemia"]
    lines += ["", f"Böhmen, the two editions over {b['pairs']} paired municipalities: "
              f"{b['blind_in_german']} carry no pictogram in German where the Czech has "
              f"{b['their_czech_pictograms']}; {b['blind_in_czech']} the reverse; both read {b['both_read']}",
              f"  {'symbol':32} {'both':>5} {'cz only':>8} {'de only':>8} {'agree':>6}"]
    for mark, r in b["by_symbol"].items():
        lines.append(f"  {mark:32} {r['both']:>5} {r['cz_only']:>8} {r['de_only']:>8} {r['agree']:>6.3f}")
    lines.append(f"  castle/poorhouse differ in {b['castle_poorhouse_differ']} municipalities both read; "
                 f"a German poorhouse against a Czech castle in {b['german_poorhouse_czech_castle']}")
    g = rec["regions"]
    e = rec["eye_recall"]
    lines += ["", f"Offices regions to re-read (headings by template, match ≥ {MATCH_MIN}, ruled by eye "
              f"where read): {g['regions']} in {g['sections']} sections {g['by_edition']}; "
              f"{g['to_confirm']} start on an unruled match under {CONFIRM_BELOW} — look before use",
              f"  the locator against the {e['ruled']} rulings by eye: headings found {e['found']}, "
              f"missed {len(e['missed'])}, matches rejected as entry lines {e['rejected_matches']}, "
              f"matches no ruling covers {len(e['unruled_matches'])}"]
    lines += [f"    missed: {m}" for m in e["missed"]]
    lines += [f"    unruled: {m}" for m in e["unruled_matches"]]
    return "\n".join(lines)


def main() -> int:
    places = pd.read_parquet(REGISTER / "register_places.parquet")
    inst = pd.read_parquet(REGISTER / "register_institutions.parquet")
    coops = pd.read_parquet(REGISTER / "register_cooperatives.parquet")
    pic = inst[inst.source == "pictogram"]
    keys = set(zip(pic.edition, pic.court_key, pic.muni_seq))
    frames, summaries, located, heads, regions = [], {}, {}, {}, []
    tpl, ruled = templates(), eye_rulings()
    for ed in EDITIONS:
        cand, summaries[ed] = candidates(ed, places, keys)
        frames.append(cand)
        for section in cand.section:
            located[section] = locate_headings(ed, section, tpl)
            heads[section] = ruled_headings(located[section], ruled.get(section, []))
        regions += offices_regions(ed, set(cand.section), heads)
    reg = pd.DataFrame(regions, columns=["edition", "section", "y_from", "y_to", "how",
                                         "heading_score", "heading_source"])
    unconfirmed = (reg.heading_source == "template") & (reg.heading_score < CONFIRM_BELOW)
    rec = {"check": __version__, "run": time.strftime("%Y-%m-%d %H:%M:%S"),
           "offices_only_tokens": len(OFFICES_ONLY), "editions": summaries,
           "bohemia": bohemia_editions(places, inst, coops),
           "regions": {"regions": len(reg), "sections": int(reg.section.nunique()),
                       "by_edition": reg.groupby("edition").size().to_dict(),
                       "to_confirm": int(unconfirmed.sum())},
           "eye_recall": eye_recall(located, ruled)}
    OUT.mkdir(parents=True, exist_ok=True)
    reg.to_csv(OUT / "offices_coverage_regions.csv", index=False, encoding="utf-8")
    cols = ["edition", "section", "page", "slug", "reads", "dino_glyphs", "dino_inst", "dino_inst_top",
            "parts_only_page", "blind_munis", "zero_reads"]
    pd.concat(frames)[cols].to_csv(OUT / "offices_coverage_candidates.csv", index=False, encoding="utf-8")
    (OUT / "offices_coverage.json").write_text(json.dumps(rec, indent=1, ensure_ascii=False, default=str),
                                               encoding="utf-8")
    text = report(rec)
    (OUT / "offices_coverage.txt").write_text(text, encoding="utf-8")
    print(text)
    print("\n  written: output/validation/offices_coverage.{json,txt}, offices_coverage_candidates.csv, "
          "offices_coverage_regions.csv")
    return 0
