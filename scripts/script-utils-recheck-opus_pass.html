How it works

The question is the route's own (run_verify_test_b): the entry's lines with the asked marks outlined and numbered, each mark enlarged, the key, the plate's glyph and cuts of each offered symbol, and per mark which is printed there? — its options, none of these (printed but none of them, or no pictogram at all: a digit, a letter, the &) or cannot tell. Two things differ, both from the author's Czech queue (D-190): the outline is drawn around the whole glyph — the reader's box padded by PAD of its size each way, since the reader's box often held only part of a symbol — and each call asks about its marks only, with no reader's reading shown.

The test (test_items, score_test): marks the author ruled by eye on the hand-check page, a fixed random sample of TEST_PER_GROUP per group (an edition × a look-alike pair), one mark per call. D-190's bars, fixed before any spend: in every group of 50 or more, Opus agrees with the author on at least 95 % with a Wilson 95 % lower bound of at least 90 %; and on marks where the author confirmed the route's value, Opus changes at most 2 %.

The pass (pass_items), only once the test has passed: one call per part, for

  • DINOv3's flags (§5, dino/flags.csv, the ones marked to send);
  • the look-alike pair marks the author has not ruled — gamekeeper and huntsman in the German volumes (every poorhouse and castle has been ruled by hand);
  • §4's disputed claims, as token questions on the re-read region (is X printed for this part?).

Everything goes under output/recheck/&lt;edition&gt;/&lt;TEST_STAGE&gt;/ or opus_pass/ (the first, pair-only test stays in opus_test/): the route's Opus folders are never written. Nothing is sent without spend, and only on the machine of record.

def whole_glyph(box: tuple, width: int, height: int) -> tuple[int, int, int, int]:
    """The reader's box widened by :data:`PAD` of its size each way, kept on the image."""
    x0, y0, x1, y1 = box
    dx, dy = max(4, round((x1 - x0) * PAD)), max(4, round((y1 - y0) * PAD))
    return max(0, x0 - dx), max(0, y0 - dy), min(width, x1 + dx), min(height, y1 + dy)

def build_call(item_id: str, unit: tuple, marks: list[dict], tokens: list[str], ctx: dict,
               regions: list[tuple] = ()) -> dict:
    """One call: ``marks`` — ``{image, box, placed, dino, extra}`` asked as *which is printed
    there?*; ``tokens`` — asked as *is it printed for this part?*, shown ``regions`` (section
    images and rows) or, with none, the bands of the marks."""
    questions, parts = [], []
    by_image = defaultdict(list)
    for k, m in enumerate(marks, 1):
        options = tb.options_for(m.get("placed", []), m.get("dino", []), unit[0], ctx["bases"],
                                 ctx["swaps"], ctx["vocabulary"])
        for extra in m.get("extra", []):
            if extra in ctx["vocabulary"] and extra not in options:
                options.append(extra)
        questions.append({"n": k, "kind": "mark", "image": m["image"], "box": list(m["box"]),
                          "placed": m.get("placed", []), "options": options})
        by_image[m["image"]].append((k, tuple(m["box"])))
    for token in tokens:
        questions.append({"n": len(questions) + 1, "kind": "token", "token": token})
    for name, ks in by_image.items():
        band = _band(name, ks)
        if band:
            parts.append(band)
    for name, y_from, y_to in regions:
        shown = _region(name, y_from, y_to)
        if shown:
            parts.append(shown)
    for k, m in enumerate(marks, 1):
        z = _zoom(k, m["image"], tuple(m["box"]))
        if z:
            parts.append(z)
    offered = sorted({t for q in questions for t in q.get("options", [q.get("token")]) if t})
    for code in offered:
        card, cuts = tb.reference_card(code, unit[0], ctx["gold"])
        if card is not None:
            parts.append((card, f"Reference for {code} ({ctx['english'].get(code, '')}): the key's glyph"
                                + (f", then {cuts} cut(s) of it from other pages" if cuts else "") + "."))
    digest = tb.image_digest(image for image, _c in parts)
    for q in questions:
        q["image_digest"] = digest
    return {"id": item_id, "unit": list(unit), "view": "band" if parts else "none",
            "questions": questions, "where": {}, "parts": parts,
            "question": tb._question_text(unit, questions, [], ctx)}

def hand_rulings() -> list[dict]:
    """Every staged pair ruling with its mark, the author's value and the route's:
    ``{id, group, edition, mark, author, route, pair}``. ``other`` / ``unsure`` are left out."""
    pairs = {g["group"]: g for g in json.loads((df.CHECKS / "hand_check_groups.json").read_text("utf-8"))}
    key = {r["id"]: r for r in _read(df.CHECKS / "hand_check_key_pairs.csv")}
    out = []
    for folder in sorted(df.STAGED.glob("r_*-[pg]")):
        group = folder.name[2:]
        for doc in sorted(folder.glob("*.json")):
            k = key.get(doc.stem)
            value = df._ruled(json.loads(doc.read_text("utf-8")).get("v", ""), pairs[group])
            if k is None or value is None:
                continue
            out.append({"id": doc.stem, "group": group, "edition": k["edition"], "mark": k["mark"],
                        "author": value, "route": k["final"], "pair": (pairs[group]["a"], pairs[group]["b"])})
    # Marks the author looked at again (``rerulings.json``, 2026-11-07): the later ruling stands.
    again = df.STAGED / "rerulings.json"
    if again.exists():
        later = json.loads(again.read_text("utf-8"))["rulings"]
        for r in out:
            r["author"] = later.get(r["id"], r["author"])
    return out

def test_sample(rulings: list[dict], per_group: int | None = TEST_PER_GROUP, seed: int = SEED) -> list[dict]:
    """A fixed random sample of each group (all of a smaller one), in id order; ``per_group=None``:
    every ruled mark (2026-11-07, after the 60-mark sample failed in two groups: D-190 sets its bars
    on the ruled marks, and 60 a group allows one miss under the Wilson bound). Never a mark on a
    sealed page (0.2.1, D-127): the hand-check page had shown 57; without them every group still
    passes (2026-11-09)."""
    sealed = {ed: rf.sealed_pages(ed) for ed in {r.get("edition") for r in rulings} if ed}
    by_group = defaultdict(list)
    for r in rulings:
        if r.get("edition") in sealed and r.get("mark", "").split("_")[0] in sealed[r["edition"]]:
            continue
        by_group[r["group"]].append(r)
    rng, out = random.Random(seed), []
    for group in sorted(by_group):
        rows = sorted(by_group[group], key=lambda r: r["id"])
        out += sorted(rows if per_group is None or len(rows) <= per_group else rng.sample(rows, per_group),
                      key=lambda r: r["id"])
    return out

def test_items(edition: str, sample: list[dict]) -> list[dict]:
    """One call per sampled mark of this edition; options as the route would offer them, both
    symbols of the pair always among them. No reading is shown."""
    ctx = _context(edition)
    marks = {r["mark"]: r for r in _read(rf.out_dir(edition, PAGES) / "ledger" / f"marks_{edition}_{PAGES}.csv")}
    record = rf.labels(PAGES, edition)["gemini"]
    dino = {led.box_mark(r["image"], r["box"]): r["dino"] for r in _read(rf.verdicts_path(edition, PAGES))
            if r["pass"] == record and r["box"]}
    items = []
    for s in (s for s in sample if s["edition"] == edition):
        m = marks.get(s["mark"])
        if m is None:
            continue
        image, box = _mark_box(s["mark"])
        named = dino.get(s["mark"], "")
        mark = {"image": image, "box": box, "placed": [s["route"]] if s["route"].startswith("[") else [],
                "dino": [named] if df.is_symbol(named) else [], "extra": lookalike_options(s["pair"])}
        items.append(build_call(f"h-{s['id']}", (m["page"], m["district"], m["muni"], m["seq"]),
                                [mark], [], ctx))
    return items

def wilson_low(k: int, n: int, z: float = 1.96) -> float:
    if n == 0:
        return 0.0
    p = k / n
    centre = p + z * z / (2 * n)
    spread = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n))
    return (centre - spread) / (1 + z * z / n)

def score_test(sample: list[dict], answers: dict) -> list[dict]:
    """Per group: n answered, agree with the author, the Wilson lower bound, and — on marks where
    the author confirmed the route — how often Opus changes the route's value. ``answers`` —
    ``{hand id: Opus's answer}``; *cannot tell* counts as a disagreement and changes nothing."""
    rows = []
    for group in sorted({s["group"] for s in sample}):
        asked = [s for s in sample if s["group"] == group and s["id"] in answers]
        agree = sum(_opus_value(answers[s["id"]]) == s["author"] for s in asked)
        confirmed = [s for s in asked if s["author"] == s["route"]]
        changed = sum(answers[s["id"]] not in (tb.UNSURE,) and _opus_value(answers[s["id"]]) != s["route"]
                      for s in confirmed)
        n = len(asked)
        low = wilson_low(agree, n)
        judged = n >= BAR["min_group"]
        rows.append({"group": group, "n": n, "agree": agree, "share": round(agree / n, 3) if n else None,
                     "wilson_low": round(low, 3), "confirmed": len(confirmed), "changed": changed,
                     "change_share": round(changed / len(confirmed), 3) if confirmed else None,
                     "passes": (not judged) or (agree / n >= BAR["agree"] and low >= BAR["wilson_low"]
                                                and (not confirmed or changed / len(confirmed) <= BAR["change_max"])),
                     "judged": judged})
    return rows

def pass_items(edition: str, out: Path = OUT) -> list[dict]:
    """One call per part: §5's flags to send, the unruled look-alike marks, §4's disputed claims."""
    ctx = _context(edition)
    run = rf.out_dir(edition, PAGES)
    marks = _read(run / "ledger" / f"marks_{edition}_{PAGES}.csv")
    record = rf.labels(PAGES, edition)["gemini"]
    dino = {led.box_mark(r["image"], r["box"]): r["dino"] for r in _read(rf.verdicts_path(edition, PAGES))
            if r["pass"] == record and r["box"]}
    by_part: dict = defaultdict(lambda: {"marks": {}, "tokens": set(), "regions": set()})
    flags = Path(out) / edition / df.SUB / "flags.csv"
    for f in (_read(flags) if flags.exists() else []):
        if f["send"] == "True":
            image, box = _mark_box(f["mark"])
            part = (f["page"], f["district"], f["muni"], f["seq"])
            by_part[part]["marks"][f["mark"]] = {"image": image, "box": box,
                                                  "placed": [f["claim"]] if f["claim"] else [],
                                                  "dino": [f["dino"]]}
    ruled = df.hand_checked(edition)
    pairs = json.loads((df.CHECKS / "hand_check_groups.json").read_text("utf-8"))
    pair_tokens = {t for g in pairs if g["edition"] == edition for t in (g["a"], g["b"])}
    for m in marks:
        if (m["final"] in pair_tokens and m["mark"] not in ruled and "#" in m["mark"]
                and m["mark"].split("#")[0].endswith(".png") and not m["rule"].startswith(("opus", "author"))):
            image, box = _mark_box(m["mark"])
            partner = [t for g in pairs if g["edition"] == edition and m["final"] in (g["a"], g["b"])
                       for t in (g["a"], g["b"])]
            named = dino.get(m["mark"], "")
            by_part[(m["page"], m["district"], m["muni"], m["seq"])]["marks"].setdefault(
                m["mark"], {"image": image, "box": box, "placed": [m["final"]],
                            "dino": [named] if df.is_symbol(named) else [],
                            "extra": lookalike_options(partner)})
    consensus = Path(out) / edition / orr.SUB / "consensus.csv"
    regions = orr.regions(edition) if edition in rf.GERMAN else []
    for c in (_read(consensus) if consensus.exists() else []):
        if c["status"] != "disputed" or c["in_route"] == "True":
            continue
        part = (c["page"], c["district"], c["muni_seq"], c["settlement_seq"])
        by_part[part]["tokens"].add(c["code"])
        for r in regions:
            if r["section"].startswith(f"{c['page']}_{c['district']}_"):
                by_part[part]["regions"].add((r["section"] + ".png", r["y_from"], r["y_to"]))
    items = []
    for part in sorted(by_part):
        p = by_part[part]
        items.append(build_call("p-" + tb.unit_id(part), part, list(p["marks"].values()),
                                sorted(p["tokens"]), ctx, sorted(p["regions"])))
    return items

def send(edition: str, stage: str, items: list[dict], spend: bool, out: Path = OUT,
         allow_other_machine: bool = False) -> dict:
    """Write the calls and, with ``spend``, send those not answered or in flight. Returns the
    counts and the price; nothing is sent without ``spend``."""
    folder = aim(edition, stage, out)
    blind = [i for i in items if i["view"] == "none"]
    items = [i for i in items if i["view"] != "none"]
    try:
        tb.write_items(items)
    except ValueError:
        # its closing print names the folder relative to the repo; a folder outside it (a
        # dry run into scratch) is written all the same
        if not (folder / "items.json").exists():
            raise
    done = tb._answered()
    in_flight = {cid for b in tb._batches() if not b.get("collected") for cid in b["ids"]}
    todo = [i for i in items if f"r1-{i['id']}" not in done | in_flight]
    low, high = price(todo)
    result = {"edition": edition, "stage": stage, "calls": len(items), "no_image": len(blind),
              "answered": sum(f"r1-{i['id']}" in done for i in items), "to_send": len(todo),
              "usd": (round(low, 2), round(high, 2)), "sent": None}
    if spend and todo:
        if bo.HOST != MACHINE_OF_RECORD and not allow_other_machine:
            raise RuntimeError(f"Opus is sent from {MACHINE_OF_RECORD} only (D-68): this is {bo.HOST}")
        result["sent"] = rf.opus_submit(todo, edition)
    (folder / "send.json").write_text(json.dumps(result, indent=1), encoding="utf-8")
    return result

def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--test", action="store_true")
    ap.add_argument("--out", default=str(OUT))

"""Opus on the marks the route left unsettled — tested first on the author's hand rulings (D-190,
notebook 17 §6).

The question is the route's own (``run_verify_test_b``): the entry's lines with the asked marks
outlined and numbered, each mark enlarged, the key, the plate's glyph and cuts of each offered
symbol, and per mark *which is printed there?* — its options, *none of these* (printed but none of
them, or no pictogram at all: a digit, a letter, the &) or *cannot tell*. Two things differ, both
from the author's Czech queue (D-190): the outline is drawn around the **whole glyph** — the
reader's box padded by :data:`PAD` of its size each way, since the reader's box often held only
part of a symbol — and each call asks about its marks only, with no reader's reading shown.

**The test** (:func:`test_items`, :func:`score_test`): marks the author ruled by eye on the
hand-check page, a fixed random sample of :data:`TEST_PER_GROUP` per group (an edition × a
look-alike pair), one mark per call. D-190's bars, fixed before any spend: in every group of 50 or
more, Opus agrees with the author on at least 95 % with a Wilson 95 % lower bound of at least 90 %;
and on marks where the author confirmed the route's value, Opus changes at most 2 %.

**The pass** (:func:`pass_items`), only once the test has passed: one call per part, for

* DINOv3's flags (§5, ``dino/flags.csv``, the ones marked to send);
* the look-alike pair marks the author has not ruled — gamekeeper and huntsman in the German
  volumes (every poorhouse and castle has been ruled by hand);
* §4's disputed claims, as token questions on the re-read region (*is X printed for this part?*).

Everything goes under ``output/recheck/<edition>/<TEST_STAGE>/`` or ``opus_pass/`` (the first,
pair-only test stays in ``opus_test/``): the route's Opus folders are never written. Nothing is sent without ``spend``, and only on the machine of record.

    python -m utils.recheck.opus_pass --test            # build, price (nothing sent)
"""

def whole_glyph(box: tuple, width: int, height: int) -> tuple[int, int, int, int]:
    """The reader's box widened by :data:`PAD` of its size each way, kept on the image."""
    x0, y0, x1, y1 = box
    dx, dy = max(4, round((x1 - x0) * PAD)), max(4, round((y1 - y0) * PAD))
    return max(0, x0 - dx), max(0, y0 - dy), min(width, x1 + dx), min(height, y1 + dy)


def _band(name: str, marks: list[tuple[int, tuple]]) -> tuple | None:
    """The rows of section image ``name`` around its asked marks, each outlined (whole glyph)
    and numbered."""
    from PIL import Image, ImageDraw

    path = tb.image_path(name)
    if path is None:
        return None
    scan = Image.open(path).convert("RGB")
    heights = [b[3] - b[1] for _k, b in marks]
    top = max(0, min(b[1] for _k, b in marks) - BAND_LINES * max(heights))
    foot = min(scan.height, max(b[3] for _k, b in marks) + BAND_LINES * max(heights))
    band = scan.crop((0, top, scan.width, foot))
    draw, font = ImageDraw.Draw(band), tb._font(44)
    for k, box in marks:
        x0, y0, x1, y1 = whole_glyph(box, scan.width, scan.height)
        draw.rectangle((x0, y0 - top, x1, y1 - top), outline=tb.OUTLINE, width=4)
        draw.text((x0, max(0, y0 - top - 50)), str(k), fill=tb.OUTLINE, font=font)
    return tb._fit(band, tb.LONG_EDGE), f"From section image {name}; the asked marks outlined and numbered."


def _zoom(k: int, name: str, box: tuple):
    from PIL import Image, ImageDraw

    path = tb.image_path(name)
    if path is None:
        return None
    scan = Image.open(path).convert("RGB")
    x0, y0, x1, y1 = whole_glyph(box, scan.width, scan.height)
    pad = max(20, y1 - y0)
    left, top = max(0, x0 - 2 * pad), max(0, y0 - pad // 2)
    zoom = scan.crop((left, top, min(scan.width, x1 + 2 * pad), min(scan.height, y1 + pad // 2)))
    ImageDraw.Draw(zoom).rectangle((x0 - left, y0 - top, x1 - left, y1 - top), outline=tb.OUTLINE, width=2)
    return tb._to_height(zoom, tb.ZOOM_HEIGHT), f"Mark {k}, enlarged."


def _region(name: str, y_from: str, y_to: str):
    from PIL import Image

    path = tb.image_path(name)
    if path is None:
        return None
    scan = Image.open(path).convert("RGB")
    top = int(float(y_from or 0))
    foot = int(float(y_to)) if str(y_to).strip() else scan.height
    return (tb._fit(scan.crop((0, top, scan.width, foot)), tb.LONG_EDGE),
            f"The offices of this court as printed (section image {name}, from row {top}).")


def build_call(item_id: str, unit: tuple, marks: list[dict], tokens: list[str], ctx: dict,
               regions: list[tuple] = ()) -> dict:
    """One call: ``marks`` — ``{image, box, placed, dino, extra}`` asked as *which is printed
    there?*; ``tokens`` — asked as *is it printed for this part?*, shown ``regions`` (section
    images and rows) or, with none, the bands of the marks."""
    questions, parts = [], []
    by_image = defaultdict(list)
    for k, m in enumerate(marks, 1):
        options = tb.options_for(m.get("placed", []), m.get("dino", []), unit[0], ctx["bases"],
                                 ctx["swaps"], ctx["vocabulary"])
        for extra in m.get("extra", []):
            if extra in ctx["vocabulary"] and extra not in options:
                options.append(extra)
        questions.append({"n": k, "kind": "mark", "image": m["image"], "box": list(m["box"]),
                          "placed": m.get("placed", []), "options": options})
        by_image[m["image"]].append((k, tuple(m["box"])))
    for token in tokens:
        questions.append({"n": len(questions) + 1, "kind": "token", "token": token})
    for name, ks in by_image.items():
        band = _band(name, ks)
        if band:
            parts.append(band)
    for name, y_from, y_to in regions:
        shown = _region(name, y_from, y_to)
        if shown:
            parts.append(shown)
    for k, m in enumerate(marks, 1):
        z = _zoom(k, m["image"], tuple(m["box"]))
        if z:
            parts.append(z)
    offered = sorted({t for q in questions for t in q.get("options", [q.get("token")]) if t})
    for code in offered:
        card, cuts = tb.reference_card(code, unit[0], ctx["gold"])
        if card is not None:
            parts.append((card, f"Reference for {code} ({ctx['english'].get(code, '')}): the key's glyph"
                                + (f", then {cuts} cut(s) of it from other pages" if cuts else "") + "."))
    digest = tb.image_digest(image for image, _c in parts)
    for q in questions:
        q["image_digest"] = digest
    return {"id": item_id, "unit": list(unit), "view": "band" if parts else "none",
            "questions": questions, "where": {}, "parts": parts,
            "question": tb._question_text(unit, questions, [], ctx)}


def _box(text: str) -> tuple:
    return tuple(int(float(v)) for v in text.split(","))


def _mark_box(mark: str) -> tuple[str, tuple]:
    image, _sep, box = mark.partition("#")
    return image, _box(box)

def hand_rulings() -> list[dict]:
    """Every staged pair ruling with its mark, the author's value and the route's:
    ``{id, group, edition, mark, author, route, pair}``. ``other`` / ``unsure`` are left out."""
    pairs = {g["group"]: g for g in json.loads((df.CHECKS / "hand_check_groups.json").read_text("utf-8"))}
    key = {r["id"]: r for r in _read(df.CHECKS / "hand_check_key_pairs.csv")}
    out = []
    for folder in sorted(df.STAGED.glob("r_*-[pg]")):
        group = folder.name[2:]
        for doc in sorted(folder.glob("*.json")):
            k = key.get(doc.stem)
            value = df._ruled(json.loads(doc.read_text("utf-8")).get("v", ""), pairs[group])
            if k is None or value is None:
                continue
            out.append({"id": doc.stem, "group": group, "edition": k["edition"], "mark": k["mark"],
                        "author": value, "route": k["final"], "pair": (pairs[group]["a"], pairs[group]["b"])})
    # Marks the author looked at again (``rerulings.json``, 2026-11-07): the later ruling stands.
    again = df.STAGED / "rerulings.json"
    if again.exists():
        later = json.loads(again.read_text("utf-8"))["rulings"]
        for r in out:
            r["author"] = later.get(r["id"], r["author"])
    return out


def test_sample(rulings: list[dict], per_group: int | None = TEST_PER_GROUP, seed: int = SEED) -> list[dict]:
    """A fixed random sample of each group (all of a smaller one), in id order; ``per_group=None``:
    every ruled mark (2026-11-07, after the 60-mark sample failed in two groups: D-190 sets its bars
    on the ruled marks, and 60 a group allows one miss under the Wilson bound). Never a mark on a
    sealed page (0.2.1, D-127): the hand-check page had shown 57; without them every group still
    passes (2026-11-09)."""
    sealed = {ed: rf.sealed_pages(ed) for ed in {r.get("edition") for r in rulings} if ed}
    by_group = defaultdict(list)
    for r in rulings:
        if r.get("edition") in sealed and r.get("mark", "").split("_")[0] in sealed[r["edition"]]:
            continue
        by_group[r["group"]].append(r)
    rng, out = random.Random(seed), []
    for group in sorted(by_group):
        rows = sorted(by_group[group], key=lambda r: r["id"])
        out += sorted(rows if per_group is None or len(rows) <= per_group else rng.sample(rows, per_group),
                      key=lambda r: r["id"])
    return out


def test_items(edition: str, sample: list[dict]) -> list[dict]:
    """One call per sampled mark of this edition; options as the route would offer them, both
    symbols of the pair always among them. No reading is shown."""
    ctx = _context(edition)
    marks = {r["mark"]: r for r in _read(rf.out_dir(edition, PAGES) / "ledger" / f"marks_{edition}_{PAGES}.csv")}
    record = rf.labels(PAGES, edition)["gemini"]
    dino = {led.box_mark(r["image"], r["box"]): r["dino"] for r in _read(rf.verdicts_path(edition, PAGES))
            if r["pass"] == record and r["box"]}
    items = []
    for s in (s for s in sample if s["edition"] == edition):
        m = marks.get(s["mark"])
        if m is None:
            continue
        image, box = _mark_box(s["mark"])
        named = dino.get(s["mark"], "")
        mark = {"image": image, "box": box, "placed": [s["route"]] if s["route"].startswith("[") else [],
                "dino": [named] if df.is_symbol(named) else [], "extra": lookalike_options(s["pair"])}
        items.append(build_call(f"h-{s['id']}", (m["page"], m["district"], m["muni"], m["seq"]),
                                [mark], [], ctx))
    return items


def lookalike_options(pair) -> list[str]:
    """The options a look-alike mark is always offered: both of the pair, then :data:`OFFERED`."""
    return list(dict.fromkeys([*pair, *OFFERED]))


def wilson_low(k: int, n: int, z: float = 1.96) -> float:
    if n == 0:
        return 0.0
    p = k / n
    centre = p + z * z / (2 * n)
    spread = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n))
    return (centre - spread) / (1 + z * z / n)


def _opus_value(answer: str) -> str:
    """Opus's answer on the author's scale: *none of these* is no symbol."""
    return led.ABSENT if answer == tb.NONE else answer


def score_test(sample: list[dict], answers: dict) -> list[dict]:
    """Per group: n answered, agree with the author, the Wilson lower bound, and — on marks where
    the author confirmed the route — how often Opus changes the route's value. ``answers`` —
    ``{hand id: Opus's answer}``; *cannot tell* counts as a disagreement and changes nothing."""
    rows = []
    for group in sorted({s["group"] for s in sample}):
        asked = [s for s in sample if s["group"] == group and s["id"] in answers]
        agree = sum(_opus_value(answers[s["id"]]) == s["author"] for s in asked)
        confirmed = [s for s in asked if s["author"] == s["route"]]
        changed = sum(answers[s["id"]] not in (tb.UNSURE,) and _opus_value(answers[s["id"]]) != s["route"]
                      for s in confirmed)
        n = len(asked)
        low = wilson_low(agree, n)
        judged = n >= BAR["min_group"]
        rows.append({"group": group, "n": n, "agree": agree, "share": round(agree / n, 3) if n else None,
                     "wilson_low": round(low, 3), "confirmed": len(confirmed), "changed": changed,
                     "change_share": round(changed / len(confirmed), 3) if confirmed else None,
                     "passes": (not judged) or (agree / n >= BAR["agree"] and low >= BAR["wilson_low"]
                                                and (not confirmed or changed / len(confirmed) <= BAR["change_max"])),
                     "judged": judged})
    return rows


def answers_by_hand_id(folder: Path) -> dict:
    """``{hand id: answer}`` from a test folder's ``answers_raw.csv`` (question 1 of each call)."""
    path = Path(folder) / "answers_raw.csv"
    if not path.exists():
        return {}
    return {r["item"][2:]: r["answer"] for r in _read(path) if str(r["question"]) == "1"}

def pass_items(edition: str, out: Path = OUT) -> list[dict]:
    """One call per part: §5's flags to send, the unruled look-alike marks, §4's disputed claims."""
    ctx = _context(edition)
    run = rf.out_dir(edition, PAGES)
    marks = _read(run / "ledger" / f"marks_{edition}_{PAGES}.csv")
    record = rf.labels(PAGES, edition)["gemini"]
    dino = {led.box_mark(r["image"], r["box"]): r["dino"] for r in _read(rf.verdicts_path(edition, PAGES))
            if r["pass"] == record and r["box"]}
    by_part: dict = defaultdict(lambda: {"marks": {}, "tokens": set(), "regions": set()})
    flags = Path(out) / edition / df.SUB / "flags.csv"
    for f in (_read(flags) if flags.exists() else []):
        if f["send"] == "True":
            image, box = _mark_box(f["mark"])
            part = (f["page"], f["district"], f["muni"], f["seq"])
            by_part[part]["marks"][f["mark"]] = {"image": image, "box": box,
                                                  "placed": [f["claim"]] if f["claim"] else [],
                                                  "dino": [f["dino"]]}
    ruled = df.hand_checked(edition)
    pairs = json.loads((df.CHECKS / "hand_check_groups.json").read_text("utf-8"))
    pair_tokens = {t for g in pairs if g["edition"] == edition for t in (g["a"], g["b"])}
    for m in marks:
        if (m["final"] in pair_tokens and m["mark"] not in ruled and "#" in m["mark"]
                and m["mark"].split("#")[0].endswith(".png") and not m["rule"].startswith(("opus", "author"))):
            image, box = _mark_box(m["mark"])
            partner = [t for g in pairs if g["edition"] == edition and m["final"] in (g["a"], g["b"])
                       for t in (g["a"], g["b"])]
            named = dino.get(m["mark"], "")
            by_part[(m["page"], m["district"], m["muni"], m["seq"])]["marks"].setdefault(
                m["mark"], {"image": image, "box": box, "placed": [m["final"]],
                            "dino": [named] if df.is_symbol(named) else [],
                            "extra": lookalike_options(partner)})
    consensus = Path(out) / edition / orr.SUB / "consensus.csv"
    regions = orr.regions(edition) if edition in rf.GERMAN else []
    for c in (_read(consensus) if consensus.exists() else []):
        if c["status"] != "disputed" or c["in_route"] == "True":
            continue
        part = (c["page"], c["district"], c["muni_seq"], c["settlement_seq"])
        by_part[part]["tokens"].add(c["code"])
        for r in regions:
            if r["section"].startswith(f"{c['page']}_{c['district']}_"):
                by_part[part]["regions"].add((r["section"] + ".png", r["y_from"], r["y_to"]))
    items = []
    for part in sorted(by_part):
        p = by_part[part]
        items.append(build_call("p-" + tb.unit_id(part), part, list(p["marks"].values()),
                                sorted(p["tokens"]), ctx, sorted(p["regions"])))
    return items

def price(items: list[dict]) -> tuple[float, float]:
    return rf.opus_estimate(items) if items else (0.0, 0.0)


def send(edition: str, stage: str, items: list[dict], spend: bool, out: Path = OUT,
         allow_other_machine: bool = False) -> dict:
    """Write the calls and, with ``spend``, send those not answered or in flight. Returns the
    counts and the price; nothing is sent without ``spend``."""
    folder = aim(edition, stage, out)
    blind = [i for i in items if i["view"] == "none"]
    items = [i for i in items if i["view"] != "none"]
    try:
        tb.write_items(items)
    except ValueError:
        # its closing print names the folder relative to the repo; a folder outside it (a
        # dry run into scratch) is written all the same
        if not (folder / "items.json").exists():
            raise
    done = tb._answered()
    in_flight = {cid for b in tb._batches() if not b.get("collected") for cid in b["ids"]}
    todo = [i for i in items if f"r1-{i['id']}" not in done | in_flight]
    low, high = price(todo)
    result = {"edition": edition, "stage": stage, "calls": len(items), "no_image": len(blind),
              "answered": sum(f"r1-{i['id']}" in done for i in items), "to_send": len(todo),
              "usd": (round(low, 2), round(high, 2)), "sent": None}
    if spend and todo:
        if bo.HOST != MACHINE_OF_RECORD and not allow_other_machine:
            raise RuntimeError(f"Opus is sent from {MACHINE_OF_RECORD} only (D-68): this is {bo.HOST}")
        result["sent"] = rf.opus_submit(todo, edition)
    (folder / "send.json").write_text(json.dumps(result, indent=1), encoding="utf-8")
    return result


def collect(edition: str, stage: str, out: Path = OUT) -> bool | None:
    """The stage's batches into ``answers_raw.csv``: ``True`` when all have ended."""
    aim(edition, stage, out)
    return tb.collect() if tb.BATCHES.exists() else None


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--test", action="store_true")
    ap.add_argument("--out", default=str(OUT))
    args = ap.parse_args(argv)
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    sample = test_sample(hand_rulings())
    print("test sample:", dict(Counter(s["group"] for s in sample)))
    for ed in EDITIONS:
        items = test_items(ed, sample) if args.test else pass_items(ed, Path(args.out))
        print(send(ed, TEST_STAGE if args.test else "opus_pass", items, spend=False, out=Path(args.out)))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
