How it works

The route decoded each reading's offices text with institution_abbreviations.csv (D-156), and an item no code reached became no claim: on the full run 1,636 distinct items in the Czech volume, 1,785 in German Böhmen, 436 in Mähren and 191 in Schlesien (report_unlisted_abbreviations.py). Some are names, most are institutions the code table lacks (a theatre, a slaughterhouse, a sugar factory) or spells otherwise (Musiksch for the music school whose code is Musiksch.). The code table is in the freeze's digest, so it is not changed: the codes added here live in reference/places/abbreviation_additions.csv, in its format.

  • census — every item no code reaches, with the places it is printed in (distinct entries, any reader) and the places all four readings print it. The additions file holds the items printed in five or more places of a volume (the author's threshold, 2026-11-12), each curated to a code.
  • claims — the readings decoded again, offline, with the code table plus the additions, by the route's own decoder (lexikon_1910_institutions.decode_entry); the German sections put under their Czech courts as the route put them (relabel_sections). A claim of an added code stands where all four readings make it, on an entry every reading read (the route's agreed, entries_read), with the readers' majority count; a claim some readings make is listed, not kept, and no model is asked (the author's rule, 2026-11-12: no new reading). The rows are final_abbreviations' own, so notebook 17 §9 puts them beside the route's and every later mend reaches them.

No model is called.

    court (``relabel_sections``), as the route keyed it (``report_unlisted_abbreviations`` does not).

    The segments of one court on one page are **one section**, joined in reading order, as the route joins them
    (``lexikon_1910_vision.sections_from_segments``, keyed on page, kind and court): a segment that opens without
    an entry number — Beneschau's offices continue past the divider of page 0031 with *Gendarmerie Abt. Kdo. Nr.
    13, BG., … Obergymn.* — is the rest of the entry before it. Cut apart, its lines fall before any entry number

    it is kept, as the route does."""

    if path is None:
        return []
    notes = pd.read_csv(path, dtype=str, keep_default_na=False)
    if "kind" in notes:
        offices = notes[notes["kind"] == "offices"].assign(
            _order=lambda d: pd.to_numeric(d["order"], errors="coerce"),
            _line=lambda d: pd.to_numeric(d["line_no"], errors="coerce"))

def route_labels(entries: list, label: str, route: set | None = None) -> list:
    """The entries under the court label the route gave the same page (0.5.0, D-237). The route keys a section by the
    label its own cut read — on Czech pages sometimes a misread heading (*kostelecmadcernymilesy* for Kostelec nad
    Černými lesy, 247 Czech entries) — and the run's court table and shared entries are keyed so; the re-cut here
    normalises the heading afresh. Where a page has exactly one label of the re-cut the route lacks and exactly one
    the re-cut lacks, the re-cut's is renamed to the route's; a page with more than one such pair is left as it is.
    ``route``: the route's entries (page, label, municipality) instead of the reading's inventory (tests)."""
    route = route if route is not None else {tuple(map(str, k)) for k in rf.entries_read(label)}
    by_page_route: dict = defaultdict(set)
    for page, district, _ in route:
        by_page_route[str(page)].add(str(district))
    by_page_mine: dict = defaultdict(set)
    for e in entries:
        by_page_mine[str(e.page)].add(str(e.district))
    rename = {}
    for page, mine in by_page_mine.items():
        extra, missing = mine - by_page_route.get(page, set()), by_page_route.get(page, set()) - mine
        if len(extra) == 1 and len(missing) == 1:
            rename[(page, next(iter(extra)))] = next(iter(missing))
    for e in entries:
        new = rename.get((str(e.page), str(e.district)))
        if new:
            e.district = new
    return entries

def census(edition: str, additions: Path | None = None, pages: str = "full") -> pd.DataFrame:
    """Every item no code reaches: ``item`` (normalised), ``beside_code``, ``places`` (distinct entries any
    reading prints it in), ``places_all`` (entries every reading prints it in), ``readings`` (occurrences),
    an ``example`` entry and the item spelled out through the volume's abbreviation parts."""
    table = code_table(edition, additions)
    parts = lxin.load_components(COMPONENTS, lang=rf._lang(edition))
    names = readers(edition, pages)
    crosswalk = lxdn.load_court_crosswalk() if edition in rf.GERMAN else None
    where: dict = defaultdict(lambda: defaultdict(set))
    seen, example = Counter(), {}
    for label in names:
        for entry in reading_entries(label, crosswalk):
            for r in lxin.unlisted_items(entry, table, parts):
                key = (_norm(r["item"]), r["beside_code"])
                place = (r["page"], r["district"], r["muni_seq"], r["settlement_seq"])
                where[key][place].add(label)
                seen[key] += 1
                example.setdefault(key, {**r, "place": "/".join(map(str, place))})
    rows = [{"item": k[0], "beside_code": k[1], "places": len(ps),
             "places_all": sum(len(ls) == len(names) for ls in ps.values()), "readings": seen[k],
             "example": example[k]["place"], "printed": example[k]["item"], "expansion": example[k]["expansion"]}
            for k, ps in where.items()]
    return pd.DataFrame(rows).sort_values(["places", "readings"], ascending=False).reset_index(drop=True)

def reading_claims(label: str, table: list, meaning: dict[str, str], crosswalk: dict | None,
                   base: list | None = None, supersedes: dict[str, str] | None = None,
                   present: set | None = None) -> dict:
    """``{(page, district, muni, part, institution): {"count", "codes", "superseded"}}`` in one reading, for every
    code whose institution (``meaning``: code → standard English) an added code names. Keyed on the institution,
    not the code: a reader who wrote *Musiksch.* and one who wrote *Musiksch* name one music school.

    ``superseded`` (0.4.0, D-237): for an added code that supersedes a general one, what the route counted for the
    **same printed item** under the general code — the entry decoded with the route's own table (``base``), the
    general code's match on the same words, its count read as the route reads it (``run_frozen.printed_reading``).
    *5 hudebních šk.* is five music schools but one bare *šk.* to the route (a word stands between the number and
    the code), so replacing it takes 1 off the general claim, not 5 (0.2.0–0.3.0 took 5, and so took other schools)."""
    out: dict = {}
    supersedes = supersedes or {}
    patterns = {a.code: lxin._pattern_for(a) for a in table if a.code in supersedes}
    for entry in reading_entries(label, crosswalk):
        if present is not None:              # the entries this reading has in its own text (D-239)
            present.add((str(entry.page), str(entry.district), str(entry.muni_seq)))
        rows = lxin.decode_entry(entry, table)
        queue = (_route_counts(entry, patterns, supersedes, base)
                 if base and any(str(r.get("code") or "") in supersedes for r in rows) else {})
        for row in rows:
            code = str(row.get("code") or "")
            if code not in meaning:
                continue
            key = tuple(map(str, lxin._entry_of(row))) + (lxin._part_of(row), meaning[code])
            hit = out.setdefault(key, {"count": 0, "codes": Counter(), "superseded": 0})
            hit["count"] += lxin._count_of(row)
            hit["codes"][code] += 1
            if queue.get(code):
                hit["superseded"] += queue[code].pop(0)
    return out

def reading_entries(label: str, crosswalk: dict | None = None) -> list[lxin.Entry]:
    """The offices entries of one reading, cut from its notes file; a German section put under its Czech
    court (``relabel_sections``), as the route keyed it (``report_unlisted_abbreviations`` does not).

    The segments of one court on one page are **one section**, joined in reading order, as the route joins them
    (``lexikon_1910_vision.sections_from_segments``, keyed on page, kind and court): a segment that opens without
    an entry number — Beneschau's offices continue past the divider of page 0031 with *Gendarmerie Abt. Kdo. Nr.
    13, BG., … Obergymn.* — is the rest of the entry before it. Cut apart, its lines fall before any entry number
    and are lost (0.3.0; 0.1.0–0.2.0 lost them, D-236). A heading is stripped and whatever its line carries after
    it is kept, as the route does."""
    path = bo.find_pass_file("notes", label)
    if path is None:
        return []
    notes = pd.read_csv(path, dtype=str, keep_default_na=False)
    if "kind" in notes:
        offices = notes[notes["kind"] == "offices"].assign(
            _order=lambda d: pd.to_numeric(d["order"], errors="coerce"),
            _line=lambda d: pd.to_numeric(d["line_no"], errors="coerce"))
        sections = []
        for (page, court), g in offices.sort_values(["page", "_order", "_line"], kind="stable").groupby(
                ["page", "court_de"], sort=False):
            lines = [tail for tail in (_DE_HEADING_LINE.sub("", t) for t in g["text"]) if tail.strip()]
            sections.append(lxnt.Section("offices", court, page, lines))
        lxdn.relabel_sections(sections, crosswalk if crosswalk is not None else lxdn.load_court_crosswalk())
        entries = [e for s in sections for e in lxin.parse_entries(s)]
    else:
        pages = [lxv.notes_from_regions(p, {r: g.loc[g["region"] == r, "text"].tolist() for r in ("left", "right", "footer")})
                 for p, g in notes.groupby("page", sort=True)]
        entries = [e for s in lxnt.collect_sections(pages) if s.kind == "offices" for e in lxin.parse_entries(s)]
    return route_labels(entries, label)


def route_labels(entries: list, label: str, route: set | None = None) -> list:
    """The entries under the court label the route gave the same page (0.5.0, D-237). The route keys a section by the
    label its own cut read — on Czech pages sometimes a misread heading (*kostelecmadcernymilesy* for Kostelec nad
    Černými lesy, 247 Czech entries) — and the run's court table and shared entries are keyed so; the re-cut here
    normalises the heading afresh. Where a page has exactly one label of the re-cut the route lacks and exactly one
    the re-cut lacks, the re-cut's is renamed to the route's; a page with more than one such pair is left as it is.
    ``route``: the route's entries (page, label, municipality) instead of the reading's inventory (tests)."""
    route = route if route is not None else {tuple(map(str, k)) for k in rf.entries_read(label)}
    by_page_route: dict = defaultdict(set)
    for page, district, _ in route:
        by_page_route[str(page)].add(str(district))
    by_page_mine: dict = defaultdict(set)
    for e in entries:
        by_page_mine[str(e.page)].add(str(e.district))
    rename = {}
    for page, mine in by_page_mine.items():
        extra, missing = mine - by_page_route.get(page, set()), by_page_route.get(page, set()) - mine
        if len(extra) == 1 and len(missing) == 1:
            rename[(page, next(iter(extra)))] = next(iter(missing))
    for e in entries:
        new = rename.get((str(e.page), str(e.district)))
        if new:
            e.district = new
    return entries


def code_table(edition: str, additions: Path | None = ADDITIONS) -> list[lxin.Abbreviation]:
    """The edition's code table as the route reads it, with the additions; longest code first."""
    lang, volume = rf._lang(edition), rf.GERMAN.get(edition, "")
    rows = lxin.load_abbreviations(ABBREVIATIONS, lang=lang, volume=volume)
    if additions and Path(additions).exists():
        rows += lxin.load_abbreviations(additions, lang=lang, volume=volume)
    return sorted(rows, key=lambda a: -len(a.key))


def _norm(item: str) -> str:
    return re.sub(r"^\d+\s+", "", " ".join(lxin._dehyphenate(item).lower().split()))


# ── the census ────────────────────────────────────────────────────────────────
def census(edition: str, additions: Path | None = None, pages: str = "full") -> pd.DataFrame:
    """Every item no code reaches: ``item`` (normalised), ``beside_code``, ``places`` (distinct entries any
    reading prints it in), ``places_all`` (entries every reading prints it in), ``readings`` (occurrences),
    an ``example`` entry and the item spelled out through the volume's abbreviation parts."""
    table = code_table(edition, additions)
    parts = lxin.load_components(COMPONENTS, lang=rf._lang(edition))
    names = readers(edition, pages)
    crosswalk = lxdn.load_court_crosswalk() if edition in rf.GERMAN else None
    where: dict = defaultdict(lambda: defaultdict(set))
    seen, example = Counter(), {}
    for label in names:
        for entry in reading_entries(label, crosswalk):
            for r in lxin.unlisted_items(entry, table, parts):
                key = (_norm(r["item"]), r["beside_code"])
                place = (r["page"], r["district"], r["muni_seq"], r["settlement_seq"])
                where[key][place].add(label)
                seen[key] += 1
                example.setdefault(key, {**r, "place": "/".join(map(str, place))})
    rows = [{"item": k[0], "beside_code": k[1], "places": len(ps),
             "places_all": sum(len(ls) == len(names) for ls in ps.values()), "readings": seen[k],
             "example": example[k]["place"], "printed": example[k]["item"], "expansion": example[k]["expansion"]}
            for k, ps in where.items()]
    return pd.DataFrame(rows).sort_values(["places", "readings"], ascending=False).reset_index(drop=True)


# ── the claims ────────────────────────────────────────────────────────────────
def reading_claims(label: str, table: list, meaning: dict[str, str], crosswalk: dict | None,
                   base: list | None = None, supersedes: dict[str, str] | None = None,
                   present: set | None = None) -> dict:
    """``{(page, district, muni, part, institution): {"count", "codes", "superseded"}}`` in one reading, for every
    code whose institution (``meaning``: code → standard English) an added code names. Keyed on the institution,
    not the code: a reader who wrote *Musiksch.* and one who wrote *Musiksch* name one music school.

    ``superseded`` (0.4.0, D-237): for an added code that supersedes a general one, what the route counted for the
    **same printed item** under the general code — the entry decoded with the route's own table (``base``), the
    general code's match on the same words, its count read as the route reads it (``run_frozen.printed_reading``).
    *5 hudebních šk.* is five music schools but one bare *šk.* to the route (a word stands between the number and
    the code), so replacing it takes 1 off the general claim, not 5 (0.2.0–0.3.0 took 5, and so took other schools)."""
    out: dict = {}
    supersedes = supersedes or {}
    patterns = {a.code: lxin._pattern_for(a) for a in table if a.code in supersedes}
    for entry in reading_entries(label, crosswalk):
        if present is not None:              # the entries this reading has in its own text (D-239)
            present.add((str(entry.page), str(entry.district), str(entry.muni_seq)))
        rows = lxin.decode_entry(entry, table)
        queue = (_route_counts(entry, patterns, supersedes, base)
                 if base and any(str(r.get("code") or "") in supersedes for r in rows) else {})
        for row in rows:
            code = str(row.get("code") or "")
            if code not in meaning:
                continue
            key = tuple(map(str, lxin._entry_of(row))) + (lxin._part_of(row), meaning[code])
            hit = out.setdefault(key, {"count": 0, "codes": Counter(), "superseded": 0})
            hit["count"] += lxin._count_of(row)
            hit["codes"][code] += 1
            if queue.get(code):
                hit["superseded"] += queue[code].pop(0)
    return out


def _route_counts(entry, patterns: dict, supersedes: dict, base: list) -> dict[str, list[int]]:
    """``{added code: [the route's count under the general code, for each printed item the code matches, in
    order]}``. Each such item is decoded alone with the route's own table and its general-code matches counted as
    the route counts them (``run_frozen.printed_reading``): the item, not the decoder's short context window, which
    cuts *šk. pro moderní ř…* before the new code's words end."""
    base_by_code = {a.code: a for a in base}
    out: dict[str, list[int]] = {}
    for k, item in enumerate(lxin.split_items(entry.text)):
        words = lxin._dehyphenate(item)
        for code, pattern in patterns.items():
            if not pattern.search(words):
                continue
            general = supersedes[code]
            mini = lxin.Entry(entry.district, entry.page, entry.muni_seq, entry.settlement_seq, item, k)
            n = sum(rf.printed_reading(r, base_by_code[general])[0] if general in base_by_code else lxin._count_of(r)
                    for r in lxin.decode_entry(mini, base) if str(r.get("code") or "") == general)
            out.setdefault(code, []).append(n)
    return out


def claims(edition: str, pages: str = "full", additions: Path = ADDITIONS, runs: dict | None = None,
           shared: set | None = None, route: pd.DataFrame | None = None,
           present: dict | None = None) -> tuple[pd.DataFrame, pd.DataFrame]:
    """``(settled, disputed)`` in ``final_abbreviations``' fields, for the institutions the added codes name.

    A claim is **settled** where every reading that has the entry in its own text makes it, and at least
    :data:`MIN_READERS` do (0.6.0, D-239; 0.1.0–0.5.0 asked all four): a Czech pass's inventory is padded with the
    text layer's rows where its reader read nothing, so the route counts a reading that never saw an entry as
    having read it, and the all-four rule took its absence for a *no* (50 of the Czech volume's 110 listed claims).
    A reading that has the entry and leaves the institution out still disputes it. Everything not settled is
    listed with ``readers_with_entry`` for Opus (``census_opus``, D-239). Where the route already holds the
    institution at the place (``route``), the route's claim stands and none is added. ``runs``, ``shared`` and
    ``present`` (``{reading: entries it has}``; by default every reading has every entry) replace the decoding
    (tests)."""
    from utils.recheck import standard_codes as std

    lang, volume = rf._lang(edition), rf.GERMAN.get(edition, "")
    added_rows = lxin.load_abbreviations(additions, lang=lang, volume=volume) if Path(additions).exists() else []
    empty = pd.DataFrame(columns=rf.ABBREVIATION_FIELDS + ["supersedes", "supersedes_count", "readers_with_entry"])
    if not added_rows:
        return empty, empty
    standard = std.load()
    table = code_table(edition, additions)
    meaning_of = {a.code: std.standard(a.english, standard) for a in table if a.kind in rf.ABBREVIATION_KINDS}
    wanted = {std.standard(a.english, standard) for a in added_rows}
    meaning = {c: m for c, m in meaning_of.items() if m in wanted}
    by_code = {a.code: a for a in table}
    supersedes = supersession(additions, lang)
    names = readers(edition, pages)
    if runs is None:
        crosswalk = lxdn.load_court_crosswalk() if edition in rf.GERMAN else None
        base = code_table(edition, None)
        present = {label: set() for label in names}
        runs = {label: reading_claims(label, table, meaning, crosswalk, base, supersedes, present[label])
                for label in names}
        shared = set.intersection(*(rf.entries_read(label) for label in names))
    if shared is None:
        shared = {c[:3] for cs in runs.values() for c in cs}
    route = route if route is not None else _route_abbreviations(edition, pages)
    held = {(r.page, r.district, str(r.muni_seq), str(r.settlement_seq), std.standard(r.english, standard))
            for r in route.itertuples()} if not route.empty else set()
    runs = {label: {c: v for c, v in cs.items() if c[:3] in shared and c not in held} for label, cs in runs.items()}
    support = Counter(c for cs in runs.values() for c in cs)
    counts = rf.majority_counts({label: {c: v["count"] for c, v in cs.items()} for label, cs in runs.items()}, list(runs))
    replaced = rf.majority_counts({label: {c: v.get("superseded", 0) for c, v in cs.items()}
                                   for label, cs in runs.items()}, list(runs))
    courts = _courts(edition, pages)
    settled, disputed = [], []
    for claim, n in sorted(support.items()):
        page, district, muni, seq, institution = claim
        codes = sum((cs[claim]["codes"] for cs in runs.values() if claim in cs), Counter())
        a = by_code[codes.most_common(1)[0][0]]
        court, how = courts.get((page, district), ("", ""))
        count, source = counts.get(claim, ("", ""))
        row = {"entry": "/".join((page, court or district, muni, seq)), "page": page, "district": district,
               "court_id": court, "court_source": how, "muni_seq": muni, "settlement_seq": seq, "code": a.code,
               "count": count, "count_source": source, "expansion": a.expansion, "english": institution,
               "category": a.category, "kind": a.kind, "establishment": a.kind in rf.ESTABLISHMENT_KINDS,
               "qualifier": "", "language_suffix": "", "language_counts": "", "rule": RULE,
               "support": f"{n} of {len(runs)}",
               "supersedes": next((supersedes[c] for c, _ in codes.most_common() if supersedes.get(c)), "")}
        row["supersedes_count"] = int(replaced.get(claim, (0, ""))[0] or 0) if row["supersedes"] else 0
        having = (sum(claim[:3] in present[label] for label in runs) if present is not None else len(runs))
        row["readers_with_entry"] = having
        if n == len(runs):
            settled.append(row)
        elif n == having and n >= MIN_READERS:
            settled.append({**row, "rule": RULE_PRESENT})
        else:
            disputed.append(row)
    cols = rf.ABBREVIATION_FIELDS + ["supersedes", "supersedes_count", "readers_with_entry"]
    return (pd.DataFrame(settled, columns=cols),
            pd.DataFrame(disputed, columns=cols).assign(rule="disputed: listed, not asked"))


def supersession(additions: Path = ADDITIONS, lang: str = "") -> dict[str, str]:
    """``{added code: the code it supersedes}`` (D-235): a code whose words the route decoded as a more general one
    (*hudebních šk.* as the bare *šk.*, *GendarmerieabtKdo.* as *Gendarm*), so adding its claim takes its count off
    that one at the same part, never counting one institution twice."""
    if not Path(additions).exists():
        return {}
    with Path(additions).open(encoding="utf-8", newline="") as fh:
        return {r["code"].strip(): (r.get("supersedes") or "").strip() for r in csv.DictReader(fh)
                if (not lang or r["lang"].strip() == lang) and (r.get("supersedes") or "").strip()}


def _route_abbreviations(edition: str, pages: str) -> pd.DataFrame:
    """The route's settled abbreviations for the edition (what a census claim must not repeat)."""
    path = rf.out_dir(edition, pages) / "final_abbreviations.csv"
    return pd.read_csv(path, dtype=str, keep_default_na=False) if path.exists() else pd.DataFrame()


def _courts(edition: str, pages: str) -> dict:
    path = rf.out_dir(edition, pages) / "courts.csv"
    if not path.exists():
        return {}
    with path.open(encoding="utf-8", newline="") as fh:
        return {(r["page"], r["district"]): (r["court_id"], r["court_source"]) for r in csv.DictReader(fh)}

def cached(edition: str, folder: Path = OUT) -> tuple[pd.DataFrame, pd.DataFrame]:
    """``(settled, disputed)`` for one volume, from ``<folder>/<edition>_{settled,disputed}.csv`` while they are newer
    than the additions, the standards and this module; decoded again (offline, a few minutes a volume) and written
    when they are not. Notebook 17 §9 (``register_rechecked.census_claims``) and the Opus step (``census_opus``)
    read the same files."""
    from utils.recheck import standard_codes as std

    paths = [Path(folder) / f"{edition}_{kind}.csv" for kind in ("settled", "disputed")]
    inputs = [x for x in (ADDITIONS, std.STANDARD, Path(__file__)) if x.exists()]
    if all(x.exists() for x in paths) and all(x.stat().st_mtime >= i.stat().st_mtime for x in paths for i in inputs):
        return tuple(pd.read_csv(x, dtype=str, keep_default_na=False) for x in paths)
    settled, disputed = claims(edition)
    Path(folder).mkdir(parents=True, exist_ok=True)
    settled.to_csv(paths[0], index=False, encoding="utf-8")
    disputed.to_csv(paths[1], index=False, encoding="utf-8")
    return tuple(pd.read_csv(x, dtype=str, keep_default_na=False) for x in paths)


def build(editions=EDITIONS, out: Path = OUT) -> dict:
    """The census and the claims for every edition with a full run; ``{edition: (settled, disputed)}``."""
    out.mkdir(parents=True, exist_ok=True)
    done = {}
    for edition in editions:
        if not (rf.out_dir(edition, "full") / "final_abbreviations.csv").exists():
            continue
        settled, disputed = claims(edition)
        settled.to_csv(out / f"{edition}_settled.csv", index=False, encoding="utf-8")
        disputed.to_csv(out / f"{edition}_disputed.csv", index=False, encoding="utf-8")
        done[edition] = (settled, disputed)
        print(f"  {edition}: {len(settled)} census-code claims all four readings make, {len(disputed)} listed")
    return done
