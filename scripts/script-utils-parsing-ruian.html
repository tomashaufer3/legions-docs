TL;DR

There are no 1910 municipal boundaries for Bohemia and Moravia, and the project does not need them: what a 1910 settlement needs is a place, and a place survives boundary changes. This module reads the ČÚZK RÚIAN state file (ST_UKSG, CC-BY 4.0) into four current-geography tables and then gives every row of settlements_1910 a location by finding it, by name, among today's municipalities and their parts.

How it works

Tables written to data/final/geo/ (key in brackets, asserted in data_checks):

  • ruian_obce [obec_kod] name, modern okres, definition point, polygon;
  • ruian_casti_obce [cast_kod] name, obec_kod, definition point;
  • ruian_zsj [zsj_kod] základní sídelní jednotka — name, KÚ, definition point. ZSJ are the statistical settlement units: many 1910 osady that are no longer an official část obce still are one;
  • ruian_ku [ku_kod] katastrální území — name, obec_kod, polygon;
  • ruian_obce_adjacency [obec_kod, neighbour_kod] obce sharing a boundary.

Geometries are GeoParquet (WKB, OGC:CRS84 lon/lat) beside a flat CSV with lat/lon.

Coordinates

RÚIAN is in S-JTSK / Křovák East-North (EPSG:5514) on the Bessel ellipsoid. The repo carries no geo stack (see utils.parsing.spatial), so the inverse Křovák projection (EPSG guidance note 7-2, method 9819) and the 7-parameter S-JTSK → WGS 84 Helmert (EPSG:1623, position-vector convention, quoted accuracy 1 m) are written out here in closed form. The projection reproduces the EPSG worked example to under a millimetre (tests/test_ruian.py).

The crosswalk

See build_crosswalk and reference/PLACE_MATCHING.md §"Today's geography". In one breath: the candidates for a 1910 settlement are the parts, ZSJ and obce whose definition point lies inside its 1910 okres (the union of its MPIDR court polygons); only when nothing of that name is there does the search widen to the obce adjacent to that area. A name that matches two different places at the same tier is refused, never broken by order (D-22). Wikidata is never overwritten: coord_source says where the point came from.

def krovak_inverse(southing, westing):
    """Křovák (south-oriented X, west-oriented Y, metres) → Bessel latitude/longitude in radians."""
    xs = np.asarray(southing, float)
    yw = np.asarray(westing, float)
    r = np.hypot(xs, yw)
    d = np.arctan2(yw, xs) / _KN
    t = 2 * (np.arctan((_R0 / r) ** (1 / _KN) * np.tan(np.pi / 4 + _PHI_P / 2)) - np.pi / 4)
    u = np.arcsin(np.cos(_ALPHA_C) * np.sin(t) - np.sin(_ALPHA_C) * np.cos(t) * np.cos(d))
    v = np.arcsin(np.cos(t) * np.sin(d) / np.cos(u))
    phi = u
    for _ in range(8):                         # converges to < 1e-12 rad in four
        es = _E_BESSEL * np.sin(phi)
        phi = 2 * (np.arctan(_T0 ** (-1 / _KB) * np.tan(u / 2 + np.pi / 4) ** (1 / _KB)
                             * ((1 + es) / (1 - es)) ** (_E_BESSEL / 2)) - np.pi / 4)
    return phi, _LAM_0 - v / _KB

def krovak_forward(phi, lam):
    """Bessel latitude/longitude (radians) → Křovák (southing, westing) in metres. Tests only."""
    es = _E_BESSEL * np.sin(phi)
    u = 2 * (np.arctan(_T0 * np.tan(phi / 2 + np.pi / 4) ** _KB
                       / ((1 + es) / (1 - es)) ** (_E_BESSEL * _KB / 2)) - np.pi / 4)
    v = _KB * (_LAM_0 - lam)
    t = np.arcsin(np.cos(_ALPHA_C) * np.sin(u) + np.sin(_ALPHA_C) * np.cos(u) * np.cos(v))
    d = np.arcsin(np.cos(u) * np.sin(v) / np.cos(t))
    r = _R0 * np.tan(np.pi / 4 + _PHI_P / 2) ** _KN / np.tan(t / 2 + np.pi / 4) ** _KN
    return r * np.cos(_KN * d), r * np.sin(_KN * d)

def sjtsk_to_wgs84(x, y):
    """EPSG:5514 (x = easting, y = northing, both negative) → WGS 84 ``(lon, lat)`` in degrees.

    EPSG:5514 is the East-North variant of Křovák: ``x = −westing``, ``y = −southing``.
    """
    phi, lam = krovak_inverse(-np.asarray(y, float), -np.asarray(x, float))
    X, Y, Z = _geodetic_to_ecef(phi, lam, _A_BESSEL, _E2_BESSEL)
    tx, ty, tz, rx, ry, rz, s = _HELMERT
    rx, ry, rz = (np.radians(v / 3600) for v in (rx, ry, rz))
    m = 1 + s * 1e-6
    X2 = tx + m * (X - rz * Y + ry * Z)                # position-vector rotation, small angles
    Y2 = ty + m * (rz * X + Y - rx * Z)
    Z2 = tz + m * (-ry * X + rx * Y + Z)
    phi2, lam2 = _ecef_to_geodetic(X2, Y2, Z2, _A_WGS, _E2_WGS)
    return np.degrees(lam2), np.degrees(phi2)

def wgs84_to_sjtsk(lon, lat):
    """WGS 84 degrees → EPSG:5514 ``(x, y)``: the inverse of :func:`sjtsk_to_wgs84`.

    The Helmert is reversed by negating its parameters — exact to millimetres at these
    rotation sizes — then the Křovák forward projection.
    """
    X, Y, Z = _geodetic_to_ecef(np.radians(np.asarray(lat, float)),
                                np.radians(np.asarray(lon, float)), _A_WGS, _E2_WGS)
    tx, ty, tz, rx, ry, rz, s = (-v for v in _HELMERT)
    rx, ry, rz = (np.radians(v / 3600) for v in (rx, ry, rz))
    m = 1 + s * 1e-6
    X2 = tx + m * (X - rz * Y + ry * Z)
    Y2 = ty + m * (rz * X + Y - rx * Z)
    Z2 = tz + m * (-ry * X + rx * Y + Z)
    phi, lam = _ecef_to_geodetic(X2, Y2, Z2, _A_BESSEL, _E2_BESSEL)
    southing, westing = krovak_forward(phi, lam)
    return -westing, -southing

def read_vfr(zip_path) -> dict[str, pd.DataFrame]:
    """Stream the ``ST_UKSG`` state file into ``okresy`` / ``obce`` / ``casti_obce`` / ``ku`` / ``zsj``.

    Polygons are kept (S-JTSK, list of ``[exterior, *interiors]``) for obce and KÚ only; ZSJ
    carry their original boundaries in the file, which nothing here needs. Every table gains
    WGS 84 ``lat``/``lon`` for its definition point.
    """
    import zipfile
    from lxml import etree

    with_polygon = {"obce", "ku"}
    rows: dict[str, list[dict]] = {k: [] for k in _OBJECTS.values()}
    with zipfile.ZipFile(zip_path) as zf:
        name = next(n for n in zf.namelist() if n.endswith(".xml"))
        with zf.open(name) as fh:
            for _ev, el in etree.iterparse(fh, events=("end",), huge_tree=True):
                kind = _OBJECTS.get(_local(el.tag))
                # An object carries a gml:id; its container does not — and two containers
                # (``vf:KatastralniUzemi``, ``vf:Zsj``) share their member's name.
                if (kind is None or f"{_GML}id" not in el.attrib
                        or not el.tag.startswith("{urn:cz:isvs:ruian:schemas:VymennyFormatTypy")):
                    continue
                rows[kind].append(_read_object(el, kind, kind in with_polygon))
                el.clear()
                while el.getprevious() is not None:          # free parsed siblings
                    del el.getparent()[0]
    out = {}
    for kind, recs in rows.items():
        df = pd.DataFrame(recs)
        lon, lat = sjtsk_to_wgs84(df["x"].to_numpy(), df["y"].to_numpy())
        df["lat"], df["lon"] = lat, lon
        out[kind] = df
    return out

def wkb_multipolygon(polygons: list[list[np.ndarray]]) -> bytes:
    """``[[exterior, *interiors], …]`` (each ring ``(N, 2)``) → little-endian ISO WKB MultiPolygon."""
    import struct

    parts = [struct.pack("<BII", 1, 6, len(polygons))]
    for rings in polygons:
        parts.append(struct.pack("<BII", 1, 3, len(rings)))
        for ring in rings:
            parts.append(struct.pack("<I", len(ring)))
            parts.append(np.ascontiguousarray(ring, dtype="<f8").tobytes())
    return b"".join(parts)

def write_geoparquet(df: pd.DataFrame, path, polygon_col: str = "polygons") -> None:
    """Write *df* as GeoParquet 1.1: *polygon_col* (S-JTSK rings) → WKB ``geometry`` in OGC:CRS84.

    The CRS is left out of the metadata on purpose — GeoParquet's default **is** OGC:CRS84
    (lon/lat, WGS 84), so any reader (geopandas, QGIS, DuckDB) places it without PROJJSON.
    """
    import json

    import pyarrow as pa
    import pyarrow.parquet as pq

    flat = df.drop(columns=[polygon_col])
    lonlat = [_polygons_lonlat(p) for p in df[polygon_col]]
    wkb = [wkb_multipolygon(p) if p else None for p in lonlat]
    allpts = [np.vstack([r for rings in p for r in rings]) for p in lonlat if p]
    bbox = (min(a[:, 0].min() for a in allpts), min(a[:, 1].min() for a in allpts),
            max(a[:, 0].max() for a in allpts), max(a[:, 1].max() for a in allpts))
    table = pa.Table.from_pandas(flat, preserve_index=False)
    table = table.append_column("geometry", pa.array(wkb, type=pa.binary()))
    meta = {"version": "1.1.0", "primary_column": "geometry",
            "columns": {"geometry": {"encoding": "WKB", "geometry_types": ["MultiPolygon"],
                                     "bbox": [float(v) for v in bbox]}}}
    table = table.replace_schema_metadata({**(table.schema.metadata or {}),
                                           b"geo": json.dumps(meta).encode()})
    pq.write_table(table, path)

def read_geoparquet_polygons(path) -> pd.DataFrame:
    """Read a table written by :func:`write_geoparquet` back, ``geometry`` decoded to lon/lat rings."""
    import struct

    df = pd.read_parquet(path)

    def decode(b):
        if b is None:
            return []
        _bo, _t, npoly = struct.unpack_from("<BII", b, 0)
        pos, polys = 9, []
        for _ in range(npoly):
            _bo, _t, nring = struct.unpack_from("<BII", b, pos)
            pos += 9
            rings = []
            for _ in range(nring):
                (n,) = struct.unpack_from("<I", b, pos)
                pos += 4
                rings.append(np.frombuffer(b, "<f8", 2 * n, pos).reshape(n, 2))
                pos += 16 * n
            polys.append(rings)
        return polys

    df["geometry"] = df["geometry"].map(decode)
    return df

def obce_adjacency(obce: pd.DataFrame, tol_m: float = 0.01) -> pd.DataFrame:
    """Pairs of obce whose boundaries meet, both directions, with the count of shared vertices.

    RÚIAN's generalised obec boundaries are topologically consistent — every vertex of a shared
    line is stored identically on both sides (measured: 491,602 vertices shared by exactly two
    obce, 12k by three) — so adjacency is vertex sharing, found with a KD-tree at *tol_m*.
    ``shared_vertices == 1`` is a corner touch; a shared edge has two or more.
    """
    from scipy.spatial import cKDTree

    xy, kod = [], []
    for k, polys in zip(obce["obec_kod"], obce["polygons"]):
        for rings in polys:
            for r in rings:
                xy.append(r[:-1])                      # the closing vertex repeats the first
                kod.append(np.full(len(r) - 1, k))
    xy, kod = np.vstack(xy), np.concatenate(kod)
    pairs = cKDTree(xy).query_pairs(r=tol_m, output_type="ndarray")
    a, b = kod[pairs[:, 0]], kod[pairs[:, 1]]
    keep = a != b
    p = pd.DataFrame({"obec_kod": np.minimum(a, b)[keep], "neighbour_kod": np.maximum(a, b)[keep]})
    p = p.value_counts().rename("shared_vertices").reset_index()
    both = pd.concat([p, p.rename(columns={"obec_kod": "neighbour_kod",
                                           "neighbour_kod": "obec_kod"})], ignore_index=True)
    return both.sort_values(["obec_kod", "neighbour_kod"]).reset_index(drop=True)

============================================================================================

* ``ruian_obce``          [``obec_kod``]  name, modern okres, definition point, polygon;

                          longer an official část obce still are one;
* ``ruian_ku``            [``ku_kod``]    katastrální území — name, ``obec_kod``, polygon;
* ``ruian_obce_adjacency`` [``obec_kod``, ``neighbour_kod``] obce sharing a boundary.

Geometries are GeoParquet (WKB, OGC:CRS84 lon/lat) beside a flat CSV with lat/lon.

Coordinates
-----------
RÚIAN is in S-JTSK / Křovák East-North (EPSG:5514) on the Bessel ellipsoid. The repo carries no
geo stack (see :mod:`utils.parsing.spatial`), so the inverse Křovák projection (EPSG guidance
note 7-2, method 9819) and the 7-parameter S-JTSK → WGS 84 Helmert (EPSG:1623, position-vector
convention, quoted accuracy 1 m) are written out here in closed form. The projection reproduces
the EPSG worked example to under a millimetre (``tests/test_ruian.py``).

The crosswalk
-------------
See :func:`build_crosswalk` and ``reference/PLACE_MATCHING.md`` §"Today's geography". In one
breath: the candidates for a 1910 settlement are the parts, ZSJ and obce whose definition point
lies inside its **1910 okres** (the union of its MPIDR court polygons); only when nothing of that
name is there does the search widen to the obce **adjacent** to that area. A name that matches
two different places at the same tier is **refused**, never broken by order (D-22). Wikidata is
never overwritten: ``coord_source`` says where the point came from.

Usage
-----
    from utils.parsing import ruian as ru
    geo = ru.read_vfr(RUIAN_ZIP)                       # dict of DataFrames (S-JTSK + WGS84)
    ru.write_tables(geo, FINAL / "geo")
    xw = ru.build_crosswalk(settlements, geo, locator)
    ru.crosswalk_report(xw)
"""

# ── S-JTSK (EPSG:5514) → WGS 84 ───────────────────────────────────────────────────
# Bessel 1841 and the Křovák defining parameters (EPSG:5514 / method 9819).
_A_BESSEL = 6377397.155
_F_BESSEL = 1 / 299.1528128
_E2_BESSEL = 2 * _F_BESSEL - _F_BESSEL ** 2
_E_BESSEL = np.sqrt(_E2_BESSEL)
_PHI_C = np.radians(49.5)                                        # latitude of projection centre
_LAM_0 = np.radians(24 + 50 / 60)                                # 42°30' E of Ferro, in Greenwich
_ALPHA_C = np.radians(30 + 17 / 60 + 17.30311 / 3600)            # co-latitude of the cone axis
_PHI_P = np.radians(78.5)                                        # pseudo standard parallel
_K_P = 0.9999                                                    # scale on the pseudo parallel

_KA = _A_BESSEL * np.sqrt(1 - _E2_BESSEL) / (1 - _E2_BESSEL * np.sin(_PHI_C) ** 2)
_KB = np.sqrt(1 + _E2_BESSEL * np.cos(_PHI_C) ** 4 / (1 - _E2_BESSEL))
_G0 = np.arcsin(np.sin(_PHI_C) / _KB)
_T0 = (np.tan(np.pi / 4 + _G0 / 2)
       * ((1 + _E_BESSEL * np.sin(_PHI_C)) / (1 - _E_BESSEL * np.sin(_PHI_C))) ** (_E_BESSEL * _KB / 2)
       / np.tan(np.pi / 4 + _PHI_C / 2) ** _KB)
_KN = np.sin(_PHI_P)
_R0 = _K_P * _KA / np.tan(_PHI_P)

# EPSG:1623 S-JTSK → WGS 84 (position vector): metres, arc-seconds, ppm.
_HELMERT = (570.8, 85.7, 462.8, 4.998, 1.587, 5.261, 3.56)
_A_WGS = 6378137.0
_F_WGS = 1 / 298.257223563
_E2_WGS = 2 * _F_WGS - _F_WGS ** 2


def krovak_inverse(southing, westing):
    """Křovák (south-oriented X, west-oriented Y, metres) → Bessel latitude/longitude in radians."""
    xs = np.asarray(southing, float)
    yw = np.asarray(westing, float)
    r = np.hypot(xs, yw)
    d = np.arctan2(yw, xs) / _KN
    t = 2 * (np.arctan((_R0 / r) ** (1 / _KN) * np.tan(np.pi / 4 + _PHI_P / 2)) - np.pi / 4)
    u = np.arcsin(np.cos(_ALPHA_C) * np.sin(t) - np.sin(_ALPHA_C) * np.cos(t) * np.cos(d))
    v = np.arcsin(np.cos(t) * np.sin(d) / np.cos(u))
    phi = u
    for _ in range(8):                         # converges to < 1e-12 rad in four
        es = _E_BESSEL * np.sin(phi)
        phi = 2 * (np.arctan(_T0 ** (-1 / _KB) * np.tan(u / 2 + np.pi / 4) ** (1 / _KB)
                             * ((1 + es) / (1 - es)) ** (_E_BESSEL / 2)) - np.pi / 4)
    return phi, _LAM_0 - v / _KB


def krovak_forward(phi, lam):
    """Bessel latitude/longitude (radians) → Křovák (southing, westing) in metres. Tests only."""
    es = _E_BESSEL * np.sin(phi)
    u = 2 * (np.arctan(_T0 * np.tan(phi / 2 + np.pi / 4) ** _KB
                       / ((1 + es) / (1 - es)) ** (_E_BESSEL * _KB / 2)) - np.pi / 4)
    v = _KB * (_LAM_0 - lam)
    t = np.arcsin(np.cos(_ALPHA_C) * np.sin(u) + np.sin(_ALPHA_C) * np.cos(u) * np.cos(v))
    d = np.arcsin(np.cos(u) * np.sin(v) / np.cos(t))
    r = _R0 * np.tan(np.pi / 4 + _PHI_P / 2) ** _KN / np.tan(t / 2 + np.pi / 4) ** _KN
    return r * np.cos(_KN * d), r * np.sin(_KN * d)


def _geodetic_to_ecef(phi, lam, a, e2):
    n = a / np.sqrt(1 - e2 * np.sin(phi) ** 2)
    return (n * np.cos(phi) * np.cos(lam), n * np.cos(phi) * np.sin(lam),
            n * (1 - e2) * np.sin(phi))


def _ecef_to_geodetic(x, y, z, a, e2):
    lam = np.arctan2(y, x)
    p = np.hypot(x, y)
    phi = np.arctan2(z, p * (1 - e2))
    for _ in range(6):
        n = a / np.sqrt(1 - e2 * np.sin(phi) ** 2)
        phi = np.arctan2(z + e2 * n * np.sin(phi), p)
    return phi, lam


def sjtsk_to_wgs84(x, y):
    """EPSG:5514 (x = easting, y = northing, both negative) → WGS 84 ``(lon, lat)`` in degrees.

    EPSG:5514 is the East-North variant of Křovák: ``x = −westing``, ``y = −southing``.
    """
    phi, lam = krovak_inverse(-np.asarray(y, float), -np.asarray(x, float))
    X, Y, Z = _geodetic_to_ecef(phi, lam, _A_BESSEL, _E2_BESSEL)
    tx, ty, tz, rx, ry, rz, s = _HELMERT
    rx, ry, rz = (np.radians(v / 3600) for v in (rx, ry, rz))
    m = 1 + s * 1e-6
    X2 = tx + m * (X - rz * Y + ry * Z)                # position-vector rotation, small angles
    Y2 = ty + m * (rz * X + Y - rx * Z)
    Z2 = tz + m * (-ry * X + rx * Y + Z)
    phi2, lam2 = _ecef_to_geodetic(X2, Y2, Z2, _A_WGS, _E2_WGS)
    return np.degrees(lam2), np.degrees(phi2)


def wgs84_to_sjtsk(lon, lat):
    """WGS 84 degrees → EPSG:5514 ``(x, y)``: the inverse of :func:`sjtsk_to_wgs84`.

    The Helmert is reversed by negating its parameters — exact to millimetres at these
    rotation sizes — then the Křovák forward projection.
    """
    X, Y, Z = _geodetic_to_ecef(np.radians(np.asarray(lat, float)),
                                np.radians(np.asarray(lon, float)), _A_WGS, _E2_WGS)
    tx, ty, tz, rx, ry, rz, s = (-v for v in _HELMERT)
    rx, ry, rz = (np.radians(v / 3600) for v in (rx, ry, rz))
    m = 1 + s * 1e-6
    X2 = tx + m * (X - rz * Y + ry * Z)
    Y2 = ty + m * (rz * X + Y - rx * Z)
    Z2 = tz + m * (-ry * X + rx * Y + Z)
    phi, lam = _ecef_to_geodetic(X2, Y2, Z2, _A_BESSEL, _E2_BESSEL)
    southing, westing = krovak_forward(phi, lam)
    return -westing, -southing


# ── VFR reader ─────────────────────────────────────────────────────────────────────
_GML = "{http://www.opengis.net/gml/3.2}"
# The object elements read from the state file, by local name → output table.
_OBJECTS = {"Okres": "okresy", "Obec": "obce", "CastObce": "casti_obce",
            "KatastralniUzemi": "ku", "Zsj": "zsj"}


def _local(tag: str) -> str:
    return tag.rsplit("}", 1)[-1]


def _child(el, name: str):
    for c in el:
        if _local(c.tag) == name:
            return c
    return None


def _child_text(el, name: str) -> str | None:
    c = _child(el, name)
    return None if c is None else (c.text or "").strip() or None


def _ref_kod(el, name: str) -> int | None:
    """``<x:Obec><obi:Kod>575640</obi:Kod></x:Obec>`` → 575640."""
    c = _child(el, name)
    if c is None:
        return None
    k = _child_text(c, "Kod")
    return int(k) if k else None


def _ring_coords(ring_el) -> np.ndarray:
    """All ``posList``/``pos`` under one ``LinearRing`` or ``Ring`` (curve members in order).

    Arc segments (``ArcString``, 7 in the whole state file) are kept as their control points —
    a chord approximation of a few metres, immaterial at municipal scale.
    """
    vals: list[float] = []
    for e in ring_el.iter():
        t = _local(e.tag)
        if t in ("posList", "pos") and e.text:
            vals.extend(float(v) for v in e.text.split())
    pts = np.asarray(vals, float).reshape(-1, 2)
    if len(pts) and not np.array_equal(pts[0], pts[-1]):
        pts = np.vstack([pts, pts[:1]])
    return pts


def _polygons(geom_el) -> list[list[np.ndarray]]:
    """A ``MultiSurface`` → list of polygons, each ``[exterior, *interiors]`` (S-JTSK metres)."""
    out = []
    for poly in geom_el.iter(f"{_GML}Polygon"):
        rings = []
        for part in poly:
            if _local(part.tag) in ("exterior", "interior"):
                for ring in part:
                    rings.append(_ring_coords(ring))
        if rings:
            out.append(rings)
    return out


def _definition_point(geom_el) -> tuple[float, float]:
    """First ``gml:pos`` under ``DefinicniBod`` → ``(x, y)`` S-JTSK, else NaNs."""
    if geom_el is None:
        return (np.nan, np.nan)
    db = _child(geom_el, "DefinicniBod")
    if db is None:
        return (np.nan, np.nan)
    pos = next(db.iter(f"{_GML}pos"), None)
    if pos is None or not pos.text:
        return (np.nan, np.nan)
    x, y = pos.text.split()[:2]
    return (float(x), float(y))


def _boundary(geom_el):
    """The most detailed boundary carried: ``OriginalniHranice``, else the lowest-numbered
    ``GeneralizovaneHraniceN`` (the state file carries one level per object type)."""
    if geom_el is None:
        return None
    cands = {_local(c.tag): c for c in geom_el}
    if "OriginalniHranice" in cands:
        return cands["OriginalniHranice"]
    gen = sorted(k for k in cands if k.startswith("GeneralizovaneHranice"))
    return cands[gen[0]] if gen else None


def _read_object(el, kind: str, with_polygon: bool) -> dict:
    geom = _child(el, "Geometrie")
    x, y = _definition_point(geom)
    rec = {"kod": int(_child_text(el, "Kod")), "name": _child_text(el, "Nazev"), "x": x, "y": y}
    if kind == "obce":
        rec["okres_kod"] = _ref_kod(el, "Okres")
        rec["status_kod"] = _child_text(el, "StatusKod")
    elif kind == "casti_obce":
        rec["obec_kod"] = _ref_kod(el, "Obec")
    elif kind == "ku":
        rec["obec_kod"] = _ref_kod(el, "Obec")
    elif kind == "zsj":
        rec["ku_kod"] = _ref_kod(el, "KatastralniUzemi")
        rec["zsj_character"] = _child_text(el, "CharakterZsjKod")
    if with_polygon:
        b = _boundary(geom)
        rec["polygons"] = _polygons(b) if b is not None else []
    return rec


def read_vfr(zip_path) -> dict[str, pd.DataFrame]:
    """Stream the ``ST_UKSG`` state file into ``okresy`` / ``obce`` / ``casti_obce`` / ``ku`` / ``zsj``.

    Polygons are kept (S-JTSK, list of ``[exterior, *interiors]``) for obce and KÚ only; ZSJ
    carry their original boundaries in the file, which nothing here needs. Every table gains
    WGS 84 ``lat``/``lon`` for its definition point.
    """
    import zipfile
    from lxml import etree

    with_polygon = {"obce", "ku"}
    rows: dict[str, list[dict]] = {k: [] for k in _OBJECTS.values()}
    with zipfile.ZipFile(zip_path) as zf:
        name = next(n for n in zf.namelist() if n.endswith(".xml"))
        with zf.open(name) as fh:
            for _ev, el in etree.iterparse(fh, events=("end",), huge_tree=True):
                kind = _OBJECTS.get(_local(el.tag))
                # An object carries a gml:id; its container does not — and two containers
                # (``vf:KatastralniUzemi``, ``vf:Zsj``) share their member's name.
                if (kind is None or f"{_GML}id" not in el.attrib
                        or not el.tag.startswith("{urn:cz:isvs:ruian:schemas:VymennyFormatTypy")):
                    continue
                rows[kind].append(_read_object(el, kind, kind in with_polygon))
                el.clear()
                while el.getprevious() is not None:          # free parsed siblings
                    del el.getparent()[0]
    out = {}
    for kind, recs in rows.items():
        df = pd.DataFrame(recs)
        lon, lat = sjtsk_to_wgs84(df["x"].to_numpy(), df["y"].to_numpy())
        df["lat"], df["lon"] = lat, lon
        out[kind] = df
    return out


# ── geometry output: WKB + GeoParquet, no geo stack ───────────────────────────────
def wkb_multipolygon(polygons: list[list[np.ndarray]]) -> bytes:
    """``[[exterior, *interiors], …]`` (each ring ``(N, 2)``) → little-endian ISO WKB MultiPolygon."""
    import struct

    parts = [struct.pack("<BII", 1, 6, len(polygons))]
    for rings in polygons:
        parts.append(struct.pack("<BII", 1, 3, len(rings)))
        for ring in rings:
            parts.append(struct.pack("<I", len(ring)))
            parts.append(np.ascontiguousarray(ring, dtype="<f8").tobytes())
    return b"".join(parts)


def _polygons_lonlat(polygons: list[list[np.ndarray]]) -> list[list[np.ndarray]]:
    out = []
    for rings in polygons:
        new = []
        for r in rings:
            lon, lat = sjtsk_to_wgs84(r[:, 0], r[:, 1])
            new.append(np.column_stack([lon, lat]))
        out.append(new)
    return out


def write_geoparquet(df: pd.DataFrame, path, polygon_col: str = "polygons") -> None:
    """Write *df* as GeoParquet 1.1: *polygon_col* (S-JTSK rings) → WKB ``geometry`` in OGC:CRS84.

    The CRS is left out of the metadata on purpose — GeoParquet's default **is** OGC:CRS84
    (lon/lat, WGS 84), so any reader (geopandas, QGIS, DuckDB) places it without PROJJSON.
    """
    import json

    import pyarrow as pa
    import pyarrow.parquet as pq

    flat = df.drop(columns=[polygon_col])
    lonlat = [_polygons_lonlat(p) for p in df[polygon_col]]
    wkb = [wkb_multipolygon(p) if p else None for p in lonlat]
    allpts = [np.vstack([r for rings in p for r in rings]) for p in lonlat if p]
    bbox = (min(a[:, 0].min() for a in allpts), min(a[:, 1].min() for a in allpts),
            max(a[:, 0].max() for a in allpts), max(a[:, 1].max() for a in allpts))
    table = pa.Table.from_pandas(flat, preserve_index=False)
    table = table.append_column("geometry", pa.array(wkb, type=pa.binary()))
    meta = {"version": "1.1.0", "primary_column": "geometry",
            "columns": {"geometry": {"encoding": "WKB", "geometry_types": ["MultiPolygon"],
                                     "bbox": [float(v) for v in bbox]}}}
    table = table.replace_schema_metadata({**(table.schema.metadata or {}),
                                           b"geo": json.dumps(meta).encode()})
    pq.write_table(table, path)


def read_geoparquet_polygons(path) -> pd.DataFrame:
    """Read a table written by :func:`write_geoparquet` back, ``geometry`` decoded to lon/lat rings."""
    import struct

    df = pd.read_parquet(path)

    def decode(b):
        if b is None:
            return []
        _bo, _t, npoly = struct.unpack_from("<BII", b, 0)
        pos, polys = 9, []
        for _ in range(npoly):
            _bo, _t, nring = struct.unpack_from("<BII", b, pos)
            pos += 9
            rings = []
            for _ in range(nring):
                (n,) = struct.unpack_from("<I", b, pos)
                pos += 4
                rings.append(np.frombuffer(b, "<f8", 2 * n, pos).reshape(n, 2))
                pos += 16 * n
            polys.append(rings)
        return polys

    df["geometry"] = df["geometry"].map(decode)
    return df

TABLE_KEYS: dict[str, list[str]] = {
    "ruian_obce": ["obec_kod"],
    "ruian_casti_obce": ["cast_kod"],
    "ruian_ku": ["ku_kod"],
    "ruian_zsj": ["zsj_kod"],
    "ruian_obce_adjacency": ["obec_kod", "neighbour_kod"],
}


def build_tables(geo: dict[str, pd.DataFrame]) -> dict[str, pd.DataFrame]:
    """The four published tables (plus adjacency) from :func:`read_vfr` output, keys asserted."""
    okres_name = dict(zip(geo["okresy"]["kod"], geo["okresy"]["name"]))
    okres_name[PRAHA_OKRES[0]] = PRAHA_OKRES[1]

    obce = geo["obce"].rename(columns={"kod": "obec_kod", "name": "obec_name"})
    obce["okres_kod"] = obce["okres_kod"].fillna(PRAHA_OKRES[0]).astype("int64")
    obce["okres_name"] = obce["okres_kod"].map(okres_name)
    obce = obce[["obec_kod", "obec_name", "status_kod", "okres_kod", "okres_name",
                 "x", "y", "lat", "lon", "polygons"]]

    obec_info = obce.set_index("obec_kod")[["obec_name", "okres_kod", "okres_name"]]
    casti = geo["casti_obce"].rename(columns={"kod": "cast_kod", "name": "cast_name"})
    casti = casti.join(obec_info, on="obec_kod", validate="m:1")
    casti = casti[["cast_kod", "cast_name", "obec_kod", "obec_name", "okres_kod", "okres_name",
                   "x", "y", "lat", "lon"]]

    ku = geo["ku"].rename(columns={"kod": "ku_kod", "name": "ku_name"})
    ku = ku.join(obec_info[["obec_name"]], on="obec_kod", validate="m:1")
    ku = ku[["ku_kod", "ku_name", "obec_kod", "obec_name", "x", "y", "lat", "lon", "polygons"]]

    zsj = geo["zsj"].rename(columns={"kod": "zsj_kod", "name": "zsj_name"})
    zsj = zsj.join(ku.set_index("ku_kod")[["ku_name", "obec_kod"]], on="ku_kod", validate="m:1")
    zsj = zsj.join(obec_info, on="obec_kod", validate="m:1")
    zsj = zsj[["zsj_kod", "zsj_name", "zsj_character", "ku_kod", "ku_name", "obec_kod",
               "obec_name", "okres_kod", "okres_name", "x", "y", "lat", "lon"]]

    out = {"ruian_obce": obce, "ruian_casti_obce": casti, "ruian_ku": ku, "ruian_zsj": zsj,
           "ruian_obce_adjacency": obce_adjacency(obce)}
    for name, key in TABLE_KEYS.items():
        d = out[name]
        assert d[key].notna().all().all() and not d.duplicated(key).any(), name
    for name in ("ruian_casti_obce", "ruian_ku", "ruian_zsj"):         # every parent found
        assert out[name]["obec_name"].notna().all(), name
    return out


def obce_adjacency(obce: pd.DataFrame, tol_m: float = 0.01) -> pd.DataFrame:
    """Pairs of obce whose boundaries meet, both directions, with the count of shared vertices.

    RÚIAN's generalised obec boundaries are topologically consistent — every vertex of a shared
    line is stored identically on both sides (measured: 491,602 vertices shared by exactly two
    obce, 12k by three) — so adjacency is vertex sharing, found with a KD-tree at *tol_m*.
    ``shared_vertices == 1`` is a corner touch; a shared edge has two or more.
    """
    from scipy.spatial import cKDTree

    xy, kod = [], []
    for k, polys in zip(obce["obec_kod"], obce["polygons"]):
        for rings in polys:
            for r in rings:
                xy.append(r[:-1])                      # the closing vertex repeats the first
                kod.append(np.full(len(r) - 1, k))
    xy, kod = np.vstack(xy), np.concatenate(kod)
    pairs = cKDTree(xy).query_pairs(r=tol_m, output_type="ndarray")
    a, b = kod[pairs[:, 0]], kod[pairs[:, 1]]
    keep = a != b
    p = pd.DataFrame({"obec_kod": np.minimum(a, b)[keep], "neighbour_kod": np.maximum(a, b)[keep]})
    p = p.value_counts().rename("shared_vertices").reset_index()
    both = pd.concat([p, p.rename(columns={"obec_kod": "neighbour_kod",
                                           "neighbour_kod": "obec_kod"})], ignore_index=True)
    return both.sort_values(["obec_kod", "neighbour_kod"]).reset_index(drop=True)


def write_tables(tables: dict[str, pd.DataFrame], out_dir) -> None:
    """GeoParquet for tables with polygons, plain Parquet otherwise, and a flat CSV beside each."""
    from pathlib import Path

    out_dir = Path(out_dir)
    for name, d in tables.items():
        flat = d.drop(columns=["polygons"], errors="ignore")
        if "polygons" in d.columns:
            write_geoparquet(d, out_dir / f"{name}.parquet")
        else:
            flat.to_parquet(out_dir / f"{name}.parquet", index=False)
        flat.to_csv(out_dir / f"{name}.csv", index=False, encoding="utf-8")


def ruian_report(tables: dict[str, pd.DataFrame]) -> None:
    """Row counts and neighbour degree."""
    for name, d in tables.items():
        print(f"    {name:<22} {len(d):>8,} rows")
    adj = tables["ruian_obce_adjacency"]
    deg = adj.groupby("obec_kod").size()
    print(f"    obce with >=1 neighbour {deg.size:,} of {len(tables['ruian_obce']):,}; "
          f"median degree {deg.median():.0f}, max {deg.max()}; "
          f"corner-only pairs {int((adj['shared_vertices'] == 1).sum() // 2)}")


# ── vectorised point-in-court (MPIDR 1910) ──────────────────────────────────────────
def _in_rings(px: np.ndarray, py: np.ndarray, rings, chunk: int = 4_000_000) -> np.ndarray:
    """Even-odd containment of many points in one polygon (all rings), vectorised over edges."""
    inside = np.zeros(len(px), bool)
    for ring in rings:
        a, b = ring[:-1], ring[1:]
        xi, yi, xj, yj = a[None, :, 0], a[None, :, 1], b[None, :, 0], b[None, :, 1]
        step = max(1, chunk // max(len(a), 1))
        for s in range(0, len(px), step):
            x, y = px[s:s + step, None], py[s:s + step, None]
            with np.errstate(divide="ignore", invalid="ignore"):
                cross = ((yi > y) != (yj > y)) & (x < (xj - xi) * (y - yi) / (yj - yi) + xi)
            inside[s:s + step] ^= (cross.sum(1) % 2).astype(bool)
    return inside


def locate_courts(lon, lat, locator) -> tuple[np.ndarray, np.ndarray]:
    """``(court_id, okres_id)`` of the MPIDR 1910 court containing each WGS 84 point; NaN outside.

    Same answer as :meth:`spatial.CourtLocator.locate` (even-odd over all rings, first court in
    file order wins), vectorised: ~90k points in seconds instead of minutes.
    """
    from utils.parsing.spatial import laea_forward

    lon, lat = np.asarray(lon, float), np.asarray(lat, float)
    court = np.full(len(lon), np.nan)
    okres = np.full(len(lon), np.nan)
    ok = ~(np.isnan(lon) | np.isnan(lat))
    px, py = np.full(len(lon), np.nan), np.full(len(lon), np.nan)
    px[ok], py[ok] = laea_forward(lon[ok], lat[ok])
    todo = ok.copy()
    for court_id, okres_code, rings, (x0, y0, x1, y1) in locator.courts:
        cand = np.flatnonzero(todo & (px >= x0) & (px <= x1) & (py >= y0) & (py <= y1))
        if not len(cand):
            continue
        hit = cand[_in_rings(px[cand], py[cand], rings)]
        r, d = okres_code.split("-")
        court[hit], okres[hit] = court_id, int(r) * 1000 + int(d)
        todo[hit] = False
    return court, okres


def locate_obce(x, y, obce: pd.DataFrame) -> np.ndarray:
    """``obec_kod`` of the RÚIAN obec containing each S-JTSK point (NaN outside every obec)."""
    out = np.full(len(x), np.nan)
    todo = ~(np.isnan(x) | np.isnan(y))
    for kod, polys in zip(obce["obec_kod"], obce["polygons"]):
        pts = np.vstack([r for rings in polys for r in rings])
        cand = np.flatnonzero(todo & (x >= pts[:, 0].min()) & (x <= pts[:, 0].max())
                              & (y >= pts[:, 1].min()) & (y <= pts[:, 1].max()))
        if not len(cand):
            continue
        inside = np.zeros(len(cand), bool)
        for rings in polys:
            inside |= _in_rings(x[cand], y[cand], rings)
        out[cand[inside]] = kod
        todo[cand[inside]] = False
    return out


def distance_to_okres(lon, lat, okres_ids, locator) -> np.ndarray:
    """Metres from each point to the nearest edge of its own 1910 okres (NaN if no okres)."""
    from utils.parsing.spatial import _ring_min_distance, laea_forward

    out = np.full(len(lon), np.nan)
    px, py = laea_forward(np.asarray(lon, float), np.asarray(lat, float))
    for i, (x, y, oid) in enumerate(zip(px, py, okres_ids)):
        if pd.isna(oid) or np.isnan(x):
            continue
        rings = [r for e in locator.by_okres.get(int(oid), []) for r in e[2]]
        if rings:
            out[i] = min(_ring_min_distance(x, y, r) for r in rings)
    return out


# ── the 1910 → current crosswalk ─────────────────────────────────────────────────────
# Where a matched place's point comes from, best first. A část obce is the settlement itself;
# an obec's definition point usually sits in its eponymous part; a ZSJ is a statistical unit
# whose point can sit in fields; Geonames and PRNG are names on a map.
SOURCE_RANK = {"ruian_part": 0, "ruian_obec": 1, "ruian_zsj": 2, "geonames": 3,
               "wikidata_vanished": 4, "prng": 5}
# Sources are consulted tier by tier inside a scope: the official register first, then the
# state map's names, then places that no longer exist, then the Polish register. A later tier is
# read only where every earlier one has nothing of that name — so a vanished hamlet can never
# outbid a living village, and a ruin never a part.
SOURCE_TIER = {"ruian_part": 0, "ruian_obec": 0, "ruian_zsj": 0, "geonames": 1,
               "wikidata_vanished": 2, "prng": 3}
# Geonames layers used as candidates. Its obec / část obce layers duplicate RÚIAN and are left out.
GEONAMES_LAYERS = ("mistni_cast", "castle", "works", "lodge")
# The project's fuzzy threshold (records_geo, D-22): token similarity ≥ 90, ties refused.
FUZZY_THRESHOLD = 90
SCOPES = ("okres_1910", "neighbour", "modern_okres")
# Wikidata and the crosswalk disagree when their points are this far apart. Read off the
# measured distribution, not chosen: where both exist the median is 8 m and the 99th
# percentile 81 m, so 5 km is far outside agreement and well inside "a different place".
WD_CONFLICT_M = 5000.0
# A neighbour-scope candidate must also lie this close to the 1910 okres it would have left.
# The ring is one municipality wide, but a municipality can be Praha. Measured before the cap:
# of 1,544 neighbour matches 1,509 sat within 5 km (1,024 within 1 km, the MPIDR edge noise);
# the 6 beyond 5 km included a Prague Háje matched to the Háje in Roztoky.
NEIGHBOUR_MAX_M = 5000.0


def name_key(v) -> str:
    """The matching key: :func:`records_geo._norm` with every space removed.

    Space-free because the ČSÚ register's text layer breaks words at random (``Kutná Hor a``),
    the fault :func:`records_geo._nospace_index` already repairs for okres names.
    """
    from utils.parsing import records_geo as rg

    return rg._norm(v).replace(" ", "")


# The register writes the capital's okres as plain "Praha"; RÚIAN as *Hlavní město Praha*.
_MODERN_OKRES_ALIAS = {"praha": "hlavnimestopraha"}


def modern_okres_key(okres_all) -> str:
    """The last okres of the register's ``okres_all`` timeline, as a :func:`name_key`.

    The timeline is the register's own record of where a settlement went, so its last entry is
    the modern okres (2011) — read off the text layer, hence the stray brackets it sometimes
    carries (``Sokolov)``) are dropped.
    """
    if not isinstance(okres_all, str) or not okres_all.strip():
        return ""
    k = name_key(okres_all.split("|")[-1].replace("(", "").replace(")", ""))
    return _MODERN_OKRES_ALIAS.get(k, k)


def candidate_places(tables: dict[str, pd.DataFrame], locator,
                     geonames: pd.DataFrame | None = None,
                     prng: pd.DataFrame | None = None,
                     vanished: pd.DataFrame | None = None) -> pd.DataFrame:
    """Every named modern place that a 1910 settlement could be, with its 1910 court and okres.

    One row per (source, place). ``obec_kod`` is the modern municipality (``<NA>`` for PRNG);
    ``court_id`` / ``okres_id_1910`` come from locating the point in the MPIDR 1910 courts.
    """
    obce, casti, zsj = tables["ruian_obce"], tables["ruian_casti_obce"], tables["ruian_zsj"]
    frames = [
        casti.assign(source="ruian_part", place_id=casti["cast_kod"].astype(str),
                     name=casti["cast_name"]),
        obce.assign(source="ruian_obec", place_id=obce["obec_kod"].astype(str),
                    name=obce["obec_name"]),
        zsj.assign(source="ruian_zsj", place_id=zsj["zsj_kod"].astype(str), name=zsj["zsj_name"]),
    ]
    if geonames is not None:
        gn = geonames[geonames["layer"].isin(GEONAMES_LAYERS)]
        # Filed under a modern obec by (obec name, okres name); a pair two obce share is dropped.
        o = obce.assign(k=obce["obec_name"].map(name_key) + "|" + obce["okres_name"].map(name_key))
        o = o[~o["k"].duplicated(keep=False)].set_index("k")
        k = gn["obec_name"].map(name_key) + "|" + gn["okres_name"].fillna("").map(name_key)
        gn = gn.assign(obec_kod=k.map(o["obec_kod"]), source="geonames",
                       place_id=gn["geonames_id"])
        # The rest (4 %: renamed obce, a name two obce of one okres share) by their point.
        if "polygons" in obce.columns and gn["obec_kod"].isna().any():
            miss = gn["obec_kod"].isna()
            gn.loc[miss, "obec_kod"] = locate_obce(gn.loc[miss, "x"].to_numpy(float),
                                                   gn.loc[miss, "y"].to_numpy(float), obce)
        gn = gn.join(obce.set_index("obec_kod")[["okres_name"]].rename(
            columns={"okres_name": "okres_modern"}), on="obec_kod")
        frames.append(gn.drop(columns=["okres_name"]).rename(columns={"okres_modern": "okres_name"}))
    if vanished is not None:
        # Settlement-like items only, one candidate row per label (Czech and German): the
        # register's headword for a German hamlet is often the German name.
        v = vanished[vanished["settlement_like"] & vanished["lat"].notna()]
        v = v.join(obce.set_index("obec_kod")[["okres_name"]], on="obec_kod")
        for col in ("label_cs", "label_de"):
            frames.append(v[v[col].notna()].assign(source="wikidata_vanished",
                                                   place_id=v["qid"], name=v[col]))
    if prng is not None:
        frames.append(prng.assign(source="prng", place_id=prng["prng_id"].astype(str),
                                  obec_kod=np.nan, okres_name=prng["powiat"]))
    cols = ["source", "place_id", "name", "obec_kod", "okres_name", "lat", "lon"]
    cand = pd.concat([f[cols] for f in frames if len(f)], ignore_index=True)
    cand["obec_kod"] = cand["obec_kod"].astype("Int64")
    cand["key"] = cand["name"].map(name_key)
    cand["okres_key"] = cand["okres_name"].map(lambda v: name_key(v) if isinstance(v, str) else "")
    court, okres = locate_courts(cand["lon"].to_numpy(), cand["lat"].to_numpy(), locator)
    cand["court_id"] = pd.array(court, dtype="Int64")
    cand["okres_id_1910"] = pd.array(okres, dtype="Int64")
    # A place's identity is its modern obec (PRNG: the locality itself). Two sources naming the
    # same key inside one obec are the same place seen twice, not two candidates.
    cand["cluster"] = np.where(cand["obec_kod"].notna(), "o" + cand["obec_kod"].astype(str),
                               "p" + cand["place_id"])
    cand["rank"] = cand["source"].map(SOURCE_RANK)
    cand = cand[cand["key"] != ""].drop_duplicates(["source", "place_id", "key"])
    return cand.reset_index(drop=True)


def _okres_rings(cand: pd.DataFrame, adjacency: pd.DataFrame) -> dict[int, set]:
    """1910 okres → the modern obce reaching into it **plus their neighbours**.

    An obec "reaches into" a 1910 okres when any of its RÚIAN points (obec, part or ZSJ) lies
    in that okres's courts. The neighbour ring is one municipality wide — enough to catch a
    hamlet moved across a boundary, not so wide that a common name finds a stranger.
    """
    ru_pts = cand[cand["source"].str.startswith("ruian") & cand["okres_id_1910"].notna()]
    foot = ru_pts.groupby("okres_id_1910")["obec_kod"].agg(lambda s: set(s.dropna()))
    nb = adjacency.groupby("obec_kod")["neighbour_kod"].agg(set)
    out = {}
    for oid, obce in foot.items():
        ring = set(obce)
        for k in obce:
            ring |= nb.get(k, set())
        out[int(oid)] = ring
    return out


def _official_first(rows: pd.DataFrame) -> pd.DataFrame:
    """The rows of the first :data:`SOURCE_TIER` present: RÚIAN, else Geonames, else the
    vanished places, else PRNG.

    A castle or a mill that shares a village's name is not a rival for the village: each later
    source is consulted only where every earlier one has nothing of that name in scope.
    """
    tier = rows["source"].map(SOURCE_TIER)
    return rows[tier == tier.min()]


def _pick(rows: pd.DataFrame, s, obec_names: dict, part_names: dict) -> tuple:
    """Choose among candidate rows of one name and scope. Returns ``(row | None, why)``.

    One cluster → that place. Several → kept only if the settlement's **own** evidence names
    one of them: its modern okres (the last entry of the register's ``okres_all`` timeline),
    then its 1910 parent commune. Never by order (D-22): what neither decides is refused.
    """
    clusters = rows["cluster"].unique()
    why = "unique"
    if len(clusters) > 1:
        if s.okres_modern:
            keep = rows[rows["okres_key"] == s.okres_modern]["cluster"].unique()
            if len(keep) == 1:
                clusters, why = keep, "modern_okres"
        if len(clusters) > 1 and s.parent_key:
            keep = [c for c in clusters if c.startswith("o") and (
                name_key(obec_names.get(int(c[1:]), "")) == s.parent_key
                or s.parent_key in part_names.get(int(c[1:]), set()))]
            if len(keep) == 1:
                clusters, why = np.array(keep), "parent_1910"
        if len(clusters) > 1:
            return None, "ambiguous"
    best = rows[rows["cluster"] == clusters[0]].sort_values("rank").iloc[0]
    return best, why


# Fallback name forms, tried only when the headword itself finds nothing anywhere.
# A place the register splits into numbered halves (``Hlásek 1. díl``, ``Království I``) is one
# place today; the halves share it.
_HALF = re.compile(r"\s+(?:\d+|[IVX]+)\s*\.?\s*(?:d[ií]l)?\s*$", re.I)
# A part named for its position in a pair (``Horní Kšely``, ``Moravský Leskovec``) is often
# filed today under the bare name — but only accepted inside its own 1910 parent's modern obec.
_PREFIXES = {"horni", "dolni", "novy", "nova", "nove", "stary", "stara", "stare", "velky",
             "velka", "velke", "maly", "mala", "male", "predni", "zadni", "vysni", "nizni",
             "cesky", "ceska", "ceske", "nemecky", "nemecka", "nemecke", "moravsky", "moravska",
             "moravske"}
VARIANT_KINDS = ("half", "prefix")   # fallback name forms; see _refuse_shared for how each shares


def half_key(cs_name) -> str:
    """The key with a trailing half number stripped; ``""`` when there is none."""
    if not isinstance(cs_name, str):
        return ""
    k = name_key(_HALF.sub("", cs_name))
    return k if k and k != name_key(cs_name) else ""


def prefix_key(cs_name) -> str:
    """The key without a leading positional word (Horní, Nový, Moravský …); ``""`` when none."""
    from utils.parsing import records_geo as rg

    toks = rg._norm(cs_name).split() if isinstance(cs_name, str) else []
    return "".join(toks[1:]) if len(toks) > 1 and toks[0] in _PREFIXES else ""


def register_courts(register_places: pd.DataFrame | None) -> dict[str, int]:
    """``settlement_id`` → the 1910 court the register files it under, where the register says so
    unambiguously.

    The 1910 register (*Podrobný seznam míst*) is printed court by court, and the
    ``register-variables`` build anchors each of its places to a ``settlement_id`` (D-148). Only
    places it **anchored** (its own ``ANCHORED`` set — not a tie, not a guess) are used, and a
    settlement the register files under two different courts gives no court at all.
    """
    if register_places is None or not len(register_places):
        return {}
    from utils.lexikon_1910.lexikon_1910_register import ANCHORED

    p = register_places[register_places["anchor_status"].isin(ANCHORED)
                        & register_places["settlement_id"].notna()
                        & register_places["court_id"].notna()]
    courts = p.groupby("settlement_id")["court_id"].agg(lambda c: set(c.astype(int)))
    return {sid: next(iter(c)) for sid, c in courts.items() if len(c) == 1}


def build_crosswalk(settlements: pd.DataFrame, tables: dict[str, pd.DataFrame], locator,
                    geonames: pd.DataFrame | None = None,
                    prng: pd.DataFrame | None = None,
                    vanished: pd.DataFrame | None = None,
                    courts_1910: dict[str, int] | None = None) -> pd.DataFrame:
    """One row per ``settlement_id``: the modern place a 1910 settlement is, and where it lies.

    Scopes, tried in order and stopping at the first that finds the name:

    1. ``okres_1910``  — candidates whose point lies in the settlement's 1910 okres;
    2. ``neighbour``   — candidates in a modern obec reaching into that okres or adjacent to one;
    3. ``modern_okres``— candidates in the modern okres the register's own timeline names
                         (``okres_all``'s last entry); the only scope for rows with no 1910 okres;
    4. ``okres_1910`` again with **fuzzy** names (ratio ≥ :data:`FUZZY_THRESHOLD` on the space-free
       key), only if every candidate at the top score is one place;
    5. a **numbered half** (``Hlásek 1. díl``) through scopes 1–3 under its bare name, and a
       **positional prefix** (``Horní Kšely`` → ``Kšely``) inside the 1910 parent's modern obec only.

    Where the 1910 register files the settlement under a court (*courts_1910*, from
    :func:`register_courts`), several places of its name in its okres are first narrowed to those
    inside that court's 1910 polygon (``match_why = "court_1910"``), and of two rows claiming one
    place the one whose court holds it keeps it.

    A scope that finds the name in several places and cannot choose by the settlement's own
    evidence **refuses** (``match_status = "ambiguous"``) — it does not fall through to a wider
    scope, which would only find more strangers. Wikidata coordinates are never replaced:
    ``lat``/``lon`` are Wikidata's where it has a point, and ``coord_source`` says which.
    """
    from rapidfuzz import fuzz, process

    cand = candidate_places(tables, locator, geonames, prng, vanished)
    rings = _okres_rings(cand, tables["ruian_obce_adjacency"])
    obec_names = dict(zip(tables["ruian_obce"]["obec_kod"], tables["ruian_obce"]["obec_name"]))
    part_names = (tables["ruian_casti_obce"].assign(k=lambda d: d["cast_name"].map(name_key))
                  .groupby("obec_kod")["k"].agg(set).to_dict())
    by_key = {k: g for k, g in cand.groupby("key")}
    by_okres = {int(k): g for k, g in cand[cand["okres_id_1910"].notna()].groupby("okres_id_1910")}

    s = settlements[["settlement_id", "cs_name", "status_1910", "okres_id", "parent_1910",
                     "okres_all"]].copy()
    s["key"] = s["cs_name"].map(name_key)
    s["parent_key"] = s["parent_1910"].map(lambda v: name_key(v) if isinstance(v, str) else "")
    s["okres_modern"] = s["okres_all"].map(modern_okres_key)
    s["half_key"] = s["cs_name"].map(half_key)
    s["prefix_key"] = s["cs_name"].map(prefix_key)
    courts_1910 = courts_1910 or {}
    s["court_1910"] = s["settlement_id"].map(courts_1910)

    near_cache: dict[tuple, bool] = {}

    def near(sub: pd.DataFrame, oid: int) -> pd.DataFrame:
        """Rows of *sub* within :data:`NEIGHBOUR_MAX_M` of 1910 okres *oid*."""
        todo = [i for i in sub.index if (i, oid) not in near_cache]
        if todo:
            d = distance_to_okres(sub.loc[todo, "lon"], sub.loc[todo, "lat"], [oid] * len(todo),
                                  locator)
            near_cache.update({(i, oid): bool(v <= NEIGHBOUR_MAX_M) for i, v in zip(todo, d)})
        return sub[[near_cache[(i, oid)] for i in sub.index]]

    empty = cand.iloc[0:0]

    def search(key: str, r, oid):
        """Scopes 1–3 for one key → ``(hit, why, scope)``; the first scope with the name decides."""
        rows = by_key.get(key)
        if rows is not None:
            scoped = [
                ("okres_1910", rows[rows["okres_id_1910"] == oid] if oid else empty),
                ("neighbour", near(rows[rows["obec_kod"].isin(rings.get(oid, set()))
                                        & (rows["okres_id_1910"] != oid).fillna(True)], oid)
                 if oid else empty),
                ("modern_okres", rows[rows["okres_key"] == r.okres_modern] if r.okres_modern
                 else empty),
            ]
            for sc, sub in scoped:
                if len(sub):
                    sub = _official_first(sub)
                    in_court = (sub[sub["court_id"] == int(r.court_1910)]
                                if sc == "okres_1910" and pd.notna(r.court_1910) else empty)
                    if len(in_court) and in_court["cluster"].nunique() < sub["cluster"].nunique():
                        hit, why = _pick(in_court, r, obec_names, part_names)
                        return hit, ("court_1910" if why == "unique" else why), sc
                    hit, why = _pick(sub, r, obec_names, part_names)
                    return hit, why, sc
        return None, "no_candidate", None

    def in_parent(sub: pd.DataFrame, parent_key: str) -> pd.DataFrame:
        """Rows of *sub* whose modern obec is the 1910 parent, or holds a part of that name."""
        ok = [pd.notna(k) and (name_key(obec_names.get(int(k), "")) == parent_key
                               or parent_key in part_names.get(int(k), set()))
              for k in sub["obec_kod"]]
        return sub[ok]

    def match_one(r) -> dict:
        oid = None if pd.isna(r.okres_id) else int(r.okres_id)
        hit, why, scope = search(r.key, r, oid)
        kind = "exact" if scope else None
        score = 100.0 if scope else np.nan
        if hit is None and why == "no_candidate" and oid in by_okres and len(r.key) >= 4:
            pool = by_okres[oid]
            top = process.extract(r.key, pool["key"].unique(), scorer=fuzz.ratio, limit=None,
                                  score_cutoff=FUZZY_THRESHOLD)
            if top:
                best = max(t[1] for t in top)
                at_top = pool[pool["key"].isin([t[0] for t in top if t[1] == best])]
                hit, why = _pick(_official_first(at_top), r, obec_names, part_names)
                scope, kind, score = "okres_1910", "fuzzy", best
        if hit is None and why == "no_candidate" and r.half_key:
            hit, why, scope = search(r.half_key, r, oid)
            kind, score = ("half", 100.0) if scope else (None, np.nan)
        if hit is None and why == "no_candidate" and r.prefix_key and r.parent_key and oid:
            rows = by_key.get(r.prefix_key)
            if rows is not None:
                sub = in_parent(rows[rows["okres_id_1910"] == oid], r.parent_key)
                if len(sub):
                    hit, why = _pick(_official_first(sub), r, obec_names, part_names)
                    scope, kind, score = "okres_1910", "prefix", 100.0
        rec = {"settlement_id": r.settlement_id, "match_scope": scope, "match_kind": kind,
               "match_score": score, "match_why": why}
        if hit is not None:
            rec.update({"match_source": hit["source"], "place_id": hit["place_id"],
                        "place_name": hit["name"], "obec_kod": hit["obec_kod"],
                        "point_lat": hit["lat"], "point_lon": hit["lon"],
                        "place_court": hit["court_id"]})
        return rec

    xw = pd.DataFrame([match_one(r) for r in s.itertuples(index=False)])
    xw["match_status"] = np.where(xw["match_source"].notna(), "matched",
                                  np.where(xw["match_why"] == "ambiguous", "ambiguous", "none"))
    xw["court_1910_register"] = pd.array(xw["settlement_id"].map(courts_1910), dtype="Int64")
    xw["place_court"] = pd.array(xw.get("place_court", pd.Series(pd.NA, index=xw.index)),
                                  dtype="Int64")
    xw = _refuse_shared(xw, settlements)
    return _finish(xw.drop(columns=["place_court"], errors="ignore"), settlements, tables,
                   locator, vanished)


def _refuse_shared(xw: pd.DataFrame, settlements: pd.DataFrame) -> pd.DataFrame:
    """One modern place, one 1910 settlement — the rule :func:`wikidata.add_settlement_coords`
    applies to a shared Wikidata point.

    Claimants with the same name, 1910 okres and parent are the register's own duplicates and
    share. A row with **no 1910 entry** (``status_1910`` missing: the headword's timeline does not
    reach 1910) yields to a row that has one; otherwise every claimant is refused.

    A **numbered half** (``Česká Lhotice 1. díl``) is part of the village whose bare name it
    matched, so it never contests the place: it rides along with whoever holds it, and is refused
    only when that holder is (two different 1910 villages of that name in one okres). Halves alone
    on a place share it within one okres. A **prefix** match (``Horní Mez`` → ``Mez``) can be a
    different place from a 1910 ``Mez``, so it yields to any 1910 row that found the place under
    its own name, and otherwise contests like any other claimant.
    """
    m = xw[xw["match_status"] == "matched"].join(
        settlements.set_index("settlement_id")[["cs_name", "okres_id", "parent_1910",
                                                "status_1910"]], on="settlement_id")
    m = m.assign(place=m["match_source"] + ":" + m["place_id"].astype(str))
    # The 1910 register's court decides between rival claimants where it can: of the rows that
    # claim one place, those whose register court is known and is not the place's court give it
    # up — provided some claimant's court is the place's court.
    if "court_1910_register" in m.columns:
        same = (m["court_1910_register"] == m["place_court"])
        own = same.fillna(False).astype(bool)
        other = (~same).fillna(False).astype(bool)        # both courts known, and they differ
        has_own = m["place"].isin(m.loc[own, "place"])
        by_court = other & has_own & m["place"].map(m.groupby("place").size()).gt(1)
    else:
        by_court = pd.Series(False, index=m.index)
    court_lost = m.loc[by_court, "settlement_id"]
    m = m[~by_court]
    half = m["match_kind"] == "half"
    prefix = m["match_kind"] == "prefix"
    direct_1910 = m.loc[~half & ~prefix & m["status_1910"].notna(), "place"].unique()
    outranked = prefix & m["place"].isin(direct_1910)
    halves, lost = m[half], m.loc[outranked, "settlement_id"]
    m = m[~half & ~outranked]
    m = m.assign(sig=np.where(m["match_kind"] == "prefix", "prefix|" + m["okres_id"].astype(str),
                              m["cs_name"].map(name_key) + "|" + m["okres_id"].astype(str) + "|"
                              + m["parent_1910"].fillna("").map(name_key)),
                 in_1910=m["status_1910"].notna())
    n_1910 = m[m["in_1910"]].groupby("place")["sig"].nunique()
    n_all = m.groupby("place")["sig"].nunique()
    m = m.assign(n_1910=m["place"].map(n_1910).fillna(0), n_all=m["place"].map(n_all))
    yields = (m["n_all"] > 1) & ~m["in_1910"] & (m["n_1910"] >= 1)
    contested = (m["n_all"] > 1) & ~yields & (m["n_1910"] != 1)
    # Halves: refused where the place's holder was contested, or where halves from two okresy
    # meet on one place with no holder.
    held = set(m["place"])
    refused_places = set(m.loc[contested, "place"])
    half_okres = halves.groupby("place")["okres_id"].nunique(dropna=False)
    bad_half = halves["place"].isin(refused_places) | (
        ~halves["place"].isin(held) & halves["place"].map(half_okres).gt(1))
    bad = pd.concat([m.loc[yields | contested, "settlement_id"], lost, court_lost,
                     halves.loc[bad_half, "settlement_id"]])
    out = xw.copy()
    hit = out["settlement_id"].isin(bad)
    out.loc[hit, "match_status"] = "shared"
    for c in ("match_source", "place_id", "place_name", "obec_kod", "point_lat", "point_lon"):
        out.loc[hit, c] = np.nan
    return out


def _haversine_m(lat1, lon1, lat2, lon2) -> np.ndarray:
    p1, p2 = np.radians(lat1), np.radians(lat2)
    a = (np.sin((p2 - p1) / 2) ** 2
         + np.cos(p1) * np.cos(p2) * np.sin(np.radians(lon2 - lon1) / 2) ** 2)
    return 2 * 6_371_008.8 * np.arcsin(np.sqrt(a))


def coord_refs(out: pd.DataFrame, s: pd.DataFrame, vanished_ids: dict) -> list:
    """A citation for each final point: the source and the record it was read from.

    ``RÚIAN část obce 32247`` · ``RÚIAN obec 569089`` · ``RÚIAN ZSJ 67067`` ·
    ``ČÚZK Geonames mistni_cast:5082`` · ``Wikidata Q123 (vanished; zanikleobce.cz 5539)`` ·
    ``PRNG 1234`` · ``Wikidata Q456`` (the point ``wikidata.py`` attached).
    """
    label = {"ruian_part": "RÚIAN část obce", "ruian_obec": "RÚIAN obec", "ruian_zsj": "RÚIAN ZSJ",
             "geonames": "ČÚZK Geonames", "prng": "PRNG"}
    qid = s["wd_qid"] if "wd_qid" in s.columns else pd.Series(dtype=object)
    refs = []
    for sid, src, pid in zip(out["settlement_id"], out["coord_source"], out["place_id"]):
        if src is None or (isinstance(src, float) and np.isnan(src)):
            refs.append(None)
        elif src == "wikidata":
            refs.append(f"Wikidata {qid.get(sid)}")
        elif src == "wikidata_vanished":
            z = vanished_ids.get(pid)
            refs.append(f"Wikidata {pid} (vanished" + (f"; zanikleobce.cz {z})" if z else ")"))
        else:
            refs.append(f"{label[src]} {pid}")
    return refs


def _finish(xw: pd.DataFrame, settlements: pd.DataFrame, tables, locator,
            vanished: pd.DataFrame | None = None) -> pd.DataFrame:
    """Modern parents, transfer flags, the court-polygon test and the Wikidata comparison."""
    s = settlements.set_index("settlement_id")
    vanished_ids = ({} if vanished is None else
                    dict(zip(vanished["qid"], vanished["zanikleobce_id"].where(
                        vanished["zanikleobce_id"].notna(), None))))
    out = xw.join(s[["cs_name", "status_1910", "okres_id", "parent_1910", "land",
                     "lat", "lon"]].rename(columns={"lat": "wd_lat", "lon": "wd_lon"}),
                  on="settlement_id", validate="1:1")
    obce = tables["ruian_obce"].set_index("obec_kod")
    out["obec_kod"] = out["obec_kod"].astype("Int64")
    out = out.join(obce[["obec_name", "okres_kod", "okres_name"]], on="obec_kod")

    # Transfers. A 1910 commune now a part of another obec; a 1910 part now under another parent.
    key = out["cs_name"].map(name_key)
    obec_key = out["obec_name"].map(lambda v: name_key(v) if isinstance(v, str) else "")
    matched = out["match_status"] == "matched"
    out["obec_absorbed"] = pd.array(np.where(
        matched & (out["status_1910"] == "obec"), key != obec_key, None), dtype="boolean")
    parts = (tables["ruian_casti_obce"].assign(k=lambda d: d["cast_name"].map(name_key))
             .groupby("obec_kod")["k"].agg(set).to_dict())
    pk = out["parent_1910"].map(lambda v: name_key(v) if isinstance(v, str) else "")
    # The register names the parent in its 1910 form (``Mühlbach``), so the parent's own row —
    # same 1910 okres, that name as its ``name_1910`` — supplies the modern headword
    # (``Pomezí nad Ohří``) to compare with as well.
    modern_of = defaultdict(set)
    for n10, cs, oid in zip(settlements["name_1910"], settlements["cs_name"], settlements["okres_id"]):
        if isinstance(n10, str) and pd.notna(oid):
            modern_of[(name_key(n10), int(oid))].add(name_key(cs))
    # The parent is unchanged when it is the modern obec or one of that obec's own parts.
    same_parent = []
    for p, o, k, oid in zip(pk, obec_key, out["obec_kod"], out["okres_id"]):
        names = {p} | (modern_of.get((p, int(oid)), set()) if pd.notna(oid) else set())
        same_parent.append(o in names or (pd.notna(k) and bool(names & parts.get(int(k), set()))))
    same_parent = np.array(same_parent)
    out["parent_changed"] = pd.array(np.where(matched & (pk != ""), ~same_parent, None),
                                     dtype="boolean")

    # The court-polygon test, on the modern point: in which 1910 court and okres does it fall?
    court, okres = locate_courts(out["point_lon"].to_numpy(float), out["point_lat"].to_numpy(float),
                                 locator)
    out["point_court_id"] = pd.array(court, dtype="Int64")
    out["point_okres_id"] = pd.array(okres, dtype="Int64")
    has = out["point_lat"].notna() & out["okres_id"].notna()
    out["in_own_okres"] = pd.array(np.where(has, out["point_okres_id"] == out["okres_id"], None),
                                   dtype="boolean")
    outside = (out["in_own_okres"] == False).fillna(False).to_numpy()   # noqa: E712
    d = np.full(len(out), np.nan)
    d[outside] = distance_to_okres(out.loc[outside, "point_lon"], out.loc[outside, "point_lat"],
                                   out.loc[outside, "okres_id"], locator)
    out["dist_own_okres_m"] = d
    # Against the 1910 register: is the point in the court the register files the settlement under?
    if "court_1910_register" in out.columns:
        has_c = out["court_1910_register"].notna() & out["point_court_id"].notna()
        out["in_register_court"] = pd.array(
            np.where(has_c, out["point_court_id"] == out["court_1910_register"], None),
            dtype="boolean")
    else:
        out["court_1910_register"] = pd.array([pd.NA] * len(out), dtype="Int64")
        out["in_register_court"] = pd.array([pd.NA] * len(out), dtype="boolean")

    # Wikidata, where it has a point: the distance is the validation, and its point is kept —
    # with one exception, stated and flagged. When the two disagree by more than
    # WD_CONFLICT_M, the Wikidata point lies OUTSIDE the settlement's 1910 okres and the
    # crosswalk point inside it, the court test has already said which one is the place.
    out["wd_dist_m"] = _haversine_m(out["wd_lat"], out["wd_lon"], out["point_lat"], out["point_lon"])
    _c, wd_okres = locate_courts(out["wd_lon"].to_numpy(float), out["wd_lat"].to_numpy(float),
                                 locator)
    wd = out["wd_lat"].notna()
    out["wd_in_own_okres"] = pd.array(
        np.where(wd & out["okres_id"].notna(), wd_okres == out["okres_id"].to_numpy(float), None),
        dtype="boolean")
    out["wd_overridden"] = ((out["wd_dist_m"] > WD_CONFLICT_M)
                            & (out["wd_in_own_okres"] == False)          # noqa: E712
                            & (out["in_own_okres"] == True)).fillna(False).astype(bool)  # noqa: E712
    keep_wd = wd & ~out["wd_overridden"]
    out["lat"] = np.where(keep_wd, out["wd_lat"], out["point_lat"])
    out["lon"] = np.where(keep_wd, out["wd_lon"], out["point_lon"])
    out["coord_source"] = np.where(keep_wd, "wikidata", out["match_source"])
    out.loc[out["lat"].isna(), "coord_source"] = None
    out["coord_ref"] = coord_refs(out, s, vanished_ids)
    cols = ["settlement_id", "cs_name", "status_1910", "land", "okres_id", "parent_1910",
            "match_status", "match_scope", "match_kind", "match_score", "match_why",
            "match_source", "place_id", "place_name", "obec_kod", "obec_name", "okres_kod",
            "okres_name", "obec_absorbed", "parent_changed", "point_lat", "point_lon",
            "point_court_id", "point_okres_id", "in_own_okres", "dist_own_okres_m",
            "court_1910_register", "in_register_court",
            "wd_lat", "wd_lon", "wd_dist_m", "wd_in_own_okres", "wd_overridden",
            "lat", "lon", "coord_source", "coord_ref"]
    out = out[cols]
    assert out["settlement_id"].is_unique and len(out) == len(settlements)
    return out


def crosswalk_report(xw: pd.DataFrame) -> pd.DataFrame:
    """Coverage before (Wikidata only) and after, by land × 1910 status; the checks in numbers."""
    def share(m):
        return f"{m.mean():6.1%}"

    g = xw.assign(land=xw["land"].fillna("(none)"), status=xw["status_1910"].fillna("(none)"))
    tab = g.groupby(["land", "status"]).agg(
        n=("settlement_id", "size"),
        wikidata=("wd_lat", lambda v: v.notna().mean()),
        after=("lat", lambda v: v.notna().mean())).reset_index()
    print("Crosswalk settlements_1910 -> RÚIAN/Geonames/PRNG")
    print(xw["match_status"].value_counts().to_string())
    print(xw.groupby(["match_scope", "match_kind"], dropna=False).size().to_string())
    print(xw["match_source"].value_counts().to_string())
    print("\nShare with a point, before (Wikidata) -> after:")
    print(tab.to_string(index=False, formatters={"wikidata": "{:.1%}".format,
                                                 "after": "{:.1%}".format}))
    m = xw[xw["match_status"] == "matched"]
    print("\nCourt-polygon test (an okres_1910 match lies inside by construction; the test bites "
          "on the wider scopes):")
    band = pd.cut(m["dist_own_okres_m"], [0, 1000, 5000, np.inf],
                  labels=["<=1 km", "1-5 km", ">5 km"]).astype(object)
    band[m["in_own_okres"].fillna(False).to_numpy(bool)] = "inside"
    band[m["in_own_okres"].isna().to_numpy()] = "no 1910 okres"
    print(pd.crosstab(m["match_scope"], band).to_string())
    rc = m["in_register_court"].dropna()
    if len(rc):
        print(f"Against the 1910 register's courts: {rc.mean():.1%} of {len(rc):,} matched points lie "
              f"in the court the register files them under; "
              f"{int((xw['match_why'] == 'court_1910').sum())} ties settled by that court")
    d = m["wd_dist_m"].dropna()
    if len(d):
        q = d.quantile([.5, .9, .99]).round(0).astype(int).tolist()
        print(f"Wikidata vs crosswalk point: n={len(d):,}, median {q[0]} m, p90 {q[1]} m, "
              f"p99 {q[2]} m, > 5 km: {(d > 5000).sum()}")
    print(f"Transfers: {int(m['obec_absorbed'].sum()):,} 1910 obce are now parts of another obec; "
          f"{int(m['parent_changed'].sum()):,} 1910 parts sit under a different modern obec")
    return tab

def build_current_geography(ruian_zip, geonames_gpkg, prng_xlsx, vanished_csv,
                            settlements: pd.DataFrame, districts: pd.DataFrame, locator,
                            out_dir, register_dir=None) -> pd.DataFrame:
    """Read RÚIAN + Geonames + PRNG + the vanished places, write every table and the crosswalk.

    Writes to *out_dir* (``data/final/geo``): ``ruian_obce`` / ``ruian_ku`` (GeoParquet),
    ``ruian_casti_obce``, ``ruian_zsj``, ``ruian_obce_adjacency``, ``geonames_places``,
    ``prng_places``, ``vanished_places`` and ``settlements_1910_geo`` (the crosswalk, keyed on
    ``settlement_id``), each with a CSV beside it. *vanished_csv* is the cached Wikidata result
    (:func:`vanished.fetch_vanished` queries only when it is missing). *register_dir* is
    ``data/final/register`` as notebook 15 last built it; where it exists, the register's courts
    break ties (:func:`register_courts`). Returns the crosswalk.
    """
    from pathlib import Path

    from utils.parsing import geonames as gn
    from utils.parsing import prng as pr
    from utils.parsing import vanished as va

    out_dir = Path(out_dir)
    tables = build_tables(read_vfr(ruian_zip))
    ruian_report(tables)
    write_tables(tables, out_dir)

    places = gn.read_geonames(geonames_gpkg)
    gn.geonames_report(places)
    prng = pr.add_court_1910(pr.read_prng(prng_xlsx), locator, pr.silesia_okres_ids(districts))
    pr.prng_report(prng)
    vanished = va.build_vanished(va.fetch_vanished(vanished_csv), locator, tables["ruian_obce"])
    va.vanished_report(vanished)
    for name, d in (("geonames_places", places), ("prng_places", prng),
                    ("vanished_places", vanished)):
        d.to_parquet(out_dir / f"{name}.parquet", index=False)
        d.to_csv(out_dir / f"{name}.csv", index=False, encoding="utf-8")

    n0 = len(settlements)
    reg_path = Path(register_dir) / "register_places.parquet" if register_dir else None
    courts = register_courts(pd.read_parquet(reg_path) if reg_path and reg_path.exists() else None)
    print(f"1910 register courts available for {len(courts):,} settlements")
    xw = build_crosswalk(settlements, tables, locator, places, prng, vanished, courts)
    assert len(xw) == n0 and xw["settlement_id"].is_unique and xw["settlement_id"].notna().all()
    crosswalk_report(xw)
    xw.to_parquet(out_dir / "settlements_1910_geo.parquet", index=False)
    xw.to_csv(out_dir / "settlements_1910_geo.csv", index=False, encoding="utf-8")
    return xw
