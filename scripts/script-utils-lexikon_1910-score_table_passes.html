How it works

The table arm of r5 (run_table_arm.py) asks how far a model reading of the seventeen printed columns beats the text layer. Channel B (the NDK ALTO text layer) and channel A (a vision model) read the same municipality-block crops, and every row each returned is in table_rows_&lt;label&gt;.csv (run_lexikon_1910_pilot.py). The gold is section 1 of the calibration and purposive sheets — 403 rows on 13 pages — read in-process by read_transcription_md.read_sheet. Certification sheets are never read here.

Matching

A gold row and a channel row are paired per page by a weighted alignment that keeps printed order: a pair scores the similarity of the normalised names, plus a bonus when the printed number agrees, and no pair under MIN_PAIR is made. A gold row with no partner is missed: the crop did not hold it, or the reader dropped it — the same for both channels when the crop is to blame. The mezi tím vojsko lines are left out on the channel side, because the gold sheets do not transcribe them.

What is compared

name: the words of the place name — runs of two or more letters, case-folded — so the bracketed place-type tokens, the footnote marks (*)), the dot leaders, the part numbers and a pictogram the text layer turned into a stray letter (Pankratz o) do not count against a reading. A misspelt word does. name_strict keeps everything but the brackets, marks and leaders, and is reported beside it. Number columns: the digits, with the printed nil . and an empty cell equal. post_key and station_note: the text with every space removed.

A cell counts when the gold or the reading has something in it, so a column that is blank on most rows does not score itself.

def norm_name(text: str) -> str:
    """The words of a name: runs of two or more letters, case-folded."""
    text = _BRACKET_RE.sub(" ", text or "")
    words = re.findall(r"[^\W\d_]{2,}", text)
    return " ".join(words).casefold()

def load_gold(roles=GOLD_ROLES) -> dict[str, list[dict]]:
    """``{page: [row, …]}`` in printed order, from the sheets themselves."""
    import read_transcription_md as rt
    from utils.lexikon_1910 import lexikon_1910_institutions as lxin

def load_rows(path: Path) -> dict[str, dict[str, list[dict]]]:
    """``{channel: {page: [row, …]}}`` in block order, *mezi tím* lines left out."""
    out: dict[str, dict[str, list[dict]]] = defaultdict(lambda: defaultdict(list))

        rows = list(csv.DictReader(fh))

def pair_score(gold: dict, row: dict) -> float:
    a, b = norm_name(gold["name"]), norm_name(row.get("name", ""))
    score = difflib.SequenceMatcher(None, a, b).ratio() if a and b else 0.0
    seq_g, seq_r = norm_number(gold["seq"]), norm_number(row.get("seq", ""))
    if seq_g and seq_g == seq_r:
        score += SEQ_BONUS
    return score

def align(gold: list[dict], rows: list[dict]) -> list[tuple[int, int]]:
    """Order-keeping pairs ``(gold index, row index)`` of greatest total score."""
    n, m = len(gold), len(rows)
    best = [[0.0] * (m + 1) for _ in range(n + 1)]
    move = [[""] * (m + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            best[i][j], move[i][j] = best[i - 1][j], "g"
            if best[i][j - 1] > best[i][j]:
                best[i][j], move[i][j] = best[i][j - 1], "r"
            s = pair_score(gold[i - 1], rows[j - 1])
            if s >= MIN_PAIR and best[i - 1][j - 1] + s > best[i][j]:
                best[i][j], move[i][j] = best[i - 1][j - 1] + s, "p"
    pairs, i, j = [], n, m
    while i and j:
        if move[i][j] == "p":
            pairs.append((i - 1, j - 1))
            i, j = i - 1, j - 1
        elif move[i][j] == "g":
            i -= 1
        else:
            j -= 1
    return pairs[::-1]

def readings(gold: dict, rows: dict[str, list[dict]]) -> dict[tuple, dict]:
    """``{(page, gold position index): row}`` — the row read for each gold row."""
    out = {}
    for page, gold_rows in gold.items():
        for gi, ri in align(gold_rows, rows.get(page, [])):
            out[(page, gi)] = rows[page][ri]
    return out

def score_reading(gold: dict, read: dict[tuple, dict], n_rows: int) -> dict:
    """Recall, row-exact and per-column accuracy of one channel of one pass."""
    n_gold = sum(len(v) for v in gold.values())
    right, counted = Counter(), Counter()
    exact = 0
    for (page, gi), row in read.items():
        g = gold[page][gi]
        whole = True
        for col in SCORED_COLS:
            truth, got = norm(col, g.get(col, "")), norm(col, row.get(col, ""))
            if not truth and not got:
                continue
            counted[col] += 1
            if truth == got:
                right[col] += 1
            else:
                whole = False
        exact += whole
        for col in EXTRA_COLS:
            truth = norm(col, g.get("name", ""))
            got = norm(col, row.get("name", ""))
            counted[col] += 1
            right[col] += truth == got
    out = {"gold_rows": n_gold, "matched": len(read), "extra": n_rows - len(read),
           "recall": len(read) / n_gold if n_gold else 0.0,
           "row_exact": exact / n_gold if n_gold else 0.0,
           "cells": sum(counted.values()),
           "cell_acc": sum(right.values()) / max(sum(counted.values()), 1),
           "number_acc": (sum(right[c] for c in NUMBER_COLS)
                          / max(sum(counted[c] for c in NUMBER_COLS), 1))}
    for col in SCORED_COLS + EXTRA_COLS:
        out[f"acc_{col}"] = right[col] / counted[col] if counted[col] else None
    return out

def cross(gold: dict, left: dict[tuple, dict], right: dict[tuple, dict]) -> Counter:
    """Cell outcomes where two readings both hold a gold row.

    ``agree_right``/``agree_wrong``: the two say the same; ``left_only`` /
    ``right_only``: one is right; ``both_wrong``: they differ and neither is.
    ``agree_wrong`` is the shared error a comparison of the two cannot see.
    """
    out = Counter()
    for key in left.keys() & right.keys():
        g = gold[key[0]][key[1]]
        for col in SCORED_COLS:
            truth = norm(col, g.get(col, ""))
            a, b = norm(col, left[key].get(col, "")), norm(col, right[key].get(col, ""))
            if not (truth or a or b):
                continue
            if a == b:
                out["agree_right" if a == truth else "agree_wrong"] += 1
            elif a == truth:
                out["left_only"] += 1
            elif b == truth:
                out["right_only"] += 1
            else:
                out["both_wrong"] += 1
    return out

def report(gold: dict, passes: list[tuple[str, dict]]) -> tuple[str, list[dict]]:
    lines = [f"table arm — {sum(len(v) for v in gold.values())} gold rows on "
             f"{len(gold)} pages ({', '.join(GOLD_ROLES)})", ""]
    table, read_by = [], {}
    seen_text_layer = False
    for label, channels in passes:
        for channel, rows in sorted(channels.items()):
            if channel == "B":
                # Every pass carries the same text layer; score it once.
                if seen_text_layer:
                    continue
                seen_text_layer = True
            reading = readings(gold, rows)
            n_rows = sum(len(rows.get(p, [])) for p in gold)
            s = score_reading(gold, reading, n_rows)
            tag = "OCR" if channel == "B" else "AI"
            read_by[(label, channel)] = reading
            table.append({"label": label, "channel": channel, "reader": tag, **s})
    head = (f"  {'pass':28s} {'ch':3s} {'recall':>7s} {'row ok':>7s} {'cells':>7s} "
            f"{'numbers':>7s} {'names':>7s} {'strict':>7s} {'seq':>7s} {'post':>7s} {'station':>7s}")
    lines += [head, "  " + "-" * (len(head) - 2)]
    for r in table:
        lines.append(f"  {r['label'][:28]:28s} {r['channel']:3s} {fmt(r['recall'])} "
                     f"{fmt(r['row_exact'])} {fmt(r['cell_acc'])} {fmt(r['number_acc'])} "
                     f"{fmt(r['acc_name'])} {fmt(r['acc_name_strict'])} {fmt(r['acc_seq'])} {fmt(r['acc_post_key'])} "
                     f"{fmt(r['acc_station_note'])}")
    lines += ["", "  per number column (share of counted cells right):",
              "  " + " ".join(f"{c[:9]:>9s}" for c in NUMBER_COLS)]
    for r in table:
        lines.append(f"  {r['label'][:20]:20s} {r['channel']} "
                     + " ".join(f"{fmt(r[f'acc_{c}']):>9s}" for c in NUMBER_COLS))
    pairs = []
    keys = list(read_by)
    for i, a in enumerate(keys):
        for b in keys[i + 1:]:
            pairs.append((a, b, cross(gold, read_by[a], read_by[b])))
    if pairs:
        lines += ["", "  two readings side by side (cells where either holds something):",
                  f"  {'left':24s} {'right':24s} {'agree ok':>9s} {'agree bad':>9s} "
                  f"{'left only':>9s} {'right only':>10s} {'both bad':>9s}"]
        for a, b, c in pairs:
            total = sum(c.values()) or 1
            lines.append(f"  {a[0][:20] + ' ' + a[1]:24s} {b[0][:20] + ' ' + b[1]:24s} "
                         + " ".join(f"{c[k] / total:9.1%}" for k in
                                    ("agree_right", "agree_wrong", "left_only",
                                     "right_only", "both_wrong")))
    return "\n".join(lines), table

def main(argv=None) -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):
            pass
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--rows", nargs="*", default=[],
                    help="table_rows_<label>.csv files; default every pass on disk")
    ap.add_argument("--out", default="", help="write the table here as CSV")
    args = ap.parse_args(argv)
    if args.rows:
        files = [Path(p) for p in args.rows]
    else:
        import bakeoff as bo
        files = pass_files([d for d in bo.BAKEOFF.iterdir() if d.is_dir()])
    if not files:
        print("no table_rows_*.csv found — run run_table_arm.py first")
        return 1
    gold = load_gold()
    passes = [(label_of(f), load_rows(f)) for f in files]
    text, table = report(gold, passes)
    print(text)
    if args.out:
        out = Path(args.out)
        out.parent.mkdir(parents=True, exist_ok=True)
        with out.open("w", encoding="utf-8", newline="") as fh:
            writer = csv.DictWriter(fh, fieldnames=list(table[0]))
            writer.writeheader()
            writer.writerows(table)
        out.with_suffix(".txt").write_text(text + "\n", encoding="utf-8")
        print(f"\n  -> {out}")
    return 0

"""Score the table readings against the gold tables, cell by cell.

The table arm of ``r5`` (``run_table_arm.py``) asks how far a model reading of
the seventeen printed columns beats the text layer. Channel B (the NDK ALTO
text layer) and channel A (a vision model) read the **same municipality-block
crops**, and every row each returned is in ``table_rows_<label>.csv``
(``run_lexikon_1910_pilot.py``). The gold is section 1 of the calibration and
purposive sheets — 403 rows on 13 pages — read in-process by
``read_transcription_md.read_sheet``. Certification sheets are never read here.

Matching
--------
A gold row and a channel row are paired per page by a weighted alignment that
keeps printed order: a pair scores the similarity of the normalised names, plus
a bonus when the printed number agrees, and no pair under ``MIN_PAIR`` is made.
A gold row with no partner is **missed**: the crop did not hold it, or the
reader dropped it — the same for both channels when the crop is to blame. The
*mezi tím vojsko* lines are left out on the channel side, because the gold
sheets do not transcribe them.

What is compared
----------------
``name``: the **words** of the place name — runs of two or more letters,
case-folded — so the bracketed place-type tokens, the footnote marks (``*)``),
the dot leaders, the part numbers and a pictogram the text layer turned into a
stray letter (``Pankratz o``) do not count against a reading. A misspelt word
does. ``name_strict`` keeps everything but the brackets, marks and leaders, and
is reported beside it.
Number columns: the digits, with the printed nil ``.`` and an empty cell equal.
``post_key`` and ``station_note``: the text with every space removed.

A cell counts when the gold **or** the reading has something in it, so a column
that is blank on most rows does not score itself.

    python utils/lexikon_1910/score_table_passes.py                 # every pass in the bake-off dirs
    python utils/lexikon_1910/score_table_passes.py --rows a.csv b.csv
"""

from __future__ import annotations

import argparse
import csv
import difflib
import re
import sys
from collections import Counter, defaultdict
from pathlib import Path

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]
sys.path.insert(0, str(ROOT))
sys.path.insert(0, str(HERE))

from utils.lexikon_1910 import lexikon_1910 as lx                    # noqa: E402

#: The roles the bake-off may look at; certification stays sealed (D-55).
GOLD_ROLES = ("calibration", "purposive")
#: A pair below this score is not made: the two rows are different places.
MIN_PAIR = 0.6
#: Added to a pair's name similarity when the printed numbers agree.
SEQ_BONUS = 0.3

NUMBER_COLS = [c for c in lx.INT_COLS]
TEXT_COLS = ["post_key", "station_note"]
SCORED_COLS = ["seq", "name", *NUMBER_COLS, *TEXT_COLS]
#: Reported, never part of a row's verdict.
EXTRA_COLS = ["name_strict"]

_BRACKET_RE = re.compile(r"\[[^\]]*\]")
_FOOTNOTE_RE = re.compile(r"\*+\)?")
_LEADER_RE = re.compile(r"\.{2,}|…")
_NOT_NAME_RE = re.compile(r"[^\w\s,.\-']")
_NIL = {"", ".", "·", "—", "–", "-"}


def norm_name(text: str) -> str:
    """The words of a name: runs of two or more letters, case-folded."""
    text = _BRACKET_RE.sub(" ", text or "")
    words = re.findall(r"[^\W\d_]{2,}", text)
    return " ".join(words).casefold()


def norm_name_strict(text: str) -> str:
    text = _BRACKET_RE.sub(" ", text or "")
    text = _FOOTNOTE_RE.sub(" ", text)
    text = _LEADER_RE.sub(" ", text)
    text = _NOT_NAME_RE.sub(" ", text).replace("_", " ")
    text = re.sub(r"\s+", " ", text).strip(" ,.:;")
    return text.casefold()


def norm_number(text: str) -> str:
    text = re.sub(r"\s+", "", text or "")
    return "" if text in _NIL else text


def norm_text(text: str) -> str:
    return re.sub(r"\s+", "", text or "")


def norm(col: str, value: str) -> str:
    if col == "name":
        return norm_name(value)
    if col == "name_strict":
        return norm_name_strict(value)
    if col in TEXT_COLS:
        return norm_text(value)
    return norm_number(value)

# Loading

def load_gold(roles=GOLD_ROLES) -> dict[str, list[dict]]:
    """``{page: [row, …]}`` in printed order, from the sheets themselves."""
    import read_transcription_md as rt
    from utils.lexikon_1910 import lexikon_1910_institutions as lxin
    from utils.lexikon_1910 import lexikon_1910_vision as lxv
    from check_transcription import OFF_ATLAS, TOPOGRAPHIC

    abbreviations = lxin.load_abbreviations(rt.ABBREVIATIONS, lang="cs")
    allowed = lxv.vocabulary(abbreviations) | {OFF_ATLAS} | TOPOGRAPHIC
    gold: dict[str, list[dict]] = defaultdict(list)
    for role in roles:
        for path in sorted((rt.GOLD_ROOT / role).glob("[0-9][0-9][0-9][0-9].md")):
            gold[path.stem] += rt.read_sheet(path, allowed)["table"]
    return dict(gold)


def load_rows(path: Path) -> dict[str, dict[str, list[dict]]]:
    """``{channel: {page: [row, …]}}`` in block order, *mezi tím* lines left out."""
    out: dict[str, dict[str, list[dict]]] = defaultdict(lambda: defaultdict(list))
    with path.open(encoding="utf-8", newline="") as fh:
        rows = list(csv.DictReader(fh))
    rows.sort(key=lambda r: (r["channel"], r["page"], int(r["block_index"]), int(r["row"])))
    for r in rows:
        if lx.OF_WHICH_RE.search(r.get("name", "")):
            continue
        out[r["channel"]][r["page"]].append(r)
    return out


def label_of(path: Path) -> str:
    return path.stem.removeprefix("table_rows_") or "unlabelled"


# --------------------------------------------------------------------------- #
# Matching
# --------------------------------------------------------------------------- #

def pair_score(gold: dict, row: dict) -> float:
    a, b = norm_name(gold["name"]), norm_name(row.get("name", ""))
    score = difflib.SequenceMatcher(None, a, b).ratio() if a and b else 0.0
    seq_g, seq_r = norm_number(gold["seq"]), norm_number(row.get("seq", ""))
    if seq_g and seq_g == seq_r:
        score += SEQ_BONUS
    return score


def align(gold: list[dict], rows: list[dict]) -> list[tuple[int, int]]:
    """Order-keeping pairs ``(gold index, row index)`` of greatest total score."""
    n, m = len(gold), len(rows)
    best = [[0.0] * (m + 1) for _ in range(n + 1)]
    move = [[""] * (m + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            best[i][j], move[i][j] = best[i - 1][j], "g"
            if best[i][j - 1] > best[i][j]:
                best[i][j], move[i][j] = best[i][j - 1], "r"
            s = pair_score(gold[i - 1], rows[j - 1])
            if s >= MIN_PAIR and best[i - 1][j - 1] + s > best[i][j]:
                best[i][j], move[i][j] = best[i - 1][j - 1] + s, "p"
    pairs, i, j = [], n, m
    while i and j:
        if move[i][j] == "p":
            pairs.append((i - 1, j - 1))
            i, j = i - 1, j - 1
        elif move[i][j] == "g":
            i -= 1
        else:
            j -= 1
    return pairs[::-1]


def readings(gold: dict, rows: dict[str, list[dict]]) -> dict[tuple, dict]:
    """``{(page, gold position index): row}`` — the row read for each gold row."""
    out = {}
    for page, gold_rows in gold.items():
        for gi, ri in align(gold_rows, rows.get(page, [])):
            out[(page, gi)] = rows[page][ri]
    return out


# --------------------------------------------------------------------------- #
# Scoring
# --------------------------------------------------------------------------- #

def score_reading(gold: dict, read: dict[tuple, dict], n_rows: int) -> dict:
    """Recall, row-exact and per-column accuracy of one channel of one pass."""
    n_gold = sum(len(v) for v in gold.values())
    right, counted = Counter(), Counter()
    exact = 0
    for (page, gi), row in read.items():
        g = gold[page][gi]
        whole = True
        for col in SCORED_COLS:
            truth, got = norm(col, g.get(col, "")), norm(col, row.get(col, ""))
            if not truth and not got:
                continue
            counted[col] += 1
            if truth == got:
                right[col] += 1
            else:
                whole = False
        exact += whole
        for col in EXTRA_COLS:
            truth = norm(col, g.get("name", ""))
            got = norm(col, row.get("name", ""))
            counted[col] += 1
            right[col] += truth == got
    out = {"gold_rows": n_gold, "matched": len(read), "extra": n_rows - len(read),
           "recall": len(read) / n_gold if n_gold else 0.0,
           "row_exact": exact / n_gold if n_gold else 0.0,
           "cells": sum(counted.values()),
           "cell_acc": sum(right.values()) / max(sum(counted.values()), 1),
           "number_acc": (sum(right[c] for c in NUMBER_COLS)
                          / max(sum(counted[c] for c in NUMBER_COLS), 1))}
    for col in SCORED_COLS + EXTRA_COLS:
        out[f"acc_{col}"] = right[col] / counted[col] if counted[col] else None
    return out


def cross(gold: dict, left: dict[tuple, dict], right: dict[tuple, dict]) -> Counter:
    """Cell outcomes where two readings both hold a gold row.

    ``agree_right``/``agree_wrong``: the two say the same; ``left_only`` /
    ``right_only``: one is right; ``both_wrong``: they differ and neither is.
    ``agree_wrong`` is the shared error a comparison of the two cannot see.
    """
    out = Counter()
    for key in left.keys() & right.keys():
        g = gold[key[0]][key[1]]
        for col in SCORED_COLS:
            truth = norm(col, g.get(col, ""))
            a, b = norm(col, left[key].get(col, "")), norm(col, right[key].get(col, ""))
            if not (truth or a or b):
                continue
            if a == b:
                out["agree_right" if a == truth else "agree_wrong"] += 1
            elif a == truth:
                out["left_only"] += 1
            elif b == truth:
                out["right_only"] += 1
            else:
                out["both_wrong"] += 1
    return out


def fmt(x) -> str:
    return "  -- " if x is None else f"{x:6.1%}"


def report(gold: dict, passes: list[tuple[str, dict]]) -> tuple[str, list[dict]]:
    lines = [f"table arm — {sum(len(v) for v in gold.values())} gold rows on "
             f"{len(gold)} pages ({', '.join(GOLD_ROLES)})", ""]
    table, read_by = [], {}
    seen_text_layer = False
    for label, channels in passes:
        for channel, rows in sorted(channels.items()):
            if channel == "B":
                # Every pass carries the same text layer; score it once.
                if seen_text_layer:
                    continue
                seen_text_layer = True
            reading = readings(gold, rows)
            n_rows = sum(len(rows.get(p, [])) for p in gold)
            s = score_reading(gold, reading, n_rows)
            tag = "OCR" if channel == "B" else "AI"
            read_by[(label, channel)] = reading
            table.append({"label": label, "channel": channel, "reader": tag, **s})
    head = (f"  {'pass':28s} {'ch':3s} {'recall':>7s} {'row ok':>7s} {'cells':>7s} "
            f"{'numbers':>7s} {'names':>7s} {'strict':>7s} {'seq':>7s} {'post':>7s} {'station':>7s}")
    lines += [head, "  " + "-" * (len(head) - 2)]
    for r in table:
        lines.append(f"  {r['label'][:28]:28s} {r['channel']:3s} {fmt(r['recall'])} "
                     f"{fmt(r['row_exact'])} {fmt(r['cell_acc'])} {fmt(r['number_acc'])} "
                     f"{fmt(r['acc_name'])} {fmt(r['acc_name_strict'])} {fmt(r['acc_seq'])} {fmt(r['acc_post_key'])} "
                     f"{fmt(r['acc_station_note'])}")
    lines += ["", "  per number column (share of counted cells right):",
              "  " + " ".join(f"{c[:9]:>9s}" for c in NUMBER_COLS)]
    for r in table:
        lines.append(f"  {r['label'][:20]:20s} {r['channel']} "
                     + " ".join(f"{fmt(r[f'acc_{c}']):>9s}" for c in NUMBER_COLS))
    pairs = []
    keys = list(read_by)
    for i, a in enumerate(keys):
        for b in keys[i + 1:]:
            pairs.append((a, b, cross(gold, read_by[a], read_by[b])))
    if pairs:
        lines += ["", "  two readings side by side (cells where either holds something):",
                  f"  {'left':24s} {'right':24s} {'agree ok':>9s} {'agree bad':>9s} "
                  f"{'left only':>9s} {'right only':>10s} {'both bad':>9s}"]
        for a, b, c in pairs:
            total = sum(c.values()) or 1
            lines.append(f"  {a[0][:20] + ' ' + a[1]:24s} {b[0][:20] + ' ' + b[1]:24s} "
                         + " ".join(f"{c[k] / total:9.1%}" for k in
                                    ("agree_right", "agree_wrong", "left_only",
                                     "right_only", "both_wrong")))
    return "\n".join(lines), table


def pass_files(dirs: list[Path]) -> list[Path]:
    return sorted(p for d in dirs if d.exists() for p in d.glob("table_rows_*.csv"))


def main(argv=None) -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):
            pass
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--rows", nargs="*", default=[],
                    help="table_rows_<label>.csv files; default every pass on disk")
    ap.add_argument("--out", default="", help="write the table here as CSV")
    args = ap.parse_args(argv)
    if args.rows:
        files = [Path(p) for p in args.rows]
    else:
        import bakeoff as bo
        files = pass_files([d for d in bo.BAKEOFF.iterdir() if d.is_dir()])
    if not files:
        print("no table_rows_*.csv found — run run_table_arm.py first")
        return 1
    gold = load_gold()
    passes = [(label_of(f), load_rows(f)) for f in files]
    text, table = report(gold, passes)
    print(text)
    if args.out:
        out = Path(args.out)
        out.parent.mkdir(parents=True, exist_ok=True)
        with out.open("w", encoding="utf-8", newline="") as fh:
            writer = csv.DictWriter(fh, fieldnames=list(table[0]))
            writer.writeheader()
            writer.writerows(table)
        out.with_suffix(".txt").write_text(text + "\n", encoding="utf-8")
        print(f"\n  -> {out}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
