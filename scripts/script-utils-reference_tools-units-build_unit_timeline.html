How it works

Fidler, Jiří (2025). Encyklopedie československých legií. Praha: Universum.

Why this exists

unit_authority.csv says that a unit existed, in which theatre and from when. It does not say where the unit stood in a given month, and that is what a design with time-varying exposure needs: the shocks of the Russian theatre (November 1917, Brest- Litovsk, May 1918, the withdrawal from the front in 1919) hit every unit at once, so the only variation with a comparison group is which unit was at the front, in a garrison or on the rails when. The encyclopedia writes exactly that, in one formula per entry, down to the battalion:

Nacházelo se v -> Borispolu, od prosince 1917 na přesunu a v -> Kajevce, od března
1918 ..., podléhalo veliteli -> X, od října 1917 veliteli -> Y. Funkci velitele
pluku zastávali plk. A (srpen 1917 – září 1917), ...

Each formula becomes rows of one long table:

kind = location     state ∈ {place, moving, field, moving_home}; value = the place,
                    in the grammatical case the source uses ("Borispolu")
kind = parent       value = the formation the unit answered to; state says which of
                    two chains it is when the source gives two (professional, general)
kind = commander    value = rank and name, with the months of the tenure; state = the
                    office when it is not the commander's (chief_of_staff, …)
kind = lifecycle    state ∈ {formed, renamed, merged, disbanded, sailed,
                    arrived_home}; value = the unit it came from or became
kind = composition  (transports) the units that sailed
kind = strength     (transports) how many men

Dates are as precise as the source: a lifecycle date is a day, a phase of the location and subordination chains opens in a month ("od prosince 1917"), and the first phase of a chain opens when the unit was formed (date_from_precision = formation). Several states inside one phase share its opening month and keep the order the source gives them in seq; the source does not date the changes inside a phase.

What it does NOT carry: detachments the source puts in brackets ("části oddílu v X"), and anything the entry says outside its formulas.

The source's slips are kept and flagged, not corrected. A location chain that runs backwards in time ("od prosince 1919 v Čeljabinsku, od ledna 1919 na přesunu" — the year of one of the two is a misprint) keeps the dates as printed; the rows outside the longest run that is in order carry date_suspect = 1. A sub-entry heading printed twice under one headword gets "#2" on its second occurrence.

def sentences(text: str) -> list[str]:
    """Split an entry into sentences, keeping 'čs.', ranks and ordinals intact."""
    out, last = [], 0
    for m in _STOP.finditer(text):
        if _NO_END.search(text[last:m.start()]):
            continue
        out.append(text[last:m.start()].strip())
        last = m.end()
    out.append(text[last:].strip(" ."))
    return [s for s in out if s]

def split_phases(chain: str) -> list[tuple[str, str]]:
    """``'v A, od března 1918 v B a od června v poli'`` ->
    ``[('', 'v A'), ('1918-03', 'v B'), ('1918-06', 'v poli')]``.

    A phase written without its year takes the year of the phase before it, or the
    next one if the month runs backwards."""
    out, last, start, year = [], 0, "", ""
    for m in _PHASE.finditer(chain):
        body = chain[last:m.start()].strip(" ,")
        if body or out:
            out.append((start, body))
        if m.group(2):
            year = m.group(2)
        elif year and start and MONTH[m.group(1)] < int(start[5:7]):
            year = str(int(year) + 1)
        start, last = (_ym(m.group(1), year) if year else ""), m.end()
    out.append((start, chain[last:].strip(" ,")))
    return [(d, b) for d, b in out if b]

def parse_states(body: str) -> list[tuple[str, str]]:
    """One phase of a location chain -> ordered ``(state, place)`` pairs."""
    out = []
    for part in _STATE_SPLIT.split(_BRACKET.sub("", body)):
        p = _strip_ref(part)
        if not p:
            continue
        if p.startswith("na přesunu do vlasti"):
            out.append(("moving_home", ""))
        elif p.startswith("na přesunu"):
            out.append(("moving", ""))
            rest = re.match(r"na přesunu\s+(?:a\s+)?(?:ve?|na|u)\s+(.+)$", p)
            if rest:
                out.append(("place", _strip_ref(rest.group(1))))
        elif p.startswith("v poli"):
            out.append(("field", ""))
        else:
            out.append(("place", _strip_ref(re.sub(r"^(?:ve?|na|u|poblíž)\s+", "", p))))
    return out

def parse_period(s: str) -> tuple[str, str]:
    """``'srpen 1917 – září 1917'`` -> ``('1917-08', '1917-09')``; open ends are ''."""
    s = s.strip()
    ms = re.findall(r"(" + _M + r")\s+(\d{4})", s)
    if not ms:
        return "", ""
    first, last = _ym(*ms[0]), _ym(*ms[-1])
    if len(ms) == 1 and s.startswith("od "):
        return first, ""
    if len(ms) == 1 and s.startswith("do "):
        return "", first
    return first, last

def parse_commanders(sentence: str) -> list[tuple[str, str, str, str]]:
    """'Funkci velitele pluku zastávali A (…), B (…), funkci náčelníka štábu zastával
    C (…)' -> ``(office, name, from, to)``; the commander's own office is ''."""
    parts = re.split(r"[Ff]unkci\s+(.+?)\s+zastával[iay]?\s+", sentence)
    out = []
    for office, body in zip(parts[1::2], parts[2::2]):
        role = next((v for k, v in _OFFICE.items() if office.startswith(k)), office)
        for name, period in _PERIODS.findall(body):
            name = _strip_ref(re.sub(r"^\s*(?:,|a\b|postupně)\s*", "", name.strip()))
            for one in re.split(r",\s*(?=(?:od |do )?(?:" + _M + r"))", period):
                start, end = parse_period(one)
                if name and (start or end):
                    out.append((role, name, start, end))
    return out

def parse_parents(sub: str) -> list[tuple[str, str, str]]:
    """The subordination chain -> ``(scope, month, formation)``.

    A unit that answered to one superior in its own branch and to another in everything
    else carries two chains ('ve věcech odborných …, ve věcech ostatních …'); ``scope``
    tells them apart and is '' when there is only one."""
    pieces = _SUBORD.split(sub)              # ['', scope, chain, scope, chain, …]
    out = []
    for scope, chain in zip(pieces[1::2], pieces[2::2]):
        for start, body in split_phases(_BRACKET.sub("", chain)):
            parent = _strip_ref(_SUPERIOR.sub("", _strip_ref(body)))
            if parent:
                out.append((_SCOPE.get(scope or "", ""), start, parent))
    return out

def parse_lifecycle(sentence: str) -> tuple[str, str, str] | None:
    """A sentence with a full date and a lifecycle verb -> ``(state, date, target)``."""
    day = _DAY.search(sentence)
    if not day:
        return None
    for state, pattern in _LIFECYCLE:
        verb = re.search(pattern, sentence)
        if verb:
            tail = sentence[max(day.end(), verb.end()):]
            target = re.search(r"\b(?:z|ze|na|do|v)\s+(.+)$", tail)
            return state, _iso(day), _strip_ref(target.group(1))[:160] if target else ""
    return None

def parse_transport(text: str) -> list[dict]:
    """A transport entry: the units that travelled and their number."""
    m = _COMPOSITION.search(text)
    if not m:
        return []
    day = _DAY.search(text)
    common = dict(state="", date_from=_iso(day) if day else "",
                  date_from_precision="day" if day else "unknown", date_to="")
    return [dict(kind="composition", value=_strip_ref(m.group(1)), **common),
            dict(kind="strength", value=re.sub(r"\D", "", m.group(2)), **common)]

def parse_entry(text: str) -> list[dict]:
    """All timeline rows of one entry or sub-entry (without the unit columns)."""
    rows: list[dict] = []
    formed = ""
    for s in sentences(text):
        loc = re.search(r"Nacház\w+ se (?:se )?(.*)$", s)
        if loc:
            cut = _SUBORD.search(loc.group(1))
            chain = loc.group(1)[:cut.start()] if cut else loc.group(1)
            for start, body in split_phases(chain):
                for state, place in parse_states(body):
                    rows.append(_phase_row("location", state, place, start, formed))
            if cut:
                for scope, start, parent in parse_parents(loc.group(1)[cut.start():]):
                    rows.append(_phase_row("parent", scope, parent, start, formed))
            continue
        if re.search(r"[Ff]unkci\b.*zastával", s):
            for office, name, start, end in parse_commanders(s):
                rows.append(dict(kind="commander", state=office, value=name,
                                 date_from=start,
                                 date_from_precision="month" if start else "unknown",
                                 date_to=end))
            continue
        life = parse_lifecycle(s)
        if life:
            state, date, target = life
            if state == "formed" and not formed:
                formed = date
            rows.append(dict(kind="lifecycle", state=state, value=target, date_from=date,
                             date_from_precision="day", date_to=""))
    if text.startswith("Dopravní jednotka"):
        rows += parse_transport(text)
    return rows

def out_of_order(months: list[str]) -> list[bool]:
    """True for the items outside the longest non-decreasing run of *months*."""
    n = len(months)
    best, prev = [1] * n, [-1] * n
    for i in range(n):
        for j in range(i):
            if months[j] <= months[i] and best[j] + 1 > best[i]:
                best[i], prev[i] = best[j] + 1, j
    keep, i = set(), max(range(n), key=best.__getitem__) if n else -1
    while i >= 0:
        keep.add(i)
        i = prev[i]
    return [k not in keep for k in range(n)]

def flag_suspect(rows: list[dict]) -> None:
    """Mark the location rows of one entry whose month breaks the chain's order."""
    chain = [r for r in rows if r["kind"] == "location" and r["date_from"]]
    for r, bad in zip(chain, out_of_order([r["date_from"][:7] for r in chain])):
        r["date_suspect"] = int(bad)

def build(epub: Path) -> pd.DataFrame:
    entries = read_entries(epub)
    auth = pd.read_csv(AUTHORITY, encoding="utf-8")
    units = set(auth["unit_cs"])
    rows, seen = [], {}
    for head, sub, text in entries.itertuples(index=False):
        if head not in units:
            continue
        sub = sub or ""
        seen[head, sub] = seen.get((head, sub), 0) + 1
        if seen[head, sub] > 1:
            sub = f"{sub} #{seen[head, sub]}"
        entry = [dict(date_suspect=0, **r) for r in parse_entry(text)]
        flag_suspect(entry)
        for i, r in enumerate(entry, start=1):
            rows.append(dict(unit_cs=head, subunit=sub, seq=i, **r))
    out = pd.DataFrame(rows, columns=COLUMNS)
    n = len(out)
    out = out.merge(auth[["unit_cs", "theatre", "number"]], on="unit_cs",
                    how="left", validate="m:1")
    assert len(out) == n and out["theatre"].notna().all()
    out["source"] = "Fidler 2025, entry " + out["unit_cs"]
    return out

build_unit_timeline.py — generate reference/units/unit_timeline.csv
===================================================================
Where every Czechoslovak Legion unit was, whom it answered to and who commanded it,
month by month, read from the unit entries of

    Fidler, Jiří (2025). *Encyklopedie československých legií.* Praha: Universum.

Why this exists
---------------
``unit_authority.csv`` says that a unit existed, in which theatre and from when. It
does not say where the unit stood in a given month, and that is what a design with
time-varying exposure needs: the shocks of the Russian theatre (November 1917, Brest-
Litovsk, May 1918, the withdrawal from the front in 1919) hit every unit at once, so
the only variation with a comparison group is *which unit was at the front, in a
garrison or on the rails when*. The encyclopedia writes exactly that, in one formula
per entry, down to the battalion:

    Nacházelo se v -> Borispolu, od prosince 1917 na přesunu a v -> Kajevce, od března
    1918 ..., podléhalo veliteli -> X, od října 1917 veliteli -> Y. Funkci velitele
    pluku zastávali plk. A (srpen 1917 – září 1917), ...

Each formula becomes rows of one long table:

    kind = location     state ∈ {place, moving, field, moving_home}; value = the place,
                        in the grammatical case the source uses ("Borispolu")
    kind = parent       value = the formation the unit answered to; state says which of
                        two chains it is when the source gives two (professional, general)
    kind = commander    value = rank and name, with the months of the tenure; state = the
                        office when it is not the commander's (chief_of_staff, …)
    kind = lifecycle    state ∈ {formed, renamed, merged, disbanded, sailed,
                        arrived_home}; value = the unit it came from or became
    kind = composition  (transports) the units that sailed
    kind = strength     (transports) how many men

and subordination chains opens in a month ("od prosince 1917"), and the first phase of
a chain opens when the unit was formed (``date_from_precision = formation``). Several
states inside one phase share its opening month and keep the order the source gives
them in ``seq``; the source does not date the changes inside a phase.

What it does NOT carry: detachments the source puts in brackets ("části oddílu v X"),
and anything the entry says outside its formulas.

The source's slips are kept and flagged, not corrected. A location chain that runs
backwards in time ("od prosince 1919 v Čeljabinsku, od ledna 1919 na přesunu" — the
year of one of the two is a misprint) keeps the dates as printed; the rows outside the
longest run that is in order carry ``date_suspect = 1``. A sub-entry heading printed
twice under one headword gets "#2" on its second occurrence.

--------------

gitignored ``library/``. This script emits facts only (unit, date, place, formation,
name) and none of the encyclopedia's prose.

           "date_from_precision", "date_to", "value", "date_suspect"]

# Month names: nominative ("srpen 1917") and genitive ("od srpna 1917", "15. srpna 1917").
_NOM = ["leden", "únor", "březen", "duben", "květen", "červen", "červenec", "srpen",
        "září", "říjen", "listopad", "prosinec"]
_GEN = ["ledna", "února", "března", "dubna", "května", "června", "července", "srpna",
        "září", "října", "listopadu", "prosince"]
MONTH = {m: i for i, m in enumerate(_NOM, start=1)} | {m: i for i, m in enumerate(_GEN, start=1)}
_M = "|".join(sorted(MONTH, key=len, reverse=True))
_UPPER = "A-ZÁČĎÉĚÍŇÓŘŠŤÚŮÝŽ"

# Abbreviations after which a full stop does not end a sentence.
_ABBR = ("čs|Čs|plk|pplk|mjr|kpt|pkpt|škpt|por|ppor|npor|prap|gen|genmjr|genpor|brig|"
         "div|rtm|šrtm|des|rtn|čet|svob|sv|tzv|např|resp|tj|Dr|Ing|JUDr|MUDr|MVDr|PhDr|"
         "RNDr|PhMr|MgPh|MUC|JUC|doc|prof|ryt|hr|sl|st|ml|č|r|s|v|[" + _UPPER + "]")
_STOP = re.compile(r"\.\s+(?=[" + _UPPER + r"(])|(?<=\])\.(?=[" + _UPPER + "])")  # '[I].Součástí'
_NO_END = re.compile(r"(?:\b(?:" + _ABBR + r")|(?<!\d)\d{1,2})$")   # 'čs.', 'plk.', '15.'

_PHASE = re.compile(r"(?:,\s*|\s+a\s+|^)od\s+(" + _M + r")(?:\s+(\d{4}))?\s+")
_DAY = re.compile(r"(\d{1,2})\.\s*(" + _M + r")\s+(\d{4})")
_SUBORD = re.compile(r"(?:,|\s+a)?\s*(?:ve věcech (\w+)\s+)?podléh\w+\s+")
_BRACKET = re.compile(r"\([^()]*\)?")
_PERIODS = re.compile(r"([^()]+?)\s*\(([^()]*\d{4}[^()]*)\)")
_STATE_SPLIT = re.compile(r",\s*|\s+a\s+(?=(?:na přesunu|v poli|ve? |na |u |poblíž ))")
_SUPERIOR = re.compile(r"^(?:přímo\s+)?(?:veliteli|velitelství|náčelníkovi|přednostovi|"
                       r"předsedovi)\s+")
_COMPOSITION = re.compile(r"Tvořil[aoy]? jej\s+(.+?),\s*celkem\s+([\d\s ]+?)\s+"
                          r"(?:legionář|osob|mužů|dobrovolník)")

_SCOPE = {"odborných": "professional", "ostatních": "general"}
_OFFICE = {"velitele": "", "náčelníka štábu": "chief_of_staff",
           "veřejného žalobce": "prosecutor", "předsedy": "president"}
_LIFECYCLE = [  # first match wins
    ("arrived_home", r"dorazil[oayi]?\b.*\bna čs\. území"),
    ("sailed", r"vyplul[oayi]?\b"),
    ("renamed", r"přejmenov[áa]n[oay]?\b|přeměněn[oay]?\b|změněn[oay]?\b"),
    ("merged", r"slouč\w+|začleněn[oay]?\b|včleněn[oay]?\b|vplynul[oay]?\b"),
    ("disbanded", r"zrušen[oay]?\b|zanikl[oay]?\b|rozpuštěn[oay]?\b|likvidov\w+"),
    ("formed", r"vznikl[oay]?\b|zřízen[oay]?\b|ustaven[oay]?\b|vytvořen[oay]?\b|"
               r"zformov\w+|postaven[oay]?\b|sestaven[oay]?\b"),
]


def sentences(text: str) -> list[str]:
    """Split an entry into sentences, keeping 'čs.', ranks and ordinals intact."""
    out, last = [], 0
    for m in _STOP.finditer(text):
        if _NO_END.search(text[last:m.start()]):
            continue
        out.append(text[last:m.start()].strip())
        last = m.end()
    out.append(text[last:].strip(" ."))
    return [s for s in out if s]


def _ym(month: str, year: str) -> str:
    return f"{year}-{MONTH[month]:02d}"


def _iso(day: re.Match) -> str:
    return f"{day.group(3)}-{MONTH[day.group(2)]:02d}-{int(day.group(1)):02d}"


def _strip_ref(s: str) -> str:
    return re.sub(r"\s+", " ", s.replace("->", " ")).strip(" ,.;")


def split_phases(chain: str) -> list[tuple[str, str]]:
    """``'v A, od března 1918 v B a od června v poli'`` ->
    ``[('', 'v A'), ('1918-03', 'v B'), ('1918-06', 'v poli')]``.

    A phase written without its year takes the year of the phase before it, or the
    next one if the month runs backwards."""
    out, last, start, year = [], 0, "", ""
    for m in _PHASE.finditer(chain):
        body = chain[last:m.start()].strip(" ,")
        if body or out:
            out.append((start, body))
        if m.group(2):
            year = m.group(2)
        elif year and start and MONTH[m.group(1)] < int(start[5:7]):
            year = str(int(year) + 1)
        start, last = (_ym(m.group(1), year) if year else ""), m.end()
    out.append((start, chain[last:].strip(" ,")))
    return [(d, b) for d, b in out if b]


def parse_states(body: str) -> list[tuple[str, str]]:
    """One phase of a location chain -> ordered ``(state, place)`` pairs."""
    out = []
    for part in _STATE_SPLIT.split(_BRACKET.sub("", body)):
        p = _strip_ref(part)
        if not p:
            continue
        if p.startswith("na přesunu do vlasti"):
            out.append(("moving_home", ""))
        elif p.startswith("na přesunu"):
            out.append(("moving", ""))
            rest = re.match(r"na přesunu\s+(?:a\s+)?(?:ve?|na|u)\s+(.+)$", p)
            if rest:
                out.append(("place", _strip_ref(rest.group(1))))
        elif p.startswith("v poli"):
            out.append(("field", ""))
        else:
            out.append(("place", _strip_ref(re.sub(r"^(?:ve?|na|u|poblíž)\s+", "", p))))
    return out


def parse_period(s: str) -> tuple[str, str]:
    """``'srpen 1917 – září 1917'`` -> ``('1917-08', '1917-09')``; open ends are ''."""
    s = s.strip()
    ms = re.findall(r"(" + _M + r")\s+(\d{4})", s)
    if not ms:
        return "", ""
    first, last = _ym(*ms[0]), _ym(*ms[-1])
    if len(ms) == 1 and s.startswith("od "):
        return first, ""
    if len(ms) == 1 and s.startswith("do "):
        return "", first
    return first, last


def parse_commanders(sentence: str) -> list[tuple[str, str, str, str]]:
    """'Funkci velitele pluku zastávali A (…), B (…), funkci náčelníka štábu zastával
    C (…)' -> ``(office, name, from, to)``; the commander's own office is ''."""
    parts = re.split(r"[Ff]unkci\s+(.+?)\s+zastával[iay]?\s+", sentence)
    out = []
    for office, body in zip(parts[1::2], parts[2::2]):
        role = next((v for k, v in _OFFICE.items() if office.startswith(k)), office)
        for name, period in _PERIODS.findall(body):
            name = _strip_ref(re.sub(r"^\s*(?:,|a\b|postupně)\s*", "", name.strip()))
            for one in re.split(r",\s*(?=(?:od |do )?(?:" + _M + r"))", period):
                start, end = parse_period(one)
                if name and (start or end):
                    out.append((role, name, start, end))
    return out


def parse_parents(sub: str) -> list[tuple[str, str, str]]:
    """The subordination chain -> ``(scope, month, formation)``.

    A unit that answered to one superior in its own branch and to another in everything
    else carries two chains ('ve věcech odborných …, ve věcech ostatních …'); ``scope``
    tells them apart and is '' when there is only one."""
    pieces = _SUBORD.split(sub)              # ['', scope, chain, scope, chain, …]
    out = []
    for scope, chain in zip(pieces[1::2], pieces[2::2]):
        for start, body in split_phases(_BRACKET.sub("", chain)):
            parent = _strip_ref(_SUPERIOR.sub("", _strip_ref(body)))
            if parent:
                out.append((_SCOPE.get(scope or "", ""), start, parent))
    return out


def parse_lifecycle(sentence: str) -> tuple[str, str, str] | None:
    """A sentence with a full date and a lifecycle verb -> ``(state, date, target)``."""
    day = _DAY.search(sentence)
    if not day:
        return None
    for state, pattern in _LIFECYCLE:
        verb = re.search(pattern, sentence)
        if verb:
            tail = sentence[max(day.end(), verb.end()):]
            target = re.search(r"\b(?:z|ze|na|do|v)\s+(.+)$", tail)
            return state, _iso(day), _strip_ref(target.group(1))[:160] if target else ""
    return None


def parse_transport(text: str) -> list[dict]:
    """A transport entry: the units that travelled and their number."""
    m = _COMPOSITION.search(text)
    if not m:
        return []
    day = _DAY.search(text)
    common = dict(state="", date_from=_iso(day) if day else "",
                  date_from_precision="day" if day else "unknown", date_to="")
    return [dict(kind="composition", value=_strip_ref(m.group(1)), **common),
            dict(kind="strength", value=re.sub(r"\D", "", m.group(2)), **common)]


def _phase_row(kind: str, state: str, value: str, start: str, formed: str) -> dict:
    if start:
        return dict(kind=kind, state=state, value=value, date_from=start,
                    date_from_precision="month", date_to="")
    return dict(kind=kind, state=state, value=value, date_from=formed,
                date_from_precision="formation" if formed else "unknown", date_to="")


def parse_entry(text: str) -> list[dict]:
    """All timeline rows of one entry or sub-entry (without the unit columns)."""
    rows: list[dict] = []
    formed = ""
    for s in sentences(text):
        loc = re.search(r"Nacház\w+ se (?:se )?(.*)$", s)
        if loc:
            cut = _SUBORD.search(loc.group(1))
            chain = loc.group(1)[:cut.start()] if cut else loc.group(1)
            for start, body in split_phases(chain):
                for state, place in parse_states(body):
                    rows.append(_phase_row("location", state, place, start, formed))
            if cut:
                for scope, start, parent in parse_parents(loc.group(1)[cut.start():]):
                    rows.append(_phase_row("parent", scope, parent, start, formed))
            continue
        if re.search(r"[Ff]unkci\b.*zastával", s):
            for office, name, start, end in parse_commanders(s):
                rows.append(dict(kind="commander", state=office, value=name,
                                 date_from=start,
                                 date_from_precision="month" if start else "unknown",
                                 date_to=end))
            continue
        life = parse_lifecycle(s)
        if life:
            state, date, target = life
            if state == "formed" and not formed:
                formed = date
            rows.append(dict(kind="lifecycle", state=state, value=target, date_from=date,
                             date_from_precision="day", date_to=""))
    if text.startswith("Dopravní jednotka"):
        rows += parse_transport(text)
    return rows


def out_of_order(months: list[str]) -> list[bool]:
    """True for the items outside the longest non-decreasing run of *months*."""
    n = len(months)
    best, prev = [1] * n, [-1] * n
    for i in range(n):
        for j in range(i):
            if months[j] <= months[i] and best[j] + 1 > best[i]:
                best[i], prev[i] = best[j] + 1, j
    keep, i = set(), max(range(n), key=best.__getitem__) if n else -1
    while i >= 0:
        keep.add(i)
        i = prev[i]
    return [k not in keep for k in range(n)]


def flag_suspect(rows: list[dict]) -> None:
    """Mark the location rows of one entry whose month breaks the chain's order."""
    chain = [r for r in rows if r["kind"] == "location" and r["date_from"]]
    for r, bad in zip(chain, out_of_order([r["date_from"][:7] for r in chain])):
        r["date_suspect"] = int(bad)


def build(epub: Path) -> pd.DataFrame:
    entries = read_entries(epub)
    auth = pd.read_csv(AUTHORITY, encoding="utf-8")
    units = set(auth["unit_cs"])
    rows, seen = [], {}
    for head, sub, text in entries.itertuples(index=False):
        if head not in units:
            continue
        sub = sub or ""
        seen[head, sub] = seen.get((head, sub), 0) + 1
        if seen[head, sub] > 1:
            sub = f"{sub} #{seen[head, sub]}"
        entry = [dict(date_suspect=0, **r) for r in parse_entry(text)]
        flag_suspect(entry)
        for i, r in enumerate(entry, start=1):
            rows.append(dict(unit_cs=head, subunit=sub, seq=i, **r))
    out = pd.DataFrame(rows, columns=COLUMNS)
    n = len(out)
    out = out.merge(auth[["unit_cs", "theatre", "number"]], on="unit_cs",
                    how="left", validate="m:1")
    assert len(out) == n and out["theatre"].notna().all()
    out["source"] = "Fidler 2025, entry " + out["unit_cs"]
    return out


def main() -> None:
    epub = Path(sys.argv[1]) if len(sys.argv) > 1 else DEFAULT_EPUB
    if not epub.exists():
        sys.exit(f"epub not found: {epub}")
    tab = build(epub)
    tab.to_csv(OUT, index=False, encoding="utf-8")
    with_sub = (tab["subunit"] != "").groupby(tab["unit_cs"]).any().sum()
    print(f"wrote {OUT.relative_to(ROOT)}  ({len(tab):,} rows, "
          f"{tab['unit_cs'].nunique()} units, {with_sub} with sub-units)")
    print(tab.groupby(["kind", "state"]).size().to_string())


if __name__ == "__main__":
    main()
