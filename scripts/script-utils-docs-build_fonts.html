How it works

Libertinus (SIL OFL 1.1) is not a system font on most machines, so the page carries its own copy. One-off: run this only to refresh the subsets; build_docs.py reads the .woff files. Source OTFs come from a MiKTeX/TeX Live libertinus-fonts install (or pass a directory).

python utils/docs/build_fonts.py [path/to/libertinus-fonts]

"""Subset the Libertinus faces the docs site embeds -> utils/docs/fonts/*.woff.

Libertinus (SIL OFL 1.1) is not a system font on most machines, so the page carries its own
copy. One-off: run this only to refresh the subsets; build_docs.py reads the .woff files.
Source OTFs come from a MiKTeX/TeX Live ``libertinus-fonts`` install (or pass a directory).

    python utils/docs/build_fonts.py [path/to/libertinus-fonts]
"""
from __future__ import annotations

__version__ = "1.0"

import sys
from pathlib import Path

from fontTools import subset

OUT = Path(__file__).parent / "fonts"
DEFAULT_SRC = Path.home() / "AppData/Local/Programs/MiKTeX/fonts/opentype/public/libertinus-fonts"
FACES = ["LibertinusSans-Regular", "LibertinusSans-Italic", "LibertinusSans-Bold",
         "LibertinusSerif-Bold"]
# Basic Latin to Latin Extended-B (Czech, Slovak, Polish, Hungarian, German), Cyrillic,
# general punctuation, currency, letterlike, arrows, math operators.
UNICODES = ("U+0020-007E,U+00A0-024F,U+0300-036F,U+0400-04FF,U+2000-206F,U+20A0-20CF,"
            "U+2100-214F,U+2190-21FF,U+2200-22FF,U+25A0-25FF")


def main(src: Path) -> None:
    OUT.mkdir(exist_ok=True)
    for face in FACES:
        opts = subset.Options(flavor="woff", layout_features=["*"], notdef_outline=True)
        font = subset.load_font(str(src / f"{face}.otf"), opts)
        sub = subset.Subsetter(opts)
        sub.populate(unicodes=subset.parse_unicodes(UNICODES))
        sub.subset(font)
        subset.save_font(font, str(OUT / f"{face}.woff"), opts)
        print(f"{face}: {(OUT / f'{face}.woff').stat().st_size // 1024} KB")


if __name__ == "__main__":
    main(Path(sys.argv[1]) if len(sys.argv) > 1 else DEFAULT_SRC)
