How it works

locate_glyphs.candidates_on_image drops a component before clustering when it looks like a rule (tall and thin), a full stop or comma (small, low), or a bracket. On the Czech test-A clicks none of the 141 clicked glyphs was lost; on the German clicks 58 of 203 fall inside no candidate (Opus review, 2026-10-31). For each such click this finds the component(s) under it and says which filter took them, with their size in line heights, and counts them by filter and token. It also lists the line height each section image was measured at (a wrong one shifts every threshold written in line heights).

Read-only: the section images of the German gold run and the clicks. Outside utils/lexikon_1910 so as not to change the route's code digests.

def filter_of(box, lines, line, lh: float, labels, k: int) -> str:
    """The first rule of ``candidates_on_image`` that removes component ``k`` (mirrors its order)."""
    import locate_glyphs as lg

    x0, y0, x1, y1 = box
    w, h = x1 - x0, y1 - y0
    baseline = lines.baselines[line]
    if h > 0.8 * lh and h > lg.RULE_ASPECT * max(1, w):
        return "rule (tall and thin)"
    small = (h <= lg.PUNCT_SIZE[0] * lh and w <= lg.PUNCT_SIZE[1] * lh and w <= lg.PUNCT_ASPECT * h
             and (y0 + y1) / 2 >= baseline - lg.PUNCT_LIFT * lh
             and (h <= lg.PUNCT_STOP * lh or y1 >= baseline + lg.PUNCT_DROP * lh))
    if small:
        return "full stop / comma"
    import numpy as np

    patch = np.isin(labels[y0:y1, x0:x1], [k])
    kind, _side = lg.bracket_kind(patch, box, baseline, lh)
    if kind:
        return f"bracket ({kind})"
    return "kept as a cluster component"

def main() -> int:
    import json

    import numpy as np
    from PIL import Image

    import locate_glyphs as lg
    from build_glyph_bank import line_height
    from german_click_recall import CLICKS

    clicks = defaultdict(list)
    for path in sorted(CLICKS.glob("*.json")):
        doc = json.loads(path.read_text(encoding="utf-8"))
        for c in doc["clicks"]:
            clicks[doc["image"]].append((c["sx"], c["sy"], c["token"]))
    by_filter, by_token, dims, lost = Counter(), Counter(), defaultdict(list), 0
    near_edge = []                  # distance of each lost click to the nearest candidate box, in line heights
    for name, pts in sorted(clicks.items()):
        image = Image.open(SECTIONS / name)
        lh = line_height(image.convert("L")) or 46
        lines = lg.line_components(image, lh)
        cands = lg.candidates_on_image(image, name, lh)
        print(f"{name}: measured line height {lh}px, {len(cands)} candidates")
        if lines is None:
            continue
        line_of = {k: j for j, comps in lines.components.items() for k in comps}
        for x, y, token in pts:
            if any(lg.inside(c.box, x, y, 0.1 * lh) for c in cands):
                continue
            lost += 1
            gap = min((max(c.box[0] - x, 0, x - c.box[2]) ** 2 + max(c.box[1] - y, 0, y - c.box[3]) ** 2) ** 0.5
                      for c in cands) / lh
            near_edge.append(gap)
            r = int(0.35 * lh)
            near = [k for k in set(np.unique(lines.labels[max(0, y - r):y + r, max(0, x - r):x + r])) - {0}
                    if k in lines.boxes]
            if not near:
                by_filter["no component under the click"] += 1
                continue
            k = max(near, key=lambda k: lines.areas[k])
            box = lines.boxes[k]
            why = filter_of(box, lines, line_of[k], lh, lines.labels, k) if k in line_of else "on no printed line"
            by_filter[why] += 1
            by_token[(why, token)] += 1
            dims[why].append(((box[2] - box[0]) / lh, (box[3] - box[1]) / lh))
    print(f"\n{lost} clicks inside no candidate, by the filter that removed the component under them:")
    for why, n in by_filter.most_common():
        w = np.median([a for a, _ in dims[why]]) if dims[why] else float("nan")
        h = np.median([b for _, b in dims[why]]) if dims[why] else float("nan")
        print(f"  {n:3d}  {why:32s} median size {w:.2f} x {h:.2f} line heights")
    buckets = [("within 0.25 lh of a candidate (an imprecise click, or a gap inside the glyph)", 0.25),
               ("0.25-0.5 lh", 0.5), ("0.5-1 lh", 1.0), ("more than 1 lh from any candidate", 1e9)]
    prev = 0.0
    print("distance of the lost clicks to the nearest candidate box:")
    for label, top in buckets:
        print(f"  {sum(prev < g <= top or (prev == 0 and g == 0) for g in near_edge):3d}  {label}")
        prev = top
    print("by filter and token:", by_token.most_common(14))
    print(f"constants: RULE_ASPECT {lg.RULE_ASPECT}, PUNCT_SIZE {lg.PUNCT_SIZE}, PUNCT_ASPECT {lg.PUNCT_ASPECT}, "
          f"JOIN_GAP {lg.JOIN_GAP}, MAX_GLYPH_WIDTH {lg.MAX_GLYPH_WIDTH}")
    return 0
