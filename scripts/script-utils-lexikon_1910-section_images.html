How it works

E3 (the locator and DINOv3) and the Opus call cut their marks from section images: one per section and column, the page's own pixels enlarged ×2, named &lt;page&gt;_&lt;court&gt;_&lt;kind&gt;_&lt;order&gt;.png. For the gold those were cut once by hand (make_gold_worksheet.py); for every other page they did not exist. The pilot now saves them as it cuts (--notes-save-sections DIR), from the very segments the readers read, so the check sees what the readers saw and nothing is re-cut.

Each save also appends to DIR/index.csv: page, court, kind, order, region, whether the segment was carried over from the previous column or page, and its box on the page — what run_e3_pages builds its sections from.

def index_from_names(directory: Path, pages: set | None = None) -> list[dict]:
    """An index for images that follow the naming but were not saved by :func:`save` —
    the gold section images, which is how this path is rehearsed on the gold."""
    import re

    pattern = re.compile(r"^(?P<page>\d{4})_(?P<district>.+)_(?P<kind>parts|offices)_(?P<order>\d+)\.png$")
    rows = []
    for p in sorted(Path(directory).glob("*.png")):
        m = pattern.match(p.name)
        if m and (pages is None or m["page"] in pages):

                         "x0": "", "y0": "", "x1": "", "y1": ""})
    return rows

"""The readers' section crops, saved as images for every page (D-112, step 3 beyond the gold).

E3 (the locator and DINOv3) and the Opus call cut their marks from **section
images**: one per section and column, the page's own pixels enlarged ×2, named
``<page>_<court>_<kind>_<order>.png``. For the gold those were cut once by hand
(``make_gold_worksheet.py``); for every other page they did not exist. The pilot
now saves them as it cuts (``--notes-save-sections DIR``), from the very segments
the readers read, so the check sees what the readers saw and nothing is re-cut.

Each save also appends to ``DIR/index.csv``: page, court, kind, order, region,
whether the segment was carried over from the previous column or page, and its
box on the page — what :mod:`run_e3_pages` builds its sections from.
"""

from __future__ import annotations

__version__ = "1.1.0"

import csv
from pathlib import Path

#: The gold section images are enlarged ×2 (``make_gold_worksheet.ENLARGE``); the
#: locator's sizes and DINOv3's bank were measured at that scale, so these match it.
ENLARGE = 2
INDEX = "index.csv"
FIELDS = ["image", "page", "district", "kind", "order", "region", "carried",
          "x0", "y0", "x1", "y1", "enlarge"]


def name_for(page: str, district_key: str, kind: str, order: int) -> str:
    """``0019_as_parts_0.png`` — the gold's own naming (``make_gold_worksheet``)."""
    return f"{int(page):04d}_{district_key or 'unknown'}_{kind}_{order}.png"


def save(image, segments, out_dir: Path, enlarge: float = ENLARGE) -> list[dict]:
    """Save each segment's crop, enlarged, and append it to the index. Returns the rows.

    ``image`` is the page as the pilot rendered it (ALTO pixels, the segments'
    coordinates); ``segments`` are :class:`lexikon_1910_notes.Segment`. A segment
    of another kind than ``parts``/``offices`` (the legend) is skipped. A German
    scan, rendered at its own resolution, passes ``enlarge = 2 / scale`` so its print
    comes out the size of the Czech print at ×2.
    """
    out_dir.mkdir(parents=True, exist_ok=True)
    rows = []
    for s in segments:
        if s.kind not in ("parts", "offices"):
            continue
        name = name_for(s.page, s.key, s.kind, s.order)
        crop = image.crop(s.box)
        crop.resize((round(crop.width * enlarge), round(crop.height * enlarge))).save(out_dir / name)
        rows.append({"image": name, "page": f"{int(s.page):04d}", "district": s.key,
                     "kind": s.kind, "order": s.order, "region": s.region,
                     "carried": bool(s.carried), "x0": s.box[0], "y0": s.box[1],
                     "x1": s.box[2], "y1": s.box[3], "enlarge": round(enlarge, 4)})
    _append(out_dir / INDEX, rows)
    return rows


def _append(path: Path, rows: list[dict]) -> None:
    """Add rows, replacing any earlier row for the same image (a re-read page)."""
    old = read_index(path.parent) if path.exists() else []
    names = {r["image"] for r in rows}
    keep = [r for r in old if r["image"] not in names] + rows
    with path.open("w", encoding="utf-8", newline="") as fh:
        w = csv.DictWriter(fh, fieldnames=FIELDS)
        w.writeheader()
        w.writerows(sorted(keep, key=lambda r: (r["page"], int(r["order"]))))


def read_index(out_dir: Path) -> list[dict]:
    path = Path(out_dir) / INDEX
    if not path.exists():
        return []
    with path.open(encoding="utf-8", newline="") as fh:
        return list(csv.DictReader(fh))
