def wilson(k: int, n: int, z: float = 1.96) -> tuple[float, float]:
    """Wilson score interval for k of n (the bound the checker rounds quote, VALIDATION.md)."""
    if n == 0:
        return float("nan"), float("nan")
    p = k / n
    den = 1 + z * z / n
    mid = (p + z * z / (2 * n)) / den
    half = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / den
    return max(0.0, mid - half), min(1.0, mid + half)

def fill(text: str, facts: dict[str, str], blocks: dict[str, Callable[[], str]], label: str,
         failed: list[str] | None = None) -> str:
    """``{{token}}`` and ``<!-- live: name -->`` filled, as ``build_docs.settlement_sheet`` does.

    A token or block the page names and nothing computes is an error (a typo must not ship as a
    blank); a block whose source is not there yet prints a sentence saying so."""
    failed = [] if failed is None else failed

    def _live(m: re.Match) -> str:
        name = m.group(1)
        if name not in blocks:
            raise KeyError(f"{label} names <!-- live: {name} -->, which site_pages does not build")
        try:
            return "\n\n" + blocks[name]() + "\n\n"
        except (FileNotFoundError, KeyError, ValueError, TypeError, ZeroDivisionError, ImportError, OSError,
                AttributeError, IndexError) as e:
            failed.append(f"table {name}: {type(e).__name__}: {e}")
            return "\n\n*This table is read from the build at build time and could not be built here.*\n\n"

    def _tok(m: re.Match) -> str:
        key = m.group(1)
        if key in facts:
            return facts[key]
        raise KeyError(f"{label} names {{{{{key}}}}}, which site_pages does not compute")

    text = re.sub(r"<!--\s*live:\s*([a-z0-9_]+)\s*-->", _live, text)
    text = re.sub(r"\{\{\s*([a-z0-9_]+)\s*\}\}", _tok, text)
    for f in failed:
        print(f"  [!] {label}: {f}")
    return text

def decision_rows(text: str) -> tuple[list[dict], list[str]]:
    """``DECISIONS.md``'s rows as {id, date, decision, status}; and the ids whose row has no Status cell.

    A ``|`` inside a code span is not a cell break; a row with extra breaks in its prose still has its
    Status second from the end."""
    rows, unread = [], []
    for line in text.splitlines():
        if not re.match(r"^\|\s*D-\d+\s*\|", line):
            continue
        masked = re.sub(r"`[^`]*`", lambda m: m.group(0).replace("|", "\x00"), line.strip())
        cells = [c.strip().replace("\x00", "|") for c in _PIPE.split(masked)[1:-1]]

            unread.append(cells[0])
            continue
        rows.append({"id": cells[0], "date": cells[1], "decision": cells[2], "status": cells[-2]})
    return rows, unread

def status_excerpt(status: str, limit: int = STATUS_MAX) -> str | None:
    """A long Status cut to its first sentence plus the clause the open match sits in (plain text);
    None when the Status is short enough to quote whole."""
    if len(status) <= limit:
        return None
    t = _plain(status)
    head = _summary(t, 220)
    m = OPEN_STATUS.search(t)
    if not m or m.end() <= len(head.rstrip(" …")):
        return head
    starts = [t.rfind(sep, 0, m.start()) for sep in ("; ", ". ", " — ")]
    s = max(starts)
    s = 0 if s < 0 else s + 2
    ends = [i for i in (t.find(sep, m.end()) for sep in ("; ", ". ")) if i != -1]
    clause = t[s:(min(ends) + 1 if ends else len(t))].strip(" ;")
    if len(clause) > 260:
        lo = max(0, m.start() - s - 120)
        clause = ("… " if lo else "") + clause[lo:lo + 260].rsplit(" ", 1)[0] + " …"
    return f"{head.removesuffix(' …')} … {clause}"

def refusal_desertion(final: Path = FIN, theatre: str = "Russia") -> dict:
    """Desertion among legionnaires whose Czech-lands birth was linked to a 1910 settlement against those
    who stated an okres and a municipality and got no link, in one theatre.

    The no-link group is of two kinds, kept apart as well as together: ``unmatched`` (no 1910 settlement
    of the written name was found) and every other ``birth_link_refusal`` (declined by a rule: an
    ambiguity, an okres conflict, a guard, a post-1949 Prague number …; not always with a candidate in hand). If either were tied
    to desertion, the linked sample would be a selected one. Rates, n, and each difference from the linked
    group with its 95 % (Wald) interval."""
    cols = ["theater", "exit_D", "birth_settlement_id", "birth_link_refusal", "birth_okres_id", "birth_municipality"]
    L = pd.read_parquet(final / "persons" / "legionnaires_clean.parquet", columns=cols)
    L = L[L["theater"] == theatre]
    d = L["exit_D"].astype("boolean").fillna(False).astype(bool)
    linked = L["birth_settlement_id"].notna()
    nolink = (L["birth_link_refusal"].notna() & ~linked & L["birth_okres_id"].notna()
              & L["birth_municipality"].notna())
    unmatched = nolink & (L["birth_link_refusal"] == "unmatched")
    groups = {"refused": nolink, "unmatched": unmatched, "declined": nolink & ~unmatched}
    p1, n1 = d[linked].mean(), int(linked.sum())
    out = {"theatre": theatre, "p_linked": p1, "n_linked": n1,
           "written": _when(final / "persons" / "legionnaires_clean.parquet")}
    for tag, g in groups.items():
        p2, n2 = d[g].mean(), int(g.sum())
        se = math.sqrt(p1 * (1 - p1) / n1 + p2 * (1 - p2) / n2)
        out.update({f"p_{tag}": p2, f"n_{tag}": n2, f"diff_{tag}": p2 - p1,
                    f"lo_{tag}": p2 - p1 - 1.96 * se, f"hi_{tag}": p2 - p1 + 1.96 * se})
    out.update(diff=out["diff_refused"], lo=out["lo_refused"], hi=out["hi_refused"])
    return out

def wilson(k: int, n: int, z: float = 1.96) -> tuple[float, float]:
    """Wilson score interval for k of n (the bound the checker rounds quote, VALIDATION.md)."""
    if n == 0:
        return float("nan"), float("nan")
    p = k / n
    den = 1 + z * z / n
    mid = (p + z * z / (2 * n)) / den
    half = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / den
    return max(0.0, mid - half), min(1.0, mid + half)


def _md_table(head: list[str], rows: list[list[str]], right_from: int = 99) -> str:
    align = ["---" if i < right_from else "---:" for i in range(len(head))]
    out = ["| " + " | ".join(head) + " |", "| " + " | ".join(align) + " |"]
    out += ["| " + " | ".join(c.replace("|", "\\|") for c in r) + " |" for r in rows]
    return "\n".join(out)


def _html_table(head: list[str], rows: list[list[str]], cls: str = "tbl") -> str:
    """Cells are HTML already (escape before passing)."""
    th = "".join(f"<th>{h}</th>" for h in head)
    body = "".join("<tr>" + "".join(f"<td>{c}</td>" for c in r) + "</tr>" for r in rows)
    return f'<div class="tblwrap"><table class="{cls}"><thead><tr>{th}</tr></thead><tbody>{body}</tbody></table></div>'


def fill(text: str, facts: dict[str, str], blocks: dict[str, Callable[[], str]], label: str,
         failed: list[str] | None = None) -> str:
    """``{{token}}`` and ``<!-- live: name -->`` filled, as ``build_docs.settlement_sheet`` does.

    A token or block the page names and nothing computes is an error (a typo must not ship as a
    blank); a block whose source is not there yet prints a sentence saying so."""
    failed = [] if failed is None else failed

    def _live(m: re.Match) -> str:
        name = m.group(1)
        if name not in blocks:
            raise KeyError(f"{label} names <!-- live: {name} -->, which site_pages does not build")
        try:
            return "\n\n" + blocks[name]() + "\n\n"
        except (FileNotFoundError, KeyError, ValueError, TypeError, ZeroDivisionError, ImportError, OSError,
                AttributeError, IndexError) as e:
            failed.append(f"table {name}: {type(e).__name__}: {e}")
            return "\n\n*This table is read from the build at build time and could not be built here.*\n\n"

    def _tok(m: re.Match) -> str:
        key = m.group(1)
        if key in facts:
            return facts[key]
        raise KeyError(f"{label} names {{{{{key}}}}}, which site_pages does not compute")

    text = re.sub(r"<!--\s*live:\s*([a-z0-9_]+)\s*-->", _live, text)
    text = re.sub(r"\{\{\s*([a-z0-9_]+)\s*\}\}", _tok, text)
    for f in failed:
        print(f"  [!] {label}: {f}")
    return text

def _inventory() -> list[dict]:
    """Every table in data/final (a parquet, or a CSV without a parquet twin), plus the panels
    notebook 04 has not written yet."""
    out = []
    for folder, *_ in FOLDERS:
        d = FIN / folder if folder else FIN
        if not d.is_dir():
            continue
        stems: dict[str, list[Path]] = {}
        for p in sorted(d.iterdir()):
            if p.is_file() and p.suffix in (".parquet", ".csv") and p.name != "register_columns.csv":
                stems.setdefault(p.stem, []).append(p)
        if folder == "":
            stems = {k: v for k, v in stems.items() if k == "data_dictionary"}
        for stem, files in stems.items():
            out.append({"folder": folder, "stem": stem, "files": files, "built": True})
    if not (PANELS / "spells.parquet").is_file():
        for stem in expected_panels():
            out.append({"folder": "panels", "stem": stem, "files": [], "built": False})
    return out


def _rows(files: list[Path]) -> tuple[int | None, list[str]]:
    pqf = [f for f in files if f.suffix == ".parquet"]
    if pqf:
        pf = pq.ParquetFile(pqf[0])
        return pf.metadata.num_rows, pf.schema_arrow.names
    df = pd.read_csv(files[0], dtype=str, keep_default_na=False, encoding="utf-8-sig")
    return len(df), list(df.columns)


def _key_holds(files: list[Path], key: list[str], columns: list[str]) -> str:
    if not key:
        return "none declared"
    missing = [k for k in key if k not in columns]
    if missing:
        return f"columns {missing} absent"
    pqf = [f for f in files if f.suffix == ".parquet"]
    df = (pd.read_parquet(pqf[0], columns=key) if pqf
          else pd.read_csv(files[0], usecols=key, dtype=str, keep_default_na=False, encoding="utf-8-sig")
          .replace("", pd.NA))
    na, dup = int(df.isna().any(axis=1).sum()), int(df.duplicated().sum())
    return "holds" if not (na or dup) else f"does not hold ({na:,} missing, {dup:,} duplicated)"


#: A line that writes a file: what makes a file-name needle a writer and not a reader (WRITERS).
_WRITE_CALL = re.compile(r"\.to_parquet\(|\.to_csv\(|\bwrite_\w*\(|\.write_text\(")


class _Notebooks:
    """The notebooks' code cells, each with the section it sits in (its last markdown heading or the
    cell's own '# §…' comment)."""

    def __init__(self) -> None:
        self.cells: dict[str, list[tuple[int, str, str]]] = {}
        for nb in sorted(NOTEBOOKS.glob("*.ipynb")):
            try:
                d = json.loads(nb.read_text(encoding="utf-8"))
            except (OSError, ValueError):
                continue
            sec, cells = "", []
            for i, c in enumerate(d.get("cells", [])):
                src = "".join(c.get("source", []))
                if c.get("cell_type") == "markdown":
                    m = re.search(r"^#+\s*(.+)$", src, re.M)
                    sec = m.group(1).strip() if m else sec
                elif c.get("cell_type") == "code":
                    own = re.match(r"\s*#\s*(§[^\n]*|\d+[a-z]?\.\s[^\n]*)", src)
                    cells.append((i, (own.group(1) if own else sec).strip(" -#"), src))
            self.cells[nb.stem] = cells

    def find(self, notebook: str, needle: str, write_only: bool = False) -> list[str]:
        """The cells whose source contains *needle*. With *write_only*, a needle that names a file
        (``….parquet"`` / ``….csv"``) counts only on a line that writes (``.to_parquet(`` …), so a cell
        that reads the table back is not listed as a writer."""
        names_file = write_only and bool(re.search(r"\.(?:parquet|csv)\b", needle))
        out = []
        for i, sec, src in self.cells.get(notebook, []):
            if names_file:
                hit = any(needle in ln and _WRITE_CALL.search(ln) for ln in src.splitlines())
            else:
                hit = needle in src
            if hit:
                sec = re.sub(r"`", "", sec)
                out.append(f"notebook {notebook.split('_')[0]}, cell {i}" + (f" ({sec[:70]})" if sec else ""))
        return out

    def readers(self, pattern: re.Pattern) -> list[str]:
        return [f"notebooks/{nb}.ipynb" for nb, cells in self.cells.items()
                if any(pattern.search(src) for _, _, src in cells)]


def _module_texts() -> dict[str, str]:
    out = {}
    for p in sorted((ROOT / "utils").rglob("*.py")):
        if "_archive" in p.parts or "docs" in p.parts or "__pycache__" in p.parts:
            continue
        with contextlib.suppress(OSError, UnicodeDecodeError):
            out[_rel(p)] = p.read_text(encoding="utf-8")
    return out


def _writer(folder: str, stem: str, nbs: _Notebooks) -> list[str]:
    path = f"{folder}/{stem}" if folder else stem
    specs = WRITERS.get(path)
    if specs is None:
        for pat, s in WRITERS.items():
            if pat.endswith("*") and path.startswith(pat[:-1]):
                specs = s
                break
    found = [hit for nb, needle in (specs or []) for hit in nbs.find(nb, needle, write_only=True)]
    return list(dict.fromkeys(found))


def _checks_for(folder: str, stem: str, checks: list[Check]) -> list[Check]:
    pats = [re.compile(rf"(?<![\w/]){re.escape(stem)}(?![\w])", re.I)]
    if stem in CHECK_ALIASES:
        pats.append(re.compile(CHECK_ALIASES[stem], re.I))
    if folder in REGISTER_FOLDER_LABEL:
        prefix = r"^register \(rechecked\)" if folder == "register_rechecked" else r"^register(?! \(rechecked\))"
        what = rf"\b{re.escape(stem)}\b" + (f"|{REGISTER_ALIASES[stem]}" if stem in REGISTER_ALIASES else "")
        pats = [re.compile(REGISTER_FOLDER_LABEL[folder], re.I), re.compile(rf"{prefix}.*(?:{what})", re.I)]
    elif folder == "panels" and stem.startswith(("spells_", "spell_periods_")):
        pats.append(re.compile(r"^spell tables", re.I))
    elif folder == "theatre":
        pats.append(re.compile(r"^theatre tables", re.I))
    return [c for c in checks if any(p.search(c[0]) for p in pats)]


def _readers(folder: str, stem: str, nbs: _Notebooks, mods: dict[str, str]) -> list[str]:
    s = re.escape(stem)
    pat = re.compile(rf"""["'/]{s}(?:\.parquet|\.csv)?["']|\b{s}\.(?:parquet|csv)\b""")
    hits = nbs.readers(pat) + [m for m, t in mods.items() if pat.search(t)]
    if folder == "panels":                         # a panel's name is a common word: 'units', 'spells'
        hits = [h for h in hits if "panels" in h or "04_panels" in h or "data_checks" in h]
    return hits

def _table_card(t: dict, keys: dict, nbs: _Notebooks, mods: dict, dd: pd.DataFrame | None,
                checks: list[Check]) -> str:
    folder, stem = t["folder"], t["stem"]
    key = keys.get(stem, [])
    rows: list[list[str]] = []
    if t["built"]:
        n, cols = _rows(t["files"])
        files = ", ".join(f"<code>{_esc(_rel(f).removeprefix('data/final/'))}</code>" for f in t["files"])
        rows.append(["Files", files + f" · written {_when(t['files'][0])}"])
        rows.append(["Rows × columns", f"{n:,} × {len(cols):,}"])
        holds = _key_holds(t["files"], key, cols)
    else:
        cols = []
        rows.append(["Files", f"<i>not built</i>: notebook 04 writes <code>panels/{_esc(stem)}.parquet</code> "
                              "and <code>.csv</code>"])
        holds = "checked once built"
    badge = ("pass" if holds == "holds" else "warn" if holds in ("none declared", "checked once built") else "fail")
    rows.insert(1, ["Key", (f"<code>{_esc(' × '.join(key))}</code> " if key else "")
                    + f'<span class="badge {badge}">{_esc(holds)}</span>'])
    g = grain(stem)
    rows.insert(2, ["Grain", _esc(("one row per " + g) if g else "unknown")])
    w = _writer(folder, stem, nbs)
    rows.append(["Written by", "<br>".join(_esc(x) for x in w) if w else "<i>unknown</i>"])
    cs = _checks_for(folder, stem, checks)
    if cs:
        ok = sum(1 for c in cs if c[1])
        items = "".join(f"<li>{'✓' if c[1] else ('✗' if c[3] else '!')} {_esc(c[0])}</li>" for c in cs)
        rows.append(["QA gate", f"<details><summary>{ok} of {len(cs)} passing</summary><ul>{items}</ul></details>"])
    else:
        rows.append(["QA gate", "<i>no check names it</i>"])
    rd = _readers(folder, stem, nbs, mods)
    rows.append(["Read by", ", ".join(f"<code>{_esc(r)}</code>" for r in rd) if rd else "<i>no reader found</i>"])
    rows.append(["Columns", _columns_cell(folder, stem, cols, dd)])
    body = "".join(f"<tr><th>{k}</th><td>{v}</td></tr>" for k, v in rows)
    return (f'<h3 id="tables-{_esc(folder or "root")}-{_esc(stem)}">{_esc(stem)}</h3>'
            f'<div class="tblwrap"><table class="tbl"><tbody>{body}</tbody></table></div>')


def grain(stem: str) -> str | None:
    """The table's grain in words, without the leading 'one' (the card says 'one row per …'). A spell
    variant (``spells_<v>``, ``spell_periods_<v>``, notebook 04 §1) has its base table's grain."""
    g = GRAIN.get(stem)
    if g is None:
        base = next((b for b in ("spell_periods", "spells") if stem.startswith(b + "_")), None)
        if base is None:
            return None
        g = f"{GRAIN[base]}, variant {stem[len(base) + 1:]}"
    return re.sub(r"^one\s+", "", g)


def _dictionary_names(final_names) -> tuple[set[str], tuple[str, ...]]:
    """The dictionary's ``final_name`` cells as exact names and prefixes: a row may name several
    columns (``lat / lon / coord_source``) or a family (``pol_*``)."""
    exact, prefixes = set(), []
    for cell in final_names:
        for part in (p.strip() for p in str(cell).split(" / ")):
            if part.endswith("*"):
                prefixes.append(part[:-1])
            elif part:
                exact.add(part)
    return exact, tuple(prefixes)


def _columns_cell(folder: str, stem: str, cols: list[str], dd: pd.DataFrame | None) -> str:
    if folder.startswith("register"):
        return 'described on <a href="#register-data">1910 register — the municipal data</a> (and the folder\'s <code>register_columns.csv</code>)'
    if folder == "census":
        return 'described in the <a href="#census-codebook">Census codebook</a>'
    if folder == "theatre":
        return 'described on <a href="#theatre">Amateur-theatre measure</a>'
    ds = DICT_DATASET.get(stem, stem)
    if folder == "analysis":
        ds = "legion" if stem.startswith("legionnaires") else "habsburg"
    if dd is None:
        return "no data dictionary"
    rows = dd[dd["dataset"] == ds]
    if rows.empty:
        return f'<a href="#dictionary">Data dictionary</a>: no rows under <code>{_esc(ds)}</code>'
    names, prefixes = _dictionary_names(rows["final_name"])
    known = sum(1 for c in cols if c in names or (prefixes and c.startswith(prefixes))) if cols else len(rows)
    tail = f" ({known:,} of {len(cols):,} columns named there)" if cols and folder != "analysis" else ""
    return (f'<a href="#dictionary">Data dictionary</a>, dataset <code>{_esc(ds)}</code>: '
            f"{len(rows):,} rows{tail}")

#: A decision is open when its Status cell says so: Open, pending, waits/awaiting, or a ruling page with
#: nothing ruled ("0 answered", "0 of 299 ruled"). Read from the Status cell only, never the prose. A bare
#: 'unruled' is not enough: a settled row may count the items it left unruled (D-219, D-251).
OPEN_STATUS = re.compile(r"\bopen\b|pending|\ba?wait|not yet ruled"
                         r"|\b0 (?:of \d[\d,]* )?(?:ruled|answered)", re.I)
#: A Status longer than this is shown as its first sentence and the clause that makes it open, with a
#: link to the full row.
STATUS_MAX = 400
_PIPE = re.compile(r"(?<!\\)\|")


def decision_rows(text: str) -> tuple[list[dict], list[str]]:
    """``DECISIONS.md``'s rows as {id, date, decision, status}; and the ids whose row has no Status cell.

    A ``|`` inside a code span is not a cell break; a row with extra breaks in its prose still has its
    Status second from the end."""
    rows, unread = [], []
    for line in text.splitlines():
        if not re.match(r"^\|\s*D-\d+\s*\|", line):
            continue
        masked = re.sub(r"`[^`]*`", lambda m: m.group(0).replace("|", "\x00"), line.strip())
        cells = [c.strip().replace("\x00", "|") for c in _PIPE.split(masked)[1:-1]]
        if len(cells) < 5:
            unread.append(cells[0])
            continue
        rows.append({"id": cells[0], "date": cells[1], "decision": cells[2], "status": cells[-2]})
    return rows, unread


def _summary(md_cell: str, limit: int = 260) -> str:
    """The cell's first sentence as plain text (markup dropped), cut at *limit*."""
    t = _plain(md_cell)
    m = re.match(r"(.+?[.!?])(?:\s|$)", t)
    t = m.group(1) if m else t
    return t if len(t) <= limit else t[:limit].rsplit(" ", 1)[0] + " …"


def _plain(md_cell: str) -> str:
    return re.sub(r"\*\*|__|`|\[([^\]]*)\]\([^)]*\)", lambda m: m.group(1) or "", md_cell)


def status_excerpt(status: str, limit: int = STATUS_MAX) -> str | None:
    """A long Status cut to its first sentence plus the clause the open match sits in (plain text);
    None when the Status is short enough to quote whole."""
    if len(status) <= limit:
        return None
    t = _plain(status)
    head = _summary(t, 220)
    m = OPEN_STATUS.search(t)
    if not m or m.end() <= len(head.rstrip(" …")):
        return head
    starts = [t.rfind(sep, 0, m.start()) for sep in ("; ", ". ", " — ")]
    s = max(starts)
    s = 0 if s < 0 else s + 2
    ends = [i for i in (t.find(sep, m.end()) for sep in ("; ", ". ")) if i != -1]
    clause = t[s:(min(ends) + 1 if ends else len(t))].strip(" ;")
    if len(clause) > 260:
        lo = max(0, m.start() - s - 120)
        clause = ("… " if lo else "") + clause[lo:lo + 260].rsplit(" ", 1)[0] + " …"
    return f"{head.removesuffix(' …')} … {clause}"


def _inline(md: Callable[[str], str], text: str) -> str:
    return re.sub(r"^<p>(.*)</p>$", r"\1", md(text).strip(), flags=re.S)


def _md_section(text: str, heading: str) -> str:
    m = re.search(rf"^## {re.escape(heading)}\s*$(.*?)(?=^## |\Z)", text, flags=re.M | re.S)
    return m.group(1).strip() if m else ""


def queue_summary() -> dict:

def spell_status(text: str) -> tuple[str, str]:
    """SPELL_FILE_DESIGN.md's Status paragraph (to its blank line, as it wraps) and its 'Not built …'
    paragraph ('' where absent)."""
    m = re.search(r"^\*\*Status:.*?(?=\n\s*\n|\Z)", text, re.M | re.S)
    status = m.group(0).strip() if m else ""
    nb = re.search(r"^Not built,.*?(?=\n\s*\n|\Z)", text, re.M | re.S)
    return status, nb.group(0).strip() if nb else ""

def refusal_desertion(final: Path = FIN, theatre: str = "Russia") -> dict:
    """Desertion among legionnaires whose Czech-lands birth was linked to a 1910 settlement against those
    who stated an okres and a municipality and got no link, in one theatre.

    The no-link group is of two kinds, kept apart as well as together: ``unmatched`` (no 1910 settlement
    of the written name was found) and every other ``birth_link_refusal`` (declined by a rule: an
    ambiguity, an okres conflict, a guard, a post-1949 Prague number …; not always with a candidate in hand). If either were tied
    to desertion, the linked sample would be a selected one. Rates, n, and each difference from the linked
    group with its 95 % (Wald) interval."""
    cols = ["theater", "exit_D", "birth_settlement_id", "birth_link_refusal", "birth_okres_id", "birth_municipality"]
    L = pd.read_parquet(final / "persons" / "legionnaires_clean.parquet", columns=cols)
    L = L[L["theater"] == theatre]
    d = L["exit_D"].astype("boolean").fillna(False).astype(bool)
    linked = L["birth_settlement_id"].notna()
    nolink = (L["birth_link_refusal"].notna() & ~linked & L["birth_okres_id"].notna()
              & L["birth_municipality"].notna())
    unmatched = nolink & (L["birth_link_refusal"] == "unmatched")
    groups = {"refused": nolink, "unmatched": unmatched, "declined": nolink & ~unmatched}
    p1, n1 = d[linked].mean(), int(linked.sum())
    out = {"theatre": theatre, "p_linked": p1, "n_linked": n1,
           "written": _when(final / "persons" / "legionnaires_clean.parquet")}
    for tag, g in groups.items():
        p2, n2 = d[g].mean(), int(g.sum())
        se = math.sqrt(p1 * (1 - p1) / n1 + p2 * (1 - p2) / n2)
        out.update({f"p_{tag}": p2, f"n_{tag}": n2, f"diff_{tag}": p2 - p1,
                    f"lo_{tag}": p2 - p1 - 1.96 * se, f"hi_{tag}": p2 - p1 + 1.96 * se})
    out.update(diff=out["diff_refused"], lo=out["lo_refused"], hi=out["hi_refused"])
    return out


def silent_okres(final: Path = FIN) -> dict:
    """An estimate of births linked into the wrong okres without any sign: a soldier who wrote a
    neighbouring okres is caught when his village's name is not in it (the link moves him, rule
    ``p6_neighbour``) and missed when it is. So the silent rate ≈ the detected rate × the chance the
    written neighbour holds a namesake.

    detected = ``p6_neighbour`` corrections / linked births; namesake share = over linked births, the share
    of the okresy bordering the settlement's 1910 okres (MPIDR outlines, ``birth_settlement.okres_adjacency``)
    that hold a 1910 settlement of the same Czech name. Per person file."""
    from utils.parsing.birth_settlement import okres_adjacency
    adj = okres_adjacency()
    if adj is None:
        raise FileNotFoundError("the MPIDR shapefile (okres adjacency) is not there")
    S = pd.read_parquet(final / "geo" / "settlements_1910.parquet", columns=["settlement_id", "cs_name", "okres_id"])
    S["k"] = S["cs_name"].str.casefold().str.strip()
    names = S.dropna(subset=["okres_id"]).groupby("okres_id")["k"].agg(set).to_dict()
    out = {}
    for tag, f in (("legion", "legionnaires_clean"), ("vua", "habsburg_clean")):
        p = pd.read_parquet(final / "persons" / f"{f}.parquet",
                            columns=["birth_settlement_id", "birth_okres_correction_rule"])
        lk = p[p["birth_settlement_id"].notna()].merge(
            S[["settlement_id", "k", "okres_id"]], left_on="birth_settlement_id", right_on="settlement_id",
            how="left", validate="m:1")
        shares = []
        for o, k in zip(lk["okres_id"], lk["k"]):
            nb = adj.get(int(o), set()) if pd.notna(o) else set()
            if nb:
                shares.append(sum(k in names.get(n, ()) for n in nb) / len(nb))
        detected = float((p["birth_okres_correction_rule"] == "p6_neighbour").sum() / len(lk))
        namesake = sum(shares) / len(shares)
        out[tag] = {"linked": len(lk), "detected": detected, "namesake": namesake, "silent": detected * namesake}
    return out

_NOT_BUILT = ("*`data/final/panels/` holds no {what} yet: notebook 04 {sec} builds it, and this table fills on "
              "the first site build after that run.*")


def _versions() -> dict[str, str]:
    out = {}
    for mod in ("spells", "units_panel", "birthplace_covariates", "panel_io", "panel_dictionary"):
        try:
            out[f"{mod}_version"] = __import__(f"utils.panels.{mod}", fromlist=["__version__"]).__version__
        except (ImportError, AttributeError):
            out[f"{mod}_version"] = "—"
    return out


def _unit_block() -> str:
    u, p = PANELS / "units.parquet", PANELS / "unit_panel.parquet"
    if not (u.is_file() and p.is_file()):
        return _NOT_BUILT.format(what="unit table", sec="§2")
    from utils.panels import units_panel as up
    units, panel = pd.read_parquet(u), pd.read_parquet(p)
    with contextlib.redirect_stdout(io.StringIO()):
        r = up.units_panel_report(units, panel)
    share = r.get("records_theatre_agree_share")
    rows = [["Units (`unit_key`)", f"{r['units']:,}"],
            ["Records in a unit", f"{r['records_in_units']:,}"],
            ["Units matched to a Fidler headword", f"{r['units_matched_fidler']:,} ({r['records_in_matched_units']:,} records)"],
            ["Records whose theatre agrees with Fidler's, in matched units",
             f"{r['records_theatre_agree']:,} of {r['records_theatre_known_in_matched_units']:,}"
             + (f" ({_pc(share)})" if share == share and share is not None else "")],
            ["Units flagged on lineage (`lineage_flag`)", f"{r['units_lineage_flag']:,}"],
            ["Units whose entries span several headwords (`lineage_spans_variants`)",
             f"{r['units_lineage_spans_variants']:,}"],
            ["Spelling-collision groups listed, not merged", f"{r['spelling_collision_groups']:,}"],
            ["Unit × month rows", f"{r['panel_rows']:,} ({r['panel_units']:,} units)"],
            ["Person-months at risk", f"{r['panel_person_periods_at_risk']:,}"],
            ["Exits placed", f"{r['panel_exits']:,}"]]
    if "exits_desertion" in panel:
        rows.append(["Desertions placed", f"{int(panel['exits_desertion'].sum()):,}"])
    return (_md_table(["", "Count"], rows, right_from=1)
            + f"\n\n*Read from `data/final/panels/units` and `unit_panel` (written {_when(p)}) through "
              "`units_panel.units_panel_report`.*")


def _cov_tables_block() -> str:
    from utils.panels import birthplace_covariates as bc
    rows = []
    for name, key in bc.TABLES.items():
        f = PANELS / f"{name}.parquet"
        if not f.is_file():
            return _NOT_BUILT.format(what="birthplace covariate tables", sec="§3")
        pf = pq.ParquetFile(f)
        holds = _key_holds([f], [key], pf.schema_arrow.names)
        rows.append([f"`{name}`", f"`{key}`", f"{pf.metadata.num_rows:,}", f"{pf.metadata.num_columns:,}", holds])
    return (_md_table(["Table", "Key", "Rows", "Columns", "Key"], rows, right_from=2)
            + f"\n\n*Read from `data/final/panels/` (written {_when(PANELS / 'birthplace_settlement_cz.parquet')}).*")


def _cov_coverage_block() -> str:
    from utils.panels import birthplace_covariates as bc
    files = {n: PANELS / f"{n}.parquet" for n in bc.TABLES}
    if not all(f.is_file() for f in files.values()):
        return _NOT_BUILT.format(what="birthplace covariate tables", sec="§3")
    tables = {n: pd.read_parquet(f, columns=[bc.TABLES[n]]) for n, f in files.items()}
    cov = bc.coverage(tables, bc.load_person_links(FIN))
    cov = cov[cov.apply(lambda r: r["column"] == bc.TABLES[r["table"]], axis=1)]
    rows = [[{"legion": "Legion", "vua": "VÚA"}.get(r.dataset, r.dataset), f"`{r.table}`", f"{r.n_linked:,}",
             f"{r.n_through_gate:,}", f"{r.n_joined:,}", _pc(r.n_joined / max(r.n_linked, 1))]
            for r in cov.itertuples()]
    return (_md_table(["Person file", "Table", "Births linked", "Through the gate", "Joined", "Share of linked"],
                      rows, right_from=2)
            + "\n\n*`birthplace_covariates.coverage` on the tables' keys, against `data/final/persons`.*")


PANEL_PAGES = {"unit-panel": "reference/UNIT_PANEL.md", "birthplace-covariates": "reference/BIRTHPLACE_COVARIATES.md"}


def panels_text(sheet_id: str) -> str:
    """UNIT_PANEL.md or BIRTHPLACE_COVARIATES.md with its tokens and live tables filled."""
    src = PANEL_PAGES[sheet_id]
    facts = _versions()
    built = (PANELS / "spells.parquet").is_file()
    facts["panels_state"] = (f"built {_when(PANELS / 'spells.parquet')}" if built
                             else "not built yet: notebook 04 writes it")
    blocks = {"unit_panel": _unit_block, "covariate_tables": _cov_tables_block,
              "covariate_coverage": _cov_coverage_block}
    return fill((ROOT / src).read_text(encoding="utf-8"), facts, blocks, src)

def _notebook_block() -> str:
    rows = []
    for nb in sorted(NOTEBOOKS.glob("*.ipynb")):
        try:
            d = json.loads(nb.read_text(encoding="utf-8"))
        except (OSError, ValueError):
            continue
        cells = d.get("cells", [])
        first = next(("".join(c["source"]) for c in cells if c.get("cell_type") == "markdown"), "")
        title = re.sub(r"^#+\s*", "", first.strip().splitlines()[0]) if first.strip() else ""
        n_code = sum(c.get("cell_type") == "code" for c in cells)
        ran = sum(bool(c.get("outputs")) for c in cells if c.get("cell_type") == "code")
        rows.append([f"`{nb.name}`", title.replace("|", "/"), f"{n_code}", f"{ran}", _when(nb)])
    return (_md_table(["Notebook", "Its title", "Code cells", "With saved output", "File modified"], rows, right_from=2)
            + "\n\n*Read from `notebooks/` at build time (archived notebooks left out).*")


TIMINGS = ROOT / "output" / "timings"


def _timings_block(slowest: int = 5) -> str:
    """One row per recorded notebook run (``utils/docs/notebook_timings.py`` → ``output/timings/*.json``):
    its start, total and slowest cells with the section each sits in."""
    files = sorted(TIMINGS.glob("*.json")) if TIMINGS.is_dir() else []
    if not files:
        return ("*No run is recorded in `output/timings/`: run a notebook headless with its log, then "
                "`python utils/docs/notebook_timings.py <log>`.*")
    rows = []
    for f in files:
        r = json.loads(f.read_text(encoding="utf-8"))
        cells = sorted(r["cells"], key=lambda c: -c["seconds"])[:slowest]

        def _cell(c: dict) -> str:
            sec = re.sub(r"^(§\S+)\s+-\s+", r"\1 ", c["section"] or "")    # '§4 - Linkage (…) -> …'
            sec = re.split(r"\s+->|\s+\(|:", sec)[0].strip()[:60]          # → '§4 Linkage'
            return f"cell {c['cell']}" + (f" ({sec})" if sec else "") + f" {c['seconds']:,.1f} s"

        status = "" if r.get("status") in (0, None) else f" (exit status {r['status']})"
        rows.append([f"`{r['notebook']}`", f"{r['started'] or '—'}", f"{r['total_s']:,.1f} s "
                     f"(about {round(r['total_s'] / 60):,} min){status}", "; ".join(map(_cell, cells)),
                     f"`{_rel(f)}` and `{r['log']}`"])
    return (_md_table(["Notebook", "Run started", "Total", f"The {slowest} slowest cells", "Recorded in"], rows)
            + "\n\n*Read from `output/timings/` at build time; a section is named where the cell still reads as "
              "the run logged it.*")


def reproduce_text(text: str, checks: list[Check] | None) -> str:
    """REPRODUCE.md with the QA tally, this build's interpreter, the notebook table and the recorded run
    timings filled."""
    import platform
    facts = {"python_version": platform.python_version(), "pandas_version": pd.__version__,
             "qa_n": f"{len(checks):,}" if checks else "—",
             "qa_pass": f"{sum(1 for c in checks if c[1]):,}" if checks else "—"}
    return fill(text, facts, {"notebooks": _notebook_block, "timings": _timings_block}, "reference/REPRODUCE.md")
