How it works

Fidler, Jiří (2025). Encyklopedie československých legií. Praha: Universum. ISBN 978-80-242-9615-4 (e-book edition).

Why this exists

csol.cz records a unit as a bare label plus a formation in parentheses (33.p.pl. ( Čs. legie v Itálii )). Nothing in the source states which units actually existed, when, or in which theatre — so a mis-keyed regiment number is indistinguishable from a real one. Fidler gives, for every Legion unit, an explicit theatre and lifespan, which turns that into a checkable claim. It is the external ground truth behind reference/VALIDATION_FIDLER.md.

Copyright note

The book is in copyright and lives in the (gitignored) library/ reading folder, NOT in the repo. This script emits facts only — unit name, theatre, kind, dates, lineage — which are not protectable expression. It deliberately does not extract or store the encyclopedia's prose. Re-run it against your own copy to regenerate.

Run: python utils/reference_tools/units/build_unit_authority.py [path/to/encyklopedie.epub] Emits: reference/units/unit_authority.csv columns: unit_cs, theatre, kind, formed, disbanded, variant, n_commanders Consumed by: utils/validation/data_checks.py (theatre agreement check) and the validation write-up; not by the cleaning pipeline itself.

def read_entries(epub: Path) -> pd.DataFrame:
    """Stream the encyclopedia's paragraphs into (headword, subentry, text) rows.

    Markup is regular: ``p.hesla`` = headword, ``p.podhesla`` = sub-entry,
    ``p.paraoverride*`` = body, ``p.popisky*`` = image caption, ``p.mezera`` = spacer.

    The ``(?<!/)>`` in the paragraph pattern matters: spacers are written
    ``<p class="mezera"/>``, self-closing. Without it the ``(.*?)</p>`` tail runs past
    the spacer and swallows the FOLLOWING paragraph's text into the spacer's body —
    which silently drops every sub-entry (~7 % of the book).
    """
    rows: list[tuple[str, str | None, str]] = []
    with zipfile.ZipFile(epub) as z:
        names = sorted(n for n in z.namelist()
                       if re.search(r"Text/.*\.x?html$", n))
        for name in names:
            doc = z.read(name).decode("utf-8", "replace")
            head: str | None = None
            sub: str | None = None
            buf: list[str] = []
            for m in re.finditer(r'<p class="([^"]+)"[^>]*?(?<!/)>(.*?)</p>',
                                 doc, re.S):
                cls, body = m.group(1), _clean(m.group(2))
                if not body or cls == "mezera" or cls.startswith("popisky"):
                    continue
                if cls == "hesla":
                    if head:
                        rows.append((head, sub, " ".join(buf)))
                    head, sub, buf = body, None, []
                elif cls == "podhesla":
                    if head:
                        rows.append((head, sub, " ".join(buf)))
                    sub, buf = body, []
                else:
                    buf.append(body)
            if head:
                rows.append((head, sub, " ".join(buf)))
    return pd.DataFrame(rows, columns=["headword", "subentry", "text"])

def build(epub: Path) -> pd.DataFrame:
    df = read_entries(epub)
    first = df["text"].str.slice(0, 220)

    theatre = pd.Series(pd.NA, index=df.index, dtype="object")
    theatre[first.str.contains(r"legie v Rusku|legii v Rusku|vojska na Rusi")] = "Russia"
    theatre[first.str.contains(r"legie v It[áa]lii|legii v It[áa]lii")] = "Italy"
    theatre[first.str.contains(r"legie ve? Francii|legii ve? Francii")] = "France"

    kind = pd.Series("other", index=df.index, dtype="object")
    kind[first.str.contains(r"^(?:Vojskov|Samostatn|Vy[šs][šs][íi] jednotka|Slo[žz]ka"
                            r"|N[áa]hradn)")] = "unit"
    kind[first.str.contains(r"Dopravn[íi] jednotka")] = "transport"
    kind[first.str.contains(r"^(?:M[ěe]sto|Vesnice|P[řr][íi]stavn[íi]|Osada|Obec"
                            r"|[Žz]elezni[čc]n[íi] stanice)")] = "place"
    kind[first.str.contains(r"^(?:Vojensk[ýy] [úu][řr]ad|[Úu][sst][řr]edn[íi] vojensk"
                            r"|Vojensk[ýy] [úu]stav)")] = "institution"
    kind[df["text"].str.match(r"^\(\d{4}")] = "person"
    df["theatre"], df["kind"] = theatre, kind

    # keep only the substantive military entries (drop places/people/sub-entries)
    keep = df["theatre"].notna() & df["kind"].isin(["unit", "transport", "institution"]) \
        & df["subentry"].isna()
    out = df[keep].copy()

    out["number"] = out["headword"].map(headword_number)
    out["variant"] = out["headword"].str.extract(r"\[(I+)\]$")[0]
    out["formed"] = out["text"].map(lambda t: _date(t, r"vznikl[oa]?|byl[oa]? z[řr][íi]zeno"))
    out["disbanded"] = out["text"].map(
        lambda t: _date(t, r"zaniklo?|byl[oa]? zru[šs]eno|bylo? p[řr]ejmenov[áa]no"))
    out["n_commanders"] = out["text"].str.count(r"zast[áa]val")

    out = out.rename(columns={"headword": "unit_cs"})
    cols = ["unit_cs", "number", "variant", "theatre", "kind",
            "formed", "disbanded", "n_commanders"]
    return out[cols].sort_values(["theatre", "kind", "unit_cs"]).reset_index(drop=True)

build_unit_authority.py — generate reference/units/unit_authority.csv
=====================================================================
Derives a **factual authority table of Czechoslovak Legion units** from

    Fidler, Jiří (2025). *Encyklopedie československých legií.* Praha: Universum.
    ISBN 978-80-242-9615-4 (e-book edition).

from __future__ import annotations

import html
import re
import sys
import zipfile
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[3]
OUT = ROOT / "reference" / "units" / "unit_authority.csv"
DEFAULT_EPUB = (ROOT / "library" / "books"
                / "Fidler-2025_Encyklopedie-ceskoslovenskych-legii.epub")

# Czech month names as they appear in the dates ("od 12. července 1918").
MONTHS = {m: i for i, m in enumerate(
    ["ledna", "února", "března", "dubna", "května", "června", "července",
     "srpna", "září", "října", "listopadu", "prosince"], start=1)}

# Ordinal words that open a unit headword ("TŘICÁTÝ PRVNÍ ČS. STŘELECKÝ PLUK" = 31st).
_ONES = {"PRVNÍ": 1, "PRVNÁ": 1, "DRUHÝ": 2, "DRUHÁ": 2, "TŘETÍ": 3, "ČTVRTÝ": 4,
         "ČTVRTÁ": 4, "PÁTÝ": 5, "PÁTÁ": 5, "ŠESTÝ": 6, "ŠESTÁ": 6, "SEDMÝ": 7,
         "SEDMÁ": 7, "OSMÝ": 8, "OSMÁ": 8, "DEVÁTÝ": 9, "DEVÁTÁ": 9, "DESÁTÝ": 10,
         "DESÁTÁ": 10, "JEDENÁCTÝ": 11, "JEDENÁCTÁ": 11, "DVANÁCTÝ": 12,
         "DVANÁCTÁ": 12}
_TENS = {"DVACÁTÝ": 20, "DVACÁTÁ": 20, "TŘICÁTÝ": 30, "TŘICÁTÁ": 30}


def _clean(s: str) -> str:
    s = re.sub(r"<[^>]+>", "", s)
    return re.sub(r"\s+", " ", html.unescape(s).replace(" ", " ")
                  .replace("→", "->")).strip()

def _date(text: str, verbs: str) -> str | None:
    """First ``D. month YYYY`` following one of *verbs* -> ISO date.

    *verbs* is wrapped in a non-capturing group: unwrapped, its ``|`` would split the
    whole pattern and a bare verb match would succeed with no date groups at all.
    """
    m = re.search(
        rf"(?:{verbs})[^.]{{0,60}}?(\d{{1,2}})\.\s*([a-zěščřžýáíéúůň]+)\s+(\d{{4}})",
        text)
    if not m or m.group(2) not in MONTHS:
        return None
    return f"{m.group(3)}-{MONTHS[m.group(2)]:02d}-{int(m.group(1)):02d}"
