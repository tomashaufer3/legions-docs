How it works

Every legionnaire left the Habsburg army once; nothing in the project records what he did inside it. Three name-level inventories of wartime military courts in the Czech lands (VÚA–VHA Praha) list the accused with rank, unit and birth year. This script

  1. parses the three PDFs (utils.parsing.vua_courts) and prints coverage — case files and accused by court and year, share military, offence flags, linkable fields;
  2. links the military accused to data/final/persons/legionnaires_clean.parquet on surname + given name (the phonetic part-by-part test of utils/linkage.py) + birth year, with the Habsburg unit as tie-breaker, and prints the counts;
  3. repeats the link with every court birth year shifted — a placebo that shows how many links name and year yield by coincidence, which is the yardstick for step 2.

A pilot, not a pipeline stage. It reads data/final and writes nothing there. The parsed tables and the candidate pairs go to data/interim/vua_courts/ so they can be read by eye (--no-write to skip); this script rebuilds them on every run.

Fetch the PDFs first — three requests, once::

python utils/scraping/fetch_vua_courts.py

Usage::

python utils/reference_tools/units/run_vua_courts_pilot.py
python utils/reference_tools/units/run_vua_courts_pilot.py --no-write --skip-link

    """Parse, report and (optionally) link; returns every table built."""
    entries, persons = vc.parse_all()
    vc.courts_report(entries, persons)
    tables = {"entries": entries, "persons": persons}
    if link:
        legion = pd.read_parquet(LEGION, columns=LEGION_COLS)
        print(f"\nlegionnaire file: {len(legion):,} records "

        cands, picks = vc.pilot_link(persons, legion)
        vc.pilot_link_report(cands, picks, persons, entries)
        tables.update(link_candidates=cands, link_picks=picks)

            c, p = vc.pilot_link(persons, legion, year_shift=shift)

                  f"{int((p['n_candidates'] > 0).sum()):>4,} accused with a candidate; "
                  f"{int(status.get('unique', 0)):>4,} unique; "
                  f"{int(status.get('unit_resolved', 0)):>3,} unit-resolved; "
                  f"{int(p['unit_agree'].fillna(False).sum()):>3,} picked with the unit agreeing")
        print("  Read the real counts against these: only the excess over the placebo, and "
              "above all the\n  picks whose unit agrees, are evidence of the same man.")
    if write:
        OUT_DIR.mkdir(parents=True, exist_ok=True)
        for name, frame in tables.items():
            frame.to_csv(OUT_DIR / f"{name}.csv", index=False, encoding="utf-8-sig")
        print(f"\nwrote {', '.join(f'{n}.csv' for n in tables)} → "
              f"{OUT_DIR.relative_to(ROOT).as_posix()}/ (nothing in data/final)")
    return tables

#!/usr/bin/env python3
"""
Pilot — the VÚA military-court inventories as a Habsburg-side record of disloyalty.

Every legionnaire left the Habsburg army once; nothing in the project records what he did
*inside* it. Three name-level inventories of wartime military courts in the Czech lands
(VÚA–VHA Praha) list the accused with rank, unit and birth year. This script

1. parses the three PDFs (:mod:`utils.parsing.vua_courts`) and prints coverage — case
   files and accused by court and year, share military, offence flags, linkable fields;
2. links the **military** accused to ``data/final/persons/legionnaires_clean.parquet`` on
   surname + given name (the phonetic part-by-part test of ``utils/linkage.py``) + birth
   year, with the Habsburg unit as tie-breaker, and prints the counts;
3. repeats the link with every court birth year shifted — a placebo that shows how many
   links name and year yield by coincidence, which is the yardstick for step 2.

**A pilot, not a pipeline stage.** It reads ``data/final`` and writes nothing there. The
parsed tables and the candidate pairs go to ``data/interim/vua_courts/`` so they can be
read by eye (``--no-write`` to skip); this script rebuilds them on every run.

Fetch the PDFs first — three requests, once::

    python utils/scraping/fetch_vua_courts.py

Usage::

    python utils/reference_tools/units/run_vua_courts_pilot.py
    python utils/reference_tools/units/run_vua_courts_pilot.py --no-write --skip-link
"""

from __future__ import annotations

import argparse
import sys
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

from utils.parsing import vua_courts as vc                            # noqa: E402

LEGION = ROOT / "data" / "final" / "persons" / "legionnaires_clean.parquet"
OUT_DIR = ROOT / "data" / "interim" / "vua_courts"
LEGION_COLS = ["legion_id", "surname", "given_name", "birth_year", "birth_municipality",
               "birth_okres", "habsburg_unit_capture", "habsburg_unit_peacetime"]
# years by which the placebo moves the court birth years; both directions, a few years
# out, so the cohort is about as large as the true one but cannot hold the same man
PLACEBO_SHIFTS = (-4, 4)


def run(*, link: bool = True, write: bool = True) -> dict[str, pd.DataFrame]:
    """Parse, report and (optionally) link; returns every table built."""
    entries, persons = vc.parse_all()
    vc.courts_report(entries, persons)
    tables = {"entries": entries, "persons": persons}
    if link:
        legion = pd.read_parquet(LEGION, columns=LEGION_COLS)
        print(f"\nlegionnaire file: {len(legion):,} records "
              f"({int(legion['birth_year'].notna().sum()):,} with a birth year)\n")
        cands, picks = vc.pilot_link(persons, legion)
        vc.pilot_link_report(cands, picks, persons, entries)
        tables.update(link_candidates=cands, link_picks=picks)
        print("\nPlacebo — the same link with the court birth years shifted "
              "(no true match is possible):")
        for shift in PLACEBO_SHIFTS:
            c, p = vc.pilot_link(persons, legion, year_shift=shift)
            status = p["status"].value_counts()
            print(f"  shift {shift:+d}: {len(c):>5,} candidate pairs; "
                  f"{int((p['n_candidates'] > 0).sum()):>4,} accused with a candidate; "
                  f"{int(status.get('unique', 0)):>4,} unique; "
                  f"{int(status.get('unit_resolved', 0)):>3,} unit-resolved; "
                  f"{int(p['unit_agree'].fillna(False).sum()):>3,} picked with the unit agreeing")
        print("  Read the real counts against these: only the excess over the placebo, and "
              "above all the\n  picks whose unit agrees, are evidence of the same man.")
    if write:
        OUT_DIR.mkdir(parents=True, exist_ok=True)
        for name, frame in tables.items():
            frame.to_csv(OUT_DIR / f"{name}.csv", index=False, encoding="utf-8-sig")
        print(f"\nwrote {', '.join(f'{n}.csv' for n in tables)} → "
              f"{OUT_DIR.relative_to(ROOT).as_posix()}/ (nothing in data/final)")
    return tables


if __name__ == "__main__":
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--skip-link", action="store_true", help="parse and report only")
    ap.add_argument("--no-write", action="store_true", help="print only, write no CSV")
    args = ap.parse_args()
    pd.set_option("display.width", 200, "display.max_columns", 30)
    run(link=not args.skip_link, write=not args.no_write)
