TL;DR

Draws a tier-stratified sample of resolved records and writes an Excel workbook a human fills in, so the pipeline's precision (not just coverage) can be measured. Stratifying by match tier is the point: the easy tiers (okres / jaras) dominate the data, so a plain random sample would barely touch the soft tiers (fuzzy / settlement / muni) that carry the real error risk. Here each tier gets a fixed quota, and an unmatched stratum is included to probe false negatives (recall).

How it works

The Slovak passes are re-run in-process (records_geo_hu) so the sample reflects the current code — including the settlement tier — even before the notebook is re-executed. The Czech columns and the rank parse are read from data/final/ (their logic is unchanged).

Output: output/validation_sample.xlsx (gitignored — it holds personal data). Three sheets: Instructions, Place, Ranks, with dropdown verdict columns. The Instructions sheet carries a build stamp (date, seed, matcher versions), because a filled workbook is evidence about a build and is worthless once it cannot be tied to one.

Re-running never overwrites judged work: if the target already holds filled verdict cells the fresh sample is written to a dated sibling instead, and the collision is announced.

Run: python utils/validation/build_validation_workbook.py

def build_place_sheet(legion: pd.DataFrame, habs: pd.DataFrame) -> pd.DataFrame:
    """Draw the tier-stratified place-matching sample, both datasets pooled.

    Stratifying is the whole point: the ``okres`` / ``jaras`` tiers dominate the data, so
    a plain random sample would barely touch the soft tiers (``fuzzy`` / ``settlement`` /
    ``muni``) where the error risk actually sits. Each tier gets a fixed quota, and an
    ``unmatched`` stratum probes false negatives.

    Args:
        legion, habs: final person frames, with the Slovak columns refreshed.

    Returns:
        One row per sampled record — the raw birthplace text, what the pipeline decided,
        and the tier it used — ready for a human verdict.
    """
    rows = []
    for df, ds, idc in [(legion, "legion", "legion_id"), (habs, "habsburg", "vua_id")]:
        d = df.copy()
        d["_tier"] = d.apply(_place_tier, axis=1)
        d["_ds"], d["_id"] = ds, d[idc]
        rows.append(d)
    allp = pd.concat(rows, ignore_index=True)
    samp = _sample(allp, "_tier", PLACE_PER_TIER)

    out = pd.DataFrame({
        "row": range(1, len(samp) + 1),
        "dataset": samp["_ds"],
        "person_id": samp["_id"],
        "name": (samp["surname"].fillna("") + " " + samp["given_name"].fillna("")).str.strip(),
        "birth_year": samp["birth_year"],
        "raw_municipality": samp["birth_municipality"],
        "raw_okres": samp["birth_okres"],
        "raw_land": samp["birth_land"],
        "PREDICTED_district": samp.apply(_pred_place, axis=1),
        "tier": samp["_tier"],
        "map": [f"https://www.google.com/maps?q={la},{lo}" if pd.notna(la) and pd.notna(lo) else ""
                for la, lo in zip(samp["birth_lat"], samp["birth_lon"])],
        "VERDICT": "",               # human: correct / wrong / unsure
        "true_district_if_wrong": "",
        "notes": "",
    })
    return out

def build_rank_sheet(legion: pd.DataFrame) -> pd.DataFrame:
    """Draw the stratified rank-parsing sample (legionnaires only — they carry both ranks).

    Strata are the parser's own tiers, plus two diagnostic groups forced in so they are
    guaranteed coverage rather than lost in the mass: ``unparsed`` (a raw rank string the
    ladder could not resolve) and ``flagged-uncertain`` (terms ``rank_ladders.md``
    records as reconstructed or contested — *podporučík*, *rotmistr*, the candidate band).
    Those two are where a rank error would actually come from.

    Returns:
        One row per sampled record — raw rank text vs. the parsed label/ordinal — for a
        human verdict.
    """
    d = legion.copy()
    d["_norm"] = d["habsburg_rank"].astype(str).str.strip().str.lower()
    d["_stratum"] = d["habsburg_rank_tier"].astype("object")
    # override stratum for the diagnostic groups so they are guaranteed coverage
    unparsed = d["habsburg_rank"].notna() & d["habsburg_rank_ord"].isna()
    uncertain = d["_norm"].isin({s.lower() for s in _UNCERTAIN_RANKS})
    d.loc[uncertain, "_stratum"] = "flagged-uncertain"
    d.loc[unparsed, "_stratum"] = "unparsed(raw≠<NA>)"
    d.loc[d["habsburg_rank"].isna(), "_stratum"] = pd.NA
    d = d[d["_stratum"].notna()]
    samp = _sample(d, "_stratum", RANK_PER_TIER)

    def fmt(r, p):
        clean, en, ordv, tier = (r.get(f"{p}_clean"), r.get(f"{p}_en"),
                                 r.get(f"{p}_ord"), r.get(f"{p}_tier"))
        if pd.isna(clean):
            return "—"
        return f"{clean} ({en}) · ord {ordv} · {tier}"

    out = pd.DataFrame({
        "row": range(1, len(samp) + 1),
        "person_id": samp["legion_id"],
        "name": (samp["surname"].fillna("") + " " + samp["given_name"].fillna("")).str.strip(),
        "stratum": samp["_stratum"],
        "raw_habsburg_rank": samp["habsburg_rank"],
        "PREDICTED_habsburg": samp.apply(lambda r: fmt(r, "habsburg_rank"), axis=1),
        "raw_legion_rank": samp["legion_rank"],
        "PREDICTED_legion": samp.apply(lambda r: fmt(r, "legion_rank"), axis=1),
        "HAB_VERDICT": "",           # human: correct / wrong / unsure
        "LEGION_VERDICT": "",
        "true_rank_notes": "",
    })
    return out

def main() -> None:
    """Build both sheets and write the workbook, echoing the realised strata.

    The printed tier/stratum counts are the check that the stratification actually bit —
    if a soft tier comes back near zero, the sample cannot say anything about it.
    """
    # The Windows console is cp1250 here, and the strata labels carry '≠' (and the place
    # rows Czech/Slovak diacritics). Report in UTF-8 rather than blunting the labels — the
    # workbook itself is fine either way, but a build that dies on its own summary line
    # looks like a failed build.
    try:
        sys.stdout.reconfigure(encoding="utf-8", errors="replace")
    except (AttributeError, OSError):                    # pragma: no cover — non-tty stdout
        pass

    legion, habs = _load_with_current_sk()
    place = build_place_sheet(legion, habs)
    ranks = build_rank_sheet(legion)
    written = _write_xlsx(place, ranks)
    print(f"wrote {written.relative_to(ROOT)}")
    print(f"  Place: {len(place)} rows across {place['tier'].nunique()} tiers")
    print(place["tier"].value_counts().to_string())
    print(f"  Ranks: {len(ranks)} rows across {ranks['stratum'].nunique()} strata")
    print(ranks["stratum"].value_counts().to_string())

"""
build_validation_workbook.py — human-validation sample for place matching + rank parsing
========================================================================================
Draws a **tier-stratified** sample of resolved records and writes an Excel workbook a human
fills in, so the pipeline's *precision* (not just coverage) can be measured. Stratifying by
match tier is the point: the easy tiers (``okres`` / ``jaras``) dominate the data, so a plain
random sample would barely touch the soft tiers (``fuzzy`` / ``settlement`` / ``muni``) that
carry the real error risk. Here each tier gets a fixed quota, and an ``unmatched`` stratum is
included to probe false negatives (recall).

The Slovak passes are **re-run in-process** (``records_geo_hu``) so the sample reflects the
current code — including the ``settlement`` tier — even before the notebook is re-executed.
The Czech columns and the rank parse are read from ``data/final/`` (their logic is unchanged).

Output: ``output/validation_sample.xlsx`` (gitignored — it holds personal data). Three sheets:
*Instructions*, *Place*, *Ranks*, with dropdown verdict columns. The *Instructions* sheet
carries a build stamp (date, seed, matcher versions), because a filled workbook is evidence
about **a build** and is worthless once it cannot be tied to one.

Re-running never overwrites judged work: if the target already holds filled verdict cells the
fresh sample is written to a dated sibling instead, and the collision is announced.

Run:
    python utils/validation/build_validation_workbook.py
"""

from __future__ import annotations

import datetime as _dt
import sys
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from utils.parsing import districts_hu as dzh, lexikon_sk as lxs, records_geo as rg, records_geo_hu as rgh

FIN = ROOT / "data" / "final"
OUT = ROOT / "output" / "validation_sample.xlsx"
SEED = 42
PLACE_PER_TIER = 25          # rows sampled per place tier
RANK_PER_TIER = 20           # rows sampled per rank stratum

# Every column a human is asked to judge. Used both to shade/validate them on the way out
# and to detect filled-in work on the way back in (see `_verdicts_filled`).
_VERDICT_COLS = {"VERDICT", "HAB_VERDICT", "LEGION_VERDICT"}

# Habsburg raw rank terms the ladder flags as reconstructed / uncertain (rank_ladders.md).
_UNCERTAIN_RANKS = {"podporučík", "rotmistr", "nápredník", "nápředník", "závodčí", "kadet-aspirant",
                    "praporčík", "důstojnický zástupce", "strážmistr", "ohněstrůjce", "nadlékař"}

def _load_with_current_sk() -> tuple[pd.DataFrame, pd.DataFrame]:
    """Load both person frames and refresh the Slovak járás columns with the current code."""
    from utils.parsing.records_geo import settlement_name_set
    backbone = dzh.build_jaras_backbone(ROOT / "data" / "raw" / "census_hu" / "shp",
                                        curation_csv=ROOT / "reference" / "places" / "district_sk_curation.csv")
    cross = lxs.build_crosswalk_sk(
        ROOT / "library" / "lexicons" /
        "Majo-2012_Historicko-demograficky-lexikon-obci-Slovenska_1880-1910.pdf", backbone)
    # same cross-frame homonym guard the pipeline (notebook 02 §4) uses
    cz_names = settlement_name_set(pd.read_parquet(FIN / "geo" / "settlements_1910.parquet"))
    out = []
    for name in ("legionnaires_clean", "habsburg_clean"):
        df = pd.read_parquet(FIN / "persons" / f"{name}.parquet")
        df = rgh.add_jaras_id(df, backbone, prefix="birth")          # seat pass (current code)
        df = rgh.add_settlement_jaras(df, cross, backbone, prefix="birth",
                                      exclude_names=cz_names)        # village pass (guarded)
        # The same two steps notebook 02 §4 runs after the passes. Without them the sample
        # judged the pre-D-19 assignment (every chartered-town birth on its járás) and a land
        # label the pipeline no longer emits (audit 2026-09-22 §B16).
        df = rgh.resolve_jaras_twins(df, backbone, cross, prefix="birth")
        df = rgh.set_land_from_jaras(df, prefix="birth")
        out.append(df)
    return out[0], out[1]


def _place_tier(r) -> str:
    """Unified place tier: Czech okres, else Slovak járás, else Slovak county-only, else unmatched."""
    if pd.notna(r["birth_okres_id"]):
        return f"cz:{r['birth_okres_match']}"
    if pd.notna(r["birth_jaras_id"]):
        return f"sk:{r['birth_jaras_match']}"
    if pd.notna(r.get("birth_megye_id")):
        return "sk:megye"                       # county resolved, district not (partial)
    return "unmatched"


def _pred_place(r) -> str:
    if pd.notna(r["birth_okres_id"]):
        s = str(r["birth_okres_cs"] or "")
        court = r.get("birth_court_cs")
        if pd.notna(court) and court and court != r["birth_okres_cs"]:
            s += f"  [court: {court}]"
        return s
    if pd.notna(r["birth_jaras_id"]):
        return f"{r['birth_jaras_sk']}  ({r['birth_jaras_hu']})"
    if pd.notna(r.get("birth_megye_id")):
        return f"county only: {r.get('birth_megye_sk')}"
    return "—"


def _sample(df: pd.DataFrame, tier_col: str, per: int) -> pd.DataFrame:
    parts = [g.sample(min(per, len(g)), random_state=SEED)
             for _, g in df.groupby(tier_col, dropna=False) if len(g)]
    return pd.concat(parts).sort_values(tier_col).reset_index(drop=True)

def _stamp_lines() -> list[tuple[str, bool]]:
    """The build stamp: which code and which draw produced this workbook.

    A filled workbook is evidence about *a build*. Without this line a verdict sheet found
    six months from now cannot be tied to the parser that earned it, and re-running the
    matcher would silently invalidate the judgements without anyone noticing.
    """
    return [
        ("PROVENANCE OF THIS SAMPLE", True),
        (f"Built {_dt.date.today().isoformat()}  ·  seed {SEED}  ·  "
         f"{PLACE_PER_TIER} place rows per tier, {RANK_PER_TIER} rank rows per stratum", False),
        (f"Matcher versions: records_geo {rg.__version__} · records_geo_hu {rgh.__version__} · "
         f"lexikon_sk {lxs.__version__} · districts_hu {dzh.__version__}", False),
        ("If any of those versions change, the verdicts below describe the OLD matcher — "
         "re-draw the sample.", False),
        ("", False),
    ]


def _verdicts_filled(path: Path) -> int:
    """Count judged cells in an existing workbook, so a re-run cannot destroy human work."""
    if not path.exists():
        return 0
    from openpyxl import load_workbook
    try:
        wb = load_workbook(path, read_only=True, data_only=True)
    except Exception:                                    # unreadable → treat as unfilled
        return 0
    n = 0
    for ws in wb.worksheets:
        rows = ws.iter_rows(values_only=True)
        header = next(rows, None)
        if not header:
            continue
        cols = [j for j, h in enumerate(header) if h in _VERDICT_COLS]
        if not cols:
            continue
        for row in rows:
            n += sum(1 for j in cols if j < len(row) and str(row[j] or "").strip())
    wb.close()
    return n


def _target_path() -> Path:
    """Where to write: ``OUT``, unless it already holds verdicts — then a dated sibling.

    The old behaviour overwrote unconditionally. The workbook is the *only* copy of a
    judgement that costs hours to produce, so a re-run must never be the thing that loses it.
    """
    filled = _verdicts_filled(OUT)
    if not filled:
        return OUT
    alt = OUT.with_name(f"{OUT.stem}_{_dt.date.today().isoformat()}{OUT.suffix}")
    print(f"!! {OUT.name} already holds {filled} filled verdict cells — NOT overwriting.")
    print(f"!! Writing the fresh sample to {alt.name} instead; merge or discard by hand.")
    return alt


def _write_xlsx(place: pd.DataFrame, ranks: pd.DataFrame) -> Path:
    from openpyxl import Workbook
    from openpyxl.styles import Font, PatternFill, Alignment
    from openpyxl.worksheet.datavalidation import DataValidation
    from openpyxl.utils import get_column_letter

    out_path = _target_path()
    wb = Workbook()

    # ── Instructions ────────────────────────────────────────────────────────
    ws = wb.active
    ws.title = "Instructions"
    for i, (text, is_head) in enumerate(_INSTRUCTIONS[:2] + _stamp_lines()
                                        + _INSTRUCTIONS[2:], start=1):
        c = ws.cell(row=i, column=1, value=text)
        if is_head:
            c.font = Font(bold=True, size=13 if i == 1 else 11)
    ws.column_dimensions["A"].width = 100

    hdr_fill = PatternFill("solid", fgColor="003657")       # CU navy
    hdr_font = Font(bold=True, color="FFFFFF")
    ask_fill = PatternFill("solid", fgColor="FDE9EC")       # light red — human columns

    def _dump(df: pd.DataFrame, title: str, verdict_cols: list[str], link_col: str | None):
        ws = wb.create_sheet(title)
        ws.append(list(df.columns))
        for _, r in df.iterrows():
            ws.append(list(r))
        for j, col in enumerate(df.columns, start=1):
            c = ws.cell(row=1, column=j)
            c.fill, c.font = hdr_fill, hdr_font
            c.alignment = Alignment(vertical="center")
            letter = get_column_letter(j)
            width = 34 if col.startswith(("PREDICTED", "raw_mun", "true_", "notes", "name")) else \
                    12 if col in ("row", "dataset", "birth_year", "tier", "VERDICT",
                                  "HAB_VERDICT", "LEGION_VERDICT") else 18
            ws.column_dimensions[letter].width = width
            if col in verdict_cols:                          # shade + dropdown on human columns
                for row in range(2, len(df) + 2):
                    ws.cell(row=row, column=j).fill = ask_fill
                dv = DataValidation(type="list", formula1='"correct,wrong,unsure"', allow_blank=True)
                ws.add_data_validation(dv)
                dv.add(f"{letter}2:{letter}{len(df)+1}")
        if link_col and link_col in df.columns:
            lj = list(df.columns).index(link_col) + 1
            for row in range(2, len(df) + 2):
                cell = ws.cell(row=row, column=lj)
                if cell.value:
                    cell.hyperlink = cell.value
                    cell.value = "map"
                    cell.font = Font(color="0563C1", underline="single")
        ws.freeze_panes = "A2"

    _dump(place, "Place", ["VERDICT"], "map")
    _dump(ranks, "Ranks", ["HAB_VERDICT", "LEGION_VERDICT"], None)

    out_path.parent.mkdir(parents=True, exist_ok=True)
    wb.save(out_path)
    return out_path


def main() -> None:
    """Build both sheets and write the workbook, echoing the realised strata.

    The printed tier/stratum counts are the check that the stratification actually bit —
    if a soft tier comes back near zero, the sample cannot say anything about it.
    """
    # The Windows console is cp1250 here, and the strata labels carry '≠' (and the place
    # rows Czech/Slovak diacritics). Report in UTF-8 rather than blunting the labels — the
    # workbook itself is fine either way, but a build that dies on its own summary line
    # looks like a failed build.
    try:
        sys.stdout.reconfigure(encoding="utf-8", errors="replace")
    except (AttributeError, OSError):                    # pragma: no cover — non-tty stdout
        pass

    legion, habs = _load_with_current_sk()
    place = build_place_sheet(legion, habs)
    ranks = build_rank_sheet(legion)
    written = _write_xlsx(place, ranks)
    print(f"wrote {written.relative_to(ROOT)}")
    print(f"  Place: {len(place)} rows across {place['tier'].nunique()} tiers")
    print(place["tier"].value_counts().to_string())
    print(f"  Ranks: {len(ranks)} rows across {ranks['stratum'].nunique()} strata")
    print(ranks["stratum"].value_counts().to_string())


if __name__ == "__main__":
    main()
