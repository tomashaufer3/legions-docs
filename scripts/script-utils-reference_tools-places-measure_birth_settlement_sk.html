How it works

records_geo_hu puts each repair behind a module switch: C3 TOWN_OKRES_TO_JARAS (a town-named district string -> the járás around the town), C3n TOWN_COMPOUND_NAMES (1.5.2: a joined town's double name and its parts are the town), C7 SUBCARPATHIA_LAND (Subcarpathian units -> land Subcarpathia; shipped off since 1.5.1), S2 DISTRICT_STRING_REPAIRS (soft signs, county adjectives, aliases, pieces, fuzzy obec in the county), S2b FOREIGN_LAND_GUARD (S2 never applies to a positively foreign row), S2f TWO_DISTRICT_FRAME (1.5.2: a string naming two frame units keeps the frame), C3q C3N_GUARDS (1.5.3: a ? compound stays undecided, a hyphen tail checked on the county's old names), S2g FRAME_ALL_PIECES (1.5.3: S2f only when every piece is in the frame). birth_settlement_sk puts each 1.0.1, 1.0.2 and 1.0.3 rule in birth_settlement_sk.RULES; a rule shipped off (1.0.3: county_fuzzy_any, an author ruling) is measured apart, as the rows it would change if turned on (link_rules.shipped_plus).

The measurement builds the Czech-placed frames once — birth (birth_settlement_sk.czech_placed_frames: legionnaires from data/interim, VÚA re-parsed from the raw cell as birth_settlement does) and home (data/interim through records_geo.add_place_standardization, as notebook 02 §4 does) — then runs notebook 02 §4's four Slovak passes for both roles:

  • every switch off — must equal records_geo_hu 1.4.2 (--baseline: no row differs);
  • cumulative, switches on in the order C3, C3n, C7, S2, S2b, S2f (each changed row goes to the first switch that changes it), leave-one-out (all on but one), and the shipped defaults;
  • invariants in every state: no birth in both frames, the twin role set exactly on the twin population, a town_okres role consistent with its unit, a district tier only with a unit;
  • the Czech link's inputs (birth_settlement, read only): okres_string_unresolved on the birth role and the A4 home refusal (home_foreign's land / Hungary reasons against HOME_REFUSING_LANDS) on the home role, each against the all-off state.

The C3 surround check links every town's village births with every town county-only (TOWN_SURROUND = {}) and compares where they fall with records_geo_hu.TOWN_SURROUND / TOWN_SURROUND_REFUSED (birth_settlement_sk.town_surround_check); for each town of SURROUND_REVISITED (a surround ruling this round changed) it counts the rows the ruling moves (the shipped state against the town refused). Then the settlement link runs: with the repairs off and on, at the shipped defaults, and with birth_settlement_sk's rules off / cumulative / leave-one-out on the shipped frames (a link row "changes" when its settlement, tier, refusal, uncertainty or correction flag differs). With every rule off the link must equal birth_settlement_sk 1.0.0 (--link-baseline).

First (unless --no-build) it rebuilds output/places_sk/ at the shipped defaults (birth_settlement_sk.build: gazetteer, Majo join, links, report, rulings queue) — the script that produces those files. Writes output/places_sk/measure_sk_&lt;birth_settlement_sk version&gt;.json (LF); --rows-dir also writes the changed rows of each item as CSV for reading by hand.

Run (about an hour)::

python utils/reference_tools/places/measure_birth_settlement_sk.py [--rgh PATH] [--baseline PATH]
    [--link-baseline PATH] [--link-baseline-101 PATH] [--link-baseline-102 PATH] [--link-baseline-103 PATH]
    [--rgh-baseline-152 PATH] [--rows-dir DIR]

def attribution(states: dict, frames: dict, p: str) -> dict:
    """Cumulative attribution, leave-one-out and total per switch and dataset."""
    order = list(SWITCHES)
    cols = [f"{p}_{c}" for c in COMPARED]
    out = {"cumulative": {}, "leave_one_out": {}, "total_changed": {}, "shipped_changed": {}}
    for d in frames:
        base, full = states["off"][d], states["all"][d]
        out["total_changed"][d] = int(changed(base, full, cols).sum())
        out["shipped_changed"][d] = int(changed(base, states["shipped"][d], cols).sum())
        prev = base
        for i, k in enumerate(order):
            cur = states["cum_" + "+".join(order[: i + 1])][d]
            ch = changed(prev, cur, cols)
            out["cumulative"].setdefault(k, {})[d] = {
                "rows": int(ch.sum()),
                "by_unit_before": {str(u): int(n) for u, n in prev.loc[ch, f"{p}_jaras_id"].fillna("<NA>")
                                   .value_counts().head(25).items()},
                "jaras_changed": int(neq(prev[f"{p}_jaras_id"], cur[f"{p}_jaras_id"]).sum()),
                "land_changed": int(neq(prev[f"{p}_land"], cur[f"{p}_land"]).sum()),
                "gained_jaras": int((prev[f"{p}_jaras_id"].isna() & cur[f"{p}_jaras_id"].notna()).sum()),
                "lost_jaras": int((prev[f"{p}_jaras_id"].notna() & cur[f"{p}_jaras_id"].isna()).sum())}
            prev = cur
        for k in order:
            out["leave_one_out"].setdefault(k, {})[d] = int(changed(states["loo_" + k][d], full, cols).sum())
    return out

def czech_inputs(states: dict, frames: dict, p: str) -> dict:
    """The Czech link's inputs against the all-off state: ``okres_string_unresolved`` (birth) or the A4 home
    refusal (home), flips per state and land transitions to the shipped state."""
    from utils.parsing import birth_settlement as bs
    fn = (lambda f: bs.okres_string_unresolved(f, p)) if p == "birth" else home_refuses
    out = {}
    for s in [k for k in states if k != "off"]:
        out[s] = {}
        for d in frames:
            a, b = fn(states["off"][d]), fn(states[s][d])
            out[s][d] = {"true_to_false": int((a & ~b).sum()), "false_to_true": int((~a & b).sum())}
    trans = {}
    for d in frames:
        a, b = states["off"][d][f"{p}_land"].astype("object"), states["shipped"][d][f"{p}_land"].astype("object")
        ct = pd.crosstab(a.fillna("<NA>"), b.fillna("<NA>"))
        trans[d] = {f"{i}->{j}": int(ct.loc[i, j]) for i in ct.index for j in ct.columns if i != j and ct.loc[i, j]}
    return {"flips_vs_all_off": out, "land_transitions_all_off_to_shipped": trans}

class Linker:
    """Links frames under a records_geo_hu state and a set of birth_settlement_sk rules, caching the bundles."""

    def __init__(self, backbone, majo, parts, earlier, rgh):
        self.backbone, self.majo, self.parts, self.earlier, self.rgh = backbone, majo, parts, earlier, rgh
        self.bundles: dict = {}

    def bundle(self, rules: dict) -> dict:
        key = (*(rules[r] for r in bss.BUNDLE_RULES), self.rgh.DISTRICT_STRING_REPAIRS)
        if key not in self.bundles:
            bss.RULES.update(rules)
            self.bundles[key] = bss.build_gazetteer_sk(self.backbone, self.majo, self.parts, earlier=self.earlier)
        return self.bundles[key]

    def link(self, frames: dict, rules: dict) -> dict:
        bss.RULES.update(rules)
        b = self.bundle(rules)
        return {d: bss.add_birth_settlement_sk_columns(f, b, self.backbone, d) for d, f in frames.items()}

def link_items(linker: Linker, frames: dict, shipped_rules: dict, rows_dir: Path | None) -> tuple[dict, dict]:
    """The link rules shipped on: off / cumulative / leave-one-out on *frames* (the rules shipped off stay off; the
    cumulative end is the shipped state); each rule shipped off (1.0.3: ``county_fuzzy_any``) turned on alone over the
    shipped state (``shipped_plus``). Returns the counts and the all-off runs."""
    on = [r for r in RULES if shipped_rules[r]]
    off = {r: False for r in RULES}
    runs = {"off": linker.link(frames, off)}
    for i in range(len(on)):
        runs["cum_" + on[i]] = linker.link(frames, {r: r in on[: i + 1] for r in RULES})
    for r in on:
        runs["loo_" + r] = linker.link(frames, {q: q in on and q != r for q in RULES})
    for r in (q for q in RULES if not shipped_rules[q]):
        runs["plus_" + r] = linker.link(frames, {q: q in on or q == r for q in RULES})
    full = runs["cum_" + on[-1]]
    out = {"cumulative": {}, "leave_one_out": {}, "total_changed": {}, "shipped_plus": {}}
    for d in frames:
        out["total_changed"][d] = int(changed(runs["off"][d], full[d], LINK_COMPARED).sum())
        for r in (q for q in RULES if not shipped_rules[q]):
            cur = runs["plus_" + r][d]
            ch = changed(full[d], cur, LINK_COMPARED) & cur["birth_settlement_sk_in_scope"]
            out["shipped_plus"].setdefault(r, {})[d] = {
                "rows": int(ch.sum()), "linked_gained": int((full[d].loc[ch, "birth_settlement_sk_id"].isna()
                                                             & cur.loc[ch, "birth_settlement_sk_id"].notna()).sum())}
            _rows_csv(rows_dir, "plus_" + r, d, cur, full[d], ch)
        prev = runs["off"][d]
        for r in on:
            cur = runs["cum_" + r][d]
            ch = changed(prev, cur, LINK_COMPARED) & cur["birth_settlement_sk_in_scope"]
            out["cumulative"].setdefault(r, {})[d] = {
                "rows": int(ch.sum()), "linked_gained": int((prev.loc[ch, "birth_settlement_sk_id"].isna()
                                                             & cur.loc[ch, "birth_settlement_sk_id"].notna()).sum()),
                "linked_lost": int((prev.loc[ch, "birth_settlement_sk_id"].notna()
                                    & cur.loc[ch, "birth_settlement_sk_id"].isna()).sum()),
                "settlement_changed": int((prev.loc[ch, "birth_settlement_sk_id"].notna()
                                           & cur.loc[ch, "birth_settlement_sk_id"].notna()
                                           & neq(prev.loc[ch, "birth_settlement_sk_id"],
                                                 cur.loc[ch, "birth_settlement_sk_id"])).sum())}
            _rows_csv(rows_dir, r, d, cur, prev, ch)
            prev = cur
        for r in on:
            out["leave_one_out"].setdefault(r, {})[d] = int(changed(runs["loo_" + r][d], full[d],
                                                                   LINK_COMPARED).sum())
    bss.RULES.update(shipped_rules)
    return out, runs["off"]

def surround_check(rgh, linker: Linker, frames: dict, shipped_on: set, shipped_rules: dict, ctx: tuple) -> dict:
    """:func:`birth_settlement_sk.town_surround_check` with every town county-only, against the constants."""

    rgh.TOWN_SURROUND, rgh.TOWN_SURROUND_RULINGS_KEEP = {}, False

"""What ``records_geo_hu``'s repairs and ``birth_settlement_sk``'s rules change, row by row.

``records_geo_hu`` puts each repair behind a module switch: **C3** :data:`TOWN_OKRES_TO_JARAS` (a town-named
district string -> the járás around the town), **C3n** :data:`TOWN_COMPOUND_NAMES` (1.5.2: a joined town's double

:data:`TWO_DISTRICT_FRAME` (1.5.2: a string naming two frame units keeps the frame), **C3q** :data:`C3N_GUARDS` (1.5.3: a
``?`` compound stays undecided, a hyphen tail checked on the county's old names), **S2g** :data:`FRAME_ALL_PIECES` (1.5.3:
S2f only when every piece is in the frame). ``birth_settlement_sk`` puts each 1.0.1, 1.0.2 and 1.0.3 rule in

The measurement builds the Czech-placed frames once — birth (:func:`birth_settlement_sk.czech_placed_frames`:
legionnaires from ``data/interim``, VÚA re-parsed from the raw cell as ``birth_settlement`` does) and **home**

round changed) it counts the rows the ruling moves (the shipped state against the town refused). Then the settlement link runs: with the repairs off and on, at

def neq(a: pd.Series, b: pd.Series) -> pd.Series:
    x, y = a.astype("object"), b.astype("object")
    return ~((x == y).fillna(False) | (x.isna() & y.isna()))


def changed(a: pd.DataFrame, b: pd.DataFrame, cols: list[str]) -> pd.Series:
    out = pd.Series(False, index=a.index)
    for c in cols:
        out |= neq(a[c], b[c])
    return out


def invariants(f: pd.DataFrame, p: str, twins: set, backbone) -> dict:
    """Counts that must be 0 in every state."""
    towns = set(backbone.loc[backbone["unit_type"] != "járás", "unit_id"])
    out = {"both_frames": int((f[f"{p}_okres_id"].notna() & f[f"{p}_jaras_id"].notna()).sum()),
           "twin_role_off_twin_population": int((f[f"{p}_jaras_id"].isin(twins) != f[f"{p}_jaras_twin"].notna()).sum()),
           "district_tier_without_unit": int((f[f"{p}_jaras_match"].astype("object").isin(
               ["jaras", "town", "fuzzy", "settlement", "settlement_fuzzy"]) & f[f"{p}_jaras_id"].isna()).sum())}
    if f"{p}_jaras_town_okres" in f.columns:
        role = f[f"{p}_jaras_town_okres"].astype("object")
        out["town_okres_city_not_a_town"] = int((role.eq("city") & ~f[f"{p}_jaras_id"].isin(towns)).sum())
        out["town_okres_ring_on_a_town"] = int((role.eq("ring") & f[f"{p}_jaras_id"].isin(towns)).sum())
    return out


def attribution(states: dict, frames: dict, p: str) -> dict:
    """Cumulative attribution, leave-one-out and total per switch and dataset."""
    order = list(SWITCHES)
    cols = [f"{p}_{c}" for c in COMPARED]
    out = {"cumulative": {}, "leave_one_out": {}, "total_changed": {}, "shipped_changed": {}}
    for d in frames:
        base, full = states["off"][d], states["all"][d]
        out["total_changed"][d] = int(changed(base, full, cols).sum())
        out["shipped_changed"][d] = int(changed(base, states["shipped"][d], cols).sum())
        prev = base
        for i, k in enumerate(order):
            cur = states["cum_" + "+".join(order[: i + 1])][d]
            ch = changed(prev, cur, cols)
            out["cumulative"].setdefault(k, {})[d] = {
                "rows": int(ch.sum()),
                "by_unit_before": {str(u): int(n) for u, n in prev.loc[ch, f"{p}_jaras_id"].fillna("<NA>")
                                   .value_counts().head(25).items()},
                "jaras_changed": int(neq(prev[f"{p}_jaras_id"], cur[f"{p}_jaras_id"]).sum()),
                "land_changed": int(neq(prev[f"{p}_land"], cur[f"{p}_land"]).sum()),
                "gained_jaras": int((prev[f"{p}_jaras_id"].isna() & cur[f"{p}_jaras_id"].notna()).sum()),
                "lost_jaras": int((prev[f"{p}_jaras_id"].notna() & cur[f"{p}_jaras_id"].isna()).sum())}
            prev = cur
        for k in order:
            out["leave_one_out"].setdefault(k, {})[d] = int(changed(states["loo_" + k][d], full, cols).sum())
    return out


def home_refuses(f: pd.DataFrame) -> pd.Series:
    """birth_settlement's A4 for the home reasons these passes move (``home_land`` / ``home_hungary``): True where
    the home refuses (:func:`birth_settlement.home_foreign` + :func:`birth_settlement.home_refuses`, first two
    branches; the name-based reasons do not read these columns)."""
    from utils.parsing import birth_settlement as bs
    hl = f["home_land"].astype("object")
    land = hl.map(lambda v: isinstance(v, str) and v.strip() not in ("", "Czech lands"))
    hung = f["home_megye_id"].notna() | f["home_jaras_id"].notna()
    return (land & hl.map(lambda v: isinstance(v, str) and v.strip() in bs.HOME_REFUSING_LANDS)) | (~land & hung)


def czech_inputs(states: dict, frames: dict, p: str) -> dict:
    """The Czech link's inputs against the all-off state: ``okres_string_unresolved`` (birth) or the A4 home
    refusal (home), flips per state and land transitions to the shipped state."""
    from utils.parsing import birth_settlement as bs
    fn = (lambda f: bs.okres_string_unresolved(f, p)) if p == "birth" else home_refuses
    out = {}
    for s in [k for k in states if k != "off"]:
        out[s] = {}
        for d in frames:
            a, b = fn(states["off"][d]), fn(states[s][d])
            out[s][d] = {"true_to_false": int((a & ~b).sum()), "false_to_true": int((~a & b).sum())}
    trans = {}
    for d in frames:
        a, b = states["off"][d][f"{p}_land"].astype("object"), states["shipped"][d][f"{p}_land"].astype("object")
        ct = pd.crosstab(a.fillna("<NA>"), b.fillna("<NA>"))
        trans[d] = {f"{i}->{j}": int(ct.loc[i, j]) for i in ct.index for j in ct.columns if i != j and ct.loc[i, j]}
    return {"flips_vs_all_off": out, "land_transitions_all_off_to_shipped": trans}


def role_states(rgh, frames: dict, prefix: str, ctx: tuple, shipped_on: set) -> dict:
    order = list(SWITCHES)
    states = {"off": run_state(rgh, frames, set(), prefix, ctx)}
    for i in range(len(order)):
        states["cum_" + "+".join(order[: i + 1])] = run_state(rgh, frames, set(order[: i + 1]), prefix, ctx)
    states["all"] = states["cum_" + "+".join(order)]
    for k in order:
        states["loo_" + k] = run_state(rgh, frames, set(order) - {k}, prefix, ctx)
    states["shipped"] = run_state(rgh, frames, shipped_on, prefix, ctx)
    return states


class Linker:
    """Links frames under a records_geo_hu state and a set of birth_settlement_sk rules, caching the bundles."""

    def __init__(self, backbone, majo, parts, earlier, rgh):
        self.backbone, self.majo, self.parts, self.earlier, self.rgh = backbone, majo, parts, earlier, rgh
        self.bundles: dict = {}

    def bundle(self, rules: dict) -> dict:
        key = (*(rules[r] for r in bss.BUNDLE_RULES), self.rgh.DISTRICT_STRING_REPAIRS)
        if key not in self.bundles:
            bss.RULES.update(rules)
            self.bundles[key] = bss.build_gazetteer_sk(self.backbone, self.majo, self.parts, earlier=self.earlier)
        return self.bundles[key]

    def link(self, frames: dict, rules: dict) -> dict:
        bss.RULES.update(rules)
        b = self.bundle(rules)
        return {d: bss.add_birth_settlement_sk_columns(f, b, self.backbone, d) for d, f in frames.items()}


def _rows_csv(rows_dir: Path | None, name: str, d: str, cur: pd.DataFrame, prev: pd.DataFrame, ch: pd.Series) -> None:
    if rows_dir is None or not ch.any():
        return
    idc = "legion_id" if d == "legion" else "vua_id"
    keep = [c for c in [idc, *ROW_COLS] if c in cur.columns]
    pd.concat([cur.loc[ch, keep], prev.loc[ch, ["birth_settlement_sk_id", "birth_settlement_sk_match",
                                                "birth_settlement_sk_refusal"]].add_suffix("_before")],
              axis=1).to_csv(rows_dir / f"link_{name}_{d}.csv", index=False, encoding="utf-8", lineterminator="\n")


def link_items(linker: Linker, frames: dict, shipped_rules: dict, rows_dir: Path | None) -> tuple[dict, dict]:
    """The link rules shipped on: off / cumulative / leave-one-out on *frames* (the rules shipped off stay off; the
    cumulative end is the shipped state); each rule shipped off (1.0.3: ``county_fuzzy_any``) turned on alone over the
    shipped state (``shipped_plus``). Returns the counts and the all-off runs."""
    on = [r for r in RULES if shipped_rules[r]]
    off = {r: False for r in RULES}
    runs = {"off": linker.link(frames, off)}
    for i in range(len(on)):
        runs["cum_" + on[i]] = linker.link(frames, {r: r in on[: i + 1] for r in RULES})
    for r in on:
        runs["loo_" + r] = linker.link(frames, {q: q in on and q != r for q in RULES})
    for r in (q for q in RULES if not shipped_rules[q]):
        runs["plus_" + r] = linker.link(frames, {q: q in on or q == r for q in RULES})
    full = runs["cum_" + on[-1]]
    out = {"cumulative": {}, "leave_one_out": {}, "total_changed": {}, "shipped_plus": {}}
    for d in frames:
        out["total_changed"][d] = int(changed(runs["off"][d], full[d], LINK_COMPARED).sum())
        for r in (q for q in RULES if not shipped_rules[q]):
            cur = runs["plus_" + r][d]
            ch = changed(full[d], cur, LINK_COMPARED) & cur["birth_settlement_sk_in_scope"]
            out["shipped_plus"].setdefault(r, {})[d] = {
                "rows": int(ch.sum()), "linked_gained": int((full[d].loc[ch, "birth_settlement_sk_id"].isna()
                                                             & cur.loc[ch, "birth_settlement_sk_id"].notna()).sum())}
            _rows_csv(rows_dir, "plus_" + r, d, cur, full[d], ch)
        prev = runs["off"][d]
        for r in on:
            cur = runs["cum_" + r][d]
            ch = changed(prev, cur, LINK_COMPARED) & cur["birth_settlement_sk_in_scope"]
            out["cumulative"].setdefault(r, {})[d] = {
                "rows": int(ch.sum()), "linked_gained": int((prev.loc[ch, "birth_settlement_sk_id"].isna()
                                                             & cur.loc[ch, "birth_settlement_sk_id"].notna()).sum()),
                "linked_lost": int((prev.loc[ch, "birth_settlement_sk_id"].notna()
                                    & cur.loc[ch, "birth_settlement_sk_id"].isna()).sum()),
                "settlement_changed": int((prev.loc[ch, "birth_settlement_sk_id"].notna()
                                           & cur.loc[ch, "birth_settlement_sk_id"].notna()
                                           & neq(prev.loc[ch, "birth_settlement_sk_id"],
                                                 cur.loc[ch, "birth_settlement_sk_id"])).sum())}
            _rows_csv(rows_dir, r, d, cur, prev, ch)
            prev = cur
        for r in on:
            out["leave_one_out"].setdefault(r, {})[d] = int(changed(runs["loo_" + r][d], full[d],
                                                                   LINK_COMPARED).sum())
    bss.RULES.update(shipped_rules)
    return out, runs["off"]


def surround_check(rgh, linker: Linker, frames: dict, shipped_on: set, shipped_rules: dict, ctx: tuple) -> dict:
    """:func:`birth_settlement_sk.town_surround_check` with every town county-only, against the constants."""
    town_state = run_state(rgh, frames, shipped_on - {"C3"}, "birth", ctx)
    keep_surround, keep_rulings = dict(rgh.TOWN_SURROUND), rgh.TOWN_SURROUND_RULINGS_KEEP
    rgh.TOWN_SURROUND, rgh.TOWN_SURROUND_RULINGS_KEEP = {}, False
    try:
        co = run_state(rgh, frames, shipped_on | {"C3"}, "birth", ctx)
    finally:
        rgh.TOWN_SURROUND, rgh.TOWN_SURROUND_RULINGS_KEEP = keep_surround, keep_rulings
    for d in co:
        co[d]["birth_okres_town"] = town_state[d]["birth_jaras_id"].values
    runs = linker.link(co, shipped_rules)
    gs = bss.gista_settlements(bss.load_gista(), linker.backbone)
    cand = bss.town_surrounding_jaras(gs, linker.backbone, set(rgh.build_jaras_twin_pairs(linker.backbone)))
    gz = linker.bundle(shipped_rules)["gazetteer"]
    check = bss.town_surround_check(runs, cand, gz, rgh.TOWN_SURROUND_MIN_SHARE, rgh.TOWN_SURROUND_MIN_LINKED)
    for t, v in check.items():
        v["constant"] = ("kept" if rgh.TOWN_SURROUND.get(t) == v["candidate"]
                         else "ruling" if rgh.TOWN_SURROUND_RULINGS.get(t) == v["candidate"]
                         else "refused" if rgh.TOWN_SURROUND_REFUSED.get(t) == v["candidate"] else "absent")
        v["constant_agrees"] = (v["constant"] == "kept") == v["keep"] if v["constant"] != "ruling" else None
    return check


def surround_revisited(rgh, linker: Linker, frames: dict, shipped_on: set, shipped_rules: dict, ctx: tuple,
                       shipped: dict) -> dict:
    """``{town: {role: {dataset: rows}}}``: the rows each :data:`SURROUND_REVISITED` ruling would move — the shipped
    state (the town refused, its villages county-only) against the same with the ruling kept
    (``records_geo_hu.TOWN_SURROUND_RULINGS`` for that town, ``TOWN_SURROUND_RULINGS_KEEP``), passes compared on
    :data:`COMPARED`, and for the birth role the settlement link too (``link_rows``)."""
    out = {}
    for town in SURROUND_REVISITED:
        keep_r, keep_k = dict(rgh.TOWN_SURROUND_RULINGS), rgh.TOWN_SURROUND_RULINGS_KEEP
        if town not in keep_r:
            out[town] = "not in TOWN_SURROUND_RULINGS"
            continue
        rgh.TOWN_SURROUND_RULINGS, rgh.TOWN_SURROUND_RULINGS_KEEP = {town: keep_r[town]}, True
        try:
            alt = {p: run_state(rgh, frames[p], shipped_on, p, ctx) for p in frames}
        finally:
            rgh.TOWN_SURROUND_RULINGS, rgh.TOWN_SURROUND_RULINGS_KEEP = keep_r, keep_k
        res = {p: {d: int(changed(alt[p][d], shipped[p][d], [f"{p}_{c}" for c in COMPARED]).sum())
                   for d in frames[p]} for p in frames}
        a, b = linker.link(alt["birth"], shipped_rules), linker.link(shipped["birth"], shipped_rules)
        res["link_rows"] = {d: int(changed(a[d], b[d], LINK_COMPARED).sum()) for d in a}
        out[town] = res
    return out


def link_baseline(path: Path, frames: dict, backbone, linker: Linker, off=None, cols=None) -> dict:
    """Rows where *path* (an earlier birth_settlement_sk file, at its own defaults) and this version with the rules
    *off* off (default: every rule — then *path* is 1.0.0) disagree on *cols* (default: settlement, tier, refusal;
    ``birth_jaras_corrected`` left out against 1.0.0, which wrote False where 1.0.1 writes <NA> without a unit)."""
    old = load_module(path, "bss_old")
    bss.RULES.update({r: r not in (RULES if off is None else off) for r in RULES})
    b = bss.build_gazetteer_sk(linker.backbone, linker.majo, linker.parts, earlier=None)
    ob = old.build_gazetteer_sk(linker.backbone, linker.majo, linker.parts, gista=bss.load_gista())   # its own
    # GISTA_DIR resolves from wherever the old file lies
    out = {}
    for d, f in frames.items():
        a = bss.add_birth_settlement_sk_columns(f, b, backbone, d)
        o = old.add_birth_settlement_sk_columns(f, ob, backbone, d)
        out[d] = int(changed(a, o, cols or LINK_COMPARED[:3]).sum())
    return out


def rgh_baseline_152(path: Path, rgh, frames: dict, shipped_on: set, ctx: tuple) -> dict:
    """Rows where a records_geo_hu 1.5.2 file (*path*, at its defaults) and this one with C3q and S2g off and the

    old = load_module(path, "rgh_152")

                               [f"{p}_{c}" for c in COMPARED] + [f"{p}_jaras_twin", f"{p}_jaras_town_okres"]).sum())
                for d, f in frames[p].items()} for p in frames}
