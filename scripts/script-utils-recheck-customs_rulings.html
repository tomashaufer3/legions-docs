How it works

One glyph — a circle on a hook — prints six offices: a dot inside is customs (in brackets, on the Mähren and Schlesien key, the customs expositure), empty is the revenue office (in brackets the cadastre office), hatched is the revenue guard (in brackets technical revenue control). The readers mix them up, and the author ruled them by eye on three pages (customs_cards), one question per volume and glyph — printed, absent or cannot tell — about the entry each volume prints:

  • Customs Dot Rulings (review_readback/customs/): German and Czech Böhmen where they disagreed on customs or the revenue office; 35 cards, four put on the page again (-r3) when their first pictures showed the wrong line.
  • Customs Rulings Mähren Schlesien (review_readback/customs_ms/): every municipality whose table holds customs or the revenue office, or where some reader read either; 94 cards, one volume each.
  • Revenue Glyph Rulings (review_readback/revenue_family/): the cadastre office, the revenue guard and technical control — Böhmen where the editions disagreed, Mähren and Schlesien where the readers did; 110 cards.

stage writes them into one review_readback/revenue_rulings.csv, a row per card, volume and glyph. apply lays them over merge's tables as the author's word on the whole municipality in the section the card showed:

  • printed, and no part holds the glyph: it is added on the municipality's own row (settlement_seq 0) on the first page the card showed;
  • absent: the glyph is removed from every part of the municipality there — the pictogram only; the same office in words (Hauptzollamt, hlavní celní úřad) is a text code and stays;
  • cannot tell changes nothing; nor does a card marked not the same place / wrong entry, unless its id is in confirmed, nor a volume whose pictures spanned a post-1910 court's section and its 1910 court's (mixed: Buschullersdorf's first Czech picture was Neustadt's no. 4, D-208).

Two answers were corrected in chat (2026-11-10, CHAT), each on a card whose margin bar or missing answer the page got wrong: Mährisch Altstadt 17 prints customs (a dotted circle with II; the bar sat on line 18), Kyjov 8 prints no customs (its (Q) is the cadastre office; left unanswered).

Card numbers are those the page prints, so merge runs this between the German moves and the post-1910 renumbering of its keys step (D-208).

def apply(t, edition: str, staged: Path = STAGED, confirmed: frozenset = frozenset()) -> dict:
    """The rulings laid over merge's ``Tables`` ``t``; returns what changed."""
    report = {"added": 0, "removed": 0, "held": 0, "no_section": []}
    for r in rulings(edition, staged, confirmed):
        section = _section(t, r)
        if not section:
            report["no_section"].append(f"{r['card']} {r['glyph']}")
            continue
        page, label = section[0]["page"], section[0]["district"]
        codes = GLYPHS[r["glyph"]][0]
        who = f"you ({r['page_set']} page, card {r['card']})"
        holding = [x for x in t.rows["inventory"] if x in section and x["code"] in codes]
        if r["verdict"] == "printed":
            if holding:
                report["held"] += 1
            elif t.add(STEP, "inventory", (page, label, r["muni_seq"], "0"), r["add_code"],
                       f"{who}: printed for the municipality"):
                report["added"] += 1
        else:
            for x in holding:
                if t.remove(STEP, "inventory", (x["page"], x["district"], str(x["muni_seq"]), str(x["settlement_seq"])),
                            x["code"], f"{who}: not printed for the municipality"):
                    report["removed"] += 1
    return report

def _read(path: Path) -> list[dict]:
    with open(path, encoding="utf-8", newline="") as fh:
        return list(csv.DictReader(fh))


def _keys(cards: Path = CARDS) -> dict[str, dict]:
    """Every card's key, all pages and rounds (the first file naming a card wins)."""
    out = {}
    for path in sorted(Path(cards).glob("**/key.csv")):
        for r in _read(path):
            out.setdefault(r["card"], r)

def _number(value) -> str:
    """A key's number as the page prints it (``"9.0"`` → ``"9"``: a pandas join wrote some as floats)."""
    n = int(float(value))
    if n >= 1000:
        raise ValueError(f"no. {n}: a card number is a printed one, never a post-1910 key (D-208)")
    return str(n)

def _latest(folder: Path) -> list[dict]:
    """The page's documents; a card put on the page again (``-r<n>``) replaces its earlier ruling."""
    latest = {}
    for path in sorted((folder / "rulings").glob("*.json")):
        doc = json.loads(path.read_text("utf-8"))
        base, _, rnd = doc["id"].partition("-r")
        if base not in latest or int(rnd or 0) > latest[base][0]:
            latest[base] = (int(rnd or 0), doc)
    return [doc for _rnd, doc in latest.values()]


def _row(page_set, card, edition, court, district, muni, pages, glyph, verdict, not_same, mixed, note) -> dict:
    verdict = CHAT.get(card, {}).get(glyph, verdict)
    if card in CHAT and glyph in CHAT[card]:

    add = GLYPHS[glyph][1]
    if glyph == "customs" and "bracket" in note.lower():

            "muni_seq": _number(muni), "pages": pages, "glyph": glyph, "verdict": verdict, "add_code": add,

def _section(t, r: dict) -> list[dict]:
    """The rows of the ruled municipality in the section the card showed: its court and number on
    those pages — under the card's own label where it names one (Mähren, Schlesien), else under the
    label most of them carry, never a post-1910 court's (whose own numbers share the pages, D-208)."""
    from utils.recheck import section_keys as sk
    court = "" if r["court_key"].startswith("unresolved:") else r["court_key"]
    rows = [x for table in ("inventory", "types") for x in t.rows[table]
            if str(x.get("court_id", "")) == court and str(x["muni_seq"]) == r["muni_seq"] and x["page"] in r["pages"]]
    if r.get("district"):
        return [x for x in rows if x["district"] == r["district"]]
    later = sk.post1910_labels("cz") | sk.post1910_labels("de-bohemia")
    labels = [x["district"] for x in rows if sk.lxnt.normalise(x["district"]) not in later]
    if not labels:
        return []
    label = max(set(labels), key=labels.count)
    return [x for x in rows if x["district"] == label]


def apply(t, edition: str, staged: Path = STAGED, confirmed: frozenset = frozenset()) -> dict:
    """The rulings laid over merge's ``Tables`` ``t``; returns what changed."""
    report = {"added": 0, "removed": 0, "held": 0, "no_section": []}
    for r in rulings(edition, staged, confirmed):
        section = _section(t, r)
        if not section:
            report["no_section"].append(f"{r['card']} {r['glyph']}")
            continue
        page, label = section[0]["page"], section[0]["district"]
        codes = GLYPHS[r["glyph"]][0]
        who = f"you ({r['page_set']} page, card {r['card']})"
        holding = [x for x in t.rows["inventory"] if x in section and x["code"] in codes]
        if r["verdict"] == "printed":
            if holding:
                report["held"] += 1
            elif t.add(STEP, "inventory", (page, label, r["muni_seq"], "0"), r["add_code"],
                       f"{who}: printed for the municipality"):
                report["added"] += 1
        else:
            for x in holding:
                if t.remove(STEP, "inventory", (x["page"], x["district"], str(x["muni_seq"]), str(x["settlement_seq"])),
                            x["code"], f"{who}: not printed for the municipality"):
                    report["removed"] += 1
    return report
