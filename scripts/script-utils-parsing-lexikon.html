How it works

The register is a narrative: each entry is keyed by an ALL-CAPS modern Czech headword and lists the settlement's status + okres for every census year, e.g. ::

BLATEC v r. 1869-1910 obec v okr. Olomouc, v r. 1921-1930 obec v okr. Olomouc-venkov, …
ABERTAMY v r. 1869-1930 pod názvem Abertham město v okr. Nejdek, …

For each entry we extract the segment whose year range contains 1910 (note 1900-1950 covers 1910 without the literal year) and read off:

  • cs_name the headword — the modern Czech settlement name;
  • name_1910 the pod názvem … form used in 1910 — the historical name, which for Sudeten settlements is the German exonym (Abertham, Adolfsgrün, Olbersdorf) and elsewhere an older Czech spelling;
  • okres_1910 the 1910 political okres (period name), mapped to okres_id via utils.parsing.records_geo (exact okres/court names + fuzzy);
  • status_1910 obec (independent commune) / osada (hamlet) / cast (part-of), with parent_1910 the parent commune where the settlement was dependent.

Cross-reference stubs (HEADWORD viz Target (Okres)) are returned separately as an alias table (alternative/older spelling → canonical name) — useful for resolving record spellings.

Because the register tracks the okres for every census year, the resulting settlement → okres_1910 map also recovers records whose birth_okres is a post-1910 administrative label (e.g. Praha-venkov, Kralupy nad Vltavou): resolve the settlement, not the anachronistic okres. That second pass lives in utils.parsing.records_geo.

from utils.parsing import lexikon as lx
settlements, aliases = lx.build_settlement_crosswalk(ABC_PDF, districts_1910)
lx.lexikon_report(settlements, aliases)

def build_settlement_crosswalk(pdf_path, districts: pd.DataFrame
                               ) -> tuple[pd.DataFrame, pd.DataFrame]:
    """Return ``(settlements, aliases)``.

    ``settlements`` — full entries with ``okres_id`` resolved from ``okres_1910`` (reusing the
    :mod:`records_geo` okres/court/fuzzy matcher) plus ``land`` from the backbone.
    ``aliases`` — cross-reference stubs (``cs_name`` alt spelling → ``viz_target`` canonical).
    """
    df = parse_lexikon(pdf_path)
    have = {str(n).strip().lower() for n in df.loc[df["kind"] == "entry", "cs_name"]}
    extra = [{k: v for k, v in r.items() if k != "force"} for r in _CURATED_EXTRA
             if r.get("force") or r["cs_name"].lower() not in have]
    if extra:                                            # e.g. Praha — absent from the register
        df = pd.concat([df, pd.DataFrame(extra)], ignore_index=True)
    aliases = df[df["kind"] == "alias"][["cs_name", "viz_target", "viz_okres"]]
    # the other names an entry's headword carries become alias rows pointing at it
    alt = df[(df["kind"] == "entry") & df["alt_names"].notna()] if "alt_names" in df else df.iloc[:0]
    alt_rows = [{"cs_name": a, "viz_target": h, "viz_okres": None}
                for h, names in zip(alt["cs_name"], alt["alt_names"]) for a in names.split("|")]
    aliases = pd.concat([aliases, pd.DataFrame(alt_rows, columns=aliases.columns)],
                        ignore_index=True)
    settlements = df[df["kind"] == "entry"].drop(
        columns=[c for c in ("viz_target", "viz_okres", "alt_names") if c in df]).copy()

    # fill the curated okres on headwords the register left without one (statutory cities);
    # only where it is missing, so a row the register does describe is never overwritten
    key = settlements["cs_name"].astype(str).str.strip().str.lower()
    fill = key.map(_CURATED_OKRES).where(settlements["okres_1910"].isna())
    if fill.notna().any():
        settlements["okres_1910"] = settlements["okres_1910"].fillna(fill)

        def _prepend(timeline, okres):        # the timeline gate reads okres_all, so the
            if not isinstance(okres, str):    # curated okres has to appear there as well
                return timeline
            parts = [p for p in str(timeline or "").split("|") if p]
            return "|".join(dict.fromkeys([okres, *parts]))

        settlements["okres_all"] = [_prepend(o, f)
                                    for o, f in zip(settlements["okres_all"], fill)]
        settlements.loc[fill.notna() & settlements["status_1910"].isna(),
                        "status_1910"] = "obec"

    # okres_1910 → okres_id via the shared matcher (exact okres + court names + fuzzy); the
    # register's own period names for a qualified okres are read as that okres first, but only on
    # rows whose timeline names it. okres_1910 itself keeps what the register prints.
    settlements["_okres_lookup"] = [_register_okres(o, t) for o, t
                                    in zip(settlements["okres_1910"], settlements["okres_all"])]
    settlements = rg.add_birth_okres_id(settlements, districts, okres_col="_okres_lookup")
    settlements = settlements.drop(columns="_okres_lookup").rename(
        columns={"birth_okres_id": "okres_id", "birth_okres_match": "okres_match"})
    # attach crown land from the backbone, keyed by okres_id
    okres_land = {
        int(r["okres_code"].split("-")[0]) * 1000 + int(r["okres_code"].split("-")[1]): r["land"]
        for _, r in districts.drop_duplicates("okres_code").iterrows()
    }
    settlements["land"] = settlements["okres_id"].map(okres_land)

    cols = ["cs_name", "name_1910", "okres_1910", "okres_id", "okres_match",
            "land", "status_1910", "parent_1910", "okres_all"]
    return settlements[cols].reset_index(drop=True), aliases

def lexikon_report(settlements: pd.DataFrame, aliases: pd.DataFrame) -> None:
    """Print coverage of the settlement crosswalk."""
    n = len(settlements)
    with_okres = settlements["okres_1910"].notna().sum()
    mapped = settlements["okres_id"].notna().sum()
    renamed = settlements["name_1910"].notna().sum()
    print(f"Settlement crosswalk (Historický lexikon)  settlements = {n}, aliases = {len(aliases)}")
    print(f"  with 1910 okres:        {with_okres}  ({with_okres/n*100:.1f}%)")
    print(f"  okres_1910 -> okres_id: {mapped}  ({mapped/with_okres*100:.1f}% of those)")
    print(f"  with 1910-era name (German exonym / old spelling): {renamed}")
    period = [isinstance(o, str) and _register_okres(o, t) != o for o, t
              in zip(settlements["okres_1910"], settlements["okres_all"])]
    print(f"  read through the register's period okres names (D-142): {sum(period)}")
    print(f"  status: {settlements['status_1910'].value_counts(dropna=False).to_dict()}")

"""
lexikon.py — settlement crosswalk from the Historický lexikon obcí ČR 1869–2011
===============================================================================
Parses the ČSÚ *Historický lexikon obcí* alphabetical register (the source PDF
``library/lexicons/CSU_Historicky-lexikon-obci-CR_1869-2011_abecedni-prehled.pdf``;
path passed in by the notebook) into a settlement-level crosswalk keyed to the 1910
political okres.

The register is a narrative: each entry is keyed by an ALL-CAPS **modern** Czech headword and
lists the settlement's status + okres for every census year, e.g. ::

    BLATEC v r. 1869-1910 obec v okr. Olomouc, v r. 1921-1930 obec v okr. Olomouc-venkov, …
    ABERTAMY v r. 1869-1930 pod názvem Abertham město v okr. Nejdek, …

For each entry we extract the segment whose **year range contains 1910** (note ``1900-1950``
covers 1910 without the literal year) and read off:

* ``cs_name``      the headword — the modern Czech settlement name;
* ``name_1910``    the ``pod názvem …`` form used in 1910 — the **historical name**, which for
                   Sudeten settlements is the German exonym (``Abertham``, ``Adolfsgrün``,
                   ``Olbersdorf``) and elsewhere an older Czech spelling;
* ``okres_1910``   the 1910 political okres (period name), mapped to ``okres_id`` via
                   :mod:`utils.parsing.records_geo` (exact okres/court names + fuzzy);
* ``status_1910``  ``obec`` (independent commune) / ``osada`` (hamlet) / ``cast`` (part-of),
                   with ``parent_1910`` the parent commune where the settlement was dependent.

Cross-reference stubs (``HEADWORD viz Target (Okres)``) are returned separately as an **alias**
table (alternative/older spelling → canonical name) — useful for resolving record spellings.

Because the register tracks the okres for *every* census year, the resulting
``settlement → okres_1910`` map also recovers records whose ``birth_okres`` is a **post-1910**
administrative label (e.g. *Praha-venkov*, *Kralupy nad Vltavou*): resolve the settlement, not
the anachronistic okres. That second pass lives in :mod:`utils.parsing.records_geo`.

Usage
-----
    from utils.parsing import lexikon as lx
    settlements, aliases = lx.build_settlement_crosswalk(ABC_PDF, districts_1910)
    lx.lexikon_report(settlements, aliases)
"""

from __future__ import annotations

import re
from pathlib import Path

import pandas as pd

from utils.parsing import records_geo as rg

try:
    from pypdf import PdfReader
except ImportError as exc:  # pragma: no cover
    raise ImportError("lexikon.py requires pypdf (`pip install pypdf`).") from exc

__version__ = "1.6.0"   # OCR-split years and "okr ." healed; Moravian "okr. Beroun" (2026-10-28)

# Settlements the abc register does not carry as headwords — notably the capital: Prague is not
# an "obec" entry there, which left every bare "Praha" birth (no okres text; ~1.8k VÚA records)
# unresolvable on the Czech side and exposed to the like-named Slovak village (okres Halič).
# Appended as curated rows; okres_1910 resolves through the normal okres matcher.
# ``force`` appends the row even when the register already carries that headword — needed when
# the register's entry of that name is a *different place*: TĚŠÍN there is an osada of Železnice
# in okres Jičín, while the town of Teschen (okres Těšín, ceded to Poland in 1920 and so absent
# from a modern Czech register) is what a 1910 record means by "Těšín".
_CURATED_EXTRA: list[dict] = [
    {"cs_name": "Praha", "kind": "entry", "viz_target": None, "viz_okres": None,
     "name_1910": "Prag", "okres_1910": "Praha", "status_1910": "obec",
     "parent_1910": None, "okres_all": "Praha"},
    {"cs_name": "Těšín", "kind": "entry", "viz_target": None, "viz_okres": None,
     "name_1910": "Teschen", "okres_1910": "Těšín", "status_1910": "obec",
     "parent_1910": None, "okres_all": "Těšín", "force": True},
    # Bielitz, the Silesian statutory city (okres 303-2), ceded to Poland in 1920 and so absent
    # from a modern Czech register for the same reason as Těšín. Without it the three Czech
    # hamlets named BÍLSKO — okr. Jičín, Litovel and Písek — are the only candidates, and a
    # Bielitz birth acquires a hamlet's coordinates ~180 km away.
    {"cs_name": "Bílsko", "kind": "entry", "viz_target": None, "viz_okres": None,
     "name_1910": "Bielitz", "okres_1910": "Bílsko (město)", "status_1910": "obec",
     "parent_1910": None, "okres_all": "Bílsko (město)", "force": True},
]

# Headwords the register DOES carry but leaves without an okres — a different failure from the
# one above, and a worse one. A *Statutarstadt* was its own political district, so the register's
# okres column is empty for it; downstream, `records_geo` indexes only settlements that HAVE an
# okres, so the city vanishes and a like-named hamlet is left as the "unique" candidate. That is
# how every bare "Brno" birth landed in okres Jilemnice (an osada of Bratrouchov) — 453 VÚA +
# 3 legion records — and every Brno-born legionnaire (775) ended up with no coordinates at all.
#
# Keyed lower-cased headword -> the 1910 okres string, filled ONLY where okres_1910 is missing;
# the string resolves through the same okres matcher as every other row. For a statutory city
# that is the CITY district — always the ``X (město)`` unit, never the rural ring, whose okres
# the register writes **bare** (every ring village around Brno records ``okr. Brno``, which is
# okres 302-4). One entry is not a statutory city but the same shape of gap: Teplá, whose only
# okres-bearing row is a different Teplá in okres Litoměřice.
_CURATED_OKRES: dict[str, str] = {
    "brno": "Brno (město)",                           # 302-5  Statutarstadt Brünn
    "frýdek": "Frýdek (město)",                       # 303-7
    "jihlava": "Jihlava (město)",                     # 302-13
    "kroměříž": "Kroměříž (město)",                   # 302-15
    "liberec": "Liberec (město)",                     # 301-80
    "olomouc": "Olomouc (město)",                     # 302-28
    "opava": "Opava (město)",                         # 303-11
    "uherské hradiště": "Uherské Hradiště (město)",   # 302-37
    "znojmo": "Znojmo (město)",                       # 302-42
    "teplá": "Teplá",                                 # 301-97
}
# The register's 1910 name for a Moravian okres that the backbone qualifies. Every row whose
# 1910 segment reads bare ``okr. Beroun`` is Moravian — all 43 carry ``Moravský Beroun`` later in
# their own timeline (BARNOV ``Šternberk|Beroun|Moravský Beroun``) — while the Bohemian town is
# filed ``v r. 1869-1930 obec v okr. Hořovice``: there was no Bohemian okres Beroun in 1910, only a
# court of that name inside okres Hořovice. The shared matcher resolves bare "Beroun" to that
# court, so the 43 villages sat in okres Hořovice. Applied only where the row's timeline names the
# qualified okres, so the rule cannot reach a row the register does not itself corroborate.
# Keyed space-free (the text layer writes ``Berou n`` on LESY). See DECISIONS D-142.
_REGISTER_OKRES_1910: dict[str, str] = {
    "beroun": "Moravský Beroun",                      # 302-2
}
# Not here: Těšín, which the register does not carry at all (see _CURATED_EXTRA above), and
# Horšovský Týn, whose row *has* an okres — the register's 1910 spelling "Horšův Týn", which
# the backbone does not know. That one belongs in records_geo.ALIAS_OVERRIDE, where it also
# rescues every record that writes the old spelling as its own district.

# ALL-CAPS headword alphabet: Czech letters **plus the German umlauts** the register's
# Sudeten-exonym headwords carry (ADLERHÜTTE, ADOLFSGRÜN, ALTBÜRGERSDORF). Omitting ÄÖÜ
# made every such headword fail to start an entry, so its text was appended to the
# PREVIOUS entry and corrupted that entry's okres timeline (531 entries; audit 2026-07-16).
_UP = "A-ZÁČĎÉĚÍŇÓŔŘŠŤÚŮÝŽÄÖÜ"
_DIV = set("A B C Č D E F G H CH I J K L M N O P Q R Ř S Š T U V W X Y Z Ž".split())
# entry start: ALL-CAPS headword, then a separator — "v r." (dated segment), "od r."
# (open-ended "from year X onwards"; 451 entries, previously unrecognised and bled into
# their predecessor), or "viz" (cross-reference stub).
#
# The headword may carry (v1.5.0) a **mixed-case parenthetical** — the other name of the
# place, "BADERWINKEL (Větrov)", "BARNOV (Olověná)", or a note, "BOBROVNÍKY (do r. 1920
# součást pruského Slezska)" — and a **``t.`` alternate** ("ANTOŇOV t. ANTONÍNOV"), and the
# PDF sometimes spaces the separator apart ("v r . 1869"). The all-caps character class
# admitted none of these, so 116 entries never started: each vanished and its text was
# appended to the previous settlement, whose ``okres_all`` timeline then carried a foreign
# okres (ALBRECHTIČKY gained Prachatice) — the D1 shape again (audit 2026-09-22 §B23).
# Groups: 1 headword · 2 parenthetical · 3 ``t.`` alternate · 4 parenthetical after the
# alternate · 5 separator (spacing inside it tolerated: "v  r.", "r.1869").
_HEAD = re.compile(
    rf"^([{_UP}][{_UP}0-9 .()/-]*?)"
    rf"(\s*\([^)]*\))?"
    rf"(\s+t\.\s+[{_UP}][{_UP}0-9 .-]*?)?"
    rf"(\s*\([^)]*\))?"
    rf"\s+(v\s*r\s*\.|od\s*r\s*\.|viz)(?=\s|\d)")


# ── PDF → entries ─────────────────────────────────────────────────────────────
def _extract_text(pdf_path) -> str:
    reader = PdfReader(str(pdf_path))
    return "\n".join((pg.extract_text() or "") for pg in reader.pages)


def _entries(text: str) -> list[str]:
    """Split the register text into whole entries (accumulating wrapped lines)."""
    out: list[str] = []
    cur: str | None = None
    for line in text.splitlines():
        s = line.strip()
        if not s or s in _DIV:
            continue
        m = _HEAD.match(s)
        is_head = bool(m) and len([c for c in m.group(1) if c.isalpha()]) >= 2
        if is_head:
            if cur:
                out.append(cur)
            cur = s
        elif cur is not None:
            cur += " " + s
    if cur:
        out.append(cur)
    return out


# An okres clause can run on into narrative. A clause is split on commas, but the sub-clauses
# describing a town's internal division are separated by **semicolons**, so the register's entry
# for Brno reads ``okr. Brno-město; v r. 1869-1890 se člení na 4 městské obvody (I.-IV. Bezirk);
# v r. 1900-1910 se nečlení na části; …`` and a greedy ``(.+)$`` swallowed the lot. Everything
# from the first ``;`` is narrative and never part of the district name.
#
# This mattered twice over: the blob landed in ``okres_1910`` *and* in ``okres_all``, whose last
# entry is the key :func:`utils.parsing.wikidata.add_settlement_coords` corroborates against — so
# BRNO's key was that whole sentence and could never equal Wikidata's ``Brno-město``. Fifteen
# headwords carried the pollution, all of them towns big enough to be divided into osady.
_OKRES_RE = re.compile(r"okr\.\s*(.+)$")


# The text layer also breaks the tokens the 1910 segment is found by, and an entry whose year or
# ``okr.`` is broken loses its 1910 okres without any error: ``v r. 186 9-1910 obec v okr.
# Budějovice`` (RUDOLFOV), ``v r. 1 869-1910`` (STRAŠKOV), ``v r. 1869-1930 obec v okr . Hradec
# Králové`` (KUKLENY), ``obec v ok r. Příbram`` (PODLESÍ). 182 entries, among them the Prague suburb
# MICHLE and LÁZNĚ BOHDANEČ. A year is healed only in the slots the register prints one — after
# ``r.`` and after the dash of a range — so a day-dated clause (``od 1.1.1980``) is never touched;
# on the register as read, the heal changes no okres that was already found (2026-10-28, D-143).
_YEAR_AFTER_R = re.compile(r"(?<=r\. )(\d)\s?(\d)\s?(\d)\s?(\d)(?=\D)")
_YEAR_AFTER_DASH = re.compile(r"(-\s*)(\d)\s?(\d)\s?(\d)\s?(\d)(?=\D)")
_OKR_BROKEN = re.compile(r"\bo\s?k\s?r\s*\.")


def _heal_ocr(body: str) -> str:
    """Rejoin the years and the ``okr.`` the PDF text layer split apart."""
    body = _YEAR_AFTER_R.sub(r"\1\2\3\4", body)
    body = _YEAR_AFTER_DASH.sub(r"\1\2\3\4\5", body)
    return _OKR_BROKEN.sub("okr.", body)


def _okres_text(clause: str) -> str | None:
    """The okres name in one register clause, cut free of any trailing narrative."""
    mo = _OKRES_RE.search(clause)
    if not mo:
        return None
    return mo.group(1).split(";")[0].strip() or None


def _alt_names(m: "re.Match[str]") -> str | None:
    """The other names a headword carries — a name-like parenthetical and a ``t.`` alternate.

    A parenthetical with a digit in it is a note ("do r. 1920 součást pruského Slezska"), not
    a name, and is dropped.
    """
    alts = []
    for g in (2, 4):
        par = (m.group(g) or "").strip()[1:-1].strip() if m.group(g) else ""
        if par and not re.search(r"\d", par) and par[:1].isupper():
            alts.append(par)
    alt = (m.group(3) or "").strip()
    if alt:
        alts.append(re.sub(r"^t\.\s+", "", alt).strip())
    return "|".join(alts) or None


def _parse_entry(entry: str) -> dict:
    m = _HEAD.match(entry)
    head, sep = m.group(1).strip(), re.sub(r"\s", "", m.group(5))
    alt_names = _alt_names(m)
    if sep == "viz":
        mm = re.match(r"^\s*(.+?)(?:\s*\(([^)]+)\))?\s*$", entry[m.end(5):])
        return {"cs_name": head, "kind": "alias",
                "viz_target": mm.group(1).strip() if mm else None,
                "viz_okres": (mm.group(2).strip() if mm and mm.group(2) else None),
                "name_1910": None, "okres_1910": None,
                "status_1910": None, "parent_1910": None, "alt_names": alt_names}

    body = re.sub(r"\b(v|od)\s*r\s*\.\s*", r"\1 r. ", entry[m.start(5):])  # heal "v  r .1869"
    body = _heal_ocr(body)                                                # "186 9", "okr ."
    body = re.sub(r"(\d{4})\s*-\s*(\d{4})", r"\1-\2", body)               # heal "1900 - 1910"
    okres = name = status = parent = None
    okres_all: list[str] = []                       # every okres this settlement ever sat in
    for clause in (c.strip() for c in body.split(",")):
        okres_txt = _okres_text(clause)
        if okres_txt:
            okres_all.append(okres_txt)             # collect okres from *all* year segments
        # "v r. 1869-1910" = that span; "v r. 1900" = that single year;
        # "od r. 1869" = from 1869 onwards (open-ended -> covers 1910).
        cm = re.match(r"(v|od) r\.\s*(\d{4})(?:-(\d{4}))?", clause)
        if not cm or okres is not None:             # only fill the 1910 fields once
            continue
        y0 = int(cm.group(2))
        y1 = int(cm.group(3)) if cm.group(3) else (9999 if cm.group(1) == "od" else y0)
        if not (y0 <= 1910 <= y1):
            continue
        okres = okres_txt
        mn = re.search(r"pod názvem\s+(.+?)\s+(?:obec|osada|část|město|v okr\.)", clause)
        if mn:
            name = re.split(r"\s+t\.\s+", mn.group(1).strip())[0].strip()  # drop "též …" alt
        mp = re.search(r"(osada|část) obce\s+(.+?)\s+v okr\.", clause)
        if mp:
            status = {"osada": "osada", "část": "cast"}[mp.group(1)]
            parent = mp.group(2).strip()
        elif re.search(r"\b(obec|město)\b", clause):
            status = "obec"
    return {"cs_name": head, "kind": "entry", "viz_target": None, "viz_okres": None,
            "name_1910": name, "okres_1910": okres, "status_1910": status, "parent_1910": parent,
            "okres_all": "|".join(dict.fromkeys(okres_all)), "alt_names": alt_names}


# ── Public API ────────────────────────────────────────────────────────────────
def parse_lexikon(pdf_path) -> pd.DataFrame:
    """Parse the register PDF → one row per entry (``kind`` = ``entry`` / ``alias``)."""
    text = _extract_text(pdf_path)
    return pd.DataFrame([_parse_entry(e) for e in _entries(text)])


def _register_okres(okres_1910, okres_all) -> object:
    """The okres string the matcher should read for one row (:data:`_REGISTER_OKRES_1910`)."""
    if not isinstance(okres_1910, str):
        return okres_1910
    target = _REGISTER_OKRES_1910.get(rg._norm(okres_1910).replace(" ", ""))
    if target is None:
        return okres_1910
    timeline = {rg._norm(t).replace(" ", "") for t in str(okres_all or "").split("|")}
    return target if rg._norm(target).replace(" ", "") in timeline else okres_1910


def build_settlement_crosswalk(pdf_path, districts: pd.DataFrame
                               ) -> tuple[pd.DataFrame, pd.DataFrame]:
    """Return ``(settlements, aliases)``.

    ``settlements`` — full entries with ``okres_id`` resolved from ``okres_1910`` (reusing the
    :mod:`records_geo` okres/court/fuzzy matcher) plus ``land`` from the backbone.
    ``aliases`` — cross-reference stubs (``cs_name`` alt spelling → ``viz_target`` canonical).
    """
    df = parse_lexikon(pdf_path)
    have = {str(n).strip().lower() for n in df.loc[df["kind"] == "entry", "cs_name"]}
    extra = [{k: v for k, v in r.items() if k != "force"} for r in _CURATED_EXTRA
             if r.get("force") or r["cs_name"].lower() not in have]
    if extra:                                            # e.g. Praha — absent from the register
        df = pd.concat([df, pd.DataFrame(extra)], ignore_index=True)
    aliases = df[df["kind"] == "alias"][["cs_name", "viz_target", "viz_okres"]]
    # the other names an entry's headword carries become alias rows pointing at it
    alt = df[(df["kind"] == "entry") & df["alt_names"].notna()] if "alt_names" in df else df.iloc[:0]
    alt_rows = [{"cs_name": a, "viz_target": h, "viz_okres": None}
                for h, names in zip(alt["cs_name"], alt["alt_names"]) for a in names.split("|")]
    aliases = pd.concat([aliases, pd.DataFrame(alt_rows, columns=aliases.columns)],
                        ignore_index=True)
    settlements = df[df["kind"] == "entry"].drop(
        columns=[c for c in ("viz_target", "viz_okres", "alt_names") if c in df]).copy()

    # fill the curated okres on headwords the register left without one (statutory cities);
    # only where it is missing, so a row the register does describe is never overwritten
    key = settlements["cs_name"].astype(str).str.strip().str.lower()
    fill = key.map(_CURATED_OKRES).where(settlements["okres_1910"].isna())
    if fill.notna().any():
        settlements["okres_1910"] = settlements["okres_1910"].fillna(fill)

        def _prepend(timeline, okres):        # the timeline gate reads okres_all, so the
            if not isinstance(okres, str):    # curated okres has to appear there as well
                return timeline
            parts = [p for p in str(timeline or "").split("|") if p]
            return "|".join(dict.fromkeys([okres, *parts]))

        settlements["okres_all"] = [_prepend(o, f)
                                    for o, f in zip(settlements["okres_all"], fill)]
        settlements.loc[fill.notna() & settlements["status_1910"].isna(),
                        "status_1910"] = "obec"

    # okres_1910 → okres_id via the shared matcher (exact okres + court names + fuzzy); the
    # register's own period names for a qualified okres are read as that okres first, but only on
    # rows whose timeline names it. okres_1910 itself keeps what the register prints.
    settlements["_okres_lookup"] = [_register_okres(o, t) for o, t
                                    in zip(settlements["okres_1910"], settlements["okres_all"])]
    settlements = rg.add_birth_okres_id(settlements, districts, okres_col="_okres_lookup")
    settlements = settlements.drop(columns="_okres_lookup").rename(
        columns={"birth_okres_id": "okres_id", "birth_okres_match": "okres_match"})
    # attach crown land from the backbone, keyed by okres_id
    okres_land = {
        int(r["okres_code"].split("-")[0]) * 1000 + int(r["okres_code"].split("-")[1]): r["land"]
        for _, r in districts.drop_duplicates("okres_code").iterrows()
    }
    settlements["land"] = settlements["okres_id"].map(okres_land)

    cols = ["cs_name", "name_1910", "okres_1910", "okres_id", "okres_match",
            "land", "status_1910", "parent_1910", "okres_all"]
    return settlements[cols].reset_index(drop=True), aliases


def lexikon_report(settlements: pd.DataFrame, aliases: pd.DataFrame) -> None:
    """Print coverage of the settlement crosswalk."""
    n = len(settlements)
    with_okres = settlements["okres_1910"].notna().sum()
    mapped = settlements["okres_id"].notna().sum()
    renamed = settlements["name_1910"].notna().sum()
    print(f"Settlement crosswalk (Historický lexikon)  settlements = {n}, aliases = {len(aliases)}")
    print(f"  with 1910 okres:        {with_okres}  ({with_okres/n*100:.1f}%)")
    print(f"  okres_1910 -> okres_id: {mapped}  ({mapped/with_okres*100:.1f}% of those)")
    print(f"  with 1910-era name (German exonym / old spelling): {renamed}")
    period = [isinstance(o, str) and _register_okres(o, t) != o for o, t
              in zip(settlements["okres_1910"], settlements["okres_all"])]
    print(f"  read through the register's period okres names (D-142): {sum(period)}")
    print(f"  status: {settlements['status_1910'].value_counts(dropna=False).to_dict()}")
