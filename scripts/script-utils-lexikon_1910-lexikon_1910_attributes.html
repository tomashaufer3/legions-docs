How it works

The offices inventory records that a part of a municipality had a school or a cooperative. The print says more, in plain type beside the pictogram, and that is where two of the variables a regression will use actually live::

5 [school] (1 č., 4 něm.) XXVI      five schools, one Czech and four German,
                                    twenty-six classes between them
2 měšť. šk. VI                      two burgher schools, six classes
[cooperative]: Schč, 3 K, Rn        a Czech Schulze-Delitzsch credit union,
                                    three consumer cooperatives, a German
                                    Raiffeisen one

Nothing scored these before 2026-10-06: lexikon_1910_institutions. score_against_reference sees the glyph and stops. This module reads the text that follows it, with one parser for the reader and for the gold sheet, so a difference between the two is a difference in what was transcribed and never in how it was parsed — the rule split_parts already follows for Části osad.

Three things about the source decide the parsing:

  • The numeral after a school is its classes, not its count. The count stands before the glyph (2 [school]); the Roman numeral after it is the number of classes (Návod, p. VII). They are kept in separate columns and never summed.
  • The language note is optional and says two different things. (č. a něm.) names the languages; (1 č., 4 něm.) also splits the count between them. Where the print gives no note the column is blank — an absent note is not "Czech", it is a school the register did not qualify.
  • The cooperative's language is one small letter. č / n after the kind in the Czech edition, b / d in the German (böhmisch / deutsch). A reader that drops the háček writes c; that is read as č and the raw letter is kept in raw so the two can be told apart later.

The frame is one row per school mention and one per cooperative kind, keyed on the entry's own (page, district, municipality, part) like every other frame of this apparatus.

def roman(numeral: str) -> int | str:
    """A Roman numeral as an integer, or blank when it is not one.

    Checked by writing the value back: ``IIII`` and ``VX`` parse to a number under
    the subtractive rule and are not numerals the register prints, so they come
    back blank rather than as a class count nobody set.
    """
    numeral = (numeral or "").strip().upper()
    if not numeral or any(ch not in _ROMAN for ch in numeral):
        return ""
    total = 0
    for here, after in zip(numeral, numeral[1:] + " "):
        value = _ROMAN[here]
        total += -value if after in _ROMAN and _ROMAN[after] > value else value
    return total if _to_roman(total) == numeral else ""

def attribute_records(parts: list[tuple], source: str = "model") -> list[dict]:
    """The frame, from ``(page, district, muni_seq, settlement_seq, text)`` tuples.

    One tuple per part of a municipality, as :func:`lexikon_1910_institutions.
    parse_entries` cuts them for the reader and as the gold ledger holds them.
    ``occurrence`` numbers repeated mentions of one thing inside a part, so two
    school mentions under one key stay two rows.

    for page, district, muni_seq, settlement_seq, text in parts:
        seen: dict[tuple, int] = {}
        for found in schools(text) + cooperatives(text):
            key = (found["attribute"], found["item"])
            seen[key] = seen.get(key, 0) + 1
            rows.append({"check": CHECK, "page": page, "district": district,
                         "muni_seq": muni_seq,
                         "settlement_seq": str(settlement_seq or "0"),
                         **found, "occurrence": seen[key], "source": source})
    return [{column: row[column] for column in COLUMNS} for row in rows]

def score_attributes(reference: list[dict], rows: list[dict],
                     entries: set | None = None) -> list[dict]:
    """One row per attribute: how much of what the reference lists the pass has,
    and how often the detail beside it agrees.

    ``entries`` restricts both sides to ``(page, district, muni_seq)`` entries both
    read, the same rule the symbol scorer follows — pass the set it computed.
    A mention is a hit when both sides have it under the same part; ``count``,
    ``classes`` and ``language`` are then compared on the hits only, so a missed
    school is charged once, as a miss, and not again in every column.
    """
    def keyed(frame):
        out = {}
        for row in frame:
            entry = (str(row["page"]), str(row["district"]), str(row["muni_seq"]))
            if entries is None or entry in entries:
                out[_key(row)] = row
        return out

    ref, mine = keyed(reference), keyed(rows)
    out = []
    for attribute in sorted({key[4] for key in ref} | {key[4] for key in mine}):
        theirs = {k for k in ref if k[4] == attribute}
        ours = {k for k in mine if k[4] == attribute}
        hit = theirs & ours
        agree = lambda column: sum(  # noqa: E731
            1 for k in hit if _same(ref[k][column], mine[k][column]))
        row = {"check": CHECK, "attribute": attribute,
               "reference": len(theirs), "claimed": len(ours), "hit": len(hit),
               "false": len(ours - theirs), "missed": len(theirs - ours),
               "count_agree": agree("count"), "language_agree": agree("language")}
        row["classes_agree"] = agree("classes") if attribute != "cooperative" else ""
        out.append(row)
    return out

def lexikon_1910_attributes_report(rows: list[dict], label: str = "") -> str:
    """Ledger for :func:`score_attributes`."""
    if not rows:
        return "  no attribute in either reading"
    lines = [f"  {label + ': ' if label else ''}what stands beside the school and "
             f"cooperative glyphs, against the reference", ""]
    lines.append(f"  {'attribute':16s} {'in ref':>6s} {'found':>6s} {'false':>6s} "
                 f"{'lost':>5s} {'count ok':>9s} {'classes ok':>11s} {'language ok':>12s}")
    for row in rows:
        of = lambda n: f"{n}/{row['hit']}" if n != "" else ""  # noqa: E731
        lines.append(f"  {row['attribute']:16s} {row['reference']:6d} {row['hit']:6d} "
                     f"{row['false']:6d} {row['missed']:5d} "
                     f"{of(row['count_agree']):>9s} {of(row['classes_agree']):>11s} "
                     f"{of(row['language_agree']):>12s}")
    lines.append("")
    lines.append("  found = listed by both under the same part. The three columns on "
                 "the right are")
    lines.append("  counted on those only: a school that was missed is charged once, "
                 "as lost.")
    return "\n".join(lines)

"""What the register prints *beside* two glyphs: a school's classes and language,
a cooperative's kind and language.

The offices inventory records that a part of a municipality had a school or a
cooperative. The print says more, in plain type beside the pictogram, and that
is where two of the variables a regression will use actually live::

    5 [school] (1 č., 4 něm.) XXVI      five schools, one Czech and four German,
                                        twenty-six classes between them
    2 měšť. šk. VI                      two burgher schools, six classes
    [cooperative]: Schč, 3 K, Rn        a Czech Schulze-Delitzsch credit union,
                                        three consumer cooperatives, a German
                                        Raiffeisen one

Nothing scored these before 2026-10-06: :func:`lexikon_1910_institutions.
score_against_reference` sees the glyph and stops. This module reads the text
that follows it, with **one parser for the reader and for the gold sheet**, so a
difference between the two is a difference in what was transcribed and never in
how it was parsed — the rule ``split_parts`` already follows for *Části osad*.

Three things about the source decide the parsing:

* **The numeral after a school is its classes, not its count.** The count stands
  *before* the glyph (``2 [school]``); the Roman numeral after it is the number of
  classes (*Návod*, p. VII). They are kept in separate columns and never summed.
* **The language note is optional and says two different things.** ``(č. a něm.)``
  names the languages; ``(1 č., 4 něm.)`` also splits the count between them.
  Where the print gives no note the column is blank — an absent note is not
  "Czech", it is a school the register did not qualify.
* **The cooperative's language is one small letter.** ``č`` / ``n`` after the kind
  in the Czech edition, ``b`` / ``d`` in the German (*böhmisch* / *deutsch*). A
  reader that drops the háček writes ``c``; that is read as ``č`` and the raw
  letter is kept in ``raw`` so the two can be told apart later.

The frame is one row per school mention and one per cooperative kind, keyed on
the entry's own ``(page, district, municipality, part)`` like every other frame
of this apparatus.
"""

from __future__ import annotations

import re

__version__ = "0.1.0"

#: The check these rows belong to in the pilot's ledger.
CHECK = "I19_attributes"

#: A school mention: an optional count, the glyph or the burgher-school
#: abbreviation, an optional language note in brackets, an optional class numeral.
#: The numeral may not run on into a word, which is what keeps ``[school] Ivanka``
#: from yielding one class.
_SCHOOL_RE = re.compile(
    r"(?:(?<![\w.])(?P<count>\d{1,2})\s*)?"
    r"(?P<kind>\[school\]|m[eěĕ]š[tť]\.?\s*šk\.)\s*"
    r"(?:\((?P<note>[^()]{1,40})\)\s*)?"
    r"(?P<classes>[IVXLC]{1,8}(?![^\W\d_]))?")

#: One language inside a note: an optional count and ``č.`` or ``něm.``. The word
#: must start there, so the ``č`` that ends another abbreviation is not a language.
_LANGUAGE_RE = re.compile(r"(?:(\d{1,2})\s*)?(?<![^\W\d_])(č|něm|nem)\b\.?",
                          re.IGNORECASE)

#: The list after the cooperative glyph, up to the semicolon or full stop that
#: closes it or the next glyph.
_COOP_LIST_RE = re.compile(r"\[cooperative\]\s*:\s*(?P<list>[^;.\[\]]*)")

#: One kind in that list: count, the letter code, the language letter.
_COOP_ITEM_RE = re.compile(
    r"^\s*(?:(?P<count>\d{1,2})\s*)?(?P<kind>Sch|St|[HJKPRBGLS])\s*"
    r"(?P<language>[čcnbd])?\s*$")

#: The language letter after a cooperative's kind. ``c`` is ``č`` with its háček
#: lost in reading; ``b`` / ``d`` are the German edition's own pair.
COOP_LANGUAGE = {"č": "cs", "c": "cs", "b": "cs", "n": "de", "d": "de"}

_ROMAN = {"I": 1, "V": 5, "X": 10, "L": 50, "C": 100}

#: The columns of the frame, in order.
COLUMNS = ("check", "page", "district", "muni_seq", "settlement_seq", "attribute",
           "item", "occurrence", "count", "classes", "language", "n_czech",
           "n_german", "raw", "source")


def roman(numeral: str) -> int | str:
    """A Roman numeral as an integer, or blank when it is not one.

    Checked by writing the value back: ``IIII`` and ``VX`` parse to a number under
    the subtractive rule and are not numerals the register prints, so they come
    back blank rather than as a class count nobody set.
    """
    numeral = (numeral or "").strip().upper()
    if not numeral or any(ch not in _ROMAN for ch in numeral):
        return ""
    total = 0
    for here, after in zip(numeral, numeral[1:] + " "):
        value = _ROMAN[here]
        total += -value if after in _ROMAN and _ROMAN[after] > value else value
    return total if _to_roman(total) == numeral else ""


def _to_roman(value: int) -> str:
    out = ""
    for number, letters in ((100, "C"), (90, "XC"), (50, "L"), (40, "XL"),
                            (10, "X"), (9, "IX"), (5, "V"), (4, "IV"), (1, "I")):
        while value >= number:
            out += letters
            value -= number
    return out


def _languages(note: str) -> tuple[str, int | str, int | str]:
    """``(languages, czech count, german count)`` from a school's note."""
    counts: dict[str, int | str] = {}
    for number, word in _LANGUAGE_RE.findall(note or ""):
        code = "cs" if word.lower() == "č" else "de"
        counts[code] = int(number) if number else counts.get(code, "")
    names = "+".join(code for code in ("cs", "de") if code in counts)
    return names, counts.get("cs", ""), counts.get("de", "")


def schools(text: str) -> list[dict]:
    """Every school mention in one part's offices text."""
    rows = []
    for mention in _SCHOOL_RE.finditer(text or ""):
        languages, czech, german = _languages(mention.group("note"))
        kind = "school" if mention.group("kind").startswith("[") else "burgher_school"
        rows.append({
            "attribute": kind, "item": "",
            "count": int(mention.group("count") or 1),
            "classes": roman(mention.group("classes")),
            "language": languages, "n_czech": czech, "n_german": german,
            "raw": mention.group(0).strip()})
    return rows


def cooperatives(text: str) -> list[dict]:
    """Every cooperative kind listed after the glyph in one part's offices text.

    An item that is not ``count kind language`` is kept with ``item="?"`` and its
    text in ``raw``: a kind the parser cannot name is a reading to look at, and
    dropping it would make the list look shorter than the print.
    """
    rows = []
    for listing in _COOP_LIST_RE.finditer(text or ""):
        for piece in listing.group("list").split(","):
            if not piece.strip():
                continue
            item = _COOP_ITEM_RE.match(piece)
            rows.append({
                "attribute": "cooperative",
                "item": item.group("kind") if item else "?",
                "count": int(item.group("count") or 1) if item else 1,
                "classes": "",
                "language": COOP_LANGUAGE.get(item.group("language") or "", "")
                            if item else "",
                "n_czech": "", "n_german": "", "raw": piece.strip()})
    return rows


def attribute_records(parts: list[tuple], source: str = "model") -> list[dict]:
    """The frame, from ``(page, district, muni_seq, settlement_seq, text)`` tuples.

    One tuple per part of a municipality, as :func:`lexikon_1910_institutions.
    parse_entries` cuts them for the reader and as the gold ledger holds them.
    ``occurrence`` numbers repeated mentions of one thing inside a part, so two
    school mentions under one key stay two rows.
    """
    rows: list[dict] = []
    for page, district, muni_seq, settlement_seq, text in parts:
        seen: dict[tuple, int] = {}
        for found in schools(text) + cooperatives(text):
            key = (found["attribute"], found["item"])
            seen[key] = seen.get(key, 0) + 1
            rows.append({"check": CHECK, "page": page, "district": district,
                         "muni_seq": muni_seq,
                         "settlement_seq": str(settlement_seq or "0"),
                         **found, "occurrence": seen[key], "source": source})
    return [{column: row[column] for column in COLUMNS} for row in rows]


def entry_parts(entries) -> list[tuple]:
    """``attribute_records`` input from the decoder's own :class:`Entry` objects."""
    return [(entry.page, entry.district, entry.muni_seq,
             entry.settlement_seq or "0", entry.text) for entry in entries]


# --------------------------------------------------------------------------- #
# Scoring against a reference
# --------------------------------------------------------------------------- #

def _key(row: dict) -> tuple:
    return (str(row["page"]), str(row["district"]), str(row["muni_seq"]),
            str(row["settlement_seq"] or "0"), row["attribute"], row["item"],
            str(row["occurrence"]))


def _same(left, right) -> bool:
    return str(left) == str(right)


def score_attributes(reference: list[dict], rows: list[dict],
                     entries: set | None = None) -> list[dict]:
    """One row per attribute: how much of what the reference lists the pass has,
    and how often the detail beside it agrees.

    ``entries`` restricts both sides to ``(page, district, muni_seq)`` entries both
    read, the same rule the symbol scorer follows — pass the set it computed.
    A mention is a hit when both sides have it under the same part; ``count``,
    ``classes`` and ``language`` are then compared on the hits only, so a missed
    school is charged once, as a miss, and not again in every column.
    """
    def keyed(frame):
        out = {}
        for row in frame:
            entry = (str(row["page"]), str(row["district"]), str(row["muni_seq"]))
            if entries is None or entry in entries:
                out[_key(row)] = row
        return out

    ref, mine = keyed(reference), keyed(rows)
    out = []
    for attribute in sorted({key[4] for key in ref} | {key[4] for key in mine}):
        theirs = {k for k in ref if k[4] == attribute}
        ours = {k for k in mine if k[4] == attribute}
        hit = theirs & ours
        agree = lambda column: sum(  # noqa: E731
            1 for k in hit if _same(ref[k][column], mine[k][column]))
        row = {"check": CHECK, "attribute": attribute,
               "reference": len(theirs), "claimed": len(ours), "hit": len(hit),
               "false": len(ours - theirs), "missed": len(theirs - ours),
               "count_agree": agree("count"), "language_agree": agree("language")}
        row["classes_agree"] = agree("classes") if attribute != "cooperative" else ""
        out.append(row)
    return out


def lexikon_1910_attributes_report(rows: list[dict], label: str = "") -> str:
    """Ledger for :func:`score_attributes`."""
    if not rows:
        return "  no attribute in either reading"
    lines = [f"  {label + ': ' if label else ''}what stands beside the school and "
             f"cooperative glyphs, against the reference", ""]
    lines.append(f"  {'attribute':16s} {'in ref':>6s} {'found':>6s} {'false':>6s} "
                 f"{'lost':>5s} {'count ok':>9s} {'classes ok':>11s} {'language ok':>12s}")
    for row in rows:
        of = lambda n: f"{n}/{row['hit']}" if n != "" else ""  # noqa: E731
        lines.append(f"  {row['attribute']:16s} {row['reference']:6d} {row['hit']:6d} "
                     f"{row['false']:6d} {row['missed']:5d} "
                     f"{of(row['count_agree']):>9s} {of(row['classes_agree']):>11s} "
                     f"{of(row['language_agree']):>12s}")
    lines.append("")
    lines.append("  found = listed by both under the same part. The three columns on "
                 "the right are")
    lines.append("  counted on those only: a school that was missed is charged once, "
                 "as lost.")
    return "\n".join(lines)
