How it works

Reads a worksheet filled in per education_gold/NAVOD.md, re-parses the same raw strings with the live parser, and reports where the two disagree.

Three things this does that a naive accuracy would not

It reweights. The sample is disproportionate by design — 40 of 952 unparsed records against 90 of 71,622 plain ones. An unweighted mean over the worksheet would describe a population that does not exist. Every headline figure is weighted by the true stratum size from _frame.csv, so it estimates the accuracy the data has; the per-stratum figures are reported beside it, unweighted, because that is where the precision actually is.

It separates reading from ranking. The coder names institutions, never a rung. The rung is then derived from their institutions with the parser's own INSTITUTIONS and TIE_BREAK. So a rung disagreement is reported twice: once as a reading difference (the two disagree about which schools are named) and once as a residual — the same institutions placed differently, which would be a bug in the ladder code rather than in the dictionary.

It refuses to score what was not coded. A blank cell is "not coded", not "no". Blank rows are counted and excluded, and the counts are printed, because a worksheet that is 60 % filled produces a very confident wrong number if the blanks are read as agreement.

Usage: python utils/reference_tools/education/score_education_gold.py calibration python utils/reference_tools/education/score_education_gold.py certification

Emits: nothing — prints the result for reference/VALIDATION_EDUCATION.md. Writes no file and touches no pipeline column.

def human_rung(insts: list[str]) -> int | None:
    """Put the coder's institutions on the ladder, using the parser's own table.

    Mirrors `parse_education`: the continuation school never sets the rung, but floors it
    at Elementary when nothing else is named.
    """
    from utils.parsing.education import INSTITUTIONS

    known = [k for k in insts if k in INSTITUTIONS]
    scored = [k for k in known if INSTITUTIONS[k][2] is not None]
    if scored:
        return max(INSTITUTIONS[k][2] for k in scored)
    if any(INSTITUTIONS[k][3] == "continuation" for k in known):
        return 1
    return None

def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("half", choices=["calibration", "certification"])
    ap.add_argument("--worksheet", type=Path, default=None,
                    help="override the worksheet path (used to smoke-test the scorer "
                         "without touching a real, sealed sheet)")
    args = ap.parse_args()

    from utils.parsing.education import parse_education, INSTITUTIONS

    sheet = pd.read_csv(args.worksheet or (GOLD / args.half / "worksheet.csv"),
                        dtype=str)
    frame = pd.read_csv(GOLD / "_frame.csv", dtype={"record_id": str})
    raw = pd.read_csv(RAW, usecols=[ID, FIELD], dtype=str).rename(
        columns={ID: "record_id", FIELD: "vzdelani_raw"})

    df = (sheet.merge(frame[["record_id", "stratum", "stratum_size"]], on="record_id",
                      how="left")
               .merge(raw, on="record_id", how="left"))

    coded = df[df["instituce"].notna() | df["poznamka"].notna()].copy()
    blank = len(df) - len(coded)
    if not len(coded):
        print(f"{args.half}: nothing coded yet — 0 of {len(df)} rows filled in.")
        print("Fill in the worksheet per education_gold/NAVOD.md, then re-run.")
        return 0

    rows = []
    for r in coded.itertuples():
        parsed = parse_education(r.vzdelani_raw)
        p_inst = ([k for k in parsed["educ_institutions"].split("|") if k]
                  if parsed else [])
        h_inst = _tokens(r.instituce)
        unknown = [k for k in h_inst if k not in INSTITUTIONS]
        h_years, p_years = _years(r.roky), {}
        if parsed:
            for slot in ("elementary", "burgher", "continuation"):
                v = parsed[f"educ_years_{slot}"]
                if v is not None:
                    p_years[slot] = int(v)
        h_ap, p_ap = _tri(r.ucen), (bool(parsed["educ_apprentice"]) if parsed else False)
        rows.append({
            "record_id": r.record_id, "stratum": r.stratum,
            "stratum_size": r.stratum_size, "vzdelani": r.vzdelani_raw,
            "inst_match": set(h_inst) == set(p_inst),
            "rung_match": human_rung(h_inst) == (parsed["educ_ord"] if parsed else None),
            "years_match": h_years == p_years,
            "appr_match": None if h_ap is None else (h_ap == p_ap),
            "human": "|".join(h_inst), "parser": "|".join(p_inst),
            "unknown_token": ",".join(unknown),
        })
    res = pd.DataFrame(rows)

    def weighted(col: str) -> float:
        d = res[res[col].notna()]
        if not len(d):
            return float("nan")
        per = d.groupby("stratum").agg(acc=(col, "mean"), size=("stratum_size", "first"))
        return float((per["acc"] * per["size"]).sum() / per["size"].sum())

    print("=" * 74)
    print(f"Education hand-coding — {args.half.upper()}")
    print("=" * 74)
    print()
    print(f"rows in worksheet ... {len(df)}")
    print(f"coded ............... {len(coded)}")
    print(f"not yet coded ....... {blank}   (excluded, not counted as agreement)")
    print()

    print("Per stratum (unweighted — this is where the precision is):")
    per = (res.groupby("stratum")
              .agg(n=("inst_match", "size"),
                   institutions=("inst_match", "mean"),
                   rung=("rung_match", "mean"),
                   years=("years_match", "mean"))
              .sort_values("n", ascending=False))
    print((per.assign(**{c: (per[c] * 100).round(1)
                         for c in ("institutions", "rung", "years")})).to_string())
    print()

    print("Record-weighted (reweighted to the true stratum sizes — the headline):")
    for label, col in (("institutions read identically", "inst_match"),
                       ("rung agrees", "rung_match"),
                       ("years agree", "years_match"),
                       ("apprentice flag agrees", "appr_match")):
        print(f"   {label:32s} {100 * weighted(col):5.1f}%")
    print()

    # A rung that agrees while the institutions differ, or vice versa, is the interesting
    # cell: it separates a dictionary problem from a ladder problem.
    reading = res[~res["inst_match"]]
    ladder = res[res["inst_match"] & ~res["rung_match"]]
    print(f"Disagreements: {len(reading)} on the reading, {len(ladder)} on the ladder "
          f"alone (same institutions, different rung).")
    if len(ladder):
        print("   ⚠ A ladder-only disagreement is a bug in the ranking code, not in the")
        print("     dictionary. These should be zero; investigate before anything else.")
    print()

    unk = sorted({t for s in res["unknown_token"] for t in s.split(",") if t})
    if unk:
        print(f"Tokens the coder used that are not in INSTITUTIONS: {', '.join(unk)}")
        print("   (either a typo, or a school the ladder has no place for — a finding)")
        print()

    if len(reading):
        print("Every reading disagreement, worst stratum first:")
        show = reading.sort_values("stratum")[
            ["stratum", "vzdelani", "human", "parser"]]
        with pd.option_context("display.max_colwidth", 44, "display.width", 200):
            print(show.to_string(index=False))
    return 0

"""Score a hand-coded education worksheet against `education.py`.

Reads a worksheet filled in per `education_gold/NAVOD.md`, re-parses the same raw strings
with the live parser, and reports where the two disagree.

Three things this does that a naive accuracy would not
------------------------------------------------------
**It reweights.** The sample is disproportionate by design — 40 of 952 `unparsed` records
against 90 of 71,622 `plain` ones. An unweighted mean over the worksheet would describe a
population that does not exist. Every headline figure is weighted by the true stratum size
from `_frame.csv`, so it estimates the accuracy the *data* has; the per-stratum figures are
reported beside it, unweighted, because that is where the precision actually is.

**It separates reading from ranking.** The coder names institutions, never a rung. The rung
is then derived from their institutions with the parser's own `INSTITUTIONS` and `TIE_BREAK`.
So a rung disagreement is reported twice: once as a *reading* difference (the two disagree
about which schools are named) and once as a *residual* — the same institutions placed
differently, which would be a bug in the ladder code rather than in the dictionary.

**It refuses to score what was not coded.** A blank cell is "not coded", not "no". Blank
rows are counted and excluded, and the counts are printed, because a worksheet that is 60 %
filled produces a very confident wrong number if the blanks are read as agreement.

Usage:
    python utils/reference_tools/education/score_education_gold.py calibration
    python utils/reference_tools/education/score_education_gold.py certification

Emits: nothing — prints the result for `reference/VALIDATION_EDUCATION.md`. Writes no file
       and touches no pipeline column.

from __future__ import annotations

import argparse
import sys
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

__version__ = "1.0.0"

GOLD = ROOT / "reference" / "education" / "education_gold"
RAW = ROOT / "data" / "raw" / "legionnaires" / "legionnaires.csv"
FIELD, ID = "Vzdělání", "ID"

YES, NO = {"a", "y", "ano", "yes", "1", "true"}, {"n", "ne", "no", "0", "false"}


def _tokens(cell) -> list[str]:
    if not isinstance(cell, str) or not cell.strip():
        return []
    return [t.strip() for t in cell.split("|") if t.strip()]


def _years(cell) -> dict[str, int]:
    out = {}
    for part in _tokens(cell):
        if "=" not in part:
            continue
        k, _, v = part.partition("=")
        try:
            out[k.strip()] = int(v)
        except ValueError:
            pass
    return out


def _tri(cell):
    """'a'/'n' -> True/False; anything else (including blank and '?') -> None."""
    if not isinstance(cell, str):
        return None
    v = cell.strip().lower()
    return True if v in YES else False if v in NO else None

    Mirrors `parse_education`: the continuation school never sets the rung, but floors it

    scored = [k for k in known if INSTITUTIONS[k][2] is not None]
    if scored:
        return max(INSTITUTIONS[k][2] for k in scored)
    if any(INSTITUTIONS[k][3] == "continuation" for k in known):
        return 1
    return None
