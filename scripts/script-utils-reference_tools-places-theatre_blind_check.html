How it works

The choice page (D-218) asked the author to pick among the readers' readings; the author found the single lines hard to rule that way (a line's leading date often belongs to something else: a pause, a note that the company still exists) and asked to rule the same clauses again from the beginning, writing an opinion instead of pressing buttons. This module:

  1. draws the 139 calibration clauses the choice page asked about (every disagreement and the 30 agreed clauses drawn, with their weights and ab_prob, so the two-phase design of D-218 carries over unchanged) into reference/places/theatre_blind_check.csv;
  2. builds the page: each clause on its whole page (D-220), no reading shown, a text box;
  3. reads the notes back (fill_notes) and, once Claude has coded each note into the four answers and the years, the author confirms or corrects each coding on the same page (coded / confirm in the same document); fill_codings writes the confirmed codings into the sheet's v_* columns.

The choice page's answers in the sample sheets are never touched: the blind rulings live in their own sheet. A coding the author has not confirmed is not a ruling.

def draw_certification(seed: int = CERT_SEED) -> pd.DataFrame:

def write_sheet(drawn: pd.DataFrame, path: Path = SHEET) -> Path:

    if Path(path).exists():
        old = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
        if old.get("note", pd.Series(dtype=str)).ne("").any():

def page_items(path: Path = SHEET) -> list[dict]:
    """Each clause on its whole page with what the database has beside it; no reading.  Each
    item says where the line comes from: a company's own page (with what the database lists
    for that company) or a place's chronicle."""
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    clauses = atm.load_clauses()
    rows = clauses[clauses["clause_key"].isin(sheet["clause_key"])]
    ctx = atm.add_context(rows.assign(open_kind="x"), context="page").set_index("clause_key")
    facts = _company_facts(rows["ensemble_id"]) if "ensemble_id" in rows else {}
    text = lambda v: v if isinstance(v, str) else ""                              # noqa: E731
    items = []
    for i, r in enumerate(sheet.itertuples(index=False)):
        c = ctx.loc[r.clause_key]
        on_page = c["source"] == "ensemble_chronicle"
        ens_id = int(c["ensemble_id"]) if on_page and pd.notna(c["ensemble_id"]) else None
        items.append({"i": i, "key": r.clause_key, "sample": r.sample, "place": text(c["place"]),
                      "ensemble": text(c["ensemble"]), "source": "company" if on_page else "place",
                      "company_facts": facts.get(ens_id, "") if ens_id is not None else "",
                      "context": c["context"], "related": text(c["related"]),
                      "clause": c["text"], "glossary": text(c["glossary"]), "date_raw": text(c["date_raw"]),
                      "years_inside": text(c["years_inside"])})
    return items

def build_page(out: Path, path: Path = SHEET, title: str | None = None) -> Path:

def fill_notes(documents: list[dict], path: Path = SHEET) -> int:
    """The page's notes (``blind/<clause_key>``) into the sheet's ``note`` column."""
    sheet, index = _sheet_index(path)
    written = 0
    for d in documents:
        key, note = d.get("clause_key"), (d.get("note") or "").strip()
        if key in index and note:
            sheet.loc[index[key], "note"] = note
            written += 1
    sheet.to_csv(path, index=False, encoding="utf-8-sig")
    return written

def fill_codings(documents: list[dict], path: Path = SHEET) -> dict:
    """Confirmed codings into ``v_*``; a coding the author corrected keeps the correction and
    stays out of ``v_*`` until it is coded again and confirmed; one coded from a note the
    author has since changed (``coded.note_seen``) is not taken."""
    sheet, index = _sheet_index(path)
    counts = {"confirmed": 0, "corrected": 0, "stale": 0, "waiting": 0}
    for d in documents:
        key, coded = d.get("clause_key"), d.get("coded") or {}
        if key not in index or not coded:
            continue
        i = index[key]
        if (coded.get("note_seen") or "").strip() != (d.get("note") or "").strip():
            counts["stale"] += 1
            continue
        sheet.loc[i, "coding_summary"] = coded.get("summary", "")
        if d.get("confirm") == "right":
            # The first 51 codings left ``date`` empty for "the leading date stands", which
            # the later ones (and ``ra.DATE_VERDICTS``) call ``right``.
            coded = {**coded, "date": coded.get("date") or "right"}
            sheet.loc[i, [f"v_{c}" for c in CODED]] = [
                "" if coded.get(c) in (None, "") else str(coded[c]) for c in CODED]
            sheet.loc[i, ["confirmed", "correction"]] = ["yes", ""]
            counts["confirmed"] += 1
        elif d.get("confirm") == "fix":
            sheet.loc[i, [f"v_{c}" for c in CODED]] = ""
            sheet.loc[i, ["confirmed", "correction"]] = ["no", d.get("correction") or ""]
            counts["corrected"] += 1
        else:
            counts["waiting"] += 1
    sheet.to_csv(path, index=False, encoding="utf-8-sig")
    return counts

def blind_verdict_rows(half: str = "certification", path: Path | None = None) -> pd.DataFrame:
    """The blind sheet of ``half`` as :func:`theatre_rule_audit.verdict_rows` scores it.

    The confirmed answers are laid over the samples' rows of that half (the sample sheets on
    disk are not touched), each marked a second-phase draw with the blind sheet's own
    ``ab_group`` / ``ab_prob``; the drawn set is the blind sheet's.  An unconfirmed coding is
    no ruling: the half is refused until every drawn clause is confirmed.  The certification
    half keeps its seal (``RULES_FROZEN`` and ``PROMPT_FROZEN``)."""
    path = (CERT_SHEET if half == "certification" else SHEET) if path is None else path
    blind = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    blind = blind[blind["half"] == half].set_index(["sample", "clause_key"])
    unconfirmed = blind.index[blind["confirmed"].ne("yes")]
    assert not len(unconfirmed), f"{half}: {len(unconfirmed)} blind codings not confirmed, e.g. {list(unconfirmed[:3])}"
    sheets = {}
    for name, sheet in ra.load_samples().items():
        s = sheet.copy()
        mine = blind.xs(name, level="sample") if name in blind.index.get_level_values(0) else blind.iloc[:0]
        rows = s["half"].eq(half)
        assert not s.loc[rows & s["clause_key"].isin(mine.index), "v_activity"].ne("").any() or half == "calibration",             f"{name}: the {half} half already holds answers of its own"
        s.loc[rows, [c for c in ra.VERDICT_COLUMNS if c in s]] = ""
        s.loc[rows, "v_source"] = ""
        on = rows & s["clause_key"].isin(mine.index)
        keys = s.loc[on, "clause_key"]
        for c in ["v_" + c for c in CODED]:
            s.loc[on, c] = keys.map(mine[c]).values
        s.loc[on, "v_source"] = "ab"
        s.loc[on, "ab_group"] = keys.map(mine["ab_group"]).values
        s.loc[on, "ab_prob"] = keys.map(mine["ab_prob"]).values
        sheets[name] = s
    return ra.verdict_rows(tuple(sheets), half=half, sheets=sheets,
                           drawn=set(blind.index.get_level_values("clause_key")))

def _company_facts(ensemble_ids) -> dict:
    """What the database lists for each company: its span on the place's list and the decades
    its own page names (*Působení*), as one plain line; the author found the source of a
    line's years decisive (2026-11-12): on a company's own page a play attests the company."""
    raw = ROOT / "data" / "raw" / "theatre"
    listed = pd.read_csv(raw / "ensembles.csv", usecols=["ensemble_id", "span_raw"]).drop_duplicates("ensemble_id")
    pages = pd.read_csv(raw / "ensemble_pages.csv", usecols=["ensemble_id", "active_raw", "ended_raw"])
    facts = listed.merge(pages, on="ensemble_id", how="outer").set_index("ensemble_id")
    out = {}
    for e in {int(x) for x in ensemble_ids if pd.notna(x)}:
        if e not in facts.index:
            continue
        f = facts.loc[e]
        bits = [f"listed {str(f['span_raw']).strip('() ')}" if isinstance(f["span_raw"], str) else "",
                f"active (its page) {f['active_raw']}" if isinstance(f["active_raw"], str) and f["active_raw"].strip() else "",
                f"ended {f['ended_raw']}" if isinstance(f["ended_raw"], str) and f["ended_raw"].strip() else ""]
        out[e] = " · ".join(b for b in bits if b)
    return out

    for that company) or a place's chronicle."""
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    clauses = atm.load_clauses()
    rows = clauses[clauses["clause_key"].isin(sheet["clause_key"])]
    ctx = atm.add_context(rows.assign(open_kind="x"), context="page").set_index("clause_key")
    facts = _company_facts(rows["ensemble_id"]) if "ensemble_id" in rows else {}
    text = lambda v: v if isinstance(v, str) else ""                              # noqa: E731
    items = []
    for i, r in enumerate(sheet.itertuples(index=False)):
        c = ctx.loc[r.clause_key]
        on_page = c["source"] == "ensemble_chronicle"
        ens_id = int(c["ensemble_id"]) if on_page and pd.notna(c["ensemble_id"]) else None
        items.append({"i": i, "key": r.clause_key, "sample": r.sample, "place": text(c["place"]),
                      "ensemble": text(c["ensemble"]), "source": "company" if on_page else "place",
                      "company_facts": facts.get(ens_id, "") if ens_id is not None else "",
                      "context": c["context"], "related": text(c["related"]),
                      "clause": c["text"], "glossary": text(c["glossary"]), "date_raw": text(c["date_raw"]),
                      "years_inside": text(c["years_inside"])})
    return items


def build_page(out: Path, path: Path = SHEET, title: str | None = None) -> Path:
    data = json.dumps(page_items(path), ensure_ascii=False).replace("</", "<\\/")
    template = TEMPLATE.read_text(encoding="utf-8")
    assert template.count(PLACEHOLDER) == 1, "template: one data placeholder"
    if title:                                  # a page of its own (its own notes), named apart
        assert template.count("<title>Theatre Clause Notes</title>") == 1, "template: the title moved"
        template = template.replace("<title>Theatre Clause Notes</title>", f"<title>{title}</title>")
    Path(out).write_text(template.replace(PLACEHOLDER, data), encoding="utf-8")
    return Path(out)


def _sheet_index(path: Path) -> tuple[pd.DataFrame, dict]:
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    return sheet, {k: i for i, k in enumerate(sheet["clause_key"])}


def fill_notes(documents: list[dict], path: Path = SHEET) -> int:
    """The page's notes (``blind/<clause_key>``) into the sheet's ``note`` column."""
    sheet, index = _sheet_index(path)
    written = 0
    for d in documents:
        key, note = d.get("clause_key"), (d.get("note") or "").strip()
        if key in index and note:
            sheet.loc[index[key], "note"] = note
            written += 1
    sheet.to_csv(path, index=False, encoding="utf-8-sig")
    return written


def fill_codings(documents: list[dict], path: Path = SHEET) -> dict:
    """Confirmed codings into ``v_*``; a coding the author corrected keeps the correction and
    stays out of ``v_*`` until it is coded again and confirmed; one coded from a note the
    author has since changed (``coded.note_seen``) is not taken."""
    sheet, index = _sheet_index(path)
    counts = {"confirmed": 0, "corrected": 0, "stale": 0, "waiting": 0}
    for d in documents:
        key, coded = d.get("clause_key"), d.get("coded") or {}
        if key not in index or not coded:
            continue
        i = index[key]
        if (coded.get("note_seen") or "").strip() != (d.get("note") or "").strip():
            counts["stale"] += 1
            continue
        sheet.loc[i, "coding_summary"] = coded.get("summary", "")
        if d.get("confirm") == "right":
            # The first 51 codings left ``date`` empty for "the leading date stands", which
            # the later ones (and ``ra.DATE_VERDICTS``) call ``right``.
            coded = {**coded, "date": coded.get("date") or "right"}
            sheet.loc[i, [f"v_{c}" for c in CODED]] = [
                "" if coded.get(c) in (None, "") else str(coded[c]) for c in CODED]
            sheet.loc[i, ["confirmed", "correction"]] = ["yes", ""]
            counts["confirmed"] += 1
        elif d.get("confirm") == "fix":
            sheet.loc[i, [f"v_{c}" for c in CODED]] = ""
            sheet.loc[i, ["confirmed", "correction"]] = ["no", d.get("correction") or ""]
            counts["corrected"] += 1
        else:
            counts["waiting"] += 1
    sheet.to_csv(path, index=False, encoding="utf-8-sig")
    return counts



def blind_verdict_rows(half: str = "certification", path: Path | None = None) -> pd.DataFrame:
    """The blind sheet of ``half`` as :func:`theatre_rule_audit.verdict_rows` scores it.

    The confirmed answers are laid over the samples' rows of that half (the sample sheets on
    disk are not touched), each marked a second-phase draw with the blind sheet's own
    ``ab_group`` / ``ab_prob``; the drawn set is the blind sheet's.  An unconfirmed coding is
    no ruling: the half is refused until every drawn clause is confirmed.  The certification
    half keeps its seal (``RULES_FROZEN`` and ``PROMPT_FROZEN``)."""
    path = (CERT_SHEET if half == "certification" else SHEET) if path is None else path
    blind = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    blind = blind[blind["half"] == half].set_index(["sample", "clause_key"])
    unconfirmed = blind.index[blind["confirmed"].ne("yes")]
    assert not len(unconfirmed), f"{half}: {len(unconfirmed)} blind codings not confirmed, e.g. {list(unconfirmed[:3])}"
    sheets = {}
    for name, sheet in ra.load_samples().items():
        s = sheet.copy()
        mine = blind.xs(name, level="sample") if name in blind.index.get_level_values(0) else blind.iloc[:0]
        rows = s["half"].eq(half)
        assert not s.loc[rows & s["clause_key"].isin(mine.index), "v_activity"].ne("").any() or half == "calibration",             f"{name}: the {half} half already holds answers of its own"
        s.loc[rows, [c for c in ra.VERDICT_COLUMNS if c in s]] = ""
        s.loc[rows, "v_source"] = ""
        on = rows & s["clause_key"].isin(mine.index)
        keys = s.loc[on, "clause_key"]
        for c in ["v_" + c for c in CODED]:
            s.loc[on, c] = keys.map(mine[c]).values
        s.loc[on, "v_source"] = "ab"
        s.loc[on, "ab_group"] = keys.map(mine["ab_group"]).values
        s.loc[on, "ab_prob"] = keys.map(mine["ab_prob"]).values
        sheets[name] = s
    return ra.verdict_rows(tuple(sheets), half=half, sheets=sheets,
                           drawn=set(blind.index.get_level_values("clause_key")))
