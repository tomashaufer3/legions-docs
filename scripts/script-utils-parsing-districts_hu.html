TL;DR

The Cisleithanian analogue is utils.parsing.districts (Czech lands). This module builds the Hungarian 1910 administrative backbone — vármegye (county) → járás (district) / chartered town — for the Felvidék (Upper Hungary), the region that became Czechoslovakia in 1918 (Slovakia + Subcarpathian Rus). It is the spine for standardising the Slovak-born soldiers the Cisleithanian census frame cannot reach.

How it works

Source & spine

GISta Hungarorum (gistory.hu; MTA BTK, after the 1910 census) shapefiles under data/raw/census_hu/shp/ — all UTF-8 (.cpg), so districts.read_dbf is called with encodings=("utf-8", …) (its default tries cp1250 first — wrong here). The spine is MO_JarasVaros (577 units = 439 járás + 112 rtv. chartered towns + 25 tjv. boroughs), because that is what the járás-level census (…_ETN.xlsx, kod) keys on 1:1 — the chartered towns (Körmöcbánya=Kremnica, Selmecbánya=Banská Štiavnica, Szakolca=Skalica, Rimaszombat=Rimavská Sobota …) are standalone census units, not folded into a járás. County names come from MO_Megye (Nagytaj == "Felvidék" isolates the 21 Upper-Hungary counties); járás seat towns from MO_Telepules via IDJarassze.

Magyar → Slovak names

GIS names are Magyar (Nyitra, Trencsén, Pozsony) while the soldier records use Slovak (Nitra, Trenčín, Bratislava). The Slovak/Rusyn name of each county and unit is a curated data file (reference/places/district_sk_curation.csv, cols level,key,sk,cs_modern) — the same auditable pattern as the Czech district_cs_curation.csv. key is IDMegye for counties and IDJaras (the unit id) for járás / towns.

Usage

from utils.parsing import districts_hu as dzh
backbone = dzh.build_jaras_backbone(SHP_DIR, curation_csv=SK_CURATION)   # 192 Felvidék units

def build_jaras_backbone(shp_dir, curation_csv=None, region: str = "Felvidék",
                         extra_megye: set[str] = _EXTRA_MEGYE) -> pd.DataFrame:
    """Assemble the járás/town backbone for *region* (default the Upper-Hungary ``Felvidék``).

    One row per census unit (járás or chartered town). Columns: ``unit_id`` (=census ``kod``) ·
    ``unit_type`` (járás / rtv. / tjv.) · ``megye_id`` · ``megye_hu`` · ``megye_sk`` (curated) ·
    ``name_hu`` (Magyar seat town for a járás, else the town's own name) · ``jaras_hu`` (Magyar
    adjectival járás name; NA for towns) · ``name_sk`` (curated Slovak) · ``name_sk_modern`` ·
    ``is_czechoslovak`` · ``nagytaj``. ``region=None`` keeps all of Hungary; *extra_megye* adds
    named counties from other macro-regions (Danube-straddling, north bank to Czechoslovakia).
    """
    shp_dir = Path(shp_dir)
    megye = pd.DataFrame(read_dbf(shp_dir / "MO_Megye.dbf", encodings=_UTF))
    jv = pd.DataFrame(read_dbf(shp_dir / "MO_JarasVaros.dbf", encodings=_UTF))
    seat = _seat_names(shp_dir)
    unit_sk, unit_alt, megye_sk, modern = _load_sk_curation(curation_csv)

    megye_hu = dict(zip(megye["IDMegye"], megye["Megye_1910"]))
    nagytaj = dict(zip(megye["IDMegye"], megye["Nagytaj"]))

    df = jv.copy()
    df["nagytaj"] = df["IDMegye"].map(nagytaj)
    df["megye_hu_tmp"] = df["IDMegye"].map(megye_hu)
    if region is not None:
        keep = (df["nagytaj"] == region) | df["megye_hu_tmp"].isin(extra_megye or set())
        df = df[keep].copy()
    df = df.drop(columns="megye_hu_tmp")

    df["unit_id"] = df["IDJaras"]
    df["unit_type"] = df["Status"]
    df["megye_id"] = df["IDMegye"]
    df["megye_hu"] = df["IDMegye"].map(megye_hu)
    # a járás's Magyar place name is its seat town (via IDJarassze); a town unit is itself
    is_jaras = df["Status"] == "járás"
    df["jaras_hu"] = df["JarasNev"].where(is_jaras)
    df["name_hu"] = df["IDJarassze"].map(seat).where(is_jaras, df["JarasNev"])
    df["megye_sk"] = df["megye_id"].map(megye_sk)
    df["name_sk"] = df["unit_id"].map(unit_sk)
    df["name_sk_alt"] = df["unit_id"].map(unit_alt)
    df["name_sk_modern"] = df["unit_id"].map(modern)
    df["is_czechoslovak"] = ~df["megye_hu"].isin(_STAYED_HUNGARY)

    cols = ["unit_id", "unit_type", "megye_id", "megye_hu", "megye_sk", "name_hu", "jaras_hu",
            "name_sk", "name_sk_alt", "name_sk_modern", "is_czechoslovak", "nagytaj"]
    return df[cols].sort_values(["megye_hu", "unit_type", "unit_id"]).reset_index(drop=True)

def write_sk_curation_template(shp_dir, path, region: str = "Felvidék") -> pd.DataFrame:
    """Write the blank Magyar→Slovak curation skeleton: one ``megye`` row per county + one
    ``jaras`` row per unit, with the Magyar name as an edit seed and an empty ``sk`` to fill."""
    bb = build_jaras_backbone(shp_dir, curation_csv=None, region=region)
    rows = []
    for mid, sub in bb.drop_duplicates("megye_id").groupby("megye_id"):
        rows.append({"level": "megye", "key": mid, "hu": sub.iloc[0]["megye_hu"],
                     "unit_type": "", "sk": "", "cs_modern": ""})
    for _, r in bb.iterrows():
        rows.append({"level": "jaras", "key": r["unit_id"], "hu": r["name_hu"],
                     "unit_type": r["unit_type"], "sk": "", "cs_modern": ""})
    tmpl = pd.DataFrame(rows, columns=["level", "key", "hu", "unit_type", "sk", "cs_modern"])
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    tmpl.to_csv(p, index=False, encoding="utf-8")
    return tmpl

def jaras_report(backbone: pd.DataFrame) -> None:
    """Print coverage of the járás backbone (counties, units, Slovak-name curation load)."""
    n_u = len(backbone)
    n_m = backbone["megye_id"].nunique()
    cs = backbone[backbone["is_czechoslovak"]]
    print(f"Járás/town backbone  (units = {n_u}, counties = {n_m}; "
          f"Czechoslovak: {cs['megye_id'].nunique()} counties / {len(cs)} units)")
    print(f"  by unit type: {dict(backbone['unit_type'].value_counts())}")
    named = backbone["name_sk"].notna().sum()
    m_named = backbone.drop_duplicates("megye_id")["megye_sk"].notna().sum()
    print(f"  Slovak name curated: {named}/{n_u} units, {m_named}/{n_m} counties")
    miss = backbone[backbone["is_czechoslovak"] & backbone["name_sk"].isna()]
    if len(miss):
        print(f"  [!] {len(miss)} Czechoslovak units still without a Slovak name:")
        for _, r in miss.head(25).iterrows():
            print(f"      {r['megye_hu']:<18} {r['unit_type']:<6} {r['name_hu']}")

districts_hu.py — 1910 Transleithanian (Upper-Hungary) district backbone, Magyar + Slovak
=========================================================================================

the **Hungarian** 1910 administrative backbone — *vármegye* (county) → *járás* (district) /
chartered town — for the **Felvidék** (Upper Hungary), the region that became Czechoslovakia in
1918 (Slovakia + Subcarpathian Rus). It is the spine for standardising the Slovak-born soldiers the
Cisleithanian census frame cannot reach.

--------------

-----

from __future__ import annotations

from pathlib import Path

import pandas as pd

from utils.parsing.districts import read_dbf

__version__ = "1.0.0"

_UTF = ("utf-8", "cp1250", "latin-1")            # these GIS dbfs are UTF-8 (unlike the MPIDR one)

# Felvidék counties that stayed Hungary in 1918 (the rest became Czechoslovakia — Slovakia +
# Subcarpathian Rus). Several border counties (Gömör, Hont, Nógrád, Abaúj-Torna, Zemplén, Ugocsa,
# Máramaros) were split; flagged Czechoslovak here at county level — records only name Slovak/Rusyn
# places, so the over-inclusion is harmless for matching.
_STAYED_HUNGARY = {"Borsod", "Heves"}

# Danube-straddling counties the census files under other macro-regions but whose north bank went
# to Czechoslovakia (Komárno, Štúrovo…) — added to the Felvidék backbone so Slovak records there
# resolve. Mostly stayed Hungary, so is_czechoslovak is left True only as a coarse eligibility hint.
_EXTRA_MEGYE = {"Komárom", "Esztergom"}


def _seat_names(shp_dir: Path) -> dict[str, str]:
    """``IDTel1910 → Magyar settlement name`` (for the járás-seat lookup)."""
    tel = read_dbf(shp_dir / "MO_Telepules.dbf", encodings=_UTF)
    return {r["IDTel1910"]: r["telepulesn"] for r in tel}


def _load_sk_curation(curation_csv):
    """Curated Slovak names → ``(unit_sk, unit_alt, megye_sk, modern)`` dicts keyed by id.

    ``unit_sk`` ``{IDJaras: sk}`` primary (seat) name · ``unit_alt`` ``{IDJaras: sk_alt}`` an extra
    Slovak name for the same unit (a non-seat settlement records use) · ``megye_sk`` ``{IDMegye: sk}``
    · ``modern`` ``{IDJaras: cs_modern}`` today's Slovak name where renamed.
    """
    unit_sk: dict[str, str] = {}
    unit_alt: dict[str, str] = {}
    megye_sk: dict[str, str] = {}
    modern: dict[str, str] = {}
    if curation_csv and Path(curation_csv).exists():
        cur = pd.read_csv(curation_csv, dtype=str).fillna("")
        has_modern = "cs_modern" in cur.columns
        has_alt = "sk_alt" in cur.columns
        for _, r in cur.iterrows():
            key, sk = r["key"].strip(), r["sk"].strip()
            if r["level"] == "megye":
                if sk:
                    megye_sk[key] = sk
                continue
            if sk:
                unit_sk[key] = sk
            if has_alt and r["sk_alt"].strip():
                unit_alt[key] = r["sk_alt"].strip()
            if has_modern and r["cs_modern"].strip():
                modern[key] = r["cs_modern"].strip()
    return unit_sk, unit_alt, megye_sk, modern
