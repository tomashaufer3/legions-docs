How it works

run_glyph_e3.py checks the readers on the gold pages: its sections come from the gold sheets, each page is placed by a locator trained on the other calibration pages, and each glyph is named against a bank that leaves its own page out. None of that exists for a page without a gold sheet. This runs the same check there:

  • sections from the section crops the pilot saved (section_images), one per page, court and kind; the municipalities each prints, in order, from the readers' own output (any reader's row for the entry, with or without a mark);
  • placement by one locator trained on every calibration page;
  • the unclaimed-mark detector fitted on every calibration section, at the threshold the gold run set;
  • naming against a bank of every gold glyph (calibration and purposive — the freeze, D-112), with one not in the bank threshold pooled over the gold.

Its output has the columns of e3_verdicts_&lt;tag&gt;.csv (gold blank), so the route and the ledger read it unchanged.

Rehearsed on the gold section images themselves (--rehearse): the same path, fed images that follow the saved crops' naming, against the leave-page-out run.

python utils/lexikon_1910/run_e3_pages.py --sections DIR --passes L1,L2 --out DIR
python utils/lexikon_1910/run_e3_pages.py --rehearse --encoders pixels

    """``(page, court, kind) -> munis in printed order``, from every reader's rows.

    Offices from ``institutions_<label>.csv``, parts from ``parts_<label>.csv``;
    a row counts whether it carries a mark or not, since a municipality printed
    without one still opens an entry the locator must count.

    from utils.lexikon_1910 import lexikon_1910_notes as lxnt

    out: dict = defaultdict(set)
    for path in paths.values():
        for stem, kind in (("institutions_", "offices"), ("parts_", "parts")):
            p = path.with_name(path.name.replace("institutions_", stem, 1))

def full_detector(located: dict, locators: dict, sections: list, cands: dict,
                  locator: lg.Locator) -> lg.Detection:
    """The token-blind mark detector fitted on every calibration section, applied to
    the new sections at the gold run's threshold (best F1 on the calibration pages)."""
    import numpy as np

    gold = lg.detect_marks(located, locators)           # the threshold, leave-page-out
    xs, ys = [], []
    for section, cs, placed in located.values():
        if section.role != "calibration" or not cs:
            continue
        cols = lg.section_text_columns(section)
        elig = np.array([lg.eligible(c, cols.get(c.image, 0.0)) for c in cs])
        pos = {i for i in placed if i is not None}
        x = lg.detector_features(cs, locators[section.scan])
        xs.append(x[elig])
        ys.append(np.array([1 if i in pos else 0 for i in range(len(cs))])[elig])
    model = lg.fit_detector(np.concatenate(xs), np.concatenate(ys))
    scores, elig, columns = {}, {}, {}
    for section in sections:
        cs = cands.get(section.key) or []
        if not cs:
            continue
        cols = lg.section_text_columns(section)
        columns.update(cols)
        scores[section.key] = model(lg.detector_features(cs, locator))
        elig[section.key] = np.array([lg.eligible(c, cols.get(c.image, 0.0)) for c in cs])
    return lg.Detection(scores, elig, {}, gold.threshold, columns)

class Namer:
    """DINOv3 (or pixels) naming against every gold glyph, base x brackets x letter."""

    def __init__(self, glyphs: list, encoder: tuple, download: bool):
        import numpy as np

        self.name, self.variant, self.feature, self.k = encoder
        self.download = download
        self.glyphs = glyphs
        feats = e3.embed(glyphs, [self.name], download, cuts=(self.variant,))
        self.bank_x = feats[(self.name, self.variant)][self.feature]
        # one threshold for pages the bank has never seen: the ABSTAIN_QUANTILE of
        # the similarity of correctly named gold glyphs, each named leave-page-out
        raw = e3.name_bases(self.bank_x, glyphs, self.bank_x, glyphs, self.k)
        right = [s for (pred, s), g in zip(raw, glyphs) if pred == g.base]
        self.threshold = float(np.quantile(right, e3.ABSTAIN_QUANTILE)) if right else -1.0
        self.lettered = [g for g in glyphs if g.letter_box is not None]
        self.letter_x = None
        if self.lettered:
            lf = e3.embed(self.lettered, [e3.LETTER_ENCODER], download, cuts=("letter",))
            key = "pixels" if e3.LETTER_ENCODER == "pixels" else "cls"
            self.letter_x = lf[(e3.LETTER_ENCODER, "letter")][key]

    def __call__(self, readers: list, _config=None) -> list[tuple]:
        import numpy as np

        if not readers:
            return []
        unique = list(dict.fromkeys(g.key for g in readers))
        first = {key: next(g for g in readers if g.key == key) for key in unique}
        got = e3.embed(list(first.values()), [self.name], self.download, cuts=(self.variant,))
        idx = {key: n for n, key in enumerate(unique)}
        x = got[(self.name, self.variant)][self.feature][[idx[g.key] for g in readers]]
        thresholds = {g.scan: self.threshold for g in readers}
        bases = e3.name_bases(x, readers, self.bank_x, self.glyphs, self.k, thresholds)
        letters = [""] * len(readers)
        lettered = [g for g in readers if g.letter_box is not None]
        if lettered and self.letter_x is not None:
            lx = e3.embed(lettered, [e3.LETTER_ENCODER], self.download, cuts=("letter",))
            key = "pixels" if e3.LETTER_ENCODER == "pixels" else "cls"
            q = lx[(e3.LETTER_ENCODER, "letter")][key]
            full = np.zeros((len(readers), q.shape[1]))
            at = {id(g): n for n, g in enumerate(lettered)}
            for n, g in enumerate(readers):
                if g.letter_box is not None:
                    full[n] = q[at[id(g)]]
            letters = e3.name_letters(full, readers, self.letter_x, self.lettered,
                                      [b for b, _s in bases])
        return e3.compose_names(readers, bases, letters)

def run(section_dir: Path, index: list[dict], paths: dict, out: Path, encoders: list[str],
        rounds: int = 4, download: bool = False) -> Path:
    """Verdict rows for every pass in ``paths`` (label -> institutions csv) on the saved sections."""
    data = lg.load_sections()
    located, locators = lg.locate_all(data, rounds)
    glyphs = e3.placed_glyphs(located)
    locator = full_locator(data, rounds)
    sections = sections_from_index(index, section_dir, printed_munis(paths))
    cands = {s.key: lg.section_candidates(s) for s in sections}
    for s in sections:
        scores = locator.glyph.score(cands[s.key]) if cands[s.key] else []
        for c, v in zip(cands[s.key], scores):
            c.glyphness = float(v)
    detection = full_detector(located, locators, sections, cands, locator)
    config = e3.OF_RECORD if e3.OF_RECORD[0] in encoders else e3.FLOOR
    namer = Namer(glyphs, config, download)
    verdicts = []
    for label, path in paths.items():
        claims = e3.read_reader_claims(path)
        on, off, extra = [], [], []
        for section in sections:
            mine = claims.get((section.scan, section.district, section.kind), [])
            listed = [lg.Claim(token, muni, seq, 0, "reader")
                      for muni, seq, _pos, token in mine if muni in section.printed]
            for n, cl in enumerate(listed):
                cl.order = n
            cs = cands[section.key]
            if not listed or not cs:
                off += [(section, cl) for cl in listed]
                continue
            placed = locator.place(cs, listed)
            a, b, c = e3.section_glyphs(section, cs, listed, placed, detection, label)
            on += a
            off += [(section, cl) for cl in b]
            extra += c
        named = namer(on + extra)
        verdicts += e3.verdict_rows(label, {}, on, named[:len(on)], off, extra,
                                    named[len(on):], {})
        print(f"  {label}: {len(on)} placed, {len(off)} unplaced, {len(extra)} unclaimed marks")
    out.mkdir(parents=True, exist_ok=True)
    path = out / "e3_verdicts_pages.csv"
    e3.write_csv(path, verdicts)
    print(f"  {len(sections)} sections, encoder {'/'.join(map(str, config))}, "
          f"not-in-bank below {namer.threshold:.3f} -> {path}")
    return path

def rehearse(encoders: list[str], download: bool = False) -> Path:
    """The page path on the gold section images, for the r5 reader of record and one Haiku read."""
    gold_dirs = [lg.GOLD / role for role in lg.ROLES]
    labels = ["gemini38_r5atlasntg1", "haiku55nh_r5atlasg1"]
    paths = {l: p for l, p in e3.reader_passes("r5").items() if l in labels}
    out = ROOT / "output" / "frozen" / "rehearsal_pages"
    results = []
    for d in gold_dirs:                  # one directory at a time: Section has one sheet_dir
        index = si.index_from_names(d)
        if index:
            results.append(run(d, index, paths, out / d.name, encoders, download=download))
    return out

def main(argv=None) -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):
            pass
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--sections", default="", help="a directory of saved section crops")
    ap.add_argument("--passes", default="", help="comma list of reader labels")
    ap.add_argument("--out", default="")
    ap.add_argument("--encoders", default="pixels,dinov3-vitb16")
    ap.add_argument("--download", action="store_true")
    ap.add_argument("--rehearse", action="store_true",
                    help="run the page path on the gold section images")
    args = ap.parse_args(argv)
    encoders = [e for e in args.encoders.split(",") if e]
    if args.rehearse:
        rehearse(encoders, args.download)
        return 0
    labels = [l for l in args.passes.split(",") if l]
    found = {}
    for l in labels:
        hit = sorted(e3.BAKEOFF.glob(f"*/institutions_{l}.csv"))
        if hit:
            found[l] = hit[0]
    directory = Path(args.sections)
    run(directory, si.read_index(directory), found, Path(args.out), encoders, download=args.download)
    return 0

        # one threshold for pages the bank has never seen: the ABSTAIN_QUANTILE of
        # the similarity of correctly named gold glyphs, each named leave-page-out
        raw = e3.name_bases(self.bank_x, glyphs, self.bank_x, glyphs, self.k)
        right = [s for (pred, s), g in zip(raw, glyphs) if pred == g.base]
        self.threshold = float(np.quantile(right, e3.ABSTAIN_QUANTILE)) if right else -1.0
        self.lettered = [g for g in glyphs if g.letter_box is not None]
        self.letter_x = None
        if self.lettered:
            lf = e3.embed(self.lettered, [e3.LETTER_ENCODER], download, cuts=("letter",))
            key = "pixels" if e3.LETTER_ENCODER == "pixels" else "cls"
            self.letter_x = lf[(e3.LETTER_ENCODER, "letter")][key]

    def __call__(self, readers: list, _config=None) -> list[tuple]:
        import numpy as np

        if not readers:
            return []
        unique = list(dict.fromkeys(g.key for g in readers))
        first = {key: next(g for g in readers if g.key == key) for key in unique}
        got = e3.embed(list(first.values()), [self.name], self.download, cuts=(self.variant,))
        idx = {key: n for n, key in enumerate(unique)}
        x = got[(self.name, self.variant)][self.feature][[idx[g.key] for g in readers]]
        thresholds = {g.scan: self.threshold for g in readers}
        bases = e3.name_bases(x, readers, self.bank_x, self.glyphs, self.k, thresholds)
        letters = [""] * len(readers)
        lettered = [g for g in readers if g.letter_box is not None]
        if lettered and self.letter_x is not None:
            lx = e3.embed(lettered, [e3.LETTER_ENCODER], self.download, cuts=("letter",))
            key = "pixels" if e3.LETTER_ENCODER == "pixels" else "cls"
            q = lx[(e3.LETTER_ENCODER, "letter")][key]
            full = np.zeros((len(readers), q.shape[1]))
            at = {id(g): n for n, g in enumerate(lettered)}
            for n, g in enumerate(readers):
                if g.letter_box is not None:
                    full[n] = q[at[id(g)]]
            letters = e3.name_letters(full, readers, self.letter_x, self.lettered,
                                      [b for b, _s in bases])
        return e3.compose_names(readers, bases, letters)

def run(section_dir: Path, index: list[dict], paths: dict, out: Path, encoders: list[str],
        rounds: int = 4, download: bool = False) -> Path:
    """Verdict rows for every pass in ``paths`` (label -> institutions csv) on the saved sections."""
    data = lg.load_sections()
    located, locators = lg.locate_all(data, rounds)
    glyphs = e3.placed_glyphs(located)
    locator = full_locator(data, rounds)
    sections = sections_from_index(index, section_dir, printed_munis(paths))
    cands = {s.key: lg.section_candidates(s) for s in sections}
    for s in sections:
        scores = locator.glyph.score(cands[s.key]) if cands[s.key] else []
        for c, v in zip(cands[s.key], scores):
            c.glyphness = float(v)
    detection = full_detector(located, locators, sections, cands, locator)
    config = e3.OF_RECORD if e3.OF_RECORD[0] in encoders else e3.FLOOR
    namer = Namer(glyphs, config, download)
    verdicts = []
    for label, path in paths.items():
        claims = e3.read_reader_claims(path)
        on, off, extra = [], [], []
        for section in sections:
            mine = claims.get((section.scan, section.district, section.kind), [])
            listed = [lg.Claim(token, muni, seq, 0, "reader")
                      for muni, seq, _pos, token in mine if muni in section.printed]
            for n, cl in enumerate(listed):
                cl.order = n
            cs = cands[section.key]
            if not listed or not cs:
                off += [(section, cl) for cl in listed]
                continue
            placed = locator.place(cs, listed)
            a, b, c = e3.section_glyphs(section, cs, listed, placed, detection, label)
            on += a
            off += [(section, cl) for cl in b]
            extra += c
        named = namer(on + extra)
        verdicts += e3.verdict_rows(label, {}, on, named[:len(on)], off, extra,
                                    named[len(on):], {})
        print(f"  {label}: {len(on)} placed, {len(off)} unplaced, {len(extra)} unclaimed marks")
    out.mkdir(parents=True, exist_ok=True)
    path = out / "e3_verdicts_pages.csv"
    e3.write_csv(path, verdicts)
    print(f"  {len(sections)} sections, encoder {'/'.join(map(str, config))}, "
          f"not-in-bank below {namer.threshold:.3f} -> {path}")
    return path


def rehearse(encoders: list[str], download: bool = False) -> Path:
    """The page path on the gold section images, for the r5 reader of record and one Haiku read."""
    gold_dirs = [lg.GOLD / role for role in lg.ROLES]
    labels = ["gemini38_r5atlasntg1", "haiku55nh_r5atlasg1"]
    paths = {l: p for l, p in e3.reader_passes("r5").items() if l in labels}
    out = ROOT / "output" / "frozen" / "rehearsal_pages"
    results = []
    for d in gold_dirs:                  # one directory at a time: Section has one sheet_dir
        index = si.index_from_names(d)
        if index:
            results.append(run(d, index, paths, out / d.name, encoders, download=download))
    return out


def main(argv=None) -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):
            pass
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--sections", default="", help="a directory of saved section crops")
    ap.add_argument("--passes", default="", help="comma list of reader labels")
    ap.add_argument("--out", default="")
    ap.add_argument("--encoders", default="pixels,dinov3-vitb16")
    ap.add_argument("--download", action="store_true")
    ap.add_argument("--rehearse", action="store_true",
                    help="run the page path on the gold section images")
    args = ap.parse_args(argv)
    encoders = [e for e in args.encoders.split(",") if e]
    if args.rehearse:
        rehearse(encoders, args.download)
        return 0
    labels = [l for l in args.passes.split(",") if l]
    found = {}
    for l in labels:
        hit = sorted(e3.BAKEOFF.glob(f"*/institutions_{l}.csv"))
        if hit:
            found[l] = hit[0]
    directory = Path(args.sections)
    run(directory, si.read_index(directory), found, Path(args.out), encoders, download=args.download)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
