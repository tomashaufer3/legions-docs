How it works

Reads what run_matice_pilot.py wrote to data/interim/matice/ and draws it on the 1910 court-district polygons of the MPIDR shapefile (reference/places/mpidr/, ETRS89 LAEA metres, read by utils.parsing.spatial). Okres borders are the court edges where the courts on either side belong to different okresy; no GIS library is needed.

matice_map_okres matice_own_z per census okres: each okres's own society's shrunk member rate, standardised within the society's territory (D-161). matice_map_court The same per court district, with the court census of 1910 as denominator. matice_map_municipal Every settlement with a member: Matice česká founders and Matice moravská members (1890), circle area proportional to the count.

Writes PDF + PNG to output/figures/matice/ (ths.save_for_latex).

Usage::

python utils/reference_tools/places/map_matice.py

def court_shapes() -> pd.DataFrame:
    """One row per 1910 court district of the Czech lands: court and okres id, land, rings."""
    rows = []
    for rec, rings in zip(read_dbf(str(MPIDR) + ".dbf"), read_polygons(str(MPIDR) + ".shp")):
        if rec.get("LAND") in LANDS and rings:
            rows.append({"court_id": int(rec["ID_1"]),
                         "okres_id": int(rec["REGION"]) * 1000 + int(rec["DISTRICT"]),
                         "land": LANDS[rec["LAND"]], "rings": rings})
    return pd.DataFrame(rows)

def border_segments(shapes: pd.DataFrame, by: str) -> np.ndarray:
    """Edges between two polygons that differ in ``by`` (okres or land), and the outer edge.

    A segment shared by two courts appears in both; it is a border when the two differ in
    ``by`` or when only one court has it. Coordinates are rounded to the metre to pair
    them."""
    seen: dict[tuple, list] = defaultdict(list)
    for key, rings in zip(shapes[by], shapes["rings"]):
        for ring in rings:
            pts = np.round(ring).astype(np.int64)
            for a, b in zip(map(tuple, pts[:-1]), map(tuple, pts[1:])):
                if a != b:
                    seen[(a, b) if a < b else (b, a)].append(key)
    return np.array([seg for seg, keys in seen.items()
                     if len(keys) == 1 or len(set(keys)) > 1], dtype=float)

def base_map(shapes: pd.DataFrame, values: pd.Series | None, title: str, cmap=None,
             norm=None, label: str = "") -> tuple[plt.Figure, plt.Axes]:
    """Court polygons shaded by ``values`` (indexed by court_id; grey where missing),
    with okres borders thin and land borders thick."""
    fig, ax = plt.subplots(figsize=(10, 6.2))
    patches = PatchCollection([_patch(r) for r in shapes["rings"]], linewidths=0)
    if values is not None:
        v = shapes["court_id"].map(values).to_numpy(dtype=float)
        patches.set_array(np.ma.masked_invalid(v))
        patches.set_cmap(cmap)
        patches.set_norm(norm)
        cmap.set_bad(ths.CU_GREY_LIGHT)
    else:
        patches.set_facecolor("#f4f4f4")
    ax.add_collection(patches)
    for by, lw, col in (("court_id", 0.15, "#ffffff"), ("okres_id", 0.45, ths.CU_GREY),
                        ("land", 1.3, ths.CU_BLACK)):
        if by == "court_id" and values is None:
            col = ths.CU_GREY_LIGHT
        seg = border_segments(shapes, by)
        ax.add_collection(LineCollection(seg.reshape(-1, 2, 2), linewidths=lw, colors=col))
    allpts = np.vstack([np.vstack(r) for r in shapes["rings"]])
    ax.set_xlim(allpts[:, 0].min() - 5e3, allpts[:, 0].max() + 5e3)
    ax.set_ylim(allpts[:, 1].min() - 5e3, allpts[:, 1].max() + 5e3)
    ax.set_aspect("equal")
    ax.axis("off")
    ax.set_title(title)
    if values is not None:
        cb = fig.colorbar(patches, ax=ax, shrink=0.6, pad=0.01, extend="both")
        cb.set_label(label)
    return fig, ax

def settlement_counts() -> pd.DataFrame:
    """Members per settlement (Matice česká founders, Matice moravská 1890 members) with
    the point to draw them at; Prague's at ``PRAGUE_LONLAT``."""
    e = pd.read_csv(INTERIM / "matice_entries.csv", dtype={"settlement_id": str},
                    keep_default_na=False, na_values=[""])
    e["is_corporate"] = e["is_corporate"].astype(str) == "True"
    e["member_no"] = pd.to_numeric(e["member_no"], errors="coerce")
    e["place"] = e["settlement_id"].where(e["place_status"] != "prague", "PRAGUE")
    groups = mt.measure_groups(e[e["place"].notna()])
    counts = pd.DataFrame({"mc": groups["mc_founders"].groupby("place").size(),
                           "mm": groups["mm_members_1890"].groupby("place").size()}).fillna(0)
    geo = pd.read_parquet(GEO / "settlements_1910_geo.parquet").set_index("settlement_id")
    lon = counts.index.map(lambda s: PRAGUE_LONLAT[0] if s == "PRAGUE" else geo["lon"].get(s))
    lat = counts.index.map(lambda s: PRAGUE_LONLAT[1] if s == "PRAGUE" else geo["lat"].get(s))
    counts["x"], counts["y"] = laea_forward(np.asarray(lon, dtype=float), np.asarray(lat, dtype=float))
    return counts.dropna(subset=["x", "y"])

#!/usr/bin/env python3
"""
Maps of the Matice measure at three levels: political okres, court district, municipality.

Reads what ``run_matice_pilot.py`` wrote to ``data/interim/matice/`` and draws it on the
1910 court-district polygons of the MPIDR shapefile (``reference/places/mpidr/``, ETRS89
LAEA metres, read by ``utils.parsing.spatial``). Okres borders are the court edges where
the courts on either side belong to different okresy; no GIS library is needed.

``matice_map_okres``
    ``matice_own_z`` per census okres: each okres's own society's shrunk member rate,
    standardised within the society's territory (D-161).
``matice_map_court``
    The same per court district, with the court census of 1910 as denominator.
``matice_map_municipal``
    Every settlement with a member: Matice česká founders and Matice moravská members
    (1890), circle area proportional to the count.

Writes PDF + PNG to ``output/figures/matice/`` (``ths.save_for_latex``).

Usage::

    python utils/reference_tools/places/map_matice.py
"""

def court_shapes() -> pd.DataFrame:
    """One row per 1910 court district of the Czech lands: court and okres id, land, rings."""
    rows = []
    for rec, rings in zip(read_dbf(str(MPIDR) + ".dbf"), read_polygons(str(MPIDR) + ".shp")):
        if rec.get("LAND") in LANDS and rings:
            rows.append({"court_id": int(rec["ID_1"]),
                         "okres_id": int(rec["REGION"]) * 1000 + int(rec["DISTRICT"]),
                         "land": LANDS[rec["LAND"]], "rings": rings})
    return pd.DataFrame(rows)


def border_segments(shapes: pd.DataFrame, by: str) -> np.ndarray:

    A segment shared by two courts appears in both; it is a border when the two differ in

def _patch(rings: list[np.ndarray]) -> PathPatch:
    verts, codes = [], []
    for ring in rings:
        verts += list(ring)
        codes += [MplPath.MOVETO] + [MplPath.LINETO] * (len(ring) - 2) + [MplPath.CLOSEPOLY]
    return PathPatch(MplPath(np.array(verts), codes))
