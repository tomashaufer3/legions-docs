How it works

Notebook 15 §18 builds data/final/register/ from the route's run folders: the register module reads each run's final_inventory.csv and final_abbreviations.csv, and it sits in utils/lexikon_1910, whose every module is held to the freeze by digest. So the rechecked tables reach the register here instead, through the module's own door: each run is loaded as the route loads it (load_run), its inventory and abbreviations are swapped for the rechecked ones, and build_register(runs=...) builds from them. Nothing in the freeze moves.

It writes to OUT_DIR, data/final/register_rechecked/, beside the route's register, which stays the build of record. D-190: the rechecked register becomes the analysis baseline only once §8 (both_ways) passes; until then it is built and checked, not used.

"""The register built from the rechecked tables (notebook 17 §9, D-196).

Notebook 15 §18 builds ``data/final/register/`` from the route's run folders: the register module
reads each run's ``final_inventory.csv`` and ``final_abbreviations.csv``, and it sits in
``utils/lexikon_1910``, whose every module is held to the freeze by digest. So the rechecked tables
reach the register here instead, through the module's own door: each run is loaded as the route
loads it (``load_run``), its inventory and abbreviations are swapped for the rechecked ones, and
``build_register(runs=...)`` builds from them. Nothing in the freeze moves.

It writes to :data:`OUT_DIR`, ``data/final/register_rechecked/``, beside the route's register, which
stays the build of record. D-190: the rechecked register becomes the analysis baseline only once §8
(``both_ways``) passes; until then it is built and checked, not used.
"""
from __future__ import annotations

__version__ = "0.1.0"

import sys
from dataclasses import replace
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT), str(ROOT / "utils" / "lexikon_1910")]

import run_frozen as rf                                                # noqa: E402
from utils.lexikon_1910 import lexikon_1910_register as reg           # noqa: E402

RECHECK = ROOT / "output" / "recheck"
OUT_DIR = reg.FINAL / "register_rechecked"


def runs(recheck: Path = RECHECK) -> list:
    """Every full run, in the register's edition order, with its rechecked inventory and
    abbreviations where §7 wrote them (the route's otherwise, and said so)."""
    out = []
    for edition in reg.EDITION_PRIORITY:
        folder = rf.out_dir(edition, "full")
        if not (folder / "final_inventory.csv").exists():
            continue
        run = reg.load_run(folder)
        inventory = Path(recheck) / edition / "final_inventory_rechecked.csv"
        abbreviations = Path(recheck) / edition / "final_abbreviations_rechecked.csv"
        if not inventory.exists():
            print(f"  {edition}: no rechecked inventory — the route's is used")
        out.append(replace(run, inventory=reg._read(inventory) if inventory.exists() else run.inventory,
                           abbreviations=reg._read(abbreviations) if abbreviations.exists() else run.abbreviations))
    return out


def build(write: bool = True, out_dir: Path = OUT_DIR, recheck: Path = RECHECK) -> dict:
    """``build_register`` on :func:`runs`, written to ``out_dir`` (never ``data/final/register``)."""
    if Path(out_dir).resolve() == (reg.FINAL / "register").resolve():
        raise ValueError("the route's register is notebook 15's to build; the rechecked one goes beside it")
    return reg.build_register(runs=runs(recheck), out_dir=Path(out_dir), write=write)
