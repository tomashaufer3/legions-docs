How it works

Notebook 15 §18 builds data/final/register/ from the route's run folders: the register module reads each run's final_inventory.csv and final_abbreviations.csv, and it sits in utils/lexikon_1910, whose every module is held to the freeze by digest. So the rechecked tables reach the register here instead, through the module's own door: each run is loaded as the route loads it (load_run), its inventory and abbreviations are swapped for the rechecked ones, and build_register(runs=...) builds from them. Nothing in the freeze moves.

It writes to OUT_DIR, data/final/register_rechecked/, beside the route's register, which stays the build of record. D-190: the rechecked register becomes the analysis baseline only once §8 (both_ways) passes; until then it is built and checked, not used.

def runs(recheck: Path = RECHECK, table_courts: bool = True, reports: list | None = None,
         run_reports: list | None = None, row_reports: list | None = None) -> list:
    """Every full run, in the register's edition order, with its rechecked inventory and
    abbreviations where §7 wrote them (the route's otherwise, and said so), and — on a German
    volume, unless ``table_courts`` is off — its table rows filed under the court the table's own
    headings print (``table_courts.mend``, D-202), then — every volume — its seat rows numbered and the
    courts a page carried or opened under another label filed under their own (``table_runs.mend``, D-207), and a
    court created after 1910 keyed ``1000 + n`` under its 1910 court (``section_keys.mend_run``, D-208). The
    table's row structure is mended around the court fixes (``table_rows``, D-211): numbered rows tagged
    as totals and rows opened on a military line before them, unnumbered rows numbered after them."""
    out = []
    for edition in reg.EDITION_PRIORITY:
        folder = rf.out_dir(edition, "full")
        if not (folder / "final_inventory.csv").exists():
            continue
        run = reg.load_run(folder)
        inventory = Path(recheck) / edition / "final_inventory_rechecked.csv"
        abbreviations = Path(recheck) / edition / "final_abbreviations_rechecked.csv"
        if not inventory.exists():
            print(f"  {edition}: no rechecked inventory — the route's is used")
        run = replace(run, inventory=reg._read(inventory) if inventory.exists() else run.inventory,
                      abbreviations=reg._read(abbreviations) if abbreviations.exists() else run.abbreviations)
        if table_courts:
            run, rows = trw.mend_structure(run)
            run, report = tc.mend(run)
            if reports is not None:
                reports.append(report)
            if report.get("rows_moved"):
                print(f"  {edition}: {report['rows_moved']} table municipalities filed by the court heading above them, "
                      f"{len(report['courts_added'])} page courts added (D-202)")
            run, filed = trn.mend(run)
            if run_reports is not None:
                run_reports.append(filed)
            if filed["seats_numbered"] or filed["rows_moved"]:
                print(f"  {edition}: {len(filed['seats_numbered'])} seat rows numbered, {filed['rows_moved']} table "
                      f"municipalities in {filed['runs_moved']} runs filed under the court they continue or open (D-207)")
            run, rows = trw.mend_numbers(run, rows, second_table(edition), trr.replies(edition, recheck))
            if row_reports is not None:
                row_reports.append(rows)
            print(f"  {edition}: {len(rows['labels_mended'])} rows under a neighbour's label, "
                  f"{len(rows['second_reader'])} rows from the second table reader, {len(rows['reread'])} "
                  f"from the page re-read (D-215)")
            print(f"  {edition}: {len(rows['numbered_totals'])} numbered rows tagged as totals, "
                  f"{len(rows['military_openings'])} rows opened on a military line, {len(rows['numbered'])} "
                  f"unnumbered rows numbered, {len(rows['unnumbered_left'])} left; {len(rows['duplicates_mended'])} "
                  f"duplicate rows mended, {len(rows['duplicates_left'])} left (D-211)")
        run, keyed = sk.mend_run(run)          # last: the table runs above are filed by the numbers printed (D-208)
        if keyed["parts_rows"] or keyed["attribute_rows"] or keyed["table"]["courts"]:
            print(f"  {edition}: post-1910 courts keyed {sk.OFFSET} + n — {keyed['parts_rows']} parts rows, "
                  f"{keyed['attribute_rows']} attribute rows, table {keyed['table']['courts']} (D-208)")
        out.append(run)
    return out

def second_table(edition: str, pages: str = "full") -> pd.DataFrame | None:
    """The run's second table reader's rows (Haiku; the register reads the reader of record's), as
    ``load_run`` reads a table, or ``None`` where the run has no second reader's pass on disk."""
    labels = list(rf.labels(pages, edition)["table"])
    path = bo.find_pass_file("table_rows", labels[1]) if len(labels) > 1 else None
    return reg._read(path) if path else None

def build(write: bool = True, out_dir: Path = OUT_DIR, recheck: Path = RECHECK, table_courts: bool = True) -> dict:
    """``build_register`` on :func:`runs`, written to ``out_dir`` (never ``data/final/register``)."""
    if Path(out_dir).resolve() == (reg.FINAL / "register").resolve():
        raise ValueError("the route's register is notebook 15's to build; the rechecked one goes beside it")
    reports: list = []
    run_reports: list = []
    row_reports: list = []
    built = runs(recheck, table_courts, reports, run_reports, row_reports)
    tables = reg.build_register(runs=built, out_dir=Path(out_dir), write=write)
    faults = pd.concat([trw.check(r, tables["register_places"]) for r in built], ignore_index=True)
    summary = trw.check_report(faults, tables["register_institutions"])
    tables["_table_check"] = faults
    print("  the table check (D-211):", summary)
    tables = pt.add(tables, built, recheck)          # D-216: after the check, which reads the places as built
    print("  place types (D-216):", tables["_meta"]["place_types"])
    if write:
        pt.write(tables, Path(out_dir))
    if write and table_courts:                       # what the table fixes moved, for the docs (0.2.1, 0.3.0)
        (Path(recheck) / "table_courts.json").write_text(json.dumps(reports, indent=1, ensure_ascii=False),
                                                         encoding="utf-8")
        (Path(recheck) / "table_runs.json").write_text(json.dumps(run_reports, indent=1, ensure_ascii=False,
                                                                  default=str), encoding="utf-8")
        (Path(recheck) / "table_rows.json").write_text(json.dumps(row_reports, indent=1, ensure_ascii=False,
                                                                  default=str), encoding="utf-8")
        faults.to_csv(Path(recheck) / "table_check.csv", index=False, encoding="utf-8")
        (Path(recheck) / "table_check.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return tables
