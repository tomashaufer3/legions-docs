How it works

The route applies a ruling as an author event on the mark its question was about (run_frozen.apply_review, notebook 15b §5) and builds its final tables from the marks the readers made (§6). Three kinds of ruling therefore never reach the tables as ruled, and one reaches them malformed:

  • A symbol typed on a whole-entry question. Asked is [poorhouse] printed for this part? and answered [castle], the route records the poorhouse as not printed and keeps the castle only in the ruling's note (C11): the question was about one symbol, and the ledger has no mark for a claim no reader made. Here the typed symbol is added — to the inventory, or to the place types on a type question — unless the part already holds it, or another of the author's rulings on the part says it is not printed (then it is listed as a conflict and nothing is added).
  • A symbol typed on an outlined mark of the list of parts that no reading named on that part. final_types writes only the types some reading named (routed_types), so the author's value replaced the box's reading and then had no row to go to. Added.
  • A type an author's ruling on a box decided is labelled opus, on the outlined mark: the value is right, the hand is not (final_types labels every box decision so). Relabelled author.
  • A code that is not a symbol of the volume's key went into the tables as typed (rc-lokalie on a Czech page: no brackets, and the Lokalie is in the Mähren and Schlesien keys only). Where the bracketed form is in the key, it is recoded; otherwise the row is removed and the mark is listed to be ruled again.

A mark ruled again after the route applied the first ruling (rerulings.json beside the staged rulings in output/review_readback/&lt;run&gt;/, one decision record per mark, the same id as the decision it replaces) is carried as the difference between the route's value and the new one: the old value's row is removed, the new symbol added. Dobříš 26/0, first typed rc-lokalie, was ruled again on 2026-11-06 as a symbol on no plate ([!]), so its row goes and nothing replaces it.

Nothing in the route changes. The result is output/recheck/&lt;edition&gt;/author_carry.csv, one row per change with its reason and the decision it comes from; notebook 17 §7 lays it over the route's final_inventory.csv and final_types.csv with apply. The rulings are the page's decisions as staged in each run's review/ — the ones apply_review read — and the re-rulings staged after them. Validation answers are scored, not applied (notebook 15b §5), and are left out here too.

def ruled_value(item: dict, decision: dict) -> str | None:
    """What a decision says is printed, as ``apply_review`` reads it before its C11 step: the
    token kept, the replacement typed (``""``: no mark), *cannot tell*, off the plate."""
    value = {"keep": item.get("token", ""), "replace": decision.get("replacement", ""),
             "unreadable": led.UNSURE, "off_atlas": "[!]"}.get(decision.get("decision"))
    return None if value is None else (value or led.ABSENT)

def changes(edition: str, rulings: list[tuple[dict, dict]], marks: dict, inventory: list[dict],
            types: list[dict], vocabulary: set, rerulings: dict | None = None) -> list[dict]:
    """The changes to the route's two final tables that carry the author's rulings (see the
    module docstring).

    ``rulings`` — ``(queue item, decision)`` for the run's rulings; ``marks`` — the run's marks
    table (``ledger/marks_<run>.csv``) by mark: its part and its ``final`` value and ``rule``;
    ``inventory``, ``types`` — the route's ``final_inventory.csv``, ``final_types.csv``;
    ``vocabulary`` — the codes of the volume's key (``run_frozen._english``); ``rerulings`` —
    ``{decision id: decision}``, the marks ruled again after the route applied the first ruling.
    """
    rerulings = rerulings or {}
    held = {"inventory": defaultdict(set), "types": defaultdict(set)}
    for name, rows in (("inventory", inventory), ("types", types)):
        for r in rows:
            held[name][(r["page"], r["district"], r["muni_seq"], r["settlement_seq"])].add(r["code"])
    # Which marks hold each inventory claim: a row is removed only where the ruled mark is all
    # that holds it.
    holders = {(r["page"], r["district"], r["muni_seq"], r["settlement_seq"], r["code"]):
               set(filter(None, r.get("marks", "").split(" | "))) for r in inventory}
    part_of = {m: _part(r) for m, r in marks.items()}
    ruled = [(item, d, ruled_value(item, d)) for item, d in rulings]
    ruled = [(item, d, v) for item, d, v in ruled if v is not None and item["mark"] in part_of]
    # What the author said is not printed, on each part, by any question (a conflict check).
    absent = defaultdict(set)
    for item, _d, v in ruled:
        if v == led.ABSENT and item.get("token"):
            absent[part_of[item["mark"]]].add(item["token"])
    out = []

    def add(table, part, code, reason, decision, mark):
        if code in held[table][part]:
            return
        if code in absent[part]:
            out.append(_change(edition, table, "conflict", part, code,
                               reason + "; another of your rulings on this part says it is not "
                               "printed — nothing added", decision, mark))
            return
        out.append(_change(edition, table, "add", part, code, reason, decision, mark))
        held[table][part].add(code)

    for item, d, v in ruled:
        mark, part = item["mark"], part_of[item["mark"]]
        if d["id"] in rerulings or mark.endswith("#abbr") or v in NOT_A_SYMBOL:
            continue                                   # an abbreviation is text, not a code
        table = _table_of(mark)
        if v not in vocabulary:
            # Only a value that reached a table needs mending: the route's final for the mark
            # is the value as typed (a C11 question's typed symbol never got there).
            if marks[mark]["final"] == v and v in held[table][part]:
                fixed = "[" + v.strip("[]") + "]"
                if fixed in vocabulary:
                    out.append(_change(edition, table, "recode", part, v,
                                       "typed without its brackets", d["id"], mark, fixed))
                else:
                    out.append(_change(edition, table, "remove", part, v,
                                       f"'{v}' is not a symbol of this volume's key: rule the mark "
                                       "again", d["id"], mark))
            continue
        if led.is_claim_mark(mark) and v != item.get("token"):
            add(table, part, v, f"typed on the question about {item.get('token')}", d["id"], mark)

    # Marks ruled again: the route holds the first ruling's value; the difference is carried.
    reruled = set()
    for item, d, _v in ruled:
        new = rerulings.get(d["id"])
        mark = item["mark"]
        if new is None or mark.endswith("#abbr"):
            continue
        reruled.add(mark)
        part, table, was = part_of[mark], _table_of(mark), marks[mark]["final"]
        now = ruled_value(item, new)
        if now is None or now == was:
            continue
        why = (f"ruled again {str(new.get('decided_at', ''))[:10]}: {_describe(now)}"
               + (f" ({new['comment']})" if new.get("comment") else ""))
        if was not in NOT_A_SYMBOL and was in held[table][part]:
            others = holders.get(part + (was,), set()) - {mark} if table == "inventory" else set()
            if others:
                out.append(_change(edition, table, "conflict", part, was,
                                   why + "; other marks still hold it — nothing removed", d["id"], mark))
            else:
                out.append(_change(edition, table, "remove", part, was, why, d["id"], mark))
                held[table][part].discard(was)
        if now not in NOT_A_SYMBOL:
            if now in vocabulary:
                add(table, part, now, why, d["id"], mark)
            else:
                out.append(_change(edition, table, "refused", part, now,
                                   f"'{now}' is not a symbol of this volume's key: rule the mark again",
                                   d["id"], mark))

    # Outlined marks of the list of parts the author ruled to a symbol: each such symbol is a
    # type of its part, whether or not a reading named it there.
    by_box = defaultdict(dict)
    for mark, r in marks.items():
        if (r["rule"] == "author" and "_parts_" in mark.split("#")[0] and mark not in reruled
                and r["final"] not in NOT_A_SYMBOL and r["final"] in vocabulary):
            by_box[part_of[mark]].setdefault(r["final"], mark)
    decision_of = {item["mark"]: d["id"] for item, d, _v in ruled}
    for part, codes in sorted(by_box.items()):
        for code, mark in sorted(codes.items()):
            add("types", part, code, "your ruling on the outlined mark; no reading named it here",
                decision_of.get(mark, ""), mark)
    for t in types:
        part = (t["page"], t["district"], t["muni_seq"], t["settlement_seq"])
        if t["rule"] == BOX_RULE and t["code"] in by_box.get(part, {}):
            mark = by_box[part][t["code"]]
            out.append(_change(edition, "types", "relabel", part, t["code"],
                               "decided by your ruling on the outlined mark, labelled opus",
                               decision_of.get(mark, ""), mark))
    return out

def apply(inventory: list[dict], types: list[dict], carry: list[dict]) -> tuple[list[dict], list[dict]]:
    """``(inventory, types)`` with ``carry``'s changes laid over the route's rows (notebook 17
    §7). New rows are keyed and filed as the route files its own (``entry`` =
    page/court/muni/part); each says ``author`` in its rule, which the register reads as the
    author's hand (``lexikon_1910_register.settled_by``)."""
    inv, typ = [dict(r) for r in inventory], [dict(r) for r in types]
    courts = _court_of(inv + typ)
    readers = _readers_of(typ)

    def key(r):
        return (r["page"], r["district"], r["muni_seq"], r["settlement_seq"], r["code"])

    for c in carry:
        rows = inv if c["table"] == "inventory" else typ
        k = (c["page"], c["district"], c["muni_seq"], c["settlement_seq"], c["code"])
        if c["action"] == "relabel":
            for r in rows:
                if key(r) == k:
                    r["rule"] = "author"
        elif c["action"] == "recode":
            for r in rows:
                if key(r) == k:
                    r["code"] = c["new_code"]
        elif c["action"] == "remove":
            rows[:] = [r for r in rows if key(r) != k]
        elif c["action"] == "add" and not any(key(r) == k for r in rows):
            court, how = courts.get((c["page"], c["district"]), ("", ""))
            base = {"entry": "/".join((c["page"], court or c["district"], c["muni_seq"],
                                       c["settlement_seq"])),
                    "page": c["page"], "district": c["district"], "court_id": court,
                    "court_source": how, "muni_seq": c["muni_seq"],
                    "settlement_seq": c["settlement_seq"], "code": c["code"]}
            if rows is inv:
                # The way back is the question the author answered: its ledger trail holds the
                # ruling, with the symbol typed in its note. No mark of the ledger holds the claim.
                inv.append({**base, "count": "1", "count_source": "author",
                            "rule": f"author ({c['reason']})", "marks": c["mark"]})
            else:
                typ.append({**base, "rule": "author", "support": f"0 of {readers}" if readers else ""})
    order = lambda r: (r["page"], r["district"], r["muni_seq"], r["settlement_seq"], r["code"])  # noqa: E731
    return sorted(inv, key=order), sorted(typ, key=order)

def load(edition: str, pages: str = "full") -> dict:
    """The run's rulings, marks, final tables and key, read from ``output/frozen/`` only."""
    run = rf.out_dir(edition, pages)
    queue = {i["id"]: i for i in json.loads((run / "review" / "queue.json").read_text("utf-8"))["items"]}
    path = run / "review" / "decisions.json"
    decisions = json.loads(path.read_text("utf-8")) if path.exists() else []
    rulings = [(queue[d["id"]], d) for d in decisions
               if d.get("id") in queue and queue[d["id"]]["source"].endswith(":ruling")]
    name = f"{edition}_{pages}"
    marks = {r["mark"]: r for r in _read(run / "ledger" / f"marks_{name}.csv")}
    staged = ROOT / "output" / "review_readback" / name / "rerulings.json"
    rerulings = {d["id"]: d for d in json.loads(staged.read_text("utf-8"))} if staged.exists() else {}
    return {"rulings": rulings, "marks": marks, "inventory": _read(run / "final_inventory.csv"),
            "types": _read(run / "final_types.csv"), "vocabulary": set(rf._english(edition)),
            "rerulings": rerulings}

def run(edition: str, pages: str = "full", out: Path = OUT) -> list[dict]:
    """Write ``<out>/<edition>/author_carry.csv`` and return its rows."""
    got = load(edition, pages)
    rows = changes(edition, got["rulings"], got["marks"], got["inventory"], got["types"],
                   got["vocabulary"], got["rerulings"])
    target = Path(out) / edition / FILE
    target.parent.mkdir(parents=True, exist_ok=True)
    with open(target, "w", encoding="utf-8", newline="") as fh:
        w = csv.DictWriter(fh, fieldnames=FIELDS, lineterminator="\n")
        w.writeheader()
        w.writerows(rows)
    return rows

def report(edition: str, rows: list[dict]) -> str:
    counts = Counter((r["table"], r["action"]) for r in rows)
    lines = [f"{edition}: " + (", ".join(f"{t} {a} {n}" for (t, a), n in sorted(counts.items()))
                               or "nothing to carry")]
    for r in rows:
        if r["action"] in ("add", "remove", "recode", "conflict", "refused") and (r["table"] == "inventory"
                                                                        or r["action"] != "add"):
            lines.append(f"  {r['action']:8} {r['table']:9} {r['page']} {r['district']} "
                         f"{r['muni_seq']}/{r['settlement_seq']} {r['code']}"
                         + (f" -> {r['new_code']}" if r["new_code"] else "") + f" — {r['reason']}")
    added = Counter(r["code"] for r in rows if r["table"] == "types" and r["action"] == "add")
    if added:
        lines.append("  types added: " + ", ".join(f"{c} {n}" for c, n in added.most_common()))
    return "\n".join(lines)

def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--edition", default="all", choices=(*EDITIONS, "all"))
    ap.add_argument("--out", default=str(OUT))
    args = ap.parse_args(argv)
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")          # a Windows console is cp1250
    for ed in EDITIONS if args.edition == "all" else (args.edition,):
        print(report(ed, run(ed, out=Path(args.out))))
    return 0

"""The author's rulings the frozen route could not carry into its final tables (D-193).

The route applies a ruling as an ``author`` event on the mark its question was about
(``run_frozen.apply_review``, notebook 15b §5) and builds its final tables from the marks the
readers made (§6). Three kinds of ruling therefore never reach the tables as ruled, and one reaches
them malformed:

* **A symbol typed on a whole-entry question.** Asked *is [poorhouse] printed for this part?* and
  answered ``[castle]``, the route records the poorhouse as not printed and keeps the castle only
  in the ruling's note (C11): the question was about one symbol, and the ledger has no mark for a
  claim no reader made. Here the typed symbol is added — to the inventory, or to the place types
  on a type question — unless the part already holds it, or another of the author's rulings on
  the part says it is not printed (then it is listed as a conflict and nothing is added).
* **A symbol typed on an outlined mark of the list of parts** that no reading named on that part.
  ``final_types`` writes only the types some reading named (``routed_types``), so the author's
  value replaced the box's reading and then had no row to go to. Added.
* **A type an author's ruling on a box decided** is labelled ``opus, on the outlined mark``: the
  value is right, the hand is not (``final_types`` labels every box decision so). Relabelled
  ``author``.
* **A code that is not a symbol of the volume's key** went into the tables as typed
  (``rc-lokalie`` on a Czech page: no brackets, and the Lokalie is in the Mähren and Schlesien keys
  only). Where the bracketed form is in the key, it is recoded; otherwise the row is removed and
  the mark is listed to be ruled again.

A mark **ruled again** after the route applied the first ruling (``rerulings.json`` beside the
staged rulings in ``output/review_readback/<run>/``, one decision record per mark, the same id as
the decision it replaces) is carried as the difference between the route's value and the new one:
the old value's row is removed, the new symbol added. Dobříš 26/0, first typed ``rc-lokalie``, was
ruled again on 2026-11-06 as a symbol on no plate (``[!]``), so its row goes and nothing replaces it.

Nothing in the route changes. The result is ``output/recheck/<edition>/author_carry.csv``, one row
per change with its reason and the decision it comes from; notebook 17 §7 lays it over the route's
``final_inventory.csv`` and ``final_types.csv`` with :func:`apply`. The rulings are the page's
decisions as staged in each run's ``review/`` — the ones ``apply_review`` read — and the re-rulings
staged after them. Validation answers are scored, not applied (notebook 15b §5), and are left out
here too.

    python -m utils.recheck.author_carry --edition all
"""
from __future__ import annotations

__version__ = "0.2.0"   # marks ruled again (rerulings.json) are carried as the difference

import argparse
import csv
import json
import re
import sys
from collections import Counter, defaultdict
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT), str(ROOT / "utils" / "lexikon_1910")]

import run_frozen as rf                                                # noqa: E402
from utils.lexikon_1910 import lexikon_1910_ledger as led             # noqa: E402

OUT = ROOT / "output" / "recheck"
EDITIONS = ("cz", *rf.GERMAN)
FILE = "author_carry.csv"
FIELDS = ("edition", "table", "action", "page", "district", "muni_seq", "settlement_seq", "code",
          "new_code", "reason", "decision", "mark")
#: Final values that name no symbol: no mark, an open question, *cannot tell*, off the plate.
NOT_A_SYMBOL = frozenset({led.ABSENT, led.OPEN, led.UNSURE, "", "[?]", "[!]"})
#: How ``final_types`` labels a type an outlined box decided, whoever ruled the box.
BOX_RULE = "opus: opus, on the outlined mark"
csv.field_size_limit(10 ** 9)


def ruled_value(item: dict, decision: dict) -> str | None:
    """What a decision says is printed, as ``apply_review`` reads it before its C11 step: the
    token kept, the replacement typed (``""``: no mark), *cannot tell*, off the plate."""
    value = {"keep": item.get("token", ""), "replace": decision.get("replacement", ""),
             "unreadable": led.UNSURE, "off_atlas": "[!]"}.get(decision.get("decision"))
    return None if value is None else (value or led.ABSENT)


def _table_of(mark: str) -> str:
    """Which final table a mark's value goes to: a type question's mark (``#type``) or an
    outlined mark of the list of parts → ``types``; any other → ``inventory``."""
    image = mark.split("#")[0]
    return "types" if mark.endswith("#type") or "_parts_" in image else "inventory"


def _describe(value: str) -> str:
    return {"[!]": "a symbol on no plate", led.ABSENT: "not printed", led.UNSURE: "cannot tell"}.get(value, value)


def _part(row: dict) -> tuple:
    return (row["page"], row["district"], str(row["muni"]), str(row["seq"]))


def _change(edition, table, action, part, code, reason, decision="", mark="", new_code="") -> dict:
    page, district, muni, seq = part
    return {"edition": edition, "table": table, "action": action, "page": page,
            "district": district, "muni_seq": muni, "settlement_seq": seq, "code": code,
            "new_code": new_code, "reason": reason, "decision": decision, "mark": mark}


def changes(edition: str, rulings: list[tuple[dict, dict]], marks: dict, inventory: list[dict],
            types: list[dict], vocabulary: set, rerulings: dict | None = None) -> list[dict]:
    """The changes to the route's two final tables that carry the author's rulings (see the
    module docstring).

    ``rulings`` — ``(queue item, decision)`` for the run's rulings; ``marks`` — the run's marks
    table (``ledger/marks_<run>.csv``) by mark: its part and its ``final`` value and ``rule``;
    ``inventory``, ``types`` — the route's ``final_inventory.csv``, ``final_types.csv``;
    ``vocabulary`` — the codes of the volume's key (``run_frozen._english``); ``rerulings`` —
    ``{decision id: decision}``, the marks ruled again after the route applied the first ruling.
    """
    rerulings = rerulings or {}
    held = {"inventory": defaultdict(set), "types": defaultdict(set)}
    for name, rows in (("inventory", inventory), ("types", types)):
        for r in rows:
            held[name][(r["page"], r["district"], r["muni_seq"], r["settlement_seq"])].add(r["code"])
    # Which marks hold each inventory claim: a row is removed only where the ruled mark is all
    # that holds it.
    holders = {(r["page"], r["district"], r["muni_seq"], r["settlement_seq"], r["code"]):
               set(filter(None, r.get("marks", "").split(" | "))) for r in inventory}
    part_of = {m: _part(r) for m, r in marks.items()}
    ruled = [(item, d, ruled_value(item, d)) for item, d in rulings]
    ruled = [(item, d, v) for item, d, v in ruled if v is not None and item["mark"] in part_of]
    # What the author said is not printed, on each part, by any question (a conflict check).
    absent = defaultdict(set)
    for item, _d, v in ruled:
        if v == led.ABSENT and item.get("token"):
            absent[part_of[item["mark"]]].add(item["token"])
    out = []

    def add(table, part, code, reason, decision, mark):
        if code in held[table][part]:
            return
        if code in absent[part]:
            out.append(_change(edition, table, "conflict", part, code,
                               reason + "; another of your rulings on this part says it is not "
                               "printed — nothing added", decision, mark))
            return
        out.append(_change(edition, table, "add", part, code, reason, decision, mark))
        held[table][part].add(code)

    for item, d, v in ruled:
        mark, part = item["mark"], part_of[item["mark"]]
        if d["id"] in rerulings or mark.endswith("#abbr") or v in NOT_A_SYMBOL:
            continue                                   # an abbreviation is text, not a code
        table = _table_of(mark)
        if v not in vocabulary:
            # Only a value that reached a table needs mending: the route's final for the mark
            # is the value as typed (a C11 question's typed symbol never got there).
            if marks[mark]["final"] == v and v in held[table][part]:
                fixed = "[" + v.strip("[]") + "]"
                if fixed in vocabulary:
                    out.append(_change(edition, table, "recode", part, v,
                                       "typed without its brackets", d["id"], mark, fixed))
                else:
                    out.append(_change(edition, table, "remove", part, v,
                                       f"'{v}' is not a symbol of this volume's key: rule the mark "
                                       "again", d["id"], mark))
            continue
        if led.is_claim_mark(mark) and v != item.get("token"):
            add(table, part, v, f"typed on the question about {item.get('token')}", d["id"], mark)

    # Marks ruled again: the route holds the first ruling's value; the difference is carried.
    reruled = set()
    for item, d, _v in ruled:
        new = rerulings.get(d["id"])
        mark = item["mark"]
        if new is None or mark.endswith("#abbr"):
            continue
        reruled.add(mark)
        part, table, was = part_of[mark], _table_of(mark), marks[mark]["final"]
        now = ruled_value(item, new)
        if now is None or now == was:
            continue
        why = (f"ruled again {str(new.get('decided_at', ''))[:10]}: {_describe(now)}"
               + (f" ({new['comment']})" if new.get("comment") else ""))
        if was not in NOT_A_SYMBOL and was in held[table][part]:
            others = holders.get(part + (was,), set()) - {mark} if table == "inventory" else set()
            if others:
                out.append(_change(edition, table, "conflict", part, was,
                                   why + "; other marks still hold it — nothing removed", d["id"], mark))
            else:
                out.append(_change(edition, table, "remove", part, was, why, d["id"], mark))
                held[table][part].discard(was)
        if now not in NOT_A_SYMBOL:
            if now in vocabulary:
                add(table, part, now, why, d["id"], mark)
            else:
                out.append(_change(edition, table, "refused", part, now,
                                   f"'{now}' is not a symbol of this volume's key: rule the mark again",
                                   d["id"], mark))

    # Outlined marks of the list of parts the author ruled to a symbol: each such symbol is a
    # type of its part, whether or not a reading named it there.
    by_box = defaultdict(dict)
    for mark, r in marks.items():
        if (r["rule"] == "author" and "_parts_" in mark.split("#")[0] and mark not in reruled
                and r["final"] not in NOT_A_SYMBOL and r["final"] in vocabulary):
            by_box[part_of[mark]].setdefault(r["final"], mark)
    decision_of = {item["mark"]: d["id"] for item, d, _v in ruled}
    for part, codes in sorted(by_box.items()):
        for code, mark in sorted(codes.items()):
            add("types", part, code, "your ruling on the outlined mark; no reading named it here",
                decision_of.get(mark, ""), mark)
    for t in types:
        part = (t["page"], t["district"], t["muni_seq"], t["settlement_seq"])
        if t["rule"] == BOX_RULE and t["code"] in by_box.get(part, {}):
            mark = by_box[part][t["code"]]
            out.append(_change(edition, "types", "relabel", part, t["code"],
                               "decided by your ruling on the outlined mark, labelled opus",
                               decision_of.get(mark, ""), mark))
    return out


def _court_of(rows: list[dict]) -> dict:
    """``{(page, district): (court_id, court_source)}`` as the route filed the section."""
    return {(r["page"], r["district"]): (r["court_id"], r["court_source"]) for r in rows}


def _readers_of(types: list[dict]) -> str:
    """How many readings the run has, as the route's ``support`` (``3 of 4``) says."""
    for t in types:
        found = re.search(r" of (\d+)$", t.get("support", ""))
        if found:
            return found.group(1)
    return ""


def apply(inventory: list[dict], types: list[dict], carry: list[dict]) -> tuple[list[dict], list[dict]]:
    """``(inventory, types)`` with ``carry``'s changes laid over the route's rows (notebook 17
    §7). New rows are keyed and filed as the route files its own (``entry`` =
    page/court/muni/part); each says ``author`` in its rule, which the register reads as the
    author's hand (``lexikon_1910_register.settled_by``)."""
    inv, typ = [dict(r) for r in inventory], [dict(r) for r in types]
    courts = _court_of(inv + typ)
    readers = _readers_of(typ)

    def key(r):
        return (r["page"], r["district"], r["muni_seq"], r["settlement_seq"], r["code"])

    for c in carry:
        rows = inv if c["table"] == "inventory" else typ
        k = (c["page"], c["district"], c["muni_seq"], c["settlement_seq"], c["code"])
        if c["action"] == "relabel":
            for r in rows:
                if key(r) == k:
                    r["rule"] = "author"
        elif c["action"] == "recode":
            for r in rows:
                if key(r) == k:
                    r["code"] = c["new_code"]
        elif c["action"] == "remove":
            rows[:] = [r for r in rows if key(r) != k]
        elif c["action"] == "add" and not any(key(r) == k for r in rows):
            court, how = courts.get((c["page"], c["district"]), ("", ""))
            base = {"entry": "/".join((c["page"], court or c["district"], c["muni_seq"],
                                       c["settlement_seq"])),
                    "page": c["page"], "district": c["district"], "court_id": court,
                    "court_source": how, "muni_seq": c["muni_seq"],
                    "settlement_seq": c["settlement_seq"], "code": c["code"]}
            if rows is inv:
                # The way back is the question the author answered: its ledger trail holds the
                # ruling, with the symbol typed in its note. No mark of the ledger holds the claim.
                inv.append({**base, "count": "1", "count_source": "author",
                            "rule": f"author ({c['reason']})", "marks": c["mark"]})
            else:
                typ.append({**base, "rule": "author", "support": f"0 of {readers}" if readers else ""})
    order = lambda r: (r["page"], r["district"], r["muni_seq"], r["settlement_seq"], r["code"])  # noqa: E731
    return sorted(inv, key=order), sorted(typ, key=order)


def _read(path: Path) -> list[dict]:
    with open(path, encoding="utf-8", newline="") as fh:
        return list(csv.DictReader(fh))


def load(edition: str, pages: str = "full") -> dict:
    """The run's rulings, marks, final tables and key, read from ``output/frozen/`` only."""
    run = rf.out_dir(edition, pages)
    queue = {i["id"]: i for i in json.loads((run / "review" / "queue.json").read_text("utf-8"))["items"]}
    path = run / "review" / "decisions.json"
    decisions = json.loads(path.read_text("utf-8")) if path.exists() else []
    rulings = [(queue[d["id"]], d) for d in decisions
               if d.get("id") in queue and queue[d["id"]]["source"].endswith(":ruling")]
    name = f"{edition}_{pages}"
    marks = {r["mark"]: r for r in _read(run / "ledger" / f"marks_{name}.csv")}
    staged = ROOT / "output" / "review_readback" / name / "rerulings.json"
    rerulings = {d["id"]: d for d in json.loads(staged.read_text("utf-8"))} if staged.exists() else {}
    return {"rulings": rulings, "marks": marks, "inventory": _read(run / "final_inventory.csv"),
            "types": _read(run / "final_types.csv"), "vocabulary": set(rf._english(edition)),
            "rerulings": rerulings}


def run(edition: str, pages: str = "full", out: Path = OUT) -> list[dict]:
    """Write ``<out>/<edition>/author_carry.csv`` and return its rows."""
    got = load(edition, pages)
    rows = changes(edition, got["rulings"], got["marks"], got["inventory"], got["types"],
                   got["vocabulary"], got["rerulings"])
    target = Path(out) / edition / FILE
    target.parent.mkdir(parents=True, exist_ok=True)
    with open(target, "w", encoding="utf-8", newline="") as fh:
        w = csv.DictWriter(fh, fieldnames=FIELDS, lineterminator="\n")
        w.writeheader()
        w.writerows(rows)
    return rows


def report(edition: str, rows: list[dict]) -> str:
    counts = Counter((r["table"], r["action"]) for r in rows)
    lines = [f"{edition}: " + (", ".join(f"{t} {a} {n}" for (t, a), n in sorted(counts.items()))
                               or "nothing to carry")]
    for r in rows:
        if r["action"] in ("add", "remove", "recode", "conflict", "refused") and (r["table"] == "inventory"
                                                                        or r["action"] != "add"):
            lines.append(f"  {r['action']:8} {r['table']:9} {r['page']} {r['district']} "
                         f"{r['muni_seq']}/{r['settlement_seq']} {r['code']}"
                         + (f" -> {r['new_code']}" if r["new_code"] else "") + f" — {r['reason']}")
    added = Counter(r["code"] for r in rows if r["table"] == "types" and r["action"] == "add")
    if added:
        lines.append("  types added: " + ", ".join(f"{c} {n}" for c, n in added.most_common()))
    return "\n".join(lines)


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--edition", default="all", choices=(*EDITIONS, "all"))
    ap.add_argument("--out", default=str(OUT))
    args = ap.parse_args(argv)
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")          # a Windows console is cp1250
    for ed in EDITIONS if args.edition == "all" else (args.edition,):
        print(report(ed, run(ed, out=Path(args.out))))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
