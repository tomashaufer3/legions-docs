How it works

The author clicked and named 203 pictograms on German gold pages 0142, 0276 and 0480 (german_click_page; output/german_click_page/clicks). Each click is cut as a glyph and named against the Czech bank and threshold exactly as E3 names a placed box (run_e3_pages.Namer), so the placement is taken out of the question. The decision configuration is fixed beforehand: the reader-of-record encoder (run_glyph_e3.OF_RECORD: DINOv3 ViT-B/16, base cut, CLS feature, k = 1) and the Czech gold's Namer.threshold unchanged. The clicks are queries only: nothing is added to the bank or tuned on them.

Two ways of getting the box from a click, both reported: P1 build_glyph_bank.tight_box on the click (the ink under it plus close neighbours): does not depend on the candidate step; a window cut then takes whatever is near; P2 the candidate cluster the click falls in (locate_glyphs.candidates_on_image), skipped when none: what the route's own box would be, and the gap to P1 is the candidate step's share.

Measured, with a 95 % interval clustered by strip: J share of glyphs whose nearest similarity reaches the threshold (the gate's "judged"); A agreement with the clicked name among the judged; A0 raw 1-NN accuracy, no threshold, over glyphs whose token the Czech bank holds: tells a threshold problem (A0 high, J low) from a discrimination problem (A0 low); per token (n >= 5), and the Czech baseline under the same protocol (the bank's own glyphs, leave-page-out). Decision (GATE_*): (a) bank fine, fix placement: J >= J_cz - 0.10 and A >= A_cz - 0.05; (b) build a German bank: A0 below about 0.85; (c) mixed: A0 >= 0.90 but J < J_cz - 0.10.

def read_clicks() -> list[dict]:
    """Every named click: ``{image, page, strip, oracle, x, y, token}`` in section-image coordinates."""
    from german_click_recall import CLICKS

    out = []
    for path in sorted(CLICKS.glob("*.json")):
        doc = json.loads(path.read_text(encoding="utf-8"))
        for c in doc["clicks"]:
            if c.get("token"):
                out.append({"image": doc["image"], "page": doc["page"], "strip": doc["strip"],
                            "oracle": bool(doc.get("oracle")), "x": c["sx"], "y": c["sy"], "token": c["token"]})
    return out

def click_glyphs(clicks: list[dict], mode: str, ink_threshold: int | None) -> tuple[list, list[dict], Counter]:
    """``(glyphs, their clicks, skipped)`` for ``mode`` ``p1`` or ``p2``."""
    import locate_glyphs as lg
    import run_glyph_e3 as e3
    from PIL import Image

    from build_glyph_bank import line_height, tight_box

    orig = lg.ink_mask
    if ink_threshold:
        lg.ink_mask = lambda image, threshold=ink_threshold: orig(image, threshold)
    images, per_image, skipped = {}, {}, Counter()
    glyphs, kept = [], []
    for n, c in enumerate(clicks):
        name = c["image"]
        if name not in images:
            images[name] = Image.open(SECTIONS / name)
            lh = line_height(images[name].convert("L")) or 46
            cands = lg.candidates_on_image(images[name], name, lh) if mode == "p2" else []
            per_image[name] = (lh, cands, lg.line_neighbours(cands) if cands else {}, lg.ink_mask(images[name]))
        lh, cands, lines, ink = per_image[name]
        modifier_boxes, letter_box = (), None
        if mode == "p1":
            box = tight_box(ink, c["x"], c["y"], lh)
            if box is None:
                skipped["no ink near the click"] += 1
                continue
        else:
            hits = [i for i, k in enumerate(cands) if lg.inside(k.box, c["x"], c["y"], 0.1 * lh) and not k.bracket]
            if not hits:
                skipped["click in no candidate"] += 1
                continue
            i = min(hits, key=lambda i: math.dist(cands[i].centre, (c["x"], c["y"])))
            box = cands[i].box
            m = lg.modifiers(cands, i, lines)
            modifier_boxes = tuple(cands[j].box for j in (m.left, m.right, m.letter) if j is not None)
            letter_box = cands[m.letter].box if m.letter is not None else None
        glyphs.append(e3.Glyph(f"de{c['page']}", "oracle", "offices", "", 0, "", n, c["token"], name,
                               tuple(box), lh, SECTIONS, "click", "", modifier_boxes, letter_box))
        kept.append(c)
    lg.ink_mask = orig
    return glyphs, kept, skipped

    clusters = list(values_by_cluster.values())
    rng = random.Random(SEED)

def measure(truth: list[str], pred: list[str], sim: list[float], holds: list[bool], threshold: float,
            strips: list[str]) -> dict:
    n = len(truth)
    judged = [s >= threshold for s in sim]
    right = [p == t for p, t in zip(pred, truth)]
    j = sum(judged) / n if n else float("nan")
    a = sum(r for r, jd in zip(right, judged) if jd) / max(sum(judged), 1)
    in_bank = [r for r, h in zip(right, holds) if h]
    a0 = sum(in_bank) / max(len(in_bank), 1)
    by_strip = defaultdict(list)
    for k, s in enumerate(strips):
        by_strip[s].append((judged[k], right[k], holds[k]))

    def stat(parts, which):
        rows = [r for part in parts for r in part]
        if which == "J":
            return sum(r[0] for r in rows) / len(rows)
        if which == "A":
            jd = [r for r in rows if r[0]]
            return sum(r[1] for r in jd) / max(len(jd), 1)
        hb = [r for r in rows if r[2]]
        return sum(r[1] for r in hb) / max(len(hb), 1)

    return {"n": n, "J": j, "A": a, "A0": a0, "bank_holds": sum(holds) / max(n, 1),
            "J_ci": interval(by_strip, lambda p: stat(p, "J")), "A_ci": interval(by_strip, lambda p: stat(p, "A")),
            "A0_ci": interval(by_strip, lambda p: stat(p, "A0"))}

def per_token(truth, pred, sim, holds, threshold, minimum=5) -> list[dict]:
    rows = []
    for token in sorted(set(truth), key=lambda t: -truth.count(t)):
        idx = [i for i, t in enumerate(truth) if t == token]
        if len(idx) < minimum:
            continue
        rows.append({"token": token, "n": len(idx), "in_bank": sum(holds[i] for i in idx),
                     "J": sum(sim[i] >= threshold for i in idx) / len(idx),
                     "A0": sum(pred[i] == token for i in idx) / len(idx),
                     "named": dict(Counter(pred[i] for i in idx if pred[i] != token).most_common(3))})
    return rows

def leave_one_page_out(x, glyphs, truth, strips, namer, bank_tokens, bank_pages) -> dict:
    """Each German page named by exemplars from the other pages: German only, and Czech bank + German.

    The threshold of a held-out page is the ``ABSTAIN_QUANTILE`` of the similarity of the *other*
    pages' correctly named glyphs, each named leave-page-out in the same arm: the rule the Czech bank
    uses (``run_glyph_e3.page_thresholds``), so nothing is tuned on the held-out page. Three pages are
    three folds: a direction, not a rate. These are gold pages the route was tuned on."""
    import numpy as np

    import run_glyph_e3 as e3

    pages = [g.scan for g in glyphs]
    n = len(glyphs)
    out = {}
    for arm in ("german only", "czech + german"):
        pred, sim, adj, holds, thresholds = [None] * n, [0.0] * n, [0.0] * n, [False] * n, {}
        for held in sorted(set(pages)):
            train = [i for i, p in enumerate(pages) if p != held]
            test = [i for i, p in enumerate(pages) if p == held]
            bank_x, bank_t, bank_p = x[train], [truth[i] for i in train], [pages[i] for i in train]
            if arm == "czech + german":
                bank_x = np.vstack([namer.bank_x, bank_x])
                bank_t, bank_p = bank_tokens + bank_t, bank_pages + bank_p
            own = e3.classify(x[train], [pages[i] for i in train], bank_x, bank_t, bank_p, namer.k)
            right = [s for (guess, s), i in zip(own, train) if guess == truth[i]]
            threshold = float(np.quantile(right, e3.ABSTAIN_QUANTILE)) if right else -1.0
            thresholds[held] = round(threshold, 3)
            got = e3.classify(x[test], [held] * len(test), bank_x, bank_t, bank_p, namer.k)
            held_tokens = set(bank_t)
            for (guess, s), i in zip(got, test):
                pred[i], sim[i], adj[i], holds[i] = guess, s, s - threshold, truth[i] in held_tokens
        m = measure(truth, pred, adj, holds, 0.0, strips)
        m["thresholds"] = thresholds
        m["tokens"] = per_token(truth, pred, adj, holds, 0.0)
        out[arm] = m
    return out

def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--encoder", choices=("pixels", "dinov3"), default="pixels")
    parser.add_argument("--download", action="store_true", help="fetch the DINOv3 weights if not cached")
    parser.add_argument("--ink", type=int, nargs="*", default=[128],
                        help="ink threshold(s) for the German crops, each run in turn (the route's is 128)")
    parser.add_argument("--rounds", type=int, default=4)
    parser.add_argument("--german-bank", action="store_true",
                        help="also: leave-one-page-out with the German clicks as exemplars (Czech only / German only / both)")
    args = parser.parse_args(argv)

    import locate_glyphs as lg
    import run_e3_pages as e3p
    import run_glyph_e3 as e3

    config = e3.OF_RECORD if args.encoder == "dinov3" else e3.FLOOR
    print(f"encoder {config}; threshold from the Czech gold's correctly named glyphs, leave-page-out")
    data = lg.load_sections()
    located, _locators = lg.locate_all(data, args.rounds)
    bank = e3.placed_glyphs(located)
    namer = e3p.Namer(bank, config, args.download)
    print(f"Czech bank: {len(bank)} placed gold glyphs; not-in-bank below {namer.threshold:.3f}")
    bank_tokens, bank_pages = [g.base for g in bank], [g.scan for g in bank]

    # the Czech baseline under the same protocol: the bank's own glyphs, each against the other pages
    raw_cz = e3.classify(namer.bank_x, bank_pages, namer.bank_x, bank_tokens, bank_pages, namer.k)
    cz_truth = [g.base for g in bank]
    cz = measure(cz_truth, [p for p, _s in raw_cz], [s for _p, s in raw_cz],
                 e3.bank_holds(cz_truth, bank_pages, bank_tokens, bank_pages), namer.threshold,
                 [f"{g.scan}-{g.kind}" for g in bank])
    result = {"scorer": __version__, "encoder": list(config), "threshold": namer.threshold, "czech": cz, "german": {}}
    lines = [f"Oracle test, {args.encoder} {config}, Czech threshold {namer.threshold:.3f}",
             f"  Czech baseline (the bank against itself, leave-page-out): n {cz['n']}  J {cz['J']:.3f}  "
             f"A {cz['A']:.3f}  A0 {cz['A0']:.3f}"]

    clicks = read_clicks()
    runs = [(mode, label, ink) for ink in args.ink
            for mode, label in (("p1", "P1 box cut from the click"), ("p2", "P2 the candidate cluster holding the click"))]
    for mode, label, ink in runs:
        key = f"{mode}@ink{ink}"
        glyphs, kept, skipped = click_glyphs(clicks, mode, ink)
        if not glyphs:
            continue
        got = e3.embed(glyphs, [namer.name], args.download, cuts=(namer.variant,))
        if (namer.name, namer.variant) not in got:
            raise SystemExit(f"the {namer.name} encoder did not load: nothing to measure")
        x = got[(namer.name, namer.variant)][namer.feature]
        raw = e3.classify(x, [g.scan for g in glyphs], namer.bank_x, bank_tokens, bank_pages, namer.k)
        truth = [g.base for g in glyphs]
        holds = e3.bank_holds(truth, [g.scan for g in glyphs], bank_tokens, bank_pages)
        m = measure(truth, [p for p, _s in raw], [s for _p, s in raw], holds, namer.threshold, [c["strip"] for c in kept])
        m["skipped"] = dict(skipped)
        m["oracle_set"] = measure([t for t, c in zip(truth, kept) if c["oracle"]],
                                  [p for (p, _s), c in zip(raw, kept) if c["oracle"]],
                                  [s for (_p, s), c in zip(raw, kept) if c["oracle"]],
                                  [h for h, c in zip(holds, kept) if c["oracle"]], namer.threshold,
                                  [c["strip"] for c in kept if c["oracle"]])
        m["tokens"] = per_token(truth, [p for p, _s in raw], [s for _p, s in raw], holds, namer.threshold)
        m["median_similarity"] = sorted(s for _p, s in raw)[len(raw) // 2]
        result["german"][key] = m
        if mode == "p1" and ink == args.ink[0]:
            held_out = (glyphs, kept, x, truth)
        lines += [f"\n  {label}, ink < {ink}: {m['n']} glyphs (skipped {dict(skipped) or 'none'}), median nearest similarity {m['median_similarity']:.3f}",
                  f"    J  judged share        {m['J']:.3f}  [{m['J_ci'][0]:.3f}, {m['J_ci'][1]:.3f}]   Czech {cz['J']:.3f}",
                  f"    A  agree among judged  {m['A']:.3f}  [{m['A_ci'][0]:.3f}, {m['A_ci'][1]:.3f}]   Czech {cz['A']:.3f}",
                  f"    A0 raw 1-NN, in bank   {m['A0']:.3f}  [{m['A0_ci'][0]:.3f}, {m['A0_ci'][1]:.3f}]   Czech {cz['A0']:.3f}"
                  f"   (the Czech bank holds the token for {m['bank_holds']:.2f} of the glyphs)",
                  f"    the 25 pre-designated oracle clicks: n {m['oracle_set']['n']}  J {m['oracle_set']['J']:.3f}  "
                  f"A {m['oracle_set']['A']:.3f}  A0 {m['oracle_set']['A0']:.3f}"]
        lines += [f"      {r['token']:20s} n {r['n']:3d}  in bank {r['in_bank']:3d}  judged {r['J']:.2f}  raw right {r['A0']:.2f}  "
                  f"named otherwise {r['named']}" for r in m["tokens"]]
    if args.german_bank and "held_out" in locals():
        glyphs, kept, x, truth = held_out
        arms = leave_one_page_out(x, glyphs, truth, [c["strip"] for c in kept], namer, bank_tokens, bank_pages)
        result["german_bank"] = arms
        lines.append(f"\n  German exemplars, leave one page out (P1 boxes, ink < {args.ink[0]}); each held-out page "
                     "is named by exemplars from the other two pages, its threshold the 5 % quantile of their correctly named glyphs")
        for arm, m in arms.items():
            lines += [f"    {arm:16s} n {m['n']}  J {m['J']:.3f} [{m['J_ci'][0]:.3f}, {m['J_ci'][1]:.3f}]  "
                      f"A {m['A']:.3f}  A0 {m['A0']:.3f} [{m['A0_ci'][0]:.3f}, {m['A0_ci'][1]:.3f}]   "
                      f"thresholds {m['thresholds']}"]
            lines += [f"      {r['token']:20s} n {r['n']:3d}  judged {r['J']:.2f}  raw right {r['A0']:.2f}" for r in m["tokens"]]
        lines.append(f"    (Czech baseline: J {cz['J']:.3f}  A {cz['A']:.3f}  A0 {cz['A0']:.3f}; gate: J >= {cz['J'] - 0.10:.3f}, A >= {cz['A'] - 0.05:.3f})")
    verdict = []
    for key, m in result["german"].items():
        if m["J"] >= cz["J"] - 0.10 and m["A"] >= cz["A"] - 0.05:
            v = "(a) the bank can judge German marks: fix placement"
        elif m["A0"] < 0.85:
            v = "(b) the bank cannot tell German marks apart: build a German bank"
        elif m["A0"] >= 0.90:
            v = "(c) neighbours are right, the threshold or the crop is wrong"
        else:
            v = "between (b) and (c): A0 in [0.85, 0.90)"
        verdict.append(f"  {key}: {v}")
    lines += ["\nDecision rule, fixed beforehand:"] + verdict
    text = "\n".join(lines)
    print(text)
    OUT.mkdir(parents=True, exist_ok=True)
    stem = f"oracle_{args.encoder}" + "".join(f"_ink{i}" for i in args.ink if i != 128)
    (OUT / f"{stem}.json").write_text(json.dumps(result, indent=1), encoding="utf-8")
    (OUT / f"{stem}.txt").write_text(text + "\n", encoding="utf-8")
    return 0

"""The oracle test: does the Czech-built bank name German glyphs when the box is certainly right?

The author clicked and named 203 pictograms on German gold pages 0142, 0276 and 0480
(``german_click_page``; ``output/german_click_page/clicks``). Each click is cut as a glyph and named
against the **Czech** bank and threshold exactly as E3 names a placed box (``run_e3_pages.Namer``), so
the placement is taken out of the question. The decision configuration is fixed beforehand: the
reader-of-record encoder (``run_glyph_e3.OF_RECORD``: DINOv3 ViT-B/16, base cut, CLS feature, k = 1)
and the Czech gold's ``Namer.threshold`` unchanged. The clicks are queries only: nothing is added to
the bank or tuned on them.

Two ways of getting the box from a click, both reported:
  P1  ``build_glyph_bank.tight_box`` on the click (the ink under it plus close neighbours): does not
      depend on the candidate step; a window cut then takes whatever is near;
  P2  the candidate cluster the click falls in (``locate_glyphs.candidates_on_image``), skipped when
      none: what the route's own box would be, and the gap to P1 is the candidate step's share.

Measured, with a 95 % interval clustered by strip:
  J   share of glyphs whose nearest similarity reaches the threshold (the gate's "judged");
  A   agreement with the clicked name among the judged;
  A0  raw 1-NN accuracy, no threshold, over glyphs whose token the Czech bank holds: tells a
      threshold problem (A0 high, J low) from a discrimination problem (A0 low);
  per token (n >= 5), and the Czech baseline under the same protocol (the bank's own glyphs,
  leave-page-out).
Decision (``GATE_*``): (a) bank fine, fix placement: J >= J_cz - 0.10 and A >= A_cz - 0.05;
(b) build a German bank: A0 below about 0.85; (c) mixed: A0 >= 0.90 but J < J_cz - 0.10.

    python utils/reference_tools/places/run_german_oracle.py --encoder pixels            # dry run, any PC
    python utils/reference_tools/places/run_german_oracle.py --encoder dinov3 --ink 128 140  # Heavyhorse6

Read-only against the gold; writes ``output/glyph_e3_de/oracle_<encoder>.{json,txt}``. Outside
``utils/lexikon_1910`` so as not to change the route's code digests.
"""

__version__ = "0.1.0"

import json
import math
import random
import sys

from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]

sys.path.insert(0, str(ROOT))
sys.path.insert(0, str(LEXIKON))

SECTIONS = ROOT / "output" / "frozen" / "de-bohemia_gold" / "sections"
OUT = ROOT / "output" / "glyph_e3_de"
DRAWS, SEED = 10_000, 1910


def read_clicks() -> list[dict]:
    """Every named click: ``{image, page, strip, oracle, x, y, token}`` in section-image coordinates."""
    from german_click_recall import CLICKS

    out = []
    for path in sorted(CLICKS.glob("*.json")):
        doc = json.loads(path.read_text(encoding="utf-8"))
        for c in doc["clicks"]:
            if c.get("token"):
                out.append({"image": doc["image"], "page": doc["page"], "strip": doc["strip"],
                            "oracle": bool(doc.get("oracle")), "x": c["sx"], "y": c["sy"], "token": c["token"]})
    return out


def click_glyphs(clicks: list[dict], mode: str, ink_threshold: int | None) -> tuple[list, list[dict], Counter]:
    """``(glyphs, their clicks, skipped)`` for ``mode`` ``p1`` or ``p2``."""
    import locate_glyphs as lg
    import run_glyph_e3 as e3
    from PIL import Image

    from build_glyph_bank import line_height, tight_box

    orig = lg.ink_mask
    if ink_threshold:
        lg.ink_mask = lambda image, threshold=ink_threshold: orig(image, threshold)
    images, per_image, skipped = {}, {}, Counter()
    glyphs, kept = [], []
    for n, c in enumerate(clicks):
        name = c["image"]
        if name not in images:
            images[name] = Image.open(SECTIONS / name)
            lh = line_height(images[name].convert("L")) or 46
            cands = lg.candidates_on_image(images[name], name, lh) if mode == "p2" else []
            per_image[name] = (lh, cands, lg.line_neighbours(cands) if cands else {}, lg.ink_mask(images[name]))
        lh, cands, lines, ink = per_image[name]
        modifier_boxes, letter_box = (), None
        if mode == "p1":
            box = tight_box(ink, c["x"], c["y"], lh)
            if box is None:
                skipped["no ink near the click"] += 1
                continue
        else:
            hits = [i for i, k in enumerate(cands) if lg.inside(k.box, c["x"], c["y"], 0.1 * lh) and not k.bracket]
            if not hits:
                skipped["click in no candidate"] += 1
                continue
            i = min(hits, key=lambda i: math.dist(cands[i].centre, (c["x"], c["y"])))
            box = cands[i].box
            m = lg.modifiers(cands, i, lines)
            modifier_boxes = tuple(cands[j].box for j in (m.left, m.right, m.letter) if j is not None)
            letter_box = cands[m.letter].box if m.letter is not None else None
        glyphs.append(e3.Glyph(f"de{c['page']}", "oracle", "offices", "", 0, "", n, c["token"], name,
                               tuple(box), lh, SECTIONS, "click", "", modifier_boxes, letter_box))
        kept.append(c)
    lg.ink_mask = orig
    return glyphs, kept, skipped


def interval(values_by_cluster: dict, stat) -> list[float]:
    """Percentile interval of ``stat(list of per-cluster lists)`` over clusters resampled."""
    clusters = list(values_by_cluster.values())
    rng = random.Random(SEED)
    draws = sorted(stat([rng.choice(clusters) for _ in clusters]) for _ in range(DRAWS))
    return [draws[int(0.025 * DRAWS)], draws[int(0.975 * DRAWS)]]


def measure(truth: list[str], pred: list[str], sim: list[float], holds: list[bool], threshold: float,
            strips: list[str]) -> dict:
    n = len(truth)
    judged = [s >= threshold for s in sim]
    right = [p == t for p, t in zip(pred, truth)]
    j = sum(judged) / n if n else float("nan")
    a = sum(r for r, jd in zip(right, judged) if jd) / max(sum(judged), 1)
    in_bank = [r for r, h in zip(right, holds) if h]
    a0 = sum(in_bank) / max(len(in_bank), 1)
    by_strip = defaultdict(list)
    for k, s in enumerate(strips):
        by_strip[s].append((judged[k], right[k], holds[k]))

    def stat(parts, which):
        rows = [r for part in parts for r in part]
        if which == "J":
            return sum(r[0] for r in rows) / len(rows)
        if which == "A":
            jd = [r for r in rows if r[0]]
            return sum(r[1] for r in jd) / max(len(jd), 1)
        hb = [r for r in rows if r[2]]
        return sum(r[1] for r in hb) / max(len(hb), 1)

    return {"n": n, "J": j, "A": a, "A0": a0, "bank_holds": sum(holds) / max(n, 1),
            "J_ci": interval(by_strip, lambda p: stat(p, "J")), "A_ci": interval(by_strip, lambda p: stat(p, "A")),
            "A0_ci": interval(by_strip, lambda p: stat(p, "A0"))}


def per_token(truth, pred, sim, holds, threshold, minimum=5) -> list[dict]:
    rows = []
    for token in sorted(set(truth), key=lambda t: -truth.count(t)):
        idx = [i for i, t in enumerate(truth) if t == token]
        if len(idx) < minimum:
            continue
        rows.append({"token": token, "n": len(idx), "in_bank": sum(holds[i] for i in idx),
                     "J": sum(sim[i] >= threshold for i in idx) / len(idx),
                     "A0": sum(pred[i] == token for i in idx) / len(idx),
                     "named": dict(Counter(pred[i] for i in idx if pred[i] != token).most_common(3))})
    return rows


def leave_one_page_out(x, glyphs, truth, strips, namer, bank_tokens, bank_pages) -> dict:
    """Each German page named by exemplars from the other pages: German only, and Czech bank + German.

    The threshold of a held-out page is the ``ABSTAIN_QUANTILE`` of the similarity of the *other*
    pages' correctly named glyphs, each named leave-page-out in the same arm: the rule the Czech bank
    uses (``run_glyph_e3.page_thresholds``), so nothing is tuned on the held-out page. Three pages are
    three folds: a direction, not a rate. These are gold pages the route was tuned on."""
    import numpy as np

    import run_glyph_e3 as e3

    pages = [g.scan for g in glyphs]
    n = len(glyphs)
    out = {}
    for arm in ("german only", "czech + german"):
        pred, sim, adj, holds, thresholds = [None] * n, [0.0] * n, [0.0] * n, [False] * n, {}
        for held in sorted(set(pages)):
            train = [i for i, p in enumerate(pages) if p != held]
            test = [i for i, p in enumerate(pages) if p == held]
            bank_x, bank_t, bank_p = x[train], [truth[i] for i in train], [pages[i] for i in train]
            if arm == "czech + german":
                bank_x = np.vstack([namer.bank_x, bank_x])
                bank_t, bank_p = bank_tokens + bank_t, bank_pages + bank_p
            own = e3.classify(x[train], [pages[i] for i in train], bank_x, bank_t, bank_p, namer.k)
            right = [s for (guess, s), i in zip(own, train) if guess == truth[i]]
            threshold = float(np.quantile(right, e3.ABSTAIN_QUANTILE)) if right else -1.0
            thresholds[held] = round(threshold, 3)
            got = e3.classify(x[test], [held] * len(test), bank_x, bank_t, bank_p, namer.k)
            held_tokens = set(bank_t)
            for (guess, s), i in zip(got, test):
                pred[i], sim[i], adj[i], holds[i] = guess, s, s - threshold, truth[i] in held_tokens
        m = measure(truth, pred, adj, holds, 0.0, strips)
        m["thresholds"] = thresholds
        m["tokens"] = per_token(truth, pred, adj, holds, 0.0)
        out[arm] = m
    return out


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--encoder", choices=("pixels", "dinov3"), default="pixels")
    parser.add_argument("--download", action="store_true", help="fetch the DINOv3 weights if not cached")
    parser.add_argument("--ink", type=int, nargs="*", default=[128],
                        help="ink threshold(s) for the German crops, each run in turn (the route's is 128)")
    parser.add_argument("--rounds", type=int, default=4)
    parser.add_argument("--german-bank", action="store_true",
                        help="also: leave-one-page-out with the German clicks as exemplars (Czech only / German only / both)")
    args = parser.parse_args(argv)

    import locate_glyphs as lg
    import run_e3_pages as e3p
    import run_glyph_e3 as e3

    config = e3.OF_RECORD if args.encoder == "dinov3" else e3.FLOOR
    print(f"encoder {config}; threshold from the Czech gold's correctly named glyphs, leave-page-out")
    data = lg.load_sections()
    located, _locators = lg.locate_all(data, args.rounds)
    bank = e3.placed_glyphs(located)
    namer = e3p.Namer(bank, config, args.download)
    print(f"Czech bank: {len(bank)} placed gold glyphs; not-in-bank below {namer.threshold:.3f}")
    bank_tokens, bank_pages = [g.base for g in bank], [g.scan for g in bank]

    # the Czech baseline under the same protocol: the bank's own glyphs, each against the other pages
    raw_cz = e3.classify(namer.bank_x, bank_pages, namer.bank_x, bank_tokens, bank_pages, namer.k)
    cz_truth = [g.base for g in bank]
    cz = measure(cz_truth, [p for p, _s in raw_cz], [s for _p, s in raw_cz],
                 e3.bank_holds(cz_truth, bank_pages, bank_tokens, bank_pages), namer.threshold,
                 [f"{g.scan}-{g.kind}" for g in bank])
    result = {"scorer": __version__, "encoder": list(config), "threshold": namer.threshold, "czech": cz, "german": {}}
    lines = [f"Oracle test, {args.encoder} {config}, Czech threshold {namer.threshold:.3f}",
             f"  Czech baseline (the bank against itself, leave-page-out): n {cz['n']}  J {cz['J']:.3f}  "
             f"A {cz['A']:.3f}  A0 {cz['A0']:.3f}"]

    clicks = read_clicks()
    runs = [(mode, label, ink) for ink in args.ink
            for mode, label in (("p1", "P1 box cut from the click"), ("p2", "P2 the candidate cluster holding the click"))]
    for mode, label, ink in runs:
        key = f"{mode}@ink{ink}"
        glyphs, kept, skipped = click_glyphs(clicks, mode, ink)
        if not glyphs:
            continue
        got = e3.embed(glyphs, [namer.name], args.download, cuts=(namer.variant,))
        if (namer.name, namer.variant) not in got:
            raise SystemExit(f"the {namer.name} encoder did not load: nothing to measure")
        x = got[(namer.name, namer.variant)][namer.feature]
        raw = e3.classify(x, [g.scan for g in glyphs], namer.bank_x, bank_tokens, bank_pages, namer.k)
        truth = [g.base for g in glyphs]
        holds = e3.bank_holds(truth, [g.scan for g in glyphs], bank_tokens, bank_pages)
        m = measure(truth, [p for p, _s in raw], [s for _p, s in raw], holds, namer.threshold, [c["strip"] for c in kept])
        m["skipped"] = dict(skipped)
        m["oracle_set"] = measure([t for t, c in zip(truth, kept) if c["oracle"]],
                                  [p for (p, _s), c in zip(raw, kept) if c["oracle"]],
                                  [s for (_p, s), c in zip(raw, kept) if c["oracle"]],
                                  [h for h, c in zip(holds, kept) if c["oracle"]], namer.threshold,
                                  [c["strip"] for c in kept if c["oracle"]])
        m["tokens"] = per_token(truth, [p for p, _s in raw], [s for _p, s in raw], holds, namer.threshold)
        m["median_similarity"] = sorted(s for _p, s in raw)[len(raw) // 2]
        result["german"][key] = m
        if mode == "p1" and ink == args.ink[0]:
            held_out = (glyphs, kept, x, truth)
        lines += [f"\n  {label}, ink < {ink}: {m['n']} glyphs (skipped {dict(skipped) or 'none'}), median nearest similarity {m['median_similarity']:.3f}",
                  f"    J  judged share        {m['J']:.3f}  [{m['J_ci'][0]:.3f}, {m['J_ci'][1]:.3f}]   Czech {cz['J']:.3f}",
                  f"    A  agree among judged  {m['A']:.3f}  [{m['A_ci'][0]:.3f}, {m['A_ci'][1]:.3f}]   Czech {cz['A']:.3f}",
                  f"    A0 raw 1-NN, in bank   {m['A0']:.3f}  [{m['A0_ci'][0]:.3f}, {m['A0_ci'][1]:.3f}]   Czech {cz['A0']:.3f}"
                  f"   (the Czech bank holds the token for {m['bank_holds']:.2f} of the glyphs)",
                  f"    the 25 pre-designated oracle clicks: n {m['oracle_set']['n']}  J {m['oracle_set']['J']:.3f}  "
                  f"A {m['oracle_set']['A']:.3f}  A0 {m['oracle_set']['A0']:.3f}"]
        lines += [f"      {r['token']:20s} n {r['n']:3d}  in bank {r['in_bank']:3d}  judged {r['J']:.2f}  raw right {r['A0']:.2f}  "
                  f"named otherwise {r['named']}" for r in m["tokens"]]
    if args.german_bank and "held_out" in locals():
        glyphs, kept, x, truth = held_out
        arms = leave_one_page_out(x, glyphs, truth, [c["strip"] for c in kept], namer, bank_tokens, bank_pages)
        result["german_bank"] = arms
        lines.append(f"\n  German exemplars, leave one page out (P1 boxes, ink < {args.ink[0]}); each held-out page "
                     "is named by exemplars from the other two pages, its threshold the 5 % quantile of their correctly named glyphs")
        for arm, m in arms.items():
            lines += [f"    {arm:16s} n {m['n']}  J {m['J']:.3f} [{m['J_ci'][0]:.3f}, {m['J_ci'][1]:.3f}]  "
                      f"A {m['A']:.3f}  A0 {m['A0']:.3f} [{m['A0_ci'][0]:.3f}, {m['A0_ci'][1]:.3f}]   "
                      f"thresholds {m['thresholds']}"]
            lines += [f"      {r['token']:20s} n {r['n']:3d}  judged {r['J']:.2f}  raw right {r['A0']:.2f}" for r in m["tokens"]]
        lines.append(f"    (Czech baseline: J {cz['J']:.3f}  A {cz['A']:.3f}  A0 {cz['A0']:.3f}; gate: J >= {cz['J'] - 0.10:.3f}, A >= {cz['A'] - 0.05:.3f})")
    verdict = []
    for key, m in result["german"].items():
        if m["J"] >= cz["J"] - 0.10 and m["A"] >= cz["A"] - 0.05:
            v = "(a) the bank can judge German marks: fix placement"
        elif m["A0"] < 0.85:
            v = "(b) the bank cannot tell German marks apart: build a German bank"
        elif m["A0"] >= 0.90:
            v = "(c) neighbours are right, the threshold or the crop is wrong"
        else:
            v = "between (b) and (c): A0 in [0.85, 0.90)"
        verdict.append(f"  {key}: {v}")
    lines += ["\nDecision rule, fixed beforehand:"] + verdict
    text = "\n".join(lines)
    print(text)
    OUT.mkdir(parents=True, exist_ok=True)
    stem = f"oracle_{args.encoder}" + "".join(f"_ink{i}" for i in args.ink if i != 128)
    (OUT / f"{stem}.json").write_text(json.dumps(result, indent=1), encoding="utf-8")
    (OUT / f"{stem}.txt").write_text(text + "\n", encoding="utf-8")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
