How it works

For every entry the place matcher left unmatched (data/interim/matice/ matice_entries.csv), cut its line(s) out of the page scan, using the ALTO line coordinates, and stack them into numbered sheets of PER_SHEET. A Matice česká founder fills one line; a Matice moravská entry may run over three, so three lines are cut and the entry's first words are printed beside the number to say which entry is meant.

Writes data/interim/matice/read_sheets/sheet_NNN.png and manifest.csv (sheet, number, entry_id, the OCR's residence). The readings made from the sheets go to reference/places/matice_residence_readings.csv and are applied by run_matice_pilot.py. The 1878 list (a PDF without a page cache) is not cut.

Usage::

python utils/reference_tools/places/build_matice_read_sheets.py

    """The scanned line(s) of one entry: its first line, and two more for a run-on list."""
    pages = LIBRARY / f"{mt.SOURCES[entry['source']]['stem']}_pages"

def build_sheets(entries: pd.DataFrame) -> pd.DataFrame:
    font = ImageFont.truetype(str(FONT), 20) if FONT.exists() else ImageFont.load_default()
    OUT.mkdir(parents=True, exist_ok=True)
    rows = []
    for s, start in enumerate(range(0, len(entries), PER_SHEET), 1):
        strips = []
        for n, (_, e) in enumerate(entries.iloc[start:start + PER_SHEET].iterrows(), 1):
            crop = crop_entry(e)
            strip = Image.new("L", (LABEL_W + crop.size[0], crop.size[1] + 6), 255)
            strip.paste(label(n, e, crop.size[1], font), (0, 0))
            strip.paste(crop, (LABEL_W, 0))
            strips.append(strip)
            rows.append({"sheet": s, "number": n, "entry_id": e["entry_id"],
                         "residence_ocr": e["residence"]})
        sheet = Image.new("L", (max(x.size[0] for x in strips), sum(x.size[1] for x in strips)), 255)
        y = 0
        for x in strips:
            sheet.paste(x, (0, y))
            y += x.size[1]
        if sheet.size[0] > 1600:                      # keep each sheet under ~1600 px wide
            sheet = sheet.resize((1600, int(sheet.size[1] * 1600 / sheet.size[0])))
        sheet.save(OUT / f"sheet_{s:03d}.png")
    return pd.DataFrame(rows)

"""
Sheets of scanned lines for re-reading the Matice residences the OCR garbled.

For every entry the place matcher left ``unmatched`` (``data/interim/matice/
matice_entries.csv``), cut its line(s) out of the page scan, using the ALTO line
coordinates, and stack them into numbered sheets of ``PER_SHEET``. A Matice česká founder

the entry's first words are printed beside the number to say which entry is meant.

``reference/places/matice_residence_readings.csv`` and are applied by

    python utils/reference_tools/places/build_matice_read_sheets.py

from __future__ import annotations

import re
import sys
from pathlib import Path

import pandas as pd
from PIL import Image, ImageDraw, ImageFont

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

from utils.parsing import matice as mt                                  # noqa: E402

__version__ = "0.1.0"

LIBRARY = ROOT / "library" / "associations"
INTERIM = ROOT / "data" / "interim" / "matice"
OUT = INTERIM / "read_sheets"
PER_SHEET = 20
LABEL_W = 260                 # px of label column left of each crop
FONT = Path("C:/Windows/Fonts/ARIALN.TTF")


def alto_boxes(alto: Path) -> tuple[int, list[tuple[int, int, int, int]]]:

        boxes.append((int(a["HPOS"]), int(a["VPOS"]), int(a["WIDTH"]), int(a["HEIGHT"])))
    return width, boxes


def crop_entry(entry: pd.Series) -> Image.Image:
    """The scanned line(s) of one entry: its first line, and two more for a run-on list."""
    pages = LIBRARY / f"{mt.SOURCES[entry['source']]['stem']}_pages"
    page, line = int(entry["page"]), int(entry["line"])

    x0 = min(b[0] for b in sel) - 10
    x1 = max(b[0] + b[2] for b in sel) + 10

    y1 = max(b[1] + b[3] for b in sel) + 8
    return img.crop((int(x0 * k), int(y0 * k), int(x1 * k), int(y1 * k)))


def label(number: int, entry: pd.Series, height: int, font) -> Image.Image:

def build_sheets(entries: pd.DataFrame) -> pd.DataFrame:
    font = ImageFont.truetype(str(FONT), 20) if FONT.exists() else ImageFont.load_default()
    OUT.mkdir(parents=True, exist_ok=True)
    rows = []
    for s, start in enumerate(range(0, len(entries), PER_SHEET), 1):
        strips = []
        for n, (_, e) in enumerate(entries.iloc[start:start + PER_SHEET].iterrows(), 1):
            crop = crop_entry(e)
            strip = Image.new("L", (LABEL_W + crop.size[0], crop.size[1] + 6), 255)
            strip.paste(label(n, e, crop.size[1], font), (0, 0))
            strip.paste(crop, (LABEL_W, 0))
            strips.append(strip)

    return pd.DataFrame(rows)


def main() -> int:
    e = pd.read_csv(INTERIM / "matice_entries.csv", dtype=str, keep_default_na=False)
    todo = e[(e["place_status"] == "unmatched") & e["source"].map(

if __name__ == "__main__":
    sys.exit(main())
