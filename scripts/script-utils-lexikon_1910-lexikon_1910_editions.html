How it works

The Czech Podrobný seznam míst and the German Spezialortsrepertorium print the same institutions for the same municipalities. Read by two readers, they are two error-prone measurements of one binary fact — municipality m has institution c — and the two-measure identification of a misclassified regressor (Kane, Rouse and Staiger; Black, Berger and Scott) needs the two errors to be independent given the truth. D-58 makes the German edition a second measurement only if that holds on the gold; this module is the test.

The pairing is by court and municipality name, never by page (D-67). Each edition numbers a court's municipalities in its own alphabetical order, so municipality 23 of Trutnov in Czech is 1 in German, and the municipalities of one Czech page fall on two or three German scans. A municipality is paired when its two printed names share a form — Pilníkov, Pilnikau and Pilnikau, Pilnikov — inside the same court, one to one, and a name that would pair with two municipalities is refused rather than given to either.

The unit is the municipality, not the settlement. Parts are numbered within a municipality in the edition's own order too, and the two editions do not always split a municipality into the same parts, so a settlement-level pairing would be a second name join on worse names. Claims are pooled over a municipality's parts and pages on both sides.

The truth is each edition's own gold. A claim both golds make is a positive, a claim neither makes but some reader does is a negative, and a claim only one gold makes is an edition disagreement — a difference between the two prints, or a slip of the transcriber's — which is listed and kept out of the test, since neither reader can be wrong about it in the same sense.

The test, for a Czech pass A and a German pass B, inside each truth class: the joint errors J = Σ e_A · e_B against what independence predicts, with a permutation p-value from shuffling B's errors (Dell and Rambachan's check). It is run twice — permuting within the truth class, and within (truth class, symbol). The second is the one that matters: two readers who both find one glyph hard err together on it while erring independently given the glyph, and pooling over glyphs reads that as dependence. A pair passes D-58 only on the second; the first is printed beside it so the size of the glyph effect shows.

def pair_units(cz: dict, de: dict) -> tuple[list[tuple], list[dict]]:
    """Pair Czech and German municipalities by court and name.

    ``cz`` and ``de`` are ``{(court slug, muni_seq): printed name}``. Returns
    ``(pairs, refused)``: ``pairs`` is ``[(cz unit, de unit)]``, one to one;
    ``refused`` lists every unit whose name matched more than one municipality
    on the other side, which is left unpaired.
    """
    cz_forms = {unit: name_forms(name) for unit, name in cz.items()}
    de_forms = {unit: name_forms(name) for unit, name in de.items()}
    matches: dict = {}
    for c, cf in cz_forms.items():
        matches[c] = [d for d, df in de_forms.items()
                      if d[0] == c[0] and cf & df]
    back = Counter(d for found in matches.values() for d in found)
    pairs, refused = [], []
    for c, found in sorted(matches.items(), key=lambda kv: (str(kv[0][0]), kv[0][1])):
        if len(found) == 1 and back[found[0]] == 1:
            pairs.append((c, found[0]))
        elif found:
            refused.append({"cz_unit": c, "cz_name": cz[c],
                            "de_candidates": [de[d] for d in found]})
    return pairs, refused

def pair_by_forms(cz: dict, de: dict) -> tuple[list[tuple], list[dict]]:
    """Pair municipalities known by **every** name form they print, not by one name.

    ``cz`` and ``de`` are ``{(court, muni_seq): set of name forms}`` — on a route's output,
    every part name the reader listed for the municipality and, where the edition's table
    was read, the municipality's own name. A municipality with only numbered parts prints
    no part-0 line, so one name per municipality left most of them unpaired (the gold
    re-run of 2026-10-25: 22 pairs where the pooled forms give 49). Each Czech unit takes
    the German unit of its own court with the highest share of shared forms (Jaccard), and
    the pair stands only if that best is strictly best on both sides; anything else is
    refused, as in :func:`pair_units`.
    """
    def jac(a, b):
        return len(a & b) / len(a | b)

    cand = {c: {d: jac(cf, df) for d, df in de.items() if d[0] == c[0] and cf & df}
            for c, cf in cz.items()}
    pairs, refused = [], []
    for c, ds in sorted(cand.items(), key=lambda kv: (str(kv[0][0]), kv[0][1])):
        if not ds:
            continue
        ranked = sorted(ds.items(), key=lambda kv: -kv[1])
        d, s = ranked[0]
        unique_here = len(ranked) == 1 or s > ranked[1][1]
        unique_there = all(other[d] < s for c2, other in cand.items() if c2 != c and d in other)
        if unique_here and unique_there:
            pairs.append((c, d))
        else:
            refused.append({"cz_unit": c, "cz_name": " / ".join(sorted(cz[c])),
                            "de_candidates": [" / ".join(sorted(de[x])) for x, _s in ranked]})
    return pairs, refused

def unit_claims(rows: list[dict]) -> tuple[set, dict]:
    """``(units read, {unit: set of codes})`` at municipality level, over pages.

    A unit is ``(court slug, muni_seq)``. A unit is read when the pass has any
    row for it, claimed symbol or not, the scorer's own definition of an entry.
    """
    read, claims = set(), {}
    for row in rows:
        muni = _muni(row)
        if muni is None:
            continue
        unit = (str(row.get("district", "")), muni)
        read.add(unit)
        code = str(row.get("code", "") or "")
        if code.startswith("[") and code != UNPLACED:
            claims.setdefault(unit, set()).add(code)
    return read, claims

def truth_items(pairs: list[tuple], cz_gold: dict, de_gold: dict,
                universe: dict) -> tuple[list[tuple], list[dict]]:
    """``[(pair index, code, truth)]`` with truth 1 or 0, and the disagreements.

    ``universe`` is ``{pair index: codes some reader claimed there}``. Both
    golds must have read the pair (``cz_gold``/``de_gold`` hold the units read
    as keys, empty set where nothing was claimed).
    """
    items, disagree = [], []
    for k, (c, d) in enumerate(pairs):
        if c not in cz_gold or d not in de_gold:
            continue
        gc, gd = cz_gold[c], de_gold[d]
        for code in sorted(gc | gd | universe.get(k, set())):
            if (code in gc) != (code in gd):
                disagree.append({"cz_unit": c, "de_unit": d, "code": code,
                                 "in": "cz" if code in gc else "de"})
                continue
            items.append((k, code, int(code in gc)))
    return items, disagree

def joint_errors(e_a: list[int], e_b: list[int], strata: list, draws: int = DRAWS,
                 seed: int = SEED) -> dict:
    """The joint-error count against independence, with a permutation p-value.

    B's errors are shuffled within each stratum; the p-value is the share of
    draws (plus the observed one) with at least the observed joint count —
    one-sided, because errors that coincide are the threat to the design.
    """
    n = len(e_a)
    observed = sum(a * b for a, b in zip(e_a, e_b))
    groups: dict = {}
    for i, s in enumerate(strata):
        groups.setdefault(s, []).append(i)
    expected = sum(sum(e_a[i] for i in idx) * sum(e_b[i] for i in idx) / len(idx)
                   for idx in groups.values())
    rng = random.Random(seed)
    at_least = 0
    for _ in range(draws):
        total = 0
        for idx in groups.values():
            shuffled = [e_b[i] for i in idx]
            rng.shuffle(shuffled)
            total += sum(e_a[i] * b for i, b in zip(idx, shuffled))
        at_least += total >= observed
    na, nb = sum(e_a), sum(e_b)
    denom = (na * (n - na) * nb * (n - nb)) ** 0.5
    phi = ((n * observed - na * nb) / denom) if denom else 0.0
    return {"n": n, "errors_a": na, "errors_b": nb, "joint": observed,
            "expected": round(expected, 2), "phi": round(phi, 3),
            "p_value": round((1 + at_least) / (1 + draws), 4) if draws else None}

def compare_pair(items: list[tuple], pairs: list[tuple], cz_pass: tuple,
                 de_pass: tuple, draws: int = DRAWS) -> list[dict]:
    """One Czech pass against one German pass: a row per truth class.

    ``cz_pass`` and ``de_pass`` are :func:`unit_claims` results. Only pairs
    both passes read are tested. ``positive`` rows test misses, ``negative``
    rows false claims.
    """
    (a_read, a_claims), (b_read, b_claims) = cz_pass, de_pass
    out = []
    for truth, cls in ((1, "positive"), (0, "negative")):
        e_a, e_b, by_class, by_code = [], [], [], []
        for k, code, t in items:
            c, d = pairs[k]
            if t != truth or c not in a_read or d not in b_read:
                continue
            said_a = code in a_claims.get(c, set())
            said_b = code in b_claims.get(d, set())
            e_a.append(int(said_a != bool(t)))
            e_b.append(int(said_b != bool(t)))
            by_class.append(cls)
            by_code.append(code)
        pooled = joint_errors(e_a, e_b, by_class, draws)
        within = joint_errors(e_a, e_b, by_code, draws)
        out.append({"truth": cls, **pooled,
                    "p_within_symbol": within["p_value"],
                    "expected_within_symbol": within["expected"],
                    "independent": (within["p_value"] is None
                                    or within["p_value"] >= ALPHA)})
    return out

def keep_codes(read_claims: tuple, shared) -> tuple:
    """A :func:`unit_claims` result with only the codes in ``shared`` (all if None)."""

        return read, claims

def compare_editions(cz_gold_rows: list[dict], de_gold_rows: list[dict],
                     cz_names: dict, de_names: dict, cz_passes: dict,
                     de_passes: dict, family=lambda label: label,
                     draws: int = DRAWS, shared=None) -> dict:
    """The whole comparison: pairing, truth, and every Czech × German pass pair.

    ``cz_names``/``de_names`` are ``{(court slug, muni_seq): printed name}``
    from each edition's gold sheets; ``*_passes`` are ``{label: inventory rows}``.
    ``family(label)`` names a pass's reader, so a row says whether the two
    readers were of one family (the control, expected to fail) or two.

    ``shared`` is the set of tokens **both** editions print. A symbol only one
    plate has — the Czech ``[cooperative]``, which the German volume writes as
    letters alone — cannot be measured twice, and left in it would turn up as an
    edition disagreement on every municipality with a cooperative.
    """
    pairs, refused = pair_units(cz_names, de_names)
    _cr, cz_gold = keep_codes(unit_claims(cz_gold_rows), shared)
    _dr, de_gold = keep_codes(unit_claims(de_gold_rows), shared)
    cz_gold = {u: cz_gold.get(u, set()) for u in _cr}
    de_gold = {u: de_gold.get(u, set()) for u in _dr}
    cz_read = {label: keep_codes(unit_claims(rows), shared)
               for label, rows in cz_passes.items()}
    de_read = {label: keep_codes(unit_claims(rows), shared)
               for label, rows in de_passes.items()}
    universe: dict = {}
    for k, (c, d) in enumerate(pairs):
        codes = set()
        for _read, claims in cz_read.values():
            codes |= claims.get(c, set())
        for _read, claims in de_read.values():
            codes |= claims.get(d, set())
        universe[k] = codes
    items, disagree = truth_items(pairs, cz_gold, de_gold, universe)
    rows = []
    for a in sorted(cz_read):
        for b in sorted(de_read):
            same = family(a) == family(b)
            for row in compare_pair(items, pairs, cz_read[a], de_read[b], draws):
                rows.append({"cz_pass": a, "de_pass": b, "cz_reader": family(a),
                             "de_reader": family(b),
                             "kind": "same family" if same else "two families",
                             **row})
    return {"pairs": pairs, "refused": refused, "items": items,
            "disagreements": disagree, "rows": rows,
            "gold_pairs": sum(1 for c, d in pairs if c in cz_gold and d in de_gold)}

def lexikon_1910_editions_report(result: dict) -> str:
    """Ledger for :func:`compare_editions`."""
    items = result["items"]
    lines = [f"lexikon_1910_editions v{__version__} — the Czech and German editions as "
             f"two measurements (D-58)", "",
             f"  municipalities paired by court and name: {len(result['pairs'])}, "
             f"{result['gold_pairs']} of them read in both golds; "
             f"{len(result['refused'])} names refused as ambiguous",
             f"  claims: {sum(1 for i in items if i[2] == 1)} positive (both golds), "
             f"{sum(1 for i in items if i[2] == 0)} negative (neither gold, some "
             f"reader), {len(result['disagreements'])} edition disagreements (one "
             f"gold only — kept out of the test)"]
    for d in result["disagreements"][:10]:
        lines.append(f"      {d['cz_unit']} ~ {d['de_unit']}  {d['code']} only in "
                     f"the {d['in']} gold")
    if not result["rows"]:
        lines += ["", "  no Czech × German pass pair on disk yet"]
        return "\n".join(lines)
    lines += ["", f"  {'Czech pass':24s} {'German pass':26s} {'truth':9s} {'n':>5s} "
                  f"{'errA':>5s} {'errB':>5s} {'joint':>5s} {'exp':>6s} {'phi':>6s} "
                  f"{'p':>6s} {'p|sym':>6s}  verdict"]
    for r in result["rows"]:
        verdict = ("—" if not r["n"] else
                   "independent" if r["independent"] else "DEPENDENT given symbol")
        lines.append(f"  {r['cz_pass']:24s} {r['de_pass']:26s} {r['truth']:9s} "
                     f"{r['n']:5d} {r['errors_a']:5d} {r['errors_b']:5d} "
                     f"{r['joint']:5d} {r['expected']:6.2f} {r['phi']:6.3f} "
                     f"{r['p_value'] if r['p_value'] is not None else '':>6} "
                     f"{r['p_within_symbol'] if r['p_within_symbol'] is not None else '':>6}"
                     f"  {verdict} ({r['kind']})")
    lines += ["", "  `p` permutes the German errors within the truth class, `p|sym` "
                  "within (truth class, symbol);",
              f"  D-58 reads `p|sym` at {ALPHA}. Municipality level; the pages are "
              f"few, so read a pass as weak evidence."]
    return "\n".join(lines)

"""The two editions of the 1910 register as two measurements of one truth (D-58, D-92).

The Czech *Podrobný seznam míst* and the German *Spezialortsrepertorium* print the
same institutions for the same municipalities. Read by two readers, they are two
error-prone measurements of one binary fact — *municipality m has institution c*
— and the two-measure identification of a misclassified regressor (Kane, Rouse
and Staiger; Black, Berger and Scott) needs the two errors to be **independent
given the truth**. D-58 makes the German edition a second measurement only if that
holds on the gold; this module is the test.

**The pairing is by court and municipality name, never by page** (D-67). Each
edition numbers a court's municipalities in its own alphabetical order, so
municipality 23 of Trutnov in Czech is 1 in German, and the municipalities of
one Czech page fall on two or three German scans. A municipality is paired when
its two printed names share a form — ``Pilníkov, Pilnikau`` and ``Pilnikau,
Pilnikov`` — inside the same court, one to one, and a name that would pair with
two municipalities is refused rather than given to either.

**The unit is the municipality, not the settlement.** Parts are numbered within
a municipality in the edition's own order too, and the two editions do not
always split a municipality into the same parts, so a settlement-level pairing
would be a second name join on worse names. Claims are pooled over a
municipality's parts and pages on both sides.

The truth is each edition's own gold. A claim both golds make is a **positive**,
a claim neither makes but some reader does is a **negative**, and a claim only
one gold makes is an **edition disagreement** — a difference between the two
prints, or a slip of the transcriber's — which is listed and kept out of the
test, since neither reader can be wrong about it in the same sense.

The test, for a Czech pass A and a German pass B, inside each truth class:
the joint errors ``J = Σ e_A · e_B`` against what independence predicts, with a
permutation p-value from shuffling B's errors (Dell and Rambachan's check). It is
run twice — permuting within the truth class, and within *(truth class,
symbol)*. The second is the one that matters: two readers who both find one
glyph hard err together on it while erring independently given the glyph, and
pooling over glyphs reads that as dependence. A pair passes D-58 only on the
second; the first is printed beside it so the size of the glyph effect shows.
"""

from __future__ import annotations

__version__ = "0.2.0"

import random
import re
from collections import Counter

from . import lexikon_1910_notes as lxnt

#: The token a reader writes for a symbol it cannot place. Never a claim.
UNPLACED = "[?]"

_BRACKET_RE = re.compile(r"\[[^\]]*\]|\*+\)|\([^)]*\)")

#: Permutation draws, and the seed the project's other bootstraps use.
DRAWS = 2000
SEED = 1910

#: The p-value below which a pair is reported as dependent given the truth.
ALPHA = 0.05


def name_forms(name: str) -> frozenset[str]:
    """The normalised forms a printed municipality name is known by.

    ``Pilnikau, Pilnikov [mesto]`` -> ``{pilnikau, pilnikov}``: the place-type
    token, a footnote mark and a parenthesis go, each comma-separated form is
    normalised (no accents, letters only), and a form under three letters is
    dropped as too short to identify anything.
    """
    clean = _BRACKET_RE.sub(" ", str(name or ""))
    forms = {lxnt.normalise(part) for part in clean.split(",")}
    return frozenset(form for form in forms if len(form) >= 3)


def pair_units(cz: dict, de: dict) -> tuple[list[tuple], list[dict]]:
    """Pair Czech and German municipalities by court and name.

    ``cz`` and ``de`` are ``{(court slug, muni_seq): printed name}``. Returns
    ``(pairs, refused)``: ``pairs`` is ``[(cz unit, de unit)]``, one to one;
    ``refused`` lists every unit whose name matched more than one municipality
    on the other side, which is left unpaired.
    """
    cz_forms = {unit: name_forms(name) for unit, name in cz.items()}
    de_forms = {unit: name_forms(name) for unit, name in de.items()}
    matches: dict = {}
    for c, cf in cz_forms.items():
        matches[c] = [d for d, df in de_forms.items()
                      if d[0] == c[0] and cf & df]
    back = Counter(d for found in matches.values() for d in found)
    pairs, refused = [], []
    for c, found in sorted(matches.items(), key=lambda kv: (str(kv[0][0]), kv[0][1])):
        if len(found) == 1 and back[found[0]] == 1:
            pairs.append((c, found[0]))
        elif found:
            refused.append({"cz_unit": c, "cz_name": cz[c],
                            "de_candidates": [de[d] for d in found]})
    return pairs, refused


def pair_by_forms(cz: dict, de: dict) -> tuple[list[tuple], list[dict]]:
    """Pair municipalities known by **every** name form they print, not by one name.

    ``cz`` and ``de`` are ``{(court, muni_seq): set of name forms}`` — on a route's output,
    every part name the reader listed for the municipality and, where the edition's table
    was read, the municipality's own name. A municipality with only numbered parts prints
    no part-0 line, so one name per municipality left most of them unpaired (the gold
    re-run of 2026-10-25: 22 pairs where the pooled forms give 49). Each Czech unit takes
    the German unit of its own court with the highest share of shared forms (Jaccard), and
    the pair stands only if that best is strictly best on both sides; anything else is
    refused, as in :func:`pair_units`.
    """
    def jac(a, b):
        return len(a & b) / len(a | b)

    cand = {c: {d: jac(cf, df) for d, df in de.items() if d[0] == c[0] and cf & df}
            for c, cf in cz.items()}
    pairs, refused = [], []
    for c, ds in sorted(cand.items(), key=lambda kv: (str(kv[0][0]), kv[0][1])):
        if not ds:
            continue
        ranked = sorted(ds.items(), key=lambda kv: -kv[1])
        d, s = ranked[0]
        unique_here = len(ranked) == 1 or s > ranked[1][1]
        unique_there = all(other[d] < s for c2, other in cand.items() if c2 != c and d in other)
        if unique_here and unique_there:
            pairs.append((c, d))
        else:
            refused.append({"cz_unit": c, "cz_name": " / ".join(sorted(cz[c])),
                            "de_candidates": [" / ".join(sorted(de[x])) for x, _s in ranked]})
    return pairs, refused


def _muni(row: dict):
    try:
        return int(str(row.get("muni_seq", "")).strip())
    except ValueError:
        return None


def unit_claims(rows: list[dict]) -> tuple[set, dict]:
    """``(units read, {unit: set of codes})`` at municipality level, over pages.

    A unit is ``(court slug, muni_seq)``. A unit is read when the pass has any
    row for it, claimed symbol or not, the scorer's own definition of an entry.
    """
    read, claims = set(), {}
    for row in rows:
        muni = _muni(row)
        if muni is None:
            continue
        unit = (str(row.get("district", "")), muni)
        read.add(unit)
        code = str(row.get("code", "") or "")
        if code.startswith("[") and code != UNPLACED:
            claims.setdefault(unit, set()).add(code)
    return read, claims


def truth_items(pairs: list[tuple], cz_gold: dict, de_gold: dict,
                universe: dict) -> tuple[list[tuple], list[dict]]:
    """``[(pair index, code, truth)]`` with truth 1 or 0, and the disagreements.

    ``universe`` is ``{pair index: codes some reader claimed there}``. Both
    golds must have read the pair (``cz_gold``/``de_gold`` hold the units read
    as keys, empty set where nothing was claimed).
    """
    items, disagree = [], []
    for k, (c, d) in enumerate(pairs):
        if c not in cz_gold or d not in de_gold:
            continue
        gc, gd = cz_gold[c], de_gold[d]
        for code in sorted(gc | gd | universe.get(k, set())):
            if (code in gc) != (code in gd):
                disagree.append({"cz_unit": c, "de_unit": d, "code": code,
                                 "in": "cz" if code in gc else "de"})
                continue
            items.append((k, code, int(code in gc)))
    return items, disagree


def joint_errors(e_a: list[int], e_b: list[int], strata: list, draws: int = DRAWS,
                 seed: int = SEED) -> dict:
    """The joint-error count against independence, with a permutation p-value.

    B's errors are shuffled within each stratum; the p-value is the share of
    draws (plus the observed one) with at least the observed joint count —
    one-sided, because errors that coincide are the threat to the design.
    """
    n = len(e_a)
    observed = sum(a * b for a, b in zip(e_a, e_b))
    groups: dict = {}
    for i, s in enumerate(strata):
        groups.setdefault(s, []).append(i)
    expected = sum(sum(e_a[i] for i in idx) * sum(e_b[i] for i in idx) / len(idx)
                   for idx in groups.values())
    rng = random.Random(seed)
    at_least = 0
    for _ in range(draws):
        total = 0
        for idx in groups.values():
            shuffled = [e_b[i] for i in idx]
            rng.shuffle(shuffled)
            total += sum(e_a[i] * b for i, b in zip(idx, shuffled))
        at_least += total >= observed
    na, nb = sum(e_a), sum(e_b)
    denom = (na * (n - na) * nb * (n - nb)) ** 0.5
    phi = ((n * observed - na * nb) / denom) if denom else 0.0
    return {"n": n, "errors_a": na, "errors_b": nb, "joint": observed,
            "expected": round(expected, 2), "phi": round(phi, 3),
            "p_value": round((1 + at_least) / (1 + draws), 4) if draws else None}


def compare_pair(items: list[tuple], pairs: list[tuple], cz_pass: tuple,
                 de_pass: tuple, draws: int = DRAWS) -> list[dict]:
    """One Czech pass against one German pass: a row per truth class.

    ``cz_pass`` and ``de_pass`` are :func:`unit_claims` results. Only pairs
    both passes read are tested. ``positive`` rows test misses, ``negative``
    rows false claims.
    """
    (a_read, a_claims), (b_read, b_claims) = cz_pass, de_pass
    out = []
    for truth, cls in ((1, "positive"), (0, "negative")):
        e_a, e_b, by_class, by_code = [], [], [], []
        for k, code, t in items:
            c, d = pairs[k]
            if t != truth or c not in a_read or d not in b_read:
                continue
            said_a = code in a_claims.get(c, set())
            said_b = code in b_claims.get(d, set())
            e_a.append(int(said_a != bool(t)))
            e_b.append(int(said_b != bool(t)))
            by_class.append(cls)
            by_code.append(code)
        pooled = joint_errors(e_a, e_b, by_class, draws)
        within = joint_errors(e_a, e_b, by_code, draws)
        out.append({"truth": cls, **pooled,
                    "p_within_symbol": within["p_value"],
                    "expected_within_symbol": within["expected"],
                    "independent": (within["p_value"] is None
                                    or within["p_value"] >= ALPHA)})
    return out


def keep_codes(read_claims: tuple, shared) -> tuple:
    """A :func:`unit_claims` result with only the codes in ``shared`` (all if None)."""
    read, claims = read_claims
    if shared is None:
        return read, claims
    return read, {u: {c for c in codes if c in shared} for u, codes in claims.items()}


def compare_editions(cz_gold_rows: list[dict], de_gold_rows: list[dict],
                     cz_names: dict, de_names: dict, cz_passes: dict,
                     de_passes: dict, family=lambda label: label,
                     draws: int = DRAWS, shared=None) -> dict:
    """The whole comparison: pairing, truth, and every Czech × German pass pair.

    ``cz_names``/``de_names`` are ``{(court slug, muni_seq): printed name}``
    from each edition's gold sheets; ``*_passes`` are ``{label: inventory rows}``.
    ``family(label)`` names a pass's reader, so a row says whether the two
    readers were of one family (the control, expected to fail) or two.

    ``shared`` is the set of tokens **both** editions print. A symbol only one
    plate has — the Czech ``[cooperative]``, which the German volume writes as
    letters alone — cannot be measured twice, and left in it would turn up as an
    edition disagreement on every municipality with a cooperative.
    """
    pairs, refused = pair_units(cz_names, de_names)
    _cr, cz_gold = keep_codes(unit_claims(cz_gold_rows), shared)
    _dr, de_gold = keep_codes(unit_claims(de_gold_rows), shared)
    cz_gold = {u: cz_gold.get(u, set()) for u in _cr}
    de_gold = {u: de_gold.get(u, set()) for u in _dr}
    cz_read = {label: keep_codes(unit_claims(rows), shared)
               for label, rows in cz_passes.items()}
    de_read = {label: keep_codes(unit_claims(rows), shared)
               for label, rows in de_passes.items()}
    universe: dict = {}
    for k, (c, d) in enumerate(pairs):
        codes = set()
        for _read, claims in cz_read.values():
            codes |= claims.get(c, set())
        for _read, claims in de_read.values():
            codes |= claims.get(d, set())
        universe[k] = codes
    items, disagree = truth_items(pairs, cz_gold, de_gold, universe)
    rows = []
    for a in sorted(cz_read):
        for b in sorted(de_read):
            same = family(a) == family(b)
            for row in compare_pair(items, pairs, cz_read[a], de_read[b], draws):
                rows.append({"cz_pass": a, "de_pass": b, "cz_reader": family(a),
                             "de_reader": family(b),
                             "kind": "same family" if same else "two families",
                             **row})
    return {"pairs": pairs, "refused": refused, "items": items,
            "disagreements": disagree, "rows": rows,
            "gold_pairs": sum(1 for c, d in pairs if c in cz_gold and d in de_gold)}


def lexikon_1910_editions_report(result: dict) -> str:
    """Ledger for :func:`compare_editions`."""
    items = result["items"]
    lines = [f"lexikon_1910_editions v{__version__} — the Czech and German editions as "
             f"two measurements (D-58)", "",
             f"  municipalities paired by court and name: {len(result['pairs'])}, "
             f"{result['gold_pairs']} of them read in both golds; "
             f"{len(result['refused'])} names refused as ambiguous",
             f"  claims: {sum(1 for i in items if i[2] == 1)} positive (both golds), "
             f"{sum(1 for i in items if i[2] == 0)} negative (neither gold, some "
             f"reader), {len(result['disagreements'])} edition disagreements (one "
             f"gold only — kept out of the test)"]
    for d in result["disagreements"][:10]:
        lines.append(f"      {d['cz_unit']} ~ {d['de_unit']}  {d['code']} only in "
                     f"the {d['in']} gold")
    if not result["rows"]:
        lines += ["", "  no Czech × German pass pair on disk yet"]
        return "\n".join(lines)
    lines += ["", f"  {'Czech pass':24s} {'German pass':26s} {'truth':9s} {'n':>5s} "
                  f"{'errA':>5s} {'errB':>5s} {'joint':>5s} {'exp':>6s} {'phi':>6s} "
                  f"{'p':>6s} {'p|sym':>6s}  verdict"]
    for r in result["rows"]:
        verdict = ("—" if not r["n"] else
                   "independent" if r["independent"] else "DEPENDENT given symbol")
        lines.append(f"  {r['cz_pass']:24s} {r['de_pass']:26s} {r['truth']:9s} "
                     f"{r['n']:5d} {r['errors_a']:5d} {r['errors_b']:5d} "
                     f"{r['joint']:5d} {r['expected']:6.2f} {r['phi']:6.3f} "
                     f"{r['p_value'] if r['p_value'] is not None else '':>6} "
                     f"{r['p_within_symbol'] if r['p_within_symbol'] is not None else '':>6}"
                     f"  {verdict} ({r['kind']})")
    lines += ["", "  `p` permutes the German errors within the truth class, `p|sym` "
                  "within (truth class, symbol);",
              f"  D-58 reads `p|sym` at {ALPHA}. Municipality level; the pages are "
              f"few, so read a pass as weak evidence."]
    return "\n".join(lines)
