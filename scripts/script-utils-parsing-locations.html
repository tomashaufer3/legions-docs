How it works

Runs after utils.parsing.clean.clean_places, which strips the trailing + verification marker but keeps the okres text and German exonym.

Field structure (most→least common)

Municipality, okres District (German exonym)
Municipality, okres District, Slovensko          (Transleithanian)
Municipality, Zemské hlavní město Brno (German)  (statutory city, no okres)
GermanName, District ? (CzechName, okres District ?; ...)  (resolved in parens)
Municipality, Čechy ?                            (region only)

Extraction strategy

The municipality + okres are anchored on the first okres occurrence anywhere in the string (regex ([^,;(]+),\s*okres\s+...). This yields the resolved Czech names even when the lead token is a German exonym whose Czech resolution sits inside the parentheses (Kamnitz, Děčín ? (Česká Kamenice, okres Děčín ?) → Česká Kamenice / Děčín) and the primary guess in multi-candidate uncertain entries.

Columns added by add_location_columns (per prefix)

{prefix}_municipality resolved Czech municipality (obec) {prefix}_okres political district {prefix}_german German exonym, if present {prefix}_land Czech lands / Slovakia / Hungary / Galicia / Austria {prefix}_cisleithanian bool — eligible for the 1910 Cisleithanian census {prefix}_loc_uncertain bool — raw contained ? {prefix}_n_candidates int — number of ;-separated alternatives

def parse_location(raw) -> dict:
    """Decompose a raw place string into structured components.

    Returns a dict with keys ``municipality``, ``okres``, ``german``, ``land``,
    ``cisleithanian``, ``uncertain``, ``n_candidates``.  Missing pieces are
    ``pd.NA``.
    """
    blank = {
        "municipality": pd.NA, "okres": pd.NA, "german": pd.NA,
        "land": pd.NA, "cisleithanian": pd.NA,
        "uncertain": False, "n_candidates": 0,
    }
    if pd.isna(raw):
        return blank

    s = str(raw).strip()
    if not s:
        return blank

    low = s.lower()
    uncertain = "?" in s
    n_candidates = s.count(";") + 1

    # -- municipality + okres (anchored on the first "okres") ------------------
    municipality: object = pd.NA
    okres: object = pd.NA
    m = _OKRES_RE.search(s)
    if m:
        municipality = m.group(1).strip().strip("?").strip()
        okres = m.group(2).strip().strip("?").strip()
    else:
        # No okres: municipality = lead comma-token of the pre-"(" part.
        head = s.split("(", 1)[0]
        municipality = head.split(",", 1)[0].strip().strip("?").strip() or pd.NA

    # -- German exonym: first paren group with no "okres" and not just "?" ------
    german: object = pd.NA
    for content in _PAREN_RE.findall(s):
        c = content.strip()
        if not c or c == "?" or "okres" in c.lower():
            continue
        c = _DRIVE_RE.sub("", c)               # drop "dříve:" prefix
        c = re.split(r"\s*[-–]\s*\?", c)[0]    # drop trailing " - ?" uncertainty
        c = c.strip().strip(";").strip()
        if c and c != "?":
            german = c
            break

    # -- land + Cisleithania eligibility --------------------------------------
    land = _classify_land(low)
    cisleithanian = land not in _TRANSLEITHANIAN

    def _norm(v):
        return pd.NA if (pd.isna(v) or str(v).strip() in ("", "?")) else v

    return {
        "municipality": _norm(municipality),
        "okres": _norm(okres),
        "german": german,
        "land": land,
        "cisleithanian": cisleithanian,
        "uncertain": uncertain,
        "n_candidates": n_candidates,
    }

def add_location_columns(
    df: pd.DataFrame,
    source_col: str,
    prefix: str,
) -> pd.DataFrame:
    """Append decomposed location columns for *source_col* under *prefix*.

    Adds ``{prefix}_municipality``, ``{prefix}_okres``, ``{prefix}_german``,
    ``{prefix}_land``, ``{prefix}_cisleithanian``, ``{prefix}_loc_uncertain``,
    and ``{prefix}_n_candidates``.  No-op (with a warning skip) if *source_col*
    is absent.
    """
    df = df.copy()
    if source_col not in df.columns:
        return df

    parsed = df[source_col].map(parse_location)
    df[f"{prefix}_municipality"]  = parsed.map(lambda d: d["municipality"])
    df[f"{prefix}_okres"]         = parsed.map(lambda d: d["okres"])
    df[f"{prefix}_german"]        = parsed.map(lambda d: d["german"])
    df[f"{prefix}_land"]          = parsed.map(lambda d: d["land"])
    df[f"{prefix}_cisleithanian"] = parsed.map(lambda d: d["cisleithanian"])
    df[f"{prefix}_loc_uncertain"] = parsed.map(lambda d: d["uncertain"])
    df[f"{prefix}_n_candidates"]  = parsed.map(lambda d: d["n_candidates"])
    return df

def location_report(df: pd.DataFrame, prefix: str) -> None:
    """Print coverage and land breakdown for a parsed location prefix."""
    muni = f"{prefix}_municipality"
    okr  = f"{prefix}_okres"
    if okr not in df.columns:
        raise ValueError(f"Run add_location_columns(df, ..., prefix='{prefix}') first.")

    n = len(df)
    n_muni = int(df[muni].notna().sum())
    n_okr  = int(df[okr].notna().sum())
    n_unc  = int(df[f"{prefix}_loc_uncertain"].sum())
    n_cis  = int((df[f"{prefix}_cisleithanian"] == True).sum())  # noqa: E712

    print(f"Location report [{prefix}]  (N = {n:,})\n")
    print(f"  municipality non-null : {n_muni:>9,}  ({n_muni/n:.1%})")
    print(f"  okres non-null        : {n_okr:>9,}  ({n_okr/n:.1%})")
    print(f"  marked uncertain (?)  : {n_unc:>9,}  ({n_unc/n:.1%})")
    print(f"  Cisleithanian (census): {n_cis:>9,}  ({n_cis/n:.1%})")

    print("\n  Land breakdown:")
    print(df[f"{prefix}_land"].value_counts(dropna=False).to_string())

    print(f"\n  Top 15 {prefix} okres:")
    print(df[okr].value_counts().head(15).to_string())

"""
locations.py — Place-name decomposition (municipality / okres / land)
=====================================================================
Decomposes a free-text Czech place string into structured components.
Designed for the VÚA ``Místo narození`` / ``Domovem`` fields but kept
dataset-agnostic so it can later parse the legionnaire ``Obec narození`` /
``Okres narození`` fields too.

Runs **after** :func:`utils.parsing.clean.clean_places`, which strips the
trailing ``+`` verification marker but keeps the okres text and German exonym.

Field structure (most→least common)
------------------------------------
    Municipality, okres District (German exonym)
    Municipality, okres District, Slovensko          (Transleithanian)
    Municipality, Zemské hlavní město Brno (German)  (statutory city, no okres)
    GermanName, District ? (CzechName, okres District ?; ...)  (resolved in parens)
    Municipality, Čechy ?                            (region only)

Extraction strategy
-------------------
The municipality + okres are anchored on the **first** ``okres`` occurrence
anywhere in the string (regex ``([^,;(]+),\\s*okres\\s+...``).  This yields the
*resolved Czech* names even when the lead token is a German exonym whose Czech
resolution sits inside the parentheses
(``Kamnitz, Děčín ? (Česká Kamenice, okres Děčín ?)`` → ``Česká Kamenice`` /
``Děčín``) and the primary guess in multi-candidate uncertain entries.

Columns added by :func:`add_location_columns` (per ``prefix``)
-------------------------------------------------------------
``{prefix}_municipality``   resolved Czech municipality (obec)
``{prefix}_okres``          political district
``{prefix}_german``         German exonym, if present
``{prefix}_land``           Czech lands / Slovakia / Hungary / Galicia / Austria
``{prefix}_cisleithanian``  bool — eligible for the 1910 Cisleithanian census
``{prefix}_loc_uncertain``  bool — raw contained ``?``
``{prefix}_n_candidates``   int  — number of ``;``-separated alternatives

Usage
-----
    from utils.parsing.locations import add_location_columns, location_report
    df = add_location_columns(df, "Místo narození", prefix="birth")
    df = add_location_columns(df, "Domovem",        prefix="home")
    location_report(df, "birth")
"""

from __future__ import annotations

import re

import pandas as pd

__version__ = "1.1.0"   # land keywords matched as words, not substrings (audit 2026-09-22)

# ── Regexes ───────────────────────────────────────────────────────────────────
# Municipality + okres anchored on the first "..., okres <name>".  The
# municipality is the token directly before ", okres" (no comma/semicolon/paren
# inside it); the okres runs until a +, (, comma, semicolon, or ? terminator.
_OKRES_RE = re.compile(r"([^,;(]+),\s*okres\s+([^+(,;?]+)")

# Parenthetical groups, e.g. "(Bleistadt)" or "(Česká Kamenice, okres Děčín ?)".
_PAREN_RE = re.compile(r"\(([^)]*)\)")

# "dříve: X" (formerly) prefix inside a German exonym parenthetical.
_DRIVE_RE = re.compile(r"^d[řr]íve:\s*", re.IGNORECASE)


# ── Land classification ───────────────────────────────────────────────────────
# Ordered (pattern, land); first hit wins.  Default → "Czech lands".
# Note: a "(dříve: <Magyar name>)" rename marker reliably flags a formerly-
# Hungarian (Slovak / Subcarpathian) place even when no explicit land is named
# — checked after the explicit Slovakia/Hungary tags so those stay precise.
#
# Every pattern must start a word (v1.1.0). The bare substring "uhersk" read the
# Moravian okres Uherské Hradiště and Uherský Brod as Hungary — 1,747 VÚA births
# flagged not-Cisleithanian (audit 2026-09-22 §A7). Hungary is now the noun
# (Uhry / Uhersko / Maďarsko), never the adjective a Moravian town name carries.
# (This land is provisional anyway: records_geo.add_land replaces it and
# re-derives the cisleithanian flag from the corrected value.)
_LAND_KEYWORDS: list[tuple[str, str]] = [
    (r"slovensk\w*", "Slovakia"),
    (r"uhry|uhrách|uhersko|uhersku", "Hungary"),
    (r"maďarsk\w*|maďar|maďaři", "Hungary"),
    (r"dříve", "Transleithania"),
    (r"halič\w*", "Galicia"),
    (r"rakous\w*", "Austria"),
    (r"štýrsk\w*", "Austria"),
    (r"korut\w*", "Austria"),
    (r"tyrol\w*", "Austria"),
]
_LAND_RES = [(re.compile(rf"(?<![^\W\d_])(?:{pat})(?![^\W\d_])"), land)
             for pat, land in _LAND_KEYWORDS]

# Lands outside Cisleithania (the Hungarian half) — excluded from the 1910
# Cisleithanian census.  "Transleithania" = Slovak / Subcarpathian places
# detected only via the rename marker (precise land unknown).
_TRANSLEITHANIAN = {"Slovakia", "Hungary", "Transleithania"}


def _classify_land(low: str) -> str:
    for rx, land in _LAND_RES:
        if rx.search(low):
            return land
    return "Czech lands"


# ── Core parser ───────────────────────────────────────────────────────────────
