How it works

matice The unit's own land's society, per 1,000 inhabitants, shrunk (Marshall 1991): Matice česká founders in Bohemia, Matice moravská members of 1890 in Moravia and Silesia (D-161). matice_own_z standardises it within each society's territory, since a 52-zl. founder and a Moravian member are not the same act. theatre Czech amateur-theatre companies active in a year (certain), averaged over THEATRE_WINDOW, per 1,000 inhabitants. legion Legionnaires born in the unit, per 1,000 inhabitants and per 1,000 Czech speakers (Umgangssprache) of 1910; the latter is left out where fewer than MIN_CZECH_SPEAKERS speak Czech. desertion Legionnaires leaving by desertion (exit code D) per 1,000 legionnaires born in the unit, shrunk; left out under MIN_LEGIONNAIRES.

Denominators are the 1910 census: the pooled okres (148 units, D-18) and the court census (330 units, the nine statutory-city courts pooled with their twin). Only 57 % of the Czech-lands births name a court, so a court's legionnaires are scaled up by its okres's ratio of all births to court-located births — i.e. a birth known only to its okres is assumed to fall among the okres's courts as the located ones do. The desertion rate is a ratio of located births and needs no scaling.

There is no settlement population in the repo yet (the 1910 register's full run will carry it), so the settlement table holds counts, plus the desertion rate, whose denominator is the legionnaires themselves.

Writes data/interim/intensity/intensity_{okres,court,settlement}.csv.

def theatre_mean(level: str, key: str) -> pd.Series:
    """Companies active per year (certain), averaged over ``THEATRE_WINDOW``, by ``key``."""
    t = pd.read_parquet(THEATRE / f"{level}_year_panel.parquet",
                        columns=[key, "year", "ensembles_active_certain"])

def add_rates(df: pd.DataFrame) -> pd.DataFrame:
    """Rates per 1,000 from the counts and the census columns ``pop_total``/``lang_czech``."""
    df["theatre_per_1000"] = 1000 * df["theatre_companies"] / df["pop_total"]
    df["legion_per_1000"] = 1000 * df["legionnaires"] / df["pop_total"]
    czech_ok = df["lang_czech"] >= MIN_CZECH_SPEAKERS
    df["legion_per_1000_czech"] = (1000 * df["legionnaires"] / df["lang_czech"]).where(czech_ok)
    shrunk = 1000 * mt.eb_rate(df["deserters_located"], df["legionnaires_located"])
    df["desertion_per_1000_leg"] = shrunk.where(df["legionnaires_located"] >= MIN_LEGIONNAIRES)
    return df

def okres_table(leg: pd.DataFrame) -> pd.DataFrame:
    """One row per pooled census okres (key ``census_okres_id``)."""
    m = pd.read_csv(MATICE / "matice_okres.csv").set_index("census_okres_id")
    df = m[["okres_cs"] + MATICE_KEEP].copy()
    xw = pd.read_csv(GEO / "okres_census_crosswalk.csv").set_index("okres_id")["census_okres_id"]
    t = theatre_mean("okres", "okres_id")
    df["theatre_companies"] = t.groupby(t.index.map(xw)).sum().reindex(df.index).fillna(0)
    by = leg.groupby("birth_census_okres_id")
    df["legionnaires"] = by.size().reindex(df.index).fillna(0).astype(int)
    df["deserters"] = by["deserted"].sum().reindex(df.index).fillna(0).astype(int)
    df["legionnaires_located"], df["deserters_located"] = df["legionnaires"], df["deserters"]
    unmatched = set(leg["birth_census_okres_id"].astype(int)) - set(df.index)
    assert not unmatched, f"legion births in no census okres: {sorted(unmatched)[:5]}"
    return add_rates(own_matice(df))[["okres_cs"] + OUT_COLS]

def court_okres(districts: pd.DataFrame) -> pd.Series:
    """Court id → census okres id, from the district backbone's ``okres_code``."""
    code = districts["okres_code"].str.split("-", expand=True).astype(int)
    xw = pd.read_csv(GEO / "okres_census_crosswalk.csv").set_index("okres_id")["census_okres_id"]
    okres = (code[0] * 1000 + code[1]).map(xw)
    return pd.Series(okres.to_numpy(), index=districts["court_id"].astype(int))

def court_table(leg: pd.DataFrame) -> pd.DataFrame:
    """One row per census court (key ``census_court_id``); legionnaires scaled to the okres
    total by the share of the okres's births that name a court."""
    m = pd.read_csv(MATICE / "matice_court.csv").set_index("census_court_id")
    df = m[["gerichtsbezirk_cs", "okres_cs"] + MATICE_KEEP].copy()
    districts = pd.read_parquet(GEO / "districts_1910.parquet")
    pool = mt.census_court_pool(districts, set(df.index))
    t = theatre_mean("court", "court_id")
    df["theatre_companies"] = t.groupby(t.index.map(pool)).sum().reindex(df.index).fillna(0)

    leg = leg.assign(census_court_id=leg["birth_court_id"].map(
        lambda c: pool.get(int(c)) if pd.notna(c) else np.nan))
    located = leg[leg["census_court_id"].notna()]
    by = located.groupby("census_court_id")
    df["legionnaires_located"] = by.size().reindex(df.index).fillna(0).astype(int)
    df["deserters_located"] = by["deserted"].sum().reindex(df.index).fillna(0).astype(int)
    okres_of = court_okres(districts).groupby(level=0).first()
    df["census_okres_id"] = df.index.map(okres_of)
    all_births = leg.groupby("birth_census_okres_id").size()
    located_births = located.groupby("birth_census_okres_id").size()
    scale = (all_births / located_births).reindex(df["census_okres_id"]).to_numpy()
    df["legion_scale"] = scale
    df["legionnaires"] = df["legionnaires_located"] * np.nan_to_num(scale, nan=1.0)
    print(f"court: {len(located):,} of {len(leg):,} births name a court "
          f"({len(located) / len(leg):.1%}); {leg['census_court_id'].isna().sum():,} scaled in")
    out = add_rates(own_matice(df))
    return out[["gerichtsbezirk_cs", "okres_cs", "census_okres_id", "legion_scale"] + OUT_COLS]

def legion_settlements(leg: pd.DataFrame, s: pd.DataFrame) -> pd.Series:
    """Each located legionnaire's ``settlement_id``: the settlement at the birth's
    coordinates, the one of the same name where several share them."""
    s = s.assign(pt=list(zip(s["lat"].round(5), s["lon"].round(5))))
    by_pt = s.groupby("pt")["settlement_id"].apply(list).to_dict()
    by_pt_name = dict(zip(zip(s["pt"], s["cs_name"]), s["settlement_id"]))

    def one(lat, lon, name):
        ids = by_pt.get((round(lat, 5), round(lon, 5)))
        if not ids:
            return None
        return ids[0] if len(ids) == 1 else by_pt_name.get(((round(lat, 5), round(lon, 5)), name))
    return pd.Series([one(a, b, n) if pd.notna(a) else None for a, b, n in
                      zip(leg["birth_lat"], leg["birth_lon"], leg["birth_municipality_cs"])],
                     index=leg.index)

def settlement_table(leg: pd.DataFrame) -> pd.DataFrame:
    """One row per 1910 settlement with anything to draw (key ``settlement_id``): counts,
    and the shrunk desertion rate where ``MIN_LEGIONNAIRES`` are born there."""
    s = pd.read_parquet(GEO / "settlements_1910.parquet").reset_index(drop=True)
    df = s[["settlement_id", "cs_name", "land", "okres_id", "court_id", "lat", "lon"]].copy()
    df.index.name = "settlement_row"          # the theatre panel's key (row position)
    df["theatre_companies"] = theatre_mean("settlement", "settlement_row").reindex(df.index).fillna(0)
    df = df.set_index("settlement_id")

    e = pd.read_csv(MATICE / "matice_entries.csv", dtype={"settlement_id": str},
                    keep_default_na=False, na_values=[""])
    e["is_corporate"] = e["is_corporate"].astype(str) == "True"
    e["member_no"] = pd.to_numeric(e["member_no"], errors="coerce")
    e["place"] = e["settlement_id"].where(e["place_status"] != "prague", "praha:1")
    groups = mt.measure_groups(e[e["place"].notna()])
    df["mc_founders"] = groups["mc_founders"].groupby("place").size().reindex(df.index).fillna(0)
    df["mm_members_1890"] = groups["mm_members_1890"].groupby("place").size().reindex(df.index).fillna(0)
    df["matice_members"] = np.where(df["land"] == "Böhmen", df["mc_founders"], df["mm_members_1890"])

    sid = legion_settlements(leg, s)
    print(f"settlement: {sid.notna().sum():,} of {len(leg):,} births placed at a settlement "
          f"({sid.notna().mean():.1%})")
    by = leg.assign(settlement_id=sid).dropna(subset=["settlement_id"]).groupby("settlement_id")
    df["legionnaires"] = by.size().reindex(df.index).fillna(0).astype(int)
    df["deserters"] = by["deserted"].sum().reindex(df.index).fillna(0).astype(int)
    shrunk = 1000 * mt.eb_rate(df["deserters"], df["legionnaires"])
    df["desertion_per_1000_leg"] = shrunk.where(df["legionnaires"] >= MIN_LEGIONNAIRES)
    keep = (df[["matice_members", "theatre_companies", "legionnaires"]] > 0).any(axis=1)
    return df[keep & df["lat"].notna()]

def main() -> int:
    OUT.mkdir(parents=True, exist_ok=True)
    leg = load_legion()
    tables = {"okres": okres_table(leg), "court": court_table(leg),
              "settlement": settlement_table(leg)}
    for level, df in tables.items():
        assert df.index.is_unique and df.index.notna().all(), f"{level}: key"
        df.to_csv(OUT / f"intensity_{level}.csv")
        print(f"wrote intensity_{level}.csv: {len(df):,} rows")
    rates = ["matice_per_1000", "theatre_per_1000", "legion_per_1000",
             "legion_per_1000_czech", "desertion_per_1000_leg"]
    for level in ("okres", "court"):
        print(f"\n{level}: Spearman across units\n"
              + tables[level][rates].corr(method="spearman").round(2).to_string())
    return 0

"""
Five intensities side by side, per 1,000, at three levels: political okres, court
district, settlement.  Provisional (D-163): every definition here is the simplest one
that can be drawn, and each will move when its measure is ruled on.

``matice``
    The unit's own land's society, per 1,000 inhabitants, shrunk (Marshall 1991):
    Matice česká founders in Bohemia, Matice moravská members of 1890 in Moravia and
    Silesia (D-161).  ``matice_own_z`` standardises it within each society's territory,
    since a 52-zl. founder and a Moravian member are not the same act.
``theatre``
    Czech amateur-theatre companies active in a year (certain), averaged over
    ``THEATRE_WINDOW``, per 1,000 inhabitants.
``legion``
    Legionnaires born in the unit, per 1,000 inhabitants and per 1,000 Czech speakers
    (Umgangssprache) of 1910; the latter is left out where fewer than
    ``MIN_CZECH_SPEAKERS`` speak Czech.
``desertion``
    Legionnaires leaving by desertion (exit code D) per 1,000 legionnaires born in the
    unit, shrunk; left out under ``MIN_LEGIONNAIRES``.

Denominators are the 1910 census: the pooled okres (148 units, D-18) and the court census
(330 units, the nine statutory-city courts pooled with their twin).  Only 57 % of the
Czech-lands births name a court, so a court's legionnaires are scaled up by its okres's
ratio of all births to court-located births — i.e. a birth known only to its okres is
assumed to fall among the okres's courts as the located ones do.  The desertion rate is
a ratio of located births and needs no scaling.

There is no settlement population in the repo yet (the 1910 register's full run will
carry it), so the settlement table holds counts, plus the desertion rate, whose
denominator is the legionnaires themselves.

Writes ``data/interim/intensity/intensity_{okres,court,settlement}.csv``.

Usage::

    python utils/reference_tools/places/run_intensities.py
"""

def okres_table(leg: pd.DataFrame) -> pd.DataFrame:
    """One row per pooled census okres (key ``census_okres_id``)."""
    m = pd.read_csv(MATICE / "matice_okres.csv").set_index("census_okres_id")
    df = m[["okres_cs"] + MATICE_KEEP].copy()
    xw = pd.read_csv(GEO / "okres_census_crosswalk.csv").set_index("okres_id")["census_okres_id"]
    t = theatre_mean("okres", "okres_id")
    df["theatre_companies"] = t.groupby(t.index.map(xw)).sum().reindex(df.index).fillna(0)
    by = leg.groupby("birth_census_okres_id")
    df["legionnaires"] = by.size().reindex(df.index).fillna(0).astype(int)
    df["deserters"] = by["deserted"].sum().reindex(df.index).fillna(0).astype(int)
    df["legionnaires_located"], df["deserters_located"] = df["legionnaires"], df["deserters"]
    unmatched = set(leg["birth_census_okres_id"].astype(int)) - set(df.index)
    assert not unmatched, f"legion births in no census okres: {sorted(unmatched)[:5]}"
    return add_rates(own_matice(df))[["okres_cs"] + OUT_COLS]


def court_okres(districts: pd.DataFrame) -> pd.Series:
    """Court id → census okres id, from the district backbone's ``okres_code``."""
    code = districts["okres_code"].str.split("-", expand=True).astype(int)
    xw = pd.read_csv(GEO / "okres_census_crosswalk.csv").set_index("okres_id")["census_okres_id"]
    okres = (code[0] * 1000 + code[1]).map(xw)
    return pd.Series(okres.to_numpy(), index=districts["court_id"].astype(int))


def court_table(leg: pd.DataFrame) -> pd.DataFrame:
    """One row per census court (key ``census_court_id``); legionnaires scaled to the okres
    total by the share of the okres's births that name a court."""
    m = pd.read_csv(MATICE / "matice_court.csv").set_index("census_court_id")
    df = m[["gerichtsbezirk_cs", "okres_cs"] + MATICE_KEEP].copy()
    districts = pd.read_parquet(GEO / "districts_1910.parquet")
    pool = mt.census_court_pool(districts, set(df.index))
    t = theatre_mean("court", "court_id")
    df["theatre_companies"] = t.groupby(t.index.map(pool)).sum().reindex(df.index).fillna(0)

    leg = leg.assign(census_court_id=leg["birth_court_id"].map(
        lambda c: pool.get(int(c)) if pd.notna(c) else np.nan))
    located = leg[leg["census_court_id"].notna()]
    by = located.groupby("census_court_id")
    df["legionnaires_located"] = by.size().reindex(df.index).fillna(0).astype(int)
    df["deserters_located"] = by["deserted"].sum().reindex(df.index).fillna(0).astype(int)
    okres_of = court_okres(districts).groupby(level=0).first()
    df["census_okres_id"] = df.index.map(okres_of)
    all_births = leg.groupby("birth_census_okres_id").size()
    located_births = located.groupby("birth_census_okres_id").size()
    scale = (all_births / located_births).reindex(df["census_okres_id"]).to_numpy()
    df["legion_scale"] = scale
    df["legionnaires"] = df["legionnaires_located"] * np.nan_to_num(scale, nan=1.0)
    print(f"court: {len(located):,} of {len(leg):,} births name a court "
          f"({len(located) / len(leg):.1%}); {leg['census_court_id'].isna().sum():,} scaled in")
    out = add_rates(own_matice(df))
    return out[["gerichtsbezirk_cs", "okres_cs", "census_okres_id", "legion_scale"] + OUT_COLS]


def legion_settlements(leg: pd.DataFrame, s: pd.DataFrame) -> pd.Series:
    """Each located legionnaire's ``settlement_id``: the settlement at the birth's
    coordinates, the one of the same name where several share them."""
    s = s.assign(pt=list(zip(s["lat"].round(5), s["lon"].round(5))))
    by_pt = s.groupby("pt")["settlement_id"].apply(list).to_dict()
    by_pt_name = dict(zip(zip(s["pt"], s["cs_name"]), s["settlement_id"]))

    def one(lat, lon, name):
        ids = by_pt.get((round(lat, 5), round(lon, 5)))
        if not ids:
            return None
        return ids[0] if len(ids) == 1 else by_pt_name.get(((round(lat, 5), round(lon, 5)), name))
    return pd.Series([one(a, b, n) if pd.notna(a) else None for a, b, n in
                      zip(leg["birth_lat"], leg["birth_lon"], leg["birth_municipality_cs"])],
                     index=leg.index)


def settlement_table(leg: pd.DataFrame) -> pd.DataFrame:
    """One row per 1910 settlement with anything to draw (key ``settlement_id``): counts,
    and the shrunk desertion rate where ``MIN_LEGIONNAIRES`` are born there."""
    s = pd.read_parquet(GEO / "settlements_1910.parquet").reset_index(drop=True)
    df = s[["settlement_id", "cs_name", "land", "okres_id", "court_id", "lat", "lon"]].copy()
    df.index.name = "settlement_row"          # the theatre panel's key (row position)
    df["theatre_companies"] = theatre_mean("settlement", "settlement_row").reindex(df.index).fillna(0)
    df = df.set_index("settlement_id")

    e = pd.read_csv(MATICE / "matice_entries.csv", dtype={"settlement_id": str},
                    keep_default_na=False, na_values=[""])
    e["is_corporate"] = e["is_corporate"].astype(str) == "True"
    e["member_no"] = pd.to_numeric(e["member_no"], errors="coerce")
    e["place"] = e["settlement_id"].where(e["place_status"] != "prague", "praha:1")
    groups = mt.measure_groups(e[e["place"].notna()])
    df["mc_founders"] = groups["mc_founders"].groupby("place").size().reindex(df.index).fillna(0)
    df["mm_members_1890"] = groups["mm_members_1890"].groupby("place").size().reindex(df.index).fillna(0)
    df["matice_members"] = np.where(df["land"] == "Böhmen", df["mc_founders"], df["mm_members_1890"])

    sid = legion_settlements(leg, s)
    print(f"settlement: {sid.notna().sum():,} of {len(leg):,} births placed at a settlement "
          f"({sid.notna().mean():.1%})")
    by = leg.assign(settlement_id=sid).dropna(subset=["settlement_id"]).groupby("settlement_id")
    df["legionnaires"] = by.size().reindex(df.index).fillna(0).astype(int)
    df["deserters"] = by["deserted"].sum().reindex(df.index).fillna(0).astype(int)
    shrunk = 1000 * mt.eb_rate(df["deserters"], df["legionnaires"])
    df["desertion_per_1000_leg"] = shrunk.where(df["legionnaires"] >= MIN_LEGIONNAIRES)
    keep = (df[["matice_members", "theatre_companies", "legionnaires"]] > 0).any(axis=1)
    return df[keep & df["lat"].notna()]


def main() -> int:
    OUT.mkdir(parents=True, exist_ok=True)
    leg = load_legion()
    tables = {"okres": okres_table(leg), "court": court_table(leg),
              "settlement": settlement_table(leg)}
    for level, df in tables.items():
        assert df.index.is_unique and df.index.notna().all(), f"{level}: key"
        df.to_csv(OUT / f"intensity_{level}.csv")
        print(f"wrote intensity_{level}.csv: {len(df):,} rows")
    rates = ["matice_per_1000", "theatre_per_1000", "legion_per_1000",
             "legion_per_1000_czech", "desertion_per_1000_leg"]
    for level in ("okres", "court"):
        print(f"\n{level}: Spearman across units\n"
              + tables[level][rates].corr(method="spearman").round(2).to_string())
    return 0


if __name__ == "__main__":
    sys.exit(main())
