How it works

Each panel has its own scale, clipped at its 97th percentile so that Prague and a few small towns do not wash the rest out; grey is a unit with no value (too few Czech speakers or legionnaires). intensity_maps_municipal Four dot maps: dot area is the count, capped at the panel's 99th percentile (a capped dot gets an outline). The settlements have no 1910 population yet, so only the desertion panel, whose denominator is the legionnaires born there, is coloured by a rate.

Writes PDF + PNG to output/figures/intensity/.

class Basemap:
    """Court polygons and their border lines, built once and drawn into any axes."""

    def __init__(self) -> None:
        self.shapes = court_shapes()
        self.borders = {by: border_segments(self.shapes, by).reshape(-1, 2, 2)
                        for by in ("court_id", "okres_id", "land")}
        pts = np.vstack([np.vstack(r) for r in self.shapes["rings"]])
        self.extent = (pts[:, 0].min() - 5e3, pts[:, 0].max() + 5e3,
                       pts[:, 1].min() - 5e3, pts[:, 1].max() + 5e3)

    def draw(self, ax, values: pd.Series | None = None, cmap=None, norm=None,
             court_lines: bool = True) -> PatchCollection:
        """Shade the courts by ``values`` (indexed by court_id, grey where missing)."""
        patches = PatchCollection([_patch(r) for r in self.shapes["rings"]], linewidths=0)
        if values is not None:
            v = self.shapes["court_id"].map(values).to_numpy(dtype=float)
            cmap = cmap.copy()
            cmap.set_bad(ths.CU_GREY_LIGHT)
            patches.set_array(np.ma.masked_invalid(v))
            patches.set_cmap(cmap)
            patches.set_norm(norm)
        else:
            patches.set_facecolor("#f2f2f2")
        ax.add_collection(patches)
        lines = [("okres_id", 0.3, ths.CU_GREY), ("land", 0.9, ths.CU_BLACK)]
        if court_lines:
            lines.insert(0, ("court_id", 0.1, "#ffffff"))
        for by, lw, col in lines:
            ax.add_collection(LineCollection(self.borders[by], linewidths=lw, colors=col))
        x0, x1, y0, y1 = self.extent
        ax.set_xlim(x0, x1)
        ax.set_ylim(y0, y1)
        ax.set_aspect("equal")
        ax.axis("off")
        return patches

def court_values(level: str, base: Basemap, column: str) -> pd.Series:
    """``column`` of each court polygon: its census okres's or its pooled census court's."""
    df = pd.read_csv(DATA / f"intensity_{level}.csv")
    if level == "okres":
        xw = pd.read_csv(GEO / "okres_census_crosswalk.csv").set_index("okres_id")["census_okres_id"]
        unit = base.shapes["okres_id"].map(xw)
        v = unit.map(df.set_index("census_okres_id")[column])
    else:
        df = df.set_index("census_court_id")
        pool = mt.census_court_pool(pd.read_parquet(GEO / "districts_1910.parquet"), set(df.index))
        v = base.shapes["court_id"].map(pool).map(df[column])
    return pd.Series(v.to_numpy(), index=base.shapes["court_id"])

def district_figure(level: str, base: Basemap) -> plt.Figure:
    fig, axes = plt.subplots(2, 3, figsize=(15, 6.8))
    for ax, (col, title, label, cmap) in zip(axes.flat, DISTRICT_PANELS):
        v = court_values(level, base, col)
        norm = Normalize(vmin=np.nanpercentile(v, 100 - CLIP_PCT) if col in FLOATING else 0.0,
                         vmax=np.nanpercentile(v, CLIP_PCT))
        patches = base.draw(ax, v, cmap, norm, court_lines=(level == "court"))
        ax.set_title(title, fontsize=10)
        cb = fig.colorbar(patches, ax=ax, shrink=0.7, pad=0.01,
                          extend="both" if col in FLOATING else "max")
        cb.set_label(label, fontsize=8)
        cb.ax.tick_params(labelsize=7)
    note = axes.flat[-1]
    note.axis("off")
    unit = {"okres": "148 political okresy (census-pooled)",
            "court": "330 court districts (census-pooled)"}[level]
    note.text(0, 0.95, f"Units: {unit}, 1910.\n\nDenominators: 1910 census.\n"
              "Legionnaires by birthplace; at court level\nscaled up for births known only "
              "to the okres.\nDesertion: exit code D, shrunk (Marshall 1991);\n"
              "grey under 20 legionnaires.\nPer Czech speaker: grey under 1,000 speakers.\n"
              f"Colour scales stop at the {CLIP_PCT}th percentile.\n\nProvisional (D-163).",
              va="top", fontsize=8, color=ths.CU_GREY, transform=note.transAxes)
    fig.suptitle(f"Nationalism, volunteering and desertion by {level}", fontsize=12)
    fig.tight_layout()
    return fig

def municipal_figure(base: Basemap) -> plt.Figure:
    pts = pd.read_csv(DATA / "intensity_settlement.csv")
    pts["x"], pts["y"] = laea_forward(pts["lon"].to_numpy(float), pts["lat"].to_numpy(float))
    fig, axes = plt.subplots(2, 2, figsize=(13, 8.4))
    for ax, (col, title, unit) in zip(axes.flat, MUNICIPAL_PANELS):
        base.draw(ax, court_lines=False)
        _dots(ax, pts, col, colour=ths.CU_NAVY)
        n = int((pts[col] > 0).sum())
        ax.set_title(f"{title}\n{pts[col].sum():,.0f} in {n:,} places", fontsize=10)
        ax.legend(title=unit, loc="lower left", frameon=False, fontsize=7, title_fontsize=7)
    ax = axes.flat[3]
    base.draw(ax, court_lines=False)
    d = pts[pts["desertion_per_1000_leg"].notna()]
    rate = d["desertion_per_1000_leg"]
    norm = Normalize(np.nanpercentile(rate, 100 - CLIP_PCT), np.nanpercentile(rate, CLIP_PCT))
    sc = _dots(ax, d, "legionnaires", cmap=SEQ_RED, norm=norm, value="desertion_per_1000_leg")
    ax.set_title(f"Desertion from the Legion\n{len(d):,} places with 20+ legionnaires; "
                 "dot = legionnaires", fontsize=10)
    ax.legend(title="legionnaires", loc="lower left", frameon=False, fontsize=7, title_fontsize=7)
    cb = fig.colorbar(sc, ax=ax, shrink=0.6, pad=0.01, extend="both")
    cb.set_label("desertions per 1,000 legionnaires (shrunk)", fontsize=8)
    fig.suptitle("Nationalism, volunteering and desertion by place (counts; no 1910 "
                 "settlement population yet)", fontsize=12)
    fig.tight_layout()
    return fig

``intensity_maps_okres`` / ``intensity_maps_court``
    Five choropleths on the 1910 court polygons: Matice, theatre, legionnaires per 1,000
    inhabitants and per 1,000 Czech speakers, desertions per 1,000 legionnaires.  Each
    panel has its own scale, clipped at its 97th percentile so that Prague and a few

``intensity_maps_municipal``

CAP_PCT = 99                  # a dot's area stops growing at this percentile of counts
DOT_AREA_AT_CAP = 36.0        # pt²
SEQ = ths.CU_SEQUENTIAL
SEQ_RED = LinearSegmentedColormap.from_list("cu_red_seq", ["#fbe9eb", ths.CU_RED, "#5e1019"])

DISTRICT_PANELS = [

    ("legion_per_1000", "Legionnaires", "per 1,000 inhabitants", SEQ),
    ("legion_per_1000_czech", "Legionnaires", "per 1,000 Czech speakers", SEQ),

     SEQ_RED),
]
#: Panels whose scale starts at their 3rd percentile, not zero: a z, and a rate no unit is
#: near zero on (desertion runs 10–25 per 1,000 everywhere).
FLOATING = {"matice_own_z", "desertion_per_1000_leg"}
#: (count column, title, legend unit) — the municipal panels; desertion is added apart.
MUNICIPAL_PANELS = [
    ("matice_members", "Matice members (own land's society)", "members"),
    ("theatre_companies", "Amateur theatre, 1880–1913", "companies active / yr"),
    ("legionnaires", "Legionnaires by birthplace", "legionnaires"),
]

def _dots(ax, pts: pd.DataFrame, count: str, colour=None, cmap=None, norm=None,
          value: str | None = None):
    """Dots of area ∝ ``count`` up to the cap; capped dots are outlined."""
    p = pts[pts[count] > 0].sort_values(count, ascending=False)
    cap = max(np.percentile(p[count], CAP_PCT), 1.0)
    area = DOT_AREA_AT_CAP * np.minimum(p[count], cap) / cap
    capped = p[count] > cap
    kw = dict(c=p[value], cmap=cmap, norm=norm) if value else dict(c=colour)
    sc = ax.scatter(p["x"], p["y"], s=np.maximum(area, 0.8), alpha=0.75, zorder=3,
                    linewidths=0, **kw)
    ax.scatter(p.loc[capped, "x"], p.loc[capped, "y"], s=DOT_AREA_AT_CAP, zorder=4,
               facecolors="none", edgecolors=ths.CU_BLACK, linewidths=0.6)
    for n in sorted({1, int(round(cap / 4)) or 1, int(round(cap))}):
        ax.scatter([], [], s=max(DOT_AREA_AT_CAP * n / cap, 0.8), c=colour or ths.CU_GREY,
                   alpha=0.75, label=f"{n:,}")
    ax.scatter([], [], s=DOT_AREA_AT_CAP, facecolors="none", edgecolors=ths.CU_BLACK,
               linewidths=0.5, label=f"> {int(round(cap)):,} (capped)")
    return sc

        print(f"wrote {OUT.relative_to(ROOT)}/{name}.pdf / .png")
    return 0
