How it works

The frozen route's gate (D-115) let coverage alone keep DINOv3 from routing anything in German: on the Böhmen gold it judged 0.115 of the reader of record's claims against the Czech 0.863. The plan was to gate it per symbol on accuracy where it does judge, tuned on the German Böhmen gold, but that gold cannot carry it: DINOv3 named a real symbol on 52 of its offices boxes, one castle and one poorhouse among them. So, by the author's choice of 2026-11-06, there is no gate: every flag goes to Opus (§6), which decides each one and is itself tested first on the author's hand rulings (D-190). A flag is one of two things, on the reader of record's own boxes:

  • disagree — DINOv3 names another symbol than the one the reader read there;
  • unclaimed — DINOv3 names a symbol on a box no reader claimed anything for.

A flag is not sent where the question is already settled: Opus ruled on the mark in the route, the author ruled it (the route's queue or the hand-check page), or the box lies in an offices region §4 reads again. DINOv3's own accuracy is reported against the author's hand rulings on poorhouse and castle (hand_check_agreement), as a check, not a gate.

    """The marks the author ruled on the hand-check page: a staged ruling naming a value (a symbol
    of the pair, a school, no symbol). A mark the page showed but the author did not rule, or
    marked *unsure* / *other*, is not settled. (0.1.1: before, every mark the page showed counted,

def flags(edition: str, verdicts: list[dict], record: str, rules: dict, ruled_by_hand: set,
          regions: dict, settled: set = frozenset()) -> list[dict]:
    """Every DINOv3 flag on the reader of record's boxes, with whether it is sent and why not.

    ``rules`` — the route's final rule per mark (``ledger/marks_<run>.csv``); ``ruled_by_hand`` —
    the hand-checked marks; ``regions`` — §4's regions (:func:`in_regions`); ``settled`` — the
    claims a token question settled (:func:`settled_claims`): a flag both of whose symbols (the
    one alone, for an unclaimed box) are settled on its part asks nothing new."""
    out = []
    for r in verdicts:
        if r["pass"] != record or r["verdict"] not in ("disagree", "unclaimed") or not is_symbol(r["dino"]):
            continue
        mark = led.box_mark(r["image"], r["box"])
        rule = rules.get(mark, "")
        y = float(r["box"].split(",")[1]) if r["box"] else -1
        part = (r["scan"], r["district"], str(r["muni"]), str(r["seq"] or "0"))
        asked = {r["dino"]} | ({r["claim"]} if r["claim"] else set())
        why = ("Opus ruled it in the route" if rule in SETTLED_RULES[:1]
               else "you ruled it (the route's queue)" if rule == "author"
               else "the route settled its symbols on this part" if all(part + (t,) in settled for t in asked)
               else "you ruled it (hand check)" if mark in ruled_by_hand
               else "in an offices region §4 reads again" if any(a <= y < b for a, b in regions.get(r["image"], []))
               else "")
        out.append({"edition": edition, "kind": r["verdict"], "section": r["section"], "page": r["scan"],
                    "district": r["district"], "muni": r["muni"], "seq": r["seq"], "mark": mark,
                    "claim": r["claim"], "dino": r["dino"], "similarity": r["similarity"],
                    "route_rule": rule, "send": not why, "why_not": why})
    return out

def run(edition: str, pages: str = "full", out: Path = OUT) -> list[dict]:
    """Write ``<out>/<edition>/dino/flags.csv`` and return its rows."""
    run_dir = rf.out_dir(edition, pages)
    record = rf.labels(pages, edition)["gemini"]
    marks = _read(run_dir / "ledger" / f"marks_{edition}_{pages}.csv")
    rules = {r["mark"]: r["rule"] for r in marks}
    rows = flags(edition, _read(rf.verdicts_path(edition, pages)), record, rules,
                 hand_checked(edition), in_regions(edition), settled_claims(marks))
    folder = Path(out) / edition / SUB
    folder.mkdir(parents=True, exist_ok=True)
    with open(folder / "flags.csv", "w", encoding="utf-8", newline="") as fh:
        w = csv.DictWriter(fh, fieldnames=FIELDS, lineterminator="\n")
        w.writeheader()
        w.writerows(rows)
    return rows

def report(edition: str, rows: list[dict]) -> str:
    sent = [r for r in rows if r["send"]]
    parts = {(r["page"], r["district"], r["muni"], r["seq"]) for r in sent}
    by_kind = Counter((r["kind"], r["section"]) for r in sent)
    held = Counter(r["why_not"] for r in rows if not r["send"])
    lines = [f"{edition}: {len(rows)} flags, {len(sent)} to Opus on {len(parts)} parts "
             f"({', '.join(f'{k} in {s} {n}' for (k, s), n in sorted(by_kind.items()))})"]
    if held:
        lines.append("  not sent: " + ", ".join(f"{why} {n}" for why, n in held.most_common()))
    top = Counter(r["dino"] for r in sent).most_common(8)
    lines.append("  DINOv3 names, most often: " + ", ".join(f"{c} {n}" for c, n in top))
    return "\n".join(lines)

def hand_check_agreement(editions=("cz", *GERMAN), pages: str = "full") -> list[dict]:
    """DINOv3 against the author's poorhouse/castle hand rulings: per symbol DINOv3 named on the
    reader of record's box, how often the author ruled that symbol. A check, not a gate."""
    pairs = {g["group"]: g for g in json.loads((CHECKS / "hand_check_groups.json").read_text("utf-8"))}
    key = {r["id"]: r for r in _read(CHECKS / "hand_check_key_pairs.csv")}
    tally = defaultdict(Counter)
    for edition in editions:
        record = rf.labels(pages, edition)["gemini"]
        named = {led.box_mark(r["image"], r["box"]): r["dino"]
                 for r in _read(rf.verdicts_path(edition, pages)) if r["pass"] == record and r["box"]}
        for folder in STAGED.glob("r_*-p"):
            group = folder.name[2:]
            if pairs[group]["edition"] != edition:
                continue
            for doc in folder.glob("*.json"):
                ruling = _ruled(json.loads(doc.read_text("utf-8")).get("v", ""), pairs[group])
                k = key.get(doc.stem)
                if ruling is None or k is None:
                    continue
                name = named.get(k["mark"], "")
                dino = name if is_symbol(name) else "no view"
                tally[(edition, dino)]["n"] += 1
                tally[(edition, dino)]["right"] += int(dino == ruling)
    return [{"edition": e, "dino": d, "n": c["n"], "right": c["right"],
             "share": round(c["right"] / c["n"], 3) if d != "no view" else None}
            for (e, d), c in sorted(tally.items(), key=lambda kv: (kv[0][0], -kv[1]["n"]))]

    rules = {r["mark"]: r["rule"] for r in marks}
    rows = flags(edition, _read(rf.verdicts_path(edition, pages)), record, rules,
                 hand_checked(edition), in_regions(edition), settled_claims(marks))
    folder = Path(out) / edition / SUB
    folder.mkdir(parents=True, exist_ok=True)
    with open(folder / "flags.csv", "w", encoding="utf-8", newline="") as fh:
        w = csv.DictWriter(fh, fieldnames=FIELDS, lineterminator="\n")
        w.writeheader()
        w.writerows(rows)
    return rows


def report(edition: str, rows: list[dict]) -> str:
    sent = [r for r in rows if r["send"]]
    parts = {(r["page"], r["district"], r["muni"], r["seq"]) for r in sent}
    by_kind = Counter((r["kind"], r["section"]) for r in sent)
    held = Counter(r["why_not"] for r in rows if not r["send"])
    lines = [f"{edition}: {len(rows)} flags, {len(sent)} to Opus on {len(parts)} parts "
             f"({', '.join(f'{k} in {s} {n}' for (k, s), n in sorted(by_kind.items()))})"]
    if held:
        lines.append("  not sent: " + ", ".join(f"{why} {n}" for why, n in held.most_common()))
    top = Counter(r["dino"] for r in sent).most_common(8)
    lines.append("  DINOv3 names, most often: " + ", ".join(f"{c} {n}" for c, n in top))
    return "\n".join(lines)


#: A hand ruling's value on the poorhouse/castle page: ``a`` / ``b`` the pair; ``school``;
#: ``number`` = no symbol; ``other`` / ``unsure`` say nothing a symbol can be held to.
def _ruled(value: str, pair: dict) -> str | None:
    return {"a": pair["a"], "b": pair["b"], "school": "[school]", "number": led.ABSENT}.get(value)


def hand_check_agreement(editions=("cz", *GERMAN), pages: str = "full") -> list[dict]:
    """DINOv3 against the author's poorhouse/castle hand rulings: per symbol DINOv3 named on the
    reader of record's box, how often the author ruled that symbol. A check, not a gate."""
    pairs = {g["group"]: g for g in json.loads((CHECKS / "hand_check_groups.json").read_text("utf-8"))}
    key = {r["id"]: r for r in _read(CHECKS / "hand_check_key_pairs.csv")}
    tally = defaultdict(Counter)
    for edition in editions:
        record = rf.labels(pages, edition)["gemini"]
        named = {led.box_mark(r["image"], r["box"]): r["dino"]
                 for r in _read(rf.verdicts_path(edition, pages)) if r["pass"] == record and r["box"]}
        for folder in STAGED.glob("r_*-p"):
            group = folder.name[2:]
            if pairs[group]["edition"] != edition:
                continue
            for doc in folder.glob("*.json"):
                ruling = _ruled(json.loads(doc.read_text("utf-8")).get("v", ""), pairs[group])
                k = key.get(doc.stem)
                if ruling is None or k is None:
                    continue
                name = named.get(k["mark"], "")
                dino = name if is_symbol(name) else "no view"
                tally[(edition, dino)]["n"] += 1
                tally[(edition, dino)]["right"] += int(dino == ruling)
    return [{"edition": e, "dino": d, "n": c["n"], "right": c["right"],
             "share": round(c["right"] / c["n"], 3) if d != "no view" else None}
            for (e, d), c in sorted(tally.items(), key=lambda kv: (kv[0][0], -kv[1]["n"]))]

    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--out", default=str(OUT))
    args = ap.parse_args(argv)
