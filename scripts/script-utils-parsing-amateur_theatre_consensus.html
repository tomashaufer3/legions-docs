How it works

The register's route reads every crop twice and lets Opus settle what the readers dispute; this does the same for the chronicle clauses:

  1. read — every clause is read by two models independently (PIPELINE: Haiku 5.5 and Gemini 3.8 Flash, utils.parsing.amateur_theatre_model), batched, the codebook cached;
  2. compare (disputes) — on what reaches the measure only, never on the 22 event types: Czech activity, whether it is certain (not hedged), a company, a performance, and for an undated clause the interval it dates. Each reading is put through amateur_theatre.code_clauses exactly as the rules' cues are, so the codebook (D-141) decides the flags. On a clause the rules already coded, the rules are a third voter, at no cost. A reader with no valid reading makes the clause disputed;
  3. judge (judge_others) — Opus 5.5 at high effort reads each disputed clause with every reading in front of it, the readers anonymised (Reader 1, Reader 2, Rules), and gives its own reading in the same schema;
  4. settle (consensus) — an agreed clause takes the first reader's reading, a disputed one the judge's; a disputed clause the judge has not read stays open.

What the pipeline cannot see is two readers agreeing and both being wrong: the check samples (theatre_rule_audit) measure how often that happens.

def reading_flags(rows: pd.DataFrame, readings: pd.DataFrame) -> pd.DataFrame:
    """One reader's flags on ``rows`` (coded clauses, keyed ``clause_key``), its readings run
    through the codebook; NaN where it has no valid reading.  Index ``clause_key``."""
    valid = readings[readings["valid"].astype(bool)] if len(readings) else readings
    read = rows[rows["clause_key"].isin(valid["clause_key"])] if len(valid) else rows.iloc[0:0]
    parts = [atm.apply_readings(g, valid, s).assign(clause_key=g["clause_key"].values)
             for s, g in read.groupby("source")]
    out = _flags(pd.concat(parts).set_index("clause_key")) if parts else pd.DataFrame(columns=FLAGS)
    return out.reindex(rows["clause_key"]).astype("float")

def disputes(rows: pd.DataFrame, readings: dict[str, pd.DataFrame], use_rules: bool = True) -> pd.DataFrame:
    """Per clause of ``rows``: each voter's flags, and whether they dispute (``disputed``) and on
    what (``on``).  Voters: the readers of ``readings`` and, on a clause the rules coded, the
    rules.  A missing reading is a dispute.  An undated clause is disputed also on its interval."""
    keys = pd.Index(rows["clause_key"])
    undated = rows.set_index("clause_key")["date_source"].eq("none")
    voters = {name: reading_flags(rows, table) for name, table in readings.items()}
    if use_rules:
        voters["rules"] = rules_flags(rows)
    dates = {name: _date_key(table, keys) for name, table in readings.items()}
    out = pd.DataFrame(index=keys)
    on = [[] for _ in range(len(keys))]
    for name in readings:
        missing = voters[name][FLAGS].isna().any(axis=1).to_numpy()
        for i in np.where(missing)[0]:
            on[i].append(f"no reading: {name}")
    for flag in FLAGS:
        values = pd.DataFrame({n: v[flag] for n, v in voters.items()}, index=keys)
        out[flag + "_votes"] = values.apply(lambda r: "|".join("" if pd.isna(x) else str(int(x)) for x in r), axis=1)
        split = values.nunique(axis=1, dropna=True) > 1
        for i in np.where(split.to_numpy())[0]:
            on[i].append(flag)
    if len(readings) > 1:
        d = pd.DataFrame(dates, index=keys)
        split = undated.reindex(keys).to_numpy() & (d.nunique(axis=1, dropna=False) > 1).to_numpy()
        for i in np.where(split)[0]:
            on[i].append("date")
    out["on"] = [", ".join(dict.fromkeys(x)) for x in on]
    out["disputed"] = out["on"] != ""
    out["voters"] = "|".join(voters)
    return out

def judge_others(disputed: pd.Index, readings: dict[str, pd.DataFrame], rows: pd.DataFrame) -> dict[str, list[dict]]:
    """For each disputed clause, the readings the judge weighs, anonymised and in a fixed order:
    ``Reader 1`` … for the models, ``Rules`` (event types and hedge) on a coded clause."""
    by = {name: t[t["valid"].astype(bool)].set_index("clause_key") for name, t in readings.items()}
    coded = rows.set_index("clause_key")
    out = {}
    for key in disputed:
        others = []
        for i, name in enumerate(readings, start=1):
            if key in by[name].index:
                r = by[name].loc[key]
                others.append({"name": f"Reader {i}", "types": r["types"], "hedged": bool(r["hedged"]),
                               "titles": r["titles"], **({k: r[k] for k in DATE_FIELDS} if coded.at[key, "date_source"] == "none" else {})})
            else:
                others.append({"name": f"Reader {i}", "types": "(no reading)", "hedged": False, "titles": ""})
        if not bool(coded.at[key, "needs_model"]):
            cues = [c for c in str(coded.at[key, "cues"] if pd.notna(coded.at[key, "cues"]) else "").split("|") if c]
            others.append({"name": "Rules", "types": "|".join(c for c in cues if c not in ("hedged", "title")),
                           "hedged": "hedged" in cues, "titles": ""})
        out[key] = others
    return out

def consensus(table: pd.DataFrame, readings: dict[str, pd.DataFrame], judged: pd.DataFrame | None) -> pd.DataFrame:
    """The readings to apply: an agreed clause takes the first reader's reading (``status``
    ``agreed``), a disputed one the judge's (``judged``); a disputed clause without a valid
    judge reading is left out (``open``: the rules stand, or the clause stays open).
    ``attrs["status"]`` counts the three."""
    first = next(iter(readings))
    lead = readings[first][readings[first]["valid"].astype(bool)].set_index("clause_key")
    judge = (judged[judged["valid"].astype(bool)].set_index("clause_key") if judged is not None and len(judged)
             else pd.DataFrame(columns=lead.columns))
    agreed = table.index[~table["disputed"]].intersection(lead.index)
    settled = table.index[table["disputed"]].intersection(judge.index)
    out = pd.concat([lead.loc[agreed].assign(status="agreed"), judge.loc[settled].assign(status="judged")])
    out = out.reset_index().rename(columns={"index": "clause_key"})
    out["reader"] = "consensus"
    out.attrs["status"] = {"agreed": len(agreed), "judged": len(settled),
                           "open": int(table["disputed"].sum()) - len(settled)}
    return out

def save_tables(stage: str, rows: pd.DataFrame, readings: dict[str, pd.DataFrame],
                table: pd.DataFrame | None = None, final: pd.DataFrame | None = None,
                pass_dir: Path | None = None) -> list[Path]:
    """Every result of a stage written out (D-217; G&S: keep every result of a slow step), under
    ``pass_dir/tables/``:

    - ``<stage>_readings_<reader>.csv`` — each reader's readings as read, one row a clause, with
      its validity, problems, tokens and cost, and the measure's flags the codebook makes of it
      (``flag_czech_activity`` …; NaN where the reading is not valid);
    - ``<stage>_disputes.csv`` — each clause's votes per flag, whether it is disputed and on what;
    - ``<stage>_consensus.csv`` — the readings the pipeline settles on, with ``status``.

    Everything comes from the cached replies.  Into the shared reply folder only on a pass
    machine (:data:`amateur_theatre_model.PASS_MACHINES`): two machines writing the same
    synced files make conflicted copies.
    """
    pass_dir = Path(pass_dir or atm.PASS_DIR)
    if pass_dir == atm.PASS_DIR:
        atm._pass_machine_guard("writing the stage tables")
    out_dir = pass_dir / "tables"
    out_dir.mkdir(parents=True, exist_ok=True)
    written = []
    for reader, t in readings.items():
        if t is None or not len(t):
            continue
        flags = reading_flags(rows, t).add_prefix("flag_")
        flat = t.merge(flags, left_on="clause_key", right_index=True, how="left", validate="1:1")
        assert len(flat) == len(t), f"{reader}: the flags merge changed the row count"
        written.append(out_dir / f"{stage}_readings_{reader}.csv")
        flat.to_csv(written[-1], index=False, encoding="utf-8-sig")
    if table is not None:
        written.append(out_dir / f"{stage}_disputes.csv")
        table.rename_axis("clause_key").reset_index().to_csv(written[-1], index=False, encoding="utf-8-sig")
    if final is not None and len(final):
        written.append(out_dir / f"{stage}_consensus.csv")
        final.to_csv(written[-1], index=False, encoding="utf-8-sig")
    return written

The register's route reads every crop twice and lets Opus settle what the readers dispute;

   decides the flags.  On a clause the rules already coded, the rules are a third voter, at
   no cost.  A reader with no valid reading makes the clause disputed;

   every reading in front of it, the readers anonymised (``Reader 1``, ``Reader 2``, ``Rules``),
   and gives its own reading in the same schema;
4. **settle** (:func:`consensus`) — an agreed clause takes the first reader's reading, a
   disputed one the judge's; a disputed clause the judge has not read stays open.

What the pipeline cannot see is two readers agreeing and both being wrong: the check

__version__ = "0.2.0"   # save_tables: every reader, the disputes, the consensus written out

from pathlib import Path

import numpy as np
import pandas as pd

from utils.parsing import amateur_theatre_model as atm

#: The two readers and the judge.
PIPELINE = ("haiku", "gemini")
JUDGE = "opus_judge"
#: What a reading is compared on: the flags the measure is built from.
FLAGS = ["czech_activity", "certain", "company", "performance"]
DATE_FIELDS = ["date_form", "start_lo", "start_hi", "end_lo", "end_hi"]


def _flags(coded: pd.DataFrame) -> pd.DataFrame:
    act = coded["czech_activity"].astype(bool)
    return pd.DataFrame({"czech_activity": act, "certain": act & ~coded["hedged"].astype(bool),
                         "company": coded["company"].astype(bool),
                         "performance": coded["performance"].astype(bool)}, index=coded.index)


def reading_flags(rows: pd.DataFrame, readings: pd.DataFrame) -> pd.DataFrame:
    """One reader's flags on ``rows`` (coded clauses, keyed ``clause_key``), its readings run
    through the codebook; NaN where it has no valid reading.  Index ``clause_key``."""
    valid = readings[readings["valid"].astype(bool)] if len(readings) else readings
    read = rows[rows["clause_key"].isin(valid["clause_key"])] if len(valid) else rows.iloc[0:0]
    parts = [atm.apply_readings(g, valid, s).assign(clause_key=g["clause_key"].values)
             for s, g in read.groupby("source")]
    out = _flags(pd.concat(parts).set_index("clause_key")) if parts else pd.DataFrame(columns=FLAGS)
    return out.reindex(rows["clause_key"]).astype("float")


def rules_flags(rows: pd.DataFrame) -> pd.DataFrame:
    """The rules' flags on ``rows``; NaN on a clause the rules left to the model."""
    out = _flags(rows.set_index("clause_key")).astype("float")
    out[rows.set_index("clause_key")["needs_model"].astype(bool)] = np.nan
    return out


def _date_key(readings: pd.DataFrame, keys: pd.Index) -> pd.Series:
    r = readings[readings["valid"].astype(bool)].set_index("clause_key").reindex(keys)
    def one(x):
        if pd.isna(x["date_form"]):
            return None
        years = ["" if pd.isna(x[k]) else str(int(x[k])) for k in DATE_FIELDS[1:]]
        return x["date_form"] + ":" + "/".join(years)
    return r.apply(one, axis=1) if len(r) else pd.Series(dtype=object)


def disputes(rows: pd.DataFrame, readings: dict[str, pd.DataFrame], use_rules: bool = True) -> pd.DataFrame:
    """Per clause of ``rows``: each voter's flags, and whether they dispute (``disputed``) and on
    what (``on``).  Voters: the readers of ``readings`` and, on a clause the rules coded, the
    rules.  A missing reading is a dispute.  An undated clause is disputed also on its interval."""

    if use_rules:
        voters["rules"] = rules_flags(rows)
    dates = {name: _date_key(table, keys) for name, table in readings.items()}

    on = [[] for _ in range(len(keys))]
    for name in readings:
        missing = voters[name][FLAGS].isna().any(axis=1).to_numpy()
        for i in np.where(missing)[0]:
            on[i].append(f"no reading: {name}")
    for flag in FLAGS:

        out[flag + "_votes"] = values.apply(lambda r: "|".join("" if pd.isna(x) else str(int(x)) for x in r), axis=1)
        split = values.nunique(axis=1, dropna=True) > 1

    - ``<stage>_disputes.csv`` — each clause's votes per flag, whether it is disputed and on what;
    - ``<stage>_consensus.csv`` — the readings the pipeline settles on, with ``status``.

    Everything comes from the cached replies.  Into the shared reply folder only on a pass
    machine (:data:`amateur_theatre_model.PASS_MACHINES`): two machines writing the same
    synced files make conflicted copies.
    """
    pass_dir = Path(pass_dir or atm.PASS_DIR)
    if pass_dir == atm.PASS_DIR:
        atm._pass_machine_guard("writing the stage tables")
    out_dir = pass_dir / "tables"
    out_dir.mkdir(parents=True, exist_ok=True)
    written = []
    for reader, t in readings.items():
        if t is None or not len(t):
            continue
        flags = reading_flags(rows, t).add_prefix("flag_")
        flat = t.merge(flags, left_on="clause_key", right_index=True, how="left", validate="1:1")
        assert len(flat) == len(t), f"{reader}: the flags merge changed the row count"
        written.append(out_dir / f"{stage}_readings_{reader}.csv")
        flat.to_csv(written[-1], index=False, encoding="utf-8-sig")
    if table is not None:
        written.append(out_dir / f"{stage}_disputes.csv")
        table.rename_axis("clause_key").reset_index().to_csv(written[-1], index=False, encoding="utf-8-sig")
    if final is not None and len(final):
        written.append(out_dir / f"{stage}_consensus.csv")
        final.to_csv(written[-1], index=False, encoding="utf-8-sig")
    return written
