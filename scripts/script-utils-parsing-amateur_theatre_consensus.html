How it works

The register's route reads every crop twice and lets Opus settle what the readers dispute; this does the same for the chronicle clauses:

  1. read — every clause is read by two models independently (PIPELINE: Haiku 5.5 and Gemini 3.8 Flash, utils.parsing.amateur_theatre_model), batched, the codebook cached;
  2. compare (disputes) — on what reaches the measure only, never on the 22 event types: Czech activity, whether it is certain (not hedged), a company, a performance, and for an undated clause the interval it dates. Each reading is put through amateur_theatre.code_clauses exactly as the rules' cues are, so the codebook (D-141) decides the flags. On a clause the rules already coded, the rules are a third voter, at no cost. A reader with no valid reading makes the clause disputed;
  3. judge (judge_others) — Opus 5.5 at high effort reads each disputed clause with every reading in front of it, the readers anonymised (Reader 1, Reader 2, Rules), and gives its own reading in the same schema;
  4. settle (consensus) — an agreed clause takes the first reader's reading, a disputed one the judge's; a disputed clause the judge has not read stays open.

What the pipeline cannot see is two readers agreeing and both being wrong: the check samples (theatre_rule_audit) measure how often that happens.

def freeze_rules_lines(clauses: pd.DataFrame, path: Path = RULES_LINES) -> Path:
    """``clause_key``, ``cues`` and ``needs_model`` of a clause build made before D-254, for the
    judge's ``Rules`` line.  Refuses to overwrite: a later build is not what the judge saw."""
    path = Path(path)
    if path.exists():
        raise FileExistsError(f"{path}: the pre-D-254 rules lines are written once")
    out = clauses[["clause_key", "cues", "needs_model"]].copy()
    out["cues"] = out["cues"].fillna("")
    out["needs_model"] = out["needs_model"].astype(bool)
    path.parent.mkdir(parents=True, exist_ok=True)
    out.to_csv(path, index=False, encoding="utf-8-sig")
    return path

def rules_lines(path: Path = RULES_LINES) -> pd.DataFrame | None:
    """:func:`freeze_rules_lines`' table, keyed ``clause_key``, or None if it was never written."""
    path = Path(path)

        return None

    out["needs_model"] = out["needs_model"].astype(str).str.lower().eq("true")
    return out.set_index("clause_key")

def reading_flags(rows: pd.DataFrame, readings: pd.DataFrame) -> pd.DataFrame:
    """One reader's flags on ``rows`` (coded clauses, keyed ``clause_key``), its readings run
    through the codebook; NaN where it has no valid reading.  Index ``clause_key``."""
    valid = readings[readings["valid"].astype(bool)] if len(readings) else readings
    read = rows[rows["clause_key"].isin(valid["clause_key"])] if len(valid) else rows.iloc[0:0]
    parts = [atm.apply_readings(g, valid, s).assign(clause_key=g["clause_key"].values)
             for s, g in read.groupby("source")]
    out = _flags(pd.concat(parts).set_index("clause_key")) if parts else pd.DataFrame(columns=FLAGS)
    return out.reindex(rows["clause_key"]).astype("float")

def reading_panel(rows: pd.DataFrame, readings: pd.DataFrame) -> pd.DataFrame:
    """One reader's panel columns on ``rows`` (:data:`PANEL_TYPES` as 0/1 and ``titles`` as a
    sorted ``|`` list), its readings run through the codebook; NaN / None where it has no
    valid reading.  Index ``clause_key``."""
    valid = readings[readings["valid"].astype(bool)] if len(readings) else readings
    read = rows[rows["clause_key"].isin(valid["clause_key"])] if len(valid) else rows.iloc[0:0]
    parts = [atm.apply_readings(g, valid, s).assign(clause_key=g["clause_key"].values)
             for s, g in read.groupby("source")]
    columns = PANEL_TYPES + ["titles"]
    if not parts:
        return pd.DataFrame(index=pd.Index(rows["clause_key"]), columns=columns, dtype=object)
    coded = pd.concat(parts).set_index("clause_key")
    out = coded[PANEL_TYPES].astype(bool).astype(float)
    out["titles"] = coded["titles"].fillna("").map(lambda s: "|".join(sorted(filter(None, str(s).split("|")))))
    return out.reindex(rows["clause_key"])

def disputes(rows: pd.DataFrame, readings: dict[str, pd.DataFrame], use_rules: bool = True) -> pd.DataFrame:
    """Per clause of ``rows``: each voter's flags, and whether they dispute (``disputed``) and on
    what (``on``).  Voters: the readers of ``readings`` and, on a clause the rules coded, the
    rules.  A missing reading is a dispute.  An undated clause is disputed also on its interval."""
    keys = pd.Index(rows["clause_key"])
    undated = rows.set_index("clause_key")["date_source"].eq("none")
    voters = {name: reading_flags(rows, table) for name, table in readings.items()}
    if use_rules:
        voters["rules"] = rules_flags(rows)
    dates = {name: _date_key(table, keys) for name, table in readings.items()}
    out = pd.DataFrame(index=keys)
    on = [[] for _ in range(len(keys))]
    for name in readings:
        missing = voters[name][FLAGS].isna().any(axis=1).to_numpy()
        for i in np.where(missing)[0]:
            on[i].append(f"no reading: {name}")
    for flag in FLAGS:
        values = pd.DataFrame({n: v[flag] for n, v in voters.items()}, index=keys)
        out[flag + "_votes"] = values.apply(lambda r: "|".join("" if pd.isna(x) else str(int(x)) for x in r), axis=1)
        split = values.nunique(axis=1, dropna=True) > 1
        for i in np.where(split.to_numpy())[0]:
            on[i].append(flag)
    if len(readings) > 1:
        d = pd.DataFrame(dates, index=keys)
        split = undated.reindex(keys).to_numpy() & (d.nunique(axis=1, dropna=False) > 1).to_numpy()
        for i in np.where(split)[0]:
            on[i].append("date")
        # audit B5 (D-225): the model readers must also agree on the panel's other columns;
        # the rules do not vote here (their cue matches are not a reading of these columns)
        panels = {name: reading_panel(rows, table) for name, table in readings.items()}
        for column in PANEL_TYPES + ["titles"]:
            values = pd.DataFrame({n: p[column] for n, p in panels.items()}, index=keys)
            split = values.nunique(axis=1, dropna=True) > 1
            for i in np.where(split.to_numpy())[0]:
                on[i].append(f"panel:{column}")
    out["on"] = [", ".join(dict.fromkeys(x)) for x in on]
    out["disputed"] = out["on"] != ""
    out["voters"] = "|".join(voters)
    return out

def reader_order(key: str, names: list[str]) -> list[str]:
    """The model readers in the order the judge sees them for one clause: a permutation seeded
    by the clause key, so it is the same on every run (the request stays cached) but no reader
    is always ``Reader 1`` (audit B7)."""
    seed = int(hashlib.sha256(str(key).encode("utf-8")).hexdigest()[:16], 16)
    return [names[i] for i in np.random.default_rng(seed).permutation(len(names))]

def judge_others(disputed: pd.Index, readings: dict[str, pd.DataFrame], rows: pd.DataFrame,
                 frozen_lines: pd.DataFrame | None | bool = True) -> dict[str, list[dict]]:
    """For each disputed clause, the readings the judge weighs, anonymised: ``Reader 1`` … for
    the models, in :func:`reader_order`, then ``Rules`` (event types and hedge) on a coded
    clause.  A clause read with ``context="window"`` keeps the fixed order of the 2026-11-11
    requests (D-217), whose replies are cached under it.  In a context read before D-254
    (:data:`FROZEN_CONTEXTS`) the ``Rules`` line is the pre-D-254 one (``frozen_lines``: by
    default :func:`rules_lines`; False uses the rows' own cues)."""
    by = {name: t[t["valid"].astype(bool)].set_index("clause_key") for name, t in readings.items()}
    coded = rows.set_index("clause_key")
    frozen = coded["context_kind"].eq("window") if "context_kind" in coded else pd.Series(True, index=coded.index)
    kind = coded["context_kind"] if "context_kind" in coded else pd.Series("window", index=coded.index)
    lines = rules_lines() if frozen_lines is True else (None if frozen_lines is False else frozen_lines)
    if lines is not None:
        old = kind.isin(FROZEN_CONTEXTS) & coded.index.isin(lines.index)
        if old.any():
            coded = coded.copy()
            keys = coded.index[old]
            coded.loc[keys, "cues"] = lines.loc[keys, "cues"].values
            coded.loc[keys, "needs_model"] = lines.loc[keys, "needs_model"].values
    out = {}
    for key in disputed:
        others = []
        names = list(readings) if frozen.get(key, True) else reader_order(key, list(readings))
        for i, name in enumerate(names, start=1):
            if key in by[name].index:
                r = by[name].loc[key]
                others.append({"name": f"Reader {i}", "types": r["types"], "hedged": bool(r["hedged"]),
                               "titles": r["titles"], **({k: r[k] for k in DATE_FIELDS} if coded.at[key, "date_source"] == "none" else {})})
            else:
                others.append({"name": f"Reader {i}", "types": "(no reading)", "hedged": False, "titles": ""})
        if not bool(coded.at[key, "needs_model"]):
            cues = [c for c in str(coded.at[key, "cues"] if pd.notna(coded.at[key, "cues"]) else "").split("|") if c]
            others.append({"name": "Rules", "types": "|".join(c for c in cues if c not in ("hedged", "title")),
                           "hedged": "hedged" in cues, "titles": ""})
        out[key] = others
    return out

def consensus(table: pd.DataFrame, readings: dict[str, pd.DataFrame], judged: pd.DataFrame | None,
              rows: pd.DataFrame | None = None) -> pd.DataFrame:
    """The readings to apply: an agreed clause takes the first reader's reading (``status``
    ``agreed``), a disputed one the judge's (``judged``); a disputed clause without a valid
    judge reading is left out (``open``: the rules stand, or the clause stays open) — unless
    it is disputed on the panel's other columns only (audit B5, D-225): then the readers agree
    on everything the measure's flags rest on, and it takes the first reader's reading with
    only what every reader gave on the disputed columns (``agreed_core``, :func:`_agreed_only`;
    needs ``rows``, the clauses, to check the flags do not move — without them, or where they
    would, the first reader's reading stands).  ``attrs["status"]`` counts each status."""
    first = next(iter(readings))
    lead = readings[first][readings[first]["valid"].astype(bool)].set_index("clause_key")
    judge = (judged[judged["valid"].astype(bool)].set_index("clause_key") if judged is not None and len(judged)
             else pd.DataFrame(columns=lead.columns))
    agreed = table.index[~table["disputed"]].intersection(lead.index)
    settled = table.index[table["disputed"]].intersection(judge.index)
    unjudged = table.index[table["disputed"]].difference(settled).intersection(lead.index)
    panel_only = [k for k in unjudged
                  if all(p.startswith("panel:") for p in table.at[k, "on"].split(", ") if p)]
    core = _agreed_only(panel_only, readings, table, rows) if panel_only else lead.iloc[0:0]
    out = pd.concat([lead.loc[agreed].assign(status="agreed"), judge.loc[settled].assign(status="judged"),
                     core.assign(status="agreed_core")])
    out = out.reset_index().rename(columns={"index": "clause_key"})
    out["reader"] = "consensus"
    out.attrs["status"] = {"agreed": len(agreed), "judged": len(settled), "agreed_core": len(core),
                           "open": int(table["disputed"].sum()) - len(settled) - len(core)}
    return out

def save_tables(stage: str, rows: pd.DataFrame, readings: dict[str, pd.DataFrame],
                table: pd.DataFrame | None = None, final: pd.DataFrame | None = None,
                pass_dir: Path | None = None) -> list[Path]:
    """Every result of a stage written out (D-217; G&S: keep every result of a slow step), under
    ``pass_dir/tables/``:

    - ``<stage>_readings_<reader>.csv`` — each reader's readings as read, one row a clause, with
      its validity, problems, tokens and cost, and the measure's flags the codebook makes of it
      (``flag_czech_activity`` …; NaN where the reading is not valid);
    - ``<stage>_disputes.csv`` — each clause's votes per flag, whether it is disputed and on what;
    - ``<stage>_consensus.csv`` — the readings the pipeline settles on, with ``status``.

    Everything comes from the cached replies.  Into the shared reply folder only on a pass
    machine (:data:`amateur_theatre_model.PASS_MACHINES`): two machines writing the same
    synced files make conflicted copies.
    """
    pass_dir = Path(pass_dir or atm.PASS_DIR)
    if pass_dir == atm.PASS_DIR:
        atm._pass_machine_guard("writing the stage tables")
    out_dir = pass_dir / "tables"
    out_dir.mkdir(parents=True, exist_ok=True)
    written = []
    for reader, t in readings.items():
        if t is None or not len(t):
            continue
        flags = reading_flags(rows, t).add_prefix("flag_")
        flat = t.merge(flags, left_on="clause_key", right_index=True, how="left", validate="1:1")
        assert len(flat) == len(t), f"{reader}: the flags merge changed the row count"
        written.append(out_dir / f"{stage}_readings_{reader}.csv")
        flat.to_csv(written[-1], index=False, encoding="utf-8-sig")
    if table is not None:
        written.append(out_dir / f"{stage}_disputes.csv")
        table.rename_axis("clause_key").reset_index().to_csv(written[-1], index=False, encoding="utf-8-sig")
    if final is not None and len(final):
        written.append(out_dir / f"{stage}_consensus.csv")
        final.to_csv(written[-1], index=False, encoding="utf-8-sig")
    return written

"""Two readers, the rules, and a judge: the theatre model pass as a pipeline (D-217).

The register's route reads every crop twice and lets Opus settle what the readers dispute;
this does the same for the chronicle clauses:

1. **read** — every clause is read by two models independently (``PIPELINE``: Haiku 5.5 and
   Gemini 3.8 Flash, :mod:`utils.parsing.amateur_theatre_model`), batched, the codebook cached;
2. **compare** (:func:`disputes`) — on what reaches the measure only, never on the 22 event
   types: Czech activity, whether it is certain (not hedged), a company, a performance, and
   for an undated clause the interval it dates.  Each reading is put through
   :func:`amateur_theatre.code_clauses` exactly as the rules' cues are, so the codebook (D-141)
   decides the flags.  On a clause the rules already coded, the rules are a third voter, at
   no cost.  A reader with no valid reading makes the clause disputed;
3. **judge** (:func:`judge_others`) — Opus 5.5 at high effort reads each disputed clause with
   every reading in front of it, the readers anonymised (``Reader 1``, ``Reader 2``, ``Rules``),
   and gives its own reading in the same schema;
4. **settle** (:func:`consensus`) — an agreed clause takes the first reader's reading, a
   disputed one the judge's; a disputed clause the judge has not read stays open.

What the pipeline cannot see is two readers agreeing and both being wrong: the check
samples (``theatre_rule_audit``) measure how often that happens.
"""

from __future__ import annotations

__version__ = "0.5.0"   # D-254: the judge's Rules line frozen for the cached contexts
# 0.4.0 D-225 (audit B5): panel columns disputed too; 0.3.0 B7 reader order; 0.2.0 save_tables

import hashlib
from pathlib import Path

import numpy as np
import pandas as pd

from utils.parsing import amateur_theatre_model as atm

#: The two readers and the judge.
PIPELINE = ("haiku", "gemini")
JUDGE = "opus_judge"
#: What a reading is compared on: the flags the measure is built from.
FLAGS = ["czech_activity", "certain", "company", "performance"]
DATE_FIELDS = ["date_form", "start_lo", "start_hi", "end_lo", "end_hi"]
#: D-254: the rules' cues as the judge saw them in the requests sent before D-254 (the window
#: check stage and the D-221 page context).  Rebuilding the clauses under the D-254 rules changes
#: the cues, and with them the judge's ``Rules`` line and its request key; the line is taken
#: from here for those contexts, so the paid replies stay readable.  Written once, from a
#: clause build made before D-254 (:func:`freeze_rules_lines`).
RULES_LINES = atm.PASS_DIR / "judge_rules_lines_pre_d254.csv"
FROZEN_CONTEXTS = ("window", "page_d221")


def freeze_rules_lines(clauses: pd.DataFrame, path: Path = RULES_LINES) -> Path:
    """``clause_key``, ``cues`` and ``needs_model`` of a clause build made before D-254, for the
    judge's ``Rules`` line.  Refuses to overwrite: a later build is not what the judge saw."""
    path = Path(path)
    if path.exists():
        raise FileExistsError(f"{path}: the pre-D-254 rules lines are written once")
    out = clauses[["clause_key", "cues", "needs_model"]].copy()
    out["cues"] = out["cues"].fillna("")
    out["needs_model"] = out["needs_model"].astype(bool)
    path.parent.mkdir(parents=True, exist_ok=True)
    out.to_csv(path, index=False, encoding="utf-8-sig")
    return path


def rules_lines(path: Path = RULES_LINES) -> pd.DataFrame | None:
    """:func:`freeze_rules_lines`' table, keyed ``clause_key``, or None if it was never written."""
    path = Path(path)
    if not path.exists():
        return None
    out = pd.read_csv(path, encoding="utf-8-sig", dtype={"cues": str}, keep_default_na=False)
    out["needs_model"] = out["needs_model"].astype(str).str.lower().eq("true")
    return out.set_index("clause_key")


def _flags(coded: pd.DataFrame) -> pd.DataFrame:
    act = coded["czech_activity"].astype(bool)
    return pd.DataFrame({"czech_activity": act, "certain": act & ~coded["hedged"].astype(bool),
                         "company": coded["company"].astype(bool),
                         "performance": coded["performance"].astype(bool)}, index=coded.index)


def reading_flags(rows: pd.DataFrame, readings: pd.DataFrame) -> pd.DataFrame:
    """One reader's flags on ``rows`` (coded clauses, keyed ``clause_key``), its readings run
    through the codebook; NaN where it has no valid reading.  Index ``clause_key``."""
    valid = readings[readings["valid"].astype(bool)] if len(readings) else readings
    read = rows[rows["clause_key"].isin(valid["clause_key"])] if len(valid) else rows.iloc[0:0]
    parts = [atm.apply_readings(g, valid, s).assign(clause_key=g["clause_key"].values)
             for s, g in read.groupby("source")]
    out = _flags(pd.concat(parts).set_index("clause_key")) if parts else pd.DataFrame(columns=FLAGS)
    return out.reindex(rows["clause_key"]).astype("float")


#: Audit B5 (D-225): the panel's other columns an agreed clause would carry.  Two readers that
#: agree on :data:`FLAGS` but not on these dispute too, so the judge settles them; ``bilingual``
#: follows from ``german`` and the clause's words, so it is not compared on its own.
PANEL_TYPES = ["founded", "statutes", "permit", "first", "proceeds", "national_theatre",
               "school_church", "travelling", "german", "latin", "attendance"]


def reading_panel(rows: pd.DataFrame, readings: pd.DataFrame) -> pd.DataFrame:
    """One reader's panel columns on ``rows`` (:data:`PANEL_TYPES` as 0/1 and ``titles`` as a
    sorted ``|`` list), its readings run through the codebook; NaN / None where it has no
    valid reading.  Index ``clause_key``."""
    valid = readings[readings["valid"].astype(bool)] if len(readings) else readings
    read = rows[rows["clause_key"].isin(valid["clause_key"])] if len(valid) else rows.iloc[0:0]
    parts = [atm.apply_readings(g, valid, s).assign(clause_key=g["clause_key"].values)
             for s, g in read.groupby("source")]
    columns = PANEL_TYPES + ["titles"]
    if not parts:
        return pd.DataFrame(index=pd.Index(rows["clause_key"]), columns=columns, dtype=object)
    coded = pd.concat(parts).set_index("clause_key")
    out = coded[PANEL_TYPES].astype(bool).astype(float)
    out["titles"] = coded["titles"].fillna("").map(lambda s: "|".join(sorted(filter(None, str(s).split("|")))))
    return out.reindex(rows["clause_key"])


def rules_flags(rows: pd.DataFrame) -> pd.DataFrame:
    """The rules' flags on ``rows``; NaN on a clause the rules left to the model."""
    out = _flags(rows.set_index("clause_key")).astype("float")
    out[rows.set_index("clause_key")["needs_model"].astype(bool)] = np.nan
    return out


def _date_key(readings: pd.DataFrame, keys: pd.Index) -> pd.Series:
    r = readings[readings["valid"].astype(bool)].set_index("clause_key").reindex(keys)
    def one(x):
        if pd.isna(x["date_form"]):
            return None
        years = ["" if pd.isna(x[k]) else str(int(x[k])) for k in DATE_FIELDS[1:]]
        return x["date_form"] + ":" + "/".join(years)
    return r.apply(one, axis=1) if len(r) else pd.Series(dtype=object)


def disputes(rows: pd.DataFrame, readings: dict[str, pd.DataFrame], use_rules: bool = True) -> pd.DataFrame:
    """Per clause of ``rows``: each voter's flags, and whether they dispute (``disputed``) and on
    what (``on``).  Voters: the readers of ``readings`` and, on a clause the rules coded, the
    rules.  A missing reading is a dispute.  An undated clause is disputed also on its interval."""
    keys = pd.Index(rows["clause_key"])
    undated = rows.set_index("clause_key")["date_source"].eq("none")
    voters = {name: reading_flags(rows, table) for name, table in readings.items()}
    if use_rules:
        voters["rules"] = rules_flags(rows)
    dates = {name: _date_key(table, keys) for name, table in readings.items()}
    out = pd.DataFrame(index=keys)
    on = [[] for _ in range(len(keys))]
    for name in readings:
        missing = voters[name][FLAGS].isna().any(axis=1).to_numpy()
        for i in np.where(missing)[0]:
            on[i].append(f"no reading: {name}")
    for flag in FLAGS:
        values = pd.DataFrame({n: v[flag] for n, v in voters.items()}, index=keys)
        out[flag + "_votes"] = values.apply(lambda r: "|".join("" if pd.isna(x) else str(int(x)) for x in r), axis=1)
        split = values.nunique(axis=1, dropna=True) > 1
        for i in np.where(split.to_numpy())[0]:
            on[i].append(flag)
    if len(readings) > 1:
        d = pd.DataFrame(dates, index=keys)
        split = undated.reindex(keys).to_numpy() & (d.nunique(axis=1, dropna=False) > 1).to_numpy()
        for i in np.where(split)[0]:
            on[i].append("date")
        # audit B5 (D-225): the model readers must also agree on the panel's other columns;
        # the rules do not vote here (their cue matches are not a reading of these columns)
        panels = {name: reading_panel(rows, table) for name, table in readings.items()}
        for column in PANEL_TYPES + ["titles"]:
            values = pd.DataFrame({n: p[column] for n, p in panels.items()}, index=keys)
            split = values.nunique(axis=1, dropna=True) > 1
            for i in np.where(split.to_numpy())[0]:
                on[i].append(f"panel:{column}")
    out["on"] = [", ".join(dict.fromkeys(x)) for x in on]
    out["disputed"] = out["on"] != ""
    out["voters"] = "|".join(voters)
    return out


def reader_order(key: str, names: list[str]) -> list[str]:
    """The model readers in the order the judge sees them for one clause: a permutation seeded
    by the clause key, so it is the same on every run (the request stays cached) but no reader
    is always ``Reader 1`` (audit B7)."""
    seed = int(hashlib.sha256(str(key).encode("utf-8")).hexdigest()[:16], 16)
    return [names[i] for i in np.random.default_rng(seed).permutation(len(names))]


def judge_others(disputed: pd.Index, readings: dict[str, pd.DataFrame], rows: pd.DataFrame,
                 frozen_lines: pd.DataFrame | None | bool = True) -> dict[str, list[dict]]:
    """For each disputed clause, the readings the judge weighs, anonymised: ``Reader 1`` … for
    the models, in :func:`reader_order`, then ``Rules`` (event types and hedge) on a coded
    clause.  A clause read with ``context="window"`` keeps the fixed order of the 2026-11-11
    requests (D-217), whose replies are cached under it.  In a context read before D-254
    (:data:`FROZEN_CONTEXTS`) the ``Rules`` line is the pre-D-254 one (``frozen_lines``: by
    default :func:`rules_lines`; False uses the rows' own cues)."""
    by = {name: t[t["valid"].astype(bool)].set_index("clause_key") for name, t in readings.items()}
    coded = rows.set_index("clause_key")
    frozen = coded["context_kind"].eq("window") if "context_kind" in coded else pd.Series(True, index=coded.index)
    kind = coded["context_kind"] if "context_kind" in coded else pd.Series("window", index=coded.index)
    lines = rules_lines() if frozen_lines is True else (None if frozen_lines is False else frozen_lines)
    if lines is not None:
        old = kind.isin(FROZEN_CONTEXTS) & coded.index.isin(lines.index)
        if old.any():
            coded = coded.copy()
            keys = coded.index[old]
            coded.loc[keys, "cues"] = lines.loc[keys, "cues"].values
            coded.loc[keys, "needs_model"] = lines.loc[keys, "needs_model"].values
    out = {}
    for key in disputed:
        others = []
        names = list(readings) if frozen.get(key, True) else reader_order(key, list(readings))
        for i, name in enumerate(names, start=1):
            if key in by[name].index:
                r = by[name].loc[key]
                others.append({"name": f"Reader {i}", "types": r["types"], "hedged": bool(r["hedged"]),
                               "titles": r["titles"], **({k: r[k] for k in DATE_FIELDS} if coded.at[key, "date_source"] == "none" else {})})
            else:
                others.append({"name": f"Reader {i}", "types": "(no reading)", "hedged": False, "titles": ""})
        if not bool(coded.at[key, "needs_model"]):
            cues = [c for c in str(coded.at[key, "cues"] if pd.notna(coded.at[key, "cues"]) else "").split("|") if c]
            others.append({"name": "Rules", "types": "|".join(c for c in cues if c not in ("hedged", "title")),
                           "hedged": "hedged" in cues, "titles": ""})
        out[key] = others
    return out


def consensus(table: pd.DataFrame, readings: dict[str, pd.DataFrame], judged: pd.DataFrame | None,
              rows: pd.DataFrame | None = None) -> pd.DataFrame:
    """The readings to apply: an agreed clause takes the first reader's reading (``status``
    ``agreed``), a disputed one the judge's (``judged``); a disputed clause without a valid
    judge reading is left out (``open``: the rules stand, or the clause stays open) — unless
    it is disputed on the panel's other columns only (audit B5, D-225): then the readers agree
    on everything the measure's flags rest on, and it takes the first reader's reading with
    only what every reader gave on the disputed columns (``agreed_core``, :func:`_agreed_only`;
    needs ``rows``, the clauses, to check the flags do not move — without them, or where they
    would, the first reader's reading stands).  ``attrs["status"]`` counts each status."""
    first = next(iter(readings))
    lead = readings[first][readings[first]["valid"].astype(bool)].set_index("clause_key")
    judge = (judged[judged["valid"].astype(bool)].set_index("clause_key") if judged is not None and len(judged)
             else pd.DataFrame(columns=lead.columns))
    agreed = table.index[~table["disputed"]].intersection(lead.index)
    settled = table.index[table["disputed"]].intersection(judge.index)
    unjudged = table.index[table["disputed"]].difference(settled).intersection(lead.index)
    panel_only = [k for k in unjudged
                  if all(p.startswith("panel:") for p in table.at[k, "on"].split(", ") if p)]
    core = _agreed_only(panel_only, readings, table, rows) if panel_only else lead.iloc[0:0]
    out = pd.concat([lead.loc[agreed].assign(status="agreed"), judge.loc[settled].assign(status="judged"),
                     core.assign(status="agreed_core")])
    out = out.reset_index().rename(columns={"index": "clause_key"})
    out["reader"] = "consensus"
    out.attrs["status"] = {"agreed": len(agreed), "judged": len(settled), "agreed_core": len(core),
                           "open": int(table["disputed"].sum()) - len(settled) - len(core)}
    return out


def _agreed_only(keys: list, readings: dict[str, pd.DataFrame], table: pd.DataFrame,
                 rows: pd.DataFrame | None) -> pd.DataFrame:
    """For clauses disputed on panel columns only: the first reader's reading less every
    disputed panel type not all readers gave, and with only the titles all gave (B5's
    agreement-only rule).  Where that would move one of the four flags (a panel type can make
    a company or Czech activity), or without ``rows`` to check it on, the reading is kept."""
    valid = {n: t[t["valid"].astype(bool)].set_index("clause_key") for n, t in readings.items()}
    first = next(iter(valid))
    merged = valid[first].loc[keys].copy()
    if rows is None:
        return merged
    for key in keys:
        on = [p.split(":", 1)[1] for p in table.at[key, "on"].split(", ") if p.startswith("panel:")]
        sets = [set(filter(None, str(valid[n].at[key, "types"]).split("|"))) for n in valid if key in valid[n].index]
        shared = set.intersection(*sets)
        drop = {t for t in on if t != "titles" and t not in shared}
        merged.at[key, "types"] = "|".join(t for t in str(merged.at[key, "types"]).split("|") if t and t not in drop)
        if "titles" in on:
            titles = [set(filter(None, str(valid[n].at[key, "titles"]).split("|"))) for n in valid if key in valid[n].index]
            merged.at[key, "titles"] = "|".join(sorted(set.intersection(*titles)))
    subset = rows[rows["clause_key"].isin(keys)]
    before = reading_flags(subset, valid[first].loc[keys].reset_index())
    after = reading_flags(subset, merged.reset_index())
    moved = (before.fillna(-1) != after.fillna(-1)).any(axis=1)
    for key in moved[moved].index:
        merged.loc[key] = valid[first].loc[key]
    return merged


def save_tables(stage: str, rows: pd.DataFrame, readings: dict[str, pd.DataFrame],
                table: pd.DataFrame | None = None, final: pd.DataFrame | None = None,
                pass_dir: Path | None = None) -> list[Path]:
    """Every result of a stage written out (D-217; G&S: keep every result of a slow step), under
    ``pass_dir/tables/``:

    - ``<stage>_readings_<reader>.csv`` — each reader's readings as read, one row a clause, with
      its validity, problems, tokens and cost, and the measure's flags the codebook makes of it
      (``flag_czech_activity`` …; NaN where the reading is not valid);
    - ``<stage>_disputes.csv`` — each clause's votes per flag, whether it is disputed and on what;
    - ``<stage>_consensus.csv`` — the readings the pipeline settles on, with ``status``.

    Everything comes from the cached replies.  Into the shared reply folder only on a pass
    machine (:data:`amateur_theatre_model.PASS_MACHINES`): two machines writing the same
    synced files make conflicted copies.
    """
    pass_dir = Path(pass_dir or atm.PASS_DIR)
    if pass_dir == atm.PASS_DIR:
        atm._pass_machine_guard("writing the stage tables")
    out_dir = pass_dir / "tables"
    out_dir.mkdir(parents=True, exist_ok=True)
    written = []
    for reader, t in readings.items():
        if t is None or not len(t):
            continue
        flags = reading_flags(rows, t).add_prefix("flag_")
        flat = t.merge(flags, left_on="clause_key", right_index=True, how="left", validate="1:1")
        assert len(flat) == len(t), f"{reader}: the flags merge changed the row count"
        written.append(out_dir / f"{stage}_readings_{reader}.csv")
        flat.to_csv(written[-1], index=False, encoding="utf-8-sig")
    if table is not None:
        written.append(out_dir / f"{stage}_disputes.csv")
        table.rename_axis("clause_key").reset_index().to_csv(written[-1], index=False, encoding="utf-8-sig")
    if final is not None and len(final):
        written.append(out_dir / f"{stage}_consensus.csv")
        final.to_csv(written[-1], index=False, encoding="utf-8-sig")
    return written
