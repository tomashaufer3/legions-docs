How it works

The blind ruling of the 139 calibration clauses (D-228) disagreed with the codebook in fourteen general ways; reference/places/theatre_codebook_questions.json sets them out with the codebook's present wording, what the blind ruling did, and the options. The author ruled them on the Theatre Codebook Rulings page (Q1) and in chat (Q2–Q14 and the rare cases, 2026-11-17); every answer is in the page's rulings collection. This module reads that collection back into reference/places/theatre_codebook_rulings.csv, one row per question, with the option text as ruled and the author's note.

Usage::

python utils/reference_tools/places/theatre_codebook_rulings.py DIR   # the saved `rulings` documents

def read_back(documents: list[dict], path: Path = SHEET) -> pd.DataFrame:
    """The ruled documents into ``path``, in question order; refuses a choice the question does
    not offer, and says which questions are still open."""
    questions = json.loads(QUESTIONS.read_text(encoding="utf-8"))["questions"]
    titles = {q["id"]: q["title"] for q in questions}
    offered = {q["id"]: {o[1] for o in q["options"]} for q in questions}
    rows = {}
    for d in documents:
        q = d.get("question")
        if q in offered and d.get("choice") not in offered[q]:
            raise ValueError(f"{q}: choice {d.get('choice')!r} is not one of {sorted(offered[q])}")
        rows[q] = {"question": q, "title": titles.get(q, "Rare cases" if q == "rare" else ""),
                   **{c: d.get(c) or "" for c in ("choice", "option", "note", "updated_at")}}
    order = [*titles, "rare"]
    out = pd.DataFrame([rows[q] for q in order if q in rows], columns=COLUMNS)
    out.to_csv(path, index=False, encoding="utf-8-sig")
    open_ = [q for q in order if q not in rows]
    print(f"{len(out)} ruled; open: {open_ or 'none'}")
    return out

#!/usr/bin/env python3
"""
The author's rulings on the fourteen codebook questions the blind check raised (D-254).

The blind ruling of the 139 calibration clauses (D-228) disagreed with the codebook in
fourteen general ways; ``reference/places/theatre_codebook_questions.json`` sets them out with
the codebook's present wording, what the blind ruling did, and the options.  The author ruled
them on the *Theatre Codebook Rulings* page (Q1) and in chat (Q2–Q14 and the rare cases,
2026-11-17); every answer is in the page's ``rulings`` collection.  This module reads that
collection back into ``reference/places/theatre_codebook_rulings.csv``, one row per question,
with the option text as ruled and the author's note.

Usage::

    python utils/reference_tools/places/theatre_codebook_rulings.py DIR   # the saved ``rulings`` documents
"""

from __future__ import annotations

__version__ = "0.1.0"   # D-254

import json
import sys
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[3]
QUESTIONS = ROOT / "reference" / "places" / "theatre_codebook_questions.json"
SHEET = ROOT / "reference" / "places" / "theatre_codebook_rulings.csv"
COLUMNS = ["question", "title", "choice", "option", "note", "updated_at"]


def read_back(documents: list[dict], path: Path = SHEET) -> pd.DataFrame:
    """The ruled documents into ``path``, in question order; refuses a choice the question does
    not offer, and says which questions are still open."""
    questions = json.loads(QUESTIONS.read_text(encoding="utf-8"))["questions"]
    titles = {q["id"]: q["title"] for q in questions}
    offered = {q["id"]: {o[1] for o in q["options"]} for q in questions}
    rows = {}
    for d in documents:
        q = d.get("question")
        if q in offered and d.get("choice") not in offered[q]:
            raise ValueError(f"{q}: choice {d.get('choice')!r} is not one of {sorted(offered[q])}")
        rows[q] = {"question": q, "title": titles.get(q, "Rare cases" if q == "rare" else ""),
                   **{c: d.get(c) or "" for c in ("choice", "option", "note", "updated_at")}}
    order = [*titles, "rare"]
    out = pd.DataFrame([rows[q] for q in order if q in rows], columns=COLUMNS)
    out.to_csv(path, index=False, encoding="utf-8-sig")
    open_ = [q for q in order if q not in rows]
    print(f"{len(out)} ruled; open: {open_ or 'none'}")
    return out


if __name__ == "__main__":
    if len(sys.argv) != 2:
        raise SystemExit(__doc__)
    docs = [json.loads(f.read_text(encoding="utf-8")) for f in sorted(Path(sys.argv[1]).glob("*.json"))]
    read_back([d.get("data") or d for d in docs])
    print("wrote", SHEET)
