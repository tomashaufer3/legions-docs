How it works

The register explains its pictograms on one plate — Výklad značek, p. XI, scan 0013 — printed three columns wide, one symbol to a line with its Czech meaning beside it::

®  místní obec.              S  obecná škola.       À  četník.
C  osada.                    ?  lékař.              1' hasičský zbor.
Ш  katastrální (berní) obec. +  pohodný.            ;šľ spořitelna.

Until now the vision channel was handed an English word list — [school] primary school — and asked to map a 1916 printed glyph onto it. It had never seen this typeface's symbols, so it matched on resemblance, which is why two readers given the identical mark returned [doctor] and [rc-curate]. This script replaces the word list with the printed forms themselves.

The plate is self-labelling, which is what makes this mostly mechanical. The OCR destroys every symbol on it (®, Ш, <3*, €, tí) and reads every gloss cleanly — and those glosses are already the expansion column of institution_abbreviations.csv. So the gloss says which token the symbol is, and the plate's own geometry says where the symbol is: each column is a narrow band of pictograms, a blank gutter, then the meanings, so the widest block of ink in a column is the gloss and everything left of it is the picture. Taking the crop boundary from the layout rather than from the symbol's own characters is what makes it work on the lines where the OCR emitted nothing for the symbol at all.

What comes out

symbols/&lt;token&gt;.png one crop per pictogram, at scan resolution symbols/manifest.csv token, printed gloss, match score, source box symbols/atlas.png the crops with their tokens beside them, as one image — this is what a vision model can be shown

And a report of the three ways the plate and the curated table disagree, which is the other thing this buys. A plate line that matches no token means the token list is incomplete and any such symbol on a page comes back [?] however good the reader is; the plate prints turistická bouda, salaš, koliha and klášterec, none of which the table currently carries. A token with no plate line means it was inferred rather than read off the legend.

def cluster_lines(words, gap: int = 18):
    """``[(text, words, top, bottom, left, right)]`` for one column."""
    rows: list[list[tuple]] = []
    for word in sorted(words, key=lambda s: (s[0] + s[3] // 2, s[1])):
        centre = word[0] + word[3] // 2
        if rows and centre - (rows[-1][-1][0] + rows[-1][-1][3] // 2) <= gap:
            rows[-1].append(word)
        else:
            rows.append([word])
    out = []
    for row in rows:
        row = sorted(row, key=lambda w: w[1])
        out.append((" ".join(w[4] for w in row).strip(), row,
                    min(w[0] for w in row), max(w[0] + w[3] for w in row),
                    min(w[1] for w in row), max(w[1] + w[2] for w in row)))
    return out

def columns_of(page, words) -> list[tuple[int, int, int]]:
    """``(column_left, gloss_left, column_right)`` for each printed column.

    Taken from where the ink is, not from where words start. The plate sets each
    column as a narrow band of pictograms, a blank gutter, then the meanings — so
    the widest block of covered x inside a column *is* the gloss, and everything to
    its left is the symbol. That makes the crop boundary a measurement rather than
    an inference, and it works on the lines where the OCR emitted nothing for the
    symbol at all, which a rule keyed on the symbol's own characters cannot.
    """
    out: list[tuple[int, int, int]] = []
    for left, right in _covered(words, _COLUMN_GAP):
        inner = [(a, b) for a, b in _covered(
            [w for w in words if left <= w[1] <= right], 4)]
        wide = [iv for iv in inner if iv[1] - iv[0] >= _MIN_GLOSS_WIDTH]
        if not wide:
            continue
        out.append((left, wide[0][0], right))
    return out

def match_gloss(gloss: str, targets):
    """Name the token whose curated expansion this printed meaning is.

    The plate's glosses *are* the ``expansion`` column — ``obecná škola``,
    ``lékař``, ``hasičský zbor`` — so this is a lookup with OCR damage allowed,
    not an interpretation. Scored on the normalised strings so accents and
    punctuation the scanner mangles cannot decide it.

    ``token_sort_ratio`` rather than a plain ratio because the plate and the table
    order a two-part meaning differently often enough to matter (``mlýn, pila,
    brusírna``), and a whole-string comparison would rank a short unrelated token
    above the right long one.
    """
    from rapidfuzz import fuzz

    key = normalise(gloss)
    if len(key) < 3:
        return "", "", 0.0
    best = ("", "", 0.0)
    for token, expansion, _en, _cat in targets:
        target = normalise(expansion)
        score = max(fuzz.ratio(target, key), fuzz.token_sort_ratio(target, key))
        # The plate elaborates where the table is terse — it prints "skupina
        # chalup (více rozptýlených domů vzdálených od sebe na vrh kamenem…)"
        # for what the table calls "skupina chalup". A whole-string comparison
        # scores that near zero, so a containment measure is allowed too — but
        # only for a target long enough that containment means something, or
        # every three-letter expansion would match every gloss on the plate.
        if len(target) >= 6:
            # Penalised, so containment never beats a good whole-string match.
            # Without this "městys" was captured by [mestys-part], whose curated
            # expansion "část městyse" *contains* it and scored an unpenalised
            # 100 — the longer code winning the shorter code's own gloss.
            score = max(score, fuzz.partial_ratio(target, key) - 6)
        if score > best[2]:
            best = (token, expansion, score)
    return best

def build(scan: int, show: bool = False) -> int:
    from PIL import Image, ImageDraw, ImageFont

    page = lx.read_alto(CZ_PAGES / f"{scan:04d}.alto.xml")
    image = lx.render_page(CZ_PDF, scan, page.width)
    targets = load_targets()
    print(f"  plate: scan {scan:04d}, {page.width}x{page.height}, "
          f"{len(page.strings)} words; {len(targets)} curated pictograms")

    bounds = columns_of(page, page.strings)
    print(f"  {len(bounds)} printed columns (symbol band | gloss): "
          + ", ".join(f"{a}-{g} | {g}-{b}" for a, g, b in bounds))

    OUT_DIR.mkdir(parents=True, exist_ok=True)
    manifest: list[dict] = []
    unmatched: list[dict] = []
    used: dict[str, int] = {}

    for left, gloss_left, right in bounds:
        column = [s for s in page.strings if left <= s[1] + s[2] // 2 <= right]
        entries: list[dict] = []
        for text, words, top, bottom, _x0, _x1 in cluster_lines(column):
            if not text.strip():
                continue
            symbol_words = [w for w in words if w[1] + w[2] <= gloss_left + 6]
            gloss_words = [w for w in words if w[1] + w[2] > gloss_left + 6]
            # A line with nothing in the symbol band is either the *wrap* of the
            # meaning above it — the plate sets long glosses over three lines —
            # or a new entry whose symbol the OCR simply failed to emit, which is
            # common enough to matter: "zřícenina, / továrna." is two symbols, not
            # one. The plate closes every finished meaning with a full stop or a
            # comma and breaks a wrap mid-phrase, so the punctuation the previous
            # line ends on decides it. Merging the two cases lost a symbol each
            # time and corrupted the gloss of the one it merged into.
            if not symbol_words and entries:
                # A WRAP IS ADJACENT BY DEFINITION. Without a distance limit this
                # merge will join any two lines separated by an unclosed gloss,
                # however far apart - and on this plate it did: the running head
                # "XI" opens an entry that closes on no punctuation, and the next
                # line carrying gloss words is `stanice pro parolodi.` 870 px
                # below, whose own symbol the OCR emitted at x=2014, six pixels
                # outside the symbol band. The two merged into one entry spanning
                # the whole column, so [steamer] was cropped from 92x920 px of
                # mostly blank paper, failed the ink test, and was the single
                # pictogram of 83 with no atlas image. A reader shown the atlas
                # therefore could not name it, on any page, at any accuracy.
                gap = top - entries[-1]["bottom"]
                if (gap <= MAX_WRAP_GAP
                        and not _ENTRY_END_RE.search(entries[-1]["gloss"])):
                    entries[-1]["gloss"] += " " + " ".join(w[4]
                                                           for w in gloss_words)
                    entries[-1]["bottom"] = bottom
                    continue
                entries.append({"gloss": " ".join(w[4] for w in gloss_words),
                                "top": top, "bottom": bottom,
                                "symbol_read": False})
                continue
            entries.append({
                "gloss": " ".join(w[4] for w in gloss_words),
                "top": top, "bottom": bottom, "symbol_read": True})

        for entry in entries:
            token, expansion, score = match_gloss(entry["gloss"], targets)
            box = (max(0, left - 10), max(0, entry["top"] - 6),
                   min(image.width, gloss_left - 8),
                   min(image.height, entry["bottom"] + 6))
            record = {"token": token, "gloss_printed": entry["gloss"][:70],
                      "gloss_curated": expansion, "score": round(score, 1),
                      "scan": f"{scan:04d}",
                      "symbol_in_text_layer": entry.get("symbol_read", True),
                      "left": box[0], "top": box[1], "right": box[2],
                      "bottom": box[3], "file": ""}
            if (score < _MIN_SCORE
                    or box[2] - box[0] < _MIN_SYMBOL_PX
                    or box[3] - box[1] < _MIN_SYMBOL_PX):
                record["verdict"] = ("no curated token matches this printed "
                                     "meaning" if score < _MIN_SCORE else
                                     "no symbol region on this line")
                unmatched.append(record)
                continue
            # A token can legitimately appear once. A second hit means one of the
            # two lines was misread; both are kept, suffixed, and reported.
            crop = image.crop(box)
            ink = ink_share(crop)
            record["ink"] = round(ink, 4)
            if ink < _MIN_INK_SHARE:
                record["verdict"] = ("the crop holds no ink — the symbol is not "
                                     "where the layout says it is")
                unmatched.append(record)
                continue
            used[token] = used.get(token, 0) + 1
            name = token.strip("[]") + ("" if used[token] == 1
                                        else f"_{used[token]}")
            path = OUT_DIR / f"{name}.png"
            crop.save(path)
            record["file"] = path.name
            record["verdict"] = "ok" if used[token] == 1 else "duplicate token"
            manifest.append(record)

    manifest.sort(key=lambda r: r["token"])
    with (OUT_DIR / "manifest.csv").open("w", newline="", encoding="utf-8") as fh:
        fields = list(manifest[0]) if manifest else ["token"]
        writer = csv.DictWriter(fh, fieldnames=fields + ["verdict"],
                                restval="", extrasaction="ignore")
        writer.writeheader()
        writer.writerows(manifest + unmatched)

    _montage(manifest, Image, ImageDraw, ImageFont)

    print(f"\n  {len(manifest)} symbols cropped and named  ->  {OUT_DIR}")
    dup = [r for r in manifest if r["verdict"] == "duplicate token"]
    if dup:
        print(f"  ! {len(dup)} tokens matched more than one plate line — one of "
              f"each pair is a misread gloss:")
        for r in dup[:6]:
            print(f"      {r['token']:20s} {r['gloss_printed'][:44]!r}")

    named = {r["token"] for r in manifest}
    missing = [t for t, _e, _en, _c in targets if t not in named]
    print(f"\n  curated tokens WITHOUT a symbol from the plate: {len(missing)}")
    for token in missing[:14]:
        print(f"      {token}")
    if len(missing) > 14:
        print(f"      ... {len(missing) - 14} more")
    print("      (inferred rather than read off the legend — treat with care)")

    blank = [r for r in unmatched if r["verdict"].startswith("the crop holds")]
    if blank:
        print(f"\n  crops rejected as blank paper: {len(blank)}")
        for r in blank:
            print(f"      {r['token']:22s} {r['gloss_printed'][:40]!r}  "
                  f"ink {r.get('ink')}")
        print("      (these tokens have NO verified symbol — the manual pass "
              "should start here)")

    real = [r for r in unmatched if r["verdict"].startswith("no curated")]
    print(f"\n  plate lines the curated table CANNOT name: {len(real)}")
    for r in real[:14]:
        print(f"      {r['gloss_printed'][:60]!r}")
    if len(real) > 14:
        print(f"      ... {len(real) - 14} more")
    print("      (the closed list is short of the printed legend — a symbol here")
    print("       comes back [?] however good the reader is)")

    if show:
        print("\n  manifest:")
        for r in manifest:
            print(f"    {r['token']:22s} {r['score']:5.1f}  "
                  f"{r['gloss_printed'][:44]!r}")
    return 0

def ink_share(crop) -> float:
    """Fraction of the crop darker than its own paper."""
    grey = crop.convert("L")
    pixels = list(grey.getdata())
    if not pixels:
        return 0.0
    paper = sorted(pixels)[int(len(pixels) * 0.9)]      # the 90th percentile is paper
    cut = paper * 0.72
    return sum(1 for p in pixels if p < cut) / len(pixels)

"""Cut the volume's own symbol legend into a labelled atlas of pictogram images.

The register explains its pictograms on one plate — *Výklad značek*, p. XI, scan
0013 — printed three columns wide, one symbol to a line with its Czech meaning
beside it::

    ®  místní obec.              S  obecná škola.       À  četník.
    C  osada.                    ?  lékař.              1' hasičský zbor.
    Ш  katastrální (berní) obec. +  pohodný.            ;šľ spořitelna.

Until now the vision channel was handed an **English word list** — ``[school]
primary school`` — and asked to map a 1916 printed glyph onto it. It had never
seen this typeface's symbols, so it matched on resemblance, which is why two
readers given the identical mark returned ``[doctor]`` and ``[rc-curate]``. This
script replaces the word list with the printed forms themselves.

**The plate is self-labelling, which is what makes this mostly mechanical.** The
OCR destroys every symbol on it (``®``, ``Ш``, ``<3*``, ``€``, ``tí``) and reads
every *gloss* cleanly — and those glosses are already the ``expansion`` column of
``institution_abbreviations.csv``. So the gloss says which token the symbol is, and
the plate's own geometry says where the symbol is: each column is a narrow band of
pictograms, a blank gutter, then the meanings, so the widest block of ink in a
column *is* the gloss and everything left of it is the picture. Taking the crop
boundary from the layout rather than from the symbol's own characters is what makes
it work on the lines where the OCR emitted nothing for the symbol at all.

What comes out
--------------
``symbols/<token>.png``   one crop per pictogram, at scan resolution
``symbols/manifest.csv``  token, printed gloss, match score, source box
``symbols/atlas.png``     the crops with their tokens beside them, as one image —
                          this is what a vision model can be shown

And a report of the three ways the plate and the curated table disagree, which is
the other thing this buys. A plate line that matches no token means the **token
list is incomplete** and any such symbol on a page comes back ``[?]`` however good
the reader is; the plate prints ``turistická bouda``, ``salaš, koliha`` and
``klášterec``, none of which the table currently carries. A token with no plate
line means it was inferred rather than read off the legend.

Usage
-----
    python utils/lexikon_1910/build_symbol_atlas.py
    python utils/lexikon_1910/build_symbol_atlas.py --scan 13 --show
"""

from __future__ import annotations

import argparse
import csv
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from utils.lexikon_1910 import lexikon_1910 as lx                   # noqa: E402
from utils.lexikon_1910 import lexikon_1910_institutions as lxin     # noqa: E402
from utils.lexikon_1910.lexikon_1910_notes import normalise          # noqa: E402

LIBRARY = ROOT / "library" / "lexicons"
CZ_PAGES = LIBRARY / "NDK_Podrobny-seznam-mist-Cechy-1910_cz_pages"
CZ_PDF = LIBRARY / "NDK_Podrobny-seznam-mist-Cechy-1910_cz.pdf"
ABBREVIATIONS = ROOT / "reference" / "places" / "institution_abbreviations.csv"
OUT_DIR = ROOT / "reference" / "places" / "symbols"

#: The plate. Page XI of the front matter; the abbreviations that are *words* are
#: on page X (scan 12) and need no atlas, because the text layer reads them.
LEGEND_SCAN = 13

#: How far below an entry a wrapped gloss line may start, in pixels. The plate
#: sets ~47 px boxes about 85 px apart, so a genuine wrap is always within one
#: line; this is generous enough to survive a looser column and still an order of
#: magnitude below the 870 px jump that produced the [steamer] defect.
MAX_WRAP_GAP = 150

#: A blank vertical band this wide separates two printed columns. Measured on the
#: plate rather than guessed: the two column separators are 62 px and 81 px, and
#: the gap between a symbol and its own gloss is 16-36 px, so anything from about
#: 45 to 60 divides them cleanly. A line must never be assembled across a column —
#: that was the first version's bug, and it produced three usable symbols out of
#: sixty-five because every line came out as two half-sentences glued together.
_COLUMN_GAP = 50

#: The narrowest a gloss can be. Used to tell the gloss block from the symbol band
#: inside a column: the glosses run ~725 px, the symbols ~65-95 px.
_MIN_GLOSS_WIDTH = 300

#: Below this the gloss did not match any curated expansion and the line is
#: reported rather than named. rapidfuzz partial ratio over normalised text.
_MIN_SCORE = 78

#: A symbol crop narrower or shorter than this is OCR noise, not a pictogram.
_MIN_SYMBOL_PX = 8

#: The least ink a crop must hold to be a pictogram. A share of pixels darker than
#: the paper, measured against the crop's own background so the scan's tone does
#: not decide it.
#:
#: This is the check that stops the atlas asserting something it does not have. The
#: geometry is sound in general and wrong in particular places — a symbol set
#: unusually far left, a line the clusterer bounded a few pixels short — and the
#: result is a crop of blank paper carrying a confident label. Counting the counts
#: cannot see that; only looking at the pixels can. It caught [monastery] (empty),
#: [steamer] (a sliver of the rule beside it) and [revenue-admin].
_MIN_INK_SHARE = 0.012

#: A finished meaning on the plate ends here. Used to tell a wrapped gloss from a
#: new entry whose symbol the OCR dropped — see the note in :func:`build`.
_ENTRY_END_RE = re.compile(r"[.,;]\s*$")


def cluster_lines(words, gap: int = 18):
    """``[(text, words, top, bottom, left, right)]`` for one column."""
    rows: list[list[tuple]] = []
    for word in sorted(words, key=lambda s: (s[0] + s[3] // 2, s[1])):
        centre = word[0] + word[3] // 2
        if rows and centre - (rows[-1][-1][0] + rows[-1][-1][3] // 2) <= gap:
            rows[-1].append(word)
        else:
            rows.append([word])
    out = []
    for row in rows:
        row = sorted(row, key=lambda w: w[1])
        out.append((" ".join(w[4] for w in row).strip(), row,
                    min(w[0] for w in row), max(w[0] + w[3] for w in row),
                    min(w[1] for w in row), max(w[1] + w[2] for w in row)))
    return out


def _covered(words, gap: int) -> list[list[int]]:
    """The x ranges the print actually occupies, merged across gaps below ``gap``."""
    merged: list[list[int]] = []
    for a, b in sorted((w[1], w[1] + w[2]) for w in words):
        if merged and a <= merged[-1][1] + gap:
            merged[-1][1] = max(merged[-1][1], b)
        else:
            merged.append([a, b])
    return merged


def columns_of(page, words) -> list[tuple[int, int, int]]:
    """``(column_left, gloss_left, column_right)`` for each printed column.

    Taken from where the ink is, not from where words start. The plate sets each
    column as a narrow band of pictograms, a blank gutter, then the meanings — so
    the widest block of covered x inside a column *is* the gloss, and everything to
    its left is the symbol. That makes the crop boundary a measurement rather than
    an inference, and it works on the lines where the OCR emitted nothing for the
    symbol at all, which a rule keyed on the symbol's own characters cannot.
    """
    out: list[tuple[int, int, int]] = []
    for left, right in _covered(words, _COLUMN_GAP):
        inner = [(a, b) for a, b in _covered(
            [w for w in words if left <= w[1] <= right], 4)]
        wide = [iv for iv in inner if iv[1] - iv[0] >= _MIN_GLOSS_WIDTH]
        if not wide:
            continue
        out.append((left, wide[0][0], right))
    return out


def load_targets():
    """``[(token, expansion, english, category)]`` for every curated pictogram."""
    rows = lxin.load_abbreviations(ABBREVIATIONS, lang="cs")
    return [(a.code, a.expansion, a.english, a.category)
            for a in rows if a.kind == "pictogram" and a.expansion]


def match_gloss(gloss: str, targets):
    """Name the token whose curated expansion this printed meaning is.

    The plate's glosses *are* the ``expansion`` column — ``obecná škola``,
    ``lékař``, ``hasičský zbor`` — so this is a lookup with OCR damage allowed,
    not an interpretation. Scored on the normalised strings so accents and
    punctuation the scanner mangles cannot decide it.

    ``token_sort_ratio`` rather than a plain ratio because the plate and the table
    order a two-part meaning differently often enough to matter (``mlýn, pila,
    brusírna``), and a whole-string comparison would rank a short unrelated token
    above the right long one.
    """
    from rapidfuzz import fuzz

    key = normalise(gloss)
    if len(key) < 3:
        return "", "", 0.0
    best = ("", "", 0.0)
    for token, expansion, _en, _cat in targets:
        target = normalise(expansion)
        score = max(fuzz.ratio(target, key), fuzz.token_sort_ratio(target, key))
        # The plate elaborates where the table is terse — it prints "skupina
        # chalup (více rozptýlených domů vzdálených od sebe na vrh kamenem…)"
        # for what the table calls "skupina chalup". A whole-string comparison
        # scores that near zero, so a containment measure is allowed too — but
        # only for a target long enough that containment means something, or
        # every three-letter expansion would match every gloss on the plate.
        if len(target) >= 6:
            # Penalised, so containment never beats a good whole-string match.
            # Without this "městys" was captured by [mestys-part], whose curated
            # expansion "část městyse" *contains* it and scored an unpenalised
            # 100 — the longer code winning the shorter code's own gloss.
            score = max(score, fuzz.partial_ratio(target, key) - 6)
        if score > best[2]:
            best = (token, expansion, score)
    return best


def build(scan: int, show: bool = False) -> int:
    from PIL import Image, ImageDraw, ImageFont

    page = lx.read_alto(CZ_PAGES / f"{scan:04d}.alto.xml")
    image = lx.render_page(CZ_PDF, scan, page.width)
    targets = load_targets()
    print(f"  plate: scan {scan:04d}, {page.width}x{page.height}, "
          f"{len(page.strings)} words; {len(targets)} curated pictograms")

    bounds = columns_of(page, page.strings)
    print(f"  {len(bounds)} printed columns (symbol band | gloss): "
          + ", ".join(f"{a}-{g} | {g}-{b}" for a, g, b in bounds))

    OUT_DIR.mkdir(parents=True, exist_ok=True)
    manifest: list[dict] = []
    unmatched: list[dict] = []
    used: dict[str, int] = {}

    for left, gloss_left, right in bounds:
        column = [s for s in page.strings if left <= s[1] + s[2] // 2 <= right]
        entries: list[dict] = []
        for text, words, top, bottom, _x0, _x1 in cluster_lines(column):
            if not text.strip():
                continue
            symbol_words = [w for w in words if w[1] + w[2] <= gloss_left + 6]
            gloss_words = [w for w in words if w[1] + w[2] > gloss_left + 6]
            # A line with nothing in the symbol band is either the *wrap* of the
            # meaning above it — the plate sets long glosses over three lines —
            # or a new entry whose symbol the OCR simply failed to emit, which is
            # common enough to matter: "zřícenina, / továrna." is two symbols, not
            # one. The plate closes every finished meaning with a full stop or a
            # comma and breaks a wrap mid-phrase, so the punctuation the previous
            # line ends on decides it. Merging the two cases lost a symbol each
            # time and corrupted the gloss of the one it merged into.
            if not symbol_words and entries:
                # A WRAP IS ADJACENT BY DEFINITION. Without a distance limit this
                # merge will join any two lines separated by an unclosed gloss,
                # however far apart - and on this plate it did: the running head
                # "XI" opens an entry that closes on no punctuation, and the next
                # line carrying gloss words is `stanice pro parolodi.` 870 px
                # below, whose own symbol the OCR emitted at x=2014, six pixels
                # outside the symbol band. The two merged into one entry spanning
                # the whole column, so [steamer] was cropped from 92x920 px of
                # mostly blank paper, failed the ink test, and was the single
                # pictogram of 83 with no atlas image. A reader shown the atlas
                # therefore could not name it, on any page, at any accuracy.
                gap = top - entries[-1]["bottom"]
                if (gap <= MAX_WRAP_GAP
                        and not _ENTRY_END_RE.search(entries[-1]["gloss"])):
                    entries[-1]["gloss"] += " " + " ".join(w[4]
                                                           for w in gloss_words)
                    entries[-1]["bottom"] = bottom
                    continue
                entries.append({"gloss": " ".join(w[4] for w in gloss_words),
                                "top": top, "bottom": bottom,
                                "symbol_read": False})
                continue
            entries.append({
                "gloss": " ".join(w[4] for w in gloss_words),
                "top": top, "bottom": bottom, "symbol_read": True})

        for entry in entries:
            token, expansion, score = match_gloss(entry["gloss"], targets)
            box = (max(0, left - 10), max(0, entry["top"] - 6),
                   min(image.width, gloss_left - 8),
                   min(image.height, entry["bottom"] + 6))
            record = {"token": token, "gloss_printed": entry["gloss"][:70],
                      "gloss_curated": expansion, "score": round(score, 1),
                      "scan": f"{scan:04d}",
                      "symbol_in_text_layer": entry.get("symbol_read", True),
                      "left": box[0], "top": box[1], "right": box[2],
                      "bottom": box[3], "file": ""}
            if (score < _MIN_SCORE
                    or box[2] - box[0] < _MIN_SYMBOL_PX
                    or box[3] - box[1] < _MIN_SYMBOL_PX):
                record["verdict"] = ("no curated token matches this printed "
                                     "meaning" if score < _MIN_SCORE else
                                     "no symbol region on this line")
                unmatched.append(record)
                continue
            # A token can legitimately appear once. A second hit means one of the
            # two lines was misread; both are kept, suffixed, and reported.
            crop = image.crop(box)
            ink = ink_share(crop)
            record["ink"] = round(ink, 4)
            if ink < _MIN_INK_SHARE:
                record["verdict"] = ("the crop holds no ink — the symbol is not "
                                     "where the layout says it is")
                unmatched.append(record)
                continue
            used[token] = used.get(token, 0) + 1
            name = token.strip("[]") + ("" if used[token] == 1
                                        else f"_{used[token]}")
            path = OUT_DIR / f"{name}.png"
            crop.save(path)
            record["file"] = path.name
            record["verdict"] = "ok" if used[token] == 1 else "duplicate token"
            manifest.append(record)

    manifest.sort(key=lambda r: r["token"])
    with (OUT_DIR / "manifest.csv").open("w", newline="", encoding="utf-8") as fh:
        fields = list(manifest[0]) if manifest else ["token"]
        writer = csv.DictWriter(fh, fieldnames=fields + ["verdict"],
                                restval="", extrasaction="ignore")
        writer.writeheader()
        writer.writerows(manifest + unmatched)

    _montage(manifest, Image, ImageDraw, ImageFont)

    print(f"\n  {len(manifest)} symbols cropped and named  ->  {OUT_DIR}")
    dup = [r for r in manifest if r["verdict"] == "duplicate token"]
    if dup:
        print(f"  ! {len(dup)} tokens matched more than one plate line — one of "
              f"each pair is a misread gloss:")
        for r in dup[:6]:
            print(f"      {r['token']:20s} {r['gloss_printed'][:44]!r}")

    named = {r["token"] for r in manifest}
    missing = [t for t, _e, _en, _c in targets if t not in named]
    print(f"\n  curated tokens WITHOUT a symbol from the plate: {len(missing)}")
    for token in missing[:14]:
        print(f"      {token}")
    if len(missing) > 14:
        print(f"      ... {len(missing) - 14} more")
    print("      (inferred rather than read off the legend — treat with care)")

    blank = [r for r in unmatched if r["verdict"].startswith("the crop holds")]
    if blank:
        print(f"\n  crops rejected as blank paper: {len(blank)}")
        for r in blank:
            print(f"      {r['token']:22s} {r['gloss_printed'][:40]!r}  "
                  f"ink {r.get('ink')}")
        print("      (these tokens have NO verified symbol — the manual pass "
              "should start here)")

    real = [r for r in unmatched if r["verdict"].startswith("no curated")]
    print(f"\n  plate lines the curated table CANNOT name: {len(real)}")
    for r in real[:14]:
        print(f"      {r['gloss_printed'][:60]!r}")
    if len(real) > 14:
        print(f"      ... {len(real) - 14} more")
    print("      (the closed list is short of the printed legend — a symbol here")
    print("       comes back [?] however good the reader is)")

    if show:
        print("\n  manifest:")
        for r in manifest:
            print(f"    {r['token']:22s} {r['score']:5.1f}  "
                  f"{r['gloss_printed'][:44]!r}")
    return 0


def ink_share(crop) -> float:
    """Fraction of the crop darker than its own paper."""
    grey = crop.convert("L")
    pixels = list(grey.getdata())
    if not pixels:
        return 0.0
    paper = sorted(pixels)[int(len(pixels) * 0.9)]      # the 90th percentile is paper
    cut = paper * 0.72
    return sum(1 for p in pixels if p < cut) / len(pixels)


def _montage(manifest, Image, ImageDraw, ImageFont, out_dir: Path | None = None,
             title: str = "Printed symbols of the 1910 register (Vyklad znacek, "
                          "p. XI) and their tokens") -> None:
    """The crops with their tokens beside them, as one image for a prompt.

    ``out_dir`` and ``title`` let the German plate (``build_symbol_atlas_de.py``)
    build its own atlas with this same layout; the defaults are the Czech one's.
    """
    out_dir = out_dir or OUT_DIR
    if not manifest:
        return
    cell_h, cell_w, cols, pad = 96, 460, 3, 12
    rows = (len(manifest) + cols - 1) // cols
    sheet = Image.new("RGB", (cols * cell_w, rows * cell_h + 46), "white")
    draw = ImageDraw.Draw(sheet)
    try:
        font = ImageFont.truetype("arial.ttf", 26)
        head = ImageFont.truetype("arialbd.ttf", 30)
    except Exception:                                   # noqa: BLE001
        font = head = None
    draw.text((pad, 8), title, fill="black", font=head)
    for i, record in enumerate(manifest):
        x = (i % cols) * cell_w
        y = (i // cols) * cell_h + 46
        try:
            crop = Image.open(out_dir / record["file"])
        except Exception:                               # noqa: BLE001
            continue
        scale = min((cell_h - 2 * pad) / max(crop.height, 1), 3.0)
        crop = crop.resize((max(1, int(crop.width * scale)),
                            max(1, int(crop.height * scale))))
        sheet.paste(crop, (x + pad, y + pad))
        draw.text((x + pad + crop.width + 16, y + cell_h // 3),
                  record["token"], fill="black", font=font)
        draw.line([(x, y), (x + cell_w, y)], fill="#dddddd")
    sheet.save(out_dir / "atlas.png")
    print(f"  atlas sheet: {out_dir / 'atlas.png'} "
          f"({sheet.width}x{sheet.height})")

    # A second, denser sheet for the prompt. The readable one above is for a
    # human; a model pays for every pixel, and at ~2,700 px tall the first sheet
    # costs more image tokens than the page crop it is meant to help with.
    cols2, cell_w2, cell_h2 = 5, 280, 62
    rows2 = (len(manifest) + cols2 - 1) // cols2
    tight = Image.new("RGB", (cols2 * cell_w2, rows2 * cell_h2 + 34), "white")
    draw2 = ImageDraw.Draw(tight)
    draw2.text((8, 6), "KEY: each printed symbol of this volume and its token",
               fill="black", font=head)
    for i, record in enumerate(manifest):
        x, y = (i % cols2) * cell_w2, (i // cols2) * cell_h2 + 34
        try:
            crop = Image.open(out_dir / record["file"])
        except Exception:                               # noqa: BLE001
            continue
        scale = min((cell_h2 - 10) / max(crop.height, 1), 1.6)
        crop = crop.resize((max(1, int(crop.width * scale)),
                            max(1, int(crop.height * scale))))
        tight.paste(crop, (x + 5, y + 5))
        draw2.text((x + 5 + crop.width + 8, y + cell_h2 // 3), record["token"],
                   fill="black", font=font)
    tight.save(out_dir / "atlas_prompt.png")
    print(f"  prompt sheet: {out_dir / 'atlas_prompt.png'} "
          f"({tight.width}x{tight.height}, ~{tight.width * tight.height // 750} "
          f"image tokens)")


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--scan", type=int, default=LEGEND_SCAN)
    ap.add_argument("--show", action="store_true", help="print the full manifest")
    args = ap.parse_args()
    return build(args.scan, args.show)


if __name__ == "__main__":
    raise SystemExit(main())
