How it works

  1. de_gloss / en_gloss are written into the curation from CZ_GLOSS.
  2. The German gloss is looked up exactly (case- and diacritic-insensitive) in the lang_code == 'GE' rows of hisco_occupation_link_combined.csv -> link_hisco. A gloss that resolves to more than one HISCO code is not auto-assigned; it is flagged review with the candidates in note.
  3. Where a gloss misses, the English gloss is matched against the authority's own label_en — exact only. No fuzzy matching anywhere: HISCO labels are formal ("Metal Moulder, General"), colloquial glosses are not, and a near-miss between two formal labels is far more likely to be a different occupation than a spelling variant. Anything that misses is flagged for hand assignment instead.
  4. HISCO_OVERRIDE is applied after the bridge and always wins, so a documented judgement call cannot be overwritten by it.
  5. ADJUDICATED is applied last, and it is the only writer of review. A key answered keep or recoded in the 2026-08-26 workbook has its flag cleared and its verdict written to the verdict column; a key answered open keeps the flag and carries the reason it survived. Anything never adjudicated keeps whatever the bridge decided, so a new ambiguous key still arrives flagged.

Hand-entered values already in the curation are preserved: the script only fills blanks unless --overwrite is passed.

def load_german_index(known: set[str] | None = None) -> dict[str, list[dict]]:
    """Folded German label -> the link rows carrying it (HISCO ids zero-padded, see :func:`_pad_hisco`)."""
    known = load_authority_codes() if known is None else known
    df = pd.read_csv(LINK, sep=";", dtype=str, keep_default_na=False)
    de = df[df["lang_code"].str.upper().isin(GERMAN_LANGS)]
    idx: dict[str, list[dict]] = {}
    n_pad = 0
    for r in de.itertuples():
        code = _pad_hisco(r.hisco_id, known)
        n_pad += code != r.hisco_id.strip()
        idx.setdefault(_fold(r.label), []).append(
            {"hisco": code, "status": r.status_id.strip(),
             "relation": r.relation_id.strip(), "product": r.product_id.strip()})
    print(f"  German link labels: {len(idx):,} distinct over {len(de):,} rows "
          f"({n_pad:,} HISCO ids zero-padded to 5 digits)")
    return idx

def bridge(overwrite: bool = False) -> pd.DataFrame:
    cur = pd.read_csv(CURATION, dtype=str).fillna("")
    de_idx = load_german_index()
    en_idx = load_label_index()

    if "verdict" not in cur.columns:
        cur["verdict"] = ""

    stats = {"gloss": 0, "de_exact": 0, "de_ambiguous": 0, "en_exact": 0,
             "override": 0, "kept": 0, "none": 0}

    for i, row in cur.iterrows():
        key = row["key"]
        has_code = row["hisco"].strip() != ""
        if has_code and not overwrite and key not in HISCO_OVERRIDE:
            stats["kept"] += 1
            continue

        gloss = CZ_GLOSS.get(key)
        if gloss:
            de, en = gloss
            cur.at[i, "de_gloss"], cur.at[i, "en_gloss"] = de, en
            stats["gloss"] += 1

            hits = de_idx.get(_fold(de), [])
            codes = sorted({h["hisco"] for h in hits})
            if len(codes) == 1:
                h = hits[0]
                cur.at[i, "link_hisco"] = codes[0]
                cur.at[i, "hisco"] = codes[0]
                cur.at[i, "status"] = _norm_modifier(h["status"])
                cur.at[i, "relation"] = _norm_modifier(h["relation"])
                cur.at[i, "product"] = _norm_modifier(h["product"])
                cur.at[i, "source"] = "link"
                stats["de_exact"] += 1
            elif len(codes) > 1:
                cur.at[i, "link_hisco"] = ";".join(codes)
                cur.at[i, "review"] = "1"
                cur.at[i, "note"] = f"German gloss {de!r} maps to {len(codes)} codes"
                stats["de_ambiguous"] += 1
            else:
                en_codes = en_idx.get(_fold(en), [])
                if len(en_codes) == 1:
                    cur.at[i, "hisco"] = en_codes[0]
                    cur.at[i, "link_hisco"] = en_codes[0]
                    cur.at[i, "source"] = "label_en"
                    stats["en_exact"] += 1
                else:
                    cur.at[i, "review"] = "1"
                    cur.at[i, "note"] = (
                        f"no German link for {de!r}; "
                        f"English label matched {len(en_codes)} codes")
                    stats["none"] += 1

    # Overrides win outright.
    for key, spec in HISCO_OVERRIDE.items():
        m = cur["key"] == key
        if not m.any():
            continue
        for col, val in spec.items():
            cur.loc[m, col] = val
        # `source` is part of the spec for the compounds curated whole under D-29, which are
        # a different kind of judgement from a plain hand override and stay labelled as one.
        cur.loc[m, "source"] = spec.get("source", "manual")
        for col in ("status", "relation", "product"):
            blank = m & (cur[col].str.strip() == "")
            cur.loc[blank, col] = "-9"
        stats["override"] += 1

    apply_adjudication(cur, stats)

    print("\n  " + "  ".join(f"{k}={v:,}" for k, v in stats.items()))
    missing_keys = sorted(set(HISCO_OVERRIDE) - set(cur["key"]))
    if missing_keys:
        print(f"  note: {len(missing_keys)} override keys are not in the skeleton: "
              f"{missing_keys[:10]}")
    unused = sorted(set(CZ_GLOSS) - set(cur["key"]))
    if unused:
        print(f"  note: {len(unused)} glossed keys are not in the skeleton: {unused[:10]}")
    return cur

def apply_adjudication(cur: pd.DataFrame, stats: dict) -> None

Write the human verdicts, and let them — and only them — clear a review flag.

Ordering matters and is deliberate. The bridge raises review wherever a German gloss reached several codes, and it must keep doing so: that is how a key added next month arrives flagged. What it must not do is re-raise a flag a human has already answered, so this runs last and overwrites.

Two guards, both for the same failure — a verdict recorded while the code it implies never lands, which would leave the curation looking settled at the old code:

  • a key marked recoded must have an entry in HISCO_OVERRIDE, since that is where the new code lives;
  • HISCO_OVERRIDE may not set review for an adjudicated key, or the two would be arguing about the flag and the winner would be whichever ran second.

def apply_adjudication(cur: pd.DataFrame, stats: dict) -> None:
    """Write the human verdicts, and let them — and only them — clear a ``review`` flag.

    Ordering matters and is deliberate. The bridge raises ``review`` wherever a German gloss
    reached several codes, and it must keep doing so: that is how a key added next month
    arrives flagged. What it must not do is re-raise a flag a human has already answered, so
    this runs last and overwrites.

    Two guards, both for the same failure — a verdict recorded while the code it implies
    never lands, which would leave the curation *looking* settled at the old code:

    * a key marked ``recoded`` must have an entry in ``HISCO_OVERRIDE``, since that is where
      the new code lives;
    * ``HISCO_OVERRIDE`` may not set ``review`` for an adjudicated key, or the two would be
      arguing about the flag and the winner would be whichever ran second.
    """
    missing = sorted(k for k, v in ADJUDICATED.items()
                     if v == "recoded" and k not in HISCO_OVERRIDE)
    if missing:
        raise SystemExit(f"ADJUDICATED marks these 'recoded' but HISCO_OVERRIDE has no entry, "
                         f"so the old code would stand: {missing}")
    contested = sorted(k for k in ADJUDICATED if "review" in HISCO_OVERRIDE.get(k, {}))
    if contested:
        raise SystemExit(f"HISCO_OVERRIDE sets `review` for adjudicated keys; ADJUDICATED owns "
                         f"the flag and these would fight over it: {contested}")

    absent, counts = [], {"keep": 0, "recoded": 0, "open": 0}
    for key, verdict in ADJUDICATED.items():
        m = cur["key"] == key
        if not m.any():
            absent.append(key)
            continue
        kind = verdict.split(":", 1)[0]
        counts[kind] = counts.get(kind, 0) + 1
        cur.loc[m, "verdict"] = f"{ADJUDICATED_ON} {verdict}"
        cur.loc[m, "review"] = "1" if kind == "open" else ""
    stats["adjudicated"] = len(ADJUDICATED) - len(absent)
    print()
    print(f"  adjudicated {ADJUDICATED_ON}: "
          + "  ".join(f"{k}={v}" for k, v in counts.items()))
    if absent:
        print(f"  note: {len(absent)} adjudicated keys are not in the skeleton: {absent[:10]}")

def bridge_report(cur: pd.DataFrame) -> None:
    coded = cur["hisco"].str.strip() != ""
    n_cov = int(pd.to_numeric(cur.loc[coded, "n"]).sum())
    n_all = int(pd.to_numeric(cur["n"]).sum())
    print(f"\nCuration after the bridge")
    print(f"  keys with a HISCO code  {int(coded.sum()):,} / {len(cur):,}")
    print(f"  records those cover     {n_cov:,} of {n_all:,} in the skeleton "
          f"({n_cov / n_all:.1%})")
    flagged = cur["review"].str.strip() != ""
    answered = cur["verdict"].str.strip() != ""
    print(f"  adjudicated             {int(answered.sum()):,} "
          f"({int(pd.to_numeric(cur.loc[answered, 'n']).sum()):,} records)")
    print(f"  still flagged for review {int(flagged.sum()):,} "
          f"({int(pd.to_numeric(cur.loc[flagged, 'n']).sum()):,} records)")
    print("\n  by source:")
    for k, v in cur.loc[coded, "source"].value_counts().items():
        print(f"    {k:<10} {v:>5,}")

    top_uncoded = cur[~coded].head(15)
    if len(top_uncoded):
        print("\n  largest keys still uncoded:")
        for r in top_uncoded.itertuples():
            print(f"    {int(float(r.n)):>6,}  {r.key}")

def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[1])
    ap.add_argument("--overwrite", action="store_true",
                    help="re-derive codes even where the curation already has one")
    args = ap.parse_args(argv)

    # A key written twice in czech_glosses.py is invisible at runtime — the dict is already
    # built by then — so check the source before trusting either dict.
    dupes = duplicate_keys()
    if dupes:
        raise SystemExit(
            "czech_glosses.py defines the same key more than once; Python keeps the last "
            "silently:\n" + "\n".join(f"  {k!r} at lines {v}" for k, v in dupes.items()))

    print("Bridging Czech glosses -> HISCO …")
    cur = bridge(overwrite=args.overwrite)
    bridge_report(cur)
    cur.to_csv(CURATION, index=False, encoding="utf-8")
    print(f"\nUpdated {CURATION.relative_to(ROOT)}")
    return 0

"""
run_link_bridge.py — resolve the Czech glosses to HISCO codes via the German link table
=======================================================================================
Applies ``czech_glosses.py`` to the curation skeleton and fills in the codes:

  1. ``de_gloss`` / ``en_gloss`` are written into the curation from ``CZ_GLOSS``.
  2. The German gloss is looked up **exactly** (case- and diacritic-insensitive) in the
     ``lang_code == 'GE'`` rows of ``hisco_occupation_link_combined.csv`` -> ``link_hisco``.
     A gloss that resolves to more than one HISCO code is *not* auto-assigned; it is
     flagged ``review`` with the candidates in ``note``.
  3. Where a gloss misses, the English gloss is matched against the authority's own
     ``label_en`` — **exact only**. No fuzzy matching anywhere: HISCO labels are formal
     ("Metal Moulder, General"), colloquial glosses are not, and a near-miss between two
     formal labels is far more likely to be a different occupation than a spelling
     variant. Anything that misses is flagged for hand assignment instead.
  4. ``HISCO_OVERRIDE`` is applied after the bridge and always wins, so a documented
     judgement call cannot be overwritten by it.
  5. ``ADJUDICATED`` is applied last, and it is the **only** writer of ``review``. A key
     answered ``keep`` or ``recoded`` in the 2026-08-26 workbook has its flag cleared and
     its verdict written to the ``verdict`` column; a key answered ``open`` keeps the flag
     and carries the reason it survived. Anything never adjudicated keeps whatever the
     bridge decided, so a new ambiguous key still arrives flagged.

Hand-entered values already in the curation are preserved: the script only fills blanks
unless ``--overwrite`` is passed.

Run:  python utils/reference_tools/occupations/run_link_bridge.py [--overwrite]
Emits: updates reference/occupations/occupation_hisco_curation.csv in place
Independent check: utils/reference_tools/occupations/run_occcanine_check.py

from __future__ import annotations

import argparse
import sys
import unicodedata
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

import pandas as pd
from utils.reference_tools.occupations.czech_glosses import (ADJUDICATED, ADJUDICATED_ON, CZ_GLOSS,
                                                 HISCO_OVERRIDE, duplicate_keys)

__version__ = "1.2.0"   # + the adjudication register

LINK = ROOT / "data" / "raw" / "hisco" / "hisco_occupation_link_combined.csv"
AUTHORITY = ROOT / "reference" / "occupations" / "hisco_authority.csv"
CURATION = ROOT / "reference" / "occupations" / "occupation_hisco_curation.csv"

GERMAN_LANGS = {"GE", "DE"}


def _fold(s: str) -> str:
    s = "".join(c for c in unicodedata.normalize("NFD", str(s))
                if unicodedata.category(c) != "Mn")
    return " ".join(s.lower().split())


def load_authority_codes() -> set[str]:
    """Every HISCO code the authority can resolve to a label and the HISCLASS/HISCAM scales."""
    return set(pd.read_csv(AUTHORITY, dtype={"hisco": str})["hisco"].dropna())


def _pad_hisco(code: str, known: set[str]) -> str:
    """Restore a leading zero the link table dropped — but only when that is demonstrably right.

    HISCO is a **5-digit** scheme and the authority stores it that way, but
    ``hisco_occupation_link_combined.csv`` writes the id unpadded: an obstetrician is ``7310``
    where the authority has ``07310``. 1,827 of its rows are 4-digit and 1,167 are 2-digit.

    Blanket ``zfill(5)`` would be wrong — the same column also carries the sentinels ``-1``,
    ``-2`` and ``0`` (no code), and coarser hierarchy levels (``1200``, ``2100``) that are not
    occupations at all. So the padded form is accepted **only if the authority actually has it**;
    anything else is returned untouched and left for the caller's validity check to reject.
    Without this, ``Schiffer`` was coded ``4217`` and ``Ökonom`` ``9010`` — both real codes
    (``04217`` Ship's Master, ``09010`` Economist) that no downstream lookup could resolve,
    which is what the ``data_checks`` HISCO-authority invariant caught.
    """
    c = (code or "").strip()
    if c and c.isdigit() and len(c) < 5:
        padded = c.zfill(5)
        if padded in known:
            return padded
    return c


def load_german_index(known: set[str] | None = None) -> dict[str, list[dict]]:

             "relation": r.relation_id.strip(), "product": r.product_id.strip()})
    print(f"  German link labels: {len(idx):,} distinct over {len(de):,} rows "
          f"({n_pad:,} HISCO ids zero-padded to 5 digits)")
    return idx


def load_label_index() -> dict[str, list[str]]:

def _norm_modifier(v: str) -> str:
    """The link table writes 'no modifier' as 0; the authority (from HSN) uses -9."""
    v = (v or "").strip()
    return "-9" if v in {"", "0", "-9"} else v

        has_code = row["hisco"].strip() != ""
        if has_code and not overwrite and key not in HISCO_OVERRIDE:
            stats["kept"] += 1
            continue

        if gloss:

            cur.at[i, "de_gloss"], cur.at[i, "en_gloss"] = de, en
            stats["gloss"] += 1

            hits = de_idx.get(_fold(de), [])
            codes = sorted({h["hisco"] for h in hits})
            if len(codes) == 1:
                h = hits[0]
                cur.at[i, "link_hisco"] = codes[0]
                cur.at[i, "hisco"] = codes[0]
                cur.at[i, "status"] = _norm_modifier(h["status"])
                cur.at[i, "relation"] = _norm_modifier(h["relation"])
                cur.at[i, "product"] = _norm_modifier(h["product"])
                cur.at[i, "source"] = "link"
                stats["de_exact"] += 1
            elif len(codes) > 1:
                cur.at[i, "link_hisco"] = ";".join(codes)
                cur.at[i, "review"] = "1"
                cur.at[i, "note"] = f"German gloss {de!r} maps to {len(codes)} codes"
                stats["de_ambiguous"] += 1
            else:
                en_codes = en_idx.get(_fold(en), [])
                if len(en_codes) == 1:
                    cur.at[i, "hisco"] = en_codes[0]
                    cur.at[i, "link_hisco"] = en_codes[0]
                    cur.at[i, "source"] = "label_en"
                    stats["en_exact"] += 1
                else:
                    cur.at[i, "review"] = "1"
                    cur.at[i, "note"] = (
                        f"no German link for {de!r}; "
                        f"English label matched {len(en_codes)} codes")
                    stats["none"] += 1

    # Overrides win outright.
    for key, spec in HISCO_OVERRIDE.items():
        m = cur["key"] == key
        if not m.any():
            continue
        for col, val in spec.items():
            cur.loc[m, col] = val
        # `source` is part of the spec for the compounds curated whole under D-29, which are
        # a different kind of judgement from a plain hand override and stay labelled as one.
        cur.loc[m, "source"] = spec.get("source", "manual")
        for col in ("status", "relation", "product"):
            blank = m & (cur[col].str.strip() == "")
            cur.loc[blank, col] = "-9"
        stats["override"] += 1

    apply_adjudication(cur, stats)

    print("\n  " + "  ".join(f"{k}={v:,}" for k, v in stats.items()))
    missing_keys = sorted(set(HISCO_OVERRIDE) - set(cur["key"]))
    if missing_keys:
        print(f"  note: {len(missing_keys)} override keys are not in the skeleton: "
              f"{missing_keys[:10]}")
    unused = sorted(set(CZ_GLOSS) - set(cur["key"]))
    if unused:
        print(f"  note: {len(unused)} glossed keys are not in the skeleton: {unused[:10]}")
    return cur


def apply_adjudication(cur: pd.DataFrame, stats: dict) -> None:
    """Write the human verdicts, and let them — and only them — clear a ``review`` flag.

    Ordering matters and is deliberate. The bridge raises ``review`` wherever a German gloss
    reached several codes, and it must keep doing so: that is how a key added next month
    arrives flagged. What it must not do is re-raise a flag a human has already answered, so
    this runs last and overwrites.

    Two guards, both for the same failure — a verdict recorded while the code it implies
    never lands, which would leave the curation *looking* settled at the old code:

    * a key marked ``recoded`` must have an entry in ``HISCO_OVERRIDE``, since that is where
      the new code lives;
    * ``HISCO_OVERRIDE`` may not set ``review`` for an adjudicated key, or the two would be
      arguing about the flag and the winner would be whichever ran second.
    """
    missing = sorted(k for k, v in ADJUDICATED.items()
                     if v == "recoded" and k not in HISCO_OVERRIDE)
    if missing:
        raise SystemExit(f"ADJUDICATED marks these 'recoded' but HISCO_OVERRIDE has no entry, "
                         f"so the old code would stand: {missing}")
    contested = sorted(k for k in ADJUDICATED if "review" in HISCO_OVERRIDE.get(k, {}))
    if contested:
        raise SystemExit(f"HISCO_OVERRIDE sets `review` for adjudicated keys; ADJUDICATED owns "
                         f"the flag and these would fight over it: {contested}")

    absent, counts = [], {"keep": 0, "recoded": 0, "open": 0}
    for key, verdict in ADJUDICATED.items():
        m = cur["key"] == key
        if not m.any():
            absent.append(key)
            continue
        kind = verdict.split(":", 1)[0]
        counts[kind] = counts.get(kind, 0) + 1
        cur.loc[m, "verdict"] = f"{ADJUDICATED_ON} {verdict}"
        cur.loc[m, "review"] = "1" if kind == "open" else ""
    stats["adjudicated"] = len(ADJUDICATED) - len(absent)
    print()
    print(f"  adjudicated {ADJUDICATED_ON}: "
          + "  ".join(f"{k}={v}" for k, v in counts.items()))
    if absent:
        print(f"  note: {len(absent)} adjudicated keys are not in the skeleton: {absent[:10]}")


def bridge_report(cur: pd.DataFrame) -> None:
    coded = cur["hisco"].str.strip() != ""
    n_cov = int(pd.to_numeric(cur.loc[coded, "n"]).sum())
    n_all = int(pd.to_numeric(cur["n"]).sum())
    print(f"\nCuration after the bridge")
    print(f"  keys with a HISCO code  {int(coded.sum()):,} / {len(cur):,}")
    print(f"  records those cover     {n_cov:,} of {n_all:,} in the skeleton "
          f"({n_cov / n_all:.1%})")
    flagged = cur["review"].str.strip() != ""
    answered = cur["verdict"].str.strip() != ""
    print(f"  adjudicated             {int(answered.sum()):,} "
          f"({int(pd.to_numeric(cur.loc[answered, 'n']).sum()):,} records)")
    print(f"  still flagged for review {int(flagged.sum()):,} "
          f"({int(pd.to_numeric(cur.loc[flagged, 'n']).sum()):,} records)")
    print("\n  by source:")
    for k, v in cur.loc[coded, "source"].value_counts().items():
        print(f"    {k:<10} {v:>5,}")

    top_uncoded = cur[~coded].head(15)
    if len(top_uncoded):
        print("\n  largest keys still uncoded:")
        for r in top_uncoded.itertuples():
            print(f"    {int(float(r.n)):>6,}  {r.key}")


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[1])
    ap.add_argument("--overwrite", action="store_true",
                    help="re-derive codes even where the curation already has one")
    args = ap.parse_args(argv)

    # A key written twice in czech_glosses.py is invisible at runtime — the dict is already
    # built by then — so check the source before trusting either dict.
    dupes = duplicate_keys()
    if dupes:
        raise SystemExit(
            "czech_glosses.py defines the same key more than once; Python keeps the last "
            "silently:\n" + "\n".join(f"  {k!r} at lines {v}" for k, v in dupes.items()))

    print("Bridging Czech glosses -> HISCO …")
    cur = bridge(overwrite=args.overwrite)
    bridge_report(cur)
    cur.to_csv(CURATION, index=False, encoding="utf-8")
    print(f"\nUpdated {CURATION.relative_to(ROOT)}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
