How it works

amateur_theatre.code_clauses reads a clause by its leading date and its cue words. It leaves two kinds open (needs_model): an undated clause that prints a year inside (B11), whose year may or may not be the event's, and a dated clause no cue reaches, so that the year is known and the event is not. Neither is evidence today. This module puts them to a model, one clause a request, through the Message Batches API, and brings the readings back into the rules' own frame:

  1. select (load_clauses, select_open) — the open clauses of both chronicles, keyed clause_key exactly as the evidence table keys them (source:geo_id:source_key), cut to a SCOPES entry;
  2. context (add_context) — the place, the ensemble a page belongs to, the lines around the clause, and the database's own key to the abbreviations they use;
  3. request (build_request) — the codebook as a cached system prompt, the clause as the message, a JSON schema for the reply; each request keyed by its own hash (request_key), so a changed prompt is a new request and an old reply is never reused for it;
  4. send and collect (submit, collect) — batches, the state saved after each one sent; each reply checked (reading_problems) and cached under its key in data/interim/theatre/model_pass/replies/&lt;reader&gt;/, so a replay costs nothing;
  5. gold (draw_gold_sample, read_gold, score) — a stratified sample the author rules on blind, split into a calibration half (the prompt is fitted on it) and a certification half, sealed until PROMPT_FROZEN;
  6. apply (apply_readings) — the model's types and dates replace the cues and the missing date of the clauses it read, and amateur_theatre.code_clauses runs again on them: the model reads, the rulings (D-141) still decide what enters the measure.

The model returns what a clause says — its event types in the codebook's vocabulary, the interval its years give, whether it hedges, quoted titles, who performed — never a role, a certain/possible call or a panel count. A reply that fails its checks is kept with its problems and not applied.

Nothing here sends a request unless spend=True is passed (the notebook's SPEND).

def glossary_patterns(abbreviations: pd.DataFrame) -> tuple[re.Pattern, dict[str, str]]:
    """One pattern for every key entry short enough to gloss the prose, and
    ``abbreviation -> "SDO = …"``.  An abbreviation matches as a whole word (one ending in
    a full stop needs no word end); the longest is tried first at each position."""
    entries: dict[str, str] = {}
    for abbreviation, meaning in abbreviations[["abbreviation", "meaning"]].itertuples(index=False):
        if not isinstance(abbreviation, str) or not isinstance(meaning, str) or not abbreviation.strip():
            continue
        if len(meaning) <= GLOSSARY_MAX_CHARS:
            entries.setdefault(abbreviation.strip(), f"{abbreviation.strip()} = {meaning}")
    alternatives = [re.escape(a) + (r"(?!\w)" if a[-1].isalnum() else "")
                    for a in sorted(entries, key=len, reverse=True)]
    return re.compile(r"(?<!\w)(" + "|".join(alternatives) + ")"), entries

def add_context(open_clauses: pd.DataFrame, interim: Path = INTERIM, raw: Path = RAW) -> pd.DataFrame:
    """The open clauses with what a reader needs beside them: ``place``, ``ensemble``,
    ``context`` (the lines around the clause's own line) and ``glossary``.  Row count kept."""
    places = pd.read_csv(Path(raw) / "places.csv", usecols=["geo_id", "name", "okres"])
    ensembles = pd.read_csv(Path(raw) / "ensembles.csv",
                            usecols=["ensemble_id", "geo_id", "name", "span_raw"])
    patterns = glossary_patterns(pd.read_csv(Path(raw) / "abbreviations.csv"))
    wanted = open_clauses[open_clauses["source"] == "ensemble_chronicle"]
    place_lines = _place_lines(interim)
    lines = [place_lines[place_lines["geo_id"].isin(open_clauses.loc[open_clauses["source"] == "chronicle", "geo_id"])]]
    if len(wanted):
        lines.append(_ensemble_lines(pd.read_csv(Path(raw) / "ensemble_pages.csv",
                                                 usecols=["ensemble_id", "chronicle_text"]), wanted))
    windows = _neighbours(pd.concat(lines, ignore_index=True))
    out = open_clauses.copy()
    out["_ens"] = out["ensemble_id"].fillna(-1).astype(int)
    n = len(out)
    out = out.merge(windows.rename(columns={"ensemble_id": "_ens"}),
                    on=["source", "_ens", "geo_id", "line_no"], how="left", validate="m:1")
    place = places.assign(place=places["name"] + " (okres " + places["okres"].fillna("?") + ")")
    out = out.merge(place[["geo_id", "place"]], on="geo_id", how="left", validate="m:1")
    ens = ensembles.drop_duplicates(["ensemble_id", "geo_id"])
    span = ens["span_raw"].fillna("no span").str.strip().str.strip("()")
    ens = ens.assign(ensemble=ens["name"].fillna("") + " (" + span + ")")
    out = out.merge(ens[["ensemble_id", "geo_id", "ensemble"]].rename(columns={"ensemble_id": "_ens"}),
                    on=["_ens", "geo_id"], how="left", validate="m:1")
    assert len(out) == n, "context: a merge changed the row count"
    missing = out["context"].isna()
    if missing.any():
        print(f"context: {int(missing.sum())} clauses found no line; they are sent with the clause alone")
        out.loc[missing, "context"] = ">> " + out.loc[missing, "text"]
    out["glossary"] = [glossary(c, patterns) for c in out["context"]]
    return out.drop(columns="_ens")

    parts = [f"Place: {row['place'] if isinstance(row['place'], str) else 'unknown'}"]

        parts.append("Abbreviations: " + row["glossary"])
    return "\n\n".join(parts)

def estimate_cost(queue: list[dict], batched: bool = True) -> dict:
    """An offline estimate for the uncached part of ``queue``, from characters (no API call).

    The system prompt is counted once as a cache write and then as a cache read on every
    other request — best case; a batch's cache hits are not guaranteed.
    """
    todo = [q for q in queue if not q["cached"]]
    if not todo:
        return {"requests": 0, "usd": 0.0}
    reader = todo[0]["reader"]
    price_in, price_out, price_read, price_write = PRICES[reader]
    system = len(system_prompt(reader)) / CHARS_PER_TOKEN
    message = sum(len(message_text(q["request"])) for q in todo) / CHARS_PER_TOKEN
    if provider(reader) == "google":
        # thinking off: the JSON reply alone; the codebook at the full input rate on every
        # request, since an implicit-cache hit is not promised (an upper bound)
        output = OUTPUT_TOKENS_GUESS_NO_THINKING * len(todo)
        usd = (message * price_in + system * len(todo) * price_in + output * price_out) / 1e6
    else:
        output = OUTPUT_TOKENS_GUESS * len(todo) * (2 if READERS[reader].get("effort") == "high" else 1)
        usd = (message * price_in + system * price_write + system * (len(todo) - 1) * price_read
               + output * price_out) / 1e6
    factor = BATCH_DISCOUNT if batched else 1.0
    return {"reader": reader, "requests": len(todo), "input_tokens_est": round(message + system * len(todo)),
            "output_tokens_est": output, "usd": round(usd * factor, 2), "batched": batched}

def drain(queue: list[dict], name: str, *, spend: bool = False, pass_dir: Path = PASS_DIR,
          poll_seconds: int = 300, max_rounds: int = 200) -> int:
    """Send, collect and send again until every request of ``queue`` is cached or nothing more
    can be sent, sleeping ``poll_seconds`` while batches run.  For Gemini's capped queue
    (:data:`GEMINI_HELD_TOKENS`), which takes the full pass in several rounds.  Without
    ``spend`` it reports once and returns.  Returns how many requests are still uncached;
    a request that keeps failing stops the loop after ``max_rounds``."""
    def left() -> int:
        return sum(not _cache_path(q["reader"], q["key"], pass_dir).exists() for q in queue)

    for _ in range(max_rounds):
        submit(queue, name, spend=spend, pass_dir=pass_dir)
        done = collect(name, pass_dir)
        remaining = left()
        if not spend or remaining == 0:
            return remaining
        in_flight = any(not b.get("collected") for b in read_state(name, pass_dir))
        if done and not in_flight:
            submit(queue, name, spend=spend, pass_dir=pass_dir)       # what the last collect freed
            if not any(not b.get("collected") for b in read_state(name, pass_dir)):
                return left()                                           # nothing sendable left
        print(f"{name}: {remaining:,} requests not yet cached; waiting {poll_seconds} s")
        time.sleep(poll_seconds)
    return left()

def draw_gold_sample(contexted: pd.DataFrame, n: int = GOLD_N, seed: int = GOLD_SEED) -> pd.DataFrame:
    """A stratified draw of the open clauses (source × kind), each stratum at least
    :data:`GOLD_MIN_PER_STRATUM` or all it has, the rest in proportion; every stratum
    alternated into a ``calibration`` and a ``certification`` half.  Reproducible."""
    rng = np.random.default_rng(seed)
    strata = contexted.groupby(["source", "open_kind"])
    sizes = strata.size()
    take = np.minimum(sizes, np.maximum(GOLD_MIN_PER_STRATUM, (sizes / sizes.sum() * n).round().astype(int)))
    parts = []
    for (source, kind), group in strata:
        order = rng.permutation(len(group))[: int(take[(source, kind)])]
        drawn = group.iloc[np.sort(order)].copy()
        drawn["half"] = ["calibration" if i % 2 == 0 else "certification" for i in range(len(drawn))]
        parts.append(drawn)
    return pd.concat(parts, ignore_index=True)

    mt = m["types"].fillna("").map(lambda s: set(filter(None, s.split("|"))))
    per_type = []

"""The amateur-theatre model pass: the chronicle clauses the rules leave open, read by a model.

:func:`amateur_theatre.code_clauses` reads a clause by its leading date and its cue words.
It leaves two kinds open (``needs_model``): an **undated** clause that prints a year inside
(B11), whose year may or may not be the event's, and a **dated** clause no cue reaches, so
that the year is known and the event is not.  Neither is evidence today.  This module puts
them to a model, one clause a request, through the Message Batches API, and brings the
readings back into the rules' own frame:

1. **select** (:func:`load_clauses`, :func:`select_open`) — the open clauses of both
   chronicles, keyed ``clause_key`` exactly as the evidence table keys them
   (``source:geo_id:source_key``), cut to a :data:`SCOPES` entry;
2. **context** (:func:`add_context`) — the place, the ensemble a page belongs to, the
   lines around the clause, and the database's own key to the abbreviations they use;
3. **request** (:func:`build_request`) — the codebook as a cached system prompt, the clause
   as the message, a JSON schema for the reply; each request keyed by its own hash
   (:func:`request_key`), so a changed prompt is a new request and an old reply is never
   reused for it;
4. **send and collect** (:func:`submit`, :func:`collect`) — batches, the state saved after
   each one sent; each reply checked (:func:`reading_problems`) and cached under its key in
   ``data/interim/theatre/model_pass/replies/<reader>/``, so a replay costs nothing;
5. **gold** (:func:`draw_gold_sample`, :func:`read_gold`, :func:`score`) — a stratified
   sample the author rules on blind, split into a calibration half (the prompt is fitted on
   it) and a certification half, sealed until :data:`PROMPT_FROZEN`;
6. **apply** (:func:`apply_readings`) — the model's types and dates replace the cues and the
   missing date of the clauses it read, and :func:`amateur_theatre.code_clauses` runs again
   on them: the model reads, **the rulings (D-141) still decide** what enters the measure.

The model returns what a clause *says* — its event types in the codebook's vocabulary, the
interval its years give, whether it hedges, quoted titles, who performed — never a role, a
certain/possible call or a panel count.  A reply that fails its checks is kept with its
problems and not applied.

Nothing here sends a request unless ``spend=True`` is passed (the notebook's ``SPEND``).
"""

nothing else (`1851-1893 Thalia`, `Od 1885 jako SDO Tyl`).
- active: players or a company were active or played, without one particular performance \
(hrálo se, byl činný, působil, pořádal představení, divadlo doloženo).
- founded: a company or theatre group was founded or re-founded (zal., založen, vznikl, \
ustaven, obnoven).
- statutes: statutes approved, a society registered (stanovy, registrace).
- permit: an official permission or licence to perform (povolení, koncese).
- first: a first of its kind — the first performance, the first report, the first Czech \
play in the place (poprvé, první představení).
- performance: a particular performance or a named play was given (sehráli, uvedli, \
představení, a title).
- count: the clause states a number of plays or performances (`12 her`, `3 představení`).
- attendance: audience size or reception (návštěva, diváků, vyprodáno).
- proceeds: the takings went to a cause (ve prospěch, výtěžek, na fond).
- national_theatre: money or a performance for the National Theatre (Národní divadlo, ND).
- school_church: the players were pupils, students, a school, clergy or a monastery.
- travelling: a professional travelling company (kočovná společnost, the company of a \
named director such as společnost J. Kramuelové).
- german: played in German, or a German company or theatre — not a German place name, \
and not the German population as background.
- latin: played in Latin.
- venue: where it was played (hostinec, sál, sokolovna, jeviště, a stage, scenery).
- building: a theatre building built, burnt, rebuilt.
- members: members of the company (their number, who joined).
- renamed: renamed, merged, taken over by another body.
- ended: the company ended or was dissolved (zanikl, ukončil činnost, rozpadl se).
- pause: activity paused or stalled (přerušena, nehrálo se, nečinný).
- ban: a performance or the company was forbidden or censored (zákaz, nepovoleno, cenzura).

A clause about something other than theatre (a person's career elsewhere, a festival \
after the war with no local company, a book) has no types.  Puppet theatre is read like \
any other: its types are the same.

## hedged

true when the clause itself hedges what it records: patrně, snad, zřejmě, údajně, prý, \
možná, pravděpodobně, or a question mark.  A hedge in a neighbouring line does not count.

## date

If the message gives the clause a leading date, its date is already read: answer form \
`leading` with all four years null.

Otherwise the clause has no leading date and the message lists the years it prints.  Give \
the interval of the event the clause records, from those years:
- point: one event at one time — start_lo..start_hi, end null.  A year is that year \
(1887 → 1887..1887).
- span: activity from one time to another — start and end intervals.
- from: activity from a time on, no end printed (od 1885) — start only.
- until: activity up to a time, no start printed (do 1910) — end only.
- source_year: the only year is the date of a source that reports the activity (an \
almanac, a newspaper, a book: `zachytil ... Divadelní almanach 1869`) — end_lo = end_hi = \
that year, start null.
- none: the printed years do not date the event (a birth year, a later commemoration, a \
publication about something undated), or no interval can be given.  All four years null.

Fuzzy years keep their width: a decade (188x, 80. léta, osmdesátá léta) is 1880..1889; \
early / mid / late in a decade (počátkem, polovinou, koncem 80. let) is 1880..1883 / \
1883..1886 / 1887..1889; a century part (počátkem / polovinou / koncem 19. stol.) is \
1800..1819 / 1840..1859 / 1880..1899; kolem 1850 (about) is 1845..1855.  A bare decade takes \
its century from the context lines.  Never narrow a fuzzy year to a single year.

## titles

Every play title the clause names, as printed, without quotation marks.  None → [].

## performer

Who performed or acted, if the clause says so:
- page_ensemble: the message is an ensemble's own page and the clause is about that \
ensemble (stated or implied).
- named_group: a company or society the clause names (SDO Tyl, Sokol, Beseda, Čtenářský \
spolek) — name it as printed.
- school_or_church: pupils, students, a school, clergy.
- travelling_company: a professional travelling company — name it as printed.
- unnamed_amateurs: local amateurs, unnamed (ochotníci, místní občané).
- not_stated: the clause does not say.
name is null except for named_group and travelling_company.

## evidence

The shortest words, copied exactly from the clause, that support your types and date.  \
Empty only when the clause has no types and the date form is `none` or `leading`.
"""

def clause_keys(clauses: pd.DataFrame, source: str) -> pd.Series:

    if source == "ensemble_chronicle":

def load_clauses(interim: Path = INTERIM) -> pd.DataFrame:

    parts = []
    for source, name in CLAUSE_FILES.items():

        frame.insert(0, "source", source)

def _has_early_year(years_inside) -> bool:
    if not isinstance(years_inside, str) or not years_inside:
        return False
    return any(at.year_span(y)[0] < at.PRE_WAR_CUTOFF for y in years_inside.split("|") if y)


#: Which open clauses go to the model.  ``pre1914``: a dated clause that may attest the
#: years before 1914, or an undated one printing a pre-1914 year — the clauses that can
#: change the 1780–1913 panel directly.  ``all``: every open clause, the post-1914 ones of a
#: pre-1914 company included (a later sign of activity can extend its certain stretch).
SCOPES = ("pre1914", "all")


def select_open(clauses: pd.DataFrame, scope: str = "pre1914") -> pd.DataFrame:
    """The clauses the rules leave open (``needs_model``), cut to ``scope``.

    Adds ``open_kind``: ``undated`` (no leading date, B11) or ``untyped`` (dated, no type).
    """
    assert scope in SCOPES, f"scope must be one of {SCOPES}"
    open_ = clauses[clauses["needs_model"].astype(bool)].copy()
    undated = open_["date_source"].eq("none")
    open_["open_kind"] = np.where(undated, "undated", "untyped")
    if scope == "pre1914":
        dated_early = ~undated & open_["before_1914"].isin(["yes", "ambiguous"])
        undated_early = undated & open_["years_inside"].map(_has_early_year)
        open_ = open_[dated_early | undated_early]
    return open_.reset_index(drop=True)


#: Scopes for a pass over the clauses the rules *did* code (D-212, the two-labeller idea:
#: Ratner et al. 2017).  ``coded_pre1914``: pre-1914 (yes/ambiguous), dated, not puppet, not
#: left open — the rule audit's population before its one-per-text cut.  ``both_pre1914``:
#: that and ``pre1914`` together.  Readings of coded clauses are written to a file of their
#: own (:func:`write_readings`) and are not applied until the combination rule is ruled.
FULL_SCOPES = ("pre1914", "all", "coded_pre1914", "both_pre1914")


def select_coded(clauses: pd.DataFrame) -> pd.DataFrame:
    """The pre-1914 clauses the rules coded, with ``open_kind = "coded"``."""
    coded = clauses[clauses["before_1914"].isin(["yes", "ambiguous"]) & ~clauses["puppet"].astype(bool)
                    & ~clauses["needs_model"].astype(bool) & clauses["date_source"].ne("none")].copy()
    coded["open_kind"] = "coded"
    return coded.reset_index(drop=True)


def select_for_pass(clauses: pd.DataFrame, scope: str) -> pd.DataFrame:
    """The clauses a full pass reads, by :data:`FULL_SCOPES`."""
    assert scope in FULL_SCOPES, f"scope must be one of {FULL_SCOPES}"
    if scope in SCOPES:
        return select_open(clauses, scope)
    if scope == "coded_pre1914":
        return select_coded(clauses)
    return pd.concat([select_open(clauses, "pre1914"), select_coded(clauses)], ignore_index=True)


def open_counts(clauses: pd.DataFrame) -> pd.DataFrame:
    """Open clauses by scope, source and kind — what each scope would send."""
    rows = []
    for scope in SCOPES:
        sel = select_open(clauses, scope)
        for (source, kind), n in sel.groupby(["source", "open_kind"]).size().items():
            rows.append({"scope": scope, "source": source, "open_kind": kind, "clauses": int(n)})
    out = pd.DataFrame(rows)
    return out.pivot_table(index=["source", "open_kind"], columns="scope", values="clauses",
                           aggfunc="sum", fill_value=0, margins=True, margins_name="total")

CONTEXT_BEFORE, CONTEXT_AFTER = 2, 1
#: A key entry longer than this is a bibliography siglum, not an abbreviation of the prose.
GLOSSARY_MAX_CHARS = 120


def _place_lines(interim: Path) -> pd.DataFrame:
    lines = pd.read_csv(Path(interim) / "chronicle_lines.csv", low_memory=False, encoding="utf-8-sig",
                        usecols=["geo_id", "line_no", "text"])
    lines.insert(0, "source", "chronicle")
    lines["ensemble_id"] = -1
    return lines


def _ensemble_lines(pages: pd.DataFrame, wanted: pd.DataFrame) -> pd.DataFrame:
    """The lines of each wanted ensemble page, read as the pipeline reads them."""
    text = pages.set_index("ensemble_id")["chronicle_text"]
    rows = []
    for ensemble_id, geo_id in wanted[["ensemble_id", "geo_id"]].drop_duplicates().itertuples(index=False):
        chronicle = text.get(int(ensemble_id))
        if not isinstance(chronicle, str):
            continue
        for r in at.read_chronicle(int(geo_id), chronicle):
            rows.append({"source": "ensemble_chronicle", "geo_id": int(geo_id), "line_no": r["line_no"],
                         "text": r["text"], "ensemble_id": int(ensemble_id)})
    return pd.DataFrame(rows, columns=["source", "geo_id", "line_no", "text", "ensemble_id"])


def _neighbours(lines: pd.DataFrame, before: int = CONTEXT_BEFORE, after: int = CONTEXT_AFTER) -> pd.DataFrame:
    """Each line with the lines around it in its chronicle, as ``context`` (the line marked >>)."""
    out = []
    for _, block in lines.sort_values("line_no").groupby(["source", "ensemble_id", "geo_id"], sort=False):
        texts = block["text"].tolist()
        for i, line_no in enumerate(block["line_no"]):
            window = [("   " if j != i else ">> ") + texts[j]
                      for j in range(max(0, i - before), min(len(texts), i + after + 1))]
            out.append((block["source"].iat[0], block["ensemble_id"].iat[0], block["geo_id"].iat[0],
                        line_no, texts[i], "\n".join(window)))
    return pd.DataFrame(out, columns=["source", "ensemble_id", "geo_id", "line_no", "line_text", "context"])


def glossary_patterns(abbreviations: pd.DataFrame) -> tuple[re.Pattern, dict[str, str]]:
    """One pattern for every key entry short enough to gloss the prose, and
    ``abbreviation -> "SDO = …"``.  An abbreviation matches as a whole word (one ending in
    a full stop needs no word end); the longest is tried first at each position."""
    entries: dict[str, str] = {}
    for abbreviation, meaning in abbreviations[["abbreviation", "meaning"]].itertuples(index=False):
        if not isinstance(abbreviation, str) or not isinstance(meaning, str) or not abbreviation.strip():
            continue
        if len(meaning) <= GLOSSARY_MAX_CHARS:
            entries.setdefault(abbreviation.strip(), f"{abbreviation.strip()} = {meaning}")
    alternatives = [re.escape(a) + (r"(?!\w)" if a[-1].isalnum() else "")
                    for a in sorted(entries, key=len, reverse=True)]
    return re.compile(r"(?<!\w)(" + "|".join(alternatives) + ")"), entries


def glossary(text: str, patterns: tuple[re.Pattern, dict[str, str]]) -> str:
    """The database's key to every abbreviation ``text`` uses, ``SDO = …; ND = …``."""
    pattern, entries = patterns
    return "; ".join(dict.fromkeys(entries[m.group(1)] for m in pattern.finditer(text)))


def add_context(open_clauses: pd.DataFrame, interim: Path = INTERIM, raw: Path = RAW) -> pd.DataFrame:
    """The open clauses with what a reader needs beside them: ``place``, ``ensemble``,
    ``context`` (the lines around the clause's own line) and ``glossary``.  Row count kept."""
    places = pd.read_csv(Path(raw) / "places.csv", usecols=["geo_id", "name", "okres"])
    ensembles = pd.read_csv(Path(raw) / "ensembles.csv",
                            usecols=["ensemble_id", "geo_id", "name", "span_raw"])
    patterns = glossary_patterns(pd.read_csv(Path(raw) / "abbreviations.csv"))
    wanted = open_clauses[open_clauses["source"] == "ensemble_chronicle"]
    place_lines = _place_lines(interim)
    lines = [place_lines[place_lines["geo_id"].isin(open_clauses.loc[open_clauses["source"] == "chronicle", "geo_id"])]]
    if len(wanted):
        lines.append(_ensemble_lines(pd.read_csv(Path(raw) / "ensemble_pages.csv",
                                                 usecols=["ensemble_id", "chronicle_text"]), wanted))
    windows = _neighbours(pd.concat(lines, ignore_index=True))
    out = open_clauses.copy()
    out["_ens"] = out["ensemble_id"].fillna(-1).astype(int)
    n = len(out)
    out = out.merge(windows.rename(columns={"ensemble_id": "_ens"}),
                    on=["source", "_ens", "geo_id", "line_no"], how="left", validate="m:1")
    place = places.assign(place=places["name"] + " (okres " + places["okres"].fillna("?") + ")")
    out = out.merge(place[["geo_id", "place"]], on="geo_id", how="left", validate="m:1")
    ens = ensembles.drop_duplicates(["ensemble_id", "geo_id"])
    span = ens["span_raw"].fillna("no span").str.strip().str.strip("()")
    ens = ens.assign(ensemble=ens["name"].fillna("") + " (" + span + ")")
    out = out.merge(ens[["ensemble_id", "geo_id", "ensemble"]].rename(columns={"ensemble_id": "_ens"}),
                    on=["_ens", "geo_id"], how="left", validate="m:1")
    assert len(out) == n, "context: a merge changed the row count"
    missing = out["context"].isna()
    if missing.any():
        print(f"context: {int(missing.sum())} clauses found no line; they are sent with the clause alone")
        out.loc[missing, "context"] = ">> " + out.loc[missing, "text"]
    out["glossary"] = [glossary(c, patterns) for c in out["context"]]
    return out.drop(columns="_ens")

def _year(value) -> str:
    return "?" if pd.isna(value) else str(int(value))


def clause_message(row) -> str:
    """The user message for one clause (``row`` from :func:`add_context`)."""
    parts = [f"Place: {row['place'] if isinstance(row['place'], str) else 'unknown'}"]
    if row["source"] == "ensemble_chronicle":
        parts.append(f"This is the ensemble's own page: {row['ensemble'] if isinstance(row['ensemble'], str) else '?'}")
    else:
        parts.append("This is the place's chronicle.")
    parts.append("Lines around the clause (>> marks its line):\n" + row["context"])
    parts.append(f"The clause: {row['text']}")
    if row["date_source"] == "none":
        years = row["years_inside"] if isinstance(row["years_inside"], str) else ""
        parts.append("It has no leading date. Years it prints: " + (years.replace("|", ", ") or "none"))
    else:
        parts.append(f"Its leading date: {row['date_raw']} (read as start {_year(row['start_lo'])}–"
                     f"{_year(row['start_hi'])}, end {_year(row['end_lo'])}–{_year(row['end_hi'])})")
    if isinstance(row["glossary"], str) and row["glossary"]:
        parts.append("Abbreviations: " + row["glossary"])
    return "\n\n".join(parts)


def system_prompt(reader: str) -> str:
    """The codebook as ``reader`` reads it: :data:`SYSTEM_PROMPT`, or for a reader whose spec
    says ``"prompt": "reversed"`` the same text with the type definitions in reverse order
    (the label-free order check, D-212).  Nothing else changes."""
    kind = READERS[reader].get("prompt")
    if kind == "judge":
        return SYSTEM_PROMPT + JUDGE_ADDENDUM
    if kind != "reversed":
        return SYSTEM_PROMPT
    head, rest = SYSTEM_PROMPT.split("## types", 1)
    title, body = rest.split("\n\n", 1)
    items, tail = body.split("\n\nA clause about", 1)
    lines = items.split("\n")
    assert all(l.startswith("- ") for l in lines), "codebook: one line per type"
    return head + "## types" + title + "\n\n" + "\n".join(reversed(lines)) + "\n\nA clause about" + tail


#: What the judge (D-217) reads after the codebook.  The rest of its request is a reader's.
JUDGE_ADDENDUM = """

## Other readings

This clause has been read by other readers: two models and, where rules already coded the \
clause, the rules (their event types only).  They disagree on something that reaches the \
measure — whether the clause records Czech theatre, a company, a performance, a hedge, or \
the date of an undated clause.  Any of them may be wrong, and so may all of them.  Read the \
clause yourself against the codebook above and give your own reading in the same schema.  \
Agree with a reader only where the clause supports it; a majority is not evidence.
"""


def _readings_note(others: list[dict]) -> str:
    """The other readings, for the judge: one line each, in a fixed order."""
    lines = []
    for o in others:
        parts = [f"types: {o.get('types') or 'none'}", f"hedged: {'yes' if o.get('hedged') else 'no'}"]
        if o.get("date_form"):
            years = [o.get(k) for k in ("start_lo", "start_hi", "end_lo", "end_hi")]
            parts.append(f"date: {o['date_form']} {'/'.join('' if pd.isna(y) else str(int(y)) for y in years)}")
        if o.get("titles"):
            parts.append(f"titles: {o['titles']}")
        lines.append(f"- {o['name']}: " + "; ".join(parts))
    return "Readings to weigh:\n" + "\n".join(lines)


def build_request(row, reader: str, others: list[dict] | None = None) -> dict:
    """The request for one clause, in the reader's provider's own form; the codebook first
    and cached, the reply a JSON schema.  ``others`` (the judge only) are the readings it weighs.

    Anthropic: a Messages API request, the codebook a cached system block.  Google: a native
    ``GenerateContentRequest`` with its ``model`` (taken off when it is written to a batch
    file), the codebook as ``systemInstruction`` (the shared prefix Gemini's implicit cache
    can serve), ``responseJsonSchema`` and the reader's thinking budget.
    """
    spec = READERS[reader]
    message = clause_message(row)
    if others:
        message += "\n\n" + _readings_note(others)
    if provider(reader) == "google":
        return {"model": spec["model"],
                "systemInstruction": {"parts": [{"text": system_prompt(reader)}]},
                "contents": [{"role": "user", "parts": [{"text": message}]}],
                "generationConfig": {"maxOutputTokens": MAX_TOKENS, "responseMimeType": "application/json",
                                     "responseJsonSchema": READING_SCHEMA,
                                     "thinkingConfig": {"thinkingBudget": spec["thinking_budget"]}}}
    return {
        "model": spec["model"],
        "max_tokens": MAX_TOKENS,
        "system": [{"type": "text", "text": system_prompt(reader), "cache_control": {"type": "ephemeral"}}],
        "thinking": {"type": "adaptive"},
        "output_config": {"effort": spec["effort"],
                          "format": {"type": "json_schema", "schema": READING_SCHEMA}},
        "messages": [{"role": "user", "content": message}],
    }


def message_text(request: dict) -> str:
    """The user message of a request of either provider."""
    if "contents" in request:
        return "".join(p.get("text", "") for p in request["contents"][0]["parts"])
    return request["messages"][0]["content"]


def request_key(request: dict) -> str:
    """The request's own hash (64 hex characters — also a valid batch ``custom_id``)."""
    return hashlib.sha256(json.dumps(request, sort_keys=True, ensure_ascii=False).encode("utf-8")).hexdigest()


def _cache_path(reader: str, key: str, pass_dir: Path) -> Path:
    return Path(pass_dir) / "replies" / reader / f"{key}.json"


def build_queue(contexted: pd.DataFrame, reader: str, pass_dir: Path = PASS_DIR,
                others: dict[str, list[dict]] | None = None) -> list[dict]:
    """One record per clause: ``clause_key``, ``key``, ``request``, and whether it is cached.
    ``others`` (``clause_key`` → readings) builds the judge's queue."""
    queue = []
    for _, row in contexted.iterrows():
        request = build_request(row, reader, (others or {}).get(row["clause_key"]))
        key = request_key(request)
        queue.append({"clause_key": row["clause_key"], "key": key, "reader": reader, "request": request,
                      "cached": _cache_path(reader, key, pass_dir).exists()})
    return queue

def estimate_cost(queue: list[dict], batched: bool = True) -> dict:
    """An offline estimate for the uncached part of ``queue``, from characters (no API call).

    The system prompt is counted once as a cache write and then as a cache read on every
    other request — best case; a batch's cache hits are not guaranteed.
    """
    todo = [q for q in queue if not q["cached"]]
    if not todo:
        return {"requests": 0, "usd": 0.0}
    reader = todo[0]["reader"]
    price_in, price_out, price_read, price_write = PRICES[reader]
    system = len(system_prompt(reader)) / CHARS_PER_TOKEN
    message = sum(len(message_text(q["request"])) for q in todo) / CHARS_PER_TOKEN
    if provider(reader) == "google":
        # thinking off: the JSON reply alone; the codebook at the full input rate on every
        # request, since an implicit-cache hit is not promised (an upper bound)
        output = OUTPUT_TOKENS_GUESS_NO_THINKING * len(todo)
        usd = (message * price_in + system * len(todo) * price_in + output * price_out) / 1e6
    else:
        output = OUTPUT_TOKENS_GUESS * len(todo) * (2 if READERS[reader].get("effort") == "high" else 1)
        usd = (message * price_in + system * price_write + system * (len(todo) - 1) * price_read
               + output * price_out) / 1e6
    factor = BATCH_DISCOUNT if batched else 1.0
    return {"reader": reader, "requests": len(todo), "input_tokens_est": round(message + system * len(todo)),
            "output_tokens_est": output, "usd": round(usd * factor, 2), "batched": batched}


def usage_cost(usage: dict, reader: str, batched: bool = True) -> float:
    """USD for one reply's recorded usage."""
    price_in, price_out, price_read, price_write = PRICES[reader]
    usd = (usage.get("input", 0) * price_in + usage.get("output", 0) * price_out
           + usage.get("cache_read", 0) * price_read + usage.get("cache_write", 0) * price_write) / 1e6
    return usd * (BATCH_DISCOUNT if batched else 1.0)

    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(".json.part")
    tmp.write_text(json.dumps(state, indent=1), encoding="utf-8")
    tmp.replace(path)


def _chunks(records: list[dict]) -> list[list[dict]]:
    out, cur, size = [], [], 0
    for r in records:
        n = len(json.dumps(r["request"], ensure_ascii=False).encode("utf-8"))
        if cur and (len(cur) >= BATCH_MAX_REQUESTS or size + n > BATCH_MAX_BYTES):
            out.append(cur)
            cur, size = [], 0
        cur.append(r)
        size += n
    if cur:
        out.append(cur)
    return out


def submit(queue: list[dict], name: str, *, spend: bool = False, pass_dir: Path = PASS_DIR) -> list[str]:
    """Send every uncached request not already in flight under ``name``, in batches.

    Without ``spend`` nothing is sent: what would be sent is printed.  The state is saved
    after every batch, so a failure on a later one cannot forget an earlier one.
    """
    state = read_state(name, pass_dir)
    in_flight = {k for b in state if not b.get("collected") for k in b["keys"]}
    todo = [q for q in queue if not _cache_path(q["reader"], q["key"], pass_dir).exists()
            and q["key"] not in in_flight]
    if not todo:
        print(f"{name}: nothing to send — every request is cached or in flight")
        return []
    chunks = _chunks(todo)
    if not spend:
        print(f"{name}: SPEND is off — {len(todo):,} requests in {len(chunks)} batch(es) would be sent")
        return []
    _pass_machine_guard("sending")
    if provider(todo[0]["reader"]) == "google":
        return _submit_gemini(todo, name, state, pass_dir)
    client, sent = _client(), []
    for chunk in chunks:
        batch = client.messages.batches.create(
            requests=[{"custom_id": q["key"], "params": q["request"]} for q in chunk])
        state.append({"id": batch.id, "reader": chunk[0]["reader"], "keys": [q["key"] for q in chunk],
                      "clause_keys": [q["clause_key"] for q in chunk], "submitted": time.time(),
                      "collected": False})
        _save_state(name, state, pass_dir)
        sent.append(batch.id)
        print(f"{name}: batch {batch.id}, {len(chunk):,} requests")
    return sent


def parse_reply(message) -> tuple[dict | None, dict, str]:
    """``(reading, usage, stop_reason)`` from one reply; ``reading`` is None for a reply
    that refused, was cut off or is not JSON — such a reply is not cached."""
    u = message.usage
    usage = {"input": u.input_tokens, "output": u.output_tokens,
             "cache_write": getattr(u, "cache_creation_input_tokens", 0) or 0,
             "cache_read": getattr(u, "cache_read_input_tokens", 0) or 0}
    if message.stop_reason in ("refusal", "max_tokens"):
        return None, usage, message.stop_reason
    text = next((b.text for b in message.content if b.type == "text"), "")
    try:
        return json.loads(text), usage, message.stop_reason
    except ValueError:
        return None, usage, message.stop_reason


def _write_reply(path: Path, record: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(".json.part")
    tmp.write_text(json.dumps(record, ensure_ascii=False), encoding="utf-8")
    tmp.replace(path)


def collect(name: str, pass_dir: Path = PASS_DIR) -> bool:
    """Cache the replies of every ended batch under ``name``.  ``True`` when none is pending.

    Retrieving costs nothing.  A failed reply is counted and left uncached, so the next
    :func:`submit` sends it again.
    """
    state = read_state(name, pass_dir)
    if not state:
        print(f"{name}: no batch has been sent")
        return True
    if pass_dir == PASS_DIR:
        _pass_machine_guard("collecting")
    client, pending = None, False
    for b in state:
        if b.get("collected"):
            continue
        if b.get("provider") == "google":
            pending |= not _collect_gemini(b, name, state, pass_dir)
            continue
        client = client or _client()
        batch = client.messages.batches.retrieve(b["id"])
        if batch.processing_status != "ended":
            print(f"{name}: batch {b['id']} {batch.processing_status}, "
                  f"{batch.request_counts.processing} processing")
            pending = True
            continue
        clause_of = dict(zip(b["keys"], b["clause_keys"]))
        cached = failed = 0
        for result in client.messages.batches.results(b["id"]):
            if result.result.type != "succeeded":
                failed += 1
                continue
            reading, usage, stop = parse_reply(result.result.message)
            if reading is None:
                failed += 1
                continue
            _write_reply(_cache_path(b["reader"], result.custom_id, pass_dir),
                         {"clause_key": clause_of.get(result.custom_id), "reader": b["reader"],
                          "model": READERS[b["reader"]]["model"], "reading": reading, "usage": usage,
                          "stop_reason": stop, "batch": b["id"], "at": time.time()})
            cached += 1
        b.update(collected=True, cached=cached, failed=failed)
        _save_state(name, state, pass_dir)
        print(f"{name}: batch {b['id']} — {cached:,} cached, {failed:,} failed (sent again next time)")
    return not pending


def drain(queue: list[dict], name: str, *, spend: bool = False, pass_dir: Path = PASS_DIR,
          poll_seconds: int = 300, max_rounds: int = 200) -> int:
    """Send, collect and send again until every request of ``queue`` is cached or nothing more
    can be sent, sleeping ``poll_seconds`` while batches run.  For Gemini's capped queue
    (:data:`GEMINI_HELD_TOKENS`), which takes the full pass in several rounds.  Without
    ``spend`` it reports once and returns.  Returns how many requests are still uncached;
    a request that keeps failing stops the loop after ``max_rounds``."""
    def left() -> int:
        return sum(not _cache_path(q["reader"], q["key"], pass_dir).exists() for q in queue)

    for _ in range(max_rounds):
        submit(queue, name, spend=spend, pass_dir=pass_dir)
        done = collect(name, pass_dir)
        remaining = left()
        if not spend or remaining == 0:
            return remaining
        in_flight = any(not b.get("collected") for b in read_state(name, pass_dir))
        if done and not in_flight:
            submit(queue, name, spend=spend, pass_dir=pass_dir)       # what the last collect freed
            if not any(not b.get("collected") for b in read_state(name, pass_dir)):
                return left()                                           # nothing sendable left
        print(f"{name}: {remaining:,} requests not yet cached; waiting {poll_seconds} s")
        time.sleep(poll_seconds)
    return left()

def _gemini_client():
    """The native client.  Held in a name for a whole call: built inline, it is collected
    mid-upload and closes its own connection (``reader_batches`` 1.2.0)."""
    import os

    from google import genai

    return genai.Client(api_key=os.environ.get("GEMINI_API_KEY") or os.environ.get("GOOGLE_API_KEY", ""))


def gemini_line(q: dict) -> str:
    """One line of a native batch input file: ``{"key", "request"}``, the request without its
    ``model`` (a native batch names the model once, for the whole job)."""
    request = {k: v for k, v in q["request"].items() if k != "model"}
    return json.dumps({"key": q["key"], "request": request}, ensure_ascii=True)


def _submit_gemini(todo: list[dict], name: str, state: list[dict], pass_dir: Path) -> list[str]:
    """One native batch job per input file: written to the temp folder (never beside the
    data, which Dropbox would sync), uploaded, sent, deleted locally; the state saved after
    every job.  A refused upload or job stops here: the jobs before it are on record."""
    import tempfile

    model = READERS[todo[0]["reader"]]["model"]
    client, sent = _gemini_client(), []
    files, cur, size, tokens = [], [], 0, 0
    for q in todo:
        line, n = gemini_line(q) + "\n", gemini_tokens(q)
        if cur and (size + len(line) > GEMINI_FILE_BYTES or tokens + n > GEMINI_JOB_TOKENS):
            files.append(cur)
            cur, size, tokens = [], 0, 0
        cur.append((q, line, n))
        size, tokens = size + len(line), tokens + n
    if cur:
        files.append(cur)
    held = sum(b.get("tokens", 0) for b in state if b.get("provider") == "google" and not b.get("collected"))
    for chunk in files:
        job_tokens = sum(n for _, _, n in chunk)
        if held + job_tokens > GEMINI_HELD_TOKENS:
            waiting = sum(len(c) for c in files[files.index(chunk):])
            print(f"{name}: {waiting:,} Gemini requests wait — {held:,} tokens are queued and not yet "
                  f"collected (this pass keeps at most {GEMINI_HELD_TOKENS:,}); collect, then submit again")
            return sent
        fd, path = tempfile.mkstemp(prefix="theatre-gemini-", suffix=".jsonl")
        try:
            with open(fd, "w", encoding="ascii") as fh:
                fh.writelines(line for _, line, _ in chunk)
            label = f"theatre-{name}-{len(state) + 1}"
            try:
                uploaded = client.files.upload(file=path, config={"display_name": label, "mime_type": "jsonl"})
                job = client.batches.create(model=model, src=uploaded.name, config={"display_name": label})
            except Exception as exc:                        # noqa: BLE001 — stop; what was sent is on record
                print(f"{name}: Gemini job not sent — {type(exc).__name__}: {str(exc)[:300]}")
                return sent
        finally:
            Path(path).unlink(missing_ok=True)
        state.append({"id": job.name, "provider": "google", "reader": chunk[0][0]["reader"],
                      "keys": [q["key"] for q, _, _ in chunk], "clause_keys": [q["clause_key"] for q, _, _ in chunk],
                      "tokens": job_tokens, "input_file": uploaded.name, "submitted": time.time(),
                      "collected": False})
        _save_state(name, state, pass_dir)
        held += job_tokens
        sent.append(job.name)
        print(f"{name}: Gemini job {job.name}, {len(chunk):,} requests, ~{job_tokens:,} tokens")
    return sent


def parse_gemini(response: dict) -> tuple[dict | None, dict, str]:
    """``(reading, usage, finish)`` from one native ``GenerateContentResponse``; ``reading`` is
    None for a reply that did not stop by itself or is not JSON — such a reply is not cached.
    ``input`` is the prompt tokens not served from the implicit cache, ``cache_read`` those that
    were; thinking tokens (none with thinking off) are booked with the output."""
    u = response.get("usageMetadata") or {}
    cached = u.get("cachedContentTokenCount", 0) or 0
    usage = {"input": (u.get("promptTokenCount", 0) or 0) - cached, "cache_read": cached, "cache_write": 0,
             "output": (u.get("candidatesTokenCount", 0) or 0) + (u.get("thoughtsTokenCount", 0) or 0)}
    candidates = response.get("candidates") or []
    if not candidates:
        return None, usage, "no candidate"
    finish = candidates[0].get("finishReason") or ""
    if finish != "STOP":
        return None, usage, finish
    parts = (candidates[0].get("content") or {}).get("parts") or []
    text = "".join(p.get("text") or "" for p in parts if not p.get("thought"))
    try:
        return json.loads(text), usage, finish
    except ValueError:
        return None, usage, "unreadable JSON"


def _collect_gemini(b: dict, name: str, state: list[dict], pass_dir: Path) -> bool:
    """Cache one native job's replies if it has ended (``True``), else report it (``False``).
    The results file is kept under ``pass_dir/gemini_results/``: the replies are what was paid for."""
    client = _gemini_client()
    job = client.batches.get(name=b["id"])
    status = str(getattr(getattr(job, "state", None), "name", getattr(job, "state", None)))
    if status not in GEMINI_ENDED:
        print(f"{name}: Gemini job {b['id']} {status}")
        return False
    local = Path(pass_dir) / "gemini_results" / (b["id"].replace("/", "_") + ".jsonl")
    file_name = getattr(getattr(job, "dest", None), "file_name", None)
    if local.exists():
        text = local.read_text(encoding="utf-8")
    elif file_name:
        text = client.files.download(file=file_name).decode("utf-8")
        local.parent.mkdir(parents=True, exist_ok=True)
        local.write_text(text, encoding="utf-8")
    else:
        text = ""
    clause_of = dict(zip(b["keys"], b["clause_keys"]))
    cached = failed = 0
    for line in filter(None, text.splitlines()):
        row = json.loads(line)
        reading, usage, finish = parse_gemini(row.get("response") or {})
        if reading is None or row.get("key") not in clause_of:
            failed += 1
            continue
        _write_reply(_cache_path(b["reader"], row["key"], pass_dir),
                     {"clause_key": clause_of[row["key"]], "reader": b["reader"],
                      "model": READERS[b["reader"]]["model"], "reading": reading, "usage": usage,
                      "stop_reason": finish, "batch": b["id"], "at": time.time()})
        cached += 1
    b.update(collected=True, state=status, cached=cached, failed=len(b["keys"]) - cached)
    _save_state(name, state, pass_dir)
    print(f"{name}: Gemini job {b['id']} ({status}) — {cached:,} cached, "
          f"{len(b['keys']) - cached:,} not (sent again next time)")
    return True

def _norm(text: str) -> str:
    return re.sub(r"\s+", " ", text or "").strip().casefold()


def reading_problems(reading: dict, clause_text: str, undated: bool) -> list[str]:
    """What is wrong with one reading, for its clause (empty: nothing)."""
    problems = []
    unknown = set(reading.get("types", [])) - set(TYPES)
    if unknown:
        problems.append(f"unknown types {sorted(unknown)}")
    date = reading.get("date") or {}
    form = date.get("form")
    years = {k: date.get(k) for k in ("start_lo", "start_hi", "end_lo", "end_hi")}
    if not undated and form != "leading":
        problems.append(f"a dated clause read as {form}")
    if undated and form == "leading":
        problems.append("an undated clause read as leading")
    need = {"point": ("start_lo", "start_hi"), "span": ("start_lo", "start_hi", "end_lo", "end_hi"),
            "from": ("start_lo", "start_hi"), "until": ("end_lo", "end_hi"),
            "source_year": ("end_lo", "end_hi")}.get(form, ())
    if any(years[k] is None for k in need):
        problems.append(f"{form} without {', '.join(k for k in need if years[k] is None)}")
    if form in ("leading", "none") and any(v is not None for v in years.values()):
        problems.append(f"{form} with years")
    for k, v in years.items():
        if v is not None and not (YEAR_RANGE[0] <= v <= YEAR_RANGE[1]):
            problems.append(f"{k} {v} out of range")
    for lo, hi in (("start_lo", "start_hi"), ("end_lo", "end_hi")):
        if years[lo] is not None and years[hi] is not None and years[lo] > years[hi]:
            problems.append(f"{lo} after {hi}")
    if years["start_lo"] is not None and years["end_hi"] is not None and years["start_lo"] > years["end_hi"]:
        problems.append("starts after it ends")
    evidence = reading.get("evidence", "")
    if evidence and _norm(evidence) not in _norm(clause_text):
        # D-217, after the check stage: readers quote in pieces ("…; …", "... "), so a quote
        # passes when nearly all its words are the clause's; words from a neighbouring line or
        # invented ones still fail it
        words = _words(evidence)
        have = set(_words(clause_text))
        if not words or sum(w in have for w in words) / len(words) < EVIDENCE_WORD_SHARE:
            problems.append("evidence not in the clause")
    return problems


#: The share of an evidence quote's words (three letters or more) that must be the clause's.
EVIDENCE_WORD_SHARE = 0.9


def _words(text: str) -> list[str]:
    return re.findall(r"\w{3,}", _norm(text))


def readings_table(open_clauses: pd.DataFrame, queue: list[dict], pass_dir: Path = PASS_DIR) -> pd.DataFrame:
    """One row per clause of ``queue`` that has a cached reply: the reading, flat, checked.

    ``valid`` is False where :func:`reading_problems` found anything; such a reading is not
    applied.  Usage and cost per reply are kept for the cost report.
    """
    text = open_clauses.set_index("clause_key")
    rows = []
    for q in queue:
        path = _cache_path(q["reader"], q["key"], pass_dir)
        if not path.exists():
            continue
        record = json.loads(path.read_text(encoding="utf-8"))
        r, clause = record["reading"], text.loc[q["clause_key"]]
        problems = reading_problems(r, clause["text"], clause["date_source"] == "none")
        date = r.get("date") or {}
        rows.append({"clause_key": q["clause_key"], "reader": q["reader"],
                     "types": "|".join(sorted(r.get("types", []))), "hedged": bool(r.get("hedged")),
                     "date_form": date.get("form"),
                     **{k: date.get(k) for k in ("start_lo", "start_hi", "end_lo", "end_hi")},
                     "titles": "|".join(r.get("titles", [])),
                     "performer_kind": (r.get("performer") or {}).get("kind"),
                     "performer_name": (r.get("performer") or {}).get("name"),
                     "evidence": r.get("evidence", ""), "valid": not problems,
                     "problems": "; ".join(problems),
                     "usd": usage_cost(record["usage"], q["reader"]), **record["usage"]})
    return pd.DataFrame(rows)


def cost_report(readings: pd.DataFrame, n_full: int | None = None) -> pd.DataFrame:
    """Measured cost per reader, and what ``n_full`` clauses would cost at the same rate."""
    out = readings.groupby("reader").agg(replies=("clause_key", "size"), usd=("usd", "sum"),
                                         output=("output", "mean"), cache_read=("cache_read", "mean"))
    out["usd_per_clause"] = out["usd"] / out["replies"]
    if n_full:
        out[f"usd_for_{n_full}"] = (out["usd_per_clause"] * n_full).round(2)
    return out.round(4)

def apply_readings(coded: pd.DataFrame, readings: pd.DataFrame, source: str) -> pd.DataFrame:
    """``coded`` clauses of one source with the valid ``readings`` applied, coded again.

    For each clause read: the model's types (plus ``title`` where it names a title and
    ``hedged`` where it hedges) replace the cues; an undated clause takes the model's
    interval and ``date_source = "model"``; then :func:`amateur_theatre.code_clauses` runs
    on the frame, so the rulings decide every flag.  The model's titles join the rules'
    quoted ones.  ``read_by``, ``performer_kind`` and ``performer_name`` are added.
    Rows and key unchanged; a clause without a valid reading comes out as the rules left it.
    """
    out = coded[[c for c in RAW_CLAUSE_COLUMNS if c in coded]].copy()
    out["text"] = out["text"].fillna("")          # an empty clause reads back from CSV as NaN
    keys = clause_keys(out, source)
    valid = readings[readings["valid"].astype(bool)] if "valid" in readings else readings
    assert valid["clause_key"].is_unique, "readings: one reading a clause"
    r = valid.set_index("clause_key").reindex(keys.values)
    r.index = out.index
    hit = r["types"].notna()

    def cues(row) -> str:
        types = [t for t in str(row["types"]).split("|") if t]
        return "|".join(types + (["title"] if row["titles"] else []) + (["hedged"] if row["hedged"] else []))

    r["titles"] = r["titles"].fillna("")
    out.loc[hit, "cues"] = r.loc[hit].apply(cues, axis=1)
    dated = hit & out["date_source"].eq("none") & r["date_form"].isin(list(DATE_FORM_RULES))
    for column in ("start_lo", "start_hi", "end_lo", "end_hi"):
        out.loc[dated, column] = pd.to_numeric(r.loc[dated, column], errors="coerce")
    out.loc[dated, "date_form"] = r.loc[dated, "date_form"].map(DATE_FORM_RULES)
    out.loc[dated, "date_source"] = "model"
    out.loc[dated, "before_1914"] = [at.attests_before(*v) for v in
                                     out.loc[dated, ["start_lo", "start_hi", "end_lo", "end_hi"]].itertuples(index=False)]
    recoded = at.code_clauses(out)
    model_titles = r["titles"].map(lambda s: [at._title_key(t) for t in s.split("|") if t])
    recoded.loc[hit, "titles"] = [
        "|".join(sorted(set(filter(None, str(a).split("|"))) | set(filter(None, b))))
        for a, b in zip(recoded.loc[hit, "titles"].fillna(""), model_titles[hit])]
    recoded.loc[hit, "needs_model"] = False
    recoded["read_by"] = r["reader"].where(hit, "")
    recoded["performer_kind"] = r["performer_kind"].where(hit)
    recoded["performer_name"] = r["performer_name"].where(hit)
    assert len(recoded) == len(coded)
    return recoded

def draw_gold_sample(contexted: pd.DataFrame, n: int = GOLD_N, seed: int = GOLD_SEED) -> pd.DataFrame:
    """A stratified draw of the open clauses (source × kind), each stratum at least
    :data:`GOLD_MIN_PER_STRATUM` or all it has, the rest in proportion; every stratum
    alternated into a ``calibration`` and a ``certification`` half.  Reproducible."""
    rng = np.random.default_rng(seed)
    strata = contexted.groupby(["source", "open_kind"])
    sizes = strata.size()
    take = np.minimum(sizes, np.maximum(GOLD_MIN_PER_STRATUM, (sizes / sizes.sum() * n).round().astype(int)))
    parts = []
    for (source, kind), group in strata:
        order = rng.permutation(len(group))[: int(take[(source, kind)])]
        drawn = group.iloc[np.sort(order)].copy()
        drawn["half"] = ["calibration" if i % 2 == 0 else "certification" for i in range(len(drawn))]
        parts.append(drawn)
    return pd.concat(parts, ignore_index=True)


def write_gold_sheet(sample: pd.DataFrame, path: Path = GOLD_SHEET) -> Path:

    Refuses to overwrite a sheet in which any ruling has been written.

        if (old[[c for c in GOLD_COLUMNS if c in old]] != "").any().any():
            raise FileExistsError(f"{path} already holds rulings; it is not redrawn")
    sheet = sample[["clause_key", "half", "source", "open_kind", "place", "ensemble", "context", "text",
                    "date_raw", "years_inside"]].rename(columns={"text": "clause"})
    for column in GOLD_COLUMNS:
        sheet[column] = ""
    path.parent.mkdir(parents=True, exist_ok=True)
    sheet.to_csv(path, index=False, encoding="utf-8-sig")

def fill_gold_sheet(rulings: list[dict], path: Path = GOLD_SHEET) -> dict:
    """Write the ruling page's documents (``rulings/<clause_key>``, as the page stores them)
    into the worksheet's ``gold_*`` columns.  Only documents marked ``ruled`` are written;
    a clause the page has not ruled keeps an empty row.  Every document must name a clause
    of the sheet.  Returns counts."""
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    index = {k: i for i, k in enumerate(sheet["clause_key"])}
    unknown = [d.get("clause_key") for d in rulings if d.get("clause_key") not in index]
    assert not unknown, f"rulings for clauses not in the sheet: {unknown[:5]}"

    def year(v) -> str:
        return "" if v is None or v == "" else str(int(v))

    written = 0
    for d in rulings:
        i = index[d["clause_key"]]
        if not d.get("ruled"):
            sheet.loc[i, GOLD_COLUMNS] = ""
            continue
        sheet.loc[i, GOLD_COLUMNS] = [
            "|".join(sorted(d.get("types") or [])), "true" if d.get("hedged") else "",
            d.get("date_form") or "", *(year(d.get(k)) for k in ("start_lo", "start_hi", "end_lo", "end_hi")),
            "|".join(d.get("titles") or []), d.get("performer_kind") or "",
            d.get("performer_name") or "", d.get("note") or ""]
        written += 1
    sheet.to_csv(path, index=False, encoding="utf-8-sig")
    return {"documents": len(rulings), "ruled": written, "sheet_rows": len(sheet)}


def read_gold(path: Path = GOLD_SHEET) -> pd.DataFrame:
    """The ruled rows of the worksheet, as readings (``reader = "gold"``), with their half.

    A row is ruled once ``gold_date_form`` is written (``leading`` for a dated clause).
    The author's entries are checked as a model's are; problems are reported, not fixed.
    """
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    ruled = sheet[sheet["gold_date_form"].str.strip() != ""]

    def year(v):
        v = str(v).strip()
        return int(v) if v else None

    rows = []
    for g in ruled.itertuples(index=False):
        types = sorted(t.strip() for t in re.split(r"[|,;]", g.gold_types) if t.strip())
        reading = {"types": types, "hedged": g.gold_hedged.strip().lower() in ("1", "true", "y", "yes", "ano"),
                   "date": {"form": g.gold_date_form.strip(), "start_lo": year(g.gold_start_lo),
                            "start_hi": year(g.gold_start_hi), "end_lo": year(g.gold_end_lo),
                            "end_hi": year(g.gold_end_hi)}}
        problems = reading_problems(reading, g.clause, undated=g.date_raw.strip() == "")
        rows.append({"clause_key": g.clause_key, "half": g.half, "reader": "gold", "types": "|".join(types),
                     "hedged": reading["hedged"], "date_form": reading["date"]["form"],
                     **{k: reading["date"][k] for k in ("start_lo", "start_hi", "end_lo", "end_hi")},
                     "titles": g.gold_titles.strip(), "performer_kind": g.gold_performer_kind.strip() or None,
                     "performer_name": g.gold_performer_name.strip() or None,
                     "valid": not problems, "problems": "; ".join(problems)})
    out = pd.DataFrame(rows)
    if len(out) and not out["valid"].all():
        print(f"gold: {int((~out['valid']).sum())} ruled rows have problems — see `problems`")
    return out


def _derived(coded_open: pd.DataFrame, readings: pd.DataFrame) -> pd.DataFrame:
    """The rules' flags for the gold clauses under one set of readings."""
    parts = []
    for source, group in coded_open.groupby("source"):
        applied = apply_readings(group, readings, source)
        applied["clause_key"] = clause_keys(applied, source).values
        parts.append(applied)
    return pd.concat(parts, ignore_index=True).set_index("clause_key")


DERIVED_FLAGS = ["czech_activity", "company", "performance", "hedged", "before_1914"]


def _interval(frame: pd.DataFrame) -> pd.DataFrame:
    """Date form and the four years, comparable across a CSV, a sheet and a reply."""
    out = pd.DataFrame({"date_form": frame["date_form"].astype(str)}, index=frame.index)
    for column in ("start_lo", "start_hi", "end_lo", "end_hi"):
        out[column] = pd.to_numeric(frame[column], errors="coerce").astype("Int64").astype(str)
    return out


def score(gold: pd.DataFrame, model: pd.DataFrame, coded_open: pd.DataFrame,
          half: str = "calibration") -> dict:
    """One reader against the author's rulings on one half.

    ``types`` — exact agreement on the set and, per type, precision and recall; ``flags`` —
    agreement on what the rulings make of each reading (Czech activity, company,
    performance, hedged, before 1914), which is what enters the panel; ``dates`` — exact
    agreement on the interval of the undated clauses.  The certification half is refused
    until :data:`PROMPT_FROZEN`.
    """
    if half == "certification" and not PROMPT_FROZEN:
        raise PermissionError("the certification half is sealed until PROMPT_FROZEN is True")
    g = gold[(gold["half"] == half) & gold["valid"].astype(bool)].set_index("clause_key")
    m = model[model["valid"].astype(bool)].set_index("clause_key").reindex(g.index)
    answered = m["types"].notna()
    gt = g["types"].map(lambda s: set(filter(None, s.split("|"))))
    mt = m["types"].fillna("").map(lambda s: set(filter(None, s.split("|"))))
    per_type = []
    for t in TYPES:
        tp = sum(t in a and t in b for a, b in zip(gt[answered], mt[answered]))
        fp = sum(t not in a and t in b for a, b in zip(gt[answered], mt[answered]))
        fn = sum(t in a and t not in b for a, b in zip(gt[answered], mt[answered]))
        if tp + fp + fn:
            per_type.append({"type": t, "gold": tp + fn, "model": tp + fp,
                             "precision": tp / (tp + fp) if tp + fp else np.nan,
                             "recall": tp / (tp + fn) if tp + fn else np.nan})
    rows = coded_open[coded_open["clause_key"].isin(g.index)]
    dg = _derived(rows, g.reset_index())
    dm = _derived(rows, m[answered].reset_index())
    flags = {f: float((dg[f].astype(str) == dm.loc[dg.index, f].astype(str)).mean()) for f in DERIVED_FLAGS}
    undated = rows.set_index("clause_key")["date_source"].eq("none")
    u = undated[undated].index.intersection(m[answered].index)
    same_date = (_interval(g.loc[u]) == _interval(m.loc[u])).all(axis=1)
    out = {"half": half, "ruled": len(g), "answered": int(answered.sum()),
           "types_exact": float((gt[answered] == mt[answered]).mean()) if answered.any() else np.nan,
           "flags": flags, "dates_exact": float(same_date.mean()) if len(u) else np.nan,
           "per_type": pd.DataFrame(per_type)}
    if GOLD_BAR is not None:
        out["passes"] = all(out["flags"].get(k, out.get(k, 0)) >= v for k, v in GOLD_BAR.items())
    return out
