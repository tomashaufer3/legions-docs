How it works

Two datasets are supported via the dataset argument:

  • "legion" — legionnaire records (csol.cz), 02_data_cleaning §1.
  • "habsburg" — fallen Habsburg soldiers (VÚA), 02_data_cleaning §2.

Design

  • *_RENAME maps the current column name (raw Czech or a parser's English output) to its final snake_case name. Only columns we keep need an entry; everything unmapped is dropped by the FINAL selection.
  • *_FINAL is the ordered list of final columns to keep. Boolean dummy families with a fixed, known code set (exit_*, cause_*) are enumerated explicitly; the data-driven political-affiliation dummies (pol_*) are appended dynamically by to_english.
  • The legionnaire theater (utils.parsing.theater) and the Habsburg theatre (utils.parsing.habsburg) are unified to the column name theater. The label sets differ — the Legion's service theatres (Russia / Italy / France / Serbia/Balkans / Britain/Dominions / Post-war (ČSA)) against the fronts a Habsburg soldier died on (Russia / Italy / Serbia / Macedonia / Romania / France / Rear) — so only Russia, Italy and France compare directly.

def to_english(df: pd.DataFrame, dataset: str) -> pd.DataFrame:
    """Rename to English and select the pruned FINAL schema.

    Call this **after** all field-specific parsers (and the small derived
    columns ``birth_year`` / ``exit_codes_str`` / ``death_year`` /
    ``age_at_death``) have run.  Political-affiliation dummies (``pol_*``) are
    kept automatically for the legionnaire dataset.

    Parameters
    ----------
    df:
        Parsed DataFrame (not modified in place).
    dataset:
        ``"legion"`` or ``"habsburg"``.

    Returns
    -------
    pd.DataFrame
        Copy with English column names, restricted to columns in FINAL that are
        actually present (plus ``pol_*`` for legion).  Missing FINAL columns are
        skipped silently — call :func:`schema_report` to audit them.
    """
    rename, final = _config(dataset)
    out = df.rename(columns=rename)

    # A raw/parsed column and a derived column can map to the same final name
    # (e.g. the frozenset ``exit_codes`` and the serialisable ``exit_codes_str``
    # both → ``exit_codes``).  Keep the last occurrence — the derived,
    # parquet-friendly one is appended later in the pipeline.
    if out.columns.duplicated().any():
        out = out.loc[:, ~out.columns.duplicated(keep="last")]

    extras: list[str] = []
    if dataset == "legion":
        extras = sorted(c for c in out.columns if c.startswith("pol_"))

    keep = [c for c in (final + extras) if c in out.columns]
    return out[keep].copy()

def schema_report(df: pd.DataFrame, dataset: str) -> None:
    """Print the final schema and flag any FINAL columns missing from *df*.

    *df* may be either the pre-rename parsed frame or the post-:func:`to_english`
    frame; columns are checked after applying the rename map.
    """
    rename, final = _config(dataset)
    out = df.rename(columns=rename)
    present = [c for c in final if c in out.columns]
    missing = [c for c in final if c not in out.columns]
    pol = sorted(c for c in out.columns if c.startswith("pol_")) if dataset == "legion" else []

    print(f"Schema report [{dataset}]  (N = {len(df):,})\n")
    print(f"  FINAL columns present : {len(present)}/{len(final)}")
    if dataset == "legion":
        print(f"  pol_* dummies kept    : {len(pol)}")
    if missing:
        print(f"\n  [!] MISSING FINAL columns ({len(missing)}) - a parser may not have run:")
        for c in missing:
            print(f"      - {c}")
    else:
        print("  [OK] all FINAL columns present")

def write_data_dictionary(path, datasets: tuple[str, ...] = ("legion", "habsburg")) -> pd.DataFrame:
    """Write a CSV data dictionary mapping final → source columns for *datasets*.

    Columns: ``dataset, final_name, source_name, description``.  The reverse of
    each ``*_RENAME`` map is used to recover the source (raw Czech or parser)
    column behind every final name.  Returns the dictionary DataFrame.
    """
    rows: list[dict[str, str]] = []
    for ds in datasets:
        rename, final = _config(ds)
        # final → first source name mapping (several sources can map to one final)
        reverse: dict[str, str] = {}
        for src, dst in rename.items():
            reverse.setdefault(dst, src)
        dynamic_note = {"legion": "pol_* — political-affiliation dummies, one per canonical in affiliations.PARTY_ORDER, appended dynamically by to_english(). Boolean with <NA> wherever pol_recorded is False. Two members are NOT dummies: pol_recorded (was the field filled in at all) and pol_atlas_codes (a string audit column carrying the Akademicky atlas plate IV.37 register codes the affiliation string is evidence for) — affiliations.py 2.1.0"}
        for col in final:
            rows.append({
                "dataset": ds,
                "final_name": col,
                "source_name": reverse.get(col, col),
                "description": _DESCRIPTIONS.get(col, ""),
            })
        if ds in dynamic_note:
            rows.append({"dataset": ds, "final_name": "pol_*",
                         "source_name": "Politická příslušnost",
                         "description": dynamic_note[ds]})
        for col in _POST_LINKAGE:
            rows.append({"dataset": ds, "final_name": col,
                         "source_name": "utils.linkage (notebook 02 §4)",
                         "description": _DESCRIPTIONS.get(col, "")})
        # derived geographic standardisation columns (added in §4, not part of to_english FINAL);
        # skip any that a dataset already carries in FINAL (e.g. habsburg birth_land from locations)
        for prefix, role in _GEO_PREFIXES.items():
            for suffix, desc in _GEO_DERIVED.items():
                name = f"{prefix}{suffix}"
                if name in final:
                    continue
                rows.append({"dataset": ds, "final_name": name,
                             "source_name": f"{prefix}_municipality / {prefix}_okres (derived)",
                             "description": desc.format(role=role)})
            for suffix, desc in _GEO_HU_DERIVED.items():
                rows.append({"dataset": ds, "final_name": f"{prefix}{suffix}",
                             "source_name": f"{prefix}_okres (derived, Transleithanian)",
                             "description": desc.format(role=role)})

    dd = pd.DataFrame(rows, columns=["dataset", "final_name", "source_name", "description"])
    from pathlib import Path
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    dd.to_csv(p, index=False, encoding="utf-8")
    return dd

"""
schema.py — Canonical English schema for the two final datasets
================================================================
Single source of truth for column naming.  The cleaning notebook
(``02_data_cleaning``, §1 legion and §2 habsburg) and the eventual master notebook call
:func:`to_english` as the **last** step of their cleaning pipeline to turn a
mixed Czech-raw / English-parsed DataFrame into a pruned, fully-English,
analysis-ready table written to ``data/final/``.

Two datasets are supported via the ``dataset`` argument:

* ``"legion"``   — legionnaire records (csol.cz), ``02_data_cleaning`` §1.
* ``"habsburg"`` — fallen Habsburg soldiers (VÚA), ``02_data_cleaning`` §2.

Design
------
* ``*_RENAME`` maps the **current** column name (raw Czech *or* a parser's
  English output) to its final snake_case name.  Only columns we keep need an
  entry; everything unmapped is dropped by the FINAL selection.
* ``*_FINAL`` is the ordered list of final columns to keep.  Boolean dummy
  families with a fixed, known code set (``exit_*``, ``cause_*``) are
  enumerated explicitly; the data-driven political-affiliation dummies
  (``pol_*``) are appended dynamically by :func:`to_english`.
* The legionnaire ``theater`` (:mod:`utils.parsing.theater`) and the Habsburg
  ``theatre`` (:mod:`utils.parsing.habsburg`) are unified to the column name
  ``theater``. **The label sets differ** — the Legion's service theatres (Russia /
  Italy / France / Serbia/Balkans / Britain/Dominions / Post-war (ČSA)) against the
  fronts a Habsburg soldier died on (Russia / Italy / Serbia / Macedonia / Romania /
  France / Rear) — so only Russia, Italy and France compare directly.

Usage
-----
    from utils.parsing.schema import to_english, schema_report, write_data_dictionary
    df_final = to_english(df, dataset="legion")
    schema_report(df, dataset="legion")          # flags any missing FINAL cols
    df_final.to_parquet("data/final/legionnaires_clean.parquet", index=False)
    write_data_dictionary("data/final/data_dictionary.csv")
"""

def _config(dataset: str) -> tuple[dict[str, str], list[str]]:
    if dataset == "legion":
        return LEGION_RENAME, LEGION_FINAL
    if dataset == "habsburg":
        return HABSBURG_RENAME, HABSBURG_FINAL
    raise ValueError(f"dataset must be 'legion' or 'habsburg', got {dataset!r}")


def to_english(df: pd.DataFrame, dataset: str) -> pd.DataFrame:
    """Rename to English and select the pruned FINAL schema.

    Call this **after** all field-specific parsers (and the small derived
    columns ``birth_year`` / ``exit_codes_str`` / ``death_year`` /
    ``age_at_death``) have run.  Political-affiliation dummies (``pol_*``) are
    kept automatically for the legionnaire dataset.

    Parameters
    ----------
    df:
        Parsed DataFrame (not modified in place).
    dataset:
        ``"legion"`` or ``"habsburg"``.

    Returns
    -------
    pd.DataFrame
        Copy with English column names, restricted to columns in FINAL that are
        actually present (plus ``pol_*`` for legion).  Missing FINAL columns are
        skipped silently — call :func:`schema_report` to audit them.
    """
    rename, final = _config(dataset)
    out = df.rename(columns=rename)

    # A raw/parsed column and a derived column can map to the same final name
    # (e.g. the frozenset ``exit_codes`` and the serialisable ``exit_codes_str``
    # both → ``exit_codes``).  Keep the last occurrence — the derived,
    # parquet-friendly one is appended later in the pipeline.
    if out.columns.duplicated().any():
        out = out.loc[:, ~out.columns.duplicated(keep="last")]

    extras: list[str] = []
    if dataset == "legion":
        extras = sorted(c for c in out.columns if c.startswith("pol_"))

    keep = [c for c in (final + extras) if c in out.columns]
    return out[keep].copy()


def schema_report(df: pd.DataFrame, dataset: str) -> None:
    """Print the final schema and flag any FINAL columns missing from *df*.

    *df* may be either the pre-rename parsed frame or the post-:func:`to_english`
    frame; columns are checked after applying the rename map.
    """
    rename, final = _config(dataset)
    out = df.rename(columns=rename)
    present = [c for c in final if c in out.columns]
    missing = [c for c in final if c not in out.columns]
    pol = sorted(c for c in out.columns if c.startswith("pol_")) if dataset == "legion" else []

    print(f"Schema report [{dataset}]  (N = {len(df):,})\n")
    print(f"  FINAL columns present : {len(present)}/{len(final)}")
    if dataset == "legion":
        print(f"  pol_* dummies kept    : {len(pol)}")
    if missing:
        print(f"\n  [!] MISSING FINAL columns ({len(missing)}) - a parser may not have run:")
        for c in missing:
            print(f"      - {c}")
    else:
        print("  [OK] all FINAL columns present")


def write_data_dictionary(path, datasets: tuple[str, ...] = ("legion", "habsburg")) -> pd.DataFrame:
    """Write a CSV data dictionary mapping final → source columns for *datasets*.

    Columns: ``dataset, final_name, source_name, description``.  The reverse of
    each ``*_RENAME`` map is used to recover the source (raw Czech or parser)
    column behind every final name.  Returns the dictionary DataFrame.
    """
    rows: list[dict[str, str]] = []
    for ds in datasets:
        rename, final = _config(ds)
        # final → first source name mapping (several sources can map to one final)
        reverse: dict[str, str] = {}
        for src, dst in rename.items():
            reverse.setdefault(dst, src)
        dynamic_note = {"legion": "pol_* — political-affiliation dummies, one per canonical in affiliations.PARTY_ORDER, appended dynamically by to_english(). Boolean with <NA> wherever pol_recorded is False. Two members are NOT dummies: pol_recorded (was the field filled in at all) and pol_atlas_codes (a string audit column carrying the Akademicky atlas plate IV.37 register codes the affiliation string is evidence for) — affiliations.py 2.1.0"}
        for col in final:
            rows.append({
                "dataset": ds,
                "final_name": col,
                "source_name": reverse.get(col, col),
                "description": _DESCRIPTIONS.get(col, ""),
            })
        if ds in dynamic_note:
            rows.append({"dataset": ds, "final_name": "pol_*",
                         "source_name": "Politická příslušnost",
                         "description": dynamic_note[ds]})
        for col in _POST_LINKAGE:
            rows.append({"dataset": ds, "final_name": col,
                         "source_name": "utils.linkage (notebook 02 §4)",
                         "description": _DESCRIPTIONS.get(col, "")})
        # derived geographic standardisation columns (added in §4, not part of to_english FINAL);
        # skip any that a dataset already carries in FINAL (e.g. habsburg birth_land from locations)
        for prefix, role in _GEO_PREFIXES.items():
            for suffix, desc in _GEO_DERIVED.items():
                name = f"{prefix}{suffix}"
                if name in final:
                    continue
                rows.append({"dataset": ds, "final_name": name,
                             "source_name": f"{prefix}_municipality / {prefix}_okres (derived)",
                             "description": desc.format(role=role)})
            for suffix, desc in _GEO_HU_DERIVED.items():
                rows.append({"dataset": ds, "final_name": f"{prefix}{suffix}",
                             "source_name": f"{prefix}_okres (derived, Transleithanian)",
                             "description": desc.format(role=role)})

    dd = pd.DataFrame(rows, columns=["dataset", "final_name", "source_name", "description"])
    from pathlib import Path
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    dd.to_csv(p, index=False, encoding="utf-8")
    return dd
