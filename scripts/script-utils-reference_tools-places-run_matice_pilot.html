How it works

Offline: reads the page caches utils/scraping/run_matice_downloads.py left in library/associations/ (and the 1878 Internet Archive PDF beside them), the 1910 place tables in data/final/geo/ and the pooled okres census, and writes nothing but data/interim/matice/:

matice_entries.csv One row per printed entry (key entry_id), every field beside the entry's own text, with entry_year_from_no (a deceased founder's entry year read off the membership numbers), number_order_outlier, the parse flags, the 1910 settlement and okres its residence names (utils.parsing.locative, with place_status and place_how), and the member's occupation_class and member_group (posted / local / student / corporate / unknown). matice_totals.csv Entries found against the totals each society printed. matice_okres.csv The measure: one row per census okres (key census_okres_id, zeros kept), member counts, rates per 1,000 inhabitants and per 1,000 Czech speakers, and empirical-Bayes shrunk rates (matice.okres_measure). matice_court.csv The same per court district (key census_court_id, matice.court_measure), with the court census of 1910 as denominator. matice_payments.csv One row per payment in the Matice česká monthly lists of the Časopis Musea (key payment_id; run_matice_downloads.py --set payments), placed and classed like the entries. matice_payment_months.csv One row per month as printed (key block_id), its parsed sum against its Úhrnem. matice_payers.csv One row per paying person (key payer_id, matice.payment_persons): first year, sum paid, whether it reached the founder's fee. A joiner (D-184) is a payer whose first payment falls in matice.JOINER_WINDOW.

and prints what it found. Nothing reaches data/final.

Usage::

python utils/reference_tools/places/run_matice_pilot.py
python utils/reference_tools/places/run_matice_pilot.py --only mc1890 mm1909

def add_places(entries: pd.DataFrame, index: dict, readings: bool = True) -> pd.DataFrame:
    """The 1910 settlement and okres of every residence (``locative.match_all``), then the
    residences re-read from the scans (``READINGS``) where the OCR's could not be placed."""
    places = lc.match_all(entries["residence"].where(entries["residence"].astype(bool)), index)
    out = pd.concat([entries, places], axis=1)
    if readings:
        out = apply_readings(out, index)
    out["okres_id"] = out["okres_id"].astype("Int64")
    return out

def apply_readings(entries: pd.DataFrame, index: dict) -> pd.DataFrame:
    """Place an unmatched entry by its re-read residence. A reading counts only if the
    matcher then places it (a settlement, Prague, abroad), or if it says the entry
    names no place (``-``); otherwise the entry stays unmatched, the reading kept in
    ``residence_read`` for a look by eye. ``place_how`` starts ``read:`` where it counted."""
    entries["residence_read"] = ""
    if not READINGS.exists():
        return entries
    read = pd.read_csv(READINGS, dtype=str, keep_default_na=False).set_index("entry_id")
    hit = entries["entry_id"].isin(read.index) & (entries["place_status"] == "unmatched")
    entries.loc[hit, "residence_read"] = entries.loc[hit, "entry_id"].map(read["residence_read"])
    cols = ["settlement_id", "okres_id", "place_status", "place_how", "n_candidates", "place_key"]
    redo = entries[hit & ~entries["residence_read"].isin(["", "?"])]
    none = redo["residence_read"] == "-"
    entries.loc[none[none].index, ["place_status", "place_how"]] = ["none", "read:no_place"]
    redo = redo[~none]
    again = lc.match_all(redo["residence_read"], index)
    ok = again["place_status"].isin(_ACCEPT)
    again.loc[ok, "place_how"] = "read:" + again.loc[ok, "place_how"].astype(str)
    entries.loc[again[ok].index, cols] = again.loc[ok, cols].to_numpy()
    return entries

def theatre_rate(measure: pd.DataFrame, years: tuple[int, int]) -> pd.Series:
    """Czech theatre companies active per year (certain), averaged over *years*, per 1,000
    inhabitants, per census okres of *measure*."""
    lo, hi = years
    t = pd.read_parquet(THEATRE, columns=["okres_id", "year", "ensembles_active_certain"])
    xw = pd.read_csv(GEO / "okres_census_crosswalk.csv")
    t = t[t["year"].between(lo, hi)].merge(xw[["okres_id", "census_okres_id"]], on="okres_id",
                                           how="left", validate="m:1")
    act = t.groupby("census_okres_id")["ensembles_active_certain"].sum() / (hi - lo + 1)
    return 1000 * measure["census_okres_id"].map(act).fillna(0) / measure["pop_total"]

def theatre_check(measure: pd.DataFrame) -> pd.DataFrame:
    """Spearman correlation of each Matice rate with Czech theatre companies active per
    1,000 inhabitants (certain) in that measure's own years (``THEATRE_YEARS``), across
    okresy — whole sample and the okresy at least half Czech-speaking."""
    czech = measure["lang_czech_share"] >= 0.5
    frames = {"all okresy": pd.Series(True, index=measure.index), "Czech share >= 0.5": czech,
              "Böhmen, Czech >= 0.5": czech & (measure["land"] == "Böhmen"),
              "Mähren, Czech >= 0.5": czech & (measure["land"] == "Mähren")}
    rows = {}
    for col, years in THEATRE_YEARS.items():
        theatre = theatre_rate(measure, years)
        rows[col] = {label: measure.loc[keep, col].corr(theatre[keep], method="spearman")
                     for label, keep in frames.items()}
    return pd.DataFrame(rows).T

def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    p.add_argument("--only", nargs="+", choices=list(mt.SOURCES), default=None)
    args = p.parse_args(argv)

    entries = mt.parse_all(LIBRARY, args.only)
    if entries.empty:
        print(f"No page caches found in {LIBRARY}; run utils/scraping/run_matice_downloads.py.")
        return 1
    assert entries["entry_id"].is_unique and entries["entry_id"].notna().all()
    entries["number_order_outlier"] = mt.mc_number_order(entries)
    entries["entry_year_from_no"] = mt.entry_year_from_number(entries)
    n = len(entries)
    index = place_index()
    entries = add_occupations(add_places(entries, index))
    assert len(entries) == n, "place matching changed the row count"
    totals = mt.totals_report(entries)

    if not PAYMENT_MANIFEST.exists():
        print(f"No {PAYMENT_MANIFEST.name}; run utils/scraping/run_matice_downloads.py "
              "--discover-payments, then --set payments.")
        return 1
    payments, months = mt.parse_payments(LIBRARY, pd.read_csv(PAYMENT_MANIFEST, dtype=str))
    assert payments["payment_id"].is_unique and months["block_id"].is_unique
    n = len(payments)
    payments = add_occupations(mt.borrow_places(add_places(payments, index, readings=False)))
    assert len(payments) == n, "place matching changed the row count"
    payers = mt.payment_persons(payments)
    assert payers["payer_id"].is_unique

    measure = mt.okres_measure(entries, pd.read_parquet(CENSUS),
                               pd.read_csv(GEO / "okres_census_crosswalk.csv"), payers)
    assert measure["census_okres_id"].is_unique
    courts = court_level(entries, payers)
    assert courts["census_court_id"].is_unique

    OUT.mkdir(parents=True, exist_ok=True)
    written = {"matice_entries.csv": entries, "matice_totals.csv": totals,
               "matice_okres.csv": measure, "matice_court.csv": courts,
               "matice_payments.csv": payments, "matice_payment_months.csv": months,
               "matice_payers.csv": payers}
    for name, df in written.items():
        df.to_csv(OUT / name, index=False, encoding="utf-8")
    report(entries, totals, measure)
    payment_report(payments, months, payers, entries)
    print(f"\nCourt districts: {len(courts)}; placed in one: "
          f"{int(courts[mt.MC_OWN].sum()):,} Matice česká joiners, "
          f"{int(courts['mc_founders'].sum()):,} founders 1831–1890, "
          f"{int(courts[mt.MM_OWN].sum()):,} Matice moravská 1909")
    print(f"\nwrote {OUT.relative_to(ROOT)}/" + ", ".join(written))
    return 0

def payment_report(payments: pd.DataFrame, months: pd.DataFrame, payers: pd.DataFrame,
                   entries: pd.DataFrame) -> None:
    """What the payment lists gave, and the two checks on them: each month against its
    printed total, and the payers who completed in 1888–1890 against the founder register
    of the end of 1890 (*Jména p.p. zakladatelův*), which lists everyone who had paid in full."""
    print(f"\nMatice česká payments: {len(payments):,} in {len(months):,} months, "
          f"{months['period_year'].min()}–{months['period_year'].max()}")
    read = months["printed_k"].notna()
    print(f"Months against their printed total: {int(months['agrees'].sum()):,} of "
          f"{int(read.sum()):,} with a total read agree within 0.5 K "
          f"({months.loc[read, 'agrees'].mean():.1%}); parsed {months['sum_k'].sum():,.0f} K "
          f"against {months['printed_k'].sum():,.0f} K printed")
    print("Months by year (count, agreeing):")
    print(months.groupby("period_year").agg(months=("chain", "size"), agree=("agrees", "sum"),
                                            payments=("entries", "sum")).T.to_string())
    print("\nPayment flags:")
    print(payments["flags"].replace("", "none").str.split(";").explode().value_counts().to_string())
    print("\nResidences (share of payments):")
    print(payments["place_status"].value_counts(normalize=True).round(3).to_string())
    print(f"of them placed by the same payer's other payments (borrowed): "
          f"{int((payments['place_how'] == 'borrowed:payer').sum()):,}")
    print(f"\nPayers (persons): {len(payers):,}; reached the founder's fee: "
          f"{int(payers['founder'].sum()):,}")
    by = payers.groupby("first_year").agg(payers=("payer_id", "size"), founders=("founder", "sum"))
    print(by.T.to_string())
    lo, hi = mt.JOINER_WINDOW
    print(f"\nJoiners {lo}–{hi} (first payment in the window): {len(mt.joiners(payers)):,}; "
          f"of them paid the fee within the series: "
          f"{len(mt.joiners(payers, founders_only=True)):,}")
    print(mt.joiners(payers)["member_group"].value_counts(normalize=True).round(3).to_string())
    register_check(payers, entries)

def register_check(payers: pd.DataFrame, entries: pd.DataFrame) -> None:
    """Share of the payers who reached the fee in 1888–1890 who are in the 1890 founder
    register with an entry year of 1888–1890, matched on folded surname and settlement."""
    done = payers[payers["founder"] & payers["founder_year"].between(1888, 1890)
                  & payers["settlement_id"].notna()]
    reg = entries[(entries["source"] == "mc1890") & (entries["list_kind"] == "founder")
                  & entries["settlement_id"].notna()]
    key = set(zip(reg["surname"].map(mt._fold), reg["settlement_id"].astype(str)))
    late = reg[pd.to_numeric(reg["entry_year"], errors="coerce").between(1886, 1891)]
    key_late = set(zip(late["surname"].map(mt._fold), late["settlement_id"].astype(str)))
    hit = [(mt._fold(s), str(p)) in key for s, p in zip(done["surname"], done["settlement_id"])]
    hit_late = [(mt._fold(s), str(p)) in key_late
                for s, p in zip(done["surname"], done["settlement_id"])]
    if len(done):
        print(f"\nPayers paid in full 1888–1890 and placed: {len(done):,}; in the 1890 founder "
              f"register by surname + settlement: {sum(hit) / len(done):.1%} (with an entry "
              f"year 1886–1891: {sum(hit_late) / len(done):.1%})")

#!/usr/bin/env python3
"""
Pilot — the Matice česká and Matice moravská member registers, read into tables and
summed to a measure per 1910 okres.

Offline: reads the page caches ``utils/scraping/run_matice_downloads.py`` left in
``library/associations/`` (and the 1878 Internet Archive PDF beside them), the 1910
place tables in ``data/final/geo/`` and the pooled okres census, and writes nothing but
``data/interim/matice/``:

``matice_entries.csv``
    One row per printed entry (key ``entry_id``), every field beside the entry's own
    text, with ``entry_year_from_no`` (a deceased founder's entry year read off the
    membership numbers), ``number_order_outlier``, the parse ``flags``, the 1910
    settlement and okres its residence names (``utils.parsing.locative``, with
    ``place_status`` and ``place_how``), and the member's ``occupation_class`` and
    ``member_group`` (posted / local / student / corporate / unknown).
``matice_totals.csv``
    Entries found against the totals each society printed.
``matice_okres.csv``
    The measure: one row per census okres (key ``census_okres_id``, zeros kept), member
    counts, rates per 1,000 inhabitants and per 1,000 Czech speakers, and
    empirical-Bayes shrunk rates (``matice.okres_measure``).
``matice_court.csv``
    The same per court district (key ``census_court_id``, ``matice.court_measure``),
    with the court census of 1910 as denominator.
``matice_payments.csv``
    One row per payment in the Matice česká monthly lists of the *Časopis Musea*
    (key ``payment_id``; ``run_matice_downloads.py --set payments``), placed and classed
    like the entries.
``matice_payment_months.csv``
    One row per month as printed (key ``block_id``), its parsed sum against its *Úhrnem*.
``matice_payers.csv``
    One row per paying person (key ``payer_id``, ``matice.payment_persons``): first year,
    sum paid, whether it reached the founder's fee. A joiner (D-184) is a payer whose
    first payment falls in ``matice.JOINER_WINDOW``.

and prints what it found. Nothing reaches ``data/final``.

Usage::

    python utils/reference_tools/places/run_matice_pilot.py
    python utils/reference_tools/places/run_matice_pilot.py --only mc1890 mm1909
"""

from __future__ import annotations

import sys
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

from utils.parsing import census                                        # noqa: E402
from utils.parsing import locative as lc                                # noqa: E402
from utils.parsing import matice as mt                                  # noqa: E402

LIBRARY = ROOT / "library" / "associations"
GEO = ROOT / "data" / "final" / "geo"
CENSUS = ROOT / "data" / "final" / "census" / "census_okres_1910_pooled.parquet"
THEATRE = ROOT / "data" / "final" / "theatre" / "theatre_okres_year.parquet"   # notebook 02 §3f (D-255)
#: The 1910 census by court district: population and Czech speakers (Umgangssprache).
COURT_CENSUS = (ROOT / "data" / "raw" / "census"
                / "Volkszaehlung Religion Umgangssprache 1910 District courts 934.xlsx")
OUT = ROOT / "data" / "interim" / "matice"
#: Where each payment list sits in the Časopis Musea (``run_matice_downloads.py --discover-payments``).
PAYMENT_MANIFEST = ROOT / "reference" / "places" / mt.PAYMENT_MANIFEST


def place_index() -> dict:
    return lc.build_index(pd.read_parquet(GEO / "settlements_1910.parquet"),
                          pd.read_csv(GEO / "settlement_aliases_1910.csv"),
                          pd.read_parquet(GEO / "districts_1910.parquet"))


def add_places(entries: pd.DataFrame, index: dict, readings: bool = True) -> pd.DataFrame:
    """The 1910 settlement and okres of every residence (``locative.match_all``), then the
    residences re-read from the scans (``READINGS``) where the OCR's could not be placed."""
    places = lc.match_all(entries["residence"].where(entries["residence"].astype(bool)), index)
    out = pd.concat([entries, places], axis=1)
    if readings:
        out = apply_readings(out, index)
    out["okres_id"] = out["okres_id"].astype("Int64")
    return out


#: Residences re-read by eye or by a vision model from the scan where the OCR's reading
#: could not be placed (``build_matice_read_sheets.py`` cuts the lines).
READINGS = ROOT / "reference" / "places" / "matice_residence_readings.csv"
_ACCEPT = {"matched", "prague", "outside", "land_only"}


def apply_readings(entries: pd.DataFrame, index: dict) -> pd.DataFrame:
    """Place an unmatched entry by its re-read residence. A reading counts only if the
    matcher then places it (a settlement, Prague, abroad), or if it says the entry
    names no place (``-``); otherwise the entry stays unmatched, the reading kept in
    ``residence_read`` for a look by eye. ``place_how`` starts ``read:`` where it counted."""
    entries["residence_read"] = ""
    if not READINGS.exists():
        return entries
    read = pd.read_csv(READINGS, dtype=str, keep_default_na=False).set_index("entry_id")
    hit = entries["entry_id"].isin(read.index) & (entries["place_status"] == "unmatched")
    entries.loc[hit, "residence_read"] = entries.loc[hit, "entry_id"].map(read["residence_read"])
    cols = ["settlement_id", "okres_id", "place_status", "place_how", "n_candidates", "place_key"]
    redo = entries[hit & ~entries["residence_read"].isin(["", "?"])]
    none = redo["residence_read"] == "-"
    entries.loc[none[none].index, ["place_status", "place_how"]] = ["none", "read:no_place"]
    redo = redo[~none]
    again = lc.match_all(redo["residence_read"], index)
    ok = again["place_status"].isin(_ACCEPT)
    again.loc[ok, "place_how"] = "read:" + again.loc[ok, "place_how"].astype(str)
    entries.loc[again[ok].index, cols] = again.loc[ok, cols].to_numpy()
    return entries


def add_occupations(entries: pd.DataFrame) -> pd.DataFrame:
    entries["occupation_class"] = [
        mt.occupation_class(o, h, bool(c)) for o, h, c in
        zip(entries["occupation"], entries["honorific"], entries["is_corporate"])]
    entries["member_group"] = entries["occupation_class"].map(mt.member_group)
    return entries


def report(entries: pd.DataFrame, totals: pd.DataFrame, measure: pd.DataFrame) -> None:
    print(f"\nmatice {mt.__version__} / locative {lc.__version__}: {len(entries):,} entries "
          f"from {entries['source'].nunique()} sources\n")
    print(entries.groupby(["source", "list_kind"]).size().rename("entries").to_string())
    print("\nAgainst the printed totals:")
    print(totals.to_string(index=False))
    mc = entries[entries["society"] == "MC"]
    numbered = mc[mc["member_no"].notna()]
    who = mt.founder_key(numbered)
    both = (set(who[numbered["source"] == "mc1873"]) & set(who[numbered["source"] == "mc1890"]))
    print(f"\nMatice česká founders in both issues (number and surname, mt.founder_key): {len(both):,}")
    print(f"Numbers whose entry year disagrees with their neighbours': "
          f"{int(entries['number_order_outlier'].sum()):,}")
    print("\nResidences (share of entries):")
    print(entries.groupby("society")["place_status"].value_counts(normalize=True)
          .round(3).unstack(fill_value=0).to_string())
    how = entries.loc[entries["place_status"] == "matched", "place_how"].str.split(":").str[-1]
    print("\nHow a settlement was chosen (matched entries):")
    print(how.value_counts().to_string())
    print("\nMember groups (persons, not corporate):")
    print(entries[entries["member_group"] != "corporate"].groupby("society")["member_group"]
          .value_counts(normalize=True).round(3).unstack(fill_value=0).to_string())
    cols = [mt.MC_OWN, "mc_founders", "mc_founders_local", "mm_members_1909"]
    print(f"\nThe measure: {len(measure)} okresy; members placed in an okres:")
    print(measure.groupby("land")[cols + [f"{mt.MC_OWN}_paid", "mc_joiners_1891_1913",
                                          "mc_corporate", "mm_members_1890"]].sum().to_string())
    bohemia = measure[measure["land"] == "Böhmen"]
    print(f"\nBohemian okresy with no joiner {mt.JOINER_WINDOW}: "
          f"{int((bohemia[mt.MC_OWN] == 0).sum())} of {len(bohemia)}; with no founder "
          f"1831–1890: {int((bohemia['mc_founders'] == 0).sum())}")
    top = measure.sort_values(f"{mt.MC_OWN}_per_1000_eb", ascending=False)
    print(f"\nHighest joiner rate per 1,000 (shrunk), 12 okresy:")
    print(top[["okres_cs", "land", "pop_total", "lang_czech_share", mt.MC_OWN,
               f"{mt.MC_OWN}_per_1000", f"{mt.MC_OWN}_per_1000_eb"]].head(12)
          .round(3).to_string(index=False))
    rates = [f"{c}_per_1000_eb" for c in cols]
    print("\nSpearman correlations across okresy (shrunk rates, Czech share):")
    print(measure[rates + ["lang_czech_share"]].corr(method="spearman").round(2).to_string())
    if THEATRE.exists():
        print("\nAgainst the theatre measure, each in its own years (THEATRE_YEARS):")
        print(theatre_check(measure).round(2).to_string())


#: The theatre years each Matice measure is set against: its own. The joiners and the 1909
#: roster stand at about 1909; the founders of 1831–1890 and the 1890 roster at 1890.
THEATRE_YEARS = {
    f"{mt.MC_OWN}_per_1000_eb": mt.JOINER_WINDOW, f"{mt.MC_OWN}_paid_per_1000_eb": mt.JOINER_WINDOW,
    "mc_founders_per_1000_eb": (1880, 1890), "mc_founders_local_per_1000_eb": (1880, 1890),
    "mm_members_1890_per_1000_eb": (1880, 1890), "mm_members_1909_per_1000_eb": mt.JOINER_WINDOW,
    "matice_own_z": mt.JOINER_WINDOW,
}


def theatre_rate(measure: pd.DataFrame, years: tuple[int, int]) -> pd.Series:
    """Czech theatre companies active per year (certain), averaged over *years*, per 1,000
    inhabitants, per census okres of *measure*."""
    lo, hi = years
    t = pd.read_parquet(THEATRE, columns=["okres_id", "year", "ensembles_active_certain"])
    xw = pd.read_csv(GEO / "okres_census_crosswalk.csv")
    t = t[t["year"].between(lo, hi)].merge(xw[["okres_id", "census_okres_id"]], on="okres_id",
                                           how="left", validate="m:1")
    act = t.groupby("census_okres_id")["ensembles_active_certain"].sum() / (hi - lo + 1)
    return 1000 * measure["census_okres_id"].map(act).fillna(0) / measure["pop_total"]


def theatre_check(measure: pd.DataFrame) -> pd.DataFrame:
    """Spearman correlation of each Matice rate with Czech theatre companies active per
    1,000 inhabitants (certain) in that measure's own years (``THEATRE_YEARS``), across
    okresy — whole sample and the okresy at least half Czech-speaking."""
    czech = measure["lang_czech_share"] >= 0.5
    frames = {"all okresy": pd.Series(True, index=measure.index), "Czech share >= 0.5": czech,
              "Böhmen, Czech >= 0.5": czech & (measure["land"] == "Böhmen"),
              "Mähren, Czech >= 0.5": czech & (measure["land"] == "Mähren")}
    rows = {}
    for col, years in THEATRE_YEARS.items():
        theatre = theatre_rate(measure, years)
        rows[col] = {label: measure.loc[keep, col].corr(theatre[keep], method="spearman")
                     for label, keep in frames.items()}
    return pd.DataFrame(rows).T


def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    p.add_argument("--only", nargs="+", choices=list(mt.SOURCES), default=None)
    args = p.parse_args(argv)

    entries = mt.parse_all(LIBRARY, args.only)
    if entries.empty:
        print(f"No page caches found in {LIBRARY}; run utils/scraping/run_matice_downloads.py.")
        return 1
    assert entries["entry_id"].is_unique and entries["entry_id"].notna().all()
    entries["number_order_outlier"] = mt.mc_number_order(entries)
    entries["entry_year_from_no"] = mt.entry_year_from_number(entries)
    n = len(entries)
    index = place_index()
    entries = add_occupations(add_places(entries, index))
    assert len(entries) == n, "place matching changed the row count"
    totals = mt.totals_report(entries)

    if not PAYMENT_MANIFEST.exists():
        print(f"No {PAYMENT_MANIFEST.name}; run utils/scraping/run_matice_downloads.py "
              "--discover-payments, then --set payments.")
        return 1
    payments, months = mt.parse_payments(LIBRARY, pd.read_csv(PAYMENT_MANIFEST, dtype=str))
    assert payments["payment_id"].is_unique and months["block_id"].is_unique
    n = len(payments)
    payments = add_occupations(mt.borrow_places(add_places(payments, index, readings=False)))
    assert len(payments) == n, "place matching changed the row count"
    payers = mt.payment_persons(payments)
    assert payers["payer_id"].is_unique

    measure = mt.okres_measure(entries, pd.read_parquet(CENSUS),
                               pd.read_csv(GEO / "okres_census_crosswalk.csv"), payers)
    assert measure["census_okres_id"].is_unique
    courts = court_level(entries, payers)
    assert courts["census_court_id"].is_unique

    OUT.mkdir(parents=True, exist_ok=True)
    written = {"matice_entries.csv": entries, "matice_totals.csv": totals,
               "matice_okres.csv": measure, "matice_court.csv": courts,
               "matice_payments.csv": payments, "matice_payment_months.csv": months,
               "matice_payers.csv": payers}
    for name, df in written.items():
        df.to_csv(OUT / name, index=False, encoding="utf-8")
    report(entries, totals, measure)
    payment_report(payments, months, payers, entries)
    print(f"\nCourt districts: {len(courts)}; placed in one: "
          f"{int(courts[mt.MC_OWN].sum()):,} Matice česká joiners, "
          f"{int(courts['mc_founders'].sum()):,} founders 1831–1890, "
          f"{int(courts[mt.MM_OWN].sum()):,} Matice moravská 1909")
    print(f"\nwrote {OUT.relative_to(ROOT)}/" + ", ".join(written))
    return 0


def payment_report(payments: pd.DataFrame, months: pd.DataFrame, payers: pd.DataFrame,
                   entries: pd.DataFrame) -> None:
    """What the payment lists gave, and the two checks on them: each month against its
    printed total, and the payers who completed in 1888–1890 against the founder register
    of the end of 1890 (*Jména p.p. zakladatelův*), which lists everyone who had paid in full."""
    print(f"\nMatice česká payments: {len(payments):,} in {len(months):,} months, "
          f"{months['period_year'].min()}–{months['period_year'].max()}")
    read = months["printed_k"].notna()
    print(f"Months against their printed total: {int(months['agrees'].sum()):,} of "
          f"{int(read.sum()):,} with a total read agree within 0.5 K "
          f"({months.loc[read, 'agrees'].mean():.1%}); parsed {months['sum_k'].sum():,.0f} K "
          f"against {months['printed_k'].sum():,.0f} K printed")
    print("Months by year (count, agreeing):")
    print(months.groupby("period_year").agg(months=("chain", "size"), agree=("agrees", "sum"),
                                            payments=("entries", "sum")).T.to_string())
    print("\nPayment flags:")
    print(payments["flags"].replace("", "none").str.split(";").explode().value_counts().to_string())
    print("\nResidences (share of payments):")
    print(payments["place_status"].value_counts(normalize=True).round(3).to_string())
    print(f"of them placed by the same payer's other payments (borrowed): "
          f"{int((payments['place_how'] == 'borrowed:payer').sum()):,}")
    print(f"\nPayers (persons): {len(payers):,}; reached the founder's fee: "
          f"{int(payers['founder'].sum()):,}")
    by = payers.groupby("first_year").agg(payers=("payer_id", "size"), founders=("founder", "sum"))
    print(by.T.to_string())
    lo, hi = mt.JOINER_WINDOW
    print(f"\nJoiners {lo}–{hi} (first payment in the window): {len(mt.joiners(payers)):,}; "
          f"of them paid the fee within the series: "
          f"{len(mt.joiners(payers, founders_only=True)):,}")
    print(mt.joiners(payers)["member_group"].value_counts(normalize=True).round(3).to_string())
    register_check(payers, entries)


def register_check(payers: pd.DataFrame, entries: pd.DataFrame) -> None:
    """Share of the payers who reached the fee in 1888–1890 who are in the 1890 founder
    register with an entry year of 1888–1890, matched on folded surname and settlement."""
    done = payers[payers["founder"] & payers["founder_year"].between(1888, 1890)
                  & payers["settlement_id"].notna()]
    reg = entries[(entries["source"] == "mc1890") & (entries["list_kind"] == "founder")
                  & entries["settlement_id"].notna()]
    key = set(zip(reg["surname"].map(mt._fold), reg["settlement_id"].astype(str)))
    late = reg[pd.to_numeric(reg["entry_year"], errors="coerce").between(1886, 1891)]
    key_late = set(zip(late["surname"].map(mt._fold), late["settlement_id"].astype(str)))
    hit = [(mt._fold(s), str(p)) in key for s, p in zip(done["surname"], done["settlement_id"])]
    hit_late = [(mt._fold(s), str(p)) in key_late
                for s, p in zip(done["surname"], done["settlement_id"])]
    if len(done):
        print(f"\nPayers paid in full 1888–1890 and placed: {len(done):,}; in the 1890 founder "
              f"register by surname + settlement: {sum(hit) / len(done):.1%} (with an entry "
              f"year 1886–1891: {sum(hit_late) / len(done):.1%})")


def court_level(entries: pd.DataFrame, payers: pd.DataFrame) -> pd.DataFrame:
    """The measure per court district (``matice.court_measure``), each settlement placed in
    the court its located point falls in (``settlements_1910_geo.point_court_id``)."""
    geo = pd.read_parquet(GEO / "settlements_1910_geo.parquet")
    courts = geo.dropna(subset=["point_court_id"]).set_index("settlement_id")["point_court_id"]
    court_census, _ = census.load_census_file(COURT_CENSUS)
    return mt.court_measure(entries, courts, court_census,
                            pd.read_parquet(GEO / "districts_1910.parquet"), payers)


if __name__ == "__main__":
    sys.exit(main())
