How it works

Offline: reads the page caches utils/scraping/run_matice_downloads.py left in library/associations/ (and the 1878 Internet Archive PDF beside them), the 1910 place tables in data/final/geo/ and the pooled okres census, and writes nothing but data/interim/matice/:

matice_entries.csv One row per printed entry (key entry_id), every field beside the entry's own text, with entry_year_from_no (a deceased founder's entry year read off the membership numbers), number_order_outlier, the parse flags, the 1910 settlement and okres its residence names (utils.parsing.locative, with place_status and place_how), and the member's occupation_class and member_group (posted / local / student / corporate / unknown). matice_totals.csv Entries found against the totals each society printed. matice_okres.csv The measure: one row per census okres (key census_okres_id, zeros kept), member counts, rates per 1,000 inhabitants and per 1,000 Czech speakers, and empirical-Bayes shrunk rates (matice.okres_measure). matice_court.csv The same per court district (key census_court_id, matice.court_measure), with the court census of 1910 as denominator.

and prints what it found. Nothing reaches data/final.

Usage::

python utils/reference_tools/places/run_matice_pilot.py
python utils/reference_tools/places/run_matice_pilot.py --only mc1890 mm1909

def add_places(entries: pd.DataFrame) -> pd.DataFrame:
    """The 1910 settlement and okres of every residence (``locative.match_all``), then the
    residences re-read from the scans (``READINGS``) where the OCR's could not be placed."""
    index = lc.build_index(pd.read_parquet(GEO / "settlements_1910.parquet"),
                           pd.read_csv(GEO / "settlement_aliases_1910.csv"),
                           pd.read_parquet(GEO / "districts_1910.parquet"))
    places = lc.match_all(entries["residence"].where(entries["residence"].astype(bool)), index)
    out = pd.concat([entries, places], axis=1)
    out = apply_readings(out, index)
    out["okres_id"] = out["okres_id"].astype("Int64")
    return out

def apply_readings(entries: pd.DataFrame, index: dict) -> pd.DataFrame:
    """Place an unmatched entry by its re-read residence. A reading counts only if the
    matcher then places it (a settlement, Prague, abroad), or if it says the entry
    names no place (``-``); otherwise the entry stays unmatched, the reading kept in
    ``residence_read`` for a look by eye. ``place_how`` starts ``read:`` where it counted."""
    entries["residence_read"] = ""
    if not READINGS.exists():
        return entries
    read = pd.read_csv(READINGS, dtype=str, keep_default_na=False).set_index("entry_id")
    hit = entries["entry_id"].isin(read.index) & (entries["place_status"] == "unmatched")
    entries.loc[hit, "residence_read"] = entries.loc[hit, "entry_id"].map(read["residence_read"])
    cols = ["settlement_id", "okres_id", "place_status", "place_how", "n_candidates", "place_key"]
    redo = entries[hit & ~entries["residence_read"].isin(["", "?"])]
    none = redo["residence_read"] == "-"
    entries.loc[none[none].index, ["place_status", "place_how"]] = ["none", "read:no_place"]
    redo = redo[~none]
    again = lc.match_all(redo["residence_read"], index)
    ok = again["place_status"].isin(_ACCEPT)
    again.loc[ok, "place_how"] = "read:" + again.loc[ok, "place_how"].astype(str)
    entries.loc[again[ok].index, cols] = again.loc[ok, cols].to_numpy()
    return entries

def report(entries: pd.DataFrame, totals: pd.DataFrame, measure: pd.DataFrame) -> None:
    print(f"\nmatice {mt.__version__} / locative {lc.__version__}: {len(entries):,} entries "
          f"from {entries['source'].nunique()} sources\n")
    print(entries.groupby(["source", "list_kind"]).size().rename("entries").to_string())
    print("\nAgainst the printed totals:")
    print(totals.to_string(index=False))
    mc = entries[entries["society"] == "MC"]
    both = (set(mc.loc[mc["source"] == "mc1873", "member_no"].dropna())
            & set(mc.loc[mc["source"] == "mc1890", "member_no"].dropna()))
    print(f"\nMatice česká membership numbers in both issues: {len(both):,}")
    print(f"Numbers whose entry year disagrees with their neighbours': "
          f"{int(entries['number_order_outlier'].sum()):,}")
    print("\nResidences (share of entries):")
    print(entries.groupby("society")["place_status"].value_counts(normalize=True)
          .round(3).unstack(fill_value=0).to_string())
    how = entries.loc[entries["place_status"] == "matched", "place_how"].str.split(":").str[-1]
    print("\nHow a settlement was chosen (matched entries):")
    print(how.value_counts().to_string())
    print("\nMember groups (persons, not corporate):")
    print(entries[entries["member_group"] != "corporate"].groupby("society")["member_group"]
          .value_counts(normalize=True).round(3).unstack(fill_value=0).to_string())
    cols = ["mc_founders", "mc_founders_local", "mm_members_1909"]
    print(f"\nThe measure: {len(measure)} okresy; members placed in an okres:")
    print(measure.groupby("land")[cols + ["mc_corporate", "mm_members_1890"]].sum().to_string())
    print("\nOkresy with no founder at all: "
          f"{int((measure['mc_founders'] == 0).sum())} of {len(measure)}")
    top = measure.sort_values("mc_founders_per_1000_eb", ascending=False)
    print("\nHighest Matice česká founder rate per 1,000 (shrunk), 12 okresy:")
    print(top[["okres_cs", "land", "pop_total", "lang_czech_share", "mc_founders",
               "mc_founders_per_1000", "mc_founders_per_1000_eb"]].head(12)
          .round(3).to_string(index=False))
    rates = [f"{c}_per_1000_eb" for c in cols]
    print("\nSpearman correlations across okresy (shrunk rates, Czech share):")
    print(measure[rates + ["lang_czech_share"]].corr(method="spearman").round(2).to_string())
    if THEATRE.exists():
        print("\nAgainst the theatre measure:")
        print(theatre_check(measure).round(2).to_string())

def theatre_check(measure: pd.DataFrame) -> pd.DataFrame:
    """Spearman correlation of the Matice rates with Czech theatre companies active per
    1,000 inhabitants, 1880–1890 (certain), across okresy — whole sample and the
    okresy at least half Czech-speaking."""
    t = pd.read_parquet(THEATRE)
    xw = pd.read_csv(GEO / "okres_census_crosswalk.csv")
    t = t[(t["year"] >= 1880) & (t["year"] <= 1890)].merge(
        xw[["okres_id", "census_okres_id"]], on="okres_id", how="left", validate="m:1")
    act = (t.groupby("census_okres_id")["ensembles_active_certain"].sum() / 11).rename("theatre")
    m = measure.merge(act, left_on="census_okres_id", right_index=True, how="left",
                      validate="1:1")
    m["theatre_per_1000"] = 1000 * m["theatre"] / m["pop_total"]
    cols = ["mc_founders_per_1000_eb", "mc_founders_local_per_1000_eb",
            "mm_members_1890_per_1000_eb", "mm_members_1909_per_1000_eb", "matice_own_z"]
    czech = m[m["lang_czech_share"] >= 0.5]
    frames = {"all okresy": m, "Czech share >= 0.5": czech,
              "Böhmen, Czech >= 0.5": czech[czech["land"] == "Böhmen"],
              "Mähren, Czech >= 0.5": czech[czech["land"] == "Mähren"]}
    return pd.DataFrame({label: f[cols].corrwith(f["theatre_per_1000"], method="spearman")
                         for label, f in frames.items()})

def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    p.add_argument("--only", nargs="+", choices=list(mt.SOURCES), default=None)
    args = p.parse_args(argv)

    entries = mt.parse_all(LIBRARY, args.only)
    if entries.empty:
        print(f"No page caches found in {LIBRARY}; run utils/scraping/run_matice_downloads.py.")
        return 1
    assert entries["entry_id"].is_unique and entries["entry_id"].notna().all()
    entries["number_order_outlier"] = mt.mc_number_order(entries)
    entries["entry_year_from_no"] = mt.entry_year_from_number(entries)
    n = len(entries)
    entries = add_occupations(add_places(entries))
    assert len(entries) == n, "place matching changed the row count"
    totals = mt.totals_report(entries)
    measure = mt.okres_measure(entries, pd.read_parquet(CENSUS),
                               pd.read_csv(GEO / "okres_census_crosswalk.csv"))
    assert measure["census_okres_id"].is_unique
    courts = court_level(entries)
    assert courts["census_court_id"].is_unique

    OUT.mkdir(parents=True, exist_ok=True)
    entries.to_csv(OUT / "matice_entries.csv", index=False, encoding="utf-8")
    totals.to_csv(OUT / "matice_totals.csv", index=False, encoding="utf-8")
    measure.to_csv(OUT / "matice_okres.csv", index=False, encoding="utf-8")
    courts.to_csv(OUT / "matice_court.csv", index=False, encoding="utf-8")
    report(entries, totals, measure)
    print(f"\nCourt districts: {len(courts)}; members placed in one: "
          f"{int(courts['mc_founders'].sum()):,} Matice česká, "
          f"{int(courts['mm_members_1890'].sum()):,} Matice moravská 1890")
    print(f"\nwrote {OUT.relative_to(ROOT)}/matice_entries.csv, matice_totals.csv, "
          "matice_okres.csv, matice_court.csv")
    return 0

    print("\nHighest Matice česká founder rate per 1,000 (shrunk), 12 okresy:")
    print(top[["okres_cs", "land", "pop_total", "lang_czech_share", "mc_founders",
               "mc_founders_per_1000", "mc_founders_per_1000_eb"]].head(12)
          .round(3).to_string(index=False))
    rates = [f"{c}_per_1000_eb" for c in cols]
    print("\nSpearman correlations across okresy (shrunk rates, Czech share):")
    print(measure[rates + ["lang_czech_share"]].corr(method="spearman").round(2).to_string())
    if THEATRE.exists():
        print("\nAgainst the theatre measure:")
        print(theatre_check(measure).round(2).to_string())


def theatre_check(measure: pd.DataFrame) -> pd.DataFrame:
    """Spearman correlation of the Matice rates with Czech theatre companies active per
    1,000 inhabitants, 1880–1890 (certain), across okresy — whole sample and the
    okresy at least half Czech-speaking."""
    t = pd.read_parquet(THEATRE)
    xw = pd.read_csv(GEO / "okres_census_crosswalk.csv")
    t = t[(t["year"] >= 1880) & (t["year"] <= 1890)].merge(
        xw[["okres_id", "census_okres_id"]], on="okres_id", how="left", validate="m:1")
    act = (t.groupby("census_okres_id")["ensembles_active_certain"].sum() / 11).rename("theatre")
    m = measure.merge(act, left_on="census_okres_id", right_index=True, how="left",
                      validate="1:1")
    m["theatre_per_1000"] = 1000 * m["theatre"] / m["pop_total"]
    cols = ["mc_founders_per_1000_eb", "mc_founders_local_per_1000_eb",
            "mm_members_1890_per_1000_eb", "mm_members_1909_per_1000_eb", "matice_own_z"]
    czech = m[m["lang_czech_share"] >= 0.5]
    frames = {"all okresy": m, "Czech share >= 0.5": czech,
              "Böhmen, Czech >= 0.5": czech[czech["land"] == "Böhmen"],
              "Mähren, Czech >= 0.5": czech[czech["land"] == "Mähren"]}
    return pd.DataFrame({label: f[cols].corrwith(f["theatre_per_1000"], method="spearman")
                         for label, f in frames.items()})


def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    p.add_argument("--only", nargs="+", choices=list(mt.SOURCES), default=None)
    args = p.parse_args(argv)

    entries = mt.parse_all(LIBRARY, args.only)
    if entries.empty:
        print(f"No page caches found in {LIBRARY}; run utils/scraping/run_matice_downloads.py.")
        return 1
    assert entries["entry_id"].is_unique and entries["entry_id"].notna().all()
    entries["number_order_outlier"] = mt.mc_number_order(entries)
    entries["entry_year_from_no"] = mt.entry_year_from_number(entries)
    n = len(entries)
    entries = add_occupations(add_places(entries))
    assert len(entries) == n, "place matching changed the row count"
    totals = mt.totals_report(entries)
    measure = mt.okres_measure(entries, pd.read_parquet(CENSUS),
                               pd.read_csv(GEO / "okres_census_crosswalk.csv"))
    assert measure["census_okres_id"].is_unique
    courts = court_level(entries)
    assert courts["census_court_id"].is_unique

    OUT.mkdir(parents=True, exist_ok=True)
    entries.to_csv(OUT / "matice_entries.csv", index=False, encoding="utf-8")
    totals.to_csv(OUT / "matice_totals.csv", index=False, encoding="utf-8")
    measure.to_csv(OUT / "matice_okres.csv", index=False, encoding="utf-8")
    courts.to_csv(OUT / "matice_court.csv", index=False, encoding="utf-8")
    report(entries, totals, measure)
    print(f"\nCourt districts: {len(courts)}; members placed in one: "
          f"{int(courts['mc_founders'].sum()):,} Matice česká, "
          f"{int(courts['mm_members_1890'].sum()):,} Matice moravská 1890")
    print(f"\nwrote {OUT.relative_to(ROOT)}/matice_entries.csv, matice_totals.csv, "
          "matice_okres.csv, matice_court.csv")
    return 0


def court_level(entries: pd.DataFrame) -> pd.DataFrame:
    """The measure per court district (``matice.court_measure``), each settlement placed in
    the court its located point falls in (``settlements_1910_geo.point_court_id``)."""
    geo = pd.read_parquet(GEO / "settlements_1910_geo.parquet")
    courts = geo.dropna(subset=["point_court_id"]).set_index("settlement_id")["point_court_id"]
    court_census, _ = census.load_census_file(COURT_CENSUS)
    return mt.court_measure(entries, courts, court_census,
                            pd.read_parquet(GEO / "districts_1910.parquet"))
