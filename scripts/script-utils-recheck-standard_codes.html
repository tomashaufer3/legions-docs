How it works

A text code's municipal columns are named by its English meaning (n_cat_txt_&lt;english&gt;, n_coop_kind_&lt;english&gt;), so two volumes already share a column wherever their codes' English agrees: the Czech b. úř. and the German Sta. are both tax office. A handful of English names differ for one institution, and those split it across two columns: the Czech cooperative letter P is "industrial cooperative" and the German G "trade cooperative", one registry class (Prague prints 70 of each in its two editions). reference/places/institution_standard.csv maps each such name to one standard, with the evidence for it; every other name is its own standard.

Applied to the rechecked text table before the register is built (register_rechecked.runs): each row's english becomes its standard, so the text and cooperative columns are one across volumes. n_mark_*, the code as printed, is untouched (each volume's own). Also here: the census claims of census_codes put beside the route's.

def bohemian_check(institutions: pd.DataFrame, places: pd.DataFrame, min_places: int = 5) -> pd.DataFrame:
    """The two Bohemian volumes print the same places, so a standard institution should stand at the same
    settlements in both. Per text institution (``common_category`` ``txt:…``) printed at ``min_places`` or more
    settlements both volumes anchor: the settlements in the Czech volume, in German Böhmen, in both, and their
    overlap (Jaccard). A low overlap is a name or a code one volume still lacks."""
    key = ["edition", "court_key", "muni_seq", "settlement_seq"]
    anchored = places.loc[places["settlement_id"].notna(), key + ["settlement_id"]]
    text = institutions[institutions["common_category"].astype(str).str.startswith("txt:")]
    text = text.merge(anchored, on=key, how="inner", validate="m:1")
    both = set(text.loc[text["edition"] == "cz", "settlement_id"]) & set(text.loc[text["edition"] == "de-bohemia", "settlement_id"])
    text = text[text["settlement_id"].isin(both)]
    rows = []
    for name, g in text.groupby("common_category"):
        cz, de = (set(g.loc[g["edition"] == e, "settlement_id"]) for e in ("cz", "de-bohemia"))
        if len(cz | de) >= min_places:
            rows.append({"institution": name[4:], "cz": len(cz), "de_bohemia": len(de), "both": len(cz & de),
                         "jaccard": round(len(cz & de) / len(cz | de), 3)})
    return pd.DataFrame(rows, columns=["institution", "cz", "de_bohemia", "both", "jaccard"]).sort_values("jaccard")

def _supersede(frame: pd.DataFrame, added: pd.DataFrame, key: list[str]) -> tuple[pd.DataFrame, int]:
    """Each added claim that supersedes a more general code (``census_codes.supersession``, D-235) takes its count off
    that code's claim at the same part; a claim brought to nothing is dropped. One music school the route read as a
    bare *šk.* is then one music school, not a school and a music school."""
    if "supersedes" not in added or not added["supersedes"].astype(str).ne("").any():
        return frame, 0
    frame = frame.reset_index(drop=True)
    count = pd.to_numeric(frame["count"], errors="coerce").fillna(1).astype(int)
    where = {}
    for idx, r in enumerate(frame[key + ["code"]].astype(str).itertuples(index=False)):
        where.setdefault(tuple(r), idx)
    taken = 0
    for r in added[added["supersedes"].astype(str).ne("")].itertuples(index=False):
        idx = where.get((*(str(getattr(r, c)) for c in key), str(r.supersedes)))
        if idx is not None:
            # what the route counted for this item under the general code (census_codes 0.4.0, D-237); the new
            # claim's own count only where a census file predates the column
            n = getattr(r, "supersedes_count", None)
            n = pd.to_numeric(n, errors="coerce") if n not in (None, "") else pd.to_numeric(r.count, errors="coerce")
            count.iloc[idx] -= int(n) if pd.notna(n) else 1
            taken += 1
    out = frame.assign(count=count.clip(lower=0).astype(str))
    return out[count > 0].reset_index(drop=True), taken
