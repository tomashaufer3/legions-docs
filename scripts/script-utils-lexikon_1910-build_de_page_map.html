How it works

The two volumes are the same census table set twice, and neither is a translation of the other's typesetting: they paginate differently and each orders an okres's municipalities alphabetically in its own language. So there is no scan-to-scan correspondence to assume — German scan 118 is not the German printing of Czech scan 118 — and a run that assumes one reads unrelated pages while looking entirely well-behaved.

What makes the pairing possible is that both volumes print both names. The Czech table sets Vysoká, Wisset; the German sets Wisset, Vysoká. So the German twin of every municipality on a Czech page can be read straight off the Czech page, and the German page that prints those same names is the page that covers it. No external crosswalk, no district-name translation table, and — this is the point — the match score is itself the evidence. A bad pairing shows up as a low score rather than as a clean-looking wrong answer.

Two guards, both from failures measured in this project:

  • The district is checked first. Village names repeat across Bohemia, and a name-only match over 481 pages will find a homonym in another okres. Candidate pages are therefore restricted to those whose running head names the right politischer Bezirk before names are counted — the same discipline records_geo uses when a birthplace could be one of several places.
  • A tie is refused, not broken. Where two DE pages score equally the row is written ambiguous and carries both. lexikon_1910_de already refuses a tie one level down rather than taking the first match, and for the same reason: on this source a plausible wrong answer is indistinguishable from a right one.

Output — reference/places/lexikon_1910_de_page_map.csv:

cz_scan  de_scans  score  n_names  n_matched  districts_cz  verdict

verdict is matched / weak / ambiguous / unmatched. Nothing downstream may use a row that is not matched without saying so.

def normalise(text: str) -> str:
    """Match key for a place name, across two orthographies.

    Accents are stripped, and the German transliterations are folded onto the
    umlauts they stand for: the volume prints ``Brüx`` while the curated tables
    write ``Bruex``, and under NFKD alone those are ``brux`` and ``bruex`` — two
    different keys for one town. That single mismatch is why
    ``lexikon_1910_de.okres_pages("Bruex")`` returns nothing.
    """
    text = unicodedata.normalize("NFKD", str(text))
    text = "".join(c for c in text if not unicodedata.combining(c))
    text = text.lower()
    for pair, single in (("ue", "u"), ("oe", "o"), ("ae", "a"), ("ss", "s")):
        text = text.replace(pair, single)
    return re.sub(r"[^a-z0-9]", "", text)

def parse_head(text: str) -> list[tuple[str, list[str]]]:
    """``"Most: Most. — Náchod: Česká Skalice."`` -> ``[(most,[most]), (nachod,[ceskaskalice])]``.

    Both sides are returned because they answer different questions: the political
    okres is what the German volume's index is keyed on, and the **court** is what
    actually identifies the page — a political okres runs to a dozen pages, a court
    to two or three.
    """
    out: list[tuple[str, list[str]]] = []
    for political, courts in _HEAD_RE.findall(text or ""):
        key = normalise(political)
        names = [normalise(c) for c in courts.split(",")]
        names = [n for n in names if len(n) >= 3]
        if len(key) >= 3 and names:
            out.append((key, names))
    return out

def court_crosswalk() -> dict[str, str]:
    """Czech court name -> German, both sides normalised.

    Read through :func:`normalise` on both sides rather than matched literally,
    which repairs two things at once: the curation writes the transliterated
    ``Goerkau`` / ``Bruex`` where the volume prints ``Görkau`` / ``Brüx``, and the
    OCR of the Czech head gives ``Chomútov`` and ``Marsov`` where the curation has
    ``Chomutov`` and ``Maršov``. Both are spelling, not identity.

    ⚠ Only the 330 **court** rows carry a German name; all 158 okres rows have the
    column empty. That is why this pairs on courts and not on the political
    district — the data for the other one does not exist.
    """
    path = ROOT / "reference" / "places" / "district_cs_curation.csv"
    out: dict[str, str] = {}
    with path.open(encoding="utf-8", newline="") as fh:
        for row in csv.DictReader(fh):
            if row.get("level") == "court" and (row.get("de") or "").strip():
                out[normalise(row["cs"])] = normalise(row["de"])
    return out

def cz_page_names(scan: int, template) -> tuple[list[str], set[str]]:
    """``(German twin names, Czech district names)`` printed on one Czech page.

    The name cell sets ``Czech, German`` — occasionally with a topographic symbol
    or a cadastral span trailing it — so the German twin is what follows the first
    comma, cut at the first character that cannot be part of a name. Rows whose
    name cell has no comma are settlements printed under one name in both
    volumes; they are skipped rather than guessed at.
    """
    page = lxp.alto_page(scan)
    if page is None:
        return [], set()
    try:
        rows = lx.extract_rows(page, template)
    except ValueError:
        # "no column rules — not a table page". The volume opens its table a few
        # scans after the main part nominally begins (15-18 carry no apparatus
        # either, which draw_gold_sample already excludes), and a page with no
        # grid has no name column to read. Reported as unmatched rather than
        # crashed on: the map has to cover every scan it was asked about, and a
        # page that cannot be paired is a fact about the volume.
        return [], set()
    names: list[str] = []
    for row in rows:
        cell = (row.cells.get("name") or "")
        if "," not in cell:
            continue
        german = re.split(r"[^A-Za-zÄÖÜäöüß'’.\-]", cell.split(",", 1)[1].strip())
        german = german[0] if german else ""
        if len(german) >= MIN_NAME:
            names.append(normalise(german))
    courts = {c for _pol, cs in parse_head(lx.running_head(page)) for c in cs}
    return list(dict.fromkeys(n for n in names if n)), courts

def match_page(names: list[str], courts_cz: set[str],
               volume: dict[int, tuple[set[str], set[str]]],
               crosswalk: dict[str, str],
               top: int = 3) -> tuple[list[int], float, int, str]:
    """Score every German page against one Czech page's names.

    The district filter runs first and is the guard against homonyms: village
    names repeat across Bohemia, and a name-only search over 481 pages will find
    one in the wrong okres. Where the filter leaves nothing — a Czech running head
    the OCR destroyed, a German head it destroyed — the search falls back to the
    whole volume and the row is flagged, because an unfiltered match is a weaker
    claim and must not be recorded as if it were the same thing.
    """
    if not names:
        return [], 0.0, 0, "unmatched"

    wanted = set(names)
    # Czech courts -> German courts, then keep only the German pages whose own
    # running head names one of them. This is the homonym guard: village names
    # repeat across Bohemia, and a name-only search over 481 pages will happily
    # find one in the wrong okres and score it as a hit.
    courts_de = {crosswalk[c] for c in courts_cz if c in crosswalk}
    candidates = {s: w for s, (w, d) in volume.items() if d & courts_de} \
        if courts_de else {}
    filtered = bool(candidates)
    if not candidates:
        candidates = {s: w for s, (w, _d) in volume.items()}

    scores = Counter({scan: len(wanted & words) for scan, words in candidates.items()})
    best = [(s, c) for s, c in scores.most_common(top) if c > 0]
    if not best:
        return [], 0.0, 0, "unmatched"

    top_count = best[0][1]
    winners = [s for s, c in best if c == top_count]
    share = top_count / len(wanted)
    verdict = ("ambiguous" if len(winners) > 1
               else "weak" if share < WEAK_BELOW or not filtered
               else "matched")
    pages = sorted({s for s, _c in best})
    return pages, round(share, 3), top_count, verdict

def build(scans: list[int], plaintext_dir: Path) -> list[dict]:
    template = lx.read_grid_template(GRID_CSV, GRID_VOLUME)
    crosswalk = court_crosswalk()
    print(f"  {len(crosswalk)} court name pairs, cs -> de")
    print(f"  loading the German main part, scans "
          f"{DE_FIRST_SCAN}-{DE_LAST_SCAN} ...")
    volume = load_de_volume(plaintext_dir)
    print(f"  {len(volume)} German pages indexed")

    rows: list[dict] = []
    for i, scan in enumerate(scans, 1):
        names, courts = cz_page_names(scan, template)
        pages, score, matched, verdict = match_page(names, courts, volume, crosswalk)
        rows.append({
            "cz_scan": f"{scan:04d}",
            "de_scans": " ".join(f"{p:04d}" for p in pages),
            "score": score, "n_names": len(names), "n_matched": matched,
            "districts_cz": " ".join(sorted(courts)), "verdict": verdict,
        })
        if i % 50 == 0 or len(scans) <= 20:
            print(f"    cz {scan:04d}  {len(names):3d} names -> "
                  f"{rows[-1]['de_scans'] or '-':20s} {score:.2f}  {verdict}")
    return rows

def report(rows: list[dict]) -> str:
    lines = [f"build_de_page_map v{__version__} — CZ page -> DE pages", ""]
    verdicts = Counter(r["verdict"] for r in rows)
    lines.append(f"  {len(rows)} Czech pages")
    for verdict in ("matched", "weak", "ambiguous", "unmatched"):
        n = verdicts.get(verdict, 0)
        if n:
            lines.append(f"    {verdict:10s} {n:4d}  ({n / len(rows):.1%})")
    scored = [r["score"] for r in rows if r["score"]]
    if scored:
        lines.append(f"  mean name overlap on a hit: "
                     f"{sum(scored) / len(scored):.2f}")
    lines += ["",
              "  A Czech page spans two or three German pages, so the best single",
              "  German page holds only part of its names — the score measures "
              "'the right",
              "  district's page', not 'this page holds everything'.",
              "",
              "  Only `matched` rows may be used without saying so. `ambiguous` "
              "means two",
              "  German pages tied and the tie was refused rather than broken on "
              "lookup order."]
    return "\n".join(lines)

def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n")[0])
    ap.add_argument("--pages", default="",
                    help="comma-separated Czech scans (default: the purposive five)")
    ap.add_argument("--all", action="store_true",
                    help=f"every Czech table page, {lxp.CZ_FIRST_SCAN}-{lxp.CZ_LAST_SCAN}")
    ap.add_argument("--write", action="store_true",
                    help=f"write {OUT_CSV.relative_to(ROOT)}")
    args = ap.parse_args()

    if args.all:
        scans = list(range(lxp.CZ_FIRST_SCAN, lxp.CZ_LAST_SCAN + 1))
    elif args.pages:
        scans = [int(p) for p in args.pages.split(",") if p.strip()]
    else:
        scans = list(lxp.PURPOSIVE_PAGES)

    plaintext = lxp.DE_PLAINTEXT
    if not plaintext.exists():
        print(f"no German plaintext at {plaintext}", file=sys.stderr)
        return 1

    rows = build(scans, plaintext)
    print()
    print(report(rows))
    if args.write:
        OUT_CSV.parent.mkdir(parents=True, exist_ok=True)
        with OUT_CSV.open("w", encoding="utf-8", newline="") as fh:
            writer = csv.DictWriter(fh, fieldnames=FIELDS)
            writer.writeheader()
            writer.writerows(rows)
        print(f"\n  wrote {OUT_CSV.relative_to(ROOT)}  ({len(rows)} rows)")
    else:
        print("\n  (dry run — pass --write to save)")
    return 0

The two volumes are the same census table set twice, and neither is a translation
of the other's typesetting: they paginate differently and each orders an okres's
municipalities alphabetically **in its own language**. So there is no scan-to-scan
correspondence to assume — German scan 118 is not the German printing of Czech
scan 118 — and a run that assumes one reads unrelated pages while looking
entirely well-behaved.

Czech table sets ``Vysoká, Wisset``; the German sets ``Wisset, Vysoká``. So the
German twin of every municipality on a Czech page can be read straight off the
Czech page, and the German page that prints those same names is the page that
covers it. No external crosswalk, no district-name translation table, and — this
is the point — **the match score is itself the evidence**. A bad pairing shows up
as a low score rather than as a clean-looking wrong answer.

Two guards, both from failures measured in this project:

* **The district is checked first.** Village names repeat across Bohemia, and a
  name-only match over 481 pages will find a homonym in another okres. Candidate
  pages are therefore restricted to those whose running head names the right
  *politischer Bezirk* before names are counted — the same discipline
  ``records_geo`` uses when a birthplace could be one of several places.
* **A tie is refused, not broken.** Where two DE pages score equally the row is
  written ``ambiguous`` and carries both. ``lexikon_1910_de`` already refuses a
  tie one level down rather than taking the first match, and for the same reason:
  on this source a plausible wrong answer is indistinguishable from a right one.

Output — ``reference/places/lexikon_1910_de_page_map.csv``:

    cz_scan  de_scans  score  n_names  n_matched  districts_cz  verdict

``verdict`` is ``matched`` / ``weak`` / ``ambiguous`` / ``unmatched``. Nothing
downstream may use a row that is not ``matched`` without saying so.

    python utils/lexikon_1910/build_de_page_map.py --pages 118,270,449,450,451
    python utils/lexikon_1910/build_de_page_map.py --all --write
"""

from __future__ import annotations

import argparse
import csv
import re
import sys
import unicodedata
from collections import Counter
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from utils.lexikon_1910 import lexikon_1910 as lx            # noqa: E402
from utils.lexikon_1910 import lexikon_1910_pages as lxp     # noqa: E402

__version__ = "1.0.0"

GRID_CSV = ROOT / "reference" / "places" / "lexikon_1910_grid.csv"
GRID_VOLUME = "cechy_1910_cz"
OUT_CSV = ROOT / "reference" / "places" / "lexikon_1910_de_page_map.csv"

#: The German volume's main part. Scan 0011–0013 is the *Zeichenerklärung*, 0496
#: opens ``Landesübersicht I`` and everything after is the appendix and the
#: alphabetical index. Measured by reading the heads at the boundary, not
#: assumed: the index at the back names every district again, so a page range
#: that includes it gives each district a dozen spurious pages.
DE_FIRST_SCAN, DE_LAST_SCAN = 15, 495

#: A name has to be this long to count. Short tokens are column headings, roman
#: numerals and OCR noise, and they match everywhere.
MIN_NAME = 4

#: Below this share of the page's German names, a match is reported but flagged.
#: A Czech page spans two or three German pages, so the best single German page
#: legitimately holds only a third to a half of them — the threshold is set for
#: "this is the right district's page", not for "this page holds everything".
WEAK_BELOW = 0.20

FIELDS = ["cz_scan", "de_scans", "score", "n_names", "n_matched",
          "districts_cz", "verdict"]


def normalise(text: str) -> str:
    """Match key for a place name, across two orthographies.

    Accents are stripped, and the German transliterations are folded onto the
    umlauts they stand for: the volume prints ``Brüx`` while the curated tables
    write ``Bruex``, and under NFKD alone those are ``brux`` and ``bruex`` — two
    different keys for one town. That single mismatch is why
    ``lexikon_1910_de.okres_pages("Bruex")`` returns nothing.
    """
    text = unicodedata.normalize("NFKD", str(text))
    text = "".join(c for c in text if not unicodedata.combining(c))
    text = text.lower()
    for pair, single in (("ue", "u"), ("oe", "o"), ("ae", "a"), ("ss", "s")):
        text = text.replace(pair, single)
    return re.sub(r"[^a-z0-9]", "", text)


_WORD_RE = re.compile(r"[A-Za-zÄÖÜäöüß][\w'’.\-]{%d,}" % (MIN_NAME - 1))

#: The running head of both volumes prints ``political okres: court, court.`` —
#: ``Chomútov: Chomútov, Jirkov.`` in Czech, ``Komotau: Görkau.`` in German. A
#: page where one district ends and the next begins prints both, separated by an
#: em dash. The trailing ``.`` is what ends the head; everything after it on the
#: line is column-heading text the OCR has run together with it.
_HEAD_RE = re.compile(
    r"([A-Za-zÁ-Žá-žÄÖÜäöüß][\w.\-’'À-ſ ]{1,28}?)\s*:\s*([^.:]{2,60})\.")


def parse_head(text: str) -> list[tuple[str, list[str]]]:
    """``"Most: Most. — Náchod: Česká Skalice."`` -> ``[(most,[most]), (nachod,[ceskaskalice])]``.

    Both sides are returned because they answer different questions: the political
    okres is what the German volume's index is keyed on, and the **court** is what
    actually identifies the page — a political okres runs to a dozen pages, a court
    to two or three.
    """
    out: list[tuple[str, list[str]]] = []
    for political, courts in _HEAD_RE.findall(text or ""):
        key = normalise(political)
        names = [normalise(c) for c in courts.split(",")]
        names = [n for n in names if len(n) >= 3]
        if len(key) >= 3 and names:
            out.append((key, names))
    return out


def court_crosswalk() -> dict[str, str]:
    """Czech court name -> German, both sides normalised.

    Read through :func:`normalise` on both sides rather than matched literally,
    which repairs two things at once: the curation writes the transliterated
    ``Goerkau`` / ``Bruex`` where the volume prints ``Görkau`` / ``Brüx``, and the
    OCR of the Czech head gives ``Chomútov`` and ``Marsov`` where the curation has
    ``Chomutov`` and ``Maršov``. Both are spelling, not identity.

    ⚠ Only the 330 **court** rows carry a German name; all 158 okres rows have the
    column empty. That is why this pairs on courts and not on the political
    district — the data for the other one does not exist.
    """
    path = ROOT / "reference" / "places" / "district_cs_curation.csv"
    out: dict[str, str] = {}
    with path.open(encoding="utf-8", newline="") as fh:
        for row in csv.DictReader(fh):
            if row.get("level") == "court" and (row.get("de") or "").strip():
                out[normalise(row["cs"])] = normalise(row["de"])
    return out


def de_page_words(path: Path) -> set[str]:
    """Every word-shaped token on one German page, normalised."""
    text = path.read_text(encoding="utf-8", errors="replace")
    return {normalise(w) for w in _WORD_RE.findall(text)}


def de_page_courts(path: Path, head_lines: int = 12) -> set[str]:
    """The *Gerichtsbezirk* names in one German page's running head."""
    head = "\n".join(
        path.read_text(encoding="utf-8", errors="replace").split("\n")[:head_lines])
    return {court for _political, courts in parse_head(head) for court in courts}


def load_de_volume(plaintext_dir: Path) -> dict[int, tuple[set[str], set[str]]]:
    """``scan -> (words, courts)`` for the German main part."""
    volume: dict[int, tuple[set[str], set[str]]] = {}
    for scan in range(DE_FIRST_SCAN, DE_LAST_SCAN + 1):
        path = plaintext_dir / f"AC01116810-{scan:04d}.txt"
        if path.exists():
            volume[scan] = (de_page_words(path), de_page_courts(path))
    return volume


def cz_page_names(scan: int, template) -> tuple[list[str], set[str]]:
    """``(German twin names, Czech district names)`` printed on one Czech page.

    The name cell sets ``Czech, German`` — occasionally with a topographic symbol
    or a cadastral span trailing it — so the German twin is what follows the first
    comma, cut at the first character that cannot be part of a name. Rows whose
    name cell has no comma are settlements printed under one name in both
    volumes; they are skipped rather than guessed at.
    """
    page = lxp.alto_page(scan)
    if page is None:
        return [], set()
    try:
        rows = lx.extract_rows(page, template)
    except ValueError:
        # "no column rules — not a table page". The volume opens its table a few
        # scans after the main part nominally begins (15-18 carry no apparatus
        # either, which draw_gold_sample already excludes), and a page with no
        # grid has no name column to read. Reported as unmatched rather than
        # crashed on: the map has to cover every scan it was asked about, and a
        # page that cannot be paired is a fact about the volume.
        return [], set()
    names: list[str] = []
    for row in rows:
        cell = (row.cells.get("name") or "")
        if "," not in cell:
            continue
        german = re.split(r"[^A-Za-zÄÖÜäöüß'’.\-]", cell.split(",", 1)[1].strip())
        german = german[0] if german else ""
        if len(german) >= MIN_NAME:
            names.append(normalise(german))
    courts = {c for _pol, cs in parse_head(lx.running_head(page)) for c in cs}
    return list(dict.fromkeys(n for n in names if n)), courts


def match_page(names: list[str], courts_cz: set[str],
               volume: dict[int, tuple[set[str], set[str]]],
               crosswalk: dict[str, str],
               top: int = 3) -> tuple[list[int], float, int, str]:
    """Score every German page against one Czech page's names.

    The district filter runs first and is the guard against homonyms: village
    names repeat across Bohemia, and a name-only search over 481 pages will find
    one in the wrong okres. Where the filter leaves nothing — a Czech running head
    the OCR destroyed, a German head it destroyed — the search falls back to the
    whole volume and the row is flagged, because an unfiltered match is a weaker
    claim and must not be recorded as if it were the same thing.
    """
    if not names:
        return [], 0.0, 0, "unmatched"

    wanted = set(names)
    # Czech courts -> German courts, then keep only the German pages whose own
    # running head names one of them. This is the homonym guard: village names
    # repeat across Bohemia, and a name-only search over 481 pages will happily
    # find one in the wrong okres and score it as a hit.
    courts_de = {crosswalk[c] for c in courts_cz if c in crosswalk}
    candidates = {s: w for s, (w, d) in volume.items() if d & courts_de} \
        if courts_de else {}
    filtered = bool(candidates)
    if not candidates:
        candidates = {s: w for s, (w, _d) in volume.items()}

    scores = Counter({scan: len(wanted & words) for scan, words in candidates.items()})
    best = [(s, c) for s, c in scores.most_common(top) if c > 0]
    if not best:
        return [], 0.0, 0, "unmatched"

    top_count = best[0][1]
    winners = [s for s, c in best if c == top_count]
    share = top_count / len(wanted)
    verdict = ("ambiguous" if len(winners) > 1
               else "weak" if share < WEAK_BELOW or not filtered
               else "matched")
    pages = sorted({s for s, _c in best})
    return pages, round(share, 3), top_count, verdict


def build(scans: list[int], plaintext_dir: Path) -> list[dict]:
    template = lx.read_grid_template(GRID_CSV, GRID_VOLUME)
    crosswalk = court_crosswalk()
    print(f"  {len(crosswalk)} court name pairs, cs -> de")
    print(f"  loading the German main part, scans "
          f"{DE_FIRST_SCAN}-{DE_LAST_SCAN} ...")
    volume = load_de_volume(plaintext_dir)
    print(f"  {len(volume)} German pages indexed")

    rows: list[dict] = []
    for i, scan in enumerate(scans, 1):
        names, courts = cz_page_names(scan, template)
        pages, score, matched, verdict = match_page(names, courts, volume, crosswalk)
        rows.append({
            "cz_scan": f"{scan:04d}",
            "de_scans": " ".join(f"{p:04d}" for p in pages),
            "score": score, "n_names": len(names), "n_matched": matched,
            "districts_cz": " ".join(sorted(courts)), "verdict": verdict,
        })
        if i % 50 == 0 or len(scans) <= 20:
            print(f"    cz {scan:04d}  {len(names):3d} names -> "
                  f"{rows[-1]['de_scans'] or '-':20s} {score:.2f}  {verdict}")
    return rows


def report(rows: list[dict]) -> str:
    lines = [f"build_de_page_map v{__version__} — CZ page -> DE pages", ""]
    verdicts = Counter(r["verdict"] for r in rows)
    lines.append(f"  {len(rows)} Czech pages")
    for verdict in ("matched", "weak", "ambiguous", "unmatched"):
        n = verdicts.get(verdict, 0)
        if n:
            lines.append(f"    {verdict:10s} {n:4d}  ({n / len(rows):.1%})")
    scored = [r["score"] for r in rows if r["score"]]
    if scored:
        lines.append(f"  mean name overlap on a hit: "
                     f"{sum(scored) / len(scored):.2f}")
    lines += ["",
              "  A Czech page spans two or three German pages, so the best single",
              "  German page holds only part of its names — the score measures "
              "'the right",
              "  district's page', not 'this page holds everything'.",
              "",
              "  Only `matched` rows may be used without saying so. `ambiguous` "
              "means two",
              "  German pages tied and the tie was refused rather than broken on "
              "lookup order."]
    return "\n".join(lines)


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n")[0])
    ap.add_argument("--pages", default="",
                    help="comma-separated Czech scans (default: the purposive five)")
    ap.add_argument("--all", action="store_true",
                    help=f"every Czech table page, {lxp.CZ_FIRST_SCAN}-{lxp.CZ_LAST_SCAN}")
    ap.add_argument("--write", action="store_true",
                    help=f"write {OUT_CSV.relative_to(ROOT)}")
    args = ap.parse_args()

    if args.all:
        scans = list(range(lxp.CZ_FIRST_SCAN, lxp.CZ_LAST_SCAN + 1))
    elif args.pages:
        scans = [int(p) for p in args.pages.split(",") if p.strip()]
    else:
        scans = list(lxp.PURPOSIVE_PAGES)

    plaintext = lxp.DE_PLAINTEXT
    if not plaintext.exists():
        print(f"no German plaintext at {plaintext}", file=sys.stderr)
        return 1

    rows = build(scans, plaintext)
    print()
    print(report(rows))
    if args.write:
        OUT_CSV.parent.mkdir(parents=True, exist_ok=True)
        with OUT_CSV.open("w", encoding="utf-8", newline="") as fh:
            writer = csv.DictWriter(fh, fieldnames=FIELDS)
            writer.writeheader()
            writer.writerows(rows)
        print(f"\n  wrote {OUT_CSV.relative_to(ROOT)}  ({len(rows)} rows)")
    else:
        print("\n  (dry run — pass --write to save)")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
