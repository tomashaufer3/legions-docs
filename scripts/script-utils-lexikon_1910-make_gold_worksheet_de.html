How it works

Companion to make_gold_worksheet, which builds the Czech sheets and is left alone. The German sheet has the same four sections and the same block grammar — pipe rows for the table, ledger fences for the two-column part, a legend and a notes fence — so read_transcription_md can cut it into the same blocks. Three things differ, and each follows from the source:

  • Nothing is pre-filled, the ledger keys included. The Czech sheet takes its municipality names and row counts from the ALTO text layer. The UBW scan has no ALTO and its embedded text is too damaged to print beside a blank cell (NiederalteDbuch, Altniynitz), so the transcriber types the key and the names as printed. An unfilled ledger line keeps the Czech sheets' __,+ template key, which the reader counts as blank and never as an entry.
  • The images are the page, not its sections. Without coordinates there is no section to cut, so each scan is rendered whole, plus an upper and a lower band that overlap, enlarged like the Czech crops.
  • The pages come from a manifest, not from a draw. A German page is not the twin of a Czech page: each edition orders a district's municipalities alphabetically in its own language. gold_sample_pages_de.csv names each German scan, the Czech scans it shares municipalities with, and whether the whole page is read or only one court's part of it.

The district in every heading is the Czech court name (soudní okres Trutnov), because that is the half of the join key the Czech gold already carries. The municipality number is not shared between the editions, so the two golds join on names, never on muni_seq.

A sheet that exists is never overwritten: it may hold an hour of reading.

def render(pdf, scan: int, out_dir: Path, force: bool = False) -> dict[str, str]:
    """Write the page and its two bands; return ``{kind: filename}``."""
    names = {kind: f"{scan:04d}_{kind}.png" for kind in ("page", "upper", "lower")}
    if not force and all((out_dir / name).exists() for name in names.values()):
        return names
    image = pdf.pages[scan - 1].to_image(resolution=RESOLUTION).original
    image.save(out_dir / names["page"])
    for kind, (top, bottom) in (("upper", UPPER_BAND), ("lower", LOWER_BAND)):
        band = image.crop((0, int(top * image.height),
                           image.width, int(bottom * image.height)))
        band = band.resize((band.width * ENLARGE, band.height * ENLARGE))
        band.save(out_dir / names[kind])
    return names

def sheet_text(row: dict, images: dict[str, str],
               volume: str = "1910 Spezialortsrepertorium, Böhmen",
               columns: list[str] | None = None) -> str:
    """The whole sheet for one manifest row."""
    scan = f"{int(row['de_scan']):04d}"
    court_de, court_cs = row["court_de"] or "________", row["court_cs"] or "________"
    where = f"Gerichtsbezirk {court_de} · soudní okres {court_cs}"
    scope = (SCOPE_COURT.format(courts_de=court_de)
             if row["read"] == "court" else "")
    header = MD_HEADER.format(
        scan=scan, role=row["role"], courts_cs=court_cs, courts_de=court_de,
        read=row["read"], cz_scans=", ".join(row["cz_scans"].split()),
        note=row["note"], scope=scope, volume=volume, twin=TWIN if row["cz_scans"] else "",
        **images)
    lines = [
        header,
        "# 1 — Tabelární část", "",
        f"image: `{images['upper']}`  (whole page: `{images['page']}`)", "",
        "One row per printed line, in printed order. The bold summary row of the",
        "court and the lines under it are rows too. Delete the rows left over;",
        "add rows where the print has more.", "",
        f"## 1 — {where}", "",
        *_table(int(row["rows"]), columns), "", "",
        "# 2 — Dvousloupcová část", "",
        f"image: `{images['lower']}`  (whole page: `{images['page']}`)", "",
        "One line per printed key. Replace `__,+` with the key as printed —",
        "municipality number, then the part number, `0` for a dash — then",
        "`municipality › part` and, after `>`, what the key prints:", "",
        "```",
        "   6,0     Hazlov               › (obec as a whole)        > Hammermühle [mill] 6-29, Hazlov, Haslau [ves] 217-2266.",
        "```", "",
        "A part that opens its own line is printed as a bare `2)` under the `1)`,",
        "with no dash. It is part 2 of the same municipality and gets its own line.",
        "", f"## 2a — Ortsbestandteile · {where}", "",
        *_ledger(), "",
        f"## 2b — Ämter, Anstalten usw. · {where}", "",
        "Symbols as atlas tokens, words verbatim. The school's class numeral and",
        "its language note, and the cooperative letters with their language",
        "suffix, are written exactly as printed.", "",
        *_ledger(), "", "",
        "# 3 — Jednosloupcová část pod čarou", "",
        "The postal districts and stations the last two table columns point at:",
        "`key) place`, in printed order. The numbering runs across the whole",
        "court, not down the page.", "",
        f"## 3 — [post] und [rail-station] · s. o. {court_cs}", "",
        "```legend", "", "```", "", "",
        "# 4 — Poznámky pod čarou", "",
        "The `*)` and `**)` notes, verbatim, one per line, with the marker.", "",
        "```notes", "", "```", "",
    ]
    return "\n".join(lines)

def build(role: str, images_only: bool = False, force_images: bool = False,
          volume: str | None = None) -> int:
    import pdfplumber

    rows = load_manifest(role, volume)
    if not rows:
        print(f"no {role} rows for {volume or MANIFEST.name}")
        return 1
    title, source, columns = "1910 Spezialortsrepertorium, Böhmen", DE_PDF, None
    out_dir = OUT_ROOT / role
    if volume is not None:
        from utils.lexikon_1910 import lexikon_1910_volume as lxvol

        profile = lxvol.load_profile(volume)
        # the volume's own columns: Silesia prints a Polish language column
        title, source, columns = profile.title, profile.source_pdf, [c.name for c in profile.columns]
        out_dir = OUT_ROOT.parent / VOLUMES[volume] / role
    out_dir.mkdir(parents=True, exist_ok=True)
    with pdfplumber.open(source) as pdf:
        for row in rows:
            scan = int(row["de_scan"])
            images = render(pdf, scan, out_dir, force=force_images)
            path = out_dir / f"{scan:04d}.md"
            if images_only:
                print(f"  {path.name}: images written")
                continue
            if path.exists():
                print(f"  {path.name}: exists — left alone")
                continue
            path.write_text(sheet_text(row, images, title, columns), encoding="utf-8")
            print(f"  {path.name}: written ({row['read']}, {row['rows']} blank rows"
                  + (f", twin of cz {row['cz_scans']})" if row["cz_scans"] else ")"))
    return 0

def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])

"""Build the gold worksheets for the German edition (Spezialortsrepertorium Böhmen).

Companion to :mod:`make_gold_worksheet`, which builds the Czech sheets and is left
alone. The German sheet has the same four sections and the same block grammar —
pipe rows for the table, ``ledger`` fences for the two-column part, a ``legend``
and a ``notes`` fence — so :mod:`read_transcription_md` can cut it into the same
blocks. Three things differ, and each follows from the source:

* **Nothing is pre-filled, the ledger keys included.** The Czech sheet takes its
  municipality names and row counts from the ALTO text layer. The UBW scan has no
  ALTO and its embedded text is too damaged to print beside a blank cell
  (``NiederalteDbuch``, ``Altniynitz``), so the transcriber types the key and the
  names as printed. An unfilled ledger line keeps the Czech sheets' ``__,+``
  template key, which the reader counts as blank and never as an entry.
* **The images are the page, not its sections.** Without coordinates there is no
  section to cut, so each scan is rendered whole, plus an upper and a lower band
  that overlap, enlarged like the Czech crops.
* **The pages come from a manifest, not from a draw.** A German page is not the
  twin of a Czech page: each edition orders a district's municipalities
  alphabetically in its own language. ``gold_sample_pages_de.csv`` names each
  German scan, the Czech scans it shares municipalities with, and whether the
  whole page is read or only one court's part of it.

The district in every heading is the **Czech** court name (``soudní okres
Trutnov``), because that is the half of the join key the Czech gold already
carries. The municipality number is *not* shared between the editions, so the two
golds join on names, never on ``muni_seq``.

A sheet that exists is never overwritten: it may hold an hour of reading.

    python utils/lexikon_1910/make_gold_worksheet_de.py --role purposive
    python utils/lexikon_1910/make_gold_worksheet_de.py --role purposive --images
    python utils/lexikon_1910/make_gold_worksheet_de.py --volume maehren_de --role certification

``--volume`` (1.1.0, D-133) cuts the sheets of Moravia's or Silesia's own drawn sample
(``gold_sample_pages_<slug>.csv``, from ``draw_gold_sample.py --volume``) into
``transcription_gold/de-moravia/`` or ``de-silesia/``. Those pages have no Czech twin and
no court is pre-filled: the court and every name are typed from the print.
"""

from __future__ import annotations

__version__ = "1.1.0"

import argparse
import csv
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from utils.lexikon_1910 import lexikon_1910 as lx                   # noqa: E402

DE_PDF = (ROOT / "library" / "lexicons"
          / "UBW_Spezialortsrepertorium-Boehmen-1910_de.pdf")
MANIFEST = ROOT / "reference" / "places" / "gold_sample_pages_de.csv"
OUT_ROOT = ROOT / "reference" / "places" / "transcription_gold" / "de"

#: The UBW scan's own resolution: its page images are ~2,106 px across a ~505 pt
#: page (``page.images[0]["srcsize"]`` on scans 477-481), which is 300 dpi.
#: Rendering finer than the scan adds pixels and no ink.
RESOLUTION = 300

#: The bands are enlarged by the same factor as the Czech section crops
#: (``make_gold_worksheet.ENLARGE``), so both editions are read at one size.
ENLARGE = 2

#: The table ends between about 55 % and 70 % of the way down the pages looked
#: at (scan 478: ~63 %). The two bands overlap across that whole range, so no
#: line is cut in both.
UPPER_BAND = (0.00, 0.70)
LOWER_BAND = (0.50, 1.00)

#: Blank ledger lines printed per section. A line is copied where more are needed.
LEDGER_LINES = 30

BLANK_KEY = "   __,+    ____________________ › ______________________________   >"

MD_HEADER = """---
scan: "{scan}"
edition: de
printed_page: ""         # as printed at the head of the page
role: {role}
districts: [{courts_cs}]
courts_de: [{courts_de}]
read: {read}            # whole = everything on the page | court = only the court above
matched_cz: [{cz_scans}]
status: draft            # draft | complete | double-coded
minutes: {{table: , parts: , offices: , legend: , footnotes: }}
images:
  page: {page}
  upper: {upper}
  lower: {lower}
---

<!--
GOLD TRANSCRIPTION — {volume} (German edition).
Scan {scan}.  {note}

Same rules as the Czech sheets. Names as printed here, German form first.{twin}

NOTHING IS PRE-FILLED. Type the sequence number, the names and the ledger keys
exactly as THIS edition prints them. The German edition numbers a district's
municipalities in its own alphabetical order, so a number here is not the number
of the same place in the Czech volume.

Fill from the IMAGE. A printed dash is a ZERO, a blank is a blank, and anything
printed but unreadable is `[?]`; never guess a digit.
{scope}-->
"""

SCOPE_COURT = """
READ ONLY GERICHTSBEZIRK {courts_de}. Everything on this scan that belongs to
another court is left out of every section.
"""

def _table(rows: int, names: list[str] | None = None) -> list[str]:
    columns = ["r", *(names or lx.COL_NAMES)]
    lines = ["| " + " | ".join(columns) + " |",
             "|" + "---|" * len(columns)]
    lines += [f"| r{k} |" + " |" * (len(columns) - 1) for k in range(1, rows + 1)]
    return lines


def _ledger() -> list[str]:
    return ["```ledger", *[BLANK_KEY] * LEDGER_LINES, "```"]

def build(role: str, images_only: bool = False, force_images: bool = False,
          volume: str | None = None) -> int:
    import pdfplumber

    rows = load_manifest(role, volume)
    if not rows:
        print(f"no {role} rows for {volume or MANIFEST.name}")
        return 1
    title, source, columns = "1910 Spezialortsrepertorium, Böhmen", DE_PDF, None
    out_dir = OUT_ROOT / role
    if volume is not None:
        from utils.lexikon_1910 import lexikon_1910_volume as lxvol

        profile = lxvol.load_profile(volume)
        # the volume's own columns: Silesia prints a Polish language column
        title, source, columns = profile.title, profile.source_pdf, [c.name for c in profile.columns]
        out_dir = OUT_ROOT.parent / VOLUMES[volume] / role
    out_dir.mkdir(parents=True, exist_ok=True)
    with pdfplumber.open(source) as pdf:
        for row in rows:
            scan = int(row["de_scan"])
            images = render(pdf, scan, out_dir, force=force_images)
            path = out_dir / f"{scan:04d}.md"
            if images_only:
                print(f"  {path.name}: images written")
                continue
            if path.exists():
                print(f"  {path.name}: exists — left alone")
                continue
            path.write_text(sheet_text(row, images, title, columns), encoding="utf-8")
            print(f"  {path.name}: written ({row['read']}, {row['rows']} blank rows"
                  + (f", twin of cz {row['cz_scans']})" if row["cz_scans"] else ")"))
    return 0


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--role", default="purposive",
                    choices=("calibration", "certification", "purposive"))
    ap.add_argument("--images", action="store_true",
                    help="re-render the page images only; touch no sheet")
    ap.add_argument("--volume", choices=sorted(VOLUMES),
                    help="Moravia's or Silesia's own drawn sample (D-133)")
    args = ap.parse_args()
    return build(args.role, images_only=args.images, force_images=args.images,
                 volume=args.volume)
