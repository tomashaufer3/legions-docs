How it works

Ostafin, Kaim, Siwek & Miklar (2020, Scientific Data 7:208; Harvard Dataverse doi:10.7910/DVN/K7YPAF, CC0) give every commune of Austrian Silesia, and the parts they drew on their maps, with the 1910 census — population present, sex, houses, the language of daily use, religion — typed from the same Spezialortsrepertorium Schlesien (1917) that the register's full run read, plus court-district totals from the census. It is the only outside transcription of a German volume of the register, so it tests what the sealed pages cannot: the table (D-160), commune by commune, over the whole volume. Three comparisons:

census the 1910 census this project uses (data/final/census/census_okres_1910, MPIDR / Rumpler-Seger) against Ostafin's court districts summed to the political district; register the full run's table readings (run_de_table, one row per printed line, each reader apart) against Ostafin's communes, joined on court and German name, or on the name alone where it is unique in the volume — the readers' court label sometimes carries a stray mark (Olbersdorf 18); lexikon Ostafin's present-day units (N_2019GP) against the ČSÚ lexicon 1869–2005, díl I (lexikon_population), whose 1910 figures are recomputed onto 2005 parts — a test of the lexicon's units, not of anyone's reading. Each difference is given its cause.

Two definitional facts carry every result and are written into the report rather than corrected: Ostafin's commune-name field changes within a commune (commune 006 of Freiwaldau is "Gröditz" on one part and "Neudorf" on the other), so their communes are grouped by number and named after the first part; and Ostafin's Arha_10HM is measured on their maps, not copied from the print, so area is not compared.

Validation only; reads data/final, the full run's pass files and data/raw, writes output/validation/ostafin_silesia.{json,txt} and nothing else.

def key(name) -> str:
    """A place name reduced to ASCII letters: entities unescaped (``&uuml;``), the Cyrillic
    ``ӓ`` Ostafin types for ``ä`` mended, accents and punctuation dropped."""
    s = html.unescape(str(name)).replace("ӓ", "ä")
    s = unicodedata.normalize("NFKD", s).encode("ascii", "ignore").decode().lower()
    return re.sub(r"[^a-z]", "", s)

def ostafin(level: str) -> pd.DataFrame:
    df = pd.DataFrame(read_dbf(OSTAFIN / f"SA_1910_{level}" / f"SA_1910_{level}.dbf",
                               encodings=("utf-8",)))
    for c in df.columns:
        if re.match(r"(P|H|Pm|Pwm|Pger|Pcms|Ppl|Pstr|Pcr|Pev|Pisr)[a-z]*_10", c):
            df[c] = pd.to_numeric(df[c])
    return df

def communes(cm: pd.DataFrame) -> pd.DataFrame:
    """Ostafin's communes (court × number), parts summed, named after the first part; ``k`` is
    court + name. A name two communes of one court share is left out."""
    first = cm.sort_values("CmPp").groupby(["N_DC", "CmP"])["N_CmP"].first()
    com = cm.groupby(["N_DC", "CmP"], as_index=False)[list(FIELDS.values())].sum()
    com["name"] = [first[(a, b)] for a, b in zip(com.N_DC, com.CmP)]
    com["k"] = com.N_DC.map(key) + "#" + com.name.map(key)
    return com[~com.k.duplicated(keep=False)]

def census_check(dc: pd.DataFrame) -> dict:
    """Our census against Ostafin's courts summed to the political district."""
    ce = pd.read_parquet(CENSUS)
    ce = ce[ce.land.astype(str).str.contains("Schles", case=False)]
    pairs = {"pop_total": "P_10DL", "lang_german": "Pger_10DL", "lang_czech": "Pcms_10DL",
             "lang_polish": "Ppl_10DL"}
    theirs = dc.groupby("N_DP")[list(pairs.values())].sum()
    theirs.index = [key(n.replace(" Stadt", "")) + ("stadt" if "Stadt" in n else "") for n in theirs.index]
    ce.index = [key(str(de).replace("ae", "a")) + ("stadt" if "město" in str(cs) else "")
                for de, cs in zip(ce.okres_de, ce.okres_cs)]
    j = ce[list(pairs)].join(theirs, how="outer")
    out = {"districts": len(j), "unmatched": int(j[["pop_total", "P_10DL"]].isna().any(axis=1).sum())}
    for ours, col in pairs.items():
        have = j[ours].notna() & j[col].notna()
        out[ours] = {"compared": int(have.sum()), "equal": int((j[ours] == j[col])[have].sum()),
                     "ours": int(j[ours][have].sum()), "ostafin": int(j[col][have].sum())}
    return out

def table_rows() -> dict[str, pd.DataFrame]:
    """The full run's table readings, one frame per reader, commune lines only. The pass
    files sit under the machine that ran it (``bakeoff/<host>/de``), named in its freeze."""
    import bakeoff as bo
    import run_frozen as rf

    manifest = rf.ROOT_OUT / f"{EDITION}_{PAGES}" / "freeze" / "manifest.json"
    host = json.loads(manifest.read_text(encoding="utf-8"))["host"]
    out = {}
    for label, model in rf.labels(PAGES, EDITION)["table"].items():
        path = bo.BAKEOFF / host / "de" / f"table_rows_{label}.csv"
        if path.is_file():
            r = pd.read_csv(path, dtype=str)
            out[model] = r[(r.block_kind == "municipality") & (r.row == "0")].copy()
    return out

def join_register(r: pd.DataFrame, com: pd.DataFrame) -> pd.DataFrame:
    """A reader's commune lines on Ostafin's communes: court + name, else the name alone where
    it is unique among Ostafin's communes and among the reader's lines. One row per commune."""
    r = r.assign(k=r.court_de.map(key) + "#" + r.name.map(german_name).map(key)).drop_duplicates("k")
    name = com.k.str.split("#").str[1]
    unique = name[~name.duplicated(keep=False)]
    alone = dict(zip(unique, com.k[unique.index]))
    rname = r.k.str.split("#").str[1]
    fall = ~r.k.isin(com.k) & rname.isin(alone) & ~rname.duplicated(keep=False)
    r.loc[fall, "k"] = rname[fall].map(alone)
    return r.drop_duplicates("k").merge(com, on="k", how="inner", validate="1:1")

def register_check(com: pd.DataFrame) -> dict:
    out = {}
    for model, r in table_rows().items():
        m = join_register(r, com)
        fields = {ours: {"equal": int((digits(m[ours]) == m[col]).sum()),
                         "ours": int(digits(m[ours]).sum()), "ostafin": int(m[col].sum())}
                  for ours, col in FIELDS.items()}
        diff = m.assign(ours=digits(m.pop_total))[lambda d: d.ours != d.P_10CL]
        out[model] = {"read": len(r), "joined": len(m), "communes": len(com), "fields": fields,
                      "differ": [{"page": p, "line": n, "ours": int(o), "ostafin": int(t)}
                                 for p, n, o, t in zip(diff.page, diff.name_x, diff.ours, diff.P_10CL)]}
    return out

def lexicon_candidates(lx: pd.DataFrame) -> pd.DataFrame:
    """The lexicon rows a present-day unit can match: every part, and each obec printed
    without parts (it is its own part)."""
    with_parts = set(zip(lx[lx.level == "part"].okres_2005, lx[lx.level == "part"].obec))
    lone = lx[(lx.level == "obec").to_numpy() & [t not in with_parts for t in zip(lx.okres_2005, lx.obec)]]
    return pd.concat([lx[lx.level == "part"], lone]).assign(fold=lambda d: d.name.str.lower().str.strip())

def join_lexicon(cm: pd.DataFrame, cand: pd.DataFrame) -> pd.DataFrame:
    """Ostafin's present-day units on the lexicon: the folded name, exact, within the 2005
    okresy the 1910 district became. A name two lexicon rows share there is a join problem."""
    units = cm.groupby(["N_DP", "N_2019GP"], as_index=False)["P_10CL"].sum()
    rows = []
    for dp, name, pop in zip(units.N_DP, units.N_2019GP, units.P_10CL):
        pool = cand[cand.okres_2005.isin(DISTRICT_OKRESY.get(html.unescape(dp).replace("ӓ", "ä"), set()))]
        hit = pool[pool.fold == lp.fold(name)]
        if len(hit):
            h = hit.iloc[0]
            rows.append({"district": dp, "name": name, "ostafin": int(pop), "lexikon": h.pop_1910,
                         "okres_2005": h.okres_2005, "lex_key": h.key, "note_1910": h.note_1910,
                         "ambiguous": len(hit) > 1})
    m = pd.DataFrame(rows)
    m["claimed_twice"] = m.duplicated(["okres_2005", "lex_key"], keep=False)
    return m[m.lexikon.notna()]

def cause(row, cm: pd.DataFrame) -> str:
    """Why a present-day unit's 1910 figure differs between Ostafin and the lexicon."""
    if row.ostafin == row.lexikon:
        return "equal"
    if row.note_1910:
        return "the lexicon's note: its 2005 part includes or moves other 1910 places"
    if row.ambiguous or row.claimed_twice:
        return "join: a name two lexicon rows share, or two units on one row"
    own = cm[(cm.N_DP == row.district) & (cm.N_2019GP == row["name"])]
    communes_of = set(zip(own.N_DC, own.CmP))
    in_commune = cm[[t in communes_of for t in zip(cm.N_DC, cm.CmP)]]
    if in_commune.P_10CL.sum() == row.lexikon:
        return "the lexicon gives the whole 1910 commune to one 2005 part"
    others = in_commune[in_commune.N_2019GP != row["name"]].groupby("N_2019GP").P_10CL.sum()
    for n in range(1, min(4, len(others)) + 1):
        if any(row.ostafin + others[list(c)].sum() == row.lexikon for c in itertools.combinations(others.index, n)):
            return "the lexicon's 2005 part holds other parts of the same 1910 commune"
    return "unexplained: one takes the commune, the other a settlement of it"

def lexicon_check(cm: pd.DataFrame) -> dict:
    m = join_lexicon(cm, lexicon_candidates(lp.read()))
    m["cause"] = [cause(r, cm) for _, r in m.iterrows()]
    larger = m[m.ostafin != m.lexikon]
    return {"units_ostafin": int(cm.groupby(["N_DP", "N_2019GP"]).ngroups),
            "matched": len(m), "equal": int((m.cause == "equal").sum()),
            "lexicon_larger": int((larger.lexikon > larger.ostafin).sum()),
            "causes": m.cause.value_counts().to_dict()}

def report(rec: dict) -> str:
    c, lines = rec["census"], [f"Austrian Silesia 1910 against Ostafin et al. (2020) — "
                               f"check_ostafin_silesia {__version__}, {rec['run']}", ""]
    lines.append(f"census: {c['districts']} political districts, {c['unmatched']} unmatched")
    for f in ("pop_total", "lang_german", "lang_czech", "lang_polish"):
        v = c[f]
        lines.append(f"  {f:12s} equal {v['equal']}/{v['compared']}   ours {v['ours']:,}  Ostafin {v['ostafin']:,}")
    for model, v in rec["register"].items():
        lines += ["", f"register, {model}: {v['joined']} of Ostafin's {v['communes']} communes joined "
                      f"({v['read']} commune lines read)"]
        for f, x in v["fields"].items():
            lines.append(f"  {f:17s} equal {x['equal'] / v['joined']:6.1%}   ours {x['ours']:>9,}  Ostafin {x['ostafin']:>9,}")
        lines.append(f"  total population differs on {len(v['differ'])}: "
                     + "; ".join(f"{d['line']} {d['ours']} vs {d['ostafin']}" for d in v["differ"][:6]))
    x = rec["lexikon"]
    lines += ["", f"lexikon 1869–2005: {x['matched']} of Ostafin's {x['units_ostafin']} present-day units "
                  f"matched, {x['equal']} equal; of the rest the lexicon is larger on {x['lexicon_larger']}"]
    lines += [f"  {n:3d}  {k}" for k, n in x["causes"].items()]
    return "\n".join(lines)

def main() -> int:
    dc, cm = ostafin("DC"), ostafin("CmP")
    rec = {"check": __version__, "run": time.strftime("%Y-%m-%d %H:%M"),
           "census": census_check(dc), "register": register_check(communes(cm)),
           "lexikon": lexicon_check(cm)}
    OUT.mkdir(parents=True, exist_ok=True)
    (OUT / "ostafin_silesia.json").write_text(json.dumps(rec, indent=1, ensure_ascii=False), encoding="utf-8")
    text = report(rec)
    (OUT / "ostafin_silesia.txt").write_text(text, encoding="utf-8")
    print(text)
    return 0

"""Austrian Silesia 1910 against Ostafin et al. (2020): an independent transcription of our source.

Ostafin, Kaim, Siwek & Miklar (2020, *Scientific Data* 7:208; Harvard Dataverse
doi:10.7910/DVN/K7YPAF, CC0) give every commune of Austrian Silesia, and the parts they drew
on their maps, with the 1910 census — population present, sex, houses, the language of daily
use, religion — **typed from the same Spezialortsrepertorium Schlesien (1917)** that the
register's full run read, plus court-district totals from the census. It is the only outside
transcription of a German volume of the register, so it tests what the sealed pages cannot:
the **table** (D-160), commune by commune, over the whole volume. Three comparisons:

``census``
    the 1910 census this project uses (``data/final/census/census_okres_1910``, MPIDR /
    Rumpler-Seger) against Ostafin's court districts summed to the political district;
``register``
    the full run's table readings (``run_de_table``, one row per printed line, each reader
    apart) against Ostafin's communes, joined on court and German name, or on the name alone
    where it is unique in the volume — the readers' court label sometimes carries a stray
    mark (``Olbersdorf 18``);
``lexikon``
    Ostafin's present-day units (``N_2019GP``) against the ČSÚ lexicon 1869–2005, díl I
    (:mod:`lexikon_population`), whose 1910 figures are recomputed onto 2005 parts — a test
    of the lexicon's units, not of anyone's reading. Each difference is given its cause.

Two definitional facts carry every result and are written into the report rather than
corrected: Ostafin's commune-name field changes within a commune (commune 006 of Freiwaldau
is "Gröditz" on one part and "Neudorf" on the other), so their communes are grouped by number
and named after the first part; and Ostafin's ``Arha_10HM`` is measured on their maps, not
copied from the print, so area is not compared.

Validation only; reads ``data/final``, the full run's pass files and ``data/raw``, writes
``output/validation/ostafin_silesia.{json,txt}`` and nothing else.

    python utils/validation/check_ostafin_silesia.py
"""
from __future__ import annotations

__version__ = "0.1.0"

import html
import itertools
import json
import re
import sys
import time
import unicodedata
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))
sys.path.insert(0, str(ROOT / "utils" / "lexikon_1910"))

from utils.parsing.districts import read_dbf                     # noqa: E402
from utils.validation import lexikon_population as lp            # noqa: E402

OSTAFIN = ROOT / "data" / "raw" / "geo" / "ostafin_silesia"
CENSUS = ROOT / "data" / "final" / "census" / "census_okres_1910.parquet"
OUT = ROOT / "output" / "validation"
EDITION, PAGES = "de-silesia", "full"
#: Our table column → Ostafin's commune field (area left out: theirs is measured on the map).
FIELDS = {"pop_total": "P_10CL", "pop_male": "Pm_10CL", "pop_female": "Pwm_10CL",
          "houses": "Hs_10CL", "lang_german": "Pger_10CL", "lang_czech": "Pcms_10CL",
          "lang_polish": "Ppl_10CL", "foreigners": "Pstr_10CL", "conf_catholic": "Pcr_10CL",
          "conf_evangelical": "Pev_10CL", "conf_israelite": "Pisr_10CL"}
#: The 2005 okresy each 1910 political district lies in, for the lexicon join (the Polish half
#: of Teschen and Bielitz has no lexicon).
DISTRICT_OKRESY = {"Freiwaldau": {"Jeseník"}, "Freudenthal": {"Bruntál", "Opava"},
                   "Jägerndorf": {"Bruntál", "Opava"}, "Troppau": {"Opava", "Nový Jičín", "Bruntál"},
                   "Troppau Stadt": {"Opava"}, "Wagstadt": {"Nový Jičín", "Opava"},
                   "Freistadt": {"Karviná", "Ostrava-město"},
                   "Friedek": {"Frýdek-Místek", "Ostrava-město", "Karviná"},
                   "Friedek Stadt": {"Frýdek-Místek"}, "Teschen": {"Karviná", "Frýdek-Místek"}}


def key(name) -> str:
    """A place name reduced to ASCII letters: entities unescaped (``&uuml;``), the Cyrillic
    ``ӓ`` Ostafin types for ``ä`` mended, accents and punctuation dropped."""
    s = html.unescape(str(name)).replace("ӓ", "ä")
    s = unicodedata.normalize("NFKD", s).encode("ascii", "ignore").decode().lower()
    return re.sub(r"[^a-z]", "", s)


def german_name(printed) -> str:
    """The German form a register line starts with: ``Orlau, Orlová, Orłowa [?]*)`` → Orlau."""
    return re.sub(r"\[\?\]|\*|\)|\(.*", "", str(printed).split(",")[0]).strip()


def digits(col: pd.Series) -> pd.Series:
    """A read cell as a number: marks dropped (``*34654`` → 34654), an empty cell 0, as the
    print's dot for nothing."""
    return pd.to_numeric(col.astype(str).str.replace(r"[^\d]", "", regex=True).replace("", None),
                         errors="coerce").fillna(0)


def ostafin(level: str) -> pd.DataFrame:
    df = pd.DataFrame(read_dbf(OSTAFIN / f"SA_1910_{level}" / f"SA_1910_{level}.dbf",
                               encodings=("utf-8",)))
    for c in df.columns:
        if re.match(r"(P|H|Pm|Pwm|Pger|Pcms|Ppl|Pstr|Pcr|Pev|Pisr)[a-z]*_10", c):
            df[c] = pd.to_numeric(df[c])
    return df


def communes(cm: pd.DataFrame) -> pd.DataFrame:
    """Ostafin's communes (court × number), parts summed, named after the first part; ``k`` is
    court + name. A name two communes of one court share is left out."""
    first = cm.sort_values("CmPp").groupby(["N_DC", "CmP"])["N_CmP"].first()
    com = cm.groupby(["N_DC", "CmP"], as_index=False)[list(FIELDS.values())].sum()
    com["name"] = [first[(a, b)] for a, b in zip(com.N_DC, com.CmP)]
    com["k"] = com.N_DC.map(key) + "#" + com.name.map(key)
    return com[~com.k.duplicated(keep=False)]


# ── census ───────────────────────────────────────────────────────────────────
def census_check(dc: pd.DataFrame) -> dict:
    """Our census against Ostafin's courts summed to the political district."""
    ce = pd.read_parquet(CENSUS)
    ce = ce[ce.land.astype(str).str.contains("Schles", case=False)]
    pairs = {"pop_total": "P_10DL", "lang_german": "Pger_10DL", "lang_czech": "Pcms_10DL",
             "lang_polish": "Ppl_10DL"}
    theirs = dc.groupby("N_DP")[list(pairs.values())].sum()
    theirs.index = [key(n.replace(" Stadt", "")) + ("stadt" if "Stadt" in n else "") for n in theirs.index]
    ce.index = [key(str(de).replace("ae", "a")) + ("stadt" if "město" in str(cs) else "")
                for de, cs in zip(ce.okres_de, ce.okres_cs)]
    j = ce[list(pairs)].join(theirs, how="outer")
    out = {"districts": len(j), "unmatched": int(j[["pop_total", "P_10DL"]].isna().any(axis=1).sum())}
    for ours, col in pairs.items():
        have = j[ours].notna() & j[col].notna()
        out[ours] = {"compared": int(have.sum()), "equal": int((j[ours] == j[col])[have].sum()),
                     "ours": int(j[ours][have].sum()), "ostafin": int(j[col][have].sum())}
    return out


# ── register ─────────────────────────────────────────────────────────────────
def table_rows() -> dict[str, pd.DataFrame]:
    """The full run's table readings, one frame per reader, commune lines only. The pass
    files sit under the machine that ran it (``bakeoff/<host>/de``), named in its freeze."""
    import bakeoff as bo
    import run_frozen as rf

    manifest = rf.ROOT_OUT / f"{EDITION}_{PAGES}" / "freeze" / "manifest.json"
    host = json.loads(manifest.read_text(encoding="utf-8"))["host"]
    out = {}
    for label, model in rf.labels(PAGES, EDITION)["table"].items():
        path = bo.BAKEOFF / host / "de" / f"table_rows_{label}.csv"
        if path.is_file():
            r = pd.read_csv(path, dtype=str)
            out[model] = r[(r.block_kind == "municipality") & (r.row == "0")].copy()
    return out


def join_register(r: pd.DataFrame, com: pd.DataFrame) -> pd.DataFrame:
    """A reader's commune lines on Ostafin's communes: court + name, else the name alone where
    it is unique among Ostafin's communes and among the reader's lines. One row per commune."""
    r = r.assign(k=r.court_de.map(key) + "#" + r.name.map(german_name).map(key)).drop_duplicates("k")
    name = com.k.str.split("#").str[1]
    unique = name[~name.duplicated(keep=False)]
    alone = dict(zip(unique, com.k[unique.index]))
    rname = r.k.str.split("#").str[1]
    fall = ~r.k.isin(com.k) & rname.isin(alone) & ~rname.duplicated(keep=False)
    r.loc[fall, "k"] = rname[fall].map(alone)
    return r.drop_duplicates("k").merge(com, on="k", how="inner", validate="1:1")


def register_check(com: pd.DataFrame) -> dict:
    out = {}
    for model, r in table_rows().items():
        m = join_register(r, com)
        fields = {ours: {"equal": int((digits(m[ours]) == m[col]).sum()),
                         "ours": int(digits(m[ours]).sum()), "ostafin": int(m[col].sum())}
                  for ours, col in FIELDS.items()}
        diff = m.assign(ours=digits(m.pop_total))[lambda d: d.ours != d.P_10CL]
        out[model] = {"read": len(r), "joined": len(m), "communes": len(com), "fields": fields,
                      "differ": [{"page": p, "line": n, "ours": int(o), "ostafin": int(t)}
                                 for p, n, o, t in zip(diff.page, diff.name_x, diff.ours, diff.P_10CL)]}
    return out


# ── lexicon ──────────────────────────────────────────────────────────────────
def lexicon_candidates(lx: pd.DataFrame) -> pd.DataFrame:
    """The lexicon rows a present-day unit can match: every part, and each obec printed
    without parts (it is its own part)."""
    with_parts = set(zip(lx[lx.level == "part"].okres_2005, lx[lx.level == "part"].obec))
    lone = lx[(lx.level == "obec").to_numpy() & [t not in with_parts for t in zip(lx.okres_2005, lx.obec)]]
    return pd.concat([lx[lx.level == "part"], lone]).assign(fold=lambda d: d.name.str.lower().str.strip())


def join_lexicon(cm: pd.DataFrame, cand: pd.DataFrame) -> pd.DataFrame:
    """Ostafin's present-day units on the lexicon: the folded name, exact, within the 2005
    okresy the 1910 district became. A name two lexicon rows share there is a join problem."""
    units = cm.groupby(["N_DP", "N_2019GP"], as_index=False)["P_10CL"].sum()
    rows = []
    for dp, name, pop in zip(units.N_DP, units.N_2019GP, units.P_10CL):
        pool = cand[cand.okres_2005.isin(DISTRICT_OKRESY.get(html.unescape(dp).replace("ӓ", "ä"), set()))]
        hit = pool[pool.fold == lp.fold(name)]
        if len(hit):
            h = hit.iloc[0]
            rows.append({"district": dp, "name": name, "ostafin": int(pop), "lexikon": h.pop_1910,
                         "okres_2005": h.okres_2005, "lex_key": h.key, "note_1910": h.note_1910,
                         "ambiguous": len(hit) > 1})
    m = pd.DataFrame(rows)
    m["claimed_twice"] = m.duplicated(["okres_2005", "lex_key"], keep=False)
    return m[m.lexikon.notna()]


def cause(row, cm: pd.DataFrame) -> str:
    """Why a present-day unit's 1910 figure differs between Ostafin and the lexicon."""
    if row.ostafin == row.lexikon:
        return "equal"
    if row.note_1910:
        return "the lexicon's note: its 2005 part includes or moves other 1910 places"
    if row.ambiguous or row.claimed_twice:
        return "join: a name two lexicon rows share, or two units on one row"
    own = cm[(cm.N_DP == row.district) & (cm.N_2019GP == row["name"])]
    communes_of = set(zip(own.N_DC, own.CmP))
    in_commune = cm[[t in communes_of for t in zip(cm.N_DC, cm.CmP)]]
    if in_commune.P_10CL.sum() == row.lexikon:
        return "the lexicon gives the whole 1910 commune to one 2005 part"
    others = in_commune[in_commune.N_2019GP != row["name"]].groupby("N_2019GP").P_10CL.sum()
    for n in range(1, min(4, len(others)) + 1):
        if any(row.ostafin + others[list(c)].sum() == row.lexikon for c in itertools.combinations(others.index, n)):
            return "the lexicon's 2005 part holds other parts of the same 1910 commune"
    return "unexplained: one takes the commune, the other a settlement of it"


def lexicon_check(cm: pd.DataFrame) -> dict:
    m = join_lexicon(cm, lexicon_candidates(lp.read()))
    m["cause"] = [cause(r, cm) for _, r in m.iterrows()]
    larger = m[m.ostafin != m.lexikon]
    return {"units_ostafin": int(cm.groupby(["N_DP", "N_2019GP"]).ngroups),
            "matched": len(m), "equal": int((m.cause == "equal").sum()),
            "lexicon_larger": int((larger.lexikon > larger.ostafin).sum()),
            "causes": m.cause.value_counts().to_dict()}


# ── report ───────────────────────────────────────────────────────────────────
def report(rec: dict) -> str:
    c, lines = rec["census"], [f"Austrian Silesia 1910 against Ostafin et al. (2020) — "
                               f"check_ostafin_silesia {__version__}, {rec['run']}", ""]
    lines.append(f"census: {c['districts']} political districts, {c['unmatched']} unmatched")
    for f in ("pop_total", "lang_german", "lang_czech", "lang_polish"):
        v = c[f]
        lines.append(f"  {f:12s} equal {v['equal']}/{v['compared']}   ours {v['ours']:,}  Ostafin {v['ostafin']:,}")
    for model, v in rec["register"].items():
        lines += ["", f"register, {model}: {v['joined']} of Ostafin's {v['communes']} communes joined "
                      f"({v['read']} commune lines read)"]
        for f, x in v["fields"].items():
            lines.append(f"  {f:17s} equal {x['equal'] / v['joined']:6.1%}   ours {x['ours']:>9,}  Ostafin {x['ostafin']:>9,}")
        lines.append(f"  total population differs on {len(v['differ'])}: "
                     + "; ".join(f"{d['line']} {d['ours']} vs {d['ostafin']}" for d in v["differ"][:6]))
    x = rec["lexikon"]
    lines += ["", f"lexikon 1869–2005: {x['matched']} of Ostafin's {x['units_ostafin']} present-day units "
                  f"matched, {x['equal']} equal; of the rest the lexicon is larger on {x['lexicon_larger']}"]
    lines += [f"  {n:3d}  {k}" for k, n in x["causes"].items()]
    return "\n".join(lines)


def main() -> int:
    dc, cm = ostafin("DC"), ostafin("CmP")
    rec = {"check": __version__, "run": time.strftime("%Y-%m-%d %H:%M"),
           "census": census_check(dc), "register": register_check(communes(cm)),
           "lexikon": lexicon_check(cm)}
    OUT.mkdir(parents=True, exist_ok=True)
    (OUT / "ostafin_silesia.json").write_text(json.dumps(rec, indent=1, ensure_ascii=False), encoding="utf-8")
    text = report(rec)
    (OUT / "ostafin_silesia.txt").write_text(text, encoding="utf-8")
    print(text)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
