TL;DR

Downloads the name-level archival inventories (inventáře) of three Habsburg military courts in the Czech lands from the Vojenský historický archiv (VÚA–VHA Praha, Tzv. rakouské fondy do roku 1918) into data/raw/vua_courts/ and records where each file came from. Parsing is utils.parsing.vua_courts; this module only fetches.

How it works

Three files, three requests. The rules it keeps:

  • robots.txt is read first and obeyed;
  • requests are sequential, at least MIN_DELAY seconds apart, one attempt each — a failure is reported, never retried in a loop;
  • a file already on disk is not fetched again (force=True to override);
  • the User-Agent names this script and its version — a script, not a browser, and not the person running it. No name, e-mail address or other personal detail is sent;
  • TLS stays verified. The server does not send its intermediate certificate, so Python's bundled CA list fails with "unable to get local issuer certificate"; the operating system's trust store completes the chain, and truststore hands that store to requests. (scraper_vuapraha.py switches verification off instead; this module does not.)

Every request is appended to fetch_log.csv (time UTC, URL, status, bytes, sha256) and SOURCE.txt is rewritten from that log, so the retrieval date in the provenance is the one the build recorded, not one typed by hand.

python utils/scraping/fetch_vua_courts.py

def write_source_note(out_dir: Path = DEFAULT_OUT_DIR) -> Path:
    """Rewrite ``SOURCE.txt`` from ``fetch_log.csv`` (last successful fetch per file)."""
    log = out_dir / "fetch_log.csv"
    last: dict[str, dict] = {}
    if log.exists():
        with log.open(encoding="utf-8", newline="") as fh:
            for row in csv.DictReader(fh):
                if row["status"] == "200":
                    last[row["file"]] = row
    lines = [
        "Source", "------",
        "Vojenský ústřední archiv – Vojenský historický archiv Praha (VÚA–VHA),",
        "Tzv. rakouské fondy do roku 1918: archival inventories (inventáře) of three",
        "Habsburg military courts in the Czech lands, published as born-digital PDFs.",
        "", "Files", "-----",
    ]
    for court, (name, url) in INVENTORIES.items():
        row = last.get(name)
        got = (f"retrieved {row['fetched_utc']} UTC, {int(row['bytes']):,} bytes, "
               f"sha256 {row['sha256']}") if row else "NOT FETCHED"
        lines += [f"{name}   [{court}]", f"  {url}", f"  {got}", ""]
    lines += [
        "Rights", "------",
        "Published by the archive on its own website as finding aids. Retrieved for",
        "scientific research (par. 90 zakon c. 121/2000 Sb.) by",
        f"utils/scraping/fetch_vua_courts.py v{__version__}: one request per file,",
        f"sequential, >= {MIN_DELAY:.0f} s apart. The PDFs stay local (data/ is gitignored).",
        "",
    ]
    path = out_dir / "SOURCE.txt"
    path.write_text("\n".join(lines), encoding="utf-8")
    return path

def fetch_inventories(out_dir: Path = DEFAULT_OUT_DIR, *, force: bool = False,
                      delay: float = MIN_DELAY) -> dict[str, Path]:
    """Fetch every inventory not already on disk; returns ``{court: path}`` of files held."""
    if delay < MIN_DELAY:
        raise ValueError(f"delay={delay} s is below the {MIN_DELAY} s floor")
    out_dir.mkdir(parents=True, exist_ok=True)
    held: dict[str, Path] = {}
    todo = []
    for court, (name, url) in INVENTORIES.items():
        path = out_dir / name
        if path.exists() and not force:
            print(f"  have   {name}")
            held[court] = path
        else:
            todo.append((court, name, url, path))
    if not todo:
        write_source_note(out_dir)
        return held

    blocked = _disallowed([url for _, _, url, _ in todo])
    if blocked:
        raise PermissionError(f"robots.txt disallows: {blocked}")
    for court, name, url, path in todo:
        time.sleep(delay)
        stamp = datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M:%S")
        row = {"fetched_utc": stamp, "court": court, "file": name, "url": url,
               "status": "", "bytes": 0, "sha256": ""}
        try:
            resp = requests.get(url, headers={"User-Agent": USER_AGENT}, timeout=TIMEOUT)
            row["status"] = str(resp.status_code)
            if resp.status_code == 200 and resp.content[:5] == b"%PDF-":
                path.write_bytes(resp.content)
                row["bytes"] = len(resp.content)
                row["sha256"] = hashlib.sha256(resp.content).hexdigest()
                held[court] = path
                print(f"  fetched {name}  ({len(resp.content):,} bytes)")
            else:
                if resp.status_code == 200:
                    row["status"] = "200-not-a-pdf"
                print(f"  FAILED  {name}  (status {row['status']}) — not retried")
        except requests.RequestException as exc:
            row["status"] = type(exc).__name__
            print(f"  FAILED  {name}  ({row['status']}) — not retried")
        _append_log(out_dir, row)
    write_source_note(out_dir)
    return held

#!/usr/bin/env python3
"""
fetch_vua_courts.py — the three VÚA military-court inventories, fetched once
============================================================================
Downloads the name-level archival inventories (*inventáře*) of three Habsburg military
courts in the Czech lands from the Vojenský historický archiv (VÚA–VHA Praha, *Tzv.
rakouské fondy do roku 1918*) into ``data/raw/vua_courts/`` and records where each file
came from. Parsing is :mod:`utils.parsing.vua_courts`; this module only fetches.

Three files, three requests. The rules it keeps:

* ``robots.txt`` is read first and obeyed;
* requests are **sequential**, at least :data:`MIN_DELAY` seconds apart, one attempt each —
  a failure is reported, never retried in a loop;
* a file already on disk is **not** fetched again (``force=True`` to override);
* the ``User-Agent`` names this script and its version — a script, not a browser, and not
  the person running it. No name, e-mail address or other personal detail is sent;
* TLS stays **verified**. The server does not send its intermediate certificate, so
  Python's bundled CA list fails with "unable to get local issuer certificate"; the
  operating system's trust store completes the chain, and ``truststore`` hands that store
  to ``requests``. (``scraper_vuapraha.py`` switches verification off instead; this
  module does not.)

Every request is appended to ``fetch_log.csv`` (time UTC, URL, status, bytes, sha256) and
``SOURCE.txt`` is rewritten from that log, so the retrieval date in the provenance is the
one the build recorded, not one typed by hand.

Usage::

    python utils/scraping/fetch_vua_courts.py
"""

from __future__ import annotations

import csv
import hashlib
import sys
import time
from datetime import datetime, timezone
from pathlib import Path
from urllib import robotparser

import requests

try:                                   # verify against the OS trust store (see docstring)
    import truststore
    truststore.inject_into_ssl()
except ImportError:                    # pragma: no cover — plain certifi; may fail on this host
    truststore = None

__version__ = "0.1.0"

ROOT = Path(__file__).resolve().parents[2]
DEFAULT_OUT_DIR = ROOT / "data" / "raw" / "vua_courts"

USER_AGENT = f"fetch_vua_courts/{__version__}"   # the script, not who runs it
MIN_DELAY = 3.0                                  # seconds between requests
TIMEOUT = 120
BASE = "https://www.vuapraha.cz"

# court key → (file name on disk, URL). The keys are the ones utils.parsing.vua_courts uses.
INVENTORIES: dict[str, tuple[str, str]] = {
    "dg_praha": ("Divizni-soud-Praha.pdf",
                 f"{BASE}/wp-content/uploads/2024/02/Divizni-soud-Praha.pdf"),
    "ldg_praha": ("Zemebranecky-divizni-soud-Praha.pdf",
                  f"{BASE}/wp-content/uploads/2025/08/Zemebranecky-divizni-soud-Praha.pdf"),
    "ldg_brno": ("Zemebranecky-divizni-soud-expozitura-Brno.pdf",
                 f"{BASE}/wp-content/uploads/2024/09/"
                 "Zemebranecky-divizni-soud-expozitura-Brno.pdf"),
}

LOG_FIELDS = ["fetched_utc", "court", "file", "url", "status", "bytes", "sha256"]


def _disallowed(urls: list[str]) -> list[str]:
    """The *urls* the site's robots.txt forbids this script (none if it publishes no file)."""
    try:
        resp = requests.get(f"{BASE}/robots.txt", headers={"User-Agent": USER_AGENT},
                            timeout=TIMEOUT)
    except requests.RequestException:
        return []                                     # no robots.txt reachable: nothing forbids
    if resp.status_code != 200:
        return []
    rp = robotparser.RobotFileParser()
    rp.parse(resp.text.splitlines())
    return [u for u in urls if not rp.can_fetch(USER_AGENT, u)]


def _append_log(out_dir: Path, row: dict) -> None:
    path = out_dir / "fetch_log.csv"
    new = not path.exists()
    with path.open("a", newline="", encoding="utf-8") as fh:
        w = csv.DictWriter(fh, fieldnames=LOG_FIELDS)
        if new:
            w.writeheader()
        w.writerow(row)
