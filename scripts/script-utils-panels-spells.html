How it works

Two tables, each at its own level (Gentzkow & Shapiro 2014):

spells one row per legionnaire, key legion_id (every person, in the file or not): entry / first / last period, outcome, censored flag and reason, interval flags, and why a record is not in the file. spell_periods one row per legionnaire x period at risk, key (legion_id, period_index): period_index, period_start, at_risk, event_type.

No person attribute (theatre, rank, unit, ...) is copied: the model merges them on legion_id at the last step. Calendar shocks live in a third, person-free table built by build_calendar (key period_start).

A censored spell ends on the last period the source attests the man survived. A record censored at entry therefore has n_periods = 0 and no rows: in the file (counted), but contributing nothing to the likelihood, exactly as if it were left out.

Every judgement call is a named parameter of build_spells; each default is the author's ruling of 2026-11-18 (D-260, reference/SPELL_FILE_DESIGN.md). Doubtful groups are flagged, not cut (the author's rule for samples): late_entry_russia_1920, entry_after_1920, entry_unit_noncombat, exit_after_theatre_return, note_says_deserter and desertion_note_conflict on spells, after_theatre_return on spell_periods. Only impossible records are excluded, each with its reason. Rules that move an event or a censoring point cannot be a flag: they are built as variants (VARIANTS, notebook 04).

def period_ordinal(dates: pd.Series, period: str) -> pd.Series:
    """Integer period number of each date (``<NA>`` where the date is missing)."""
    d = pd.to_datetime(dates)
    if period == "month":
        o = d.dt.year * 12 + d.dt.month - 1
    elif period == "quarter":
        o = d.dt.year * 4 + (d.dt.month - 1) // 3
    elif period == "week":
        o = (d - _WEEK_ORIGIN).dt.days // 7
    else:
        raise ValueError(f"period must be one of {PERIODS}, not {period!r}")
    return o.astype("Int64")

def period_start(ordinals, period: str) -> pd.Series:

def interval_end(dates: pd.Series, precision: pd.Series) -> pd.Series:
    """Last day a partial date can mean: a month -> its last day, a year -> 31 December.

    dates.py / exit_codes.py store a month as its first day and a year as 1 January
    (``*_precision`` says which); a day-precise date is its own interval."""
    d = pd.to_datetime(dates).copy()
    p = precision.astype("string")
    m, y = p.eq("month").fillna(False), p.eq("year").fillna(False)
    d[m] = d[m] + pd.offsets.MonthEnd(0)
    d[y] = d[y] + pd.offsets.YearEnd(0)
    return d

def resolve_outcome(codes: str | None, outcome_map: dict[str, str] = OUTCOME_MAP,
                    precedence: tuple[str, ...] = OUTCOME_PRECEDENCE):
    """``';'``-joined exit codes -> one competing outcome (``<NA>`` when there is no code)."""
    if codes is None or (not isinstance(codes, str) and pd.isna(codes)) or not str(codes).strip():
        return pd.NA
    outs = {outcome_map[c] for c in str(codes).split(";") if c}
    for o in precedence:
        if o in outs:
            return o
    raise ValueError(f"exit codes {codes!r} map to {outs}, none of which is in the precedence list")

def death_after_exit(persons: pd.DataFrame, outcome_map: dict[str, str] = OUTCOME_MAP,
                     days: int = DEATH_IN_SERVICE_DAYS) -> pd.Series:
    """For a death + demobilisation pair: True if the death (``death_date_raw``) certainly falls
    more than ``days`` after ``exit_date``, False if it can fall within; ``<NA>`` where there is
    no such pair, no exit date or no readable death date (never read as 'not after').

    A partial death date is taken at its first possible day, so True means *certainly* later."""
    out = pd.Series(pd.NA, index=persons.index, dtype="boolean")
    if "death_date_raw" not in persons:
        return out
    pair = death_demob_pair(persons["exit_codes"], outcome_map) & persons["exit_date"].notna()
    if not pair.any():
        return out
    lo, hi = DEATH_RAW_RANGE
    death = pd.to_datetime(persons.loc[pair, "death_date_raw"].map(
        lambda v: parse_date(v, lo, hi, partial=True)[0]))
    gap = (death - pd.to_datetime(persons.loc[pair, "exit_date"])).dt.days
    known = gap.notna()
    out[known[known].index] = (gap[known] > days).to_numpy()
    return out

def build_calendar(first: pd.Timestamp, last: pd.Timestamp, period: str = "month",
                   events: dict[str, pd.Timestamp] = CALENDAR_EVENTS) -> pd.DataFrame:
    """One row per period from ``first`` to ``last``: ``post_<event>`` is True from the
    period that contains the event onwards."""
    o = period_ordinal(pd.Series([pd.Timestamp(first), pd.Timestamp(last)]), period)
    ords = np.arange(int(o.iloc[0]), int(o.iloc[1]) + 1)

    for name, when in events.items():
        ev = int(period_ordinal(pd.Series([when]), period).iloc[0])
        cal[f"post_{name}"] = ords >= ev
    return cal

"""
spells.py — the person-period spell file for the discrete-time competing-risks hazard
=====================================================================================
One row per legionnaire x period at risk, from entry to the Legion to exit or censoring
(Costa & Kahn 2003; CLAUDE.md "Econometric Specification"). Built from
``data/final/persons/legionnaires_clean.parquet`` only.

Two tables, each at its own level (Gentzkow & Shapiro 2014):

``spells``          one row per legionnaire, key ``legion_id`` (every person, in the file or not):
                    entry / first / last period, outcome, censored flag and reason, interval
                    flags, and why a record is not in the file.
``spell_periods``   one row per legionnaire x period at risk, key ``(legion_id, period_index)``:
                    ``period_index``, ``period_start``, ``at_risk``, ``event_type``.

No person attribute (theatre, rank, unit, ...) is copied: the model merges them on
``legion_id`` at the last step. Calendar shocks live in a third, person-free table built by
:func:`build_calendar` (key ``period_start``).

A censored spell ends on the last period the source attests the man survived. A record
censored at entry therefore has ``n_periods = 0`` and no rows: in the file (counted), but
contributing nothing to the likelihood, exactly as if it were left out.

Every judgement call is a named parameter of :func:`build_spells`; each default is the
author's ruling of 2026-11-18 (D-260, ``reference/SPELL_FILE_DESIGN.md``). Doubtful groups are
flagged, not cut (the author's rule for samples): ``late_entry_russia_1920``,
``entry_after_1920``, ``entry_unit_noncombat``, ``exit_after_theatre_return``,
``note_says_deserter`` and ``desertion_note_conflict`` on ``spells``, ``after_theatre_return`` on
``spell_periods``. Only impossible records are excluded, each with
its reason. Rules that move an event or a censoring point cannot be a flag: they are built as
variants (:data:`VARIANTS`, notebook 04).

Usage
-----
    from utils.panels.spells import build_spells, build_calendar, spells_report, check_spells
    spells, periods = build_spells(persons)            # persons = legionnaires_clean
    spells_report(spells, periods, persons)
    assert all(ok for _, ok, _ in check_spells(spells, periods))

#: and central camps (ústř.tábor Rus.Ostrov, tábor; not a výcvikový tábor, a training camp), guard
#: (stráž.pr., strážní prapor, ochranná rota, ochr.rota) and collecting units (sborná rota). The
#: counts are in spells_report and reference/SPELL_FILE_DESIGN.md (f).
ENTRY_UNIT_NONCOMBAT = re.compile(
    r"(?<!\w)(?:prac(?:\.|\s|posádk|ovn(?!ík))|inv\.|inval|posádk|pos\.|(?<!výcvikový\s)tábor|stráž|sborn"
    r"|ochr(?:\.|\s|an\.|ann[áý]|aná|ana(?!\w)))", re.IGNORECASE)
#: An entry "unit" that names a person's role, not a unit (náborový pracovník, důvěrník
#: polit.prac.): the flag is <NA>, as where no unit is recorded.
ENTRY_UNIT_ROLE = re.compile(r"(?<!\w)(?:pracovník|polit\.\s*prac)", re.IGNORECASE)

VARIANTS: dict[str, dict] = {

    "interval_end": {"interval_rule": "end"},                 # (e) December

PERIODS = ("week", "month", "quarter")
CLOCKS = ("duration", "calendar")
_WEEK_ORIGIN = pd.Timestamp("1900-01-01")   # a Monday: weeks run Monday to Sunday


# ---------------------------------------------------------------------------------------
# Period arithmetic
# ---------------------------------------------------------------------------------------
def period_ordinal(dates: pd.Series, period: str) -> pd.Series:
    """Integer period number of each date (``<NA>`` where the date is missing)."""
    d = pd.to_datetime(dates)
    if period == "month":
        o = d.dt.year * 12 + d.dt.month - 1
    elif period == "quarter":
        o = d.dt.year * 4 + (d.dt.month - 1) // 3
    elif period == "week":
        o = (d - _WEEK_ORIGIN).dt.days // 7
    else:
        raise ValueError(f"period must be one of {PERIODS}, not {period!r}")
    return o.astype("Int64")


def period_start(ordinals, period: str) -> pd.Series:
    """First day of each period number (inverse of :func:`period_ordinal`)."""
    o = pd.Series(ordinals).astype("int64")
    if period == "week":
        return _WEEK_ORIGIN + pd.to_timedelta(o * 7, unit="D")
    per_year = {"month": 12, "quarter": 4}[period]
    months_per = 12 // per_year
    return pd.to_datetime(pd.DataFrame({"year": o // per_year,
                                        "month": (o % per_year) * months_per + 1, "day": 1}))


def interval_end(dates: pd.Series, precision: pd.Series) -> pd.Series:
    """Last day a partial date can mean: a month -> its last day, a year -> 31 December.

    dates.py / exit_codes.py store a month as its first day and a year as 1 January
    (``*_precision`` says which); a day-precise date is its own interval."""
    d = pd.to_datetime(dates).copy()
    p = precision.astype("string")
    m, y = p.eq("month").fillna(False), p.eq("year").fillna(False)
    d[m] = d[m] + pd.offsets.MonthEnd(0)
    d[y] = d[y] + pd.offsets.YearEnd(0)
    return d


# ---------------------------------------------------------------------------------------
# Outcome of a record's exit codes
# ---------------------------------------------------------------------------------------
def resolve_outcome(codes: str | None, outcome_map: dict[str, str] = OUTCOME_MAP,
                    precedence: tuple[str, ...] = OUTCOME_PRECEDENCE):
    """``';'``-joined exit codes -> one competing outcome (``<NA>`` when there is no code)."""
    if codes is None or (not isinstance(codes, str) and pd.isna(codes)) or not str(codes).strip():
        return pd.NA
    outs = {outcome_map[c] for c in str(codes).split(";") if c}
    for o in precedence:
        if o in outs:
            return o
    raise ValueError(f"exit codes {codes!r} map to {outs}, none of which is in the precedence list")


def death_demob_pair(codes: pd.Series, outcome_map: dict[str, str] = OUTCOME_MAP) -> pd.Series:
    """True where a record's codes name both a death and a demobilisation/return (``DM;S``)."""
    outs = codes.astype("string").fillna("").str.split(";").map(
        lambda cs: {outcome_map[c] for c in cs if c})
    return outs.map(lambda o: {"death", "demob_return"} <= o).astype(bool)


def death_after_exit(persons: pd.DataFrame, outcome_map: dict[str, str] = OUTCOME_MAP,
                     days: int = DEATH_IN_SERVICE_DAYS) -> pd.Series:
    """For a death + demobilisation pair: True if the death (``death_date_raw``) certainly falls
    more than ``days`` after ``exit_date``, False if it can fall within; ``<NA>`` where there is
    no such pair, no exit date or no readable death date (never read as 'not after').

    A partial death date is taken at its first possible day, so True means *certainly* later."""
    out = pd.Series(pd.NA, index=persons.index, dtype="boolean")
    if "death_date_raw" not in persons:
        return out
    pair = death_demob_pair(persons["exit_codes"], outcome_map) & persons["exit_date"].notna()
    if not pair.any():
        return out
    lo, hi = DEATH_RAW_RANGE
    death = pd.to_datetime(persons.loc[pair, "death_date_raw"].map(
        lambda v: parse_date(v, lo, hi, partial=True)[0]))
    gap = (death - pd.to_datetime(persons.loc[pair, "exit_date"])).dt.days
    known = gap.notna()
    out[known[known].index] = (gap[known] > days).to_numpy()
    return out


def _recorded_outcome(persons, outcome_map, precedence, death_with_demob, death_days):
    """The outcome the codes name, with the death + demobilisation pair resolved by rule."""
    codes = persons["exit_codes"].astype("string")
    uniq = {c: resolve_outcome(c, outcome_map, precedence) for c in codes.dropna().unique()}
    rec = codes.map(uniq).astype("string")
    late = death_after_exit(persons, outcome_map, death_days)
    if death_with_demob == "demob_return":
        flip = death_demob_pair(codes, outcome_map)
    elif death_with_demob == "by_death_date":
        flip = late.fillna(False).astype(bool)
    else:                                                    # "death": precedence alone
        flip = pd.Series(False, index=persons.index)
    rec[rec.eq("death").fillna(False) & flip] = "demob_return"
    return rec, late


def note_dated_exits(persons: pd.DataFrame, outcome: pd.Series) -> pd.DataFrame:
    """The exit date each record carries once a desertion is dated by its note.

    For a record whose outcome is desertion and whose note states a desertion date
    (:mod:`utils.panels.desertion_notes`), the note's date is **accepted** unless it falls
    **after** the recorded exit's interval (a later note date on a dated exit is a write-off,
    ``vypsán jako zběh dnem ...``) where the note does not call the recorded date the last record
    of him (``Poslední údaj ke dni ...``), or **before** the entry's interval. An accepted date
    replaces ``exit_date``, except that a finer recorded date inside it is kept (``4.1918
    dezertoval`` on an exit of 7 April 1918 keeps the day). Returns ``exit_date``,
    ``exit_date_precision``, ``desertion_note_date`` (NaT where none), ``exit_dated_by_note``
    (True where accepted; False on a desertion it does not date; ``<NA>`` on any other outcome)
    and ``desertion_note_conflict`` (on a D-alone record whose note states a date: True where the
    two desertion dates' intervals do not overlap, whichever was used; False otherwise; ``<NA>``
    where the note states none)."""
    idx = persons.index
    out = pd.DataFrame({"exit_date": pd.to_datetime(persons["exit_date"]),
                        "exit_date_precision": persons["exit_date_precision"].astype("string"),
                        "desertion_note_date": pd.Series(pd.NaT, index=idx, dtype="datetime64[ns]"),
                        "exit_dated_by_note": pd.Series(pd.NA, index=idx, dtype="boolean"),
                        "desertion_note_conflict": pd.Series(pd.NA, index=idx, dtype="boolean")})
    des = outcome.eq("desertion").fillna(False).astype(bool)
    out.loc[des, "exit_dated_by_note"] = False
    if "notes" not in persons or not des.any():
        return out
    nd = desertion_note_dates(persons.loc[des, "notes"])
    nd = nd[nd["date"].notna()]
    out.loc[nd.index, "desertion_note_date"] = nd["date"]
    use, keep_recorded, conflict = _note_date_use(persons.loc[nd.index], nd)
    out.loc[nd.index, "desertion_note_conflict"] = conflict.to_numpy()
    replace = use.index[use & ~keep_recorded]
    out.loc[replace, "exit_date"] = nd.loc[replace, "date"]
    out.loc[replace, "exit_date_precision"] = nd.loc[replace, "precision"]
    out.loc[use.index[use], "exit_dated_by_note"] = True
    return out


def _note_date_use(p: pd.DataFrame, nd: pd.DataFrame) -> tuple[pd.Series, pd.Series, pd.Series]:
    """``(use, keep_recorded, conflict)`` for the records whose note states a desertion date."""
    note_lo, note_hi = nd["date"], interval_end(nd["date"], nd["precision"])
    ex_lo = pd.to_datetime(p["exit_date"])
    ex_hi = interval_end(ex_lo, p["exit_date_precision"])
    entry_lo = pd.to_datetime(p["enroll_date"])
    last = note_readings(p["notes"])["last_record"]
    not_after = ex_lo.isna() | (note_lo <= ex_hi) | last
    not_before = entry_lo.isna() | (note_hi >= entry_lo)
    use = (not_after & not_before).fillna(False).astype(bool)
    keep = (ex_lo.notna() & (ex_lo >= note_lo) & (ex_hi <= note_hi)
            & ((note_hi - note_lo) > (ex_hi - ex_lo))).fillna(False).astype(bool)
    overlap = ((note_lo <= ex_hi) & (ex_lo <= note_hi)).fillna(False).astype(bool)
    single = p["exit_codes"].astype("string").eq("D").fillna(False).astype(bool)
    conflict = single & ex_lo.notna() & ~overlap
    return use, keep, conflict


def note_says_deserter(persons: pd.DataFrame) -> pd.Series:
    """True where a record with **no D code** has a note that says he deserted
    (:func:`utils.panels.desertion_notes.says_deserter`); False where the note does not say so, or
    the record is coded D; ``<NA>`` where there is no note."""
    if "notes" not in persons:
        return pd.Series(pd.NA, index=persons.index, dtype="boolean")
    has_d = persons["exit_codes"].astype("string").fillna("").str.split(";").map(lambda cs: "D" in cs)
    says = note_readings(persons["notes"])["says_deserter"]
    return says.mask(has_d.astype(bool) & says.notna(), False).astype("boolean")


def _recode_note_deserters(persons: pd.DataFrame, rec: pd.Series, says: pd.Series) -> pd.Series:
    """The variant ``note_deserters="desertion"``: a record whose note says deserter becomes a
    desertion, unless the note's desertion date falls before his entry (he deserted earlier and
    came back: this spell did not end in it). Dated afterwards by :func:`note_dated_exits`."""
    cand = says.fillna(False).astype(bool) & rec.ne("desertion").fillna(True).astype(bool)
    if not cand.any():
        return rec
    nd = desertion_note_dates(persons.loc[cand, "notes"])
    entry = pd.to_datetime(persons.loc[cand, "enroll_date"])
    before = (interval_end(nd["date"], nd["precision"]) < entry).fillna(False).astype(bool)
    out = rec.copy()
    out[before.index[~before]] = "desertion"
    return out


def sample_flags(persons: pd.DataFrame) -> pd.DataFrame:
    """The flags for doubtful groups (flag, don't cut), one row per record.

    ``late_entry_russia_1920``: theatre Russia and an entry dated in 1920; ``entry_after_1920``:
    an entry that certainly falls after 1920 (its first possible day); ``entry_unit_noncombat``:
    the entry unit is a labour, invalid, garrison, camp, guard or collecting unit
    (:data:`ENTRY_UNIT_NONCOMBAT`). ``<NA>`` where the source gives no entry date (no unit, or a
    person's role in place of a unit, :data:`ENTRY_UNIT_ROLE`)."""
    entry = pd.to_datetime(persons["enroll_date"])
    dated = entry.notna()
    theatre = persons["theater"].astype("string")
    late = (theatre.eq(LATE_ENTRY_THEATRE).fillna(False) & entry.dt.year.eq(LATE_ENTRY_YEAR))
    unit = (persons[ENTRY_UNIT_COL].astype("string") if ENTRY_UNIT_COL in persons
            else pd.Series(pd.NA, index=persons.index, dtype="string"))
    noncombat = unit.map(lambda u: pd.NA if pd.isna(u) or ENTRY_UNIT_ROLE.search(u)
                         else bool(ENTRY_UNIT_NONCOMBAT.search(u)))
    return pd.DataFrame({
        "late_entry_russia_1920": late.astype("boolean").where(dated),
        "entry_after_1920": (entry > ENTRY_AFTER).astype("boolean").where(dated),
        "entry_unit_noncombat": noncombat.astype("boolean"),
    }, index=persons.index)


def _spell_inputs(persons: pd.DataFrame, period: str, outcome_map, precedence,
                  death_with_demob: str, death_days: int, note_dates: bool,
                  note_deserters: str = "flag") -> pd.DataFrame:
    """Per-person entry and exit intervals as period numbers, and the recorded outcome."""
    p = persons
    codes = p["exit_codes"].astype("string")
    rec, late = _recorded_outcome(p, outcome_map, precedence, death_with_demob, death_days)
    says = note_says_deserter(p)
    if note_deserters == "desertion":
        rec = _recode_note_deserters(p, rec, says)
    n_codes = codes.str.count(";").add(1).astype("Int64")
    entry_end = interval_end(p["enroll_date"], p["enroll_date_precision"])
    ex = note_dated_exits(p, rec) if note_dates else note_dated_exits(
        p.drop(columns=["notes"], errors="ignore"), rec)
    dup = (p["dup_group"].astype("Int64") if "dup_group" in p
           else pd.Series(pd.NA, index=p.index, dtype="Int64"))
    repat = pd.to_datetime(p["theater"].astype("string").map(THEATRE_REPATRIATION))
    out = pd.DataFrame({
        KEY: p[KEY],
        "theater": p["theater"].astype("string"),
        "dup_group": dup,
        "entry_lo": period_ordinal(p["enroll_date"], period),
        "entry_hi": period_ordinal(entry_end, period),
        "entry_start_date": pd.to_datetime(p["enroll_date"]),
        "entry_end_date": entry_end,
        "exit_lo": period_ordinal(ex["exit_date"], period),
        "exit_hi": period_ordinal(interval_end(ex["exit_date"], ex["exit_date_precision"]), period),
        "outcome_recorded": rec,
        "exit_multi": n_codes.gt(1).astype("boolean"),   # <NA> where no code
        "death_after_exit": late,
        "desertion_note_date": ex["desertion_note_date"],
        "exit_dated_by_note": ex["exit_dated_by_note"],
        "desertion_note_conflict": ex["desertion_note_conflict"],
        "note_says_deserter": says,
        # the exit (its first possible day) certainly after the theatre's last moving-home month
        "exit_after_theatre_return": (ex["exit_date"] > repat).astype("boolean")
        .where(ex["exit_date"].notna() & repat.notna()),
    })
    out = pd.concat([out, sample_flags(p)], axis=1)
    return out.reset_index(drop=True)

# ---------------------------------------------------------------------------------------
def _ceiling(theater: pd.Series, entry_ceiling) -> pd.Series:
    """The latest admissible entry date per record (NaT = no ceiling for that theatre)."""
    if isinstance(entry_ceiling, dict):
        return pd.to_datetime(theater.map({k: pd.Timestamp(v) for k, v in entry_ceiling.items()}))
    return pd.Series(pd.Timestamp(entry_ceiling), index=theater.index)


def _exclusions(s: pd.DataFrame, *, entry_interval_rule: str, entry_floor, entry_ceiling,
                exit_before_entry: str, undated_rule: str, censor_rule: str, duplicates: str,
                window_p: pd.Series) -> pd.Series:
    """The first reason, in a fixed order, why each record is not in the file (<NA> = in file)."""
    reason = pd.Series(pd.NA, index=s.index, dtype="string")

    def mark(mask, why):
        reason[mask.fillna(False).astype(bool) & reason.isna()] = why

    has_code, has_date = s["outcome_recorded"].notna(), s["exit_lo"].notna()
    mark(s["entry_lo"].isna(), "no_entry_date")
    if entry_interval_rule == "exclude":
        mark(s["entry_lo"] != s["entry_hi"], "entry_imprecise")
    if entry_floor is not None:
        mark(s["entry_end_date"] < pd.Timestamp(entry_floor), "entry_before_floor")
    if entry_ceiling is not None:
        mark(s["entry_start_date"] > _ceiling(s["theater"], entry_ceiling), "entry_after_ceiling")
    mark(s["entry_lo"] > window_p, "entry_after_window")
    if exit_before_entry == "exclude":
        mark(s["exit_hi"] < s["entry_lo"], "exit_before_entry")
    if undated_rule == "exclude":
        mark(has_code & ~has_date, "exit_undated")
    if censor_rule == "exclude":
        mark(~has_code & ~has_date, "no_exit_record")
    in_group = s["dup_group"].notna()
    if duplicates == "drop":
        mark(in_group, "duplicate_group")
    elif duplicates == "representative":
        mark(in_group & (s["dup_group"] != s[KEY]), "duplicate_not_representative")
    return reason

def _theatre_period(theater: pd.Series, dates: dict, period: str) -> pd.Series:
    """Period number of a theatre-level date (<NA> for a theatre the source does not date)."""
    return period_ordinal(pd.to_datetime(theater.map(dates)), period)


def _resolve_end(s: pd.DataFrame, *, entry_p: pd.Series, interval_rule: str, censor_rule: str,
                 censor_date, uncoded_rule: str, censor_period: str, theatre_end_p: pd.Series,
                 period: str) -> pd.DataFrame:
    """Last period, outcome, censoring reason and the first period of unknown risk per spell.

    Under ``censor_period="last_survived"`` a censored spell ends on the last period the source
    attests he survived: the period *before* entry for one censored at entry (zero rows), the
    period before the exit for an exit whose type or timing is unknown."""
    has_code, has_date = s["outcome_recorded"].notna(), s["exit_lo"].notna()
    lo = np.maximum(s["exit_lo"], entry_p)
    hi = np.maximum(s["exit_hi"], entry_p)
    survived = censor_period == "last_survived"
    at_entry = entry_p - 1 if survived else entry_p          # censoring point "at entry"
    interval = has_date & (s["exit_lo"] != s["exit_hi"])
    r = pd.DataFrame({"end": pd.Series(pd.NA, index=s.index, dtype="Int64"),
                      "outcome": s["outcome_recorded"].where(has_code, UNCODED).astype("string"),
                      "censor_reason": pd.Series(pd.NA, index=s.index, dtype="string"),
                      "risk_unknown_from": pd.Series(pd.NA, index=s.index, dtype="Int64"),
                      "exit_interval": interval.where(has_date, pd.NA).astype("boolean")})
    exact = has_date & ~interval
    r.loc[exact, "end"] = lo[exact]          # an exit before entry lands on the entry period
    rule_end = {"start": lo, "end": hi, "interval": hi, "midpoint": (lo + hi) // 2,
                "censor_at_start": np.maximum(at_entry, lo - 1)}
    if interval_rule not in rule_end:
        raise ValueError(f"interval_rule {interval_rule!r} not in {sorted(rule_end)}")
    r.loc[interval, "end"] = rule_end[interval_rule][interval]
    if interval_rule == "censor_at_start":
        r.loc[interval, "censor_reason"] = "exit_interval"
    if interval_rule == "interval":
        r.loc[interval, "risk_unknown_from"] = lo[interval]
    if uncoded_rule == "censor":
        unc = has_date & ~has_code
        r.loc[unc, "censor_reason"] = "exit_uncoded"
        if survived:                         # he left in [lo, hi] for a reason unknown
            r.loc[unc, "end"] = lo[unc] - 1
            r.loc[unc, "risk_unknown_from"] = pd.NA
    undated = has_code & ~has_date
    r.loc[undated, "censor_reason"] = "exit_undated"
    r.loc[undated, "end"] = at_entry[undated]
    _censor_no_exit(r, ~has_code & ~has_date, entry_p, at_entry, censor_rule, censor_date,
                    theatre_end_p, period)
    return r


def _censor_no_exit(r, none, entry_p, at_entry, censor_rule, censor_date, theatre_end_p,
                    period) -> None:
    """Censoring point of a record with neither exit code nor exit date (in place)."""
    r.loc[none, "censor_reason"] = "no_exit_record"
    r.loc[none, "end"] = at_entry[none]
    if censor_rule == "theatre_end":
        ok = none & theatre_end_p.notna() & (theatre_end_p >= entry_p)
        r.loc[ok, "end"] = theatre_end_p[ok]
        r.loc[none & ~ok, "censor_reason"] = "no_exit_record_fallback"
    elif censor_rule == "fixed":
        fixed = int(period_ordinal(pd.Series([pd.Timestamp(censor_date)]), period).iloc[0])
        ok = none & (entry_p <= fixed)
        r.loc[ok, "end"] = fixed
        r.loc[none & ~ok, "censor_reason"] = "no_exit_record_fallback"
    elif censor_rule not in ("last_observed", "exclude"):
        raise ValueError(f"censor_rule {censor_rule!r} unknown")


def _apply_window(r: pd.DataFrame, window_p: pd.Series) -> None:
    """Administrative censoring at the window's last period (in place)."""
    past = (r["end"] > window_p).fillna(False)
    r.loc[past, "end"] = window_p[past]
    r.loc[past, "censor_reason"] = "window_end"
    over = (r["risk_unknown_from"] > r["end"]).fillna(False)
    r.loc[over, "risk_unknown_from"] = pd.NA

def _check_args(period, clock, interval_rule, censor_rule, undated_rule, uncoded_rule,
                entry_interval_rule, exit_before_entry, duplicates, censor_date, censor_period,
                death_with_demob, death_in_service_days, desertion_date_from_note=True,
                note_deserters="flag") -> None:
    allowed = {"period": (period, PERIODS), "clock": (clock, CLOCKS),
               "censor_period": (censor_period, ("last_survived", "observed_period")),
               "death_with_demob": (death_with_demob, ("by_death_date", "death", "demob_return")),
               "note_deserters": (note_deserters, ("flag", "desertion")),
               "interval_rule": (interval_rule, ("censor_at_start", "start", "midpoint", "end", "interval")),
               "censor_rule": (censor_rule, ("last_observed", "theatre_end", "fixed", "exclude")),
               "undated_rule": (undated_rule, ("censor", "exclude")),
               "uncoded_rule": (uncoded_rule, ("outcome", "censor")),
               "entry_interval_rule": (entry_interval_rule, ("exclude", "start", "end")),
               "exit_before_entry": (exit_before_entry, ("exclude", "at_entry")),
               "duplicates": (duplicates, ("keep", "representative", "drop"))}
    for name, (val, ok) in allowed.items():
        if val not in ok:
            raise ValueError(f"{name}={val!r}; allowed: {ok}")
    if censor_rule == "fixed" and censor_date is None:
        raise ValueError("censor_rule='fixed' needs censor_date")
    if not (isinstance(death_in_service_days, int) and death_in_service_days >= 0):
        raise ValueError("death_in_service_days must be a non-negative integer")
    if not isinstance(desertion_date_from_note, bool):
        raise ValueError("desertion_date_from_note must be True or False")


def _window_period(theater: pd.Series, window_end, period: str) -> pd.Series:
    """Last period of observation per record (<NA> = no administrative cap)."""
    if window_end is None:
        return pd.Series(pd.NA, index=theater.index, dtype="Int64")
    if window_end == "repatriation":
        return _theatre_period(theater, THEATRE_REPATRIATION, period)
    o = int(period_ordinal(pd.Series([pd.Timestamp(window_end)]), period).iloc[0])
    return pd.Series(o, index=theater.index, dtype="Int64")


def build_spells(persons: pd.DataFrame, *, period: str = "month", clock: str = "duration",
                 outcome_map: dict[str, str] = OUTCOME_MAP,
                 precedence: tuple[str, ...] = OUTCOME_PRECEDENCE,
                 death_with_demob: str = "demob_return",
                 death_in_service_days: int = DEATH_IN_SERVICE_DAYS,
                 desertion_date_from_note: bool = True, note_deserters: str = "flag",
                 censor_rule: str = "last_observed", censor_date=None,
                 censor_period: str = "last_survived",
                 interval_rule: str = "censor_at_start", undated_rule: str = "censor",
                 uncoded_rule: str = "outcome", entry_interval_rule: str = "exclude",
                 entry_floor=WAR_START, entry_ceiling=None, exit_before_entry: str = "exclude",
                 left_truncate_at_formation: bool = False, window_end=None,
                 duplicates: str = "keep") -> tuple[pd.DataFrame, pd.DataFrame]:
    """``(spells, spell_periods)`` from the legionnaire persons table.

    Every keyword is a design decision; see reference/SPELL_FILE_DESIGN.md for the options,
    the counts behind them and the author's ruling each default encodes (D-260).
    ``entry_ceiling`` is a date, or a ``{theatre: date}`` dict, after which an entry is excluded
    (None = no ceiling). ``desertion_date_from_note`` dates a desertion by the date its note
    states (:func:`note_dated_exits`). ``note_deserters="desertion"`` recodes a record whose
    note says deserter but carries no D code (:func:`note_says_deserter`) as a desertion; the
    default ``"flag"`` only flags it."""
    _check_args(period, clock, interval_rule, censor_rule, undated_rule, uncoded_rule,
                entry_interval_rule, exit_before_entry, duplicates, censor_date, censor_period,
                death_with_demob, death_in_service_days, desertion_date_from_note, note_deserters)
    if not persons[KEY].is_unique or persons[KEY].isna().any():
        raise ValueError(f"{KEY} must be unique and never missing")
    s = _spell_inputs(persons, period, outcome_map, precedence, death_with_demob,
                      death_in_service_days, desertion_date_from_note, note_deserters)
    window_p = _window_period(s["theater"], window_end, period)
    excluded = _exclusions(s, entry_interval_rule=entry_interval_rule, entry_floor=entry_floor,
                           entry_ceiling=entry_ceiling, exit_before_entry=exit_before_entry,
                           undated_rule=undated_rule, censor_rule=censor_rule,
                           duplicates=duplicates, window_p=window_p)
    entry_p = s["entry_hi"] if entry_interval_rule == "end" else s["entry_lo"]
    r = _resolve_end(s, entry_p=entry_p, interval_rule=interval_rule, censor_rule=censor_rule,
                     censor_date=censor_date, uncoded_rule=uncoded_rule,
                     censor_period=censor_period,
                     theatre_end_p=_theatre_period(s["theater"], THEATRE_REPATRIATION, period),
                     period=period)
    _apply_window(r, window_p)
    form_p = _theatre_period(s["theater"], THEATRE_FORMATION, period)
    first = np.maximum(entry_p, form_p).where(form_p.notna(), entry_p) \
        if left_truncate_at_formation else entry_p
    spells = _assemble_spells(s, r, excluded, entry_p, first, form_p, period)
    spells["_repat"] = _theatre_period(s["theater"], THEATRE_REPATRIATION, period).where(
        spells["in_file"]).astype("Int64")
    return _public(spells), _expand(spells, period, clock)


def _assemble_spells(s, r, excluded, entry_p, first, form_p, period) -> pd.DataFrame:
    """The one-row-per-person table; excluded records keep their key and reason only.

    A spell censored before its entry period is in the file with ``n_periods = 0``: zero rows,
    which in the likelihood is the same as leaving it out, but the record stays counted."""
    zero = (r["end"] < entry_p).fillna(False)
    inf = excluded.isna() & ((r["end"] >= first).fillna(False) | zero)
    excluded = excluded.where(inf | excluded.notna(), "truncated_before_exit")
    censored = r["censor_reason"].notna()

    def keep(x, dtype):
        return pd.Series(x, index=s.index).where(inf).astype(dtype)

    def start(o, mask):
        o = pd.Series(o, index=s.index)
        out = pd.Series(pd.NaT, index=s.index, dtype="datetime64[ns]")
        out[mask] = period_start(o[mask], period).to_numpy()
        return out

    return pd.DataFrame({
        KEY: s[KEY],
        "in_file": inf.astype(bool),
        "excluded_reason": excluded.astype("string"),
        "entry_period_start": start(entry_p, inf),
        "first_period_start": start(first, inf),
        "last_period_start": start(r["end"], inf & ~zero),
        "n_periods": keep((r["end"] - first + 1).where(~zero, 0), "Int64"),
        "outcome": keep(r["outcome"].where(~censored, CENSORED), "string"),
        "outcome_recorded": s["outcome_recorded"].astype("string"),
        "censored": keep(censored, "boolean"),
        "censor_reason": keep(r["censor_reason"], "string"),
        "entry_interval": keep(s["entry_lo"] != s["entry_hi"], "boolean"),
        "exit_interval": keep(r["exit_interval"], "boolean"),
        "exit_multi": s["exit_multi"].astype("boolean"),
        "death_after_exit": s["death_after_exit"].astype("boolean"),
        "desertion_note_date": s["desertion_note_date"],
        "exit_dated_by_note": s["exit_dated_by_note"].astype("boolean"),
        "desertion_note_conflict": s["desertion_note_conflict"].astype("boolean"),
        "note_says_deserter": s["note_says_deserter"].astype("boolean"),
        "exit_after_theatre_return": s["exit_after_theatre_return"].astype("boolean"),
        "late_entry_russia_1920": s["late_entry_russia_1920"].astype("boolean"),
        "entry_after_1920": s["entry_after_1920"].astype("boolean"),
        "entry_unit_noncombat": s["entry_unit_noncombat"].astype("boolean"),
        "entry_before_formation": (s["entry_hi"] < form_p).astype("boolean"),
        "left_truncated": keep(first > entry_p, "boolean"),
        "_first": first.where(inf).astype("Int64"),
        "_entry": entry_p.where(inf).astype("Int64"),
        "_risk_unknown_from": r["risk_unknown_from"].where(inf).astype("Int64"),
    })


def _expand(spells: pd.DataFrame, period: str, clock: str) -> pd.DataFrame:
    """Explode each spell into its periods; the event sits on the last one."""
    sp = spells[spells["in_file"] & spells["n_periods"].gt(0).fillna(False)]
    first = sp["_first"].to_numpy("int64")
    n = sp["n_periods"].to_numpy("int64")
    starts = np.cumsum(n) - n
    pos = np.arange(n.sum()) - np.repeat(starts, n)
    ordn = np.repeat(first, n) + pos
    last = pos == np.repeat(n - 1, n)
    origin = (np.repeat(sp["_entry"].to_numpy("int64"), n) if clock == "duration"
              else int(period_ordinal(pd.Series([WAR_START]), period).iloc[0]))
    unknown = np.repeat(sp["_risk_unknown_from"].to_numpy("float64", na_value=np.nan), n)
    at_risk = pd.array(np.ones(len(ordn), bool), dtype="boolean")
    at_risk[ordn >= unknown] = pd.NA                     # inside an exit interval: unknown
    repat = np.repeat(sp["_repat"].to_numpy("float64", na_value=np.nan), n)
    after = pd.array(ordn > repat, dtype="boolean")      # period_start after the moving-home month
    after[np.isnan(repat)] = pd.NA                       # a theatre the source does not date
    outcome = np.repeat(sp["outcome"].to_numpy(object), n)
    event = np.where(last & (outcome != CENSORED), outcome, NO_EVENT)
    return pd.DataFrame({
        KEY: np.repeat(sp[KEY].to_numpy(), n),
        "period_index": (ordn - origin).astype("int32"),
        "period_start": _starts(ordn, period),
        "at_risk": at_risk,
        "event_type": pd.array(event, dtype="string"),
        "after_theatre_return": after,
    })


def _starts(ordn: np.ndarray, period: str) -> np.ndarray:
    """period_start for many repeated period numbers (computed once per distinct number)."""
    if len(ordn) == 0:
        return np.array([], dtype="datetime64[ns]")
    uniq, inv = np.unique(ordn, return_inverse=True)
    return period_start(uniq, period).to_numpy()[inv]


def _public(spells: pd.DataFrame) -> pd.DataFrame:
    """The spells table without the builder's internal period numbers."""
    return spells.drop(columns=[c for c in spells.columns if c.startswith("_")])

def build_calendar(first: pd.Timestamp, last: pd.Timestamp, period: str = "month",
                   events: dict[str, pd.Timestamp] = CALENDAR_EVENTS) -> pd.DataFrame:
    """One row per period from ``first`` to ``last``: ``post_<event>`` is True from the
    period that contains the event onwards."""
    o = period_ordinal(pd.Series([pd.Timestamp(first), pd.Timestamp(last)]), period)
    ords = np.arange(int(o.iloc[0]), int(o.iloc[1]) + 1)
    cal = pd.DataFrame({"period_start": period_start(ords, period).to_numpy()})
    for name, when in events.items():
        ev = int(period_ordinal(pd.Series([when]), period).iloc[0])
        cal[f"post_{name}"] = ords >= ev
    return cal

def _key_checks(spells: pd.DataFrame, periods: pd.DataFrame) -> list[tuple[str, bool, str]]:
    k = periods[list(PERIOD_KEY)]
    return [("spells key unique, never missing",
             bool(spells[KEY].is_unique and spells[KEY].notna().all()), ""),
            ("periods key unique, never missing",
             bool(not k.duplicated().any() and k.notna().all().all()), "")]


def _event_checks(spells: pd.DataFrame, periods: pd.DataFrame) -> list[tuple[str, bool, str]]:
    """One event at most, on the last row, and the last row says how the spell ends."""
    ev = periods["event_type"].ne(NO_EVENT)
    per = periods.groupby(KEY, sort=False)
    is_last = periods["period_start"].eq(per["period_start"].transform("max"))
    last = periods.loc[is_last].set_index(KEY)["event_type"]
    sp = spells[spells["in_file"] & spells["n_periods"].gt(0).fillna(False)].set_index(KEY)
    want = sp["outcome"].where(~sp["censored"].astype(bool), NO_EVENT)
    got = last.reindex(want.index)
    n_bad = int((got.ne(want) | got.isna()).sum())
    return [("at most one event per person", bool(ev.groupby(periods[KEY]).sum().le(1).all()), ""),
            ("an event only on the person's last period", bool(not (ev & ~is_last).any()), ""),
            ("the last period carries the outcome (none if censored)", n_bad == 0,
             f"{n_bad} spell(s) whose last row disagrees with outcome/censored")]


def _span_checks(spells: pd.DataFrame, periods: pd.DataFrame) -> list[tuple[str, bool, str]]:
    """Contiguous periods, none before the first, as many rows as ``n_periods`` says."""
    span = periods.groupby(KEY, sort=False)["period_index"].agg(["min", "max", "size"])
    ent = periods[KEY].map(spells.set_index(KEY)["first_period_start"])
    sp = spells[spells["in_file"]].set_index(KEY)
    rows = span["size"].reindex(sp.index).fillna(0)
    with_rows = set(sp.index[sp["n_periods"].gt(0)])
    zero = sp["n_periods"].eq(0)
    return [("periods contiguous", bool((span["max"] - span["min"] + 1).eq(span["size"]).all()), ""),
            ("no period before the spell's first period", bool((periods["period_start"] >= ent).all()), ""),
            ("n_periods never negative", bool(sp["n_periods"].ge(0).all()), ""),
            ("n_periods equals the rows in the panel", bool(rows.eq(sp["n_periods"]).all()), ""),
            ("rows exactly for the in-file spells with n_periods > 0",
             bool(set(span.index) == with_rows), ""),
            ("a zero-row spell is censored", bool(sp.loc[zero, "censored"].astype(bool).all()), "")]


def check_spells(spells: pd.DataFrame, periods: pd.DataFrame) -> list[tuple[str, bool, str]]:
    """``(name, passed, detail)`` for every invariant the two tables must satisfy."""
    sp = spells[spells["in_file"]]
    out = _key_checks(spells, periods) + _event_checks(spells, periods) \
        + _span_checks(spells, periods)
    out.append(("censored iff outcome == censored",
                bool(sp["censored"].eq(sp["outcome"].eq(CENSORED)).all()), ""))
    out.append(("an excluded record has a reason, an included one none",
                bool(spells["excluded_reason"].isna().eq(spells["in_file"]).all()), ""))
    return out


def spells_report(spells: pd.DataFrame, periods: pd.DataFrame,
                  persons: pd.DataFrame | None = None) -> None:
    """Print counts: in/out of the file, events by type, censoring by reason (and theatre)."""
    print(f"spells.py {__version__}: {len(spells):,} records, {int(spells['in_file'].sum()):,} in the file, "
          f"{len(periods):,} person-periods")
    print("\nNot in the file, by reason:")
    print(spells["excluded_reason"].value_counts().to_string())
    sp = spells[spells["in_file"]]
    print("\nSpell outcome (event types, and censored):")
    print(sp["outcome"].value_counts().to_string())
    print("\nCensoring by reason (all / with zero rows):")
    zero = sp["n_periods"].eq(0)
    print(pd.DataFrame({"all": sp["censor_reason"].value_counts(),
                        "zero_rows": sp.loc[zero, "censor_reason"].value_counts()})
          .fillna(0).astype(int).to_string())
    print(f"\nInterval flags: entry {int(sp['entry_interval'].sum()):,}, "
          f"exit {int(sp['exit_interval'].fillna(False).sum()):,}; multi-code {int(sp['exit_multi'].fillna(False).sum()):,}")
    dae = spells["death_after_exit"]
    print(f"death + DM/V pairs with a readable death date: {int(dae.notna().sum()):,} "
          f"(death > {DEATH_IN_SERVICE_DAYS} days after exit: {int(dae.fillna(False).sum()):,})")
    print(f"at_risk unknown rows: {int(periods['at_risk'].isna().sum()):,}")
    byn = spells["exit_dated_by_note"]
    print(f"desertions whose note states a date: {int(spells['desertion_note_date'].notna().sum()):,}; "
          f"exit dated by the note: {int(byn.fillna(False).sum()):,} "
          f"(in the file as a desertion event: {int((byn.fillna(False) & spells['outcome'].eq('desertion').fillna(False)).sum()):,})")
    conf = spells["desertion_note_conflict"]
    print(f"desertion_note_conflict (D alone, two desertion dates disagree): {int(conf.fillna(False).sum()):,} "
          f"of {int(conf.notna().sum()):,} with a note date")
    nsd = spells["note_says_deserter"].fillna(False).astype(bool)
    print(f"note_says_deserter (no D code, the note says deserter): {int(nsd.sum()):,} records, "
          f"{int((nsd & spells['in_file']).sum()):,} in the file; by outcome in the file: "
          + ", ".join(f"{k} {v:,}" for k, v in spells.loc[nsd & spells["in_file"], "outcome"]
                      .value_counts().items()))
    for flag in ("late_entry_russia_1920", "entry_after_1920", "entry_unit_noncombat",
                 "exit_after_theatre_return"):
        f, inf = spells[flag], spells["in_file"]
        print(f"{flag}: {int(f.fillna(False).sum()):,} records ({int(f[inf].fillna(False).sum()):,} "
              f"in the file; {int((f.fillna(False) & spells['outcome'].eq('desertion').fillna(False)).sum()):,} "
              f"desertion events); <NA> {int(f.isna().sum()):,}")
    atr = periods["after_theatre_return"]
    print(f"after_theatre_return rows: {int(atr.fillna(False).sum()):,} of {len(periods):,} "
          f"(<NA> {int(atr.isna().sum()):,}); desertion events on them: "
          f"{int((atr.fillna(False) & periods['event_type'].eq('desertion')).sum()):,}")
    if persons is not None:
        before = len(sp)
        m = sp.merge(persons[[KEY, "theater"]], on=KEY, how="left", validate="1:1")
        assert len(m) == before, "theatre merge changed the row count"
        print(f"\nCensoring reason x theatre (merge {before:,} -> {len(m):,} rows, "
              f"{int(m['theater'].isna().sum())} unmatched):")
        print(pd.crosstab(m["censor_reason"].fillna("(event)"), m["theater"], margins=True).to_string())
    for name, ok, det in check_spells(spells, periods):
        print(f"  [{'ok' if ok else 'FAIL'}] {name} {det if not ok else ''}")

def fidler_theatre_dates(reference_dir: str | Path = "reference") -> tuple[dict, dict]:
    """``(formation, repatriation)`` per theatre, rebuilt from the Fidler 2025 tables."""
    ref = Path(reference_dir) / "units"
    auth = pd.read_csv(ref / "unit_authority.csv")
    auth["formed"] = pd.to_datetime(auth["formed"], errors="coerce")
    formation = auth.groupby("theatre")["formed"].min().dropna().to_dict()
    tl = pd.read_csv(ref / "unit_timeline.csv")
    home = tl[tl["state"].eq("moving_home") & tl["date_suspect"].eq(0)].copy()
    home["d"] = pd.to_datetime(home["date_from"], errors="coerce")
    repatriation = home.groupby("theatre")["d"].max().dropna().to_dict()
    return formation, repatriation
