How it works

The four-question page (theatre_rule_audit, D-212) asks the author to answer every clause from scratch. Once the rules and the models have read the clauses, most of that is wasted on clauses where every reader already agrees. This page asks only:

  • on a clause where the readers disagree — about the four questions, Czech activity (yes / hedged / German / no), a company, a performance, and an undated clause's date — which of their readings is right: one click, or neither (the four questions open) or can't tell;
  • on a random N_AGREED of the clauses where every reader agrees: is that reading right? This measures the pipeline's blind spot, all readers wrong alike.

Readers (from the cached replies, read-only): the rules (on a coded clause), Haiku, Gemini, Opus alone and the judge where it has read the clause. A reader's answer is its reading run through amateur_theatre.code_clauses, as everywhere in the pipeline. The page hides which reader gave which reading until the author has chosen.

Weights. Every disagreement of the half is shown, and the agreed clauses with a known probability (N_AGREED / agreed clauses in the half), so each answer carries ab_prob and theatre_rule_audit.verdict_rows divides the clause's weight by it and stratifies the bootstrap by ab_group: a two-phase sample, still weighted back to the populations. The answers fill the same v_* columns as the four-question page.

What it costs in rigour: the author chooses among the readers' readings; neither exists, but a reading nobody proposed is less likely to be given than when ruling from scratch. A dated clause's leading date is not asked (the readers do not re-date it).

Usage::

python utils/reference_tools/places/theatre_ab_check.py draw [calibration|certification]

def reader_answers(rows: pd.DataFrame) -> dict[str, pd.DataFrame]:
    """Each reader's answers on ``rows`` (coded clause rows of both samples), from the cache."""
    rows = rows.assign(_undated=rows["date_source"].eq("none"))
    stage = atm.add_context(rows.assign(open_kind=np.where(rows["needs_model"].astype(bool), "open", "coded")))
    out = {"rules": _answers(rows.set_index("clause_key").loc[~rows["needs_model"].astype(bool).to_numpy()])}
    readings = {r: atm.readings_table(stage, atm.build_queue(stage, r)) for r in cons.PIPELINE}
    readings["opus"] = atm.readings_table(stage, atm.build_queue(stage, "opus"))
    table = cons.disputes(stage, {r: readings[r] for r in cons.PIPELINE})
    disputed = table.index[table["disputed"]]
    others = cons.judge_others(disputed, {r: readings[r] for r in cons.PIPELINE}, stage)
    judge_q = atm.build_queue(stage[stage["clause_key"].isin(disputed)], cons.JUDGE, others=others)
    readings[cons.JUDGE] = atm.readings_table(stage, judge_q)
    for name in MODEL_READERS:
        valid = readings[name][readings[name]["valid"].astype(bool)] if len(readings[name]) else readings[name]
        if not len(valid):
            continue
        read = rows[rows["clause_key"].isin(valid["clause_key"])]
        coded = pd.concat([atm.apply_readings(g, valid, s).assign(clause_key=g["clause_key"].values,
                                                                   _undated=g["_undated"].values)
                           for s, g in read.groupby("source")])
        out[name] = _answers(coded.set_index("clause_key", drop=False), valid).set_index(coded["clause_key"].values)
    return out

def candidates(answers: dict[str, pd.DataFrame]) -> pd.DataFrame:
    """Per clause: the distinct readings (``candidates``, each with the readers that gave it)."""
    keys = sorted(set().union(*[a.index for a in answers.values()]))
    rows = []
    for key in keys:
        found: dict[tuple, list[str]] = {}
        for name, a in answers.items():
            if key in a.index:
                r = a.loc[key]
                found.setdefault((r["activity"], r["company"], r["performance"], r["date"], r["years"]), []).append(name)
        cands = [{"activity": t[0], "company": t[1], "performance": t[2], "date": t[3], "years": t[4], "readers": rs}
                 for t, rs in sorted(found.items(), key=lambda kv: -len(kv[1]))]
        rows.append({"clause_key": key, "n_readings": len(cands), "candidates": cands,
                     "readers": sum(len(c["readers"]) for c in cands)})
    return pd.DataFrame(rows).set_index("clause_key")

def draw(half: str = "calibration", seed: int = SEED) -> pd.DataFrame:
    """Every disagreement of ``half`` and :data:`N_AGREED` agreed clauses, drawn at random, with
    ``ab_group`` and ``ab_prob``; shuffled."""
    samples = ra.load_samples()
    sample_of = pd.concat([s.loc[s["half"] == half, ["clause_key"]].assign(sample=n) for n, s in samples.items()])
    clauses = atm.load_clauses()
    rows = clauses[clauses["clause_key"].isin(sample_of["clause_key"])]
    c = candidates(reader_answers(rows))
    c["sample"] = sample_of.set_index("clause_key")["sample"].reindex(c.index)
    c = c[c["readers"] >= 2]                         # a clause with one reader has nothing to compare
    disputed = c[c["n_readings"] > 1].assign(ab_group="disputed", ab_prob=1.0)
    agreed_all = c[c["n_readings"] == 1]
    rng = np.random.default_rng(seed)
    take = min(N_AGREED, len(agreed_all))
    agreed = agreed_all.iloc[np.sort(rng.permutation(len(agreed_all))[:take])].assign(
        ab_group="agreed", ab_prob=take / len(agreed_all))
    out = pd.concat([disputed, agreed])
    out = out.iloc[rng.permutation(len(out))]
    out.attrs.update(disputed=len(disputed), agreed=len(agreed_all), drawn_agreed=take, half=half)
    return out.reset_index()

def write_sheet(drawn: pd.DataFrame, path: Path = SHEET) -> Path:
    """The draw, frozen: each clause's candidate readings as drawn (JSON), group and probability.
    Refuses to replace a sheet whose half already has answers in the sample sheets."""
    sheet = drawn[["clause_key", "sample", "ab_group", "ab_prob", "n_readings"]].copy()
    sheet["half"] = drawn.attrs.get("half", "calibration")
    sheet["candidates"] = drawn["candidates"].map(lambda c: json.dumps(c, ensure_ascii=False))
    if Path(path).exists():
        old = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
        answered = set().union(*[set(s.loc[s["v_source"].eq("ab"), "clause_key"]) if "v_source" in s else set()
                                 for s in ra.load_samples().values()])
        if answered & set(old["clause_key"]):
            raise FileExistsError(f"{path}: some of its clauses are answered; it is not redrawn")
        sheet = pd.concat([old[old["half"] != sheet["half"].iat[0]], sheet], ignore_index=True)
    sheet.to_csv(path, index=False, encoding="utf-8-sig")
    return Path(path)

def page_items(path: Path = SHEET, half: str = "calibration") -> list[dict]:
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    sheet = sheet[sheet["half"] == half]
    clauses = atm.load_clauses()
    rows = clauses[clauses["clause_key"].isin(sheet["clause_key"])]
    ctx = atm.add_context(rows.assign(open_kind="x")).set_index("clause_key")
    items = []
    rng = np.random.default_rng(SEED)
    for i, r in enumerate(sheet.itertuples(index=False)):
        c = ctx.loc[r.clause_key]
        cands = json.loads(r.candidates)
        # in a fixed random order, so the first choice is not the majority's reading
        cands = [cands[j] for j in rng.permutation(len(cands))]
        items.append({"i": i, "key": r.clause_key, "sample": r.sample, "group": r.ab_group,
                      "place": c["place"] if isinstance(c["place"], str) else "",
                      "ensemble": c["ensemble"] if isinstance(c["ensemble"], str) else "",
                      "context": c["context"], "clause": c["text"], "glossary": c["glossary"] or "",
                      "date_raw": c["date_raw"] if isinstance(c["date_raw"], str) else "",
                      "years_inside": c["years_inside"] if isinstance(c["years_inside"], str) else "",
                      "candidates": [{**x, "text": _sentence(x),
                                      "who": ", ".join(READER_NAMES[n] for n in x["readers"])} for x in cands]})
    return items

def build_page(out: Path, half: str = "calibration") -> Path:
    data = json.dumps(page_items(half=half), ensure_ascii=False).replace("</", "<\\/")
    template = TEMPLATE.read_text(encoding="utf-8")

def fill_sheets(documents: list[dict], path: Path = SHEET) -> dict:
    """Write the page's answers (``choices/<clause_key>``) into the sample sheets' ``v_*``
    columns, with ``v_source = "ab"``, ``ab_group`` and ``ab_prob``.  Only documents marked
    ``ruled``; an answer is the four questions as the page resolved the choice."""
    drawn = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False).set_index("clause_key")
    counts = {}
    for name, sheet_path in ra.SAMPLES.items():
        sheet = pd.read_csv(sheet_path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
        for column in ("v_source", "ab_group", "ab_prob", "v_choice"):
            if column not in sheet:
                sheet[column] = ""
        index = {k: i for i, k in enumerate(sheet["clause_key"])}
        written = 0
        for d in documents:
            key = d.get("clause_key")
            if not d.get("ruled") or key not in index:
                continue
            a = d.get("answer") or {}
            i = index[key]
            sheet.loc[i, ra.VERDICT_COLUMNS] = [
                a.get("activity", ""), a.get("company", ""), a.get("performance", ""), a.get("date", ""),
                *(("" if a.get(k) in (None, "") else str(int(a[k]))) for k in ("start_lo", "start_hi", "end_lo", "end_hi")),
                d.get("note") or ""]
            sheet.loc[i, ["v_source", "ab_group", "ab_prob", "v_choice"]] = [
                "ab", drawn.at[key, "ab_group"], drawn.at[key, "ab_prob"], d.get("choice", "")]
            written += 1
        sheet.to_csv(sheet_path, index=False, encoding="utf-8-sig")
        counts[name] = written
    return counts

The four-question page (``theatre_rule_audit``, D-212) asks the author to answer every clause

- on a clause where the readers **disagree** — about the four questions, Czech activity

def _answers(coded: pd.DataFrame, readings: pd.DataFrame | None = None) -> pd.DataFrame:
    """The four questions' answers from coded clause rows (index ``clause_key``); with
    ``readings``, the date of an undated clause is the reading's."""
    act = [ra._activity_word(a, h, g) for a, h, g in zip(coded["czech_activity"], coded["hedged"], coded["german"])]
    out = pd.DataFrame({"activity": act,
                        "company": np.where(coded["company"].astype(bool), "yes", "no"),
                        "performance": np.where(coded["performance"].astype(bool), "yes", "no")}, index=coded.index)
    out["date"], out["years"] = "", ""
    undated = coded["date_source"].eq("none") if readings is None else coded["clause_key"].isin([])
    if readings is not None:
        r = readings.set_index("clause_key").reindex(coded.index)
        und = coded["_undated"].to_numpy()
        form = r["date_form"].where(und, None)
        given = form.notna() & ~form.isin(["none", "leading"])
        out.loc[und & ~given.to_numpy(), "date"] = "none"
        out.loc[given, "date"] = "given"
        years = r[["start_lo", "start_hi", "end_lo", "end_hi"]].apply(
            lambda x: "/".join("" if pd.isna(v) else str(int(v)) for v in x), axis=1)
        out.loc[given, "years"] = years[given]
    return out

    disputed = table.index[table["disputed"]]

    rows = clauses[clauses["clause_key"].isin(sample_of["clause_key"])]
    c = candidates(reader_answers(rows))
    c["sample"] = sample_of.set_index("clause_key")["sample"].reindex(c.index)
    c = c[c["readers"] >= 2]                         # a clause with one reader has nothing to compare
    disputed = c[c["n_readings"] > 1].assign(ab_group="disputed", ab_prob=1.0)
    agreed_all = c[c["n_readings"] == 1]
    rng = np.random.default_rng(seed)
    take = min(N_AGREED, len(agreed_all))
    agreed = agreed_all.iloc[np.sort(rng.permutation(len(agreed_all))[:take])].assign(
        ab_group="agreed", ab_prob=take / len(agreed_all))
    out = pd.concat([disputed, agreed])
    out = out.iloc[rng.permutation(len(out))]
    out.attrs.update(disputed=len(disputed), agreed=len(agreed_all), drawn_agreed=take, half=half)
    return out.reset_index()


def write_sheet(drawn: pd.DataFrame, path: Path = SHEET) -> Path:
    """The draw, frozen: each clause's candidate readings as drawn (JSON), group and probability.
    Refuses to replace a sheet whose half already has answers in the sample sheets."""
    sheet = drawn[["clause_key", "sample", "ab_group", "ab_prob", "n_readings"]].copy()
    sheet["half"] = drawn.attrs.get("half", "calibration")
    sheet["candidates"] = drawn["candidates"].map(lambda c: json.dumps(c, ensure_ascii=False))
    if Path(path).exists():
        old = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
        answered = set().union(*[set(s.loc[s["v_source"].eq("ab"), "clause_key"]) if "v_source" in s else set()
                                 for s in ra.load_samples().values()])
        if answered & set(old["clause_key"]):
            raise FileExistsError(f"{path}: some of its clauses are answered; it is not redrawn")
        sheet = pd.concat([old[old["half"] != sheet["half"].iat[0]], sheet], ignore_index=True)
    sheet.to_csv(path, index=False, encoding="utf-8-sig")
    return Path(path)


def _sentence(c: dict) -> str:
    act = {"yes": "Czech theatre here", "hedged": "Czech theatre here, hedged (patrně, snad, ?)",
           "german": "German theatre, not Czech", "no": "No Czech theatre"}[c["activity"]]
    parts = [act]
    if c["activity"] in ("yes", "hedged", "german"):
        parts.append("a company existing over time" if c["company"] == "yes" else "no company")
        parts.append("a performance or a number of plays" if c["performance"] == "yes" else "no performance")
    elif c["company"] == "yes" or c["performance"] == "yes":
        parts.append(" and ".join(p for p, v in (("a company", c["company"]), ("a performance", c["performance"])) if v == "yes"))
    if c["date"] == "none":
        parts.append("no usable date")
    elif c["date"] == "given":
        s_lo, s_hi, e_lo, e_hi = (c["years"].split("/") + ["", "", "", ""])[:4]
        span = lambda a, b: a if a == b or not b else f"{a}–{b}"            # noqa: E731
        when = " to ".join(x for x in (span(s_lo, s_hi) and f"from {span(s_lo, s_hi)}", span(e_lo, e_hi) and f"until {span(e_lo, e_hi)}") if x)
        parts.append(f"dated {when}" if when else "dated")
    return " · ".join(parts)


def page_items(path: Path = SHEET, half: str = "calibration") -> list[dict]:
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    sheet = sheet[sheet["half"] == half]
    clauses = atm.load_clauses()
    rows = clauses[clauses["clause_key"].isin(sheet["clause_key"])]
    ctx = atm.add_context(rows.assign(open_kind="x")).set_index("clause_key")
    items = []
    rng = np.random.default_rng(SEED)
    for i, r in enumerate(sheet.itertuples(index=False)):
        c = ctx.loc[r.clause_key]
        cands = json.loads(r.candidates)
        # in a fixed random order, so the first choice is not the majority's reading
        cands = [cands[j] for j in rng.permutation(len(cands))]
        items.append({"i": i, "key": r.clause_key, "sample": r.sample, "group": r.ab_group,
                      "place": c["place"] if isinstance(c["place"], str) else "",
                      "ensemble": c["ensemble"] if isinstance(c["ensemble"], str) else "",
                      "context": c["context"], "clause": c["text"], "glossary": c["glossary"] or "",
                      "date_raw": c["date_raw"] if isinstance(c["date_raw"], str) else "",
                      "years_inside": c["years_inside"] if isinstance(c["years_inside"], str) else "",
                      "candidates": [{**x, "text": _sentence(x),
                                      "who": ", ".join(READER_NAMES[n] for n in x["readers"])} for x in cands]})
    return items


def build_page(out: Path, half: str = "calibration") -> Path:
    data = json.dumps(page_items(half=half), ensure_ascii=False).replace("</", "<\\/")
    template = TEMPLATE.read_text(encoding="utf-8")
    assert template.count(PLACEHOLDER) == 1, "template: one data placeholder"
    Path(out).write_text(template.replace(PLACEHOLDER, data), encoding="utf-8")
    return Path(out)


def fill_sheets(documents: list[dict], path: Path = SHEET) -> dict:
    """Write the page's answers (``choices/<clause_key>``) into the sample sheets' ``v_*``
    columns, with ``v_source = "ab"``, ``ab_group`` and ``ab_prob``.  Only documents marked
    ``ruled``; an answer is the four questions as the page resolved the choice."""
    drawn = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False).set_index("clause_key")
    counts = {}
    for name, sheet_path in ra.SAMPLES.items():
        sheet = pd.read_csv(sheet_path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
        for column in ("v_source", "ab_group", "ab_prob", "v_choice"):
            if column not in sheet:
                sheet[column] = ""
        index = {k: i for i, k in enumerate(sheet["clause_key"])}
        written = 0
        for d in documents:
            key = d.get("clause_key")
            if not d.get("ruled") or key not in index:
                continue
            a = d.get("answer") or {}
            i = index[key]
            sheet.loc[i, ra.VERDICT_COLUMNS] = [
                a.get("activity", ""), a.get("company", ""), a.get("performance", ""), a.get("date", ""),
                *(("" if a.get(k) in (None, "") else str(int(a[k]))) for k in ("start_lo", "start_hi", "end_lo", "end_hi")),
                d.get("note") or ""]
            sheet.loc[i, ["v_source", "ab_group", "ab_prob", "v_choice"]] = [
                "ab", drawn.at[key, "ab_group"], drawn.at[key, "ab_prob"], d.get("choice", "")]
            written += 1
        sheet.to_csv(sheet_path, index=False, encoding="utf-8-sig")
        counts[name] = written
    return counts
