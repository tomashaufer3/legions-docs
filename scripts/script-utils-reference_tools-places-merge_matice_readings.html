How it works

The curation file keeps one row per entry: entry_id, the OCR's residence, the reading, who read it and when. A reading already in the file is kept unless --replace is given, so a correction made by eye is never overwritten by a later machine pass.

#!/usr/bin/env python3
"""
Merge the residence readings made from ``build_matice_read_sheets.py``'s sheets into the
curation file ``reference/places/matice_residence_readings.csv``.

Each reader writes ``data/interim/matice/read_sheets/readings_NN.csv`` (sheet, number,
reading); the manifest ties sheet and number to the entry. The curation file keeps one
row per entry: ``entry_id``, the OCR's residence, the reading, who read it and when. A
reading already in the file is kept unless ``--replace`` is given, so a correction made
by eye is never overwritten by a later machine pass.

Usage::

    python utils/reference_tools/places/merge_matice_readings.py --reader "claude-haiku-5-5 (vision)"
"""

from __future__ import annotations

import argparse
import sys
from datetime import date
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[3]
SHEETS = ROOT / "data" / "interim" / "matice" / "read_sheets"
CURATION = ROOT / "reference" / "places" / "matice_residence_readings.csv"
COLUMNS = ["entry_id", "residence_ocr", "residence_read", "reader", "read_on"]


def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    p.add_argument("--reader", required=True, help="who or what read the sheets")
    p.add_argument("--replace", action="store_true", help="overwrite readings already held")
    args = p.parse_args(argv)

    manifest = pd.read_csv(SHEETS / "manifest.csv", dtype=str, keep_default_na=False)
    parts = [pd.read_csv(f, dtype=str, keep_default_na=False) for f in sorted(SHEETS.glob("readings_*.csv"))]
    read = pd.concat(parts, ignore_index=True)
    read["reading"] = read["reading"].str.strip()
    new = manifest.merge(read, on=["sheet", "number"], how="left", validate="1:1")
    missing = new["reading"].isna().sum()
    new = new[new["reading"].notna()].rename(columns={"reading": "residence_read"})
    new["reader"], new["read_on"] = args.reader, date.today().isoformat()
    new = new[COLUMNS]

    if CURATION.exists():
        old = pd.read_csv(CURATION, dtype=str, keep_default_na=False)
        keep = old if not args.replace else old[~old["entry_id"].isin(new["entry_id"])]
        new = pd.concat([keep, new[~new["entry_id"].isin(keep["entry_id"])]], ignore_index=True)
    assert new["entry_id"].is_unique
    new.sort_values("entry_id").to_csv(CURATION, index=False, encoding="utf-8")
    print(f"{len(new)} readings in {CURATION.relative_to(ROOT)}; {missing} manifest rows unread")
    print(f"no place named: {int((new['residence_read'] == '-').sum())}; "
          f"unreadable: {int((new['residence_read'] == '?').sum())}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
