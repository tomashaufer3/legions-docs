How it works

The full run reads every table page of a volume, the sealed certification pages among them, with fresh reader calls: its labels are not the certification run's, so nothing comes back from that run's cache. Scored against the same sealed sheets, by the same metrics as score_certification.py, it is a second measurement of the route on the same pages — what the route doc planned ("a test–retest, not an independent measure"). It is reported beside the certification score and never in its place:

  • it is not blind: the certification run of these pages was scored and the route changed on what it showed (the German colon and position rules, D-169, were found on these pages), so it measures the route's stability across reads, not its error on unseen pages;
  • the readers sample, so two reads of one page on one route differ by a few marks, and the gap between the two scores is that variation, not a change of route.

score_certification reads a run from FROZEN / f"{edition}_certification" and cannot be edited here: it lives in utils/lexikon_1910, whose every module is digested into the freeze, so a change would take the full run off its gold's route. This module therefore copies each run's outputs into a temporary folder under that name and points the scorer at it. Nothing is written into a run folder: the result goes to OUT, rewritten each run.

python utils/validation/score_sealed_retest.py                 # every edition with a full run
python utils/validation/score_sealed_retest.py --edition de-silesia

def score_run(edition: str, folder: str) -> tuple[dict, dict] | None:
    """``(result, sides)`` for one run folder scored on the edition's sealed sheets, or
    ``None`` when the folder has no final inventory. The result has the shape
    ``score_certification.score`` writes, with the run folder named."""
    source = sc.FROZEN / folder
    if not (source / "final_inventory.csv").is_file():
        return None
    real = sc.FROZEN

        target = Path(tmp) / f"{edition}_certification"
        for name in COPIED:
            if (source / name).is_file():
                (target / name).parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(source / name, target / name)
        sc.FROZEN = Path(tmp)
        try:
            both, categories = sc.sides(edition), sc._categories()
            freeze = sc.freeze_record(edition)
            result = {"edition": f"{edition} — {folder}", "scorer": sc.__version__,
                      "retest": __version__, "scored": time.strftime("%Y-%m-%d %H:%M:%S"),
                      "forced": False, "sheets": [p.name for p in sc.sealed_paths(edition)],
                      "freeze": freeze, "off_plate": both["off_plate"], **sc.measure(both),
                      "marks_awaiting_a_ruling": sc.waiting_marks(edition),
                      "by_category": sc.measure(sc.by_category(both, categories))}

    return result, both

def report(edition: str, results: dict, diffs: list[dict]) -> str:
    lines = [f"Sealed-page test–retest — {edition} (score_sealed_retest {__version__}, D-181)",
             "  the full run's readings of the sealed pages beside the certification run's; "
             "in sample, never the record", ""]
    for name, result in results.items():
        lines += [f"######## {name} run", sc.report(result), ""]
    lines.append("  between the runs (claims one run makes and the other does not)")
    for row in diffs:
        verdict = "right" if row["sheet"] else "false"

    if not diffs:
        lines.append("    none")
    return "\n".join(lines)

def retest(edition: str) -> dict | None:
    """Both runs of one edition scored, the record and its report written to :data:`OUT`."""
    paths = sc.sealed_paths(edition)
    drafts = [p.name for p in paths if sc.sheet_status(p) != "complete"]
    if not paths or drafts:
        print(f"{edition}: sealed sheets not complete ({', '.join(drafts) or 'none'}) — skipped")
        return None
    scored = {name: score_run(edition, folder.format(edition=edition))
              for name, folder in RUNS.items()}
    if scored["full"] is None:
        print(f"{edition}: no full run with a final inventory — skipped")
        return None
    results = {name: pair[0] for name, pair in scored.items() if pair}
    diffs = between(scored["full"][1], scored["certification"][1]) if scored["certification"] else []
    record = {"edition": edition, "retest": __version__, "runs": results, "between": diffs}
    OUT.mkdir(parents=True, exist_ok=True)
    (OUT / f"{edition}.json").write_text(json.dumps(record, indent=1, ensure_ascii=False),
                                         encoding="utf-8")
    text = report(edition, results, diffs)
    (OUT / f"{edition}.txt").write_text(text, encoding="utf-8")
    print(text)
    print(f"  written: {sc._shown(OUT / edition)}.{{json,txt}}")
    return record
