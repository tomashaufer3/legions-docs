TL;DR

Encodes the accented-Czech names for the 1910 Czech-lands districts, keyed to the MPIDR backbone. This is a first pass for verification (Claude + Tomáš): most names are accent restorations of MPIDR's de-accented Czech; ~30 German-only Silesian/Sudeten seats are exonym lookups. Names in REVIEW are the ones to double-check.

def main():
    mp = dz.parse_mpidr(DBF)
    court_de = mp.drop_duplicates("gb_de").set_index("gb_de")["land"].to_dict()
    okres_land = mp.drop_duplicates("okres_code").set_index("okres_code")["land"].to_dict()
    mpidr_cs = mp.drop_duplicates("gb_de").set_index("gb_de")["gb_cs_mpidr"].to_dict()

    # okres modern name = its seat court's modern name (okres shares the seat's name)
    seat_by_okres = {code: cs for code, cs in OKRES_CS.items()}
    court_by_cs = {cs: de for de, cs in COURT_CS.items()}
    okres_modern = {code: MODERN.get(court_by_cs.get(cs, ""), "") for code, cs in seat_by_okres.items()}

    rows = []
    for de, cs in COURT_CS.items():
        rows.append({"level": "court", "key": de, "cs": cs, "cs_modern": MODERN.get(de, ""),
                     "land": court_de.get(de, "?"), "de": de, "review": de in REVIEW})
    for code, cs in OKRES_CS.items():
        rows.append({"level": "okres", "key": code, "cs": cs, "cs_modern": okres_modern.get(code, ""),
                     "land": okres_land.get(code, "?"), "de": "", "review": code in REVIEW})
    out = pd.DataFrame(rows, columns=["level", "key", "cs", "cs_modern", "land", "de", "review"])
    path = ROOT / "reference" / "places" / "district_cs_curation.csv"
    out.to_csv(path, index=False, encoding="utf-8")

    # coverage check against the MPIDR backbone
    miss_court = sorted(set(mp["gb_de"]) - set(COURT_CS))
    miss_okres = sorted(set(mp["okres_code"]) - set(OKRES_CS))
    print(f"wrote {path.relative_to(ROOT)}  ({len(COURT_CS)} courts, {len(OKRES_CS)} okresy, "
          f"{len(REVIEW)} flagged)")
    print(f"MPIDR courts not covered by COURT_CS: {len(miss_court)}  {miss_court[:20]}")
    print(f"MPIDR okresy not covered by OKRES_CS: {len(miss_okres)}  {miss_okres[:20]}")

        rows.append({"level": "court", "key": de, "cs": cs, "cs_modern": MODERN.get(de, ""),
                     "land": court_de.get(de, "?"), "de": de, "review": de in REVIEW})
    for code, cs in OKRES_CS.items():
        rows.append({"level": "okres", "key": code, "cs": cs, "cs_modern": okres_modern.get(code, ""),
                     "land": okres_land.get(code, "?"), "de": "", "review": code in REVIEW})
    out = pd.DataFrame(rows, columns=["level", "key", "cs", "cs_modern", "land", "de", "review"])
    path = ROOT / "reference" / "places" / "district_cs_curation.csv"
    out.to_csv(path, index=False, encoding="utf-8")

    # coverage check against the MPIDR backbone
    miss_court = sorted(set(mp["gb_de"]) - set(COURT_CS))
    miss_okres = sorted(set(mp["okres_code"]) - set(OKRES_CS))
    print(f"wrote {path.relative_to(ROOT)}  ({len(COURT_CS)} courts, {len(OKRES_CS)} okresy, "
          f"{len(REVIEW)} flagged)")
    print(f"MPIDR courts not covered by COURT_CS: {len(miss_court)}  {miss_court[:20]}")
    print(f"MPIDR okresy not covered by OKRES_CS: {len(miss_okres)}  {miss_okres[:20]}")
