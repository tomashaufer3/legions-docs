How it works

  • notebooks/02_data_cleaning calls assert_data_ok as its last cell, so a run that produced broken data stops there instead of reporting success;
  • utils/docs/build_docs.py renders the same list on the site's Data QA sheet.

Why a gate and not just a report

Every bug found in the 2026-07-16 audit was silent: the fabricated exit_N produced a perfectly plausible 20 % "Missing" category, and 6,320 double-placed records still carried valid-looking okres names. Nothing crashed. A check that is only displayed is found whenever someone next opens the page — which may be after the figures, tables and regressions have been built on top of it. A check that fails the run is found at the moment it breaks.

Hard vs soft

A gate that cries wolf gets switched off, so only invariants that should never be violated are hard (hard=True): frame exclusivity, key uniqueness, no fabricated codes, no join blow-up, derived columns present. Coordinate accuracy is soft — it can drift for honest external reasons (Wikidata changes), so it warns rather than fails.

Note the scope: these run after the notebook has written data/final, so they cannot prevent bad data being written. What they prevent is unknowingly building on it.

from utils.validation.data_checks import assert_data_ok, run_checks, report

def run_checks(final: Path | None = None) -> list[Check]:
    """Evaluate every invariant against the built data and return the results.

    Args:
        final: the ``data/final`` directory; defaults to the repo's own.

    Returns:
        One :class:`Check` per invariant, in reporting order.
    """
    from utils.parsing.exit_codes import parse_exit_codes

    fin = Path(final) if final else FINAL
    L = pd.read_parquet(fin / "persons" / "legionnaires_clean.parquet")
    H = pd.read_parquet(fin / "persons" / "habsburg_clean.parquet")
    cen = pd.read_parquet(fin / "census" / "census_okres_1910.parquet")
    setl = pd.read_parquet(fin / "geo" / "settlements_1910.parquet")

    out: list[Check] = []

    # ── the build is current ───────────────────────────────────────────────────
    # A data/final written by an older version of the code lacks the columns the newer code
    # emits, and until 2026-09-22 every check guarded on such a column was simply skipped —
    # the gate reported 72/72 on a build that predated two releases (audit 2026-09-22 §A6).
    # So the schema itself is checked first: every column the code promises must be there.
    from utils.parsing.schema import HABSBURG_FINAL, LEGION_FINAL
    for nm, d, final, extra in (("Legionnaires", L, LEGION_FINAL, ["pol_recorded", "dup_group"]),
                                ("Habsburg fallen", H, HABSBURG_FINAL, ["dup_group"])):
        missing = [c for c in list(final) + extra if c not in d.columns]
        out.append(Check(f"{nm}: data/final carries every column the current code emits",
                         not missing,
                         f"{len(missing)} column(s) missing — data/final predates the code; "
                         f"re-run notebook 02: {missing[:6]}"))

    # A record belongs to at most one 1910 census frame, and must not carry the other
    # frame's attributes. Both were violated before the audit (6,320 + 124 records).
    for nm, d in (("Legionnaires", L), ("Habsburg fallen", H)):
        for role in ("birth", "home"):
            n = int((d[f"{role}_okres_id"].notna() & d[f"{role}_jaras_id"].notna()).sum())
            out.append(Check(f"{nm} · {role}: placed in at most one 1910 frame", n == 0,
                             f"{n} record(s) hold both a Czech okres_id and a Slovak jaras_id"))
            leak = int((d[f"{role}_jaras_id"].notna() &
                        (d[f"{role}_court_id"].notna() | d[f"{role}_lat"].notna() |
                         d[f"{role}_municipality_cs"].notna())).sum())
            out.append(Check(f"{nm} · {role}: no Czech attributes on Slovak-placed records",
                             leak == 0, f"{leak} record(s) leak cross-frame attributes"))

    out.append(Check("Legionnaire primary key unique", bool(L["legion_id"].is_unique),
                     "duplicate legion_id values"))
    out.append(Check("Habsburg primary key unique", bool(H["vua_id"].is_unique),
                     "duplicate vua_id values"))

    # the exact shape of audit finding A3: pd.NA stringified to "<NA>" and its N was
    # scanned into a fabricated "Missing" exit code on every empty cell
    out.append(Check("Missing values do not fabricate an exit code",
                     len(parse_exit_codes(pd.NA)) == 0 and len(parse_exit_codes("?")) == 0,
                     "a missing/placeholder cell parses to a non-empty code set"))
    n_N = int(L["exit_N"].sum())
    out.append(Check(f"exit_N plausible (= {n_N:,}, not ~27k)", n_N < 5000,
                     "exit_N looks fabricated from empty cells — see audit A3"))

    # ── record provenance (Zdroj informací) ───────────────────────────────────────
    # Same failure shape again: an unrecorded source must not become "document absent",
    # and the anchored strip must not leave a document name inside the extra citation.
    if "source_spis" in L.columns:
        from utils.parsing.provenance import KARTA, SPIS, provenance_selftest

        fails = provenance_selftest()
        out.append(Check("Provenance parser self-test", not fails,
                         f"{len(fails)} provenance behaviour(s) regressed: {fails[:3]}"))

        recorded = L["record_source"].notna()
        for col in ("source_spis", "source_karta", "source_extra"):
            mism = int((L[col].notna() != recorded).sum())
            out.append(Check(f"{col} is set exactly where a source is recorded", mism == 0,
                             f"{mism} row(s) where the flag and record_source disagree"))

        mism = int((L["source_extra_ref"].notna() != L["source_extra"].fillna(False)).sum())
        out.append(Check("source_extra_ref present exactly where source_extra is True",
                         mism == 0, f"{mism} row(s) disagree"))

        ref = L["source_extra_ref"].dropna().str.lower()
        leaked = int(ref.str.contains(SPIS, regex=False).sum()
                     + ref.str.contains(KARTA, regex=False).sum())
        out.append(Check("No personnel-document name left in the extra citation", leaked == 0,
                         f"{leaked} extra citation(s) still contain a document name"))

    # ── political affiliation (Politická příslušnost) ─────────────────────────────
    # The same failure shape a third time: the field is blank on 70 % of records and
    # a False dummy there asserts "not a member" about a source that said nothing.
    if "pol_recorded" in L.columns:
        from utils.parsing.affiliations import PARTY_ORDER, affiliation_selftest

        fails = affiliation_selftest()
        out.append(Check("Affiliation parser self-test", not fails,
                         f"{len(fails)} affiliation behaviour(s) regressed: {fails[:3]}"))

        dummies = [f"pol_{c}" for c in PARTY_ORDER if f"pol_{c}" in L.columns]
        missing = [c for c in PARTY_ORDER if f"pol_{c}" not in L.columns]
        out.append(Check("Every canonical in PARTY_ORDER has a dummy column",
                         not missing,
                         f"missing: {missing}"))

        rec = L["pol_recorded"].fillna(False).astype(bool)
        bad = sum(int((L[c].notna() != rec).sum()) for c in dummies)
        out.append(Check("pol_* dummies are <NA> exactly where pol_recorded is False",
                         bad == 0,
                         f"{bad} flag/row disagreement(s) across {len(dummies)} dummies"))

        stray = int((L.loc[~rec, "pol_atlas_codes"].notna()).sum())
        out.append(Check("pol_atlas_codes is <NA> on unrecorded rows", stray == 0,
                         f"{stray} unrecorded row(s) carry atlas codes"))

    # ── education ladder (Vzdělání) ───────────────────────────────────────────────
    from utils.parsing.education import EDUC_ORDER, INSTITUTIONS

    # These five run against any build. The 2.1.0 checks below are guarded on a column
    # that build may not have; these need only the columns education.py has emitted
    # since 2.0.0, so a stale data/final is still held to them.
    ord_ = L["education_ord"]

    # The label and the ordinal are one fact stored twice, and nothing recomputes the
    # pair after `schema.to_english` renames them — so drift between the two would be
    # invisible in every table that reports one and every model that uses the other.
    out.append(Check("education_ord within 0..5",
                     bool(ord_.dropna().between(0, len(EDUC_ORDER) - 1).all()),
                     "an education_ord falls outside the six-rung ladder"))
    rank = L["education"].map({lab: i for i, lab in enumerate(EDUC_ORDER)})
    drift = int((rank.notna() & ord_.notna() & (rank != ord_)).sum())
    out.append(Check("education label and education_ord agree", drift == 0,
                     f"{drift:,} record(s) where the label is not EDUC_ORDER[education_ord]"))

    # The v1.0.0 defect this module was rewritten to remove: `pok.` scored Upper
    # secondary on 8,805 records. The continuation school is off the ordinal and may act
    # only as a *floor* of Elementary, never a promotion — so an apprentice sitting below
    # rung 1 means the floor rule has been inverted again.
    appr = L["education_apprentice"].fillna(False).astype(bool)
    below = int((appr & (ord_.fillna(-1) < 1)).sum())
    out.append(Check("Apprentice flag never leaves a record below Elementary", below == 0,
                     f"{below:,} record(s) flagged apprentice sit below rung 1"))

    # Audit A3 again, on the newest module: an empty `Vzdělání` must not acquire a
    # ladder, a track or a year count on the way through the parser.
    companions = [c for c in L.columns if c.startswith("education") and c != "education"]
    leaked = {c: int(L.loc[L["education"].isna(), c].notna().sum()) for c in companions}
    leaked = {c: n for c, n in leaked.items() if n}
    out.append(Check("A missing education fabricates no companion value", not leaked,
                     f"columns filled where `education` is missing: {leaked}"))

    # `education_complete` is deliberately unanswerable across the vocational branch:
    # Fortbildungsschule hours and Fachschule course lengths varied by Land and by trade
    # (regional laws 1907-1931), so the standard is not established and must stay <NA>
    # rather than be guessed — the same refusal as dropping HISCAM.
    voc = L["education_track"].eq("vocational") & ord_.eq(3)
    guessed = int(L.loc[voc, "education_complete"].notna().sum())
    out.append(Check("No completion claimed for a vocational course of unknown length",
                     guessed == 0,
                     f"{guessed:,} vocational record(s) carry a completion verdict"))

    # education_top names the rung that set the ordinal, so its own INSTITUTIONS entry
    # must agree; a disagreement means the audit column no longer explains the variable.
    tops = L["education_top"].map({k: v[2] for k, v in INSTITUTIONS.items()})
    unknown = int((L["education_top"].notna() & tops.isna()).sum())
    mism = int((tops.notna() & ord_.notna() & (tops != ord_)).sum())
    out.append(Check("education_top's own rung equals education_ord",
                     mism == 0 and unknown == 0,
                     f"{mism:,} record(s) disagree; {unknown:,} name an unknown institution"))

    # educ_top is imputed on the apprentice floor; everywhere else it must be a
    # school the record actually names, and the top rung must be uniquely decided.
    # Guarded on the 2.1.0 column, not on education_top: a data/final built
    # before this release has the latter and not the former.
    if "education_top_imputed" in L.columns:
        from utils.parsing.education import INSTITUTIONS, TIE_BREAK

        out.append(Check("TIE_BREAK covers every institution exactly once",
                         set(TIE_BREAK) == set(INSTITUTIONS)
                         and len(TIE_BREAK) == len(INSTITUTIONS),
                         "TIE_BREAK and INSTITUTIONS disagree — the tie-break is "
                         "not total and ties fall back on string order"))

        has = L["education_top"].notna()
        listed = [
            top in str(inst).split("|")
            for top, inst in zip(L.loc[has, "education_top"],
                                 L.loc[has, "education_institutions"])
        ]
        imputed = L.loc[has, "education_top_imputed"].fillna(False).astype(bool)
        bad = int((~pd.Series(listed, index=imputed.index) & ~imputed).sum())
        out.append(Check("education_top is named in education_institutions "
                         "unless it is imputed", bad == 0,
                         f"{bad} row(s) where educ_top is absent and not flagged"))

        from utils.parsing.education import education_selftest

        fails = education_selftest()
        out.append(Check("Education parser self-test", not fails,
                         f"{len(fails)} education behaviour(s) regressed: {fails[:3]}"))

    # ── occupation -> HISCO / HISCLASS ────────────────────────────────────────────
    # Same failure shape as A3: a missing occupation must not acquire a social class.
    if "occupation_hisco" in L.columns:
        from utils.parsing.occupations import (code_occupation, load_hisco_authority,
                                               matcher_selftest)

        out.append(Check("Missing occupation does not fabricate a HISCO code",
                         code_occupation(pd.NA) is None and code_occupation("") is None,
                         "a missing/empty cell codes to a real occupation"))

        # The matcher's own behaviours, exercised without touching the frames. Every case
        # in there is a bug that shipped once — unanchored abbreviation expansion,
        # `major` inside `generálmajor`, an adjective hijacking its compound.
        fails = matcher_selftest()
        out.append(Check("Occupation matcher self-test", not fails,
                         f"{len(fails)} matcher behaviour(s) regressed: {fails[:3]}"))

        auth = load_hisco_authority()
        known = set(auth.index.get_level_values("hisco"))

        # Both coded fields, not just the enrolment one. The interwar family went
        # unvalidated at first, which is exactly how a HISCO code absent from the
        # authority (`četnictvo` → 58210) survived: it never appears at enrolment, only
        # 58× in `occupation_interwar`, so a check reading `occupation_hisco` alone could
        # not see it. Any prefix-scoped invariant here has to iterate over both.
        for src, prefix, min_cov in (("occupation", "occupation", 0.93),
                                     ("occupation_interwar", "occupation_interwar", 0.70)):
            code_col = f"{prefix}_hisco"
            if code_col not in L.columns:
                continue
            # columns of *this* family only — occupation_* would otherwise swallow
            # occupation_interwar_* and compare them against the wrong source column
            fam = [c for c in L.columns
                   if c.startswith(f"{prefix}_")
                   and not (prefix == "occupation" and c.startswith("occupation_interwar"))]

            empty = L[src].isna()
            leak = [c for c in fam if L.loc[empty, c].notna().any()]
            out.append(Check(f"No HISCO output on empty {src} cells", not leak,
                             f"{len(leak)} column(s) non-null where {src} is <NA>: {leak[:5]}"))

            nonempty = int(L[src].notna().sum())
            coded = int(L[code_col].notna().sum())
            share = coded / nonempty if nonempty else 0.0
            out.append(Check(f"{src} coding coverage ({share:.1%} of non-empty)",
                             share >= min_cov,
                             f"only {coded:,}/{nonempty:,} coded — the curation has regressed"))

            unknown = sorted(set(L[code_col].dropna()) - known)
            out.append(Check(f"Every {src} HISCO code exists in the authority", not unknown,
                             f"{len(unknown)} code(s) absent from hisco_authority.csv: "
                             f"{unknown[:5]}"))

            hc = L[f"{prefix}_hisclass"].dropna()
            out.append(Check(f"{src} HISCLASS within 1..13", bool(hc.between(1, 13).all()),
                             "a HISCLASS value falls outside the documented range"))
            if f"{prefix}_hisclass_5" in L.columns:
                h5 = L[f"{prefix}_hisclass_5"].dropna()
                out.append(Check(f"{src} HISCLASS_5 within 1..5",
                                 bool(h5.between(1, 5).all()),
                                 "a HISCLASS_5 value falls outside 1..5"))

            # A real code must carry its major group. The authority used to label a code
            # from its ancestor group without backfilling the ancestry *codes*, so 553
            # records held a valid HISCO code and a HISCLASS but no `_hisco_major` — a
            # silent hole in a delivered column that no check could see. Sentinels
            # (-1 no occupation, -2 employer named instead, -3) have no major by design.
            major_col = f"{prefix}_hisco_major"
            if major_col in L.columns:
                real = L[code_col].notna() & ~L[code_col].isin(["-1", "-2", "-3"])
                gap = int((real & L[major_col].isna()).sum())
                out.append(Check(f"Every coded {src} carries a HISCO major group", gap == 0,
                                 f"{gap:,} record(s) hold a HISCO code but no major group"))

            # ...and the converse: a sentinel must carry *no* stratification scale. The
            # exclusion was believed to be enforced and was not — it rested on HSN happening
            # to withhold HISCLASS from (-1, status 41), which it does not do for status 42.
            # 3,098 students were SOCPO 42 "middle class, supervisors of skilled workers" and
            # 160 school leavers were HISCLASS 2 / SOCPO 5 Elite. No invariant covered SOCPO
            # at all, so nothing could see it. Both scales, both fields, from now on.
            sentinel = L[code_col].isin(["-1", "-2", "-3"])
            ranked = [c for c in (f"{prefix}_hisclass", f"{prefix}_hisclass_5",
                                  f"{prefix}_socpo", f"{prefix}_hisclass_high")
                      if c in L.columns]
            bad = {c: int(L.loc[sentinel, c].notna().sum()) for c in ranked}
            # `_hisclass_high` may legitimately survive: it is the better of two codes, and a
            # `dual` whose primary is a sentinel still has a real secondary occupation.
            leaked = {c: n for c, n in bad.items() if n and not c.endswith("_hisclass_high")}
            out.append(Check(f"{src}: no social class on a HISCO sentinel", not leaked,
                             f"a HISCO sentinel carries a stratification scale: {leaked}"))

            # Modifiers must be codes HISCO actually defines. `abiturient` and `žák` were
            # coded status 43, which does not exist — the list is 11-13, 21-24, 31-34, 41,
            # 42, 51, 52 — and the invalid key then fell back to the base code unnoticed.
            for mod in ("status", "relation"):
                mod_col = f"{prefix}_{mod}"
                if mod_col not in L.columns:
                    continue
                # The code list itself, not the combinations HSN happened to observe:
                # a legitimate modifier HSN never saw must not fail this check.
                a = auth.reset_index()
                valid = set(a.loc[a[f"{mod}_label"].notna(), mod].astype(str)) | {"-9"}
                bad = sorted(set(L[mod_col].dropna().astype(str)) - valid)
                out.append(Check(f"Every {src} HISCO {mod} code is real", not bad,
                                 f"{mod} value(s) absent from the HISCO code list: {bad[:5]}"))

        # The dual-occupation convenience column must never invent a class of its own.
        both = L[["occupation_hisclass", "occupation_hisclass_2"]]
        expect = both.min(axis=1)
        mism = int((L["occupation_hisclass_high"].fillna(-1) != expect.fillna(-1)).sum())
        out.append(Check("occupation_hisclass_high is the better of the two", mism == 0,
                         f"{mism} row(s) where the convenience column disagrees"))

        # ── qualifier / career stage (occupations 1.2.0) ──────────────────────────
        # Both are descriptive columns derived from the same normalised key, and both
        # have the shape that has produced every real bug in this repo: a derived value
        # that can quietly disagree with the thing it is derived from.
        from utils.parsing.occupations import CAREER_STAGE_ORD, _norm

        if "occupation_career_stage" in L.columns:
            # The ordinal is the map of the label, nowhere else. Checked in both
            # directions so neither an invented rung nor a dropped one passes.
            stage = L["occupation_career_stage"]
            expect_ord = stage.map(CAREER_STAGE_ORD).astype("Int64")
            bad = int((L["occupation_career_stage_ord"].fillna(-1)
                       != expect_ord.fillna(-1)).sum())
            out.append(Check("occupation_career_stage_ord is the map of the label",
                             bad == 0,
                             f"{bad:,} row(s) where the ordinal disagrees with the stage"))
            unknown = sorted(set(stage.dropna()) - set(CAREER_STAGE_ORD))
            out.append(Check("Every career stage is a documented rung", not unknown,
                             f"stage value(s) absent from CAREER_STAGE_ORD: {unknown[:5]}"))

        if "occupation_qualifier" in L.columns:
            q = L["occupation_qualifier"]
            # A `dual` is two occupations, not a trade and its modifier. If a qualifier
            # ever appears on one, the second occupation is being demoted to an adjective
            # and silently dropped from the analysis — the `_2` columns already hold it.
            on_dual = int((q.notna() & L["occupation_match"].eq("dual")).sum())
            out.append(Check("No qualifier on a dual match", on_dual == 0,
                             f"{on_dual:,} dual row(s) carry a qualifier — the second "
                             f"occupation is being read as a modifier"))
            # The qualifier is a *proper* part of the string: if it ever equalled the
            # whole normalised key, the head noun would be empty and the record would be
            # coded from nothing.
            keys = L.loc[q.notna(), "occupation"].map(_norm)
            whole = int((keys.values == q.dropna().values).sum())
            out.append(Check("Qualifier is never the whole occupation string", whole == 0,
                             f"{whole:,} row(s) where the qualifier swallowed the head noun"))

    out.append(Check("illegitimacy_rate built", "illegitimacy_rate" in cen.columns,
                     "derived rate missing from the census panel"))
    out.append(Check("pop_total has no zero rows", int((cen["pop_total"] == 0).sum()) == 0,
                     "an all-missing okres summed to 0 instead of NA"))

    # ── city/rural twins ───────────────────────────────────────────────────────
    # The court-level census sources keep ONE row per twin pair, so the pooled unit is the
    # finest one those covariates are defined on. Three ways this went wrong while being built:
    # blanking the members destroyed the total the pooling needed; pooling by averaging the two
    # shares weighted a 5k city like a 107k ring; and the twin flag drifted from the twin set.
    pooled_p = fin / "census" / "census_okres_1910_pooled.parquet"
    if pooled_p.exists():
        pool = pd.read_parquet(pooled_p)
        n_pairs = int(pool["twin_pooled"].sum())
        out.append(Check(f"Census pooled to the twin unit ({len(pool)} rows, {n_pairs} pooled)",
                         len(pool) == len(cen) - n_pairs and n_pairs == 10,
                         f"{len(pool)} pooled rows vs {len(cen)} okresy — pairs not collapsed"))
        n_na = int(pool["lang_czech_share"].isna().sum())
        out.append(Check("Pooled panel has no missing language share", n_na == 0,
                         f"{n_na} pooled unit(s) still <NA> — the pair total was lost, not merged"))
        # every share must be a ratio of the POOLED counts, never an average of the members'
        bad = int((pool["lang_czech"] / pool["lang_total"]
                   - pool["lang_czech_share"]).abs().gt(1e-9).sum())
        out.append(Check("Pooled shares recomputed from pooled counts", bad == 0,
                         f"{bad} pooled share(s) do not equal counts/total — averaged, not rebuilt"))

    for nm, d in (("Legionnaires", L), ("Habsburg fallen", H)):
        if "birth_okres_twin" not in d.columns:
            continue
        from utils.parsing import records_geo as _rg
        _d = pd.read_parquet(fin / "geo" / "districts_1910.parquet")
        twin_ids = set(_rg.build_twin_pairs(_d))
        n = int((d["birth_okres_id"].isin(twin_ids) != d["birth_okres_twin"].notna()).sum())
        out.append(Check(f"{nm} · birth: twin role set exactly on the twin population", n == 0,
                         f"{n} record(s) disagree between birth_okres_id and birth_okres_twin"))

    # ── Greater Prague (D-36) and the court near-miss repair (D-35) ────────────
    # Both add a column whose whole purpose is to let a decision be reversed downstream. A
    # flag that is set on the wrong rows is worse than no flag, because it invites a filter
    # that silently keeps or drops the wrong ones.
    for nm, d in (("Legionnaires", L), ("Habsburg fallen", H)):
        if "birth_okres_prague" in d.columns:
            tagged = d["birth_okres_prague"].notna()
            # every tagged record must sit in Praha itself or in one of the okresy the rule
            # moves to; anything else means the tag outlived the id it describes
            from utils.parsing import records_geo as _rg
            dest = set(_rg.load_prague_suburbs().values()) | {_rg.PRAHA_OKRES_ID}
            stray = int((tagged & ~d["birth_okres_id"].isin(dest)).sum())
            out.append(Check(f"{nm} · birth: Prague tag only on Prague-area okresy", stray == 0,
                             f"{stray} record(s) tagged birth_okres_prague sit outside "
                             f"okres Praha and its suburb destinations"))
            moved = int((d["birth_okres_prague"] == "suburb").sum())
            bad = int(((d["birth_okres_prague"] == "suburb")
                       & d["birth_okres_id"].eq(_rg.PRAHA_OKRES_ID)).sum())
            out.append(Check(f"{nm} · birth: every 'suburb' record actually left okres Praha "
                             f"({moved:,} moved)", bad == 0,
                             f"{bad} record(s) tagged 'suburb' are still in okres Praha"))
        # A label read off an id goes stale the moment anything rewrites the id, and it goes
        # stale *quietly* — the id is right, the merge is right, and only the human reading a
        # table sees "Praha" against okres Žižkov. The first run of D-36 shipped exactly that
        # on 1,468 records. This is the general guard, not a Prague one.
        if {"birth_okres_id", "birth_okres_cs"} <= set(d.columns):
            _dist = pd.read_parquet(fin / "geo" / "districts_1910.parquet")
            _dist = _dist.drop_duplicates("okres_code")
            _oid = _dist["okres_code"].map(
                lambda c: int(c.split("-")[0]) * 1000 + int(c.split("-")[1]))
            expect = d["birth_okres_id"].map(dict(zip(_oid, _dist["okres_cs"])))
            have = d["birth_okres_cs"]
            n = int((expect.notna() & have.notna() & (expect != have)).sum())
            out.append(Check(f"{nm} · birth: okres name matches okres id", n == 0,
                             f"{n} record(s) carry a district label that is not their id's"))
        if "birth_court_snapped" in d.columns:
            mism = int((d["birth_court_id"].notna() != d["birth_court_snapped"].notna()).sum())
            out.append(Check(f"{nm} · birth: court snap flag present exactly where a court is",
                             mism == 0,
                             f"{mism} record(s) have a court without a flag, or the reverse"))

    # ── járás / chartered-town twins (Slovak lane) ─────────────────────────────
    # Same shape as the Czech twins, and the same two ways of going wrong. The crosswalk one is
    # not hypothetical: the Majo lexicon prints a county's chartered towns as a trailing block,
    # and a section header the parser did not recognise left 27 of them carrying the previous
    # okres — obec Nitra filed under okres Nové Mesto nad Váhom, in the right county and wholly
    # plausible. A name that IS a chartered town of its own county cannot belong to a járás.
    bb_hu_p = fin / "geo" / "districts_hu_1910.parquet"
    setl_hu_p = fin / "geo" / "settlements_hu_1910.parquet"
    if bb_hu_p.exists() and setl_hu_p.exists():
        from utils.parsing import records_geo_hu as _rgh
        from utils.parsing.records_geo import _norm

        bb_hu = pd.read_parquet(bb_hu_p)
        setl_hu = pd.read_parquet(setl_hu_p)
        towns = bb_hu[(bb_hu["unit_type"] != "járás") & bb_hu["name_sk"].notna()]
        # (normalised town name, every county form it can be written in) -> the town's unit_id
        town_by_key: dict[tuple[str, str], str] = {}
        for _, t in towns.iterrows():
            for ck in _rgh._district_keys(t["megye_sk"]):
                town_by_key[(_norm(t["name_sk"]), ck)] = t["unit_id"]
        misfiled = 0
        for _, s in setl_hu.iterrows():
            k = _norm(s["obec_sk"])
            for ck in _rgh._district_keys(s.get("zupa_sk")):
                want = town_by_key.get((k, ck))
                if want is not None and s["unit_id"] != want:
                    misfiled += 1
                    break
        out.append(Check("Chartered towns sit in their own unit, not a járás", misfiled == 0,
                         f"{misfiled} obec row(s) named after a chartered town of their own county "
                         "are filed on a járás — suspect an unmatched section header"))

        # An unprinted column is a true zero, so the categories must reconcile to the total. They
        # are read by x-coordinate, and a bucketing rule that drifts narrow values silently
        # rewrites a village's ethnic composition (Nitra's 9 Croats + 13 Serbs once read as 913
        # Serbs). This is the one check that catches that class at all.
        n_bad = int((~setl_hu["sum_ok"].astype(bool)).sum())
        out.append(Check(f"Obec mother-tongue counts reconcile to the total ({len(setl_hu):,} rows)",
                         n_bad == 0,
                         f"{n_bad} obec row(s) do not sum to their population — suspect the "
                         "x-coordinate column read (lexikon_sk._read_numbers)"))

        for nm, d in (("Legionnaires", L), ("Habsburg fallen", H)):
            if "birth_jaras_twin" not in d.columns:
                continue
            twin_hu = set(_rgh.build_jaras_twin_pairs(bb_hu))
            n = int((d["birth_jaras_id"].isin(twin_hu)
                     != d["birth_jaras_twin"].notna()).sum())
            out.append(Check(f"{nm} · birth: járás twin role set exactly on the twin population",
                             n == 0,
                             f"{n} record(s) disagree between birth_jaras_id and birth_jaras_twin"))

    for nm, d in (("legionnaires", L), ("habsburg", H)):
        w = pd.read_parquet(fin / "analysis" / f"{nm}_census_1910.parquet")
        out.append(Check(f"{nm}_census_1910: census join does not duplicate rows",
                         len(w) == len(d), f"{len(w):,} rows vs {len(d):,} persons — join blowup"))

    # ── rank ladder ────────────────────────────────────────────────────────────
    # Every canonical label must resolve to itself. The rules match by substring in a
    # fixed order, so a mis-ordered or over-broad token silently re-points a whole rung
    # (this is what scored Legion generals as majors before ranks 1.4.0) — and it fails
    # silently, because the wrong answer is still a valid rank.
    from utils.parsing import ranks as R

    bad_rt = [(sys_, lab) for sys_, ladder in
              (("habsburg", R.RANK_ORDINAL_HABSBURG), ("legion", R.RANK_ORDINAL_LEGION))
              for lab in ladder if R.clean_rank(lab, system=sys_) != lab]
    out.append(Check("Every rank label resolves to itself", not bad_rt,
                     f"{len(bad_rt)} label(s) resolve elsewhere, e.g. {bad_rt[:3]}"))

    # No general may score below the general band — the exact shape of the pre-1.4.0
    # bug, asserted on the built data rather than on the table.
    gen_bad = 0
    for col, sys_ in (("legion_rank", "legion"), ("last_legion_rank", "legion"),
                      ("habsburg_rank", "habsburg")):
        if col not in L.columns:
            continue
        v = L[col].astype("string")
        # "generál" as a whole word (excludes "sluha gen. Kornilova"-style mentions)
        named = v.str.contains(r"\bgener[áa]l", case=False, na=False, regex=True)
        ordv = L[f"{col}_common"] if f"{col}_common" in L.columns else None
        if ordv is None:
            continue
        gen_bad += int((named & ordv.notna() & (ordv < 18)).sum())
    out.append(Check("No general scores below the OF-6 band", gen_bad == 0,
                     f"{gen_bad} record(s) name a general but score as a lower grade"))

    # ── unit identity ──────────────────────────────────────────────────────────
    # `regiment` is a display label shared across armies (the Foreign Legion also had a
    # 1st regiment); `unit_key` is the analysable identity. If it silently stops being
    # built, unit-level work would fall back to the ambiguous label.
    if "unit_key" in L.columns:
        orphan = int((L["regiment"].notna() & L["unit_key"].isna()).sum())
        out.append(Check("unit_key present wherever a regiment is", orphan == 0,
                         f"{orphan} record(s) have a regiment but no unit_key"))
    else:
        out.append(Check("unit_key present wherever a regiment is", False,
                         "unit_key column missing — see utils/parsing/units.py 1.1.0"))

    # ── audit 2026-09-22: land, linkage, dates, documentation ─────────────────────
    out.extend(_audit_2026_09_22_checks(fin, L, H))

    # SOFT: external ground truth (Fidler 2025). Only Czechoslovak Legion records are
    # compared — foreign armies numbered their own regiments and legitimately collide.
    out.append(_theatre_vs_fidler(L))

    # SOFT: depends on an external source (Wikidata) that can legitimately shift, so a
    # dip is a prompt to look, not a reason to fail the pipeline. Rows the D-35 near-miss
    # repair snapped into their own okres agree *by construction*, so they are counted as
    # what their raw coordinate was — a disagreement. With them counted as agreeing, the
    # figure measured the repair as much as the coordinates (audit 2026-09-22 §B9); dropping
    # them instead would remove exactly the disagreements and inflate it the other way.
    both = setl[setl["court_id"].notna() & setl["okres_id"].notna()]
    agree = (both["court_id"].astype("Int64") // 1000 == both["okres_id"].astype("Int64"))
    if "court_snapped" in both.columns:
        agree = agree & ~both["court_snapped"].fillna(False).astype(bool)
    acc = float(agree.mean()) if len(both) else 0.0
    out.append(Check(f"Settlement coordinates land in their own okres, before snapping ({acc:.1%})",
                     acc >= 0.85,
                     "coordinate accuracy below 85% — suspect an unguarded name-match fallback",
                     hard=False))
    return out

def report(checks: list[Check] | None = None) -> list[Check]:
    """Print every invariant with its verdict; return the checks."""
    checks = checks if checks is not None else run_checks()
    hard_bad = [c for c in checks if not c.ok and c.hard]
    soft_bad = [c for c in checks if not c.ok and not c.hard]
    npass = sum(c.ok for c in checks)

    print(f"Data QA  ({npass}/{len(checks)} passing)\n")
    for c in checks:
        tag = "pass" if c.ok else ("FAIL" if c.hard else "warn")
        print(f"  [{tag:>4}] {c.label}")
        if not c.ok:
            print(f"         -> {c.detail}")
    if soft_bad:
        print(f"\n  {len(soft_bad)} warning(s) — look, but not a blocker.")
    if hard_bad:
        print(f"\n  {len(hard_bad)} INVARIANT(S) BROKEN — data/final is not safe to build on.")
    return checks

def assert_data_ok(final: Path | None = None, verbose: bool = True) -> list[Check]:
    """Run every invariant and raise :class:`DataCheckError` if a hard one fails.

    This is the pipeline's gate: call it as the final cell of ``02_data_cleaning`` so a
    run cannot report success while having produced unusable data.

    Args:
        final: the ``data/final`` directory; defaults to the repo's own.
        verbose: print the full table (pass ``False`` for just the raise/return).

    Returns:
        The checks, when everything hard passed.

    Raises:
        DataCheckError: listing every hard invariant that failed.
    """
    checks = run_checks(final)
    if verbose:
        report(checks)
    broken = [c for c in checks if not c.ok and c.hard]
    if broken:
        raise DataCheckError(
            f"{len(broken)} data invariant(s) failed:\n"
            + "\n".join(f"  - {c.label}: {c.detail}" for c in broken))
    return checks

"""
data_checks.py — the invariants ``data/final`` must satisfy
===========================================================
One definition of "is the built data sound?", used in two places:

* ``notebooks/02_data_cleaning`` calls :func:`assert_data_ok` as its **last cell**, so a
  run that produced broken data stops there instead of reporting success;
* ``utils/docs/build_docs.py`` renders the same list on the site's *Data QA* sheet.

Why a gate and not just a report
--------------------------------
Every bug found in the 2026-07-16 audit was **silent**: the fabricated ``exit_N`` produced
a perfectly plausible 20 % "Missing" category, and 6,320 double-placed records still
carried valid-looking okres names. Nothing crashed. A check that is only *displayed* is
found whenever someone next opens the page — which may be after the figures, tables and
regressions have been built on top of it. A check that **fails the run** is found at the
moment it breaks.

Hard vs soft
------------
A gate that cries wolf gets switched off, so only invariants that should *never* be
violated are hard (``hard=True``): frame exclusivity, key uniqueness, no fabricated codes,
no join blow-up, derived columns present. Coordinate accuracy is **soft** — it can drift
for honest external reasons (Wikidata changes), so it warns rather than fails.

Note the scope: these run *after* the notebook has written ``data/final``, so they cannot
prevent bad data being written. What they prevent is unknowingly building on it.

Usage
-----
    from utils.validation.data_checks import assert_data_ok, run_checks, report
    assert_data_ok()          # raises DataCheckError listing the hard failures
    report()                  # print the full table, return the checks
"""

from __future__ import annotations

import sys
from dataclasses import dataclass
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))

__version__ = "1.12.0"  # + stale-build, linkage, land, date-range, dictionary/codebook checks (audit 2026-09-22)

FINAL = ROOT / "data" / "final"


class DataCheckError(AssertionError):
    """Raised by :func:`assert_data_ok` when a hard invariant fails."""


@dataclass(frozen=True)
class Check:
    """One invariant: what it asserts, whether it holds, and what a failure means."""

    label: str
    ok: bool
    detail: str            # what it means if it fails — phrased for the failure case
    hard: bool = True      # hard: must never happen. soft: warn only.


AUTHORITY = ROOT / "reference" / "units" / "unit_authority.csv"


def _theatre_vs_fidler(L: pd.DataFrame) -> Check:
    """Compare theatre against the Fidler (2025) unit authority table.

    SOFT by design. It is an *external* yardstick: the authority file is derived from a
    third-party reference work, and a handful of csol.cz records genuinely contradict
    it (a regiment number that only ever existed in France, tagged as Russian). Those
    are source errors we want surfaced, not a reason to fail the build.

    Only Czechoslovak Legion records are compared — the French Foreign Legion, French,
    Serbian and Russian armies numbered their own regiments, so their numbers overlap
    the Legion's by coincidence, not by error. That is what ``unit_key`` separates.
    """
    label = "Theatre agrees with Fidler (2025) unit authority"
    if not AUTHORITY.exists():
        return Check(f"{label} — not run", True,
                     f"authority file absent ({AUTHORITY.name}); "
                     "run utils/reference_tools/units/build_unit_authority.py", hard=False)
    if not {"regiment", "legion_army", "theater"} <= set(L.columns):
        return Check(f"{label} — not run", True, "required columns missing", hard=False)

    auth = pd.read_csv(AUTHORITY, encoding="utf-8")
    regs = auth[(auth["kind"] == "unit") & auth["number"].notna()
                & auth["unit_cs"].str.contains(r"\bPLUK\b", na=False)
                & ~auth["unit_cs"].str.contains("ZVLÁŠTNÍ", na=False)]
    exp: dict[int, set[str]] = {}
    for n, th in zip(regs["number"].astype(int), regs["theatre"]):
        exp.setdefault(n, set()).add(th)

    reg = L["regiment"].astype("string")
    num = reg.str.extract(r"^(\d+)\.")[0].astype("Float64")
    is_cs = L["legion_army"].astype("string").str.startswith("CS Legion", na=False)
    m = num.notna() & is_cs & reg.str.contains(r"(?:stř|p)\.pl\.", na=False)

    agree = conflict = 0
    for n_, th in zip(num[m].astype("Int64"), L.loc[m, "theater"]):
        want = exp.get(int(n_))
        if want is None:
            continue
        agree, conflict = (agree + 1, conflict) if th in want else (agree, conflict + 1)
    tot = agree + conflict
    rate = agree / tot if tot else 1.0
    return Check(f"{label} ({rate:.1%} of {tot:,})", rate >= 0.98,
                 f"agreement {rate:.1%} — {conflict:,} record(s) place a regiment in a "
                 "theatre where Fidler says it never existed", hard=False)


def run_checks(final: Path | None = None) -> list[Check]:
    """Evaluate every invariant against the built data and return the results.

    Args:
        final: the ``data/final`` directory; defaults to the repo's own.

    Returns:
        One :class:`Check` per invariant, in reporting order.
    """
    from utils.parsing.exit_codes import parse_exit_codes

    fin = Path(final) if final else FINAL
    L = pd.read_parquet(fin / "persons" / "legionnaires_clean.parquet")
    H = pd.read_parquet(fin / "persons" / "habsburg_clean.parquet")
    cen = pd.read_parquet(fin / "census" / "census_okres_1910.parquet")
    setl = pd.read_parquet(fin / "geo" / "settlements_1910.parquet")

    out: list[Check] = []

    # ── the build is current ───────────────────────────────────────────────────
    # A data/final written by an older version of the code lacks the columns the newer code
    # emits, and until 2026-09-22 every check guarded on such a column was simply skipped —
    # the gate reported 72/72 on a build that predated two releases (audit 2026-09-22 §A6).
    # So the schema itself is checked first: every column the code promises must be there.
    from utils.parsing.schema import HABSBURG_FINAL, LEGION_FINAL
    for nm, d, final, extra in (("Legionnaires", L, LEGION_FINAL, ["pol_recorded", "dup_group"]),
                                ("Habsburg fallen", H, HABSBURG_FINAL, ["dup_group"])):
        missing = [c for c in list(final) + extra if c not in d.columns]
        out.append(Check(f"{nm}: data/final carries every column the current code emits",
                         not missing,
                         f"{len(missing)} column(s) missing — data/final predates the code; "
                         f"re-run notebook 02: {missing[:6]}"))

    # A record belongs to at most one 1910 census frame, and must not carry the other
    # frame's attributes. Both were violated before the audit (6,320 + 124 records).
    for nm, d in (("Legionnaires", L), ("Habsburg fallen", H)):
        for role in ("birth", "home"):
            n = int((d[f"{role}_okres_id"].notna() & d[f"{role}_jaras_id"].notna()).sum())
            out.append(Check(f"{nm} · {role}: placed in at most one 1910 frame", n == 0,
                             f"{n} record(s) hold both a Czech okres_id and a Slovak jaras_id"))
            leak = int((d[f"{role}_jaras_id"].notna() &
                        (d[f"{role}_court_id"].notna() | d[f"{role}_lat"].notna() |
                         d[f"{role}_municipality_cs"].notna())).sum())
            out.append(Check(f"{nm} · {role}: no Czech attributes on Slovak-placed records",
                             leak == 0, f"{leak} record(s) leak cross-frame attributes"))

    out.append(Check("Legionnaire primary key unique", bool(L["legion_id"].is_unique),
                     "duplicate legion_id values"))
    out.append(Check("Habsburg primary key unique", bool(H["vua_id"].is_unique),
                     "duplicate vua_id values"))

    # the exact shape of audit finding A3: pd.NA stringified to "<NA>" and its N was
    # scanned into a fabricated "Missing" exit code on every empty cell
    out.append(Check("Missing values do not fabricate an exit code",
                     len(parse_exit_codes(pd.NA)) == 0 and len(parse_exit_codes("?")) == 0,
                     "a missing/placeholder cell parses to a non-empty code set"))
    n_N = int(L["exit_N"].sum())
    out.append(Check(f"exit_N plausible (= {n_N:,}, not ~27k)", n_N < 5000,
                     "exit_N looks fabricated from empty cells — see audit A3"))

    # ── record provenance (Zdroj informací) ───────────────────────────────────────
    # Same failure shape again: an unrecorded source must not become "document absent",
    # and the anchored strip must not leave a document name inside the extra citation.
    if "source_spis" in L.columns:
        from utils.parsing.provenance import KARTA, SPIS, provenance_selftest

        fails = provenance_selftest()
        out.append(Check("Provenance parser self-test", not fails,
                         f"{len(fails)} provenance behaviour(s) regressed: {fails[:3]}"))

        recorded = L["record_source"].notna()
        for col in ("source_spis", "source_karta", "source_extra"):
            mism = int((L[col].notna() != recorded).sum())
            out.append(Check(f"{col} is set exactly where a source is recorded", mism == 0,
                             f"{mism} row(s) where the flag and record_source disagree"))

        mism = int((L["source_extra_ref"].notna() != L["source_extra"].fillna(False)).sum())
        out.append(Check("source_extra_ref present exactly where source_extra is True",
                         mism == 0, f"{mism} row(s) disagree"))

        ref = L["source_extra_ref"].dropna().str.lower()
        leaked = int(ref.str.contains(SPIS, regex=False).sum()
                     + ref.str.contains(KARTA, regex=False).sum())
        out.append(Check("No personnel-document name left in the extra citation", leaked == 0,
                         f"{leaked} extra citation(s) still contain a document name"))

    # ── political affiliation (Politická příslušnost) ─────────────────────────────
    # The same failure shape a third time: the field is blank on 70 % of records and
    # a False dummy there asserts "not a member" about a source that said nothing.
    if "pol_recorded" in L.columns:
        from utils.parsing.affiliations import PARTY_ORDER, affiliation_selftest

        fails = affiliation_selftest()
        out.append(Check("Affiliation parser self-test", not fails,
                         f"{len(fails)} affiliation behaviour(s) regressed: {fails[:3]}"))

        dummies = [f"pol_{c}" for c in PARTY_ORDER if f"pol_{c}" in L.columns]
        missing = [c for c in PARTY_ORDER if f"pol_{c}" not in L.columns]
        out.append(Check("Every canonical in PARTY_ORDER has a dummy column",
                         not missing,
                         f"missing: {missing}"))

        rec = L["pol_recorded"].fillna(False).astype(bool)
        bad = sum(int((L[c].notna() != rec).sum()) for c in dummies)
        out.append(Check("pol_* dummies are <NA> exactly where pol_recorded is False",
                         bad == 0,
                         f"{bad} flag/row disagreement(s) across {len(dummies)} dummies"))

        stray = int((L.loc[~rec, "pol_atlas_codes"].notna()).sum())
        out.append(Check("pol_atlas_codes is <NA> on unrecorded rows", stray == 0,
                         f"{stray} unrecorded row(s) carry atlas codes"))

    # ── education ladder (Vzdělání) ───────────────────────────────────────────────
    from utils.parsing.education import EDUC_ORDER, INSTITUTIONS

    # These five run against any build. The 2.1.0 checks below are guarded on a column
    # that build may not have; these need only the columns education.py has emitted
    # since 2.0.0, so a stale data/final is still held to them.
    ord_ = L["education_ord"]

    # The label and the ordinal are one fact stored twice, and nothing recomputes the
    # pair after `schema.to_english` renames them — so drift between the two would be
    # invisible in every table that reports one and every model that uses the other.
    out.append(Check("education_ord within 0..5",
                     bool(ord_.dropna().between(0, len(EDUC_ORDER) - 1).all()),
                     "an education_ord falls outside the six-rung ladder"))
    rank = L["education"].map({lab: i for i, lab in enumerate(EDUC_ORDER)})
    drift = int((rank.notna() & ord_.notna() & (rank != ord_)).sum())
    out.append(Check("education label and education_ord agree", drift == 0,
                     f"{drift:,} record(s) where the label is not EDUC_ORDER[education_ord]"))

    # The v1.0.0 defect this module was rewritten to remove: `pok.` scored Upper
    # secondary on 8,805 records. The continuation school is off the ordinal and may act
    # only as a *floor* of Elementary, never a promotion — so an apprentice sitting below
    # rung 1 means the floor rule has been inverted again.
    appr = L["education_apprentice"].fillna(False).astype(bool)
    below = int((appr & (ord_.fillna(-1) < 1)).sum())
    out.append(Check("Apprentice flag never leaves a record below Elementary", below == 0,
                     f"{below:,} record(s) flagged apprentice sit below rung 1"))

    # Audit A3 again, on the newest module: an empty `Vzdělání` must not acquire a
    # ladder, a track or a year count on the way through the parser.
    companions = [c for c in L.columns if c.startswith("education") and c != "education"]
    leaked = {c: int(L.loc[L["education"].isna(), c].notna().sum()) for c in companions}
    leaked = {c: n for c, n in leaked.items() if n}
    out.append(Check("A missing education fabricates no companion value", not leaked,
                     f"columns filled where `education` is missing: {leaked}"))

    # `education_complete` is deliberately unanswerable across the vocational branch:
    # Fortbildungsschule hours and Fachschule course lengths varied by Land and by trade
    # (regional laws 1907-1931), so the standard is not established and must stay <NA>
    # rather than be guessed — the same refusal as dropping HISCAM.
    voc = L["education_track"].eq("vocational") & ord_.eq(3)
    guessed = int(L.loc[voc, "education_complete"].notna().sum())
    out.append(Check("No completion claimed for a vocational course of unknown length",
                     guessed == 0,
                     f"{guessed:,} vocational record(s) carry a completion verdict"))

    # education_top names the rung that set the ordinal, so its own INSTITUTIONS entry
    # must agree; a disagreement means the audit column no longer explains the variable.
    tops = L["education_top"].map({k: v[2] for k, v in INSTITUTIONS.items()})
    unknown = int((L["education_top"].notna() & tops.isna()).sum())
    mism = int((tops.notna() & ord_.notna() & (tops != ord_)).sum())
    out.append(Check("education_top's own rung equals education_ord",
                     mism == 0 and unknown == 0,
                     f"{mism:,} record(s) disagree; {unknown:,} name an unknown institution"))

    # educ_top is imputed on the apprentice floor; everywhere else it must be a
    # school the record actually names, and the top rung must be uniquely decided.
    # Guarded on the 2.1.0 column, not on education_top: a data/final built
    # before this release has the latter and not the former.
    if "education_top_imputed" in L.columns:
        from utils.parsing.education import INSTITUTIONS, TIE_BREAK

        out.append(Check("TIE_BREAK covers every institution exactly once",
                         set(TIE_BREAK) == set(INSTITUTIONS)
                         and len(TIE_BREAK) == len(INSTITUTIONS),
                         "TIE_BREAK and INSTITUTIONS disagree — the tie-break is "
                         "not total and ties fall back on string order"))

        has = L["education_top"].notna()
        listed = [
            top in str(inst).split("|")
            for top, inst in zip(L.loc[has, "education_top"],
                                 L.loc[has, "education_institutions"])
        ]
        imputed = L.loc[has, "education_top_imputed"].fillna(False).astype(bool)
        bad = int((~pd.Series(listed, index=imputed.index) & ~imputed).sum())
        out.append(Check("education_top is named in education_institutions "
                         "unless it is imputed", bad == 0,
                         f"{bad} row(s) where educ_top is absent and not flagged"))

        from utils.parsing.education import education_selftest

        fails = education_selftest()
        out.append(Check("Education parser self-test", not fails,
                         f"{len(fails)} education behaviour(s) regressed: {fails[:3]}"))

    # ── occupation -> HISCO / HISCLASS ────────────────────────────────────────────
    # Same failure shape as A3: a missing occupation must not acquire a social class.
    if "occupation_hisco" in L.columns:
        from utils.parsing.occupations import (code_occupation, load_hisco_authority,
                                               matcher_selftest)

        out.append(Check("Missing occupation does not fabricate a HISCO code",
                         code_occupation(pd.NA) is None and code_occupation("") is None,
                         "a missing/empty cell codes to a real occupation"))

        # The matcher's own behaviours, exercised without touching the frames. Every case
        # in there is a bug that shipped once — unanchored abbreviation expansion,
        # `major` inside `generálmajor`, an adjective hijacking its compound.
        fails = matcher_selftest()
        out.append(Check("Occupation matcher self-test", not fails,
                         f"{len(fails)} matcher behaviour(s) regressed: {fails[:3]}"))

        auth = load_hisco_authority()
        known = set(auth.index.get_level_values("hisco"))

        # Both coded fields, not just the enrolment one. The interwar family went
        # unvalidated at first, which is exactly how a HISCO code absent from the
        # authority (`četnictvo` → 58210) survived: it never appears at enrolment, only
        # 58× in `occupation_interwar`, so a check reading `occupation_hisco` alone could
        # not see it. Any prefix-scoped invariant here has to iterate over both.
        for src, prefix, min_cov in (("occupation", "occupation", 0.93),
                                     ("occupation_interwar", "occupation_interwar", 0.70)):
            code_col = f"{prefix}_hisco"
            if code_col not in L.columns:
                continue
            # columns of *this* family only — occupation_* would otherwise swallow
            # occupation_interwar_* and compare them against the wrong source column
            fam = [c for c in L.columns
                   if c.startswith(f"{prefix}_")
                   and not (prefix == "occupation" and c.startswith("occupation_interwar"))]

            empty = L[src].isna()
            leak = [c for c in fam if L.loc[empty, c].notna().any()]
            out.append(Check(f"No HISCO output on empty {src} cells", not leak,
                             f"{len(leak)} column(s) non-null where {src} is <NA>: {leak[:5]}"))

            nonempty = int(L[src].notna().sum())
            coded = int(L[code_col].notna().sum())
            share = coded / nonempty if nonempty else 0.0
            out.append(Check(f"{src} coding coverage ({share:.1%} of non-empty)",
                             share >= min_cov,
                             f"only {coded:,}/{nonempty:,} coded — the curation has regressed"))

            unknown = sorted(set(L[code_col].dropna()) - known)
            out.append(Check(f"Every {src} HISCO code exists in the authority", not unknown,
                             f"{len(unknown)} code(s) absent from hisco_authority.csv: "
                             f"{unknown[:5]}"))

            hc = L[f"{prefix}_hisclass"].dropna()
            out.append(Check(f"{src} HISCLASS within 1..13", bool(hc.between(1, 13).all()),
                             "a HISCLASS value falls outside the documented range"))
            if f"{prefix}_hisclass_5" in L.columns:
                h5 = L[f"{prefix}_hisclass_5"].dropna()
                out.append(Check(f"{src} HISCLASS_5 within 1..5",
                                 bool(h5.between(1, 5).all()),
                                 "a HISCLASS_5 value falls outside 1..5"))

            # A real code must carry its major group. The authority used to label a code
            # from its ancestor group without backfilling the ancestry *codes*, so 553
            # records held a valid HISCO code and a HISCLASS but no `_hisco_major` — a
            # silent hole in a delivered column that no check could see. Sentinels
            # (-1 no occupation, -2 employer named instead, -3) have no major by design.
            major_col = f"{prefix}_hisco_major"
            if major_col in L.columns:
                real = L[code_col].notna() & ~L[code_col].isin(["-1", "-2", "-3"])
                gap = int((real & L[major_col].isna()).sum())
                out.append(Check(f"Every coded {src} carries a HISCO major group", gap == 0,
                                 f"{gap:,} record(s) hold a HISCO code but no major group"))

            # ...and the converse: a sentinel must carry *no* stratification scale. The
            # exclusion was believed to be enforced and was not — it rested on HSN happening
            # to withhold HISCLASS from (-1, status 41), which it does not do for status 42.
            # 3,098 students were SOCPO 42 "middle class, supervisors of skilled workers" and
            # 160 school leavers were HISCLASS 2 / SOCPO 5 Elite. No invariant covered SOCPO
            # at all, so nothing could see it. Both scales, both fields, from now on.
            sentinel = L[code_col].isin(["-1", "-2", "-3"])
            ranked = [c for c in (f"{prefix}_hisclass", f"{prefix}_hisclass_5",
                                  f"{prefix}_socpo", f"{prefix}_hisclass_high")
                      if c in L.columns]
            bad = {c: int(L.loc[sentinel, c].notna().sum()) for c in ranked}
            # `_hisclass_high` may legitimately survive: it is the better of two codes, and a
            # `dual` whose primary is a sentinel still has a real secondary occupation.
            leaked = {c: n for c, n in bad.items() if n and not c.endswith("_hisclass_high")}
            out.append(Check(f"{src}: no social class on a HISCO sentinel", not leaked,
                             f"a HISCO sentinel carries a stratification scale: {leaked}"))

            # Modifiers must be codes HISCO actually defines. `abiturient` and `žák` were
            # coded status 43, which does not exist — the list is 11-13, 21-24, 31-34, 41,
            # 42, 51, 52 — and the invalid key then fell back to the base code unnoticed.
            for mod in ("status", "relation"):
                mod_col = f"{prefix}_{mod}"
                if mod_col not in L.columns:
                    continue
                # The code list itself, not the combinations HSN happened to observe:
                # a legitimate modifier HSN never saw must not fail this check.
                a = auth.reset_index()
                valid = set(a.loc[a[f"{mod}_label"].notna(), mod].astype(str)) | {"-9"}
                bad = sorted(set(L[mod_col].dropna().astype(str)) - valid)
                out.append(Check(f"Every {src} HISCO {mod} code is real", not bad,
                                 f"{mod} value(s) absent from the HISCO code list: {bad[:5]}"))

        # The dual-occupation convenience column must never invent a class of its own.
        both = L[["occupation_hisclass", "occupation_hisclass_2"]]
        expect = both.min(axis=1)
        mism = int((L["occupation_hisclass_high"].fillna(-1) != expect.fillna(-1)).sum())
        out.append(Check("occupation_hisclass_high is the better of the two", mism == 0,
                         f"{mism} row(s) where the convenience column disagrees"))

        # ── qualifier / career stage (occupations 1.2.0) ──────────────────────────
        # Both are descriptive columns derived from the same normalised key, and both
        # have the shape that has produced every real bug in this repo: a derived value
        # that can quietly disagree with the thing it is derived from.
        from utils.parsing.occupations import CAREER_STAGE_ORD, _norm

        if "occupation_career_stage" in L.columns:
            # The ordinal is the map of the label, nowhere else. Checked in both
            # directions so neither an invented rung nor a dropped one passes.
            stage = L["occupation_career_stage"]
            expect_ord = stage.map(CAREER_STAGE_ORD).astype("Int64")
            bad = int((L["occupation_career_stage_ord"].fillna(-1)
                       != expect_ord.fillna(-1)).sum())
            out.append(Check("occupation_career_stage_ord is the map of the label",
                             bad == 0,
                             f"{bad:,} row(s) where the ordinal disagrees with the stage"))
            unknown = sorted(set(stage.dropna()) - set(CAREER_STAGE_ORD))
            out.append(Check("Every career stage is a documented rung", not unknown,
                             f"stage value(s) absent from CAREER_STAGE_ORD: {unknown[:5]}"))

        if "occupation_qualifier" in L.columns:
            q = L["occupation_qualifier"]
            # A `dual` is two occupations, not a trade and its modifier. If a qualifier
            # ever appears on one, the second occupation is being demoted to an adjective
            # and silently dropped from the analysis — the `_2` columns already hold it.
            on_dual = int((q.notna() & L["occupation_match"].eq("dual")).sum())
            out.append(Check("No qualifier on a dual match", on_dual == 0,
                             f"{on_dual:,} dual row(s) carry a qualifier — the second "
                             f"occupation is being read as a modifier"))
            # The qualifier is a *proper* part of the string: if it ever equalled the
            # whole normalised key, the head noun would be empty and the record would be
            # coded from nothing.
            keys = L.loc[q.notna(), "occupation"].map(_norm)
            whole = int((keys.values == q.dropna().values).sum())
            out.append(Check("Qualifier is never the whole occupation string", whole == 0,
                             f"{whole:,} row(s) where the qualifier swallowed the head noun"))

    out.append(Check("illegitimacy_rate built", "illegitimacy_rate" in cen.columns,
                     "derived rate missing from the census panel"))
    out.append(Check("pop_total has no zero rows", int((cen["pop_total"] == 0).sum()) == 0,
                     "an all-missing okres summed to 0 instead of NA"))

    # ── city/rural twins ───────────────────────────────────────────────────────
    # The court-level census sources keep ONE row per twin pair, so the pooled unit is the
    # finest one those covariates are defined on. Three ways this went wrong while being built:
    # blanking the members destroyed the total the pooling needed; pooling by averaging the two
    # shares weighted a 5k city like a 107k ring; and the twin flag drifted from the twin set.
    pooled_p = fin / "census" / "census_okres_1910_pooled.parquet"
    if pooled_p.exists():
        pool = pd.read_parquet(pooled_p)
        n_pairs = int(pool["twin_pooled"].sum())
        out.append(Check(f"Census pooled to the twin unit ({len(pool)} rows, {n_pairs} pooled)",
                         len(pool) == len(cen) - n_pairs and n_pairs == 10,
                         f"{len(pool)} pooled rows vs {len(cen)} okresy — pairs not collapsed"))
        n_na = int(pool["lang_czech_share"].isna().sum())
        out.append(Check("Pooled panel has no missing language share", n_na == 0,
                         f"{n_na} pooled unit(s) still <NA> — the pair total was lost, not merged"))
        # every share must be a ratio of the POOLED counts, never an average of the members'
        bad = int((pool["lang_czech"] / pool["lang_total"]
                   - pool["lang_czech_share"]).abs().gt(1e-9).sum())
        out.append(Check("Pooled shares recomputed from pooled counts", bad == 0,
                         f"{bad} pooled share(s) do not equal counts/total — averaged, not rebuilt"))

    for nm, d in (("Legionnaires", L), ("Habsburg fallen", H)):
        if "birth_okres_twin" not in d.columns:
            continue
        from utils.parsing import records_geo as _rg
        _d = pd.read_parquet(fin / "geo" / "districts_1910.parquet")
        twin_ids = set(_rg.build_twin_pairs(_d))
        n = int((d["birth_okres_id"].isin(twin_ids) != d["birth_okres_twin"].notna()).sum())
        out.append(Check(f"{nm} · birth: twin role set exactly on the twin population", n == 0,
                         f"{n} record(s) disagree between birth_okres_id and birth_okres_twin"))

    # ── Greater Prague (D-36) and the court near-miss repair (D-35) ────────────
    # Both add a column whose whole purpose is to let a decision be reversed downstream. A
    # flag that is set on the wrong rows is worse than no flag, because it invites a filter
    # that silently keeps or drops the wrong ones.
    for nm, d in (("Legionnaires", L), ("Habsburg fallen", H)):
        if "birth_okres_prague" in d.columns:
            tagged = d["birth_okres_prague"].notna()
            # every tagged record must sit in Praha itself or in one of the okresy the rule
            # moves to; anything else means the tag outlived the id it describes
            from utils.parsing import records_geo as _rg
            dest = set(_rg.load_prague_suburbs().values()) | {_rg.PRAHA_OKRES_ID}
            stray = int((tagged & ~d["birth_okres_id"].isin(dest)).sum())
            out.append(Check(f"{nm} · birth: Prague tag only on Prague-area okresy", stray == 0,
                             f"{stray} record(s) tagged birth_okres_prague sit outside "
                             f"okres Praha and its suburb destinations"))
            moved = int((d["birth_okres_prague"] == "suburb").sum())
            bad = int(((d["birth_okres_prague"] == "suburb")
                       & d["birth_okres_id"].eq(_rg.PRAHA_OKRES_ID)).sum())
            out.append(Check(f"{nm} · birth: every 'suburb' record actually left okres Praha "
                             f"({moved:,} moved)", bad == 0,
                             f"{bad} record(s) tagged 'suburb' are still in okres Praha"))
        # A label read off an id goes stale the moment anything rewrites the id, and it goes
        # stale *quietly* — the id is right, the merge is right, and only the human reading a
        # table sees "Praha" against okres Žižkov. The first run of D-36 shipped exactly that
        # on 1,468 records. This is the general guard, not a Prague one.
        if {"birth_okres_id", "birth_okres_cs"} <= set(d.columns):
            _dist = pd.read_parquet(fin / "geo" / "districts_1910.parquet")
            _dist = _dist.drop_duplicates("okres_code")
            _oid = _dist["okres_code"].map(
                lambda c: int(c.split("-")[0]) * 1000 + int(c.split("-")[1]))
            expect = d["birth_okres_id"].map(dict(zip(_oid, _dist["okres_cs"])))
            have = d["birth_okres_cs"]
            n = int((expect.notna() & have.notna() & (expect != have)).sum())
            out.append(Check(f"{nm} · birth: okres name matches okres id", n == 0,
                             f"{n} record(s) carry a district label that is not their id's"))
        if "birth_court_snapped" in d.columns:
            mism = int((d["birth_court_id"].notna() != d["birth_court_snapped"].notna()).sum())
            out.append(Check(f"{nm} · birth: court snap flag present exactly where a court is",
                             mism == 0,
                             f"{mism} record(s) have a court without a flag, or the reverse"))

    # ── járás / chartered-town twins (Slovak lane) ─────────────────────────────
    # Same shape as the Czech twins, and the same two ways of going wrong. The crosswalk one is
    # not hypothetical: the Majo lexicon prints a county's chartered towns as a trailing block,
    # and a section header the parser did not recognise left 27 of them carrying the previous
    # okres — obec Nitra filed under okres Nové Mesto nad Váhom, in the right county and wholly
    # plausible. A name that IS a chartered town of its own county cannot belong to a járás.
    bb_hu_p = fin / "geo" / "districts_hu_1910.parquet"
    setl_hu_p = fin / "geo" / "settlements_hu_1910.parquet"
    if bb_hu_p.exists() and setl_hu_p.exists():
        from utils.parsing import records_geo_hu as _rgh
        from utils.parsing.records_geo import _norm

        bb_hu = pd.read_parquet(bb_hu_p)
        setl_hu = pd.read_parquet(setl_hu_p)
        towns = bb_hu[(bb_hu["unit_type"] != "járás") & bb_hu["name_sk"].notna()]
        # (normalised town name, every county form it can be written in) -> the town's unit_id
        town_by_key: dict[tuple[str, str], str] = {}
        for _, t in towns.iterrows():
            for ck in _rgh._district_keys(t["megye_sk"]):
                town_by_key[(_norm(t["name_sk"]), ck)] = t["unit_id"]
        misfiled = 0
        for _, s in setl_hu.iterrows():
            k = _norm(s["obec_sk"])
            for ck in _rgh._district_keys(s.get("zupa_sk")):
                want = town_by_key.get((k, ck))
                if want is not None and s["unit_id"] != want:
                    misfiled += 1
                    break
        out.append(Check("Chartered towns sit in their own unit, not a járás", misfiled == 0,
                         f"{misfiled} obec row(s) named after a chartered town of their own county "
                         "are filed on a járás — suspect an unmatched section header"))

        # An unprinted column is a true zero, so the categories must reconcile to the total. They
        # are read by x-coordinate, and a bucketing rule that drifts narrow values silently
        # rewrites a village's ethnic composition (Nitra's 9 Croats + 13 Serbs once read as 913
        # Serbs). This is the one check that catches that class at all.
        n_bad = int((~setl_hu["sum_ok"].astype(bool)).sum())
        out.append(Check(f"Obec mother-tongue counts reconcile to the total ({len(setl_hu):,} rows)",
                         n_bad == 0,
                         f"{n_bad} obec row(s) do not sum to their population — suspect the "
                         "x-coordinate column read (lexikon_sk._read_numbers)"))

        for nm, d in (("Legionnaires", L), ("Habsburg fallen", H)):
            if "birth_jaras_twin" not in d.columns:
                continue
            twin_hu = set(_rgh.build_jaras_twin_pairs(bb_hu))
            n = int((d["birth_jaras_id"].isin(twin_hu)
                     != d["birth_jaras_twin"].notna()).sum())
            out.append(Check(f"{nm} · birth: járás twin role set exactly on the twin population",
                             n == 0,
                             f"{n} record(s) disagree between birth_jaras_id and birth_jaras_twin"))

    for nm, d in (("legionnaires", L), ("habsburg", H)):
        w = pd.read_parquet(fin / "analysis" / f"{nm}_census_1910.parquet")
        out.append(Check(f"{nm}_census_1910: census join does not duplicate rows",
                         len(w) == len(d), f"{len(w):,} rows vs {len(d):,} persons — join blowup"))

    # ── rank ladder ────────────────────────────────────────────────────────────
    # Every canonical label must resolve to itself. The rules match by substring in a
    # fixed order, so a mis-ordered or over-broad token silently re-points a whole rung
    # (this is what scored Legion generals as majors before ranks 1.4.0) — and it fails
    # silently, because the wrong answer is still a valid rank.
    from utils.parsing import ranks as R

    bad_rt = [(sys_, lab) for sys_, ladder in
              (("habsburg", R.RANK_ORDINAL_HABSBURG), ("legion", R.RANK_ORDINAL_LEGION))
              for lab in ladder if R.clean_rank(lab, system=sys_) != lab]
    out.append(Check("Every rank label resolves to itself", not bad_rt,
                     f"{len(bad_rt)} label(s) resolve elsewhere, e.g. {bad_rt[:3]}"))

    # No general may score below the general band — the exact shape of the pre-1.4.0
    # bug, asserted on the built data rather than on the table.
    gen_bad = 0
    for col, sys_ in (("legion_rank", "legion"), ("last_legion_rank", "legion"),
                      ("habsburg_rank", "habsburg")):
        if col not in L.columns:
            continue
        v = L[col].astype("string")
        # "generál" as a whole word (excludes "sluha gen. Kornilova"-style mentions)
        named = v.str.contains(r"\bgener[áa]l", case=False, na=False, regex=True)
        ordv = L[f"{col}_common"] if f"{col}_common" in L.columns else None
        if ordv is None:
            continue
        gen_bad += int((named & ordv.notna() & (ordv < 18)).sum())
    out.append(Check("No general scores below the OF-6 band", gen_bad == 0,
                     f"{gen_bad} record(s) name a general but score as a lower grade"))

    # ── unit identity ──────────────────────────────────────────────────────────
    # `regiment` is a display label shared across armies (the Foreign Legion also had a
    # 1st regiment); `unit_key` is the analysable identity. If it silently stops being
    # built, unit-level work would fall back to the ambiguous label.
    if "unit_key" in L.columns:
        orphan = int((L["regiment"].notna() & L["unit_key"].isna()).sum())
        out.append(Check("unit_key present wherever a regiment is", orphan == 0,
                         f"{orphan} record(s) have a regiment but no unit_key"))
    else:
        out.append(Check("unit_key present wherever a regiment is", False,
                         "unit_key column missing — see utils/parsing/units.py 1.1.0"))

    # ── audit 2026-09-22: land, linkage, dates, documentation ─────────────────────
    out.extend(_audit_2026_09_22_checks(fin, L, H))

    # SOFT: external ground truth (Fidler 2025). Only Czechoslovak Legion records are
    # compared — foreign armies numbered their own regiments and legitimately collide.
    out.append(_theatre_vs_fidler(L))

    # SOFT: depends on an external source (Wikidata) that can legitimately shift, so a
    # dip is a prompt to look, not a reason to fail the pipeline. Rows the D-35 near-miss
    # repair snapped into their own okres agree *by construction*, so they are counted as
    # what their raw coordinate was — a disagreement. With them counted as agreeing, the
    # figure measured the repair as much as the coordinates (audit 2026-09-22 §B9); dropping
    # them instead would remove exactly the disagreements and inflate it the other way.
    both = setl[setl["court_id"].notna() & setl["okres_id"].notna()]
    agree = (both["court_id"].astype("Int64") // 1000 == both["okres_id"].astype("Int64"))
    if "court_snapped" in both.columns:
        agree = agree & ~both["court_snapped"].fillna(False).astype(bool)
    acc = float(agree.mean()) if len(both) else 0.0
    out.append(Check(f"Settlement coordinates land in their own okres, before snapping ({acc:.1%})",
                     acc >= 0.85,
                     "coordinate accuracy below 85% — suspect an unguarded name-match fallback",
                     hard=False))
    return out


def _audit_2026_09_22_checks(fin: Path, L: pd.DataFrame, H: pd.DataFrame) -> list[Check]:
    """The invariants behind the defects the 2026-09-22 audit found (see its §A/§B)."""
    from utils import linkage as lk
    from utils.parsing import habsburg as hb
    from utils.parsing import ranks as R
    from utils.parsing import records_geo as rg
    from utils.parsing import records_geo_hu as rgh
    from utils.parsing import theater as th

    out: list[Check] = []

    # Self-tests of the behaviours that shipped wrong.
    for name, fails in (("Habsburg note parser", hb.habsburg_selftest()),
                        ("Theatre classifier", th.theater_selftest()),
                        ("Linkage name test", lk.linkage_selftest())):
        out.append(Check(f"{name} self-test", not fails,
                         f"{len(fails)} behaviour(s) regressed: {fails[:3]}"))
    bb_hu_p = fin / "geo" / "districts_hu_1910.parquet"
    if bb_hu_p.exists():
        fails = rgh.records_geo_hu_selftest(pd.read_parquet(bb_hu_p))
        out.append(Check("Slovak district matcher self-test", not fails,
                         f"{len(fails)} behaviour(s) regressed: {fails[:3]}"))

    # §A4/§B6: a record placed in a Slovak járás carries the Slovak land — which rules out
    # the 214 Galicia-born legionnaires once filed under járás Halič.
    for nm, d in (("Legionnaires", L), ("Habsburg fallen", H)):
        for role in ("birth", "home"):
            jid, land = f"{role}_jaras_id", f"{role}_land"
            if jid not in d.columns or land not in d.columns:
                continue
            bad = int((d[jid].notna() & d[land].ne("Slovakia")).sum())
            out.append(Check(f"{nm} · {role}: every járás-placed record has land Slovakia",
                             bad == 0, f"{bad} record(s) sit in a Slovak járás with another land"))

    # §A7: the Cisleithanian flag is a function of the corrected land, never of raw text.
    for role in ("birth", "home"):
        flag, land = f"{role}_cisleithanian", f"{role}_land"
        if flag in H.columns and land in H.columns:
            want = H[land].map(lambda v: rg._CISLEITHANIAN.get(v, pd.NA) if pd.notna(v) else pd.NA)
            have = H[flag].astype("object")
            bad = int(sum(1 for w, h in zip(want, have)
                          if not ((pd.isna(w) and pd.isna(h)) or (pd.notna(w) and pd.notna(h)
                                                                   and bool(w) == bool(h)))))
            out.append(Check(f"Habsburg fallen · {role}_cisleithanian agrees with {role}_land",
                             bad == 0, f"{bad} record(s) where the flag contradicts the land"))

    # §A1/§A3: the linkage is one-to-one, never links two different full dates of birth,
    # and never rests on a surname below the floor.
    mp = fin / "persons" / "overlap_matches.csv"
    if mp.exists():
        M = pd.read_csv(mp)
        dup = int(M["vua_id"].duplicated().sum() + M["legion_id"].duplicated().sum())
        out.append(Check("Linkage is one-to-one (no id matched twice on either side)", dup == 0,
                         f"{dup} surplus link(s)"))
        lb = lk._full_dob(L).set_axis(L["legion_id"])
        hbd = lk._full_dob(H).set_axis(H["vua_id"])
        a, b = M["legion_id"].map(lb).fillna(""), M["vua_id"].map(hbd).fillna("")
        conflict = int(((a != "") & (b != "") & (a != b)).sum())
        out.append(Check("No linked pair has two full dates of birth that disagree", conflict == 0,
                         f"{conflict} linked pair(s) with conflicting full DOBs"))
        sur = [lk.names_compatible(x, "x", y, "x", corroborated=bool(o))[1]
               for x, y, o in zip(M["surname_legion"], M["surname_vua"], M["okres_match"])]
        floor = [lk.SURNAME_MIN_CORROBORATED if bool(o) else lk.SURNAME_MIN
                 for o in M["okres_match"]]
        low = int(sum(s < f for s, f in zip(sur, floor)))
        out.append(Check("Every linked pair clears the surname floor", low == 0,
                         f"{low} linked pair(s) with a surname below the floor"))
        # both sides tell the same story
        lm = L.set_index("legion_id")["match_id"].dropna()
        hm = H.set_index("vua_id")["match_id"].dropna()
        asym = int(sum(1 for lid, vid in lm.items() if hm.get(vid) != lid))
        out.append(Check("Legion-side and VÚA-side links agree", asym == 0,
                         f"{asym} legion link(s) the VÚA side does not return"))

    # §A8/§B15: every event date inside its range, and a precision wherever there is a date.
    for col, lo, hi in (("exit_date", "1914-01-01", "1939-12-31"),
                        ("capture_date", "1914-01-01", "1918-12-31"),
                        ("enroll_date", "1914-01-01", "1922-12-31")):
        if col not in L.columns:
            continue
        s = L[col]
        bad = int((s.notna() & ~s.between(lo, hi)).sum())
        out.append(Check(f"{col} within {lo[:4]}–{hi[:4]}", bad == 0,
                         f"{bad} date(s) outside the range"))
        pc = f"{col}_precision"
        if pc in L.columns:
            mism = int((s.notna() != L[pc].notna()).sum())
            out.append(Check(f"{pc} present exactly where {col} is", mism == 0,
                             f"{mism} row(s) disagree"))
    if {"exit_date", "enroll_date"} <= set(L.columns):
        # day-precise dates only: a year-only exit sits on 1 January and would "precede" any
        # enrolment later that year without contradicting it
        day = pd.Series(True, index=L.index)
        for pc in ("exit_date_precision", "enroll_date_precision"):
            if pc in L.columns:
                day &= L[pc].eq("day").fillna(False)
        early = int((day & (L["exit_date"] < L["enroll_date"])).sum())
        out.append(Check(f"Exit before enrolment, day-precise dates ({early:,} — source contradictions)",
                         early < 500, "exit dates before enrolment have grown — look at the parse",
                         hard=False))

    # §B18: every rank on either ladder reaches the common scale the reset differences.
    miss = [lab for ladder in (R.RANK_ORDINAL_HABSBURG, R.RANK_ORDINAL_LEGION)
            for lab in ladder if lab not in R.RANK_COMMON]
    out.append(Check("Every ladder rank has a common grade", not miss,
                     f"rank(s) with no common grade: {miss}"))

    # §B19/§B20: every delivered column is documented.
    dd_p = fin / "data_dictionary.csv"
    if dd_p.exists():
        dd = pd.read_csv(dd_p)
        for nm, ds, d in (("Legionnaires", "legion", L), ("Habsburg fallen", "habsburg", H)):
            names = set(dd.loc[dd["dataset"] == ds, "final_name"])
            undoc = [c for c in d.columns if c not in names and not c.startswith("pol_")]
            out.append(Check(f"{nm}: every column is in the data dictionary", not undoc,
                             f"{len(undoc)} undocumented column(s): {undoc[:6]}"))
    cb_p = ROOT / "reference" / "census" / "census_codebook.csv"
    if cb_p.exists():
        import fnmatch
        cb = pd.read_csv(cb_p)
        ids = {"okres_id", "land", "okres_code", "okres_cs", "okres_de", "census_okres_id",
               "twin_pooled", "unit_id", "megye_id", "megye_hu", "name_hu", "name_sk"}
        for frame, path in (("cis", fin / "census" / "census_okres_1910.parquet"),
                            ("hu", fin / "census" / "census_hu_jaras_1910.parquet")):
            if not path.exists():
                continue
            cols = pd.read_parquet(path).columns
            pats = [p.strip() for e in cb.loc[cb["frame"] == frame, "english_column"]
                    for p in str(e).split("/")]
            undoc = [c for c in cols if c not in ids
                     and not any(fnmatch.fnmatchcase(c, p) for p in pats)]
            out.append(Check(f"Census codebook ({frame}) documents every panel column", not undoc,
                             f"{len(undoc)} undocumented column(s): {undoc[:6]}"))
    return out


def report(checks: list[Check] | None = None) -> list[Check]:
    """Print every invariant with its verdict; return the checks."""
    checks = checks if checks is not None else run_checks()
    hard_bad = [c for c in checks if not c.ok and c.hard]
    soft_bad = [c for c in checks if not c.ok and not c.hard]
    npass = sum(c.ok for c in checks)

    print(f"Data QA  ({npass}/{len(checks)} passing)\n")
    for c in checks:
        tag = "pass" if c.ok else ("FAIL" if c.hard else "warn")
        print(f"  [{tag:>4}] {c.label}")
        if not c.ok:
            print(f"         -> {c.detail}")
    if soft_bad:
        print(f"\n  {len(soft_bad)} warning(s) — look, but not a blocker.")
    if hard_bad:
        print(f"\n  {len(hard_bad)} INVARIANT(S) BROKEN — data/final is not safe to build on.")
    return checks


def assert_data_ok(final: Path | None = None, verbose: bool = True) -> list[Check]:
    """Run every invariant and raise :class:`DataCheckError` if a hard one fails.

    This is the pipeline's gate: call it as the final cell of ``02_data_cleaning`` so a
    run cannot report success while having produced unusable data.

    Args:
        final: the ``data/final`` directory; defaults to the repo's own.
        verbose: print the full table (pass ``False`` for just the raise/return).

    Returns:
        The checks, when everything hard passed.

    Raises:
        DataCheckError: listing every hard invariant that failed.
    """
    checks = run_checks(final)
    if verbose:
        report(checks)
    broken = [c for c in checks if not c.ok and c.hard]
    if broken:
        raise DataCheckError(
            f"{len(broken)} data invariant(s) failed:\n"
            + "\n".join(f"  - {c.label}: {c.detail}" for c in broken))
    return checks
