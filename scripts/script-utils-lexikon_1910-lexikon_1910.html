How it works

The 1910 special place register is a 17-column table printed one politický okres at a time: a bold municipality row carrying the area in hectares, followed by its osady (settlements), which carry no area. This module reconstructs that table from the NDK ALTO layer of the Czech volume and grades the result against the arithmetic the table itself must satisfy.

Three facts about the source do the structural work, and none of them depend on reading a character correctly:

  1. The printed rules are in the ALTO. GraphicalElement entries taller than they are wide are the column rules; the tall ones bound the table body. Column boundaries are therefore measured, not inferred from a text histogram.
  2. The area column is printed only on municipality rows. That is the block delimiter — typography is not usable here, because the NDK TextStyle entries are synthetic (Times New Roman, sizes 5–10) and carry no weight.
  3. Column 1 runs two interleaved integer sequences — municipalities incrementing across the okres, osady restarting at 1 inside each. The OCR mangles them (o, í), Г) o), but the sequence is known a priori, so row identity is reconstructed rather than read.

The invariants in INVARIANTS are the grader. They are deliberately kept out of the transcription prompt (see run_lexikon_1910_pilot.py): a model told what the numbers must add up to will produce numbers that add up.

Column labels are provisional — the groupings the invariants use are certain (they are what the rules enclose), but the label on columns 9/10 in particular is read off the header and awaits confirmation against the preface, pp. VIII–XIV.

def summary_label(name: str) -> tuple[str, str]:
    """``(kind, district)`` for a summary row, or ``("", "")`` for anything else.

    ``kind`` is ``"okres"`` or ``"court"``. An okres that is simultaneously its own
    court district matches both patterns and is reported as ``"okres"`` — the wider
    unit wins, and the caller sets the court to the same name.

    The name is taken from the end of the header, and the header can be one marker
    or two. Two adjacent markers are one compound header — ``Politický okres
    zároveň s. o. Kraslice`` names an okres that is also its own court — so the cut
    goes after the second. But a marker far downstream is *not* part of the header:
    it is the footnote soup printed after the name, as in ``P. 0. Kralupy nad
    Vltavou, Kralup an der Moldau *) . 2 s. 0.. 42 Ф``. Cutting at the furthest
    marker swallowed that name whole and returned nothing, so distance decides.
    """
    text = str(name or "")
    okres = SUMMARY_OKRES_RE.search(text)
    court = SUMMARY_COURT_RE.search(text)
    if not (okres or court):
        return "", ""
    marks = sorted((m for m in (okres, court) if m), key=lambda m: m.start())
    end = marks[0].end()
    for mark in marks[1:]:
        if mark.start() - end > _ADJACENT_MARKER:
            break
        end = mark.end()
    label = _ENUM_PREFIX_RE.sub("", text[end:].split(",")[0])
    # Keep the leading run of words and stop at the first symbol: the register
    # prints the Czech and German names together and the OCR fills the gap with
    # glyph noise ("Jirkov^ Grörkau.. .. 28 ©"). Multi-word names survive whole,
    # which matters for "Kralupy nad Vltavou" and "Jablonec nad Nisou".
    words = re.match(r"[^\W\d_]+(?:[\s\-][^\W\d_]+)*", label.strip(" ."))
    return ("okres" if okres else "court"), (words.group(0) if words else "")

def parse_int_cell(raw: str) -> tuple[int | None, str]:
    """Parse a numeric cell strictly.

    Returns ``(value, status)`` where status is ``"blank"``, ``"ok"`` or ``"bad"``.
    Anything that is not pure digits after removing separators is ``"bad"`` — the
    OCR's ``23S``, ``íttiS``, ``58-4`` and ``07`` must surface as unreadable, not
    be coerced into a number.

    **A printed dash is a zero, not an absence**, and is returned as ``(0, "ok")``.
    The register sets it where a figure really is none, so reading it as unreadable
    threw away a value the press had stated. It is rare in this table — the nil
    marker here is usually a centred dot, and the OCR usually drops that entirely —
    but where the dash survives it is the strongest evidence available that the
    cell is zero rather than missing.

    An *empty* cell still reports ``"blank"``. Blank also means zero in this table,
    but it is not the same evidence: it is equally the trace of a nil marker the
    scanner lost and of a cell the parser never filled, so counting it as a read
    zero would let a row of nothing satisfy an identity trivially.
    """
    if raw is None:
        return None, "blank"
    text = str(raw).strip()
    if text and _DASH_RE.match(text):
        return 0, "ok"
    s = _STRIP_RE.sub("", str(raw))
    if not s:
        return None, "blank"
    return (int(s), "ok") if _DIGITS_RE.match(s) else (None, "bad")

def read_alto(path: str | Path) -> AltoPage:
    """Parse an NDK ALTO file into words and printed rules."""
    from lxml import etree

    path = Path(path)
    root = etree.parse(str(path)).getroot()
    ns = root.nsmap.get(None)

    def each(tag: str):
        return root.iter(f"{{{ns}}}{tag}" if ns else tag)

    _KEYS = ("HEIGHT", "WIDTH", "VPOS", "HPOS")

    def box(el) -> dict | None:
        """Geometry dict, or None when the engine emitted an element without one."""
        values = [el.get(k) for k in _KEYS]
        if any(v is None for v in values):
            return None
        return dict(zip(_KEYS, (int(v) for v in values)))

    page = next(each("Page"))
    graphics = [b for b in (box(e) for e in each("GraphicalElement")) if b]
    strings = [
        (b["VPOS"], b["HPOS"], b["WIDTH"], b["HEIGHT"], e.get("CONTENT") or "")
        for e, b in ((e, box(e)) for e in each("String")) if b
    ]
    print_space = next((box(e) for e in each("PrintSpace")), None)
    page_width = int(page.get("WIDTH"))
    return AltoPage(
        path=path,
        width=page_width,
        height=int(page.get("HEIGHT")),
        strings=strings,
        verticals=[g for g in graphics if g["HEIGHT"] > g["WIDTH"] * 3],
        horizontals=[g for g in graphics if g["WIDTH"] > g["HEIGHT"] * 3],
        print_left=print_space["HPOS"] if print_space else 0,
        print_width=print_space["WIDTH"] if print_space else page_width,
    )

def column_boundaries(page: AltoPage, merge_within: int = 30) -> list[int]:
    """x positions of the table's column rules, left to right.

    The frame contributes the outer two boundaries; the body rules contribute the
    interior ones. The engine often prints one rule as an adjacent pair, so
    positions closer than ``merge_within`` pixels are collapsed.
    """
    frame, body = _rule_sets(page)
    centres = sorted(v["HPOS"] + v["WIDTH"] // 2 for v in frame + body)
    cols: list[int] = []
    for centre in centres:
        if cols and centre - cols[-1] < merge_within:
            cols[-1] = (cols[-1] + centre) // 2
        else:
            cols.append(centre)
    return cols

def column_template(alto_paths, expected: int = N_COLS + 1) -> tuple[list[float], int]:
    """Median normalised boundary positions over the pages with a complete rule set.

    Returns ``(template, n_pages)`` with positions expressed as a fraction of
    ``PrintSpace`` width.
    """
    import statistics

    observed: list[list[float]] = []
    for path in alto_paths:
        try:
            page = read_alto(path)
            cols = column_boundaries(page)
        except Exception:
            continue
        if len(cols) == expected and page.print_width:
            observed.append([(c - page.print_left) / page.print_width for c in cols])
    if len(observed) < 20:
        raise ValueError(f"only {len(observed)} clean pages — too few to fit a grid")
    template = [statistics.median(row[i] for row in observed) for i in range(expected)]
    return template, len(observed)

def fit_columns(page: AltoPage, template: list[float],
                snap_share: float = 0.35) -> tuple[list[int], int]:
    """Fit ``template`` to one page, snapping to the rules the engine did find.

    Each predicted boundary is pulled onto the nearest observed rule within
    ``snap_share`` of the narrowest column; boundaries with no rule nearby keep the
    predicted position. A median residual correction is applied first so a page
    whose scan is shifted relative to its ``PrintSpace`` still lands correctly.

    Returns ``(boundaries, n_observed)``.
    """
    predicted = [page.print_left + f * page.print_width for f in template]
    # Snap against *every* vertical rule, not just the full-height ones: the engine
    # frequently reports a column rule broken into shorter segments, and a segment
    # is still evidence of where that column is.
    tallest = max((v["HEIGHT"] for v in page.verticals), default=0)
    rules = sorted(v["HPOS"] + v["WIDTH"] // 2 for v in page.verticals
                   if v["HEIGHT"] >= 0.15 * tallest)
    if not rules:
        return [round(p) for p in predicted], 0

    tolerance = snap_share * min(b - a for a, b in zip(predicted, predicted[1:]))

    def nearest(x: float) -> int | None:
        best = min(rules, key=lambda r: abs(r - x))
        return best if abs(best - x) <= tolerance else None

    residuals = [n - p for p in predicted if (n := nearest(p)) is not None]
    if residuals:
        import statistics
        shift = statistics.median(residuals)
        predicted = [p + shift for p in predicted]

    boundaries, observed = [], 0
    for p in predicted:
        match = nearest(p)
        if match is None:
            boundaries.append(round(p))
        else:
            boundaries.append(match)
            observed += 1
    return boundaries, observed

def body_band(page: AltoPage) -> tuple[int, int]:
    """``(top, bottom)`` of the table body in page pixels.

    The band is where *every* column rule is present: it starts below the deepest
    header sub-division and ends where the shortest body rule stops, above the
    footnote block. Taking the intersection rather than the union keeps the
    footnote apparatus and the running head out of the row clustering.
    """
    _frame, body = _rule_sets(page)
    tops = sorted(v["VPOS"] for v in body)
    bottoms = sorted(v["VPOS"] + v["HEIGHT"] for v in body)
    # Percentiles rather than max/min: one over-long or over-short rule should not
    # swallow a fifth of the page's rows.
    top = tops[int(0.85 * (len(tops) - 1))]
    bottom = bottoms[int(0.15 * (len(bottoms) - 1))]
    wide = [h["VPOS"] + h["HEIGHT"] for h in page.horizontals
            if h["WIDTH"] > page.width * 0.5 and h["VPOS"] < bottom * 0.4]
    if wide:
        top = max(top, max(wide))
    if bottom - top < page.height * 0.2:
        raise ValueError(f"{page.stem}: implausible body band {top}–{bottom}")
    return top + 5, bottom - 5

                 line_gap: int = 15, foot_from_rules: bool = True,
                 top_from_header: bool = True) -> list[Row]:
    """Assign every body word to a column, then cluster words into printed lines.

    ``template`` is the fitted grid from :func:`column_template`; without it the
    page's own rules must already resolve to exactly 18 boundaries, which they do
    on about a fifth of pages. ``line_gap`` is the vertical distance at which a new
    baseline starts. Wrapped name lines — a long municipality name spilling onto a
    second line with no numbers beside it — are folded into the preceding row.

    The body ends at :func:`table_bottom`, where the apparatus opens.
    ``foot_from_rules=False`` ends it at :func:`body_band`'s foot, as 0.1.0 did,
    for replaying a pass cached under it: that loses the last rows on half the
    table scans and reads apparatus lines as rows on 22.

    ``top_from_header=True`` (the default since 0.4.0) opens the body at :func:`table_top`, under the
    header's closing rule, instead of at :func:`body_band`'s top, which misses
    the first rows on 15 table scans. **Off by default** (0.3.0): switching it
    on changes the rows, and so the block crops, of every page where the two
    tops differ — see ``utils/lexikon_1910/_archive/measure_table_top.py``.
    """
    if template is not None:
        cols, _observed = fit_columns(page, template)
    else:
        cols = column_boundaries(page)
    if len(cols) != N_COLS + 1:
        raise ValueError(
            f"{page.stem}: found {len(cols) - 1} columns, expected {N_COLS}"
        )
    top, bottom = body_band(page)
    if foot_from_rules:
        bottom = table_bottom(page)
    if top_from_header:
        top = table_top(page)

    def column_of(hpos: int, width: int) -> int | None:
        centre = hpos + width // 2
        for i in range(N_COLS):
            if cols[i] <= centre < cols[i + 1]:
                return i
        return None

    # Cluster on the *vertical centre* of each word, not its top. Digits, capitals
    # and descenders in one printed line start at visibly different heights — tops
    # scatter by 40 px inside a single row, centres by under 10 — so clustering on
    # tops tears one row into two and scatters its cells across both.
    body = sorted((s for s in page.strings if top < s[0] < bottom),
                  key=lambda s: s[0] + s[3] // 2)
    lines: list[list[tuple]] = []
    for word in body:
        centre = word[0] + word[3] // 2
        if lines and centre - (lines[-1][-1][0] + lines[-1][-1][3] // 2) <= line_gap:
            lines[-1].append(word)
        else:
            lines.append([word])

    def to_row(line: list[tuple]) -> Row:
        grouped: dict[int, list[str]] = {}
        for _vpos, hpos, width, _h, text in sorted(line, key=lambda w: w[1]):
            idx = column_of(hpos, width)
            if idx is not None:
                grouped.setdefault(idx, []).append(text)
        return Row(
            page=page.stem,
            top=min(w[0] for w in line),
            bottom=max(w[0] + w[3] for w in line),
            cells={COL_NAMES[i]: " ".join(parts).strip()
                   for i, parts in sorted(grouped.items())},
        )

    def has_figures(row: Row) -> bool:
        return any(_STRIP_RE.sub("", row.get(c)) for c in INT_COLS)

    # Long names wrap onto a second printed line, and the figures may sit on either
    # half — ``také Bobkov, Doubkov, / O Dobkau 324 33 198`` carries the sequence
    # number on the first line and every number on the second. Join in whichever
    # direction the figures actually lie rather than assuming the wrap trails.
    candidates = [to_row(line) for line in lines]
    rows: list[Row] = []
    skip = False
    for i, row in enumerate(candidates):
        if skip:
            skip = False
            continue
        following = candidates[i + 1] if i + 1 < len(candidates) else None
        if not has_figures(row) and row.get("name"):
            if following is not None and not following.get("seq") and has_figures(following):
                merged = Row(page=row.page, top=row.top, bottom=following.bottom,
                             cells=dict(following.cells))
                merged.cells["name"] = f"{row.get('name')} {following.get('name')}".strip()

                    merged.cells["seq"] = row.get("seq")
                rows.append(merged)
                skip = True
                continue
            if rows and not row.get("seq"):
                previous = rows[-1]
                previous.cells["name"] = f"{previous.get('name')} {row.get('name')}".strip()
                previous.bottom = row.bottom
                continue
        rows.append(row)
    return rows

def grade_sequence(blocks: list[Block]) -> list[dict]:
    """The two interleaved integer sequences in column 1.

    Municipality numbers increase by one down the okres; settlement numbers restart
    at 1 inside each block and increase by one. This is a layout checksum that costs
    nothing and needs no character to be read correctly — where the OCR fails the
    expected value is still known, and a *spurious* block (a settlement row wrongly
    promoted because something landed in its area column) breaks the run
    immediately. It is therefore the check that measures segmentation, which no
    arithmetic identity can see.

    The first municipality on a page is a free anchor: page breaks fall inside an
    okres and okres breaks reset the count, and neither is legible from the table
    body alone.
    """
    findings: list[dict] = []
    printed = [parse_int_cell(b.municipality.get("seq")) for b in blocks]
    previous: int | None = None
    for i, block in enumerate(blocks):
        seq, status = printed[i]
        expected = None if previous is None else previous + 1
        following = printed[i + 1][0] if i + 1 < len(blocks) else None

        if expected is None:
            verdict, detail = "na", "sequence anchor"
        elif status != "ok":
            verdict, detail = "unparseable", f"printed={block.municipality.get('seq')!r}"
        elif seq == expected:
            verdict, detail = "pass", ""
        elif following is not None and following == seq + 1:
            # The printed run continues from here, just not from where we were:
            # an okres boundary resets the count, and okres headings are not
            # legible from the table body. Re-anchor rather than score a failure.
            verdict, detail = "na", f"re-anchored at {seq}"
        else:
            verdict = "fail"
            detail = f"printed={block.municipality.get('seq')!r} expected={expected}"

        findings.append({
            "block_id": block.block_id, "row_kind": "municipality",
            "row_name": block.name, "check": "I5_sequence",
            "verdict": verdict, "detail": detail,
        })
        # Carry the *run* forward, not a value we just judged wrong — otherwise one
        # misread digit scores twice, once where it is read and once on the next
        # block that disagrees with it.
        if verdict in ("pass",) or detail.startswith("re-anchored"):
            previous = seq
        elif expected is not None:
            previous = expected
        elif status == "ok":
            previous = seq

        for n, row in enumerate(block.settlements, start=1):
            value, status = parse_int_cell(row.get("seq"))
            if status != "ok":
                verdict = "unparseable"
            else:
                verdict = "pass" if value == n else "fail"
            findings.append({
                "block_id": block.block_id, "row_kind": "settlement",
                "row_name": row.get("name"), "check": "I5_sequence",
                "verdict": verdict,
                "detail": "" if verdict == "pass"
                          else f"printed={row.get('seq')!r} expected={n}",
            })
    return findings

               target_width: int = VISION_LONG_EDGE,

    :func:`crop_block` plus a controlled resize.

    which row is which, and the tiles could only be realigned by position — an
    off-by-one would silently shift a whole block's figures onto the wrong
    settlements. Repeating the name costs a few hundred pixels and makes each tile
    independently readable.

    top = max(0, block.top - pad_y)
    bottom = min(image.height, block.bottom + pad_y)

                           min(image.width, right + pad_x), bottom))

        canvas.paste(body, (label.width, 0))

    return out

def lexikon_1910_report(records: list[dict]) -> str:
    """Human-readable pass/fail summary over a list of grading records."""
    from collections import Counter

    by_check: dict[str, Counter] = {}
    for rec in records:
        by_check.setdefault(rec["check"].split(":")[0], Counter())[rec["verdict"]] += 1

    lines = [f"lexikon_1910 v{__version__} — invariant report", ""]
    header = f"{'check':<16}{'pass':>7}{'fail':>7}{'unparse':>9}{'n/a':>7}{'rate':>9}"

    grand = Counter()
    for check, counter in sorted(by_check.items()):
        grand += counter
        graded = counter["pass"] + counter["fail"] + counter["unparseable"]
        rate = f"{counter['pass'] / graded:6.1%}" if graded else "     --"

            f"{check:<16}{counter['pass']:>7}{counter['fail']:>7}"
            f"{counter['unparseable']:>9}{counter['na']:>7}{rate:>9}"

"""Podrobný seznam míst / Spezialortsrepertorium 1910 — grid, blocks, invariants.

The 1910 special place register is a 17-column table printed one *politický okres*
at a time: a bold municipality row carrying the area in hectares, followed by its
osady (settlements), which carry no area. This module reconstructs that table from
the NDK ALTO layer of the Czech volume and grades the result against the arithmetic
the table itself must satisfy.

Three facts about the source do the structural work, and none of them depend on
reading a character correctly:

1. **The printed rules are in the ALTO.** ``GraphicalElement`` entries taller than
   they are wide are the column rules; the tall ones bound the table body. Column
   boundaries are therefore measured, not inferred from a text histogram.
2. **The area column is printed only on municipality rows.** That is the block
   delimiter — typography is not usable here, because the NDK ``TextStyle`` entries
   are synthetic (Times New Roman, sizes 5–10) and carry no weight.
3. **Column 1 runs two interleaved integer sequences** — municipalities incrementing
   across the okres, osady restarting at 1 inside each. The OCR mangles them
   (``o``, ``í)``, ``Г) o``), but the sequence is known a priori, so row identity is
   reconstructed rather than read.

The invariants in :data:`INVARIANTS` are the grader. They are deliberately kept out
of the transcription prompt (see ``run_lexikon_1910_pilot.py``): a model told what
the numbers must add up to will produce numbers that add up.

Column labels are **provisional** — the groupings the invariants use are certain
(they are what the rules enclose), but the label on columns 9/10 in particular is
read off the header and awaits confirmation against the preface, pp. VIII–XIV.
"""

from __future__ import annotations

import re
from dataclasses import dataclass, field
from pathlib import Path

__version__ = "0.5.0"

# --------------------------------------------------------------------------- #
# Column schema (provisional — confirm against the preface)
# --------------------------------------------------------------------------- #

#: ``(name, kind)`` in printed order, left to right. 17 columns, 18 rule positions.
COLUMNS: list[tuple[str, str]] = [
    ("seq",              "text"),  # Pořadné číslo
    ("name",             "text"),  # okres / soudní okres / obec / osada  (+ symbols)
    ("area_ha",          "int"),   # Plocha v hektarech — municipality rows only
    ("houses",           "int"),   # Domy
    ("pop_total",        "int"),   # Přítomné obyvatelstvo: dohromady
    ("pop_male",         "int"),   #   mužské
    ("pop_female",       "int"),   #   ženské
    ("conf_catholic",    "int"),   # katolické
    ("conf_evangelical", "int"),   # evangelické
    ("conf_israelite",   "int"),   # (provisional)
    ("conf_other",       "int"),   # (provisional)
    ("lang_german",      "int"),   # státní příslušníci dle obcovací řeči: německé
    ("lang_czech",       "int"),   #   česko-, morav.-, slovácké
    ("lang_other",       "int"),   #   jiné
    ("foreigners",       "int"),   # cizozemci
    ("post_key",         "text"),  # doručovací poštovní obvod (footnote key)
    ("station_note",     "text"),  # nejbližší stanice + vzdálenost v pochod. hodinách
]

N_COLS = len(COLUMNS)
COL_NAMES = [c for c, _ in COLUMNS]
COL_INDEX = {c: i for i, (c, _) in enumerate(COLUMNS)}
INT_COLS = [c for c, k in COLUMNS if k == "int"]

#: Columns a settlement row contributes to its municipality's total. ``area_ha`` is
#: excluded because it is printed on the municipality row only — that is the block
#: delimiter, not an additive quantity.
BLOCK_SUM_COLS = [c for c in INT_COLS if c != "area_ha"]

#: Rows that are *of which* breakdowns, not additive members. ``mezi tím vojsko``
#: ("of that, military") sits inside a municipality block and would break every
#: block sum if counted.
OF_WHICH_RE = re.compile(r"mezi\s*t[ií]m|darunter", re.IGNORECASE)

#: Aggregate rows printed at the foot of an okres or soudní okres.
TOTAL_ROW_RE = re.compile(r"\b(úhrnem|uhrnem|summe|zusammen|celkem)\b", re.IGNORECASE)

#: The register opens each district with a **summary row** carrying the aggregate
#: for everything that follows — ``P. o. Krumlov`` for the politický okres,
#: ``I. s. o. Kralupy nad Vltavou`` for the soudní okres. These rows have the area
#: column filled, so :attr:`Row.is_municipality` is true of them and they were
#: being segmented as municipalities: an okres's own total added back into the
#: okres. Summing okres Kraslice with its three summary rows left in gave roughly
#: four times the true population.
#:
#: They are headers, not footers — the summary precedes the municipalities it
#: covers — which is what makes them usable as district delimiters in the row
#: stream. ``TOTAL_ROW_RE`` does not catch them: it looks for *úhrnem* / *summe*,
#: and these carry the district abbreviation instead.
#:
#: The ``o`` is routinely OCR'd as a zero (``P. 0. Kralupy``) and the spacing is
#: unreliable (``s.o.`` as often as ``s. o.``), so both are matched loosely. The
#: spelled-out forms appear too, in the okres that is also its own court district:
#: ``Politický okres zároveň s. o. Kraslice``.
#:
#: The word *okres* is itself OCR'd wrong often enough to matter — ``II. soudní
#: oleres Tannwald`` was being read as a municipality — so it is matched as
#: "o, a few letters, s" rather than literally. The bound keeps that from
#: swallowing an ordinary word: it has to start with o, end with s, and be no
#: longer than *okres* plus a letter or two.
SUMMARY_OKRES_RE = re.compile(
    r"politick[ýy]\s+o\w{2,5}s|\bP\s*\.\s*[oO0]\s*\.", re.IGNORECASE)
SUMMARY_COURT_RE = re.compile(
    r"soudn[íi]\s+o\w{2,5}s|\bs\s*\.\s*[oO0]\s*\.", re.IGNORECASE)

#: Leading enumeration on a summary row — ``IV.``, ``7 O. I.`` — stripped before
#: the district name. The **period is mandatory**: without it the pattern eats the
#: leading letter of any name that starts with a roman numeral, turning Vršovice
#: into "ršovice" and Ivančice into "ančice".
_ENUM_PREFIX_RE = re.compile(r"^(?:\s*(?:\d+|[IVXLO]+)\s*\.\s*)+", re.IGNORECASE)

#: How close a second district marker must be to the first to count as part of the
#: same header rather than as footnote noise following the name. See
#: :func:`summary_label`.
_ADJACENT_MARKER = 12


def summary_label(name: str) -> tuple[str, str]:
    """``(kind, district)`` for a summary row, or ``("", "")`` for anything else.

    ``kind`` is ``"okres"`` or ``"court"``. An okres that is simultaneously its own
    court district matches both patterns and is reported as ``"okres"`` — the wider
    unit wins, and the caller sets the court to the same name.

    The name is taken from the end of the header, and the header can be one marker
    or two. Two adjacent markers are one compound header — ``Politický okres
    zároveň s. o. Kraslice`` names an okres that is also its own court — so the cut
    goes after the second. But a marker far downstream is *not* part of the header:
    it is the footnote soup printed after the name, as in ``P. 0. Kralupy nad
    Vltavou, Kralup an der Moldau *) . 2 s. 0.. 42 Ф``. Cutting at the furthest
    marker swallowed that name whole and returned nothing, so distance decides.
    """
    text = str(name or "")
    okres = SUMMARY_OKRES_RE.search(text)
    court = SUMMARY_COURT_RE.search(text)
    if not (okres or court):
        return "", ""
    marks = sorted((m for m in (okres, court) if m), key=lambda m: m.start())
    end = marks[0].end()
    for mark in marks[1:]:
        if mark.start() - end > _ADJACENT_MARKER:
            break
        end = mark.end()
    label = _ENUM_PREFIX_RE.sub("", text[end:].split(",")[0])
    # Keep the leading run of words and stop at the first symbol: the register
    # prints the Czech and German names together and the OCR fills the gap with
    # glyph noise ("Jirkov^ Grörkau.. .. 28 ©"). Multi-word names survive whole,
    # which matters for "Kralupy nad Vltavou" and "Jablonec nad Nisou".
    words = re.match(r"[^\W\d_]+(?:[\s\-][^\W\d_]+)*", label.strip(" ."))
    return ("okres" if okres else "court"), (words.group(0) if words else "")

_DIGITS_RE = re.compile(r"^\d+$")
_STRIP_RE = re.compile(r"[\s.,'’`]+")

#: A cell holding only a dash. The register prints one where the figure is nil, so
#: this is a stated zero — see :func:`parse_int_cell`. Every dash the OCR emits is
#: included, hyphen through horizontal bar and minus sign.
_DASH_RE = re.compile(r"^[\s\-‐-―−]+$")


def parse_int_cell(raw: str) -> tuple[int | None, str]:
    """Parse a numeric cell strictly.

    Returns ``(value, status)`` where status is ``"blank"``, ``"ok"`` or ``"bad"``.
    Anything that is not pure digits after removing separators is ``"bad"`` — the
    OCR's ``23S``, ``íttiS``, ``58-4`` and ``07`` must surface as unreadable, not
    be coerced into a number.

    **A printed dash is a zero, not an absence**, and is returned as ``(0, "ok")``.
    The register sets it where a figure really is none, so reading it as unreadable
    threw away a value the press had stated. It is rare in this table — the nil
    marker here is usually a centred dot, and the OCR usually drops that entirely —
    but where the dash survives it is the strongest evidence available that the
    cell is zero rather than missing.

    An *empty* cell still reports ``"blank"``. Blank also means zero in this table,
    but it is not the same evidence: it is equally the trace of a nil marker the
    scanner lost and of a cell the parser never filled, so counting it as a read
    zero would let a row of nothing satisfy an identity trivially.
    """
    if raw is None:
        return None, "blank"
    text = str(raw).strip()
    if text and _DASH_RE.match(text):
        return 0, "ok"
    s = _STRIP_RE.sub("", str(raw))
    if not s:
        return None, "blank"
    return (int(s), "ok") if _DIGITS_RE.match(s) else (None, "bad")

@dataclass
class AltoPage:
    """One ALTO page: its frame, its printed rules, and its words."""

    path: Path
    width: int
    height: int
    strings: list[tuple[int, int, int, int, str]]  # vpos, hpos, width, height, text
    verticals: list[dict]
    horizontals: list[dict]
    print_left: int = 0
    print_width: int = 0

    @property
    def stem(self) -> str:
        return self.path.name.split(".")[0]


def read_alto(path: str | Path) -> AltoPage:
    """Parse an NDK ALTO file into words and printed rules."""
    from lxml import etree

    path = Path(path)
    root = etree.parse(str(path)).getroot()
    ns = root.nsmap.get(None)

    def each(tag: str):
        return root.iter(f"{{{ns}}}{tag}" if ns else tag)

    _KEYS = ("HEIGHT", "WIDTH", "VPOS", "HPOS")

    def box(el) -> dict | None:
        """Geometry dict, or None when the engine emitted an element without one."""
        values = [el.get(k) for k in _KEYS]
        if any(v is None for v in values):
            return None
        return dict(zip(_KEYS, (int(v) for v in values)))

    page = next(each("Page"))
    graphics = [b for b in (box(e) for e in each("GraphicalElement")) if b]
    strings = [
        (b["VPOS"], b["HPOS"], b["WIDTH"], b["HEIGHT"], e.get("CONTENT") or "")
        for e, b in ((e, box(e)) for e in each("String")) if b
    ]
    print_space = next((box(e) for e in each("PrintSpace")), None)
    page_width = int(page.get("WIDTH"))
    return AltoPage(
        path=path,
        width=page_width,
        height=int(page.get("HEIGHT")),
        strings=strings,
        verticals=[g for g in graphics if g["HEIGHT"] > g["WIDTH"] * 3],
        horizontals=[g for g in graphics if g["WIDTH"] > g["HEIGHT"] * 3],
        print_left=print_space["HPOS"] if print_space else 0,
        print_width=print_space["WIDTH"] if print_space else page_width,
    )


#: A vertical rule taller than this share of the page is the printed *frame*, which
#: runs past the table into the footnote apparatus below it. Its x positions are the
#: table's outer edges, but its vertical extent is not the body band.
_FRAME_SHARE = 0.70

#: Body rules are kept relative to the tallest non-frame rule rather than against a
#: fixed pixel height: scan height, trim and rule length all vary page to page, and
#: a fixed threshold silently drops whole pages.
_BODY_SHARE = 0.80


def _rule_sets(page: AltoPage) -> tuple[list[dict], list[dict]]:
    """Split the vertical rules into ``(frame, body)``."""
    frame = [v for v in page.verticals if v["HEIGHT"] >= _FRAME_SHARE * page.height]
    inner = [v for v in page.verticals if v not in frame]
    if not inner:
        raise ValueError(f"{page.stem}: no column rules — not a table page")
    tallest = max(v["HEIGHT"] for v in inner)
    body = [v for v in inner if v["HEIGHT"] >= _BODY_SHARE * tallest]
    return frame, body


def column_boundaries(page: AltoPage, merge_within: int = 30) -> list[int]:
    """x positions of the table's column rules, left to right.

    The frame contributes the outer two boundaries; the body rules contribute the
    interior ones. The engine often prints one rule as an adjacent pair, so
    positions closer than ``merge_within`` pixels are collapsed.
    """
    frame, body = _rule_sets(page)
    centres = sorted(v["HPOS"] + v["WIDTH"] // 2 for v in frame + body)
    cols: list[int] = []
    for centre in centres:
        if cols and centre - cols[-1] < merge_within:
            cols[-1] = (cols[-1] + centre) // 2
        else:
            cols.append(centre)
    return cols


# --------------------------------------------------------------------------- #
# The grid template
# --------------------------------------------------------------------------- #
#
# The OCR engine detects the printed rules unreliably — it finds all 18 on only a
# fifth of pages and drops one or more on the rest. But the grid is a property of
# the *book*, not of the page: one press, one typesetting, and normalised against
# the ALTO ``PrintSpace`` the 18 boundaries land within ~1% of table width of each
# other across the volume. So the template is measured once from the pages where
# the rules are unambiguous, then fitted to every page. Pages keep a count of how
# many boundaries were actually observed rather than interpolated — that number
# goes in the ledger, because an interpolated grid is weaker evidence than a
# printed one even when it happens to be right.

def column_template(alto_paths, expected: int = N_COLS + 1) -> tuple[list[float], int]:
    """Median normalised boundary positions over the pages with a complete rule set.

    Returns ``(template, n_pages)`` with positions expressed as a fraction of
    ``PrintSpace`` width.
    """
    import statistics

    observed: list[list[float]] = []
    for path in alto_paths:
        try:
            page = read_alto(path)
            cols = column_boundaries(page)
        except Exception:
            continue
        if len(cols) == expected and page.print_width:
            observed.append([(c - page.print_left) / page.print_width for c in cols])
    if len(observed) < 20:
        raise ValueError(f"only {len(observed)} clean pages — too few to fit a grid")
    template = [statistics.median(row[i] for row in observed) for i in range(expected)]
    return template, len(observed)


def fit_columns(page: AltoPage, template: list[float],
                snap_share: float = 0.35) -> tuple[list[int], int]:
    """Fit ``template`` to one page, snapping to the rules the engine did find.

    Each predicted boundary is pulled onto the nearest observed rule within
    ``snap_share`` of the narrowest column; boundaries with no rule nearby keep the
    predicted position. A median residual correction is applied first so a page
    whose scan is shifted relative to its ``PrintSpace`` still lands correctly.

    Returns ``(boundaries, n_observed)``.
    """
    predicted = [page.print_left + f * page.print_width for f in template]
    # Snap against *every* vertical rule, not just the full-height ones: the engine
    # frequently reports a column rule broken into shorter segments, and a segment
    # is still evidence of where that column is.
    tallest = max((v["HEIGHT"] for v in page.verticals), default=0)
    rules = sorted(v["HPOS"] + v["WIDTH"] // 2 for v in page.verticals
                   if v["HEIGHT"] >= 0.15 * tallest)
    if not rules:
        return [round(p) for p in predicted], 0

    tolerance = snap_share * min(b - a for a, b in zip(predicted, predicted[1:]))

    def nearest(x: float) -> int | None:
        best = min(rules, key=lambda r: abs(r - x))
        return best if abs(best - x) <= tolerance else None

    residuals = [n - p for p in predicted if (n := nearest(p)) is not None]
    if residuals:
        import statistics
        shift = statistics.median(residuals)
        predicted = [p + shift for p in predicted]

    boundaries, observed = [], 0
    for p in predicted:
        match = nearest(p)
        if match is None:
            boundaries.append(round(p))
        else:
            boundaries.append(match)
            observed += 1
    return boundaries, observed


def write_grid_template(path: str | Path, template: list[float], volume: str,
                        n_pages: int) -> Path:
    """Write the fitted grid to a tracked CSV under ``reference/places/``."""
    import csv

    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", newline="", encoding="utf-8") as fh:
        writer = csv.writer(fh)
        writer.writerow(["volume", "boundary", "position", "left_column",
                         "n_pages_fitted"])
        for i, position in enumerate(template):
            left = COL_NAMES[i] if i < N_COLS else ""
            writer.writerow([volume, i, f"{position:.6f}", left, n_pages])
    return path


def read_grid_template(path: str | Path, volume: str) -> list[float]:
    """Read a grid template written by :func:`write_grid_template`."""
    import csv

    with Path(path).open(encoding="utf-8") as fh:
        rows = [r for r in csv.DictReader(fh) if r["volume"] == volume]
    if not rows:
        raise ValueError(f"no grid template for volume {volume!r} in {path}")
    return [float(r["position"]) for r in sorted(rows, key=lambda r: int(r["boundary"]))]


def body_band(page: AltoPage) -> tuple[int, int]:
    """``(top, bottom)`` of the table body in page pixels.

    The band is where *every* column rule is present: it starts below the deepest
    header sub-division and ends where the shortest body rule stops, above the
    footnote block. Taking the intersection rather than the union keeps the
    footnote apparatus and the running head out of the row clustering.
    """
    _frame, body = _rule_sets(page)
    tops = sorted(v["VPOS"] for v in body)
    bottoms = sorted(v["VPOS"] + v["HEIGHT"] for v in body)
    # Percentiles rather than max/min: one over-long or over-short rule should not
    # swallow a fifth of the page's rows.
    top = tops[int(0.85 * (len(tops) - 1))]
    bottom = bottoms[int(0.15 * (len(bottoms) - 1))]
    wide = [h["VPOS"] + h["HEIGHT"] for h in page.horizontals
            if h["WIDTH"] > page.width * 0.5 and h["VPOS"] < bottom * 0.4]
    if wide:
        top = max(top, max(wide))
    if bottom - top < page.height * 0.2:
        raise ValueError(f"{page.stem}: implausible body band {top}–{bottom}")
    return top + 5, bottom - 5


def table_bottom(page: AltoPage) -> int:
    """Where the table's rows end: where the footnote apparatus opens.

    :func:`body_band` is not that line, in either direction (audit of
    2026-10-14). Its 15th-percentile foot sits above the lowest column rules on
    227 of 454 table scans, so the last one to three rows were never read — scan
    0196 lost Hrnčíře and Šeberov, 0023 Tisem, 0083 Nové Záboří. And on the 22
    scans whose divider only ``lexikon_1910_notes.fallback_divider`` finds, it
    runs on into the apparatus (to 4,007 px on scan 0026 against a divider at
    3,375), so apparatus lines were read as table rows.

    The bottom is therefore the apparatus band's top,
    ``lexikon_1910_notes.apparatus_band`` — the lowest foot among the column rules
    that start well up in the table and end above the divider — so a word above
    the divider is a table word or an apparatus word, never both and never
    neither. The protection the percentile gave is kept by other means: a column
    rule that runs on past the table is one that ends below the divider's top, and
    ``table_foot`` does not count it. A page with no divider keeps
    :func:`body_band`'s foot, since there is nothing to end the table against.
    """
    from utils.lexikon_1910 import lexikon_1910_notes as lxnt

    _top, bottom = body_band(page)
    found = lxnt.apparatus_band(page)
    return bottom if found is None else found[1]


#: The header's foot is looked for in this top share of the page. The column
#: header closes 520–640 px down a 4,290 px scan; nothing frame-wide is printed
#: between it and this line except the table's own top rule.
_HEADER_ZONE = 0.30

#: Above this share of the page a frame-wide "rule" is the scan's edge.
_SHEET_EDGE = 0.03

#: Frame-wide rules closer than this are one printed (double) rule.
_DOUBLE_RULE = 20

#: The header is at least this tall, so a lone top rule is never taken for its
#: foot (the header runs ~300 px from the top rule to its foot).
_HEADER_MIN = 200


def _header_rules(page: AltoPage) -> tuple[int, int] | None:
    """``(top of the table's top rule, foot of the header's closing rule)``.

    A "rule" in the top ``_SHEET_EDGE`` of the page is the edge of the scan
    (scan 0396 reports one at y = 0); the page number and running head are
    printed above the table's top rule, so it never stands there.
    """
    wide = sorted((h["VPOS"], h["VPOS"] + h["HEIGHT"]) for h in page.horizontals
                  if h["WIDTH"] > page.width * 0.5
                  and _SHEET_EDGE * page.height < h["VPOS"] < _HEADER_ZONE * page.height)
    groups: list[list[int]] = []
    for start, end in wide:
        if groups and start - groups[-1][1] <= _DOUBLE_RULE:
            groups[-1][1] = max(groups[-1][1], end)
        else:
            groups.append([start, end])
    if len(groups) < 2 or groups[-1][0] - groups[0][0] < _HEADER_MIN:
        return None
    return groups[0][0], groups[-1][1]


def header_foot(page: AltoPage) -> int | None:
    """The foot of the frame-wide (double) rule that closes the column header.

    The table opens with two frame-wide rules: its top, and the double rule under
    the column header. The second is the last frame-wide rule in the top
    ``_HEADER_ZONE`` of the page, at least ``_HEADER_MIN`` below the first.
    None where the page has no such pair (a prose page; scan 0108, whose header
    rule the engine did not report).

    Checked against the header's foot read off the image alone
    (``lexikon_1910_image``) on the 477 table scans where both exist: within
    15 px on 476, 16 px on the last (2026-10-14).
    """
    found = _header_rules(page)
    return None if found is None else found[1]


def table_top(page: AltoPage) -> int:
    """Where the table's rows start: just under the header's closing rule.

    :func:`body_band` takes the top from the 85th percentile of the body rules'
    tops, and the body rules are those at least 80 % as tall as the tallest
    non-frame rule. On 15 table scans one over-tall rule leaves exactly one rule
    in that set, starting mid-table, and the band opens 410–1,040 px under the
    header: the table's first rows were never read (scan 0399: Dejvice,
    Dobrovíz, Dolní Liboc…; audit of 2026-10-14). The header's own rule says
    where the table starts. A page with no header rule keeps :func:`body_band`'s
    top; so does one where that top is the higher of the two, which a header
    rule never is.
    """
    top, _bottom = body_band(page)
    foot = header_foot(page)
    return top if foot is None else min(top, foot + 5)

@dataclass
class Row:
    """One printed line, its cells assigned to columns."""

    page: str
    top: int
    bottom: int
    cells: dict[str, str] = field(default_factory=dict)
    continuation: bool = False   # a wrapped name line, merged into the row above

    def get(self, col: str) -> str:
        return self.cells.get(col, "")

    @property
    def is_municipality(self) -> bool:
        return bool(_STRIP_RE.sub("", self.get("area_ha")))

    @property
    def is_of_which(self) -> bool:
        return bool(OF_WHICH_RE.search(self.get("name")))

    @property
    def is_total(self) -> bool:
        return bool(TOTAL_ROW_RE.search(self.get("name")))

    @property
    def is_summary(self) -> bool:
        """A district aggregate — never a municipality, anywhere in the pipeline.

        Two guards, both load-bearing. A summary row carries an **area**, like the
        municipalities it aggregates. And a ``mezi tím vojsko`` row never is one,
        however much it looks like it: those carry the footnote symbol soup, which
        routinely contains a literal ``s. o.`` among the glyph noise
        (``mezi tím vojsko 2 s. o., 78 ®, 139 €``). Without the guards those four
        rows on twenty pages were classified as district headers and dropped —
        deleting real data, which is worse than the double-count this fixes.
        """
        if self.is_of_which or not self.is_municipality:
            return False
        return bool(summary_label(self.get("name"))[0])


def extract_rows(page: AltoPage, template: list[float] | None = None,
                 line_gap: int = 15, foot_from_rules: bool = True,
                 top_from_header: bool = True) -> list[Row]:
    """Assign every body word to a column, then cluster words into printed lines.

    ``template`` is the fitted grid from :func:`column_template`; without it the
    page's own rules must already resolve to exactly 18 boundaries, which they do
    on about a fifth of pages. ``line_gap`` is the vertical distance at which a new
    baseline starts. Wrapped name lines — a long municipality name spilling onto a
    second line with no numbers beside it — are folded into the preceding row.

    The body ends at :func:`table_bottom`, where the apparatus opens.
    ``foot_from_rules=False`` ends it at :func:`body_band`'s foot, as 0.1.0 did,
    for replaying a pass cached under it: that loses the last rows on half the
    table scans and reads apparatus lines as rows on 22.

    ``top_from_header=True`` (the default since 0.4.0) opens the body at :func:`table_top`, under the
    header's closing rule, instead of at :func:`body_band`'s top, which misses
    the first rows on 15 table scans. **Off by default** (0.3.0): switching it
    on changes the rows, and so the block crops, of every page where the two
    tops differ — see ``utils/lexikon_1910/_archive/measure_table_top.py``.
    """
    if template is not None:
        cols, _observed = fit_columns(page, template)
    else:
        cols = column_boundaries(page)
    if len(cols) != N_COLS + 1:
        raise ValueError(
            f"{page.stem}: found {len(cols) - 1} columns, expected {N_COLS}"
        )
    top, bottom = body_band(page)
    if foot_from_rules:
        bottom = table_bottom(page)
    if top_from_header:
        top = table_top(page)

    def column_of(hpos: int, width: int) -> int | None:
        centre = hpos + width // 2
        for i in range(N_COLS):
            if cols[i] <= centre < cols[i + 1]:
                return i
        return None

    # Cluster on the *vertical centre* of each word, not its top. Digits, capitals
    # and descenders in one printed line start at visibly different heights — tops
    # scatter by 40 px inside a single row, centres by under 10 — so clustering on
    # tops tears one row into two and scatters its cells across both.
    body = sorted((s for s in page.strings if top < s[0] < bottom),
                  key=lambda s: s[0] + s[3] // 2)
    lines: list[list[tuple]] = []
    for word in body:
        centre = word[0] + word[3] // 2
        if lines and centre - (lines[-1][-1][0] + lines[-1][-1][3] // 2) <= line_gap:
            lines[-1].append(word)
        else:
            lines.append([word])

    def to_row(line: list[tuple]) -> Row:
        grouped: dict[int, list[str]] = {}
        for _vpos, hpos, width, _h, text in sorted(line, key=lambda w: w[1]):
            idx = column_of(hpos, width)
            if idx is not None:
                grouped.setdefault(idx, []).append(text)
        return Row(
            page=page.stem,
            top=min(w[0] for w in line),
            bottom=max(w[0] + w[3] for w in line),
            cells={COL_NAMES[i]: " ".join(parts).strip()
                   for i, parts in sorted(grouped.items())},
        )

    def has_figures(row: Row) -> bool:
        return any(_STRIP_RE.sub("", row.get(c)) for c in INT_COLS)

    # Long names wrap onto a second printed line, and the figures may sit on either
    # half — ``také Bobkov, Doubkov, / O Dobkau 324 33 198`` carries the sequence
    # number on the first line and every number on the second. Join in whichever
    # direction the figures actually lie rather than assuming the wrap trails.
    candidates = [to_row(line) for line in lines]
    rows: list[Row] = []
    skip = False
    for i, row in enumerate(candidates):
        if skip:
            skip = False
            continue
        following = candidates[i + 1] if i + 1 < len(candidates) else None
        if not has_figures(row) and row.get("name"):
            if following is not None and not following.get("seq") and has_figures(following):
                merged = Row(page=row.page, top=row.top, bottom=following.bottom,
                             cells=dict(following.cells))
                merged.cells["name"] = f"{row.get('name')} {following.get('name')}".strip()
                if row.get("seq"):
                    merged.cells["seq"] = row.get("seq")
                rows.append(merged)
                skip = True
                continue
            if rows and not row.get("seq"):
                previous = rows[-1]
                previous.cells["name"] = f"{previous.get('name')} {row.get('name')}".strip()
                previous.bottom = row.bottom
                continue
        rows.append(row)
    return rows

@dataclass
class Block:
    """A municipality row plus the settlement rows printed beneath it.

    This is the unit of work: the smallest object that carries its own checksum,
    and small enough to crop into one image.
    """

    page: str
    index: int
    municipality: Row
    settlements: list[Row] = field(default_factory=list)
    of_which: list[Row] = field(default_factory=list)
    #: The district this block was printed under, read from the last summary row
    #: above it. A page is not a district — the register runs straight on, so a
    #: page routinely carries the end of one okres and the start of the next — and
    #: assigning blocks by page put a neighbouring okres's municipalities into
    #: Kraslice's totals. These labels come from the row stream itself, where the
    #: boundary actually is.
    okres_label: str = ""
    court_label: str = ""

    @property
    def rows(self) -> list[Row]:
        return [self.municipality, *self.settlements]

    @property
    def top(self) -> int:
        return min(r.top for r in self.rows + self.of_which)

    @property
    def bottom(self) -> int:
        return max(r.bottom for r in self.rows + self.of_which)

    @property
    def name(self) -> str:
        return self.municipality.get("name")

    @property
    def block_id(self) -> str:
        return f"{self.page}#{self.index:02d}"


def segment_blocks(rows: list[Row], okres_label: str = "",
                   court_label: str = "") -> tuple[list[Block], list[Row]]:
    """Split a page's rows into municipality blocks.

    Returns ``(blocks, orphans)``. Orphans are settlement rows appearing before the
    first municipality row on the page — the tail of a block that began on the
    previous page. They are excluded from block-level grading rather than graded
    against a municipality that is not there.

    **District summary rows are not municipalities** and are dropped here rather
    than downstream. They carry the area column, so without this they segment as
    municipalities and add a whole district back into itself. Each one instead sets
    the label carried by the blocks that follow it, which is what gives block-level
    district assignment — a district boundary falls mid-page, so a page-level
    assignment is wrong by construction at every boundary.

    ``okres_label`` and ``court_label`` carry the district in from the previous
    page, since a page that opens mid-district has no summary row of its own. Read
    them back off the last block to carry forward.
    """
    blocks: list[Block] = []
    orphans: list[Row] = []
    for row in rows:
        # Classified before anything else: an "of which" row's symbol soup can
        # contain a literal "s. o." and would otherwise read as a district header,
        # dropping a real row.
        if row.is_of_which:
            (blocks[-1].of_which if blocks else orphans).append(row)
            continue
        kind, label = (summary_label(row.get("name")) if row.is_summary
                       else ("", ""))
        if kind == "okres":
            # An okres that is also its own court district prints one row for
            # both, so the court is set here too rather than left stale from the
            # previous okres.
            okres_label, court_label = label, label
            continue
        if kind == "court":
            court_label = label
            continue
        if row.is_total:
            continue
        if row.is_municipality:
            blocks.append(Block(page=row.page, index=len(blocks) + 1,
                                municipality=row, okres_label=okres_label,
                                court_label=court_label))
        elif not blocks:
            orphans.append(row)
        else:
            blocks[-1].settlements.append(row)
    return blocks, orphans


def carry_block(block: Block, orphans: list[Row]) -> Block:
    """``block`` with the next page's orphan rows appended — one block again.

    A municipality printed at the foot of a page has the rest of its settlements
    at the head of the next one, where :func:`segment_blocks` returns them as
    orphans. Graded one page at a time the block reads short in every column it
    sums (scan 0124's Holetín; Trutnov on the gold sheets 0450-0451). The rows
    are not moved or changed: a new :class:`Block` is returned for grading, and
    an ``of which`` row among the orphans stays an ``of which`` row.
    """
    from dataclasses import replace

    return replace(block,
                   settlements=block.settlements + [r for r in orphans
                                                    if not r.is_of_which],
                   of_which=block.of_which + [r for r in orphans if r.is_of_which])


def summary_rows(rows: list[Row]) -> list[tuple[str, str, Row]]:
    """``(kind, district, row)`` for every district summary row on a page.

    Kept accessible rather than merely skipped: these rows are the register's own
    district totals, printed in the volume. Okres Kraslice's summary reads 39,216
    — exactly what ``census_okres_1910`` reports for it — so they are a totals
    target that needs no external source at all, and one that also works for the
    Morava and Slezsko volumes.
    """
    found: list[tuple[str, str, Row]] = []
    for row in rows:
        if not row.is_summary:
            continue
        kind, label = summary_label(row.get("name"))
        found.append((kind, label, row))
    return found

def grade_row(row: Row, invariant: Invariant) -> tuple[str, str]:
    """Apply one within-row identity. Returns ``(verdict, detail)``."""
    whole, whole_status = parse_int_cell(row.get(invariant.whole))
    if whole_status == "bad":
        return "unparseable", f"{invariant.whole}={row.get(invariant.whole)!r}"
    if whole_status == "blank":
        return "na", "total blank"

    total, bad, seen = 0, [], False
    for col in invariant.parts:
        value, status = parse_int_cell(row.get(col))
        if status == "bad":
            bad.append(f"{col}={row.get(col)!r}")
        elif status == "ok":
            total += value
            seen = True
    if bad:
        return "unparseable", "; ".join(bad)
    if not seen:
        return "na", "all parts blank"
    if total == whole:
        return "pass", ""
    return "fail", f"parts={total} whole={whole}"


def grade_block_sum(block: Block) -> dict[str, tuple[str, str]]:
    """Settlements sum to their municipality, column by column.

    Only applied when the block actually has settlement rows; a single-settlement
    municipality prints no breakdown and has nothing to check.
    """
    if not block.settlements:
        return {col: ("na", "no settlement rows") for col in BLOCK_SUM_COLS}

    out: dict[str, tuple[str, str]] = {}
    for col in BLOCK_SUM_COLS:
        whole, whole_status = parse_int_cell(block.municipality.get(col))
        if whole_status == "bad":
            out[col] = ("unparseable", f"municipality={block.municipality.get(col)!r}")
            continue
        if whole_status == "blank":
            out[col] = ("na", "municipality blank")
            continue

        total, bad, seen = 0, [], False
        for row in block.settlements:
            value, status = parse_int_cell(row.get(col))
            if status == "bad":
                bad.append(f"{row.get('name')[:18]!r}={row.get(col)!r}")
            elif status == "ok":
                total += value
                seen = True
        if bad:
            out[col] = ("unparseable", "; ".join(bad))
        elif not seen:
            out[col] = ("na", "all settlements blank")
        elif total == whole:
            out[col] = ("pass", "")
        else:
            out[col] = ("fail", f"settlements={total} municipality={whole}")
    return out


def grade_sequence(blocks: list[Block]) -> list[dict]:
    """The two interleaved integer sequences in column 1.

    Municipality numbers increase by one down the okres; settlement numbers restart
    at 1 inside each block and increase by one. This is a layout checksum that costs
    nothing and needs no character to be read correctly — where the OCR fails the
    expected value is still known, and a *spurious* block (a settlement row wrongly
    promoted because something landed in its area column) breaks the run
    immediately. It is therefore the check that measures segmentation, which no
    arithmetic identity can see.

    The first municipality on a page is a free anchor: page breaks fall inside an
    okres and okres breaks reset the count, and neither is legible from the table
    body alone.
    """
    findings: list[dict] = []
    printed = [parse_int_cell(b.municipality.get("seq")) for b in blocks]
    previous: int | None = None
    for i, block in enumerate(blocks):
        seq, status = printed[i]
        expected = None if previous is None else previous + 1
        following = printed[i + 1][0] if i + 1 < len(blocks) else None

        if expected is None:
            verdict, detail = "na", "sequence anchor"
        elif status != "ok":
            verdict, detail = "unparseable", f"printed={block.municipality.get('seq')!r}"
        elif seq == expected:
            verdict, detail = "pass", ""
        elif following is not None and following == seq + 1:
            # The printed run continues from here, just not from where we were:
            # an okres boundary resets the count, and okres headings are not
            # legible from the table body. Re-anchor rather than score a failure.
            verdict, detail = "na", f"re-anchored at {seq}"
        else:
            verdict = "fail"
            detail = f"printed={block.municipality.get('seq')!r} expected={expected}"

        findings.append({
            "block_id": block.block_id, "row_kind": "municipality",
            "row_name": block.name, "check": "I5_sequence",
            "verdict": verdict, "detail": detail,
        })
        # Carry the *run* forward, not a value we just judged wrong — otherwise one
        # misread digit scores twice, once where it is read and once on the next
        # block that disagrees with it.
        if verdict in ("pass",) or detail.startswith("re-anchored"):
            previous = seq
        elif expected is not None:
            previous = expected
        elif status == "ok":
            previous = seq

        for n, row in enumerate(block.settlements, start=1):
            value, status = parse_int_cell(row.get("seq"))
            if status != "ok":
                verdict = "unparseable"
            else:
                verdict = "pass" if value == n else "fail"
            findings.append({
                "block_id": block.block_id, "row_kind": "settlement",
                "row_name": row.get("name"), "check": "I5_sequence",
                "verdict": verdict,
                "detail": "" if verdict == "pass"
                          else f"printed={row.get('seq')!r} expected={n}",
            })
    return findings


def grade_page(blocks: list[Block]) -> list[dict]:
    """Every check for one page's blocks, one dict per check."""
    records: list[dict] = []
    for block in blocks:
        for row in block.rows:
            kind = "municipality" if row is block.municipality else "settlement"
            for inv in INVARIANTS:
                verdict, detail = grade_row(row, inv)
                records.append({
                    "block_id": block.block_id, "row_kind": kind,
                    "row_name": row.get("name"), "check": inv.key,
                    "verdict": verdict, "detail": detail,
                })
        for col, (verdict, detail) in grade_block_sum(block).items():
            records.append({
                "block_id": block.block_id, "row_kind": "block",
                "row_name": block.name, "check": f"I4_block:{col}",
                "verdict": verdict, "detail": detail,
            })
    records += grade_sequence(blocks)
    return records

def render_page(pdf_path: str | Path, page_number: int, target_width: int):
    """Render one PDF page at exactly ``target_width`` pixels.

    The NDK ALTO frame is the scan's own pixel grid, so rendering to the ALTO width
    makes every coordinate in this module land 1:1 on the image with no scale
    factor to carry around.
    """
    import fitz
    from PIL import Image

    with fitz.open(str(pdf_path)) as doc:
        page = doc[page_number - 1]
        zoom = target_width / page.rect.width
        pix = page.get_pixmap(matrix=fitz.Matrix(zoom, zoom))
        return Image.frombytes("RGB", (pix.width, pix.height), pix.samples)


def running_head(page: AltoPage, above_table: bool = False) -> str:
    """The text printed above the table body — page number and running head.

    The head names the okres and the soudní okres (``Chotěbor: Chotěboř.``), which
    is how a page is assigned to a district. Its vertical position varies by tens of
    pixels across the volume, so the band is taken from the fitted table body rather
    than a fixed cutoff.

    That band also takes in the column header's labels on every page, and on the
    pages whose :func:`body_band` opens late (see :func:`table_top`) the table's
    first rows: 2,865 table words on 50 scans. ``above_table=True`` ends the head
    at the table's top rule instead, where the ALTO has one (:func:`header_foot`'s
    pair). **Off by default** (0.3.0), and measured as a mixed change
    (2026-10-14): ``lexikon_1910_notes.court_of`` reads the right soudní okres on
    4 more pages (each had matched a spurious ``as``, Aš, in the captions; a fifth
    is sealed), but ``lexikon_1910_names.assign_okresy`` loses Prague on scan 0015,
    whose head had been right only because it swallowed the first table row.
    """
    try:
        top, _bottom = body_band(page)
    except ValueError:
        top = int(page.height * 0.15)
    if above_table:
        found = _header_rules(page)
        if found is not None:
            top = min(top, found[0])
    return " ".join(text for vpos, _h, _w, height, text in page.strings
                    if vpos + height < top)


def crop_block(image, page: AltoPage, block: Block,
               template: list[float] | None = None,
               pad_x: int = 18, pad_y: int = 6):
    """Crop one municipality block out of the rendered page.

    Vertical padding is deliberately tighter than horizontal: the rows are ~44 px
    apart, so a generous pad slices the neighbouring municipality's figures into
    the image and invites the reader to transcribe half a row that belongs to a
    different block. The horizontal pad only adds white margin.
    """
    cols = fit_columns(page, template)[0] if template else column_boundaries(page)
    left = max(0, cols[0] - pad_x)
    right = min(image.width, cols[-1] + pad_x)
    top = max(0, block.top - pad_y)
    bottom = min(image.height, block.bottom + pad_y)
    return image.crop((left, top, right, bottom))


#: The vision API resamples anything longer than this on its long edge. A block
#: crop is ~2672 px wide, so *every* block was being silently downscaled before the
#: model saw it — on exactly the digits the run exists to read.
VISION_LONG_EDGE = 2576


def scale_to_width(image, target_width: int):
    """Resize to ``target_width``, preserving aspect, with a good filter.

    Done here rather than left to the API so the resample is ours and known. Above
    the ceiling this is a downscale we control; below it, an upscale that adds no
    information but gives the vision encoder more patches per glyph — which is the
    whole point of tiling, since the scan is already at its native 399 DPI and
    there is no further detail to recover.
    """
    from PIL import Image

    if image.width == target_width:
        return image
    ratio = target_width / image.width
    return image.resize((target_width, max(1, round(image.height * ratio))),
                        Image.LANCZOS)


def tile_block(image, page: AltoPage, block: Block,
               template: list[float] | None = None, tiles: int = 1,
               target_width: int = VISION_LONG_EDGE,
               pad_x: int = 18, pad_y: int = 6) -> list[tuple]:
    """Cut one block into ``tiles`` images, each scaled to the vision ceiling.

    Returns ``[(image, first_col, last_col), …]``. With ``tiles=1`` this is
    :func:`crop_block` plus a controlled resize.

    Above one tile the block is split by *column group*, and **columns 1–2 (the
    sequence number and the name) are pasted onto the front of every tile**. Split
    naively, the second tile would be a field of bare digits with nothing to say
    which row is which, and the tiles could only be realigned by position — an
    off-by-one would silently shift a whole block's figures onto the wrong
    settlements. Repeating the name costs a few hundred pixels and makes each tile
    independently readable.
    """
    from PIL import Image

    cols = fit_columns(page, template)[0] if template else column_boundaries(page)
    top = max(0, block.top - pad_y)
    bottom = min(image.height, block.bottom + pad_y)

    def strip(left: int, right: int):
        return image.crop((max(0, left - pad_x), top,
                           min(image.width, right + pad_x), bottom))

    if tiles <= 1:
        return [(scale_to_width(strip(cols[0], cols[-1]), target_width), 0, N_COLS - 1)]

    # Columns 0–1 are the label; 2..N_COLS-1 are split between the tiles.
    first_data = 2
    data = list(range(first_data, N_COLS))
    per = -(-len(data) // tiles)                      # ceiling division
    label = strip(cols[0], cols[first_data])

    out: list[tuple] = []
    for i in range(0, len(data), per):
        group = data[i:i + per]
        body = strip(cols[group[0]], cols[group[-1] + 1])
        canvas = Image.new("RGB", (label.width + body.width,
                                   max(label.height, body.height)), "white")
        canvas.paste(label, (0, 0))
        canvas.paste(body, (label.width, 0))
        out.append((scale_to_width(canvas, target_width), group[0], group[-1]))
    return out


def crop_header(image, page: AltoPage, template: list[float] | None = None,
                pad: int = 8):
    """Crop the printed column-header band above the table body."""
    cols = fit_columns(page, template)[0] if template else column_boundaries(page)
    band_top, _ = body_band(page)
    _frame, body = _rule_sets(page)
    top = max(0, min(v["VPOS"] for v in body) - pad)
    return image.crop((max(0, cols[0] - pad), top,
                       min(image.width, cols[-1] + pad), band_top + pad))

def lexikon_1910_report(records: list[dict]) -> str:
    """Human-readable pass/fail summary over a list of grading records."""
    from collections import Counter

    by_check: dict[str, Counter] = {}
    for rec in records:
        by_check.setdefault(rec["check"].split(":")[0], Counter())[rec["verdict"]] += 1

    lines = [f"lexikon_1910 v{__version__} — invariant report", ""]
    header = f"{'check':<16}{'pass':>7}{'fail':>7}{'unparse':>9}{'n/a':>7}{'rate':>9}"
    lines += [header, "-" * len(header)]
    grand = Counter()
    for check, counter in sorted(by_check.items()):
        grand += counter
        graded = counter["pass"] + counter["fail"] + counter["unparseable"]
        rate = f"{counter['pass'] / graded:6.1%}" if graded else "     --"
        lines.append(
            f"{check:<16}{counter['pass']:>7}{counter['fail']:>7}"
            f"{counter['unparseable']:>9}{counter['na']:>7}{rate:>9}"
        )
    graded = grand["pass"] + grand["fail"] + grand["unparseable"]
    rate = f"{grand['pass'] / graded:6.1%}" if graded else "     --"
    lines += ["-" * len(header),
              f"{'ALL':<16}{grand['pass']:>7}{grand['fail']:>7}"
              f"{grand['unparseable']:>9}{grand['na']:>7}{rate:>9}",
              "",
              "n/a = check not applicable (cells blank); never counted as a pass."]
    return "\n".join(lines)
