How it works

What this measures, and what it does not

education.py does two separable things, and they need separate evidence:

  1. it reads a string — finds the schools named in ob. 5 mě. 3 pok. 3 and the years attached to each;
  2. it places them on a ladder — decides that a Bürgerschule outranks a Volksschule and that a Fortbildungsschule is not a rung at all.

(2) is checked against the Austrian administration's own school census in run_unterricht_check.py, which is the only thing that could contradict it. This sample checks (1), which no external source can: whether the tokeniser read what a Czech reader reads. The two together cover the variable; neither covers it alone.

So the human is asked to name the institutions and the years, not to give an ordinal. The scorer then puts their institutions on the ladder with the same INSTITUTIONS table the parser uses. A disagreement is then unambiguously a reading disagreement, which is what we can act on; if the coder were asked for a rung, a disagreement would confound the two.

The worksheet is blind

It carries the raw string and nothing else. Showing the coder what the parser produced would measure assent, not accuracy — people confirm a plausible answer in front of them. This is the same reason run_occcanine_check.py codes from the gloss rather than from our HISCO.

Disproportionate stratified sampling

A simple random draw of 300 records would spend ~270 of them on plain single-school strings and land two or three on the strata where the parser actually makes decisions. The strata below are sized by risk, not by frequency, and the scorer reweights back to the true stratum sizes for the headline number — so the tail is measured precisely without inflating or deflating the overall figure.

plain            71,622   one school, no complication — the bulk
multi_rung       17,341   two or more scored schools: the itinerary and the D-51 tie-break
elem_with_years  14,516   elementary with a year count — the population D-53 is about
imputed_floor     1,112   `educ_top` inferred from the apprentice floor, not named
unparsed            952   non-empty and the parser returned nothing
rung0               449   No schooling / illiterate — small and high-consequence

Two disjoint halves, assigned by alternating within each stratum so both keep the spread:

CALIBRATION    fix the dictionary here; re-code freely, iterate
CERTIFICATION  sealed. Coded once, at the end, and reported

They must not be the same records. An accuracy measured on the strings whose failures you just fixed is a number about the fixing.

Run: python utils/reference_tools/education/draw_education_sample.py python utils/reference_tools/education/draw_education_sample.py --seed 1910 --scale 2

Emits: education_gold/{calibration,certification}/worksheet.csv (blind, for a human) education_gold/_frame.csv (ids + strata, for reweighting) Writes no answers and touches no pipeline column.

def classify(raw: str) -> str:
    """Which stratum a raw string falls in. Mirrors the docstring table."""
    from utils.parsing.education import parse_education, INSTITUTIONS

    if not isinstance(raw, str) or not raw.strip():
        return "empty"
    parsed = parse_education(raw)
    if parsed is None:
        return "unparsed"
    if parsed.get("educ_top_imputed"):
        return "imputed_floor"
    if parsed["educ_ord"] == 0:
        return "rung0"
    named = [k for k in parsed["educ_institutions"].split("|") if k]
    scored = [k for k in named if INSTITUTIONS[k][2] is not None]
    if len(scored) >= 2:
        return "multi_rung"
    if parsed["educ_top"] == "elementary" and parsed["educ_years_elementary"] is not None:
        return "elem_with_years"
    return "plain"

def systematic(items: list, n: int, rng: random.Random) -> list:
    """Systematic draw with a random start — spreads over the frame by construction.

    A simple random draw can cluster; `Vzdělání` strings are ordered by scrape order,
    which tracks the record id and therefore, loosely, the archive's own ordering.
    """
    if n >= len(items):
        return list(items)
    step = len(items) / n
    start = rng.uniform(0, step)
    return [items[min(int(start + i * step), len(items) - 1)] for i in range(n)]

def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--seed", type=int, default=SEED)
    ap.add_argument("--scale", type=float, default=1.0,
                    help="multiply every stratum's allocation")
    args = ap.parse_args()
    rng = random.Random(args.seed)

    df = pd.read_csv(RAW, usecols=[ID, FIELD], dtype=str)
    df["stratum"] = [classify(s) for s in df[FIELD]]
    pool = df[df["stratum"] != "empty"]

    frame_rows, sheets = [], {"calibration": [], "certification": []}
    print(f"seed={args.seed}  scale={args.scale}")
    print()
    print(f"{'stratum':18s} {'in frame':>9s} {'drawn':>6s}  {'calib':>5s} {'cert':>5s}")
    for stratum, base in ALLOCATION.items():
        want = int(round(base * args.scale))
        rows = pool[pool["stratum"] == stratum].to_dict("records")
        rows.sort(key=lambda r: str(r[ID]))
        drawn = systematic(rows, want, rng)
        for i, r in enumerate(drawn):
            half = "calibration" if i % 2 == 0 else "certification"
            sheets[half].append({
                "record_id": r[ID], "vzdelani": r[FIELD],
                "instituce": "", "roky": "", "ucen": "", "dokonceno": "", "poznamka": "",
            })
            frame_rows.append({"record_id": r[ID], "stratum": stratum, "half": half,
                               "stratum_size": len(rows)})
        n_cal = sum(1 for i in range(len(drawn)) if i % 2 == 0)
        print(f"{stratum:18s} {len(rows):9,} {len(drawn):6d}  {n_cal:5d} "
              f"{len(drawn) - n_cal:5d}")

    for half, rows in sheets.items():
        d = OUT / half
        d.mkdir(parents=True, exist_ok=True)
        rng.shuffle(rows)          # so the coder cannot infer the stratum from position
        with open(d / "worksheet.csv", "w", newline="", encoding="utf-8") as fh:
            w = csv.DictWriter(fh, fieldnames=WORKSHEET_FIELDS)
            w.writeheader()
            w.writerows(rows)

    OUT.mkdir(parents=True, exist_ok=True)
    with open(OUT / "_frame.csv", "w", newline="", encoding="utf-8") as fh:
        w = csv.DictWriter(fh, fieldnames=["record_id", "stratum", "half", "stratum_size"])
        w.writeheader()
        w.writerows(frame_rows)

    print()
    print(f"wrote {(OUT / 'calibration' / 'worksheet.csv').relative_to(ROOT)}"
          f"  ({len(sheets['calibration'])} rows)")
    print(f"wrote {(OUT / 'certification' / 'worksheet.csv').relative_to(ROOT)}"
          f"  ({len(sheets['certification'])} rows)")
    print(f"wrote {(OUT / '_frame.csv').relative_to(ROOT)}")
    print()
    print("The worksheets carry no parser output. Read education_gold/NAVOD.md before coding,")
    print("code CALIBRATION first, and leave CERTIFICATION sealed until the dictionary is final.")
    return 0

from __future__ import annotations

import argparse
import csv
import random
import sys
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

__version__ = "1.0.0"

OUT = ROOT / "reference" / "education" / "education_gold"
RAW = ROOT / "data" / "raw" / "legionnaires" / "legionnaires.csv"
FIELD = "Vzdělání"
ID = "ID"   # the csol.cz record id, as scraped

SEED = 1910

# Draws per stratum at --scale 1, split half calibration / half certification.
# Sized by how much judgement the parser exercises there, not by how common it is.
ALLOCATION = {
    "plain": 90,
    "multi_rung": 60,
    "elem_with_years": 60,
    "imputed_floor": 30,
    "unparsed": 40,
    "rung0": 20,
}

# The worksheet deliberately carries no `stratum`: it is the parser's own classification,
# and "unparsed" in particular would tell the coder the machine had already failed. The
# stratum lives in `_frame.csv`, which the scorer joins back on `record_id`.
WORKSHEET_FIELDS = [
    "record_id", "vzdelani",
    "instituce", "roky", "ucen", "dokonceno", "poznamka",
]
