How it works

German E3 failed on finding the glyph (D-175): the locator, the entry finder and detection by naming all fail on the German print, while DINOv3 names a correctly boxed German glyph well once German exemplars are in the bank. This tests the one route around the finding: the reader of record (Gemini 3.8 Flash, thinking off, the German offices prompt and the volume's key, as the frozen route reads) is also asked for a box around every mark it writes, and DINOv3 judges that box.

Three stages, each on the three German gold pages the author clicked (0142, 0276, 0480; offices sections, the route's own section images; never certification or reserve pages):

read one Gemini call per section image (paid, cents; cached in output/glyph_e3_de/reader_boxes, so a rerun costs nothing). Run on Heavyhorse6, the machine of record for paid reads. boxes each returned mark against the author's 203 named clicks, one to one (a click goes to the box whose centre is nearest, among boxes holding it within 0.1 line heights): right the box holds a click of the token the reader wrote misread the box holds a click of another token (a false claim on a real glyph) box off no click in the box, a click of its token within 1.5 line heights (a true claim, badly boxed) phantom no click of its token within 1.5 line heights (a false claim) judge DINOv3 (or the pixel floor) names every box against the Czech bank + the German click exemplars of the other pages + a reject class ([not-glyph]: candidate clusters on the other pages' exhaustively clicked strips that hold no click), each page held out in turn, its threshold the 5 % quantile of the other pages' correctly named exemplars.

The bars, fixed before any read (2026-10-31, in reference/LEXIKON_1910_ROUTE.md): box right / (right + box off) >= 0.90 cover judged share of the right claims >= 0.835 (Czech 0.935 - 0.10) keep agreement among judged right claims >= 0.936 (Czech 0.986 - 0.05) catch disagreement among judged false claims (misread + phantom) >= 0.50, judged only with >= 10 A miss on any bar closes German E3 for this thesis (D-175 stands). Three folds, in-sample gold pages: a pass is a reason to build it and certify it on the sealed reserve pages, not a certification.

python utils/reference_tools/places/run_german_reader_boxes.py --stage read            # Heavyhorse6, paid
python utils/reference_tools/places/run_german_reader_boxes.py --stage boxes           # anywhere
python utils/reference_tools/places/run_german_reader_boxes.py --stage judge --encoder dinov3

Result 2026-11-01 (D-176): box, cover and catch pass, keep 0.872 fails. The queries had been cut from Gemini's raw box (median IoU 0.35 with the ink-tight box of the matched click, 1.6x its width) while the bank's German exemplars are cut by build_glyph_bank.tight_box at the click.

--snap is the one declared follow-up (D-177, rule and bars fixed before the run): each query is re-cut by the bank's own operator, tight_box on the same ink mask (--ink, as the exemplars) at the integer centre of Gemini's box, with no search and no parameter; where it finds no ink the raw box is kept and counted. Categories stay as Gemini's raw box gives them, the D-176 bars and the threshold rule are unchanged, and the D-176 fail stays on the record. Run once.

Outside utils/lexikon_1910 so as not to change the route's code digests.

def read(pages: list[str], reasoning: str) -> None:
    from PIL import Image

    import bakeoff as bo
    import run_frozen as rf
    import run_lexikon_1910_pilot as pilot
    import run_r5_german_notes as r5
    from utils.lexikon_1910 import lexikon_1910_institutions as lxin

    abbreviations = lxin.load_abbreviations(ROOT / "reference" / "places" / "institution_abbreviations.csv",
                                            lang=rf._lang(EDITION), volume=VOLUME)
    system = r5.systems(abbreviations, volume=VOLUME)["offices"] + BOX_NOTE
    url, key_env = pilot.READER_ENDPOINTS["gemini"]
    reader = pilot.OpenAICompatibleReader(MODEL, url, key_env, system=system, schema=SCHEMA,
                                          payload_key="marks", schema_name="marks", max_tokens=8000,
                                          temperature=0.0, reasoning_effort=reasoning)
    atlas = Image.open(bo.de_atlas(VOLUME))
    CACHE.mkdir(parents=True, exist_ok=True)
    spent = 0.0
    for name in section_images(pages):
        hit = CACHE / f"{Path(name).stem}_{reasoning}.json"
        if hit.exists():
            print(f"  {name}: cached")
            continue
        crop = Image.open(SECTIONS / name)
        for attempt in range(3):
            try:
                marks, usage = reader([(atlas, r5.ATLAS_CAPTION), (crop, "")], "Transcribe this image.")
                break
            except pilot.Truncated:
                reader.max_tokens = min(reader.max_tokens * 2, 32000)
            except pilot.RateLimited:
                raise
            except Exception as exc:                                    # noqa: BLE001
                if attempt == 2:
                    raise
                print(f"  {name}: {str(exc)[:120]}; retrying")
                time.sleep(3 * (attempt + 1))
        cost = (usage.get("input", 0) * PRICE[0] + usage.get("output", 0) * PRICE[1]) / 1e6
        spent += cost
        hit.write_text(json.dumps({"image": name, "size": list(crop.size), "marks": marks, "usage": usage,
                                   "model": MODEL, "reasoning": reasoning, "cost": cost}), encoding="utf-8")
        print(f"  {name}: {len(marks)} marks, ${cost:.4f}")
    print(f"read: ${spent:.4f} spent this run")

def reader_marks(pages: list[str], reasoning: str) -> list[dict]:
    """Every returned mark: ``{image, page, token, box (section px), lh}``."""
    from PIL import Image

    from build_glyph_bank import line_height

    out = []
    for name in section_images(pages):
        hit = CACHE / f"{Path(name).stem}_{reasoning}.json"
        if not hit.exists():
            raise SystemExit(f"{hit.name} missing: run --stage read first (Heavyhorse6)")
        record = json.loads(hit.read_text(encoding="utf-8"))
        w, h = record["size"]
        lh = line_height(Image.open(SECTIONS / name).convert("L")) or 46
        for m in record["marks"]:
            box = m.get("box_2d") or []
            if len(box) != 4:
                continue
            y0, x0, y1, x1 = box
            out.append({"image": name, "page": name[:4], "token": m.get("token", ""),
                        "box": (round(x0 / 1000 * w), round(y0 / 1000 * h), round(x1 / 1000 * w), round(y1 / 1000 * h)),
                        "lh": lh})
    return out

def classify_marks(marks: list[dict]) -> list[dict]:
    """Each mark's category against the clicks, one to one."""
    import german_click_recall as rec

    clicks = rec.load_clicks()
    by_image = defaultdict(list)
    for i, m in enumerate(marks):
        by_image[m["image"]].append(i)
    for name, idx in by_image.items():
        pts = clicks.get(name, [])
        pairs = []
        for i in idx:
            x0, y0, x1, y1 = marks[i]["box"]
            pad = 0.1 * marks[i]["lh"]
            cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
            for j, (x, y, *_r) in enumerate(pts):
                if x0 - pad <= x <= x1 + pad and y0 - pad <= y <= y1 + pad:
                    pairs.append((math.dist((cx, cy), (x, y)), i, j))
        used_mark, used_click = set(), set()
        for _d, i, j in sorted(pairs):
            if i in used_mark or j in used_click:
                continue
            used_mark.add(i)
            used_click.add(j)
            marks[i]["click"] = pts[j][2]
        for i in idx:
            m = marks[i]
            if "click" in m:
                m["category"] = "right" if m["click"] == m["token"] else "misread"
                continue
            x0, y0, x1, y1 = m["box"]
            cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
            near = [math.dist((cx, cy), (x, y)) for x, y, tok, *_r in pts if tok == m["token"]]
            m["category"] = "box off" if near and min(near) <= 1.5 * m["lh"] else "phantom"
    return marks

def box_report(marks: list[dict], n_clicks: int) -> tuple[dict, list[str]]:
    count = Counter(m["category"] for m in marks)
    box = count["right"] / max(count["right"] + count["box off"], 1)
    lines = [f"  reader marks {len(marks)}: " + ", ".join(f"{k} {count[k]}" for k in ("right", "box off", "misread", "phantom")),
             f"  box bar: right / (right + box off) = {box:.3f}  (bar {BARS['box']})  {'PASS' if box >= BARS['box'] else 'FAIL'}",
             f"  clicked glyphs the reader boxed and named right: {count['right']} of {n_clicks} ({count['right'] / max(n_clicks, 1):.3f})"]
    return {"counts": dict(count), "box": box}, lines

def snap_boxes(marks: list[dict], ink: int) -> tuple[dict, list[str]]:
    """Re-cut every mark's box by the bank's operator (D-177): ``tight_box`` on the ink mask at
    ``ink`` at the integer centre of Gemini's box. The raw box is kept in ``raw_box``; where no ink
    is near, the raw box stays and the mark is counted."""
    from PIL import Image

    import locate_glyphs as lg
    from build_glyph_bank import tight_box

    masks, fallback = {}, 0
    for m in marks:
        if m["image"] not in masks:
            masks[m["image"]] = lg.ink_mask(Image.open(SECTIONS / m["image"]), ink)
        x0, y0, x1, y1 = m["raw_box"] = m["box"]
        box = tight_box(masks[m["image"]], (x0 + x1) // 2, (y0 + y1) // 2, m["lh"])
        if box is None:
            fallback += 1
            continue
        m["box"] = tuple(int(v) for v in box)
    by_cat = Counter(m["category"] for m in marks if m["box"] == m["raw_box"])
    return ({"snapped": len(marks) - fallback, "raw_kept": fallback},
            [f"  snap (D-177): {len(marks) - fallback} of {len(marks)} boxes re-cut by tight_box at ink < {ink}; "
             f"{fallback} kept raw (no ink near the centre){': ' + dict(by_cat).__repr__() if fallback else ''}"])

def negative_glyphs(pages: list[str], ink: int, clear_lines: float = 0.5) -> list:
    """Candidate clusters on the clicked sections that hold no click and lie off every click:
    the reject class. The strips were clicked exhaustively, so these are not pictograms."""
    from PIL import Image

    import german_click_recall as rec
    import locate_glyphs as lg
    import run_glyph_e3 as e3
    from build_glyph_bank import line_height

    clicks = rec.load_clicks()
    orig = lg.ink_mask
    lg.ink_mask = lambda image, threshold=ink: orig(image, threshold)
    out = []
    try:
        for name in section_images(pages):
            image = Image.open(SECTIONS / name)
            lh = line_height(image.convert("L")) or 46
            for k, c in enumerate(lg.candidates_on_image(image, name, lh)):
                if c.bracket or lg.is_speck(c) or c.features["log_w"] > math.log(lg.MAX_GLYPH_WIDTH):
                    continue
                pad = clear_lines * lh
                if any(c.box[0] - pad <= x <= c.box[2] + pad and c.box[1] - pad <= y <= c.box[3] + pad
                       for x, y, *_r in clicks.get(name, [])):
                    continue
                out.append(e3.Glyph(f"de{name[:4]}", "negative", "offices", "", 0, "", k, NOT_GLYPH, name,
                                    c.box, lh, SECTIONS, "negative"))
    finally:
        lg.ink_mask = orig
    return out

def judge(marks: list[dict], encoder: str, download: bool, ink: int, rounds: int) -> tuple[dict, list[str]]:
    import numpy as np

    import locate_glyphs as lg
    import run_e3_pages as e3p
    import run_german_oracle as oracle
    import run_glyph_e3 as e3

    config = e3.OF_RECORD if encoder == "dinov3" else e3.FLOOR
    data = lg.load_sections()
    located, _ = lg.locate_all(data, rounds)
    bank = e3.placed_glyphs(located)
    namer = e3p.Namer(bank, config, download)

    def embed(glyphs):
        got = e3.embed(glyphs, [namer.name], download, cuts=(namer.variant,))
        return got[(namer.name, namer.variant)][namer.feature]

    pages = sorted({m["page"] for m in marks})
    positives, _kept, _skipped = oracle.click_glyphs(oracle.read_clicks(), "p1", ink)
    negatives = negative_glyphs(pages, ink)
    queries = [e3.Glyph(f"de{m['page']}", "query", "offices", "", 0, "", i, m["token"], m["image"],
                        m["box"], m["lh"], SECTIONS, "reader") for i, m in enumerate(marks)]
    pos_x, neg_x, q_x = embed(positives), embed(negatives), embed(queries)
    pos_t, pos_p = [g.base for g in positives], [g.scan for g in positives]
    neg_p = [g.scan for g in negatives]
    cz_t, cz_p = [g.base for g in bank], [g.scan for g in bank]

    verdicts = [None] * len(marks)
    thresholds = {}
    for held in sorted({q.scan for q in queries}):
        keep_pos = [i for i, p in enumerate(pos_p) if p != held]
        keep_neg = [i for i, p in enumerate(neg_p) if p != held]
        bank_x = np.vstack([namer.bank_x, pos_x[keep_pos], neg_x[keep_neg]])
        bank_t = cz_t + [pos_t[i] for i in keep_pos] + [NOT_GLYPH] * len(keep_neg)
        bank_p = cz_p + [pos_p[i] for i in keep_pos] + [neg_p[i] for i in keep_neg]
        own = e3.classify(pos_x[keep_pos], [pos_p[i] for i in keep_pos], bank_x, bank_t, bank_p, namer.k)
        right = [s for (g, s), i in zip(own, keep_pos) if g == pos_t[i]]
        tau = float(np.quantile(right, e3.ABSTAIN_QUANTILE)) if right else -1.0
        thresholds[held] = round(tau, 3)
        idx = [i for i, q in enumerate(queries) if q.scan == held]
        for (name, s), i in zip(e3.classify(q_x[idx], [held] * len(idx), bank_x, bank_t, bank_p, namer.k), idx):
            base = lg.base_of(marks[i]["token"])
            verdicts[i] = ("not judged" if s < tau else "agree" if name == base else "disagree", name, s)
    for m, v in zip(marks, verdicts):
        m["verdict"], m["dino"], m["similarity"] = v

    def share(category_set, verdict):
        rows = [m for m in marks if m["category"] in category_set and m["verdict"] != "not judged"]
        return (sum(m["verdict"] == verdict for m in rows) / len(rows)) if rows else None, len(rows)

    right_rows = [m for m in marks if m["category"] == "right"]
    cover = sum(m["verdict"] != "not judged" for m in right_rows) / max(len(right_rows), 1)
    keep, n_keep = share({"right"}, "agree")
    catch, n_false = share({"misread", "phantom"}, "disagree")
    off_agree, n_off = share({"box off"}, "agree")
    result = {"encoder": list(config), "thresholds": thresholds, "cover": cover, "keep": keep,
              "catch": catch, "judged_false": n_false, "box_off_agree": off_agree,
              "verdicts": {f"{c}/{v}": n for (c, v), n in sorted(Counter((m["category"], m["verdict"]) for m in marks).items())}}

    def bar(value, name):
        return "—" if value is None else ("PASS" if value >= BARS[name] else "FAIL")
    lines = [f"  DINOv3 judging ({encoder} {config}); thresholds by held-out page {thresholds}",
             f"  cover: judged share of right claims {cover:.3f} (bar {BARS['cover']})  {bar(cover, 'cover')}",
             f"  keep:  agreement among {n_keep} judged right claims {keep if keep is None else round(keep, 3)} "
             f"(bar {BARS['keep']})  {bar(keep, 'keep')}",
             f"  catch: disagreement among {n_false} judged false claims {catch if catch is None else round(catch, 3)} "
             f"(bar {BARS['catch']}, judged only with >= {MIN_FALSE})  "
             + (bar(catch, 'catch') if n_false >= MIN_FALSE else "not judged: too few false claims"),
             f"  box off (true claims, badly boxed): agreed {off_agree if off_agree is None else round(off_agree, 3)} of {n_off} judged",
             "  category x verdict: " + ", ".join(f"{k} {n}" for k, n in result["verdicts"].items())]
    return result, lines

def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--stage", choices=("read", "boxes", "judge", "all"), default="all")
    parser.add_argument("--encoder", choices=("pixels", "dinov3"), default="dinov3")
    parser.add_argument("--reasoning", default="none", help="Gemini reasoning_effort (the route's: none)")
    parser.add_argument("--download", action="store_true")
    parser.add_argument("--ink", type=int, default=128)
    parser.add_argument("--rounds", type=int, default=4)
    parser.add_argument("--snap", action="store_true", help="D-177: re-cut each box by tight_box before judging")
    args = parser.parse_args(argv)

    pages = clicked_pages()
    if args.stage in ("read", "all"):
        read(pages, args.reasoning)
    if args.stage == "read":
        return 0
    import german_click_recall as rec

    n_clicks = sum(len(v) for v in rec.load_clicks().values())
    marks = classify_marks(reader_marks(pages, args.reasoning))
    summary, lines = box_report(marks, n_clicks)
    lines = [f"Reader boxes: {MODEL}, reasoning {args.reasoning}, pages {pages}"] + lines
    if args.snap:
        summary["snap"], more = snap_boxes(marks, args.ink)
        lines += more
    if args.stage in ("judge", "all"):
        result, more = judge(marks, args.encoder, args.download, args.ink, args.rounds)
        summary["judge"] = result
        lines += more
    text = "\n".join(lines)
    print(text)
    OUT.mkdir(parents=True, exist_ok=True)
    stem = (f"reader_boxes_{args.reasoning}" + (f"_{args.encoder}" if args.stage in ("judge", "all") else "")
            + ("_snap" if args.snap else ""))
    (OUT / f"{stem}.json").write_text(json.dumps({"summary": summary, "marks": marks}, indent=1, default=str), encoding="utf-8")
    (OUT / f"{stem}.txt").write_text(text + "\n", encoding="utf-8")
    return 0

"""The reader gives the box: can DINOv3 check German claims on the boxes Gemini itself returns?

German E3 failed on *finding* the glyph (D-175): the locator, the entry finder and detection by
naming all fail on the German print, while DINOv3 names a correctly boxed German glyph well once
German exemplars are in the bank. This tests the one route around the finding: the reader of record
(Gemini 3.8 Flash, thinking off, the German offices prompt and the volume's key, as the frozen route
reads) is also asked for a box around every mark it writes, and DINOv3 judges that box.

Three stages, each on the three German gold pages the author clicked (0142, 0276, 0480; offices
sections, the route's own section images; never certification or reserve pages):

  read    one Gemini call per section image (paid, cents; cached in ``output/glyph_e3_de/reader_boxes``,
          so a rerun costs nothing). Run on Heavyhorse6, the machine of record for paid reads.
  boxes   each returned mark against the author's 203 named clicks, one to one (a click goes to the
          box whose centre is nearest, among boxes holding it within 0.1 line heights):
            right    the box holds a click of the token the reader wrote
            misread  the box holds a click of another token           (a false claim on a real glyph)
            box off  no click in the box, a click of its token within 1.5 line heights (a true claim, badly boxed)
            phantom  no click of its token within 1.5 line heights    (a false claim)
  judge   DINOv3 (or the pixel floor) names every box against the Czech bank + the German click
          exemplars of the *other* pages + a reject class (``[not-glyph]``: candidate clusters on the
          other pages' exhaustively clicked strips that hold no click), each page held out in turn,
          its threshold the 5 % quantile of the other pages' correctly named exemplars.

The bars, fixed before any read (2026-10-31, in ``reference/LEXIKON_1910_ROUTE.md``):
  box      right / (right + box off)                   >= 0.90
  cover    judged share of the ``right`` claims        >= 0.835   (Czech 0.935 - 0.10)
  keep     agreement among judged ``right`` claims     >= 0.936   (Czech 0.986 - 0.05)
  catch    disagreement among judged false claims (misread + phantom) >= 0.50, judged only with >= 10
A miss on any bar closes German E3 for this thesis (D-175 stands). Three folds, in-sample gold pages:
a pass is a reason to build it and certify it on the sealed reserve pages, not a certification.

    python utils/reference_tools/places/run_german_reader_boxes.py --stage read            # Heavyhorse6, paid
    python utils/reference_tools/places/run_german_reader_boxes.py --stage boxes           # anywhere
    python utils/reference_tools/places/run_german_reader_boxes.py --stage judge --encoder dinov3

Result 2026-11-01 (D-176): box, cover and catch pass, keep 0.872 fails. The queries had been cut from
Gemini's raw box (median IoU 0.35 with the ink-tight box of the matched click, 1.6x its width) while
the bank's German exemplars are cut by ``build_glyph_bank.tight_box`` at the click.

``--snap`` is the one declared follow-up (D-177, rule and bars fixed before the run): each query is
re-cut by the bank's own operator, ``tight_box`` on the same ink mask (``--ink``, as the exemplars)
at the integer centre of Gemini's box, with no search and no parameter; where it finds no ink the
raw box is kept and counted. Categories stay as Gemini's raw box gives them, the D-176 bars and the
threshold rule are unchanged, and the D-176 fail stays on the record. Run once.

Outside ``utils/lexikon_1910`` so as not to change the route's code digests.
"""

SECTIONS = ROOT / "output" / "frozen" / "de-bohemia_gold" / "sections"
OUT = ROOT / "output" / "glyph_e3_de"
CACHE = OUT / "reader_boxes"
MODEL, PRICE = "gemini-3.8-flash", (0.75, 3.75)          # USD per million tokens in / out (run_r5_german_notes)
VOLUME, EDITION = "boehmen_de", "de-bohemia"
NOT_GLYPH = "[not-glyph]"
BARS = {"box": 0.90, "cover": 0.835, "keep": 0.936, "catch": 0.50}
MIN_FALSE = 10

BOX_NOTE = """

POSITIONS. Besides the lines, return "marks": one object for every bracketed symbol token you wrote,
in the order you wrote them. "token" is the token exactly as written; "line" is the 0-based index
of the line in "lines" it is on; "box_2d" is the box around that printed symbol in the crop (the last
image), as [ymin, xmin, ymax, xmax] on a 0-1000 scale of the crop's height and width. Box the symbol
alone, not the letters, digits, brackets or punctuation beside it. A symbol printed twice is two marks.
"""

SCHEMA = {
    "type": "object",
    "properties": {
        "lines": {"type": "array", "items": {"type": "string"}},

            "type": "object",
            "properties": {"token": {"type": "string"}, "line": {"type": "integer"},

            "required": ["token", "line", "box_2d"], "additionalProperties": False}},

    "required": ["lines", "marks"],
    "additionalProperties": False,
}


def section_images(pages: list[str]) -> list[str]:
    import csv

    with (SECTIONS / "index.csv").open(encoding="utf-8", newline="") as fh:
        return [r["image"] for r in csv.DictReader(fh) if r["page"] in pages and r["kind"] == "offices"]


def clicked_pages() -> list[str]:
    import german_click_recall as rec

    return sorted({p for v in rec.load_clicks().values() for _x, _y, _t, p, _s in v})

def read(pages: list[str], reasoning: str) -> None:
    from PIL import Image

    import bakeoff as bo
    import run_frozen as rf
    import run_lexikon_1910_pilot as pilot
    import run_r5_german_notes as r5
    from utils.lexikon_1910 import lexikon_1910_institutions as lxin

    abbreviations = lxin.load_abbreviations(ROOT / "reference" / "places" / "institution_abbreviations.csv",
                                            lang=rf._lang(EDITION), volume=VOLUME)
    system = r5.systems(abbreviations, volume=VOLUME)["offices"] + BOX_NOTE
    url, key_env = pilot.READER_ENDPOINTS["gemini"]
    reader = pilot.OpenAICompatibleReader(MODEL, url, key_env, system=system, schema=SCHEMA,
                                          payload_key="marks", schema_name="marks", max_tokens=8000,
                                          temperature=0.0, reasoning_effort=reasoning)
    atlas = Image.open(bo.de_atlas(VOLUME))
    CACHE.mkdir(parents=True, exist_ok=True)
    spent = 0.0
    for name in section_images(pages):
        hit = CACHE / f"{Path(name).stem}_{reasoning}.json"
        if hit.exists():
            print(f"  {name}: cached")
            continue
        crop = Image.open(SECTIONS / name)
        for attempt in range(3):
            try:
                marks, usage = reader([(atlas, r5.ATLAS_CAPTION), (crop, "")], "Transcribe this image.")
                break
            except pilot.Truncated:
                reader.max_tokens = min(reader.max_tokens * 2, 32000)
            except pilot.RateLimited:
                raise
            except Exception as exc:                                    # noqa: BLE001
                if attempt == 2:
                    raise
                print(f"  {name}: {str(exc)[:120]}; retrying")
                time.sleep(3 * (attempt + 1))
        cost = (usage.get("input", 0) * PRICE[0] + usage.get("output", 0) * PRICE[1]) / 1e6
        spent += cost
        hit.write_text(json.dumps({"image": name, "size": list(crop.size), "marks": marks, "usage": usage,
                                   "model": MODEL, "reasoning": reasoning, "cost": cost}), encoding="utf-8")
        print(f"  {name}: {len(marks)} marks, ${cost:.4f}")
    print(f"read: ${spent:.4f} spent this run")


def reader_marks(pages: list[str], reasoning: str) -> list[dict]:
    """Every returned mark: ``{image, page, token, box (section px), lh}``."""
    from PIL import Image

    from build_glyph_bank import line_height

    out = []
    for name in section_images(pages):
        hit = CACHE / f"{Path(name).stem}_{reasoning}.json"
        if not hit.exists():
            raise SystemExit(f"{hit.name} missing: run --stage read first (Heavyhorse6)")
        record = json.loads(hit.read_text(encoding="utf-8"))
        w, h = record["size"]
        lh = line_height(Image.open(SECTIONS / name).convert("L")) or 46
        for m in record["marks"]:
            box = m.get("box_2d") or []
            if len(box) != 4:
                continue
            y0, x0, y1, x1 = box
            out.append({"image": name, "page": name[:4], "token": m.get("token", ""),
                        "box": (round(x0 / 1000 * w), round(y0 / 1000 * h), round(x1 / 1000 * w), round(y1 / 1000 * h)),
                        "lh": lh})
    return out

def classify_marks(marks: list[dict]) -> list[dict]:
    """Each mark's category against the clicks, one to one."""
    import german_click_recall as rec

    clicks = rec.load_clicks()
    by_image = defaultdict(list)
    for i, m in enumerate(marks):
        by_image[m["image"]].append(i)
    for name, idx in by_image.items():
        pts = clicks.get(name, [])
        pairs = []
        for i in idx:
            x0, y0, x1, y1 = marks[i]["box"]
            pad = 0.1 * marks[i]["lh"]
            cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
            for j, (x, y, *_r) in enumerate(pts):
                if x0 - pad <= x <= x1 + pad and y0 - pad <= y <= y1 + pad:
                    pairs.append((math.dist((cx, cy), (x, y)), i, j))
        used_mark, used_click = set(), set()
        for _d, i, j in sorted(pairs):
            if i in used_mark or j in used_click:
                continue
            used_mark.add(i)
            used_click.add(j)
            marks[i]["click"] = pts[j][2]
        for i in idx:
            m = marks[i]
            if "click" in m:
                m["category"] = "right" if m["click"] == m["token"] else "misread"
                continue
            x0, y0, x1, y1 = m["box"]
            cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
            near = [math.dist((cx, cy), (x, y)) for x, y, tok, *_r in pts if tok == m["token"]]
            m["category"] = "box off" if near and min(near) <= 1.5 * m["lh"] else "phantom"
    return marks


def box_report(marks: list[dict], n_clicks: int) -> tuple[dict, list[str]]:
    count = Counter(m["category"] for m in marks)
    box = count["right"] / max(count["right"] + count["box off"], 1)
    lines = [f"  reader marks {len(marks)}: " + ", ".join(f"{k} {count[k]}" for k in ("right", "box off", "misread", "phantom")),
             f"  box bar: right / (right + box off) = {box:.3f}  (bar {BARS['box']})  {'PASS' if box >= BARS['box'] else 'FAIL'}",
             f"  clicked glyphs the reader boxed and named right: {count['right']} of {n_clicks} ({count['right'] / max(n_clicks, 1):.3f})"]
    return {"counts": dict(count), "box": box}, lines


def snap_boxes(marks: list[dict], ink: int) -> tuple[dict, list[str]]:
    """Re-cut every mark's box by the bank's operator (D-177): ``tight_box`` on the ink mask at
    ``ink`` at the integer centre of Gemini's box. The raw box is kept in ``raw_box``; where no ink
    is near, the raw box stays and the mark is counted."""
    from PIL import Image

    import locate_glyphs as lg
    from build_glyph_bank import tight_box

    masks, fallback = {}, 0
    for m in marks:
        if m["image"] not in masks:
            masks[m["image"]] = lg.ink_mask(Image.open(SECTIONS / m["image"]), ink)
        x0, y0, x1, y1 = m["raw_box"] = m["box"]
        box = tight_box(masks[m["image"]], (x0 + x1) // 2, (y0 + y1) // 2, m["lh"])
        if box is None:
            fallback += 1
            continue
        m["box"] = tuple(int(v) for v in box)
    by_cat = Counter(m["category"] for m in marks if m["box"] == m["raw_box"])
    return ({"snapped": len(marks) - fallback, "raw_kept": fallback},
            [f"  snap (D-177): {len(marks) - fallback} of {len(marks)} boxes re-cut by tight_box at ink < {ink}; "
             f"{fallback} kept raw (no ink near the centre){': ' + dict(by_cat).__repr__() if fallback else ''}"])

def negative_glyphs(pages: list[str], ink: int, clear_lines: float = 0.5) -> list:
    """Candidate clusters on the clicked sections that hold no click and lie off every click:
    the reject class. The strips were clicked exhaustively, so these are not pictograms."""
    from PIL import Image

    import german_click_recall as rec
    import locate_glyphs as lg
    import run_glyph_e3 as e3
    from build_glyph_bank import line_height

    clicks = rec.load_clicks()
    orig = lg.ink_mask
    lg.ink_mask = lambda image, threshold=ink: orig(image, threshold)
    out = []
    try:
        for name in section_images(pages):
            image = Image.open(SECTIONS / name)
            lh = line_height(image.convert("L")) or 46
            for k, c in enumerate(lg.candidates_on_image(image, name, lh)):
                if c.bracket or lg.is_speck(c) or c.features["log_w"] > math.log(lg.MAX_GLYPH_WIDTH):
                    continue
                pad = clear_lines * lh
                if any(c.box[0] - pad <= x <= c.box[2] + pad and c.box[1] - pad <= y <= c.box[3] + pad
                       for x, y, *_r in clicks.get(name, [])):
                    continue
                out.append(e3.Glyph(f"de{name[:4]}", "negative", "offices", "", 0, "", k, NOT_GLYPH, name,
                                    c.box, lh, SECTIONS, "negative"))
    finally:
        lg.ink_mask = orig
    return out


def judge(marks: list[dict], encoder: str, download: bool, ink: int, rounds: int) -> tuple[dict, list[str]]:
    import numpy as np

    import locate_glyphs as lg
    import run_e3_pages as e3p
    import run_german_oracle as oracle
    import run_glyph_e3 as e3

    config = e3.OF_RECORD if encoder == "dinov3" else e3.FLOOR
    data = lg.load_sections()
    located, _ = lg.locate_all(data, rounds)
    bank = e3.placed_glyphs(located)
    namer = e3p.Namer(bank, config, download)

    def embed(glyphs):
        got = e3.embed(glyphs, [namer.name], download, cuts=(namer.variant,))
        return got[(namer.name, namer.variant)][namer.feature]

    pages = sorted({m["page"] for m in marks})
    positives, _kept, _skipped = oracle.click_glyphs(oracle.read_clicks(), "p1", ink)
    negatives = negative_glyphs(pages, ink)
    queries = [e3.Glyph(f"de{m['page']}", "query", "offices", "", 0, "", i, m["token"], m["image"],
                        m["box"], m["lh"], SECTIONS, "reader") for i, m in enumerate(marks)]
    pos_x, neg_x, q_x = embed(positives), embed(negatives), embed(queries)
    pos_t, pos_p = [g.base for g in positives], [g.scan for g in positives]
    neg_p = [g.scan for g in negatives]
    cz_t, cz_p = [g.base for g in bank], [g.scan for g in bank]

    verdicts = [None] * len(marks)
    thresholds = {}
    for held in sorted({q.scan for q in queries}):
        keep_pos = [i for i, p in enumerate(pos_p) if p != held]
        keep_neg = [i for i, p in enumerate(neg_p) if p != held]
        bank_x = np.vstack([namer.bank_x, pos_x[keep_pos], neg_x[keep_neg]])
        bank_t = cz_t + [pos_t[i] for i in keep_pos] + [NOT_GLYPH] * len(keep_neg)
        bank_p = cz_p + [pos_p[i] for i in keep_pos] + [neg_p[i] for i in keep_neg]
        own = e3.classify(pos_x[keep_pos], [pos_p[i] for i in keep_pos], bank_x, bank_t, bank_p, namer.k)
        right = [s for (g, s), i in zip(own, keep_pos) if g == pos_t[i]]
        tau = float(np.quantile(right, e3.ABSTAIN_QUANTILE)) if right else -1.0
        thresholds[held] = round(tau, 3)
        idx = [i for i, q in enumerate(queries) if q.scan == held]
        for (name, s), i in zip(e3.classify(q_x[idx], [held] * len(idx), bank_x, bank_t, bank_p, namer.k), idx):
            base = lg.base_of(marks[i]["token"])
            verdicts[i] = ("not judged" if s < tau else "agree" if name == base else "disagree", name, s)
    for m, v in zip(marks, verdicts):
        m["verdict"], m["dino"], m["similarity"] = v

    def share(category_set, verdict):
        rows = [m for m in marks if m["category"] in category_set and m["verdict"] != "not judged"]
        return (sum(m["verdict"] == verdict for m in rows) / len(rows)) if rows else None, len(rows)

    right_rows = [m for m in marks if m["category"] == "right"]
    cover = sum(m["verdict"] != "not judged" for m in right_rows) / max(len(right_rows), 1)
    keep, n_keep = share({"right"}, "agree")
    catch, n_false = share({"misread", "phantom"}, "disagree")
    off_agree, n_off = share({"box off"}, "agree")
    result = {"encoder": list(config), "thresholds": thresholds, "cover": cover, "keep": keep,
              "catch": catch, "judged_false": n_false, "box_off_agree": off_agree,
              "verdicts": {f"{c}/{v}": n for (c, v), n in sorted(Counter((m["category"], m["verdict"]) for m in marks).items())}}

    def bar(value, name):
        return "—" if value is None else ("PASS" if value >= BARS[name] else "FAIL")
    lines = [f"  DINOv3 judging ({encoder} {config}); thresholds by held-out page {thresholds}",
             f"  cover: judged share of right claims {cover:.3f} (bar {BARS['cover']})  {bar(cover, 'cover')}",
             f"  keep:  agreement among {n_keep} judged right claims {keep if keep is None else round(keep, 3)} "
             f"(bar {BARS['keep']})  {bar(keep, 'keep')}",
             f"  catch: disagreement among {n_false} judged false claims {catch if catch is None else round(catch, 3)} "
             f"(bar {BARS['catch']}, judged only with >= {MIN_FALSE})  "
             + (bar(catch, 'catch') if n_false >= MIN_FALSE else "not judged: too few false claims"),
             f"  box off (true claims, badly boxed): agreed {off_agree if off_agree is None else round(off_agree, 3)} of {n_off} judged",
             "  category x verdict: " + ", ".join(f"{k} {n}" for k, n in result["verdicts"].items())]
    return result, lines


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--stage", choices=("read", "boxes", "judge", "all"), default="all")
    parser.add_argument("--encoder", choices=("pixels", "dinov3"), default="dinov3")
    parser.add_argument("--reasoning", default="none", help="Gemini reasoning_effort (the route's: none)")
    parser.add_argument("--download", action="store_true")
    parser.add_argument("--ink", type=int, default=128)
    parser.add_argument("--rounds", type=int, default=4)
    parser.add_argument("--snap", action="store_true", help="D-177: re-cut each box by tight_box before judging")
    args = parser.parse_args(argv)

    pages = clicked_pages()
    if args.stage in ("read", "all"):
        read(pages, args.reasoning)
    if args.stage == "read":
        return 0
    import german_click_recall as rec

    n_clicks = sum(len(v) for v in rec.load_clicks().values())
    marks = classify_marks(reader_marks(pages, args.reasoning))
    summary, lines = box_report(marks, n_clicks)
    lines = [f"Reader boxes: {MODEL}, reasoning {args.reasoning}, pages {pages}"] + lines
    if args.snap:
        summary["snap"], more = snap_boxes(marks, args.ink)
        lines += more
    if args.stage in ("judge", "all"):
        result, more = judge(marks, args.encoder, args.download, args.ink, args.rounds)
        summary["judge"] = result
        lines += more
    text = "\n".join(lines)
    print(text)
    OUT.mkdir(parents=True, exist_ok=True)
    stem = (f"reader_boxes_{args.reasoning}" + (f"_{args.encoder}" if args.stage in ("judge", "all") else "")
            + ("_snap" if args.snap else ""))
    (OUT / f"{stem}.json").write_text(json.dumps({"summary": summary, "marks": marks}, indent=1, default=str), encoding="utf-8")
    (OUT / f"{stem}.txt").write_text(text + "\n", encoding="utf-8")
    return 0
