"""
panel_io.py — writing one table to ``data/final/panels`` with its key asserted
==============================================================================
Every table notebook 04 writes goes through :func:`write_panel`: the declared key is checked
unique and never missing **before** anything is written, the table is written as parquet and as
a UTF-8 CSV with LF line ends, and the parquet is read back to confirm the row count.
"""
from __future__ import annotations

from pathlib import Path

import pandas as pd

__version__ = "1.0.0"   # 2026-11-18 (D-260)


def assert_key(table: pd.DataFrame, key: list[str], name: str) -> None:
    """Raise if ``key`` is absent, missing anywhere, or not unique in ``table``."""
    absent = [k for k in key if k not in table.columns]
    if absent:
        raise KeyError(f"{name}: key column(s) {absent} absent")
    na = int(table[key].isna().any(axis=1).sum())
    dup = int(table.duplicated(key).sum())
    if na or dup:
        raise ValueError(f"{name}: key {key} has {na} missing and {dup} duplicate row(s)")


def write_panel(table: pd.DataFrame, folder: Path, name: str, key: list[str],
                csv: bool = True) -> int:
    """Write ``folder/name.parquet`` (and ``.csv``, LF) after asserting ``key``; return the rows."""
    assert_key(table, key, name)
    folder = Path(folder)
    folder.mkdir(parents=True, exist_ok=True)
    table.to_parquet(folder / f"{name}.parquet", index=False)
    if csv:
        table.to_csv(folder / f"{name}.csv", index=False, encoding="utf-8", lineterminator="\n")
    back = pd.read_parquet(folder / f"{name}.parquet", columns=key)
    if len(back) != len(table):
        raise AssertionError(f"{name}: {len(back):,} rows read back, {len(table):,} written")
    print(f"wrote panels/{name}: {len(table):,} rows, key {key}")
    return len(table)
