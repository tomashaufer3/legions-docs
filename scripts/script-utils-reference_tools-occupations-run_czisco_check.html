How it works

What this check adds that the other two cannot. The German link bridge and OccCANINE both begin from a German gloss this project wrote, so neither can test the step that actually carries the risk: whether the Czech word means what we said it means. CZ-ISCO is Czech→ English under one code, authored by ČSÚ, so it tests exactly that step and nothing else.

What it cannot do, and the write-up must keep saying so:

  • It is the 2025 classification. The historical core of this corpus is absent at every level — rolník, hajný, příručí, tkadlec, mlynář, kolář, bednář, cihlář, havíř, soustružník. Roughly two thirds of records are reachable; the words that carry the biggest coding decisions are not.
  • It cannot corroborate a residual. Where our code is HSN's "no further information" (99900/99910/99920) or a sentinel, no ISCO code can agree or disagree, because ISCO has no residual category. Those keys are scored n/a rather than counted as failures.
  • It reaches no HISCO code. See build_czisco_bridge.py for why the ISCO-08 → 88 → 68 chain is not attempted.

def best_match(key: str, b: pd.DataFrame) -> tuple[pd.Series, int] | tuple[None, int]:
    """The most on-point CZ-ISCO row whose head noun is an inflection of ``key``.

    Ranked by, in order: whether the head carries a qualifying adjective, how many
    occupations the label names, how long it is, and finally specificity of level. That
    ordering is what puts ``malíř`` on ``Malíři interiérů`` rather than ``Umělečtí
    malíři`` — both have head *malíři*, but the qualified one is a different trade — and
    what prefers ``Zedníci`` (71121, a single token once its "(kromě …)" exclusion is
    stripped) over ``Kamnáři, zedníci ohnivzdorného zdiva``.

    Also returns how many codes matched at all, so an ambiguous key is visible as such.
    """
    hit = b[b["heads"].map(lambda hs: any(matches_head(key, h) for h in hs))]
    if hit.empty:
        return None, 0
    hit = hit.copy()
    hit["qualified"] = ~hit["bare"].map(lambda hs: any(matches_head(key, h) for h in hs))
    ranked = hit.sort_values(["qualified", "n_heads", "n_tok", "level"],
                             ascending=[True, True, True, False])
    return ranked.iloc[0], len(hit)

def check() -> pd.DataFrame:
    cur = pd.read_csv(CURATION, dtype=str).fillna("")
    hisco_before = cur["hisco"].copy()
    b = load_bridge()
    label_of = _our_labels()

    coded = cur[cur["hisco"].str.strip() != ""].copy()
    print(f"  {len(coded):,} curated keys with a code; {len(b):,} CZ-ISCO codes")

    rows = {}
    for i, r in coded.iterrows():
        m, n_hits = best_match(r["key"], b)
        if m is None:
            continue
        ours = label_of.get(r["hisco"], "")
        # A residual cannot be corroborated: ISCO has no "no further information" group, so
        # `dělník` -> 99900 meeting `Kvalifikovaní stavební dělníci` is neither agreement
        # nor disagreement. Score it n/a rather than let it depress the headline.
        residual = r["hisco"].startswith("999") or r["hisco"].startswith("-")
        hb, ib = _band(r["hisco"], m["code"])
        if residual:
            agree = "n/a"
        else:
            overlap = bool(_content(ours) & _content(m["en"]))
            agree = "1" if overlap else ("band" if (hb and hb == ib) else "0")
        rows[i] = dict(czisco_code=m["code"], czisco_cs=m["cs"], czisco_en=m["en"],
                       czisco_isco08=m["isco08_4d"] if pd.notna(m["isco08_4d"]) else "",
                       czisco_agree=agree, czisco_n_hits=str(n_hits))

    add = pd.DataFrame.from_dict(rows, orient="index")
    # Blank the whole family first. Writing only the rows that matched would leave a key
    # that *stopped* matching — because the rule was tightened, say — holding the hit it
    # got on an earlier run, and the report would still show it. Derived columns must be
    # rebuilt, never patched.
    for col in ("czisco_code", "czisco_cs", "czisco_en", "czisco_isco08", "czisco_agree",
                "czisco_n_hits"):
        cur[col] = ""
        if len(add):
            cur.loc[add.index, col] = add[col]

    assert cur["hisco"].equals(hisco_before), "run_czisco_check must never change `hisco`"
    _report(cur, add)
    return cur

def tail_candidates() -> pd.DataFrame:
    """CZ-ISCO hits among the enrolment strings nothing currently codes.

    Emitted for hand adjudication only. Accepted rows are hand-entered into
    ``czech_glosses.py``'s ``HISCO_OVERRIDE`` like every other code, and the HISCO code is
    still read off ``hisco_authority.csv`` — never off CZ-ISCO.
    """
    from utils.parsing.occupations import _norm

    final = ROOT / "data" / "final" / "persons" / "legionnaires_clean.parquet"
    if not final.exists():
        raise SystemExit(f"{final} not found — run notebook 02 first.")
    d = pd.read_parquet(final, columns=["occupation", "occupation_hisco"])
    un = d.loc[d["occupation"].notna() & d["occupation_hisco"].isna(), "occupation"]
    vc = un.value_counts()
    print(f"  uncoded tail: {len(un):,} records over {len(vc):,} distinct strings")

    b = load_bridge()
    out = []
    for raw, c in vc.items():
        key = _norm(raw)
        if not key:
            continue
        m, n_hits = best_match(key, b)
        if m is None:
            continue
        out.append(dict(raw=raw, key=key, n=c, czisco_code=m["code"],
                        czisco_cs=m["cs"], czisco_en=m["en"],
                        czisco_isco08=m["isco08_4d"], czisco_n_hits=n_hits,
                        accept="", hisco="", note=""))
    df = pd.DataFrame(out).sort_values("n", ascending=False, ignore_index=True)
    print(f"  CZ-ISCO proposes a reading for {len(df):,} strings / "
          f"{int(df['n'].sum()):,} records")
    print("\n  top candidates (nothing is applied — `accept` and `hisco` are yours to fill):")
    for r in df.head(20).itertuples():
        print(f"    {r.n:>4}  {r.raw[:24]:<24} -> {r.czisco_code:<6} "
              f"{r.czisco_cs[:32]:<32} = {str(r.czisco_en)[:40]}")
    return df

run_czisco_check.py — CZ-ISCO as a third coder, on the Czech side
=================================================================

official Czech-statistical-office English label beside our HISCO label, so the two can be

BRIDGE = ROOT / "reference" / "occupations" / "czisco_authority.csv"

TAIL_OUT = ROOT / "reference" / "occupations" / "czisco_tail_candidates.csv"

# Words that carry no occupational content in either scheme's label style. Dropping them
# before comparing is what lets "Shoemakers and related workers" meet "Shoe-maker, General".
_STOP = {
    "and", "or", "of", "the", "in", "not", "elsewhere", "classified", "nec", "other",
    "others", "related", "worker", "workers", "general", "specialisation", "specialization",
    "unknown", "similar", "trades", "occupations", "professional", "professionals",
    "associate", "n", "e", "c", "except", "including", "etc",
}

# HISCO major group -> a band that is comparable with ISCO-08. The raw major digits are
# **not** comparable: HISCO 7/8/9 are all production workers, while ISCO-08 splits the same
# people across 7 craft, 8 operators, 9 elementary, and HISCO 4/5 (sales, service) both fall
# in ISCO-08's single major 5. Only these coarse bands mean the same thing on both sides.
_HISCO_BAND = {"0": "prof", "1": "prof", "2": "manage", "3": "clerk",
               "4": "sales", "5": "sales", "6": "agri",
               "7": "manual", "8": "manual", "9": "manual"}
_ISCO_BAND = {"1": "manage", "2": "prof", "3": "prof", "4": "clerk", "5": "sales",
              "6": "agri", "7": "manual", "8": "manual", "9": "manual", "0": "protect"}


def _band(hisco: str, isco: str) -> tuple[str | None, str | None]:
    """Coarse bands for our HISCO code and the CZ-ISCO code, or ``None`` where not scoreable."""
    h = str(hisco).strip()
    i = str(isco).strip()
    hb = "protect" if h.startswith("58") else _HISCO_BAND.get(h[:1])
    ib = "protect" if (i.startswith("0") or i.startswith("541")) else _ISCO_BAND.get(i[:1])
    return hb, ib


def _content(label: str) -> set[str]:
    """Content-word stems of an English occupational label, singularised crudely."""
    words = re.findall(r"[a-z]+", str(label).replace("-", "").lower())
    return {w[:-1] if len(w) > 4 and w.endswith("s") else w
            for w in words if w not in _STOP} - _STOP

def _our_labels() -> dict[str, str]:
    """HISCO code -> our English label, at default modifiers."""
    a = pd.read_csv(AUTHORITY, dtype={"hisco": str})
    base = a[(a[["status", "relation", "product"]].astype(str) == "-9").all(axis=1)]

def _report(cur: pd.DataFrame, add: pd.DataFrame) -> None:
    n = pd.to_numeric(cur["n"], errors="coerce")
    coded = cur["hisco"].str.strip() != ""
    matched = cur["czisco_code"].str.strip() != ""
    print(f"\n  CZ-ISCO reached {int((coded & matched).sum()):,} of {int(coded.sum()):,} "
          f"coded keys — {int(n[coded & matched].sum()):,} of "
          f"{int(n[coded].sum()):,} records ({n[coded & matched].sum() / n[coded].sum():.1%})")

    print("\n  agreement on the English reading")
    for lab, val in (("label words overlap", "1"), ("same broad band only", "band"),
                     ("disagree", "0"), ("residual, not comparable", "n/a")):
        m = cur["czisco_agree"].eq(val)
        if m.any():
            print(f"    {lab:<26} {int(m.sum()):>4} keys   {int(n[m].sum()):>7,} records")

    dis = cur[cur["czisco_agree"].eq("0")].assign(n_i=n).nlargest(20, "n_i")
    if len(dis):
        print("\n  largest disagreements — read these, they are the point of the check:")
        for r in dis.itertuples():
            print(f"    {int(r.n_i):>6,}  {r.key:<20} ours={r.hisco:<6} "
                  f"{str(r.czisco_code):<6} {r.czisco_cs[:30]:<30} = {str(r.czisco_en)[:38]}")


def tail_candidates() -> pd.DataFrame:
    """CZ-ISCO hits among the enrolment strings nothing currently codes.

    Emitted for hand adjudication only. Accepted rows are hand-entered into
    ``czech_glosses.py``'s ``HISCO_OVERRIDE`` like every other code, and the HISCO code is
    still read off ``hisco_authority.csv`` — never off CZ-ISCO.
    """
    from utils.parsing.occupations import _norm

    final = ROOT / "data" / "final" / "persons" / "legionnaires_clean.parquet"
    if not final.exists():
        raise SystemExit(f"{final} not found — run notebook 02 first.")
    d = pd.read_parquet(final, columns=["occupation", "occupation_hisco"])
    un = d.loc[d["occupation"].notna() & d["occupation_hisco"].isna(), "occupation"]
    vc = un.value_counts()
    print(f"  uncoded tail: {len(un):,} records over {len(vc):,} distinct strings")

    b = load_bridge()
    out = []
    for raw, c in vc.items():
        key = _norm(raw)
        if not key:
            continue
        m, n_hits = best_match(key, b)
        if m is None:
            continue
        out.append(dict(raw=raw, key=key, n=c, czisco_code=m["code"],
                        czisco_cs=m["cs"], czisco_en=m["en"],
                        czisco_isco08=m["isco08_4d"], czisco_n_hits=n_hits,
                        accept="", hisco="", note=""))
    df = pd.DataFrame(out).sort_values("n", ascending=False, ignore_index=True)
    print(f"  CZ-ISCO proposes a reading for {len(df):,} strings / "
          f"{int(df['n'].sum()):,} records")
    print("\n  top candidates (nothing is applied — `accept` and `hisco` are yours to fill):")
    for r in df.head(20).itertuples():
        print(f"    {r.n:>4}  {r.raw[:24]:<24} -> {r.czisco_code:<6} "
              f"{r.czisco_cs[:32]:<32} = {str(r.czisco_en)[:40]}")
    return df


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[1])
    ap.add_argument("--tail", action="store_true",
                    help="also emit candidates for the uncoded enrolment tail")
    args = ap.parse_args(argv)

    print("CZ-ISCO third-coder check …")
    cur = check()
    cur.to_csv(CURATION, index=False, encoding="utf-8")
    print(f"\nUpdated {CURATION.relative_to(ROOT)}")

    if args.tail:
        print("\nTail candidates …")
        tail_candidates().to_csv(TAIL_OUT, index=False, encoding="utf-8")
        print(f"Wrote {TAIL_OUT.relative_to(ROOT)}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
