def extract_theater(unit) -> str:
    """Regex classifier on a unit / transfer-unit string (Pass 1 & 4).

    Substring matching is CASE-SENSITIVE by design — that is what keeps the two
    fragile short tokens safe on the observed data: ``"RA"`` (Russia) does not fire
    inside lowercase words, and ``"Fra"`` (France) is checked before the Russia
    block so "Francie" never reaches ``"RA"``. If the source strings ever change
    shape (e.g. an ALL-CAPS re-scrape), re-validate these two rules first. Note
    also that Pass 2 (regiment-mate majority vote) PROPAGATES any Pass-1 error to
    every unknown-theatre soldier of that regiment — QA new tokens against the
    residual report before adding them.
    """
    if pd.isna(unit):
        return "Unknown"
    s = str(unit)
    if any(x in s for x in ["Itálii", "Itá", "Italská", "( Li )", "Li )"]):
        return "Italy"
    if any(x in s for x in ["Cizinecké legie", "Fra", "( CL )", "Armée polonaise"]):
        return "France"
    # "Polská" (the Polish Legion) is deliberately NOT here any more. It sent all 41 Polish-
    # Legion records to Serbia/Balkans, whose units are Haller's Armée polonaise (France) and
    # Russian formations (Dutova armáda, Krymský pl.); none is Balkan. Those records now take
    # whatever the other cues and passes say, and Unknown where nothing does.
    if any(x in s for x in ["Srbská", "Rumunská"]):
        return "Serbia/Balkans"
    if any(x in s for x in ["Rus", "Kornilo", "RA", "Česká družina",
                            "( Lr )", "Lr )"]):
        return "Russia"
    if any(x in s for x in ["Britská", "Kanadská", "Americká", "Australská",
                            "Fusiliers", "K.O.Y.L", "Lincolns", "Battalion"]):
        return "Britain/Dominions"
    if "ČSA" in s:
        return "Post-war (ČSA)"
    return "Unknown"

def theater_selftest() -> list[str]:
    """Behaviours that shipped wrong once; empty list means all hold."""
    fails = [f"extract_theater({u!r}) = {extract_theater(u)!r}, expected {want!r}"
             for u, want in _SELFTEST if extract_theater(u) != want]
    for city in ("Chicago", "Cleveland", "USA"):
        if CITY_THEATER.get(city) != "France":
            fails.append(f"CITY_THEATER[{city!r}] = {CITY_THEATER.get(city)!r}, expected 'France'")
    return fails

def add_theater_column(df: pd.DataFrame,
                       unit_col: str = "Jednotka v době zařazení",
                       app_col: str = "Misto podání přihlášky do legií",
                       transfer_col: str = "Převelen k útvaru",
                       ) -> pd.DataFrame:
    """Append the four-pass ``theater`` column (not modified in place).

    Args:
        df:           input DataFrame (raw Czech columns)
        unit_col:     enrolment-unit column (Pass 1 + the regiment key of Pass 2)
        app_col:      application-place column (Pass 3)
        transfer_col: transfer-unit column (Pass 4)

    Returns:
        Copy of *df* with a ``theater`` column.
    """
    df = df.copy()

    # Pass 1 — regex on enrolment unit.
    df["theater"] = df[unit_col].apply(extract_theater)

    # Pass 2 — regiment-mate majority vote for residual unknowns.
    regiment_key = df[unit_col].str.extract(r"^([^(]+)")[0].str.strip()
    regiment_theater = (
        df[df["theater"] != "Unknown"]
        .groupby(regiment_key[df["theater"] != "Unknown"])["theater"]
        .agg(lambda x: x.value_counts().index[0])
    )
    unknown_mask = df["theater"] == "Unknown"
    df.loc[unknown_mask, "theater"] = regiment_key[unknown_mask].map(regiment_theater)
    df["theater"] = df["theater"].fillna("Unknown")

    # Pass 3 — application city.
    unknown_mask = df["theater"] == "Unknown"
    df.loc[unknown_mask, "theater"] = (
        df.loc[unknown_mask, app_col].map(CITY_THEATER)
    )
    df["theater"] = df["theater"].fillna("Unknown")

    # Pass 4 — transfer unit (same regex as Pass 1).
    unknown_mask = df["theater"] == "Unknown"
    df.loc[unknown_mask, "theater"] = (
        df.loc[unknown_mask, transfer_col].apply(extract_theater)
    )
    df["theater"] = df["theater"].fillna("Unknown")
    return df

def theater_report(df: pd.DataFrame) -> None:
    """Print the theatre distribution."""
    if "theater" not in df.columns:
        print("No theater column. Run add_theater_column() first.")
        return
    n = len(df)
    print(f"Theatre distribution ({n:,} records):")
    for t, c in df["theater"].value_counts().items():
        print(f"  {t:<18} {c:>7,}  ({c/n:.1%})")

    residual report before adding them.

_SELFTEST: tuple[tuple[str, str], ...] = (
    ("33.p.pl. ( Francouská armáda )", "France"),
    ("Armée polonaise ( Polská legie )", "France"),
    ("( Polská legie )", "Unknown"),
    ("1.stř.pl. ( Čs. legie v Rusku )", "Russia"),
    ("Royal Fusiliers (R.F.) ( Britská armáda )", "Britain/Dominions"),
