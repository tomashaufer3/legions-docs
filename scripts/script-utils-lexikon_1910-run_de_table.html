How it works

The Czech table pass (run_lexikon_1910_pilot.py channel A, driven by run_table_arm.py and the frozen route) crops each municipality block where the ALTO found it and has a vision model read its columns. The German volumes have no ALTO and an OCR layer whose names are garbage, so the blocks are cut from the image (lexikon_1910_de_table) and the model reads everything, names included. The readers are the Czech table arm's two, with its settings:

gemini38   Gemini 3.8 Flash, thinking off, temperature 0     the reader of record
haiku55    Claude Haiku 5.5, thinking adaptive at high        the second reader

One crop a block, frame to frame, scaled to the 2,576 px vision ceiling (one tile, as the Czech default); the prompt is the Czech one with the German volume's title, columns and the rules for its two name forms (lexikon_1910_de_table.build_system). A block's reply is cached under a key of the crop, the instruction, the model, the prompt and every reading setting, so a batch (reader_batches.py) fills the cache and the next run reads from it, and a rerun spends nothing.

Writes table_rows_&lt;label&gt;.csv (channel A, one row per printed line, keyed by the block, with the block's kind and its district) and table_checks_&lt;label&gt;.csv (the volume's identities on every row, and each municipality's parts against it) to the German pass folder, where bakeoff.find_pass_file finds them for the register. score_table_passes.py --german scores a pass against the German gold tables.

The frozen route calls frozen_pass. By hand, on Heavyhorse6 (D-68):

python utils/lexikon_1910/run_de_table.py --dry-run --volume boehmen_de --pages 142,276
python utils/lexikon_1910/run_de_table.py --volume boehmen_de --pages gold --reader gemini38

def make_reader(arm: Arm, profile):
    """The pilot's reader classes with the German prompt and the volume's row schema."""
    system, schema = lxdt.build_system(profile), lxdt.row_schema(profile)
    if arm.route == "claude":

def cache_key(crop, instruction: str, reader) -> str:
    """The Czech table cache's key (``run_lexikon_1910_pilot``, D-115): the crop's bytes, the
    instruction, the model, the prompt and every reading setting."""
    h = hashlib.sha256()
    for piece in (pilot.encode_png(crop) + "\0", instruction, str(getattr(reader, "model", "")),
                  str(getattr(reader, "system", "")), str(getattr(reader, "temperature", "")),
                  str(getattr(reader, "reasoning_effort", "") or ""),
                  str(getattr(reader, "thinking", "")), str(getattr(reader, "effort", ""))):
        h.update(piece.encode("utf-8"))
        h.update(bytes(1))
    return h.hexdigest()

def cut(pages: list[int], profile) -> list[tuple]:
    """``[(block, crop), …]`` in reading order, and the blocks by page for the report."""
    jobs, by_page = [], {}
    for scan in pages:
        stem = f"{scan:04d}"
        image = lxdn.render_native(profile.source_pdf, scan)
        geo, blocks = lxdt.page_blocks(image, profile, stem)
        by_page[stem] = blocks
        jobs += [(block, lxdt.crop_block(image, geo, block)) for block in blocks]
    return jobs, by_page

def read_blocks(jobs: list[tuple], reader, cache: Path, queue: Path | None = None,
                workers: int = WORKERS) -> tuple[list, Counter]:
    """``[(block, rows or None), …]`` and the usage. With ``queue`` every block not in the
    cache is written there as a batch line (``reader_batches.py``) and none is sent."""
    cache.mkdir(parents=True, exist_ok=True)
    lock, usage = threading.Lock(), Counter()

    def one(job):
        block, crop = job
        instruction = lxdt.INSTRUCTIONS[block.kind]
        hit = cache / f"{cache_key(crop, instruction, reader)}.json"
        if hit.exists():
            record = json.loads(hit.read_text(encoding="utf-8"))
            return block, record["lines"], {"cached": 1}
        if queue is not None:
            make = getattr(reader, "params", None) or getattr(reader, "body")
            line = {"key": hit.stem, "reader": reader.name, "cache_dir": str(cache),
                    "slot": [block.page, block.block_id], "kind": "table",
                    "model": getattr(reader, "model", ""), "payload_key": "rows",
                    "request": make([(crop, "")], instruction)}
            with lock, Path(queue).open("a", encoding="utf-8") as fh:
                fh.write(json.dumps(line) + "\n")
            return block, None, {"queued": 1}
        for attempt in range(RETRIES):
            try:
                rows, spent = reader([(crop, "")], instruction)
                tmp = hit.with_suffix(".json.part")
                tmp.write_text(json.dumps({"lines": rows, "usage": spent,
                                           "slot": [block.page, block.block_id], "kind": "table",
                                           "model": getattr(reader, "model", ""),
                                           "at": time.time()}), encoding="utf-8")
                tmp.replace(hit)
                return block, rows, spent
            except pilot.Truncated:
                reader.max_tokens = min(getattr(reader, "max_tokens", 8000) * 2, 32000)
            except pilot.RateLimited:
                raise
            except Exception as exc:                                    # noqa: BLE001
                if attempt == RETRIES - 1:
                    return block, None, {"error": str(exc)[:300]}
                time.sleep(2 * (attempt + 1))
        return block, None, {"error": "truncated"}

    with ThreadPoolExecutor(max_workers=workers) as pool:
        results = list(pool.map(one, jobs))
    out = []
    for block, rows, spent in results:
        for key, value in spent.items():
            usage[key] += value if isinstance(value, (int, float)) else 1
        out.append((block, rows))
    return out, usage

def assemble(name: str, read: list[tuple], profile, crosswalk: dict, out: Path) -> dict:
    """Label every block with its district, check it, write the pass's two files."""
    labels = lxdt.court_labels([(b, rows or []) for b, rows in read])
    page_reads, checks = [], []
    for (block, rows), (okres, court) in zip(read, labels):
        court_cs = lxdn.court_name(court, crosswalk)[0] if court else ""
        page_reads.append((block, rows or [], okres, court, court_cs))
        for c in lxdt.grade_block(rows or [], profile, block.kind):
            checks.append({"page": block.page, "block_id": block.block_id,
                           "block_kind": block.kind, **c})
    rows = lxdt.table_rows(page_reads, profile)
    _write(out / f"table_rows_{name}.csv", rows)
    _write(out / f"table_checks_{name}.csv", checks)
    verdicts = Counter(c["verdict"] for c in checks)
    return {"blocks": len(read), "unread": sum(r is None for _b, r in read), "rows": len(rows),
            **{f"check_{k}": v for k, v in verdicts.items()}}

def frozen_pass(label: str, arm_name: str, pages: list[int], profile_name: str,
                out: Path | None = None, queue: Path | None = None) -> bool:
    """One table pass of the frozen route over any pages of one German volume. The runner that
    calls it (``run_frozen.py``) chooses the pages and keeps certification apart. ``False``
    if a quota stopped it; a rerun goes on from the cache."""
    out = out or bo.DE_PASS_DIR
    profile = lxvol.load_profile(profile_name)
    arm = ARMS[arm_name]
    print("\n" + "=" * 78 + f"\nGERMAN TABLE (frozen)  {label}   ({arm.model}, {profile.title}, "
          f"{len(pages)} pages)\n" + "=" * 78)
    started = time.time()
    jobs, by_page = cut(pages, profile)
    print("  " + lxdt.lexikon_1910_de_table_report(by_page, profile))
    reader = make_reader(arm, profile)
    try:
        read, usage = read_blocks(jobs, reader, out / "table_cache" / label, queue=queue)
    except pilot.RateLimited as exc:
        print(f"  rate limited: {exc}; run again to go on from the cache")
        return False
    if queue is not None:
        print(f"  {usage.get('queued', 0)} blocks queued to {Path(queue).name}, "
              f"{usage.get('cached', 0)} already in the cache")
        return True
    counts = assemble(label, read, profile, lxdn.load_court_crosswalk(), out)
    cost = (usage.get("input", 0) * arm.price[0] + usage.get("output", 0) * arm.price[1]) / 1e6
    print(f"  {counts}; {usage.get('cached', 0)} from the cache; ${cost:.2f}; "
          f"{(time.time() - started) / 60:.1f} min")
    return counts["unread"] == 0

def estimate(pages: list[int], profile, arm: Arm) -> str:
    """Blocks and dollars for one pass over ``pages``; renders and cuts, sends nothing."""
    jobs, by_page = cut(pages, profile)
    tokens = [min(pilot_image_cap(arm), crop.width * crop.height // 750) for _b, crop in jobs]
    out_tokens = EST_OUTPUT + (pilot.THINKING_TOKENS.get(arm.effort, 800) if arm.thinking == "adaptive" else 0)
    dollars = sum((t + 1500) * arm.price[0] + out_tokens * arm.price[1] for t in tokens) / 1e6
    per_page = len(jobs) / max(1, len(pages))
    return (lxdt.lexikon_1910_de_table_report(by_page, profile)
            + f"\n  {arm.model}: ~${dollars:.2f} for these {len(pages)} pages "
              f"({per_page:.1f} blocks a page; the volume's {profile.table_last - profile.table_first + 1} "
              f"table pages ~${dollars / max(1, len(pages)) * (profile.table_last - profile.table_first + 1):,.0f}, "
              f"half that batched)")

def page_spec(spec: str, profile) -> list[int]:
    """``gold`` (Böhmen's calibration and purposive pages), ``calibration`` (Mähren's or
    Schlesien's drawn pages, D-154), or ``142,276`` / ``21-30``. Never a sealed page."""
    if spec == "gold":
        pages = bo.de_gold_pages()
    elif spec == "calibration":
        path = ROOT / "reference" / "places" / f"gold_sample_pages_{profile.slug}.csv"
        with path.open(encoding="utf-8", newline="") as fh:
            pages = sorted(int(r["scan"]) for r in csv.DictReader(fh) if r["role"] == "calibration")
    else:
        pages = []
        for part in spec.split(","):
            first, _, last = part.partition("-")
            pages += range(int(first), int(last or first) + 1)
    sealed = set(profile.sealed_pages)
    assert not sealed & set(pages), f"sealed pages asked for: {sorted(sealed & set(pages))}"
    return pages

def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n")[0])
    ap.add_argument("--volume", default="boehmen_de",
                    choices=("boehmen_de", "maehren_de", "schlesien_de"))
    ap.add_argument("--pages", default="gold")
    ap.add_argument("--reader", default="gemini38", choices=sorted(ARMS))
    ap.add_argument("--label", default="", help="pass label; default de_table_<reader>_<volume>")
    ap.add_argument("--dry-run", action="store_true", help="cut and estimate; send nothing")
    args = ap.parse_args(argv)
    profile = lxvol.load_profile(args.volume)
    pages = page_spec(args.pages, profile)
    if args.dry_run:
        print(estimate(pages, profile, ARMS[args.reader]))
        return 0
    label = args.label or f"de_table_{args.reader}_{args.volume}"
    return 0 if frozen_pass(label, args.reader, pages, args.volume) else 2

    return pilot.OpenAICompatibleReader(arm.model, url, key_env, system=system, schema=schema,
                                        payload_key="rows", schema_name="block",
                                        max_tokens=arm.max_tokens, temperature=0.0,
                                        reasoning_effort=arm.reasoning or None)


def cache_key(crop, instruction: str, reader) -> str:
    """The Czech table cache's key (``run_lexikon_1910_pilot``, D-115): the crop's bytes, the
    instruction, the model, the prompt and every reading setting."""
    h = hashlib.sha256()
    for piece in (pilot.encode_png(crop) + "\0", instruction, str(getattr(reader, "model", "")),
                  str(getattr(reader, "system", "")), str(getattr(reader, "temperature", "")),
                  str(getattr(reader, "reasoning_effort", "") or ""),
                  str(getattr(reader, "thinking", "")), str(getattr(reader, "effort", ""))):
        h.update(piece.encode("utf-8"))
        h.update(bytes(1))
    return h.hexdigest()


def cut(pages: list[int], profile) -> list[tuple]:
    """``[(block, crop), …]`` in reading order, and the blocks by page for the report."""
    jobs, by_page = [], {}
    for scan in pages:
        stem = f"{scan:04d}"
        image = lxdn.render_native(profile.source_pdf, scan)
        geo, blocks = lxdt.page_blocks(image, profile, stem)
        by_page[stem] = blocks
        jobs += [(block, lxdt.crop_block(image, geo, block)) for block in blocks]
    return jobs, by_page


def read_blocks(jobs: list[tuple], reader, cache: Path, queue: Path | None = None,
                workers: int = WORKERS) -> tuple[list, Counter]:
    """``[(block, rows or None), …]`` and the usage. With ``queue`` every block not in the
    cache is written there as a batch line (``reader_batches.py``) and none is sent."""
    cache.mkdir(parents=True, exist_ok=True)
    lock, usage = threading.Lock(), Counter()

    def one(job):
        block, crop = job
        instruction = lxdt.INSTRUCTIONS[block.kind]
        hit = cache / f"{cache_key(crop, instruction, reader)}.json"
        if hit.exists():
            record = json.loads(hit.read_text(encoding="utf-8"))
            return block, record["lines"], {"cached": 1}
        if queue is not None:
            make = getattr(reader, "params", None) or getattr(reader, "body")
            line = {"key": hit.stem, "reader": reader.name, "cache_dir": str(cache),
                    "slot": [block.page, block.block_id], "kind": "table",
                    "model": getattr(reader, "model", ""), "payload_key": "rows",
                    "request": make([(crop, "")], instruction)}
            with lock, Path(queue).open("a", encoding="utf-8") as fh:
                fh.write(json.dumps(line) + "\n")
            return block, None, {"queued": 1}
        for attempt in range(RETRIES):
            try:
                rows, spent = reader([(crop, "")], instruction)
                tmp = hit.with_suffix(".json.part")
                tmp.write_text(json.dumps({"lines": rows, "usage": spent,
                                           "slot": [block.page, block.block_id], "kind": "table",
                                           "model": getattr(reader, "model", ""),
                                           "at": time.time()}), encoding="utf-8")
                tmp.replace(hit)
                return block, rows, spent
            except pilot.Truncated:
                reader.max_tokens = min(getattr(reader, "max_tokens", 8000) * 2, 32000)
            except pilot.RateLimited:
                raise
            except Exception as exc:                                    # noqa: BLE001
                if attempt == RETRIES - 1:
                    return block, None, {"error": str(exc)[:300]}
                time.sleep(2 * (attempt + 1))
        return block, None, {"error": "truncated"}

    with ThreadPoolExecutor(max_workers=workers) as pool:
        results = list(pool.map(one, jobs))
    out = []
    for block, rows, spent in results:
        for key, value in spent.items():
            usage[key] += value if isinstance(value, (int, float)) else 1
        out.append((block, rows))
    return out, usage


def assemble(name: str, read: list[tuple], profile, crosswalk: dict, out: Path) -> dict:
    """Label every block with its district, check it, write the pass's two files."""
    labels = lxdt.court_labels([(b, rows or []) for b, rows in read])
    page_reads, checks = [], []
    for (block, rows), (okres, court) in zip(read, labels):
        court_cs = lxdn.court_name(court, crosswalk)[0] if court else ""
        page_reads.append((block, rows or [], okres, court, court_cs))
        for c in lxdt.grade_block(rows or [], profile, block.kind):
            checks.append({"page": block.page, "block_id": block.block_id,
                           "block_kind": block.kind, **c})
    rows = lxdt.table_rows(page_reads, profile)
    _write(out / f"table_rows_{name}.csv", rows)
    _write(out / f"table_checks_{name}.csv", checks)
    verdicts = Counter(c["verdict"] for c in checks)
    return {"blocks": len(read), "unread": sum(r is None for _b, r in read), "rows": len(rows),
            **{f"check_{k}": v for k, v in verdicts.items()}}


def _write(path: Path, rows: list[dict]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    fields = list(dict.fromkeys(k for r in rows for k in r)) or ["page"]
    with path.open("w", encoding="utf-8", newline="") as fh:
        writer = csv.DictWriter(fh, fieldnames=fields)
        writer.writeheader()
        writer.writerows(rows)


def frozen_pass(label: str, arm_name: str, pages: list[int], profile_name: str,
                out: Path | None = None, queue: Path | None = None) -> bool:
    """One table pass of the frozen route over any pages of one German volume. The runner that
    calls it (``run_frozen.py``) chooses the pages and keeps certification apart. ``False``
    if a quota stopped it; a rerun goes on from the cache."""
    out = out or bo.DE_PASS_DIR
    profile = lxvol.load_profile(profile_name)
    arm = ARMS[arm_name]
    print("\n" + "=" * 78 + f"\nGERMAN TABLE (frozen)  {label}   ({arm.model}, {profile.title}, "
          f"{len(pages)} pages)\n" + "=" * 78)
    started = time.time()
    jobs, by_page = cut(pages, profile)
    print("  " + lxdt.lexikon_1910_de_table_report(by_page, profile))
    reader = make_reader(arm, profile)
    try:
        read, usage = read_blocks(jobs, reader, out / "table_cache" / label, queue=queue)
    except pilot.RateLimited as exc:
        print(f"  rate limited: {exc}; run again to go on from the cache")
        return False
    if queue is not None:
        print(f"  {usage.get('queued', 0)} blocks queued to {Path(queue).name}, "
              f"{usage.get('cached', 0)} already in the cache")
        return True
    counts = assemble(label, read, profile, lxdn.load_court_crosswalk(), out)
    cost = (usage.get("input", 0) * arm.price[0] + usage.get("output", 0) * arm.price[1]) / 1e6
    print(f"  {counts}; {usage.get('cached', 0)} from the cache; ${cost:.2f}; "
          f"{(time.time() - started) / 60:.1f} min")
    return counts["unread"] == 0


def estimate(pages: list[int], profile, arm: Arm) -> str:
    """Blocks and dollars for one pass over ``pages``; renders and cuts, sends nothing."""
    jobs, by_page = cut(pages, profile)
    tokens = [min(pilot_image_cap(arm), crop.width * crop.height // 750) for _b, crop in jobs]
    out_tokens = EST_OUTPUT + (pilot.THINKING_TOKENS.get(arm.effort, 800) if arm.thinking == "adaptive" else 0)
    dollars = sum((t + 1500) * arm.price[0] + out_tokens * arm.price[1] for t in tokens) / 1e6
    per_page = len(jobs) / max(1, len(pages))
    return (lxdt.lexikon_1910_de_table_report(by_page, profile)
            + f"\n  {arm.model}: ~${dollars:.2f} for these {len(pages)} pages "
              f"({per_page:.1f} blocks a page; the volume's {profile.table_last - profile.table_first + 1} "
              f"table pages ~${dollars / max(1, len(pages)) * (profile.table_last - profile.table_first + 1):,.0f}, "
              f"half that batched)")


def pilot_image_cap(arm: Arm) -> int:
    """The image-token ceiling the estimate assumes for one crop."""
    return 4784 if arm.route == "gemini" else 1600


def page_spec(spec: str, profile) -> list[int]:
    """``gold`` (Böhmen's calibration and purposive pages), ``calibration`` (Mähren's or
    Schlesien's drawn pages, D-154), or ``142,276`` / ``21-30``. Never a sealed page."""
    if spec == "gold":
        pages = bo.de_gold_pages()
    elif spec == "calibration":
        path = ROOT / "reference" / "places" / f"gold_sample_pages_{profile.slug}.csv"
        with path.open(encoding="utf-8", newline="") as fh:
            pages = sorted(int(r["scan"]) for r in csv.DictReader(fh) if r["role"] == "calibration")
    else:
        pages = []
        for part in spec.split(","):
            first, _, last = part.partition("-")
            pages += range(int(first), int(last or first) + 1)
    sealed = set(profile.sealed_pages)
    assert not sealed & set(pages), f"sealed pages asked for: {sorted(sealed & set(pages))}"
    return pages


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n")[0])
    ap.add_argument("--volume", default="boehmen_de",
                    choices=("boehmen_de", "maehren_de", "schlesien_de"))
    ap.add_argument("--pages", default="gold")
    ap.add_argument("--reader", default="gemini38", choices=sorted(ARMS))
    ap.add_argument("--label", default="", help="pass label; default de_table_<reader>_<volume>")
    ap.add_argument("--dry-run", action="store_true", help="cut and estimate; send nothing")
    args = ap.parse_args(argv)
    profile = lxvol.load_profile(args.volume)
    pages = page_spec(args.pages, profile)
    if args.dry_run:
        print(estimate(pages, profile, ARMS[args.reader]))
        return 0
    label = args.label or f"de_table_{args.reader}_{args.volume}"
    return 0 if frozen_pass(label, args.reader, pages, args.volume) else 2


if __name__ == "__main__":
    raise SystemExit(main())
