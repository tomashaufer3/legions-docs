How it works

Usage (from the project root)::

python utils/docs/notebook_timings.py path/to/run02.log path/to/run03.log

Writes, per log, output/timings/&lt;notebook&gt;.log (the log, verbatim) and output/timings/&lt;notebook&gt;.json (the run's start, end, total and every cell's seconds, with the section the cell sits in). A later run of the same notebook replaces both. The JSON stores only the notebook's stem; the .log copy is verbatim and keeps the path its header recorded.

def parse_log(text: str) -> dict:
    """The run a log records: notebook stem, start, finish, exit status, total seconds and each cell's
    seconds. A log cut short (no ``# finished`` line) has ``finished`` None and ``total`` summed."""
    run: dict = {"notebook": None, "started": None, "finished": None, "status": None, "total_s": None,
                 "cells": []}
    first: dict[int, str] = {}
    for line in text.splitlines():
        if (m := _HEAD.match(line)) and run["notebook"] is None:
            run["notebook"] = Path(m["path"].replace("\\", "/")).stem
            run["started"] = m["started"]
        elif m := _CELL.match(line):
            first[int(m["cell"])] = m["first"].strip()
        elif m := _DONE.match(line):
            c = int(m["cell"])
            run["cells"].append({"cell": c, "first_line": first.get(c, ""), "status": m["status"],
                                 "seconds": float(m["sec"].replace(",", ""))})
        elif m := _END.match(line):
            run.update(finished=m["finished"], status=int(m["status"]), total_s=float(m["total"].replace(",", "")))
    if run["notebook"] is None:
        raise ValueError("not a run log: no '# <notebook>.ipynb | … | started …' header")
    if run["total_s"] is None:
        run["total_s"] = round(sum(c["seconds"] for c in run["cells"]), 1)
    return run

"""
notebook_timings.py — record a notebook run's cell timings where the docs site can read them
=============================================================================================

The *Reproduce the build* sheet says how long each notebook takes. Those numbers come from the log a
headless run writes (one ``=== cell N | first line`` header and one ``--- cell N: ok in S s`` footer per
code cell, a ``# finished … | total S s`` last line). A log kept in a session's scratch folder can vanish,
so this script copies it into ``output/timings/`` and writes a small JSON beside it; ``site_pages`` reads
the JSON at build time, so the sheet's figures come from a file in the project, not from typed prose.

Usage (from the project root)::

    python utils/docs/notebook_timings.py path/to/run02.log path/to/run03.log

Writes, per log, ``output/timings/<notebook>.log`` (the log, verbatim) and ``output/timings/<notebook>.json``
(the run's start, end, total and every cell's seconds, with the section the cell sits in). A later run of
the same notebook replaces both. The JSON stores only the notebook's stem; the .log copy is verbatim and keeps the path its header recorded.

Changelog
---------
1.0.0 (2026-11-18)  First version.
"""
from __future__ import annotations

import json
import re
import shutil
import sys
from pathlib import Path

__version__ = "1.0.0"

ROOT = Path(__file__).resolve().parents[2]
OUT = ROOT / "output" / "timings"

_HEAD = re.compile(r"^#\s*(?P<path>.+?\.ipynb)\s*\|.*?\|\s*started\s+(?P<started>[\d-]+ [\d:]+)\s*$")
_CELL = re.compile(r"^=== cell (?P<cell>\d+) \| (?P<first>.*)$")
_DONE = re.compile(r"^--- cell (?P<cell>\d+): (?P<status>\w+) in (?P<sec>[\d.,]+) s\s*$")
_END = re.compile(r"^# finished (?P<finished>[\d-]+ [\d:]+) \| status (?P<status>-?\d+) \| total (?P<total>[\d.,]+) s")


def parse_log(text: str) -> dict:
    """The run a log records: notebook stem, start, finish, exit status, total seconds and each cell's
    seconds. A log cut short (no ``# finished`` line) has ``finished`` None and ``total`` summed."""
    run: dict = {"notebook": None, "started": None, "finished": None, "status": None, "total_s": None,
                 "cells": []}
    first: dict[int, str] = {}
    for line in text.splitlines():
        if (m := _HEAD.match(line)) and run["notebook"] is None:
            run["notebook"] = Path(m["path"].replace("\\", "/")).stem
            run["started"] = m["started"]
        elif m := _CELL.match(line):
            first[int(m["cell"])] = m["first"].strip()
        elif m := _DONE.match(line):
            c = int(m["cell"])
            run["cells"].append({"cell": c, "first_line": first.get(c, ""), "status": m["status"],
                                 "seconds": float(m["sec"].replace(",", ""))})
        elif m := _END.match(line):
            run.update(finished=m["finished"], status=int(m["status"]), total_s=float(m["total"].replace(",", "")))
    if run["notebook"] is None:
        raise ValueError("not a run log: no '# <notebook>.ipynb | … | started …' header")
    if run["total_s"] is None:
        run["total_s"] = round(sum(c["seconds"] for c in run["cells"]), 1)
    return run


def _sections(stem: str) -> dict[int, tuple[str, str]]:
    """Each code cell's (section, first line) in the notebook as it is on disk now (``site_pages._Notebooks``)."""
    sys.path.insert(0, str(ROOT))
    from utils.docs.site_pages import _Notebooks
    return {i: (sec, (src.strip().splitlines() or [""])[0].strip())
            for i, sec, src in _Notebooks().cells.get(stem, [])}


def record(log: Path) -> Path:
    """Copy *log* into ``output/timings/`` and write its JSON; return the JSON's path."""
    run = parse_log(Path(log).read_text(encoding="utf-8"))
    secs = _sections(run["notebook"])
    for c in run["cells"]:
        sec, first = secs.get(c["cell"], ("", None))
        # a notebook edited since the run may have moved its cells: name a section only where the cell's
        # first line still reads as the log recorded it (the log cuts it, so a prefix match)
        same = first is not None and first[:60].strip() == c["first_line"][:60].strip()
        c["section"] = sec if same else ""
    run["log"] = f"output/timings/{run['notebook']}.log"
    run["recorded_by"] = f"utils/docs/notebook_timings.py {__version__}"
    OUT.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(log, OUT / f"{run['notebook']}.log")
    dest = OUT / f"{run['notebook']}.json"
    dest.write_text(json.dumps(run, ensure_ascii=False, indent=1) + "\n", encoding="utf-8", newline="\n")
    return dest


if __name__ == "__main__":
    if len(sys.argv) < 2:
        sys.exit(__doc__)
    for a in sys.argv[1:]:
        p = record(Path(a))
        r = json.loads(p.read_text(encoding="utf-8"))
        print(f"{r['notebook']}: {len(r['cells'])} cells, total {r['total_s']:,.1f} s -> {p.relative_to(ROOT).as_posix()}")
