How it works

The plates print a cooperative's language letter (č/n in the Czech volume, b/d/p in the German ones) for about one cooperative in five; the rest are not_stated (D-146), never a default. Measured on the rechecked register (2026-11-12/16): the two Bohemian volumes letter the same cooperatives identically (98.8 % of 3,805 place × kind pairs), and lettering follows the language mix of the court — 10 % of cooperatives lettered where the court's minority is under 2 %, 66 % where it is 25–50 % — so the printed letters mostly mark the language border. Where a letter is printed, it is the municipality's majority language in 91–95 % of cases, and in 99 % where the municipality's minority is under 5 % (Bohemia; 94–99 % in Mähren and Schlesien). The author's ruling (2026-11-16):

  • language_imputed — the printed language where a letter is printed; else the municipality's majority language (Czech, German or Polish, off the register's own table figures, lang_czech/lang_german/lang_polish) where its minority is under THRESHOLD; else not_stated. A cooperative of a part takes its municipality's figures; a municipality without figures imputes nothing.
  • language_source — printed, municipality_majority or not_stated.
  • n_coop_lang_imputed_&lt;language&gt; on register_municipalities (and so register_okres): the counts by language_imputed; they sum to n_coop.

language itself is untouched, and every analysis can be run on it as the robustness check. Each build re-measures the rule's own basis (_meta["coop_language"]["printed_agreement"]): how often a printed letter in a municipality under the threshold is its majority. Blind to: whether unlettered cooperatives differ from lettered ones in the same kind of municipality, which the agreement on lettered ones cannot show.

def municipality_languages(places: pd.DataFrame) -> pd.DataFrame:
    """Per municipality (edition, court_key, muni_seq): ``majority`` language and ``minority`` share, off the
    municipality's own row (``settlement_seq`` 0) of the table; ``<NA>`` where it prints no language figures."""
    own = places[places["settlement_seq"] == 0].copy()
    counts = pd.DataFrame({lang: pd.to_numeric(own[col], errors="coerce") if col in own else pd.Series(pd.NA, index=own.index)
                           for lang, col in FIGURES.items()}, index=own.index).astype("Float64")
    seen = counts.notna().any(axis=1)
    counts = counts.fillna(0)
    total = counts.sum(axis=1)
    usable = seen & (total > 0)
    out = own[MUNI].astype({"court_key": str}).copy()
    out["majority"] = counts.idxmax(axis=1).where(usable)
    out["minority"] = (1 - counts.max(axis=1) / total.where(usable)).where(usable)
    return out.drop_duplicates(MUNI).reset_index(drop=True)

def impute(cooperatives: pd.DataFrame, places: pd.DataFrame, threshold: float = THRESHOLD) -> tuple[pd.DataFrame, dict]:
    """``register_cooperatives`` with ``language_imputed`` and ``language_source``, and a report."""
    langs = municipality_languages(places)
    coop = cooperatives.assign(court_key=cooperatives["court_key"].astype(str))
    n0 = len(coop)
    j = coop.merge(langs, on=MUNI, how="left", validate="m:1")
    assert len(j) == n0, "coop_language: the municipality join changed the row count"
    printed = j["language"].astype(str).ne("not_stated") & j["language"].astype(str).ne("unknown")
    eligible = ~printed & j["minority"].notna() & (j["minority"] < threshold)
    j["language_imputed"] = j["language"].where(printed, j["majority"].where(eligible, "not_stated")).astype(str)
    j["language_source"] = "not_stated"
    j.loc[printed, "language_source"] = "printed"
    j.loc[eligible, "language_source"] = "municipality_majority"
    count = pd.to_numeric(j["count"], errors="coerce").fillna(0)
    basis = printed & j["minority"].notna() & (j["minority"] < threshold)
    agree = (j["language"].astype(str) == j["majority"].astype(str)).astype(float)
    agreement = {}
    for e in sorted(j["edition"].unique()):
        rows = basis & (j["edition"] == e)
        agreement[e] = {"lettered": int(count[rows].sum()),
                        "majority": round(float((count[rows] * agree[rows]).sum() / max(1.0, count[rows].sum())), 3)}
    report = {"threshold": threshold, "cooperatives": int(count.sum()),
              "by_source": {k: int(v) for k, v in count.groupby(j["language_source"]).sum().items()},
              "printed_agreement": agreement}
    out = j.drop(columns=["majority", "minority"])
    return out[list(cooperatives.columns) + ["language_imputed", "language_source"]], report

def municipal_columns(municipalities: pd.DataFrame, cooperatives: pd.DataFrame, places: pd.DataFrame) -> pd.DataFrame:
    """``register_municipalities`` with ``n_coop_lang_imputed_<language>`` (czech, german, polish, not_stated), the
    cooperatives of the municipality and its parts by ``language_imputed``; ``<NA>`` where ``n_coop`` is. Counted
    from the one volume the municipal table takes the municipality from (its ``edition``, ``court_id``,
    ``muni_seq``): a place both Bohemian volumes print is never summed twice."""
    owner = municipalities[["settlement_id", "edition", "court_id", "muni_seq"]].assign(
        court_key=municipalities["court_id"].astype("string").fillna("").astype(str))[["settlement_id"] + MUNI]
    c = cooperatives.assign(court_key=cooperatives["court_key"].astype(str)).merge(
        owner, on=MUNI, how="inner", validate="m:1")
    c = c.assign(count=pd.to_numeric(c["count"], errors="coerce").fillna(0))
    wide = c.pivot_table(index="settlement_id", columns="language_imputed", values="count", aggfunc="sum", fill_value=0)
    out = municipalities.drop(columns=[x for x in municipalities.columns if x.startswith("n_coop_lang_imputed_")])
    for lang in (*LANGUAGES, "not_stated"):
        col = f"n_coop_lang_imputed_{lang}"
        values = wide[lang] if lang in wide else pd.Series(dtype=float)
        out[col] = out["settlement_id"].map(values).fillna(0).astype("Int64")
        out.loc[out["n_coop"].isna(), col] = pd.NA
    return out

    coop, report = impute(tables["register_cooperatives"], tables["register_places"])
    muni = municipal_columns(tables["register_municipalities"], coop, tables["register_places"])
    out = {**tables, "register_cooperatives": coop, "register_municipalities": muni}
    out["_meta"] = {**tables.get("_meta", {}), "coop_language": report}
    return out


def write(tables: dict, out_dir: Path) -> None:
    """The cooperatives table (the municipalities are written by ``place_types.write``, after this step)."""
    tables["register_cooperatives"].to_parquet(Path(out_dir) / "register_cooperatives.parquet", index=False)
    tables["register_cooperatives"].to_csv(Path(out_dir) / "register_cooperatives.csv", index=False, encoding="utf-8")
