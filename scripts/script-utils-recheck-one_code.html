How it works

institution_abbreviations.csv holds some institutions under more than one code: the Czech volume's burgher school as měšť. šk. and měšt. šk. (D-156: as two readings print it), the real school and higher real school as reální/reálná, the gymnasium as gymn./gymnasium/gymnasiem, German Böhmen's district council as BezVer./BezVertrg., and a few more. Each code is a spelling, not a different institution: the rows of one group share expansion, kind and category. Kept apart, they split one institution across two mark columns of the register, and where one part holds two spellings — one institution the readers wrote two ways — the register counts it twice. The Czech sealed pages caught one (Chabařovice 1, D-230).

So, as the merge's last step: every code of a group becomes the group's first code in the table's order, and where a part then holds the institution twice, one row stays with the larger count (the same institution read twice, not two). Every rename and every merge is logged in changes.csv. Pictograms are untouched; cooperative letters and qualifiers are never grouped.

    earlier one in ``codes`` (``run_frozen.abbreviation_table``, in the vocabulary's order)."""
    first: dict = {}

            continue
        group = (a.expansion, a.kind, a.category)
        if group in first:
            out[code] = first[group]
        else:
            first[group] = code

def apply(t) -> dict:
    """Rename and merge in ``t.rows["abbreviations"]`` (a ``merge.Tables``); returns the counts."""
    from utils.recheck.merge import _part
    to = aliases(t.codes)
    if not to:
        return {"renamed": 0, "merged": 0}
    renamed = merged = 0
    kept: dict = {}
    out = []
    for r in t.rows["abbreviations"]:
        code = r["code"]
        canon = to.get(code, code)
        part = _part(r)
        if canon != code:
            renamed += 1
            t._log(STEP, "abbreviations", "alias", part, code, f"{code} is a spelling of {canon} (D-231)")
            r = dict(r, code=canon)
        key = (part, canon)
        if key in kept:
            first = kept[key]
            merged += 1
            t._log(STEP, "abbreviations", "remove", part, canon,
                   f"held twice under two spellings: one {canon}, count {max(_count(first), _count(r))} (D-231)")
            if _count(r) > _count(first):
                first.update(count=r["count"], count_source=r.get("count_source", ""))
            first["rule"] = f"{first.get('rule', '')} + {STEP}: two spellings held as one (D-231)".strip(" +")
            continue
        kept[key] = r
        out.append(r)
    t.rows["abbreviations"] = out
    return {"renamed": renamed, "merged": merged}

Pictograms are untouched; cooperative letters and qualifiers are never grouped.
"""
from __future__ import annotations

__version__ = "0.1.0"   # 0.1.0 one code per institution in the text table (D-231)

STEP = "one code"
KINDS = ("institution", "course")


def aliases(codes: dict) -> dict:
    """``{code: first code of its group}`` for every code that shares expansion, kind and category with an
    earlier one in ``codes`` (``run_frozen.abbreviation_table``, in the vocabulary's order)."""
    first: dict = {}
    out = {}
    for code, a in codes.items():
        if getattr(a, "kind", "") not in KINDS:
            continue
        group = (a.expansion, a.kind, a.category)
        if group in first:
            out[code] = first[group]
        else:
            first[group] = code
    return out


def _count(row: dict) -> int:
    try:
        return int(str(row.get("count") or "1").strip() or 1)
    except ValueError:
        return 1
