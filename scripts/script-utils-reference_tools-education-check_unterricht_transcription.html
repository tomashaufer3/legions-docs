How it works

Three tables of the volume are transcribed by hand from an image-only scan into unterricht_t{19,20,34}_*.csv. Reading a printed table off a photographed page is the one step in this project with a known failure mode: check_atlas_parties.py exists because a row-offset in a table read off a curved page produced a plausible set of numbers that only a cross-reference caught. The same risk applies here and is larger, because these are numeric columns where a slipped row leaves no grammatical trace at all.

What makes the check possible is that the volume over-determines itself. Every table prints both the parts and their roll-up: Küstenland is Triest + Görz und Gradisca + Istrien, Tirol und Vorarlberg is its two halves, the last row is the column total, and in Table 34 the eight class-count columns must sum across to the school total on the same row. None of that is used to build the CSVs — it is arithmetic the printer already did, so agreement is evidence the read is right rather than evidence the file is self-consistent.

Every failure is reported with the row and column, because the point is to find the slipped line, not to learn that one exists.

Run: python utils/reference_tools/education/check_unterricht_transcription.py Emits: nothing — prints PASS/FAIL per invariant. This is a transcription gate, never a pipeline input; it writes no file and touches no code column.

def check_table(path: Path, rollups: dict[str, list[str]], res: Result) -> None:
    name = path.name
    df = pd.read_csv(path)
    lands = df[df["row_kind"] == "land"].set_index("land")
    rolls = df[df["row_kind"] == "rollup"].set_index("land")
    total = df[df["row_kind"] == "total"]
    cols = _numeric(df)

    res.add(len(lands) == N_LANDS, f"{name}: {N_LANDS} crownland rows",
            f"found {len(lands)}")
    res.add(len(total) == 1, f"{name}: exactly one printed total row",
            f"found {len(total)}")
    res.add(not df["land"].duplicated().any(), f"{name}: no duplicated row label",
            f"duplicates: {sorted(df.loc[df['land'].duplicated(), 'land'])}")

    # 1. every column sums across the 17 crownlands to the printed total
    bad = {}
    for c in cols:
        got, want = int(lands[c].sum()), int(total[c].iloc[0])
        if got != want:
            bad[c] = f"{got} vs printed {want}"
    res.add(not bad, f"{name}: every column sums to the printed total", str(bad))

    # 2. every printed roll-up equals its printed parts, except where the printed volume
    #    itself is known to be one out (see KNOWN_SOURCE_ERRATA)
    bad, errata = {}, {}
    for roll, parts in rollups.items():
        if roll not in rolls.index:
            bad[roll] = "roll-up row absent"
            continue
        for c in cols:
            got, want = int(lands.loc[parts, c].sum()), int(rolls.loc[roll, c])
            if got == want:
                continue
            cell = f"{roll}.{c}"
            note = f"parts {got} vs printed {want}"
            if (name, cell) in KNOWN_SOURCE_ERRATA:
                errata[cell] = note
            else:
                bad[cell] = note
    res.add(not bad, f"{name}: printed roll-ups equal their parts", str(bad))
    if errata:
        res.errata.append((name, errata))

    # 3. column families add up across each row, on every row including the roll-ups
    bad = {}
    for parts, whole in ROW_SUMS.get(name, []):
        for _, row in df.iterrows():
            got, want = int(sum(row[p] for p in parts)), int(row[whole])
            if got != want:
                bad[f"{row['land']}.{whole}"] = f"{got} vs printed {want}"
    res.add(not bad, f"{name}: row families sum to their printed total", str(bad))

    # 4. a named subset never exceeds the whole it is drawn from
    bad = {}
    for parts, whole in SUBSETS.get(name, []):
        for _, row in df.iterrows():
            if sum(row[p] for p in parts) > row[whole]:
                bad[f"{row['land']}.{whole}"] = f"subset {sum(row[p] for p in parts)} > {row[whole]}"
    res.add(not bad, f"{name}: named subsets do not exceed their whole", str(bad))

    # 5. nothing negative, nothing missing — a blank cell in the source is a printed
    #    "." meaning zero, and must have been transcribed as 0 rather than left empty
    holes = {c: int(df[c].isna().sum()) for c in cols if df[c].isna().any()}
    res.add(not holes, f"{name}: no empty numeric cell", str(holes))
    neg = {c: int((df[c] < 0).sum()) for c in cols if (df[c] < 0).any()}
    res.add(not neg, f"{name}: no negative count", str(neg))

def main() -> int:
    res = Result()
    for fname, rollups in TABLES.items():
        path = HERE / fname
        if not path.exists():
            res.add(False, f"{fname}: present", "file not found")
            continue
        check_table(path, rollups, res)

    width = max(len(lbl) for _, lbl, _ in res)
    for ok, label, detail in res:
        line = f"{'PASS' if ok else 'FAIL'}  {label:<{width}}"
        print(line if ok else f"{line}  <- {detail}")

    failed = [lbl for ok, lbl, _ in res if not ok]
    print()
    print(f"{len(res) - len(failed)}/{len(res)} invariants passed")
    if res.errata:
        print()
        print("Known errors in the printed volume (transcription follows the page):")
        for fname, cells in res.errata:
            for cell, note in cells.items():
                print(f"  {fname}: {cell} -> {note}")
    if failed:
        print("FAILED:", "; ".join(failed))
    return 1 if failed else 0

"""Invariants on the transcription of Österreichische Statistik XCIII/1 (1908/09).

Three tables of the volume are transcribed by hand from an image-only scan into
``unterricht_t{19,20,34}_*.csv``.  Reading a printed table off a photographed page is the
one step in this project with a known failure mode: `check_atlas_parties.py` exists because
a row-offset in a table read off a curved page produced a *plausible* set of numbers that
only a cross-reference caught.  The same risk applies here and is larger, because these are
numeric columns where a slipped row leaves no grammatical trace at all.

What makes the check possible is that the volume over-determines itself.  Every table prints
both the parts **and** their roll-up: *Küstenland* is Triest + Görz und Gradisca + Istrien,
*Tirol und Vorarlberg* is its two halves, the last row is the column total, and in Table 34
the eight class-count columns must sum across to the school total on the same row.  None of
that is used to *build* the CSVs — it is arithmetic the printer already did, so agreement is
evidence the read is right rather than evidence the file is self-consistent.

Every failure is reported with the row and column, because the point is to find the slipped
line, not to learn that one exists.

Run:   python utils/reference_tools/education/check_unterricht_transcription.py
Emits: nothing — prints PASS/FAIL per invariant.  This is a transcription gate, never a
       pipeline input; it writes no file and touches no code column.

from __future__ import annotations

import sys
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[3]
HERE = ROOT / "reference" / "education"

__version__ = "1.0.0"

# The volume prints these roll-ups alongside their parts, so each is a free check on the read.
ROLLUPS = {
    "Küstenland": ["Triest", "Görz und Gradisca", "Istrien"],
    "Tirol und Vorarlberg": ["Tirol", "Vorarlberg"],
}
# Table 34 spells one of them differently; the transcription keeps the printed spelling.
ROLLUPS_T34 = {
    "Küstenland": ["Triest", "Görz und Gradiska", "Istrien"],
    "Tirol und Vorarlberg": ["Tirol", "Vorarlberg"],
}

TABLES = {
    "unterricht_t34_volksschulen.csv": ROLLUPS_T34,
    "unterricht_t20_gewerbeschulen.csv": ROLLUPS,
    "unterricht_t19_handelslehranstalten.csv": ROLLUPS,
}

# Column families that must add up across a row to a printed total on that same row.
ROW_SUMS = {
    "unterricht_t34_volksschulen.csv": [
        ([f"klassen_{i}" for i in range(1, 9)], "volksschulen_total"),
    ],
    "unterricht_t20_gewerbeschulen.csv": [
        (["staatsgewerbe_anstalten", "fachschulen_anstalten",
          "handwerker_anstalten", "fortbildung_anstalten"], "zusammen_anstalten"),
        (["staatsgewerbe_lehrkraefte", "fachschulen_lehrkraefte",
          "handwerker_lehrkraefte", "fortbildung_lehrkraefte"], "zusammen_lehrkraefte"),
        (["staatsgewerbe_schueler", "fachschulen_schueler",
          "handwerker_schueler", "fortbildung_schueler"], "zusammen_schueler"),
    ],
    "unterricht_t19_handelslehranstalten.csv": [
        (["hoehere_handelsschulen", "tagesschulen_oeffentliche",
          "tagesschulen_vereine_privat", "kaufm_fortbildungsschulen"], "zusammen"),
    ],
}

# Columns that are a named subset of another and so may never exceed it.
SUBSETS = {
    "unterricht_t34_volksschulen.csv": [
        (["buergerschulen_deutsch", "buergerschulen_cechoslavisch"], "buergerschulen_total"),
        (["notschulen"], "volksschulen_total"),
    ],
}

# The 17 crownlands, once each. Roll-ups and the total are excluded from column sums so a
# column is never counted twice — the mistake that would make every total agree trivially.
N_LANDS = 17

# Errors in the printed volume, not in the transcription.
#
# The transcription is faithful to the page, so a genuine arithmetic slip by the k.k.
# Statistische Zentralkommission surfaces here as a failure.  Each entry below was checked
# against the scan at high magnification and, crucially, against a *second* arithmetic path
# that decides which of the two printed numbers is wrong — otherwise "the source is wrong"
# is unfalsifiable and becomes a way to make any bad read pass.
#
# `20/Tirol und Vorarlberg`: the page prints Tirol 30 and Vorarlberg 13 gewerbliche
# Fortbildungs-/Zeichenschulen but rolls them up to 42, and correspondingly 52 + 14 = 66
# Anstalten altogether printed as 65.  Every other cell of the same roll-up adds up exactly
# (Lehrkräfte 83 + 53 = 136, Schüler 1.839 + 447 = 2.286, Zusammen 250 + 60 = 310 and
# 3.237 + 516 = 3.753), and the empire total of 1.200 Anstalten agrees with the crownland
# rows and not with the roll-up.  So the two crownland rows are authoritative and the
# roll-up is one short; the transcription keeps both as printed.
KNOWN_SOURCE_ERRATA = {
    ("unterricht_t20_gewerbeschulen.csv", "Tirol und Vorarlberg.fortbildung_anstalten"),
    ("unterricht_t20_gewerbeschulen.csv", "Tirol und Vorarlberg.zusammen_anstalten"),
}


class Result(list):
    def __init__(self):
        super().__init__()
        self.errata: list[tuple[str, dict]] = []

    def add(self, ok: bool, label: str, detail: str = "") -> None:
        self.append((ok, label, detail))


def _numeric(df: pd.DataFrame) -> list[str]:
    return [c for c in df.columns
            if c not in {"table_no", "folio", "pdf_page", "land", "row_kind"}]


def check_table(path: Path, rollups: dict[str, list[str]], res: Result) -> None:
    name = path.name
    df = pd.read_csv(path)
    lands = df[df["row_kind"] == "land"].set_index("land")
    rolls = df[df["row_kind"] == "rollup"].set_index("land")
    total = df[df["row_kind"] == "total"]
    cols = _numeric(df)

    res.add(len(lands) == N_LANDS, f"{name}: {N_LANDS} crownland rows",
            f"found {len(lands)}")
    res.add(len(total) == 1, f"{name}: exactly one printed total row",
            f"found {len(total)}")
    res.add(not df["land"].duplicated().any(), f"{name}: no duplicated row label",
            f"duplicates: {sorted(df.loc[df['land'].duplicated(), 'land'])}")

    # 1. every column sums across the 17 crownlands to the printed total
    bad = {}
    for c in cols:
        got, want = int(lands[c].sum()), int(total[c].iloc[0])
        if got != want:
            bad[c] = f"{got} vs printed {want}"
    res.add(not bad, f"{name}: every column sums to the printed total", str(bad))

    # 2. every printed roll-up equals its printed parts, except where the printed volume
    #    itself is known to be one out (see KNOWN_SOURCE_ERRATA)
    bad, errata = {}, {}
    for roll, parts in rollups.items():
        if roll not in rolls.index:
            bad[roll] = "roll-up row absent"
            continue
        for c in cols:
            got, want = int(lands.loc[parts, c].sum()), int(rolls.loc[roll, c])
            if got == want:
                continue
            cell = f"{roll}.{c}"
            note = f"parts {got} vs printed {want}"
            if (name, cell) in KNOWN_SOURCE_ERRATA:
                errata[cell] = note
            else:
                bad[cell] = note
    res.add(not bad, f"{name}: printed roll-ups equal their parts", str(bad))
    if errata:
        res.errata.append((name, errata))

    # 3. column families add up across each row, on every row including the roll-ups
    bad = {}
    for parts, whole in ROW_SUMS.get(name, []):
        for _, row in df.iterrows():
            got, want = int(sum(row[p] for p in parts)), int(row[whole])
            if got != want:
                bad[f"{row['land']}.{whole}"] = f"{got} vs printed {want}"
    res.add(not bad, f"{name}: row families sum to their printed total", str(bad))

    # 4. a named subset never exceeds the whole it is drawn from
    bad = {}
    for parts, whole in SUBSETS.get(name, []):
        for _, row in df.iterrows():
            if sum(row[p] for p in parts) > row[whole]:
                bad[f"{row['land']}.{whole}"] = f"subset {sum(row[p] for p in parts)} > {row[whole]}"
    res.add(not bad, f"{name}: named subsets do not exceed their whole", str(bad))

    # 5. nothing negative, nothing missing — a blank cell in the source is a printed
    #    "." meaning zero, and must have been transcribed as 0 rather than left empty
    holes = {c: int(df[c].isna().sum()) for c in cols if df[c].isna().any()}
    res.add(not holes, f"{name}: no empty numeric cell", str(holes))
    neg = {c: int((df[c] < 0).sum()) for c in cols if (df[c] < 0).any()}
    res.add(not neg, f"{name}: no negative count", str(neg))


def main() -> int:
    res = Result()
    for fname, rollups in TABLES.items():
        path = HERE / fname
        if not path.exists():
            res.add(False, f"{fname}: present", "file not found")
            continue
        check_table(path, rollups, res)

    width = max(len(lbl) for _, lbl, _ in res)
    for ok, label, detail in res:
        line = f"{'PASS' if ok else 'FAIL'}  {label:<{width}}"
        print(line if ok else f"{line}  <- {detail}")

    failed = [lbl for ok, lbl, _ in res if not ok]
    print()
    print(f"{len(res) - len(failed)}/{len(res)} invariants passed")
    if res.errata:
        print()
        print("Known errors in the printed volume (transcription follows the page):")
        for fname, cells in res.errata:
            for cell, note in cells.items():
                print(f"  {fname}: {cell} -> {note}")
    if failed:
        print("FAILED:", "; ".join(failed))
    return 1 if failed else 0


if __name__ == "__main__":
    raise SystemExit(main())
