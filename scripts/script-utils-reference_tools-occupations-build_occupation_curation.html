How it works

It reads data/raw/legionnaires/legionnaires.csv, not data/final/: the final frame is produced by notebook 02, which consumes this curation, so building the skeleton from the final frame would make the pipeline circular.

Refresh is non-destructive. Re-running merges on key and keeps every hand-entered value; only n / cum_share / rank are recomputed, and new keys are appended with empty code columns. A key that has dropped out of the top N but already carries a code is kept, never dropped — the same round-trip property districts.write_curation_template gives the places curation.

def key_frequencies() -> pd.DataFrame:
    """Normalised key -> record counts, over both occupation fields."""
    if not RAW.exists():
        raise SystemExit(f"Missing {RAW.relative_to(ROOT)} — the csol scrape must be present.")
    df = pd.read_csv(RAW, dtype=str, usecols=SOURCE_COLS)

    frames = []
    for col, name in zip(SOURCE_COLS, ["n_enrol", "n_interwar"]):
        keys = df[col].map(_norm).dropna()
        frames.append(keys.value_counts().rename(name))
    out = pd.concat(frames, axis=1).fillna(0).astype(int)
    out["n"] = out["n_enrol"] + out["n_interwar"]
    out = out.sort_values("n", ascending=False)
    out.index.name = "key"
    return out.reset_index()

def build(top: int) -> pd.DataFrame:
    freq = key_frequencies()
    total = int(freq["n"].sum())
    freq["rank"] = range(1, len(freq) + 1)
    freq["cum_share"] = (freq["n"].cumsum() / total).round(5)

    print(f"  {len(freq):,} distinct normalised keys over {total:,} coded-field values")
    for n in (200, 500, 1000, 1500, 2000):
        if n <= len(freq):
            print(f"    top {n:>5} keys -> {freq['cum_share'].iloc[n - 1]:.2%}")

    skeleton = freq.head(top).copy()
    for c in CURATED_COLS:
        skeleton[c] = ""

    if OUT.exists():
        old = pd.read_csv(OUT, dtype=str).fillna("")
        curated = old[old["hisco"].str.strip() != ""]
        print(f"  existing curation: {len(old):,} rows, "
              f"{len(curated):,} already carry a HISCO code")

        # Hand-entered values win; counts are always recomputed.
        merged = skeleton.merge(old[["key", *CURATED_COLS]], on="key",
                                how="outer", suffixes=("", "_old"))
        for c in CURATED_COLS:
            old_c = merged[f"{c}_old"].fillna("")
            merged[c] = old_c.where(old_c.str.strip() != "", merged[c].fillna(""))
            merged = merged.drop(columns=[f"{c}_old"])

        # Keys that fell out of the top N but are already coded stay in.
        fresh = freq.set_index("key")
        for c in COUNT_COLS:
            merged[c] = merged["key"].map(fresh[c]) if c in fresh.columns else merged.get(c)
        kept = merged["key"].isin(skeleton["key"]) | (merged["hisco"].str.strip() != "")
        dropped = int((~kept).sum())
        if dropped:
            print(f"  dropping {dropped:,} uncoded keys no longer in the top {top:,}")
        skeleton = merged[kept].copy()

    skeleton = skeleton.sort_values("n", ascending=False, ignore_index=True)
    return skeleton[["key", *COUNT_COLS, *CURATED_COLS]]

====================================================================================================

raw csol.cz ``Zaměstnání`` (and ``Povolání v letech 1918 - 1938``) field, with its record

Run:  python utils/reference_tools/occupations/build_occupation_curation.py [--top 1000]
Emits: reference/occupations/occupation_hisco_curation.csv
Consumed by utils.parsing.occupations.load_occupation_curation.

from __future__ import annotations

import argparse
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

import pandas as pd
from utils.parsing.occupations import _norm

__version__ = "1.0.0"

RAW = ROOT / "data" / "raw" / "legionnaires" / "legionnaires.csv"
OUT = ROOT / "reference" / "occupations" / "occupation_hisco_curation.csv"

SOURCE_COLS = ["Zaměstnání", "Povolání v letech 1918 - 1938"]

# Hand-filled columns, preserved verbatim across refreshes.
CURATED_COLS = ["de_gloss", "en_gloss", "hisco", "status", "relation", "product",
                "source", "link_hisco", "occcanine_hisco", "occcanine_p",
                "agree", "review", "note", "verdict"]
COUNT_COLS = ["rank", "n", "n_enrol", "n_interwar", "cum_share"]


def key_frequencies() -> pd.DataFrame:

    out = pd.concat(frames, axis=1).fillna(0).astype(int)
    out["n"] = out["n_enrol"] + out["n_interwar"]
    out = out.sort_values("n", ascending=False)
    out.index.name = "key"
    return out.reset_index()
