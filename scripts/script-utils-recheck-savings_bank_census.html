How it works

The error sample (D-226) found the German volumes' savings bank to be mostly a misread cooperative: the register's savings bank was printed on 1 of 6 sampled German entries against 18 of 18 Czech ones. The cooperative's glyph is the savings-bank beehive with wavy lines beside it, so a faint or cropped wave turns one into the other. The error sample's bar for the stratum the readers doubted was not met in either German volume on the blind ruling (31/35, 32/35 against 33), and the pre-registered consequence is a targeted fix. This is that fix: a census, not a sample — every entry (municipality or part) of Mähren and Schlesien to which the rechecked offices table (final_inventory_rechecked.csv) gives a [savings-bank], one card each, never one on a sealed, reserve, calibration or gold page (skipped_pages).

The card asks about the entry, not about a reader's box: the court's offices list in a window opened near the entry (at a mark of the entry where one exists; for a court whose offices sit inside its parts section, at the region §4 found, D-192), all the court's sections on the page beside it, and the volume's own plate glyphs for both symbols. What does this entry's offices line print — a savings bank, a cooperative, both, or neither? (or cannot tell). The answer settles both symbols at the entry, the analysis's level.

Not yet merged. The answers are staged in output/review_readback/savings_bank/rulings.json (with SOURCE.txt) and laid over the register by notebook 17's merge as author rulings in a later step; changes lists what they would change.

python -m utils.recheck.savings_bank_census build    # key (reference/), windows and page (output/)
python -m utils.recheck.savings_bank_census changes  # what the staged answers would change

def census(edition: str, recheck: Path = RECHECK, key: Path = KEY) -> list[dict]:
    """Every entry the volume's rechecked offices table gives a savings bank, not on a page :func:`skipped_pages`
    keeps out, in page and entry order. An id once given stays with its entry (``key``): a rebuild never
    renumbers a card that may already carry an answer."""
    rows = _read(Path(recheck) / edition / "final_inventory_rechecked.csv")
    coop = {r["entry"] for r in rows if r["code"] == COOPERATIVE}
    named = names(edition)
    skip = skipped_pages(edition)
    claims = sorted((r for r in rows if r["code"] == TOKEN and r["page"] not in skip),
                    key=lambda r: (r["page"], r["court_id"], int(r["muni_seq"]), int(r["settlement_seq"] or 0)))
    given = {(k["edition"], k["entry"]): k["id"] for k in _read(key)}
    taken = {int(i.rsplit("-", 1)[1]) for (e, _x), i in given.items() if e == edition}
    fresh = iter(n for n in range(1, 10_000) if n not in taken)
    out = []
    for r in claims:
        # a post-1910 court has no court_id in the offices table, and the Bielitz, Skotschau and Strumen courts have
        # no municipality in the rechecked register: those cards carry the court's label as the run filed it
        court_cs, printed = named.get((r["court_id"], r["muni_seq"]), (r["district"].capitalize(), ""))
        cid = given.get((edition, r["entry"])) or f"{SHORT[edition]}-sb-{next(fresh):03d}"
        out.append({"id": cid, "edition": edition, "entry": r["entry"], "page": r["page"],
                    "district": r["district"], "court_id": r["court_id"], "muni_seq": r["muni_seq"],
                    "settlement_seq": r["settlement_seq"], "court_cs": court_cs, "printed_name": printed,
                    "rule": r["rule"], "marks": r["marks"], "cooperative_held": r["entry"] in coop})
    return out

    """Each card's window, opened near the entry: at a boxed mark of the entry where one exists, else at

    import run_verify_test_b as tb

def page(items: list[dict], out: Path = OUT) -> Path:
    """``index.html``: the cards and, per volume, the plate's savings-bank and cooperative glyphs."""
    from PIL import Image
    from utils.recheck import family_rulings as fr
    glyphs = {}
    for ed in EDITIONS:
        glyphs[ed] = {}
        for token in ("savings-bank", "cooperative"):
            source = fr.SYMBOLS / PLATES[ed] / f"{token}.png"
            target = Path(out) / "glyph" / SHORT[ed] / source.name
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(source.read_bytes())
            glyphs[ed][token] = f"glyph/{SHORT[ed]}/{source.name}"
    shown = []
    for it in items:
        item = {k: it.get(k) for k in SHOWN}
        if item["one"].get("section"):
            with Image.open(Path(out) / item["one"]["section"]) as im:
                item["one"] = {**item["one"], "w": im.width, "h": im.height}
        shown.append(item)
    html = (TEMPLATE.read_text(encoding="utf-8")
            .replace("__ITEMS__", json.dumps(shown, ensure_ascii=False))
            .replace("__GLYPHS__", json.dumps(glyphs, ensure_ascii=False)))
    path = Path(out) / "index.html"
    path.write_text(html, encoding="utf-8")
    return path

def build(recheck: Path = RECHECK, out: Path = OUT, key: Path = KEY) -> dict:
    items = [it for ed in EDITIONS for it in census(ed, recheck, key)]
    windows(items, recheck, out)
    write_key(items, key)
    path = page(items, out)
    return {"entries": {ed: sum(i["edition"] == ed for i in items) for ed in EDITIONS},
            "left_out_sealed_or_reserve": {ed: sum(r["code"] == TOKEN and r["page"] in skipped_pages(ed)
                                                   for r in _read(Path(recheck) / ed / "final_inventory_rechecked.csv"))
                                           for ed in EDITIONS},
            "opened_at": {s: sum((i["one"] or {}).get("from") == s for i in items)
                          for s in ("entry", "region", "neighbour", "court")},
            "no_window": [i["id"] for i in items if not (i["one"] or {}).get("section")],
            "key": str(Path(key).relative_to(ROOT)), "page": str(path.relative_to(ROOT))}

def changes(key: list[dict] | None = None, answers: dict | None = None) -> dict:
    """What the staged answers would change at each entry, for both symbols: the register's savings bank kept
    (*savings bank*, *both*) or dropped (*cooperative*, *neither*); a cooperative added where the answer prints
    one and the register has none, dropped where the register has one and the answer prints none; *cannot
    tell* and unanswered cards change nothing."""
    key = read_key() if key is None else key
    answers = read_rulings() if answers is None else answers
    out = {"keep_savings_bank": [], "drop_savings_bank": [], "add_cooperative": [], "drop_cooperative": [], "open": []}
    for it in key:
        v = str((answers.get(it["id"]) or {}).get("value") or "")
        if v not in ANSWERS or v == "cannot-tell":
            out["open"].append(it["id"])
            continue
        coop_held = str(it.get("cooperative_held")) == "True"
        out["keep_savings_bank" if v in ("savings-bank", "both") else "drop_savings_bank"].append(it["id"])
        if v in ("cooperative", "both") and not coop_held:
            out["add_cooperative"].append(it["id"])
        if v in ("savings-bank", "neither") and coop_held:
            out["drop_cooperative"].append(it["id"])
    return out

def as_rulings(edition: str, key: Path = KEY, rulings: Path = RULINGS, recheck: Path = RECHECK,
               problems: list | None = None) -> list[dict]:
    """The census answers as merge's author-step records (``merge.apply_chat``): ``{part, code, verdict, ruled,
    note, item}`` for both symbols at each answered entry — the savings bank *printed* on *savings bank* and
    *both*, *not printed* on *cooperative* and *neither*; the cooperative *printed* on *cooperative* and *both*,
    *not printed* on the other two. *Cannot tell* and unanswered cards give none.

    The key holds each entry as the merge's output wrote it, after its ``keys`` step; the author step meets
    it on the number the page prints, so the step is undone here as for the look-alike families
    (:func:`family_rulings.keys_undo`): a post-1910 court's number back to the page's, a German opening
    entry's row back to the court it stood under. A row moved in from two courts is a problem, left out."""
    from utils.recheck import family_rulings as fr
    if edition not in EDITIONS or not Path(key).exists():
        return []
    answers = read_rulings(rulings)
    if not answers:
        return []
    renumbered, moved = fr.keys_undo(_read(Path(recheck) / edition / "changes.csv"))
    out = []
    for it in (k for k in read_key(key) if k["edition"] == edition):
        answer = answers.get(it["id"]) or {}
        value = str(answer.get("value") or "")
        if value not in ANSWERS or value == "cannot-tell":
            continue
        page, district, muni, seq = it["page"], it["district"], it["muni_seq"], it["settlement_seq"]
        muni = renumbered.get((page, district, muni, seq), muni)
        before = moved.get((page, district, muni, seq), {}).get(TOKEN, set())
        if len(before) > 1:
            if problems is not None:
                problems.append({"id": it["id"], "edition": edition, "entry": it["entry"],
                                 "problem": f"the keys step moved the savings bank here from {sorted(before)}"})
            continue
        part = (page, next(iter(before)) if before else district, muni, seq)
        note = str(answer.get("note") or "")
        verdicts = {TOKEN: value in ("savings-bank", "both"), COOPERATIVE: value in ("cooperative", "both")}
        for code, printed in verdicts.items():
            out.append({"part": part, "code": code, "verdict": "printed" if printed else "not printed",
                        "ruled": RULED_ON, "note": note, "item": it["id"]})
    return out

"""Every entry where the German volumes' register claims a savings bank, read again: savings bank, cooperative,
both or neither (D-229, notebook 17 §6f).

The error sample (D-226) found the German volumes' savings bank to be mostly a misread cooperative: the
register's savings bank was printed on 1 of 6 sampled German entries against 18 of 18 Czech ones. The
cooperative's glyph is the savings-bank beehive with wavy lines beside it, so a faint or cropped wave turns
one into the other. The error sample's bar for the stratum the readers doubted was not met in either German
volume on the blind ruling (31/35, 32/35 against 33), and the pre-registered consequence is a targeted fix.
This is that fix: a census, not a sample — every entry (municipality or part) of Mähren and Schlesien to
which the rechecked offices table (``final_inventory_rechecked.csv``) gives a ``[savings-bank]``, one card
each, never one on a sealed, reserve, calibration or gold page (:func:`skipped_pages`).

**The card** asks about the entry, not about a reader's box: the court's offices list in a window opened near
the entry (at a mark of the entry where one exists; for a court whose offices sit inside its parts section,
at the region §4 found, D-192), all the court's sections on the page beside it, and the volume's own plate
glyphs for both symbols. *What does this entry's offices line print — a savings bank, a cooperative, both,
or neither?* (or *cannot tell*). The answer settles both symbols at the entry, the analysis's level.

**Not yet merged.** The answers are staged in ``output/review_readback/savings_bank/rulings.json`` (with
``SOURCE.txt``) and laid over the register by notebook 17's merge as author rulings in a later step;
:func:`changes` lists what they would change.

    python -m utils.recheck.savings_bank_census build    # key (reference/), windows and page (output/)
    python -m utils.recheck.savings_bank_census changes  # what the staged answers would change
"""
from __future__ import annotations

__version__ = "0.2.0"   # 0.1.0 the census of the German volumes' savings-bank claims, one card per reader's box (D-229)
#                         0.2.0 one question per entry (savings bank, cooperative, both, neither), no outline; sealed and
#                               reserve pages left out; ids kept across rebuilds; windows at §4's regions, all the court's
#                               sections on the page beside them

import argparse
import csv
import json
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT), str(ROOT / "utils" / "lexikon_1910")]

EDITIONS = ("de-moravia", "de-silesia")
SHORT = {"de-moravia": "mo", "de-silesia": "si"}
TOKEN, COOPERATIVE = "[savings-bank]", "[cooperative]"
RECHECK = ROOT / "output" / "recheck"
OUT = RECHECK / "savings_bank"
KEY = ROOT / "reference" / "places" / "savings_bank" / "census_key.csv"
RULINGS = ROOT / "output" / "review_readback" / "savings_bank" / "rulings.json"
TEMPLATE = Path(__file__).with_name("savings_bank_census_page.html")
PLATES = {"de-moravia": "de_maehren", "de-silesia": "de_schlesien"}
ANSWERS = ("savings-bank", "cooperative", "both", "neither", "cannot-tell")
KEY_FIELDS = ("id", "edition", "entry", "page", "district", "court_id", "muni_seq", "settlement_seq", "court_cs",
              "printed_name", "rule", "marks", "cooperative_held")


def _read(path: Path) -> list[dict]:
    if not Path(path).exists():
        return []
    with open(path, encoding="utf-8", newline="") as fh:
        return list(csv.DictReader(fh))


def names(edition: str) -> dict:
    """``(court_id, printed muni) → (court_cs, printed_name)`` from the rechecked register's municipalities
    (a post-1910 court's key is offset; :func:`error_sample.printed_number` takes it back to the page)."""
    import pandas as pd
    from utils.recheck import error_sample as es
    m = pd.read_parquet(ROOT / "data" / "final" / "register_rechecked" / "register_municipalities.parquet")
    m = m[m["edition"] == edition]
    return {(str(c), es.printed_number(str(s))): (cs, n)
            for c, s, cs, n in zip(m["court_id"], m["muni_seq"], m["court_cs"], m["printed_name"])}


def skipped_pages(edition: str) -> set:
    """Pages no card may show: sealed, reserve, calibration and gold (``run_frozen.sealed_pages | in_sample_pages``),
    as in the error sample's frame. No ruling of the author's may fall on them."""
    import run_frozen as rf
    return set(rf.sealed_pages(edition)) | set(rf.in_sample_pages(edition))


def census(edition: str, recheck: Path = RECHECK, key: Path = KEY) -> list[dict]:
    """Every entry the volume's rechecked offices table gives a savings bank, not on a page :func:`skipped_pages`
    keeps out, in page and entry order. An id once given stays with its entry (``key``): a rebuild never
    renumbers a card that may already carry an answer."""
    rows = _read(Path(recheck) / edition / "final_inventory_rechecked.csv")
    coop = {r["entry"] for r in rows if r["code"] == COOPERATIVE}
    named = names(edition)
    skip = skipped_pages(edition)
    claims = sorted((r for r in rows if r["code"] == TOKEN and r["page"] not in skip),
                    key=lambda r: (r["page"], r["court_id"], int(r["muni_seq"]), int(r["settlement_seq"] or 0)))
    given = {(k["edition"], k["entry"]): k["id"] for k in _read(key)}
    taken = {int(i.rsplit("-", 1)[1]) for (e, _x), i in given.items() if e == edition}
    fresh = iter(n for n in range(1, 10_000) if n not in taken)
    out = []
    for r in claims:
        # a post-1910 court has no court_id in the offices table, and the Bielitz, Skotschau and Strumen courts have
        # no municipality in the rechecked register: those cards carry the court's label as the run filed it
        court_cs, printed = named.get((r["court_id"], r["muni_seq"]), (r["district"].capitalize(), ""))
        cid = given.get((edition, r["entry"])) or f"{SHORT[edition]}-sb-{next(fresh):03d}"
        out.append({"id": cid, "edition": edition, "entry": r["entry"], "page": r["page"],
                    "district": r["district"], "court_id": r["court_id"], "muni_seq": r["muni_seq"],
                    "settlement_seq": r["settlement_seq"], "court_cs": court_cs, "printed_name": printed,
                    "rule": r["rule"], "marks": r["marks"], "cooperative_held": r["entry"] in coop})
    return out


def _court_sections(edition: str, page: str, labels: set) -> list[str]:
    """Every section image (offices first, then parts) the run cut for the court on the page."""
    import run_frozen as rf
    folder = rf.out_dir(edition, "full") / "sections"
    found = [p.name for p in sorted(folder.glob(f"{page}_*.png")) if p.name.split("_")[1] in labels]
    return sorted(found, key=lambda n: ("_offices_" not in n, n))


def _window_mark(edition: str, it: dict, labels: set) -> tuple[str, str]:
    """``(mark, from)`` for a card no mark of its entry places: §4's region of the court on the page (D-192: the
    offices list inside the parts section), else the neighbouring entry's offices mark of the same court
    (:func:`error_sample._fallback`, kept only when it is the same court), else the court's first section."""
    from utils.recheck import error_sample as es
    from utils.recheck import offices_reread as orr
    regions = [r for r in orr.regions(edition) if r["section"].startswith(f"{it['page']}_")
               and r["section"].split("_")[1] in labels and r.get("how") != "continuation"]
    if regions:
        y = int(float(regions[0]["y_from"] or 0))
        return f"{regions[0]['section']}.png#0,{y},1,{y + 1}", "region"
    mark, _own = es._fallback(edition, it["page"], it["district"], it["muni_seq"], it["court_id"])
    if mark and Path(mark.split("#")[0]).name.split("_")[1] in labels:
        return mark, "neighbour"
    sections = _court_sections(edition, it["page"], labels)
    return (f"{Path(sections[0]).stem}.png#0,0,1,1", "court") if sections else ("", "")


def windows(items: list[dict], recheck: Path = RECHECK, out: Path = OUT) -> list[dict]:
    """Each card's window, opened near the entry: at a boxed mark of the entry where one exists, else at
    :func:`_window_mark`; and beside it every other section the run cut for the court on that page (the
    rest of the offices list, and the parts sections a court's offices can run into)."""
    import run_verify_test_b as tb
    from PIL import Image
    from utils.recheck import error_sample as es
    from utils.recheck import family_rulings as fr
    from utils.recheck import opus_pass as op
    folder = Path(out) / "img"
    folder.mkdir(parents=True, exist_ok=True)
    for ed in EDITIONS:
        op._context(ed)
        by_entry = {}
        for r in _read(Path(recheck) / ed / "final_inventory_rechecked.csv"):
            if ".png#" in r["marks"]:
                by_entry.setdefault(r["entry"], []).append(r["marks"])
        for it in (i for i in items if i["edition"] == ed):
            labels = {it["district"]} | es._same_court(ed, it["page"], it["court_id"])
            side = fr._side(ed, " | ".join(by_entry.get(it["entry"], [])), it["id"], folder)
            if side:
                side["from"] = "entry"
            else:
                mark, how = _window_mark(ed, it, labels)
                side = fr._side(ed, mark, it["id"], folder) if mark else {}
                if side:
                    side["from"] = how
            it["one"] = side or {}
            shown = Path(side["section"]).stem + ".png" if side else ""
            it["more"] = []
            for name in _court_sections(ed, it["page"], labels):
                if name == shown:
                    continue
                target = Path(out) / "sec" / f"{Path(name).stem}.jpg"
                if not target.exists():
                    with Image.open(tb.image_path(name)) as scan:
                        fr._jpeg(scan.convert("RGB"), target, width=1000)
                it["more"].append(f"sec/{target.name}")
    return items


def write_key(items: list[dict], path: Path = KEY) -> Path:
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w", encoding="utf-8", newline="") as fh:
        w = csv.DictWriter(fh, fieldnames=KEY_FIELDS, extrasaction="ignore")
        w.writeheader()
        w.writerows(items)
    return Path(path)


def read_key(path: Path = KEY) -> list[dict]:
    return _read(path)


SHOWN = ("id", "edition", "court_cs", "printed_name", "page", "muni_seq", "settlement_seq", "one", "more")


def page(items: list[dict], out: Path = OUT) -> Path:
    """``index.html``: the cards and, per volume, the plate's savings-bank and cooperative glyphs."""
    from PIL import Image
    from utils.recheck import family_rulings as fr
    glyphs = {}
    for ed in EDITIONS:
        glyphs[ed] = {}
        for token in ("savings-bank", "cooperative"):
            source = fr.SYMBOLS / PLATES[ed] / f"{token}.png"
            target = Path(out) / "glyph" / SHORT[ed] / source.name
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(source.read_bytes())
            glyphs[ed][token] = f"glyph/{SHORT[ed]}/{source.name}"
    shown = []
    for it in items:
        item = {k: it.get(k) for k in SHOWN}
        if item["one"].get("section"):
            with Image.open(Path(out) / item["one"]["section"]) as im:
                item["one"] = {**item["one"], "w": im.width, "h": im.height}
        shown.append(item)
    html = (TEMPLATE.read_text(encoding="utf-8")
            .replace("__ITEMS__", json.dumps(shown, ensure_ascii=False))
            .replace("__GLYPHS__", json.dumps(glyphs, ensure_ascii=False)))
    path = Path(out) / "index.html"
    path.write_text(html, encoding="utf-8")
    return path


def build(recheck: Path = RECHECK, out: Path = OUT, key: Path = KEY) -> dict:
    items = [it for ed in EDITIONS for it in census(ed, recheck, key)]
    windows(items, recheck, out)
    write_key(items, key)
    path = page(items, out)
    return {"entries": {ed: sum(i["edition"] == ed for i in items) for ed in EDITIONS},
            "left_out_sealed_or_reserve": {ed: sum(r["code"] == TOKEN and r["page"] in skipped_pages(ed)
                                                   for r in _read(Path(recheck) / ed / "final_inventory_rechecked.csv"))
                                           for ed in EDITIONS},
            "opened_at": {s: sum((i["one"] or {}).get("from") == s for i in items)
                          for s in ("entry", "region", "neighbour", "court")},
            "no_window": [i["id"] for i in items if not (i["one"] or {}).get("section")],
            "key": str(Path(key).relative_to(ROOT)), "page": str(path.relative_to(ROOT))}


def read_rulings(path: Path = RULINGS) -> dict:
    if not Path(path).exists():
        return {}
    data = json.loads(Path(path).read_text("utf-8"))
    return data.get("rulings", data)


def changes(key: list[dict] | None = None, answers: dict | None = None) -> dict:
    """What the staged answers would change at each entry, for both symbols: the register's savings bank kept
    (*savings bank*, *both*) or dropped (*cooperative*, *neither*); a cooperative added where the answer prints
    one and the register has none, dropped where the register has one and the answer prints none; *cannot
    tell* and unanswered cards change nothing."""
    key = read_key() if key is None else key
    answers = read_rulings() if answers is None else answers
    out = {"keep_savings_bank": [], "drop_savings_bank": [], "add_cooperative": [], "drop_cooperative": [], "open": []}
    for it in key:
        v = str((answers.get(it["id"]) or {}).get("value") or "")
        if v not in ANSWERS or v == "cannot-tell":
            out["open"].append(it["id"])
            continue
        coop_held = str(it.get("cooperative_held")) == "True"
        out["keep_savings_bank" if v in ("savings-bank", "both") else "drop_savings_bank"].append(it["id"])
        if v in ("cooperative", "both") and not coop_held:
            out["add_cooperative"].append(it["id"])
        if v in ("savings-bank", "neither") and coop_held:
            out["drop_cooperative"].append(it["id"])
    return out


RULED_ON = "savings-bank census, 2026-11-12 (D-229)"


def as_rulings(edition: str, key: Path = KEY, rulings: Path = RULINGS, recheck: Path = RECHECK,
               problems: list | None = None) -> list[dict]:
    """The census answers as merge's author-step records (``merge.apply_chat``): ``{part, code, verdict, ruled,
    note, item}`` for both symbols at each answered entry — the savings bank *printed* on *savings bank* and
    *both*, *not printed* on *cooperative* and *neither*; the cooperative *printed* on *cooperative* and *both*,
    *not printed* on the other two. *Cannot tell* and unanswered cards give none.

    The key holds each entry as the merge's output wrote it, after its ``keys`` step; the author step meets
    it on the number the page prints, so the step is undone here as for the look-alike families
    (:func:`family_rulings.keys_undo`): a post-1910 court's number back to the page's, a German opening
    entry's row back to the court it stood under. A row moved in from two courts is a problem, left out."""
    from utils.recheck import family_rulings as fr
    if edition not in EDITIONS or not Path(key).exists():
        return []
    answers = read_rulings(rulings)
    if not answers:
        return []
    renumbered, moved = fr.keys_undo(_read(Path(recheck) / edition / "changes.csv"))
    out = []
    for it in (k for k in read_key(key) if k["edition"] == edition):
        answer = answers.get(it["id"]) or {}
        value = str(answer.get("value") or "")
        if value not in ANSWERS or value == "cannot-tell":
            continue
        page, district, muni, seq = it["page"], it["district"], it["muni_seq"], it["settlement_seq"]
        muni = renumbered.get((page, district, muni, seq), muni)
        before = moved.get((page, district, muni, seq), {}).get(TOKEN, set())
        if len(before) > 1:
            if problems is not None:
                problems.append({"id": it["id"], "edition": edition, "entry": it["entry"],
                                 "problem": f"the keys step moved the savings bank here from {sorted(before)}"})
            continue
        part = (page, next(iter(before)) if before else district, muni, seq)
        note = str(answer.get("note") or "")
        verdicts = {TOKEN: value in ("savings-bank", "both"), COOPERATIVE: value in ("cooperative", "both")}
        for code, printed in verdicts.items():
            out.append({"part": part, "code": code, "verdict": "printed" if printed else "not printed",
                        "ruled": RULED_ON, "note": note, "item": it["id"]})
    return out


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n")[0])
    ap.add_argument("what", choices=("build", "changes"))
    args = ap.parse_args(argv)
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    if args.what == "build":
        print(json.dumps(build(), indent=1, ensure_ascii=False))
    else:
        print(json.dumps({k: len(v) for k, v in changes().items()}, indent=1))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
