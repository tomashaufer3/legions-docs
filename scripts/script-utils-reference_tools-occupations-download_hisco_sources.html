How it works

  • History of Work — HISCO (IISG / Datalegend, CC-BY-SA 4.0). Five CSV assets served straight off the druid asset endpoint. This script fetches them. They supply the HISCO hierarchy (major → minor → unit → micro, with English labels and task descriptions), the 246k multilingual occupational title → HISCO links, and the status / relation / product modifier code lists.
  • HSN standardised, HISCO-coded occupational titles, release 2020.02 (IISG). Behind a click-through on the IISG site, so it is not fetched here — the script only checks the two CSVs are present and points at the landing page if they are not. Its value to this project is the HISCO → HISCLASS / HISCAM / SOCPO crosswalk; its Dutch title strings are of no use for Czech input.

Every file present afterwards — downloaded or hand-placed — is recorded in MANIFEST.TXT with byte size, SHA-256, source URL, retrieval date and licence, so the provenance of the whole folder lives in one auditable place.

Run: python utils/reference_tools/occupations/download_hisco_sources.py [--force] Emits: data/raw/hisco/{hisco_45_full_03.csv, hisco_occupation_link_combined.csv, status.csv, relation.csv, product.csv, MANIFEST.TXT} Consumed by utils/reference_tools/occupations/build_hisco_authority.py.

Idempotent: a file already on disk whose SHA-256 matches the recorded one is skipped. --force re-downloads regardless.

def sha256(path: Path) -> str:
    """SHA-256 of a file, read in chunks — these are tens of megabytes."""
    h = hashlib.sha256()
    with path.open("rb") as fh:
        for chunk in iter(lambda: fh.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()

    """Stream ``url`` to ``dest`` via a temp file, so a failed download leaves nothing.

    Returns True if the transfer was TLS-verified. Verification is attempted first and
    only relaxed if the local trust store rejects the chain — which it does on this
    project's Windows machine, where TLS is intercepted and even the ``certifi`` bundle

    ``utils/scraping/scraper_vuapraha.py``, and the ``curl --ssl-no-revoke`` note in

    expected byte size — check them if the provenance matters to you.

    try:
        _stream(url, dest, verify=True)
        return True
    except requests.exceptions.SSLError:
        import urllib3
        from urllib3.exceptions import InsecureRequestWarning
        urllib3.disable_warnings(InsecureRequestWarning)

def download_druid(force: bool = False) -> dict[str, str]:
    """Fetch the five druid assets. Returns filename → sha256 for those on disk."""
    RAW.mkdir(parents=True, exist_ok=True)
    digests: dict[str, str] = {}
    unverified = 0
    for name, (asset_id, expected_size, _role) in DRUID_ASSETS.items():
        dest = RAW / name
        if dest.exists() and not force:
            digests[name] = sha256(dest)
            print(f"  skip     {name:<38} {dest.stat().st_size:>10,} B  (present)")
            continue
        print(f"  download {name:<38} ...", end="", flush=True)
        verified = fetch(DRUID_BASE + asset_id, dest)
        unverified += not verified
        size = dest.stat().st_size
        digests[name] = sha256(dest)
        flag = "" if size == expected_size else f"  ** size differs from {expected_size:,} **"
        tls = "" if verified else "  [TLS unverified]"
        print(f"\r  download {name:<38} {size:>10,} B{flag}{tls}")
    if unverified:
        print(f"\n  note: {unverified} file(s) fetched with TLS verification relaxed — the "
              f"local trust store rejects this host's chain.\n"
              f"        Sizes matched the published values; SHA-256 recorded in MANIFEST.TXT.")
    return digests

def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[1])
    ap.add_argument("--force", action="store_true",
                    help="re-download the druid assets even if already on disk")
    args = ap.parse_args(argv)

    print(f"HISCO sources -> {RAW}")
    print("\nHistory of Work (druid):")
    digests = download_druid(force=args.force)
    print("\nHSN release 2020.02:")
    digests |= check_hsn()

    manifest = write_manifest(digests)
    print(f"\nManifest: {manifest}  ({len(digests)} files recorded)")
    return 0

=================================================================================

def sha256(path: Path) -> str:
    """SHA-256 of a file, read in chunks — these are tens of megabytes."""
    h = hashlib.sha256()
    with path.open("rb") as fh:
        for chunk in iter(lambda: fh.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def _stream(url: str, dest: Path, verify: bool) -> None:
    import requests

    tmp = dest.with_suffix(dest.suffix + ".part")
    with requests.get(url, stream=True, timeout=120, verify=verify) as r:
        r.raise_for_status()
        with tmp.open("wb") as fh:
            for chunk in r.iter_content(chunk_size=1 << 20):
                fh.write(chunk)
    tmp.replace(dest)


def fetch(url: str, dest: Path) -> bool:

    only relaxed if the local trust store rejects the chain — which it does on this
    project's Windows machine, where TLS is intercepted and even the ``certifi`` bundle
    fails. The same relaxation is already needed elsewhere in the repo (see
    ``utils/scraping/scraper_vuapraha.py``, and the ``curl --ssl-no-revoke`` note in
    DATA_SOURCES.md §3). Integrity is covered instead by the recorded SHA-256 and the
    expected byte size — check them if the provenance matters to you.

    try:
        _stream(url, dest, verify=True)
        return True
    except requests.exceptions.SSLError:
        import urllib3
        from urllib3.exceptions import InsecureRequestWarning
        urllib3.disable_warnings(InsecureRequestWarning)
        _stream(url, dest, verify=False)
        return False

                    help="re-download the druid assets even if already on disk")
    args = ap.parse_args(argv)

    print(f"HISCO sources -> {RAW}")

    digests |= check_hsn()

    print(f"\nManifest: {manifest}  ({len(digests)} files recorded)")
    return 0
