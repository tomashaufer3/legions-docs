How it works

Stage 1 (statutory_prose.py) left one verified transcription per page. This turns each edition's text into one row per printed institution: the city, the exact words (quote), the printed count, the language split where one is printed (18 Bürgerschulen (14 deutsch, 4 böhmisch)), a category, and the code of institution_abbreviations.csv whose meaning it is, where one is.

read one call per edition and reader, Gemini 3.8 Flash and Opus 5.5, independently, with the edition's text, its cities in printed order and the vocabulary of its language. Cached; paid only with --spend, on Heavyhorse6 (D-68). About $2.50-3 for both: ~2,100 rows of ~45 tokens, mostly Opus output at $20 per million. check each row against the text, by rule: the quote occurs in the text verbatim; the count is the number the quote opens with (1 where it opens with none); a language split sums to the count; the code is in the vocabulary. compare the two readers row by row on (city, quote): a row both give alike is settled; a row one reader gives, or the two give differently, is an open group. judge a third, blind reading of the open groups by Sonnet 5.5, neither reader; a group it settles with every row clean is adjudicated; a group the author rules on (statutory_institution_rulings.csv) is author; the rest go to review.csv. cross Prague and Liberec are printed in both Bohemian editions: their counts by category, Czech against German, are a check neither reader saw.

Writes output/statutory_prose/institutions/: &lt;edition&gt;_&lt;model&gt;.json (the replies), institutions.csv (every row, both readers side by side), review.csv and summary.txt.

    parts = []

        path = sp.OUT / f"{edition}_{scan:04d}" / "aligned.txt"

def read(edition: str, model: str, spend: bool) -> dict | None:
    """One reader's rows for one edition, cached by prompt and model."""
    import hashlib

    text = prompt(edition)
    key = hashlib.sha256((text + model).encode()).hexdigest()[:16]
    path = OUT / f"{edition}_{model}_{key}.json"
    if path.exists():
        return json.loads(path.read_text(encoding="utf-8"))
    if not spend:
        return None
    from utils.lexikon_1910 import lexikon_1910_pages as lxp

    lxp.load_credentials()
    if model == sp.GEMINI:
        from openai import OpenAI

        client = OpenAI(base_url=sp.GEMINI_URL, api_key=os.environ["GEMINI_API_KEY"])
        # As the route's OpenAI-compatible reader: a strict schema first, plain JSON if the endpoint
        # rejects the format (run_lexikon_1910_pilot.OpenAICompatibleReader.formats).
        formats = [{"type": "json_schema", "json_schema": {"name": "rows", "schema": schema(edition), "strict": True}},
                   {"type": "json_object"}]
        for k, fmt in enumerate(formats):
            try:
                reply = client.chat.completions.create(
                    model=model, temperature=0.0, reasoning_effort="none", max_tokens=MAX_TOKENS,
                    response_format=fmt, messages=[{"role": "user", "content": text}])
                break
            except Exception as exc:                                      # noqa: BLE001
                if k + 1 == len(formats) or getattr(exc, "status_code", None) != 400:
                    raise
                print(f"  {edition} {model}: the endpoint rejected the schema ({exc}); plain JSON")
        if reply.choices[0].finish_reason == "length":
            raise SystemExit(f"{edition} {model}: the reply hit {MAX_TOKENS} tokens and was cut off; nothing cached")
        raw = reply.choices[0].message.content or ""
        tokens_in, tokens_out = reply.usage.prompt_tokens, reply.usage.total_tokens - reply.usage.prompt_tokens
    else:
        import anthropic

        client = anthropic.Anthropic(default_headers={"Accept-Encoding": "gzip, deflate"})   # as statutory_prose
        with client.messages.stream(
                model=model, max_tokens=MAX_TOKENS,
                output_config={"effort": "low", "format": {"type": "json_schema", "schema": schema(edition)}},
                messages=[{"role": "user", "content": text}]) as stream:
            reply = stream.get_final_message()
        if reply.stop_reason == "max_tokens":
            raise SystemExit(f"{edition} {model}: the reply hit {MAX_TOKENS} tokens and was cut off; nothing cached")
        raw = "".join(b.text for b in reply.content if getattr(b, "type", "") == "text")
        tokens_in, tokens_out = reply.usage.input_tokens, reply.usage.output_tokens
    p_in, p_out = sp.PRICE[model]
    record = {"edition": edition, "model": model, "rows": json.loads(raw)["items"],
              "tokens_in": tokens_in, "tokens_out": tokens_out,
              "cost_usd": round((tokens_in * p_in + tokens_out * p_out) / 1e6, 4)}
    OUT.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(record, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"  {edition} {model}: {len(record['rows'])} rows, ${record['cost_usd']:.3f}")
    return record

def check(row: dict, text_flat: str, codes: set[str], lenient: bool = False) -> list[str]:
    """What is wrong with one row, by rule; empty when nothing is. ``lenient`` reads the count
    by :data:`LENIENT_COUNT`: the judge's rows are accepted by it, while the readers' rows keep
    the strict rule, so which groups go to the judge — and so its cached prompt — is unchanged."""
    out = []
    quote = _flat(row["quote"])
    if not quote or quote not in text_flat:
        out.append("quote not in the text")
    lead = (LENIENT_COUNT if lenient else STRICT_COUNT).match(quote)
    if (int(lead.group(1)) if lead else 1) != row["count"]:
        out.append(f"count {row['count']} is not the printed {lead.group(1) if lead else 'none (1)'}")

        out.append("languages do not sum to the count")
    if row["code"] and row["code"] not in codes:
        out.append(f"code {row['code']!r} not in the vocabulary")
    return out

    if code and code not in codes and expansions and code in expansions:
        return expansions[code]
    return code

def span_groups(text_flat: str, gemini: list[dict], opus: list[dict], codes: set[str]) -> list[dict]:
    """Both readers' rows grouped by where their words sit in the text: rows whose spans overlap
    are one group, so a mention one reader cut finer, or quoted a word longer, pairs with the
    other's. Each reader's rows are found in printed order (a cursor), the same words printed twice
    taking the next occurrence. ``{"start", "end", "city", "gemini": [...], "opus": [...]}``; a row
    whose words are nowhere in the text is a group of its own at the end."""
    placed, lost = [], []
    for name, rows in (("gemini", gemini), ("opus", opus)):
        cursor = 0
        for row in rows:
            row = {**row, "code": canonical_code(row["code"], codes)}
            quote = _flat(row["quote"])
            at = text_flat.find(quote, cursor) if quote else -1
            at = text_flat.find(quote) if at < 0 and quote else at
            if at < 0:
                lost.append({"start": -1, "end": -1, "city": row["city"], "gemini": [], "opus": [], name: [row]})
                continue
            cursor = at + len(quote)
            placed.append((at, at + len(quote), name, row))
    groups: list[dict] = []
    for start, end, name, row in sorted(placed, key=lambda t: (t[0], t[1])):
        if groups and start < groups[-1]["end"]:
            g = groups[-1]
            g["end"] = max(g["end"], end)
        else:
            g = {"start": start, "end": end, "city": row["city"], "gemini": [], "opus": []}
            groups.append(g)
        g[name].append(row)
    return groups + lost

PROMPT = """Below is the verified transcription of the statutory cities' section of a 1910 gazetteer of
Austria, in {language}. The cities, in printed order: {cities}. Each page is headed "=== page N ===".

List every office, authority, court, school, church or religious house, hospital or charitable
institution, museum or library, association, bank or savings bank, enterprise, military command or
unit, transport or postal office that the text names. One row per printed mention, in printed order:

- city: the city the mention belongs to, written exactly as in the list above;
- quote: the words of the mention copied from the text character for character, including a number

- count: the number the mention opens with, or 1 where it opens with none;
- languages: where the text prints how the count divides by language ("(14 deutsch, 4 böhmisch)",
  "(český a německý)"), one entry per language with its count; else an empty list;
- category: one of {categories};
- code: the code from the vocabulary below whose meaning the mention is, or "" where none is.

Leave out the city's districts and their populations, altitudes, distances, footnotes, and the city
itself. A mention of branches ("se 2 pobočkami", "mit 2 Exposituren") is a row of its own.
Return one JSON object: {{"items": [the rows]}}.

Vocabulary (code | English | {language} expansion):
{vocabulary}

The text:
{text}"""


def vocabulary(lang: str) -> list[dict]:
    with VOCABULARY.open(encoding="utf-8", newline="") as fh:
        return [r for r in csv.DictReader(fh)
                if r["lang"] == lang and r["category"] not in ("settlement", "unit")
                and r["kind"] not in ("qualifier", "unit")]


def edition_text(edition: str) -> str:
    """The verified pages of an edition, in order, each headed."""
    parts = []
    for scan in sp.PAGES[edition]:
        path = sp.OUT / f"{edition}_{scan:04d}" / "aligned.txt"
        if not path.exists():
            raise SystemExit(f"{path.relative_to(ROOT)} missing: run statutory_prose.py first")
        text = path.read_text(encoding="utf-8")
        if "[[" in text:
            raise SystemExit(f"{path.relative_to(ROOT)} has an unsettled disagreement: rule it first")
        parts.append(f"=== page {scan} ===\n{text.strip()}")
    return "\n\n".join(parts)


def prompt(edition: str) -> str:
    lang = LANGUAGE[edition]
    vocab = "\n".join(f"{r['code']} | {r['english']} | {r['expansion']}" for r in vocabulary(lang))
    return PROMPT.format(language="Czech" if lang == "cs" else "German", cities=", ".join(sp.CITIES[edition]),
                         categories=", ".join(CATEGORIES), vocabulary=vocab, text=edition_text(edition))


def schema(edition: str) -> dict:
    row = {"type": "object", "additionalProperties": False,
           "required": ["city", "quote", "count", "languages", "category", "code"],
           "properties": {
               "city": {"type": "string", "enum": list(sp.CITIES[edition])},
               "quote": {"type": "string"}, "count": {"type": "integer"},
               "languages": {"type": "array", "items": {
                   "type": "object", "additionalProperties": False, "required": ["language", "count"],
                   "properties": {"language": {"type": "string", "enum": list(LANGUAGES)},
                                  "count": {"type": "integer"}}}},
               "category": {"type": "string", "enum": list(CATEGORIES)},
               "code": {"type": "string"}}}
    return {"type": "object", "additionalProperties": False, "required": ["items"],
            "properties": {"items": {"type": "array", "items": row}}}

def read(edition: str, model: str, spend: bool) -> dict | None:
    """One reader's rows for one edition, cached by prompt and model."""
    import hashlib

    text = prompt(edition)
    key = hashlib.sha256((text + model).encode()).hexdigest()[:16]
    path = OUT / f"{edition}_{model}_{key}.json"
    if path.exists():
        return json.loads(path.read_text(encoding="utf-8"))
    if not spend:
        return None
    from utils.lexikon_1910 import lexikon_1910_pages as lxp

    lxp.load_credentials()
    if model == sp.GEMINI:
        from openai import OpenAI

        client = OpenAI(base_url=sp.GEMINI_URL, api_key=os.environ["GEMINI_API_KEY"])
        # As the route's OpenAI-compatible reader: a strict schema first, plain JSON if the endpoint
        # rejects the format (run_lexikon_1910_pilot.OpenAICompatibleReader.formats).
        formats = [{"type": "json_schema", "json_schema": {"name": "rows", "schema": schema(edition), "strict": True}},
                   {"type": "json_object"}]
        for k, fmt in enumerate(formats):
            try:
                reply = client.chat.completions.create(
                    model=model, temperature=0.0, reasoning_effort="none", max_tokens=MAX_TOKENS,
                    response_format=fmt, messages=[{"role": "user", "content": text}])
                break
            except Exception as exc:                                      # noqa: BLE001
                if k + 1 == len(formats) or getattr(exc, "status_code", None) != 400:
                    raise
                print(f"  {edition} {model}: the endpoint rejected the schema ({exc}); plain JSON")
        if reply.choices[0].finish_reason == "length":
            raise SystemExit(f"{edition} {model}: the reply hit {MAX_TOKENS} tokens and was cut off; nothing cached")
        raw = reply.choices[0].message.content or ""
        tokens_in, tokens_out = reply.usage.prompt_tokens, reply.usage.total_tokens - reply.usage.prompt_tokens
    else:
        import anthropic

        client = anthropic.Anthropic(default_headers={"Accept-Encoding": "gzip, deflate"})   # as statutory_prose
        with client.messages.stream(
                model=model, max_tokens=MAX_TOKENS,
                output_config={"effort": "low", "format": {"type": "json_schema", "schema": schema(edition)}},
                messages=[{"role": "user", "content": text}]) as stream:
            reply = stream.get_final_message()
        if reply.stop_reason == "max_tokens":
            raise SystemExit(f"{edition} {model}: the reply hit {MAX_TOKENS} tokens and was cut off; nothing cached")
        raw = "".join(b.text for b in reply.content if getattr(b, "type", "") == "text")
        tokens_in, tokens_out = reply.usage.input_tokens, reply.usage.output_tokens
    p_in, p_out = sp.PRICE[model]
    record = {"edition": edition, "model": model, "rows": json.loads(raw)["items"],
              "tokens_in": tokens_in, "tokens_out": tokens_out,
              "cost_usd": round((tokens_in * p_in + tokens_out * p_out) / 1e6, 4)}
    OUT.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(record, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"  {edition} {model}: {len(record['rows'])} rows, ${record['cost_usd']:.3f}")
    return record

def _flat(text: str) -> str:
    """Text compared for a quote: one space, the hyphen of a joined line break ignored."""
    text = sp.normalise(text).replace("¶", " ")
    return re.sub(r"\s+", " ", re.sub(r"\s*-\s*", "-", text)).strip()


#: Where a mention's count is printed. Strict, as the readers were told: the number the words open
#: with. Lenient, for the judge's rows (0.4.0): also after an opening bracket or a preposition —
#: *se 7 pobočkami*, *(mit 3 Filialen)*, *s 2 odbory* — where the count is printed all the same.
STRICT_COUNT = re.compile(r"(\d+)\s")
LENIENT_COUNT = re.compile(r"\(?(?:(?:se|s|ze|z|mit)\s+)?(\d+)\s")


def check(row: dict, text_flat: str, codes: set[str], lenient: bool = False) -> list[str]:
    """What is wrong with one row, by rule; empty when nothing is. ``lenient`` reads the count
    by :data:`LENIENT_COUNT`: the judge's rows are accepted by it, while the readers' rows keep
    the strict rule, so which groups go to the judge — and so its cached prompt — is unchanged."""
    out = []
    quote = _flat(row["quote"])
    if not quote or quote not in text_flat:
        out.append("quote not in the text")
    lead = (LENIENT_COUNT if lenient else STRICT_COUNT).match(quote)
    if (int(lead.group(1)) if lead else 1) != row["count"]:
        out.append(f"count {row['count']} is not the printed {lead.group(1) if lead else 'none (1)'}")
    if row["languages"] and sum(x["count"] for x in row["languages"]) != row["count"]:
        out.append("languages do not sum to the count")
    if row["code"] and row["code"] not in codes:
        out.append(f"code {row['code']!r} not in the vocabulary")
    return out


def _langs(row: dict) -> str:
    return ", ".join(f"{x['language']} {x['count']}" for x in sorted(row["languages"], key=lambda x: x["language"]))


def canonical_code(code: str, codes: set[str], expansions: dict[str, str] | None = None) -> str:
    """A reader's code as the vocabulary writes it: ``cadastre-office`` → ``[cadastre-office]``;
    with ``expansions`` (:func:`expansion_codes`), a code given as its expansion → the code
    (``Steueramt`` → ``Sta.``)."""
    code = (code or "").strip()
    if code and code not in codes and f"[{code}]" in codes:
        return f"[{code}]"
    if code and code not in codes and expansions and code in expansions:
        return expansions[code]
    return code


def expansion_codes(vocab: list[dict]) -> dict[str, str]:
    """``{expansion: code}`` for every expansion all of whose codes mean the same in English:
    ``Steueramt`` is printed as ``Sta.`` and ``StA.``, both *tax office*, and takes the first in
    the vocabulary's order. An expansion two meanings share maps to nothing."""
    by: dict[str, list[dict]] = defaultdict(list)
    for v in vocab:
        if v.get("expansion"):
            by[v["expansion"]].append(v)
    return {e: rows[0]["code"] for e, rows in by.items() if len({r.get("english") for r in rows}) == 1}


def span_groups(text_flat: str, gemini: list[dict], opus: list[dict], codes: set[str]) -> list[dict]:
    """Both readers' rows grouped by where their words sit in the text: rows whose spans overlap
    are one group, so a mention one reader cut finer, or quoted a word longer, pairs with the
    other's. Each reader's rows are found in printed order (a cursor), the same words printed twice
    taking the next occurrence. ``{"start", "end", "city", "gemini": [...], "opus": [...]}``; a row
    whose words are nowhere in the text is a group of its own at the end."""
    placed, lost = [], []
    for name, rows in (("gemini", gemini), ("opus", opus)):
        cursor = 0
        for row in rows:
            row = {**row, "code": canonical_code(row["code"], codes)}
            quote = _flat(row["quote"])
            at = text_flat.find(quote, cursor) if quote else -1
            at = text_flat.find(quote) if at < 0 and quote else at
            if at < 0:
                lost.append({"start": -1, "end": -1, "city": row["city"], "gemini": [], "opus": [], name: [row]})
                continue
            cursor = at + len(quote)
            placed.append((at, at + len(quote), name, row))
    groups: list[dict] = []
    for start, end, name, row in sorted(placed, key=lambda t: (t[0], t[1])):
        if groups and start < groups[-1]["end"]:
            g = groups[-1]
            g["end"] = max(g["end"], end)
        else:
            g = {"start": start, "end": end, "city": row["city"], "gemini": [], "opus": []}
            groups.append(g)
        g[name].append(row)
    return groups + lost


def _same(a: dict, b: dict) -> list[str]:
    """The fields two rows differ in."""
    return [f for f in ("count", "category", "code") if a[f] != b[f]] + (["languages"] if _langs(a) != _langs(b) else [])


def compare(edition: str, gemini: list[dict], opus: list[dict]) -> list[dict]:
    """Both readers' rows compared group by group (:func:`span_groups`). A group of one row from
    each, alike in count, category, code and language split and clean by :func:`check`, is
    ``agree``; else the group is open: ``differ: <fields>``, ``split`` (one reader cut it finer),
    ``gemini only`` / ``opus only``. One output row per group, with both readers' rows as JSON."""
    text_flat = _flat(edition_text(edition))
    codes = {r["code"] for r in vocabulary(LANGUAGE[edition])}
    out = []
    for k, g in enumerate(span_groups(text_flat, gemini, opus, codes)):
        a, b = g["gemini"], g["opus"]
        problems = sorted({p for row in a + b for p in check(row, text_flat, codes)})
        if len(a) == 1 and len(b) == 1:
            differ = _same(a[0], b[0])
            status = "differ: " + ", ".join(differ) if differ else "agree"
        else:
            status = "split" if a and b else "gemini only" if a else "opus only"
        base = (a or b)[0]
        out.append({"edition": edition, "city": g["city"], "group": k, "quote": base["quote"],
                    "count": base["count"], "languages": _langs(base), "category": base["category"],
                    "code": base["code"], "status": status, "problems": "; ".join(problems),
                    "start": g["start"], "end": g["end"],
                    "gemini_rows": json.dumps(a, ensure_ascii=False), "opus_rows": json.dumps(b, ensure_ascii=False)})
    return out

#: The judge of the open groups. Sonnet 5.5 by default, the model that is neither reader: Opus judging
#: the groups where it disagreed with Gemini would rule on its own answers (``--judge opus`` keeps it
#: as an option). The record (2026-11-01) is Sonnet's, so a cached re-run (notebook 15 §21) is free
#: and changes nothing. Either sees the two readings as A and B, in an order set per group.
JUDGES = {"opus": sp.OPUS, "sonnet": "claude-sonnet-5-5"}
JUDGE_PRICE = {sp.OPUS: sp.PRICE[sp.OPUS], "claude-sonnet-5-5": (2.00, 10.00)}   # USD per Mtok in / out

JUDGE_PROMPT = """Two readers listed the institutions named in a 1910 gazetteer's description of a
statutory city ({language}). Below are the stretches of the text where they disagree. For each
stretch, give the correct rows, by the same rules the readers followed:

- one row per printed mention; a mention of branches ("se 2 pobočkami", "mit 2 Exposituren") is a
  row of its own;
- quote: the mention's words copied from the stretch character for character, including a number
  that opens it; count: that number, or 1;
- languages: the language split where the text prints one, else an empty list;
- category: one of {categories}; code: the vocabulary's code whose meaning the mention is, or "".
Reading A and reading B are suggestions, either may be wrong; follow the text and the rules. A
stretch that names no institution (a place, a number, a footnote) gets no rows.

Vocabulary (code | English | {language} expansion):
{vocabulary}

The stretches:
{groups}

Return one JSON object: {{"groups": [{{"group": <its number>, "rows": [the rows]}}]}}."""


def _order(edition: str, group: int) -> bool:
    """Whether Gemini is reading A for this group: fixed by a hash, so a re-run asks the same."""
    import hashlib

    return hashlib.sha256(f"{edition}:{group}".encode()).digest()[0] % 2 == 0


def judge_prompt(edition: str, open_rows: list[dict]) -> str:
    text_flat = _flat(edition_text(edition))
    lang = LANGUAGE[edition]
    blocks = []
    for r in open_rows:
        start, end = int(r["start"]), int(r["end"])
        excerpt = text_flat[max(0, start - 60):end + 60] if start >= 0 else r["quote"]
        a, b = json.loads(r["gemini_rows"]), json.loads(r["opus_rows"])
        if not _order(edition, int(r["group"])):
            a, b = b, a
        strip = lambda rows: [{k: v for k, v in x.items() if k != "city"} for x in rows]  # noqa: E731
        blocks.append(f"group {r['group']} (city: {r['city']})\n  text: …{excerpt}…\n"
                      f"  reading A: {json.dumps(strip(a), ensure_ascii=False)}\n"
                      f"  reading B: {json.dumps(strip(b), ensure_ascii=False)}")
    vocab = "\n".join(f"{v['code']} | {v['english']} | {v['expansion']}" for v in vocabulary(lang))
    return JUDGE_PROMPT.format(language="Czech" if lang == "cs" else "German", categories=", ".join(CATEGORIES),
                               vocabulary=vocab, groups="\n\n".join(blocks))


def judge_schema() -> dict:
    row = {"type": "object", "additionalProperties": False,
           "required": ["quote", "count", "languages", "category", "code"],
           "properties": {"quote": {"type": "string"}, "count": {"type": "integer"},
                          "languages": schema("cz")["properties"]["items"]["items"]["properties"]["languages"],
                          "category": {"type": "string", "enum": list(CATEGORIES)}, "code": {"type": "string"}}}
    group = {"type": "object", "additionalProperties": False, "required": ["group", "rows"],
             "properties": {"group": {"type": "integer"}, "rows": {"type": "array", "items": row}}}
    return {"type": "object", "additionalProperties": False, "required": ["groups"],
            "properties": {"groups": {"type": "array", "items": group}}}


def judge(edition: str, open_rows: list[dict], model: str, spend: bool) -> dict | None:
    """The judge's rows for the open groups of one edition, cached by prompt and model."""
    import hashlib

    if not open_rows:
        return {"groups": [], "cost_usd": 0.0}
    text = judge_prompt(edition, open_rows)
    key = hashlib.sha256((text + model).encode()).hexdigest()[:16]
    path = OUT / f"judge_{edition}_{model}_{key}.json"
    if path.exists():
        return json.loads(path.read_text(encoding="utf-8"))
    if not spend:
        return None
    import anthropic
    from utils.lexikon_1910 import lexikon_1910_pages as lxp

    lxp.load_credentials()
    client = anthropic.Anthropic(default_headers={"Accept-Encoding": "gzip, deflate"})   # as statutory_prose
    params = dict(model=model, max_tokens=MAX_TOKENS, messages=[{"role": "user", "content": text}],
                  output_config={"effort": "medium", "format": {"type": "json_schema", "schema": judge_schema()}})
    try:                         # a refused request re-runs on the fallback model (server-side, default routing)
        with client.beta.messages.stream(**params, betas=["server-side-fallback-2026-07-01"],
                                         fallbacks="default") as stream:
            reply = stream.get_final_message()
    except TypeError:            # an SDK without the fallbacks parameter: the plain request
        with client.messages.stream(**params) as stream:
            reply = stream.get_final_message()
    if reply.stop_reason in ("max_tokens", "refusal"):
        raise SystemExit(f"{edition} judge {model}: stop_reason {reply.stop_reason}; nothing cached")
    raw = "".join(b.text for b in reply.content if getattr(b, "type", "") == "text")
    p_in, p_out = JUDGE_PRICE[model]
    record = {"edition": edition, "model": reply.model, "groups": json.loads(raw)["groups"],
              "tokens_in": reply.usage.input_tokens, "tokens_out": reply.usage.output_tokens,
              "cost_usd": round((reply.usage.input_tokens * p_in + reply.usage.output_tokens * p_out) / 1e6, 4)}
    path.write_text(json.dumps(record, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"  {edition} judge {model}: {len(record['groups'])} groups, ${record['cost_usd']:.3f}")
    return record


def adjudicated(edition: str, open_rows: list[dict], verdict: dict) -> tuple[list[dict], list[dict]]:
    """``(settled rows, groups still open)``: a group is settled when the judge returned it and
    every row it gave is clean by :func:`check` and lies inside the group's stretch of text."""
    text_flat = _flat(edition_text(edition))
    vocab = vocabulary(LANGUAGE[edition])
    codes, expansions = {r["code"] for r in vocab}, expansion_codes(vocab)
    given = {int(g["group"]): g["rows"] for g in verdict.get("groups", [])}
    settled, still = [], []
    for r in open_rows:
        rows = given.get(int(r["group"]))
        start, end = int(r["start"]), int(r["end"])
        stretch = text_flat[max(0, start - 60):end + 60] if start >= 0 else ""
        rows = [{**x, "code": canonical_code(x["code"], codes, expansions)} for x in rows or []]
        ok = rows is not None and int(r["group"]) in given and all(
            not check(x, text_flat, codes, lenient=True) and _flat(x["quote"]) in stretch for x in rows)
        if not ok:
            still.append({**r, "status": r["status"] + " (judge unsettled)"})
            continue
        settled += [{"edition": edition, "city": r["city"], "group": r["group"], "quote": x["quote"],
                     "count": x["count"], "languages": _langs(x), "category": x["category"],
                     "code": x["code"], "status": "adjudicated", "problems": "", "start": start, "end": end,
                     "gemini_rows": r["gemini_rows"], "opus_rows": r["opus_rows"]} for x in rows]
    return settled, still


#: The author's rulings on groups neither the readers nor the judge settled (0.5.0): one line per
#: row the group becomes, keyed by edition, city and the group's quote as ``review.csv`` prints it.
RULINGS = ROOT / "reference" / "places" / "statutory_institution_rulings.csv"


def load_rulings(path: Path = RULINGS) -> dict[tuple[str, str, str], list[dict]]:
    """``{(edition, city, group quote): [rows]}``; a language split is written ``czech 1; german 1``."""
    out: dict[tuple[str, str, str], list[dict]] = defaultdict(list)
    if not Path(path).exists():
        return out
    with Path(path).open(encoding="utf-8", newline="") as fh:
        for r in csv.DictReader(fh):
            langs = [{"language": lang, "count": int(n)}
                     for lang, n in (part.strip().rsplit(" ", 1) for part in r["languages"].split(";") if part.strip())]
            out[(r["edition"], r["city"], r["group_quote"])].append(
                {"quote": r["quote"], "count": int(r["count"]), "languages": langs,
                 "category": r["category"], "code": r["code"], "ruled_by": r["ruled_by"]})
    return out


def apply_rulings(edition: str, open_rows: list[dict],
                  rulings: dict | None = None) -> tuple[list[dict], list[dict], list[str]]:
    """``(ruled rows, groups still open, rulings that matched no open group)``. A ruling is the
    author's: its count need not be printed (two departments named, not numbered), but its words
    must lie in the group's stretch of text and a language split must sum to its count."""
    rulings = load_rulings() if rulings is None else rulings
    text_flat = _flat(edition_text(edition))
    ruled, still, used = [], [], set()
    for r in open_rows:
        key = (edition, r["city"], r["quote"])
        rows = rulings.get(key)
        start, end = int(r["start"]), int(r["end"])
        stretch = text_flat[max(0, start - 60):end + 60] if start >= 0 else ""
        if not rows or not all(_flat(x["quote"]) in stretch and
                               (not x["languages"] or sum(l["count"] for l in x["languages"]) == x["count"])
                               for x in rows):
            still.append(r)
            continue
        used.add(key)
        ruled += [{"edition": edition, "city": r["city"], "group": r["group"], "quote": x["quote"],
                   "count": x["count"], "languages": _langs(x), "category": x["category"],
                   "code": x["code"], "status": "author", "problems": "", "start": start, "end": end,
                   "gemini_rows": r["gemini_rows"], "opus_rows": r["opus_rows"]} for x in rows]
    unused = [" / ".join(k) for k in rulings if k[0] == edition and k not in used]
    return ruled, still, unused


#: What introduces a printed enumeration after a count: "181 zapsaných … družstev a sice: 74 …",
#: "181 registrierte … und zwar: 74 …", "5 železničních stanic: státní nádraží …".
ENUMERATION = re.compile(r"\s*(?:\([^)]*\)\s*)?(?:a sice|und zwar|darunter|davon|z toho|mezi nimi)?\s*:")


def mark_totals(edition: str, rows: list[dict]) -> list[dict]:
    """``is_total`` on every row whose words the text follows with an enumeration (``ENUMERATION``):
    a total printed before its own members, which the readers mostly list as rows of their own.
    Sum either the totals or the members, never both (the German Prague total of 181
    cooperatives beside its breakdown was the whole of a 180-cooperative gap to the Czech edition)."""
    text_flat = _flat(edition_text(edition))
    for r in rows:
        quote = _flat(r["quote"])
        at = text_flat.find(quote, max(0, int(r.get("start") or 0) - 60)) if quote else -1
        r["is_total"] = bool(at >= 0 and ENUMERATION.match(text_flat, at + len(quote)))
    return rows


def by_category(rows: list[dict], city: str) -> dict[str, int]:
    """Settled institutions by category, totals left out (their members are counted)."""
    out = defaultdict(int)
    for r in rows:
        if (r["city"] == city and r["status"] in ("agree", "adjudicated", "author") and not r["problems"]
                and not r.get("is_total")):
            out[r["category"]] += int(r["count"])
    return dict(out)

def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--spend", action="store_true", help="read what is not cached (Heavyhorse6)")
    parser.add_argument("--judge", choices=sorted(JUDGES), default="sonnet",
                        help="the model that settles the open groups (sonnet: the default, neither reader)")
    parser.add_argument("--allow-other-machine", action="store_true", help="break D-68 on purpose")
    args = parser.parse_args(argv)
    if args.spend and sp._host() != sp.MACHINE_OF_RECORD and not args.allow_other_machine:
        raise SystemExit(f"this is {sp._host()}, not {sp.MACHINE_OF_RECORD} (D-68); nothing was sent")

    judge_model = JUDGES[args.judge]
    rows, lines, spent, missing, judged = [], [], 0.0, [], 0.0
    for edition in sp.PAGES:
        got = {m: read(edition, m, args.spend) for m in (sp.GEMINI, sp.OPUS)}
        if any(v is None for v in got.values()):
            words = len(prompt(edition).split())
            missing.append(f"{edition} (~{words:,} words of prompt)")
            continue
        spent += sum(v["cost_usd"] for v in got.values())
        groups = compare(edition, got[sp.GEMINI]["rows"], got[sp.OPUS]["rows"])
        agree = [r for r in groups if r["status"] == "agree" and not r["problems"]]
        open_rows = [r for r in groups if r not in agree]
        verdict = judge(edition, open_rows, judge_model, args.spend)
        if verdict is None:
            settled, still = [], open_rows
            missing.append(f"{edition} judge ({len(open_rows)} open groups)")
        else:
            judged += verdict.get("cost_usd", 0.0)
            settled, still = adjudicated(edition, open_rows, verdict)
        ruled, still, unused = apply_rulings(edition, still)
        if unused:
            lines.append(f"  !! {edition}: rulings that match no open group (the grouping changed?): "
                         + "; ".join(unused))
        mine = mark_totals(edition, agree + settled + ruled) + [{**r, "is_total": False} for r in still]
        rows += mine
        for city in sp.CITIES[edition]:
            a = [r for r in agree if r["city"] == city]
            s = [r for r in settled if r["city"] == city]
            u = [r for r in ruled if r["city"] == city]
            o = [r for r in still if r["city"] == city]
            counted = [r for r in a + s + u if not r.get("is_total")]
            lines.append(f"  {edition} {city}: {len(a)} agreed, {len(s)} adjudicated, "
                         f"{len({r['group'] for r in u})} ruled by the author, {len(o)} open groups; "
                         f"{sum(int(r['count']) for r in counted)} institutions settled "
                         f"({sum(1 for r in a + s + u if r.get('is_total'))} printed totals left out)")
    head = [f"Statutory-city institutions, stage 2 ({__version__}): readers {sp.GEMINI} + {sp.OPUS}, judge {judge_model}"]
    if missing:
        head.append(f"  not done yet: {', '.join(missing)}; --spend on {sp.MACHINE_OF_RECORD}")
    head.append(f"  spent: readers ${spent:.2f}, judge ${judged:.2f}")
    cross = []
    if any(r["edition"] == "cz" for r in rows) and any(r["edition"] == "de-bohemia" for r in rows):
        for cz_city, de_city in (("Praha", "Prag"), ("Liberec", "Reichenberg")):
            a = by_category([r for r in rows if r["edition"] == "cz"], cz_city)
            b = by_category([r for r in rows if r["edition"] == "de-bohemia"], de_city)
            diff = {c: (a.get(c, 0), b.get(c, 0)) for c in sorted(set(a) | set(b)) if a.get(c, 0) != b.get(c, 0)}
            cross.append(f"  {cz_city}/{de_city}, settled institutions by category, Czech against German: "
                         + ("the same" if not diff else "; ".join(f"{c} {x}/{y}" for c, (x, y) in diff.items())))
    text = "\n".join(head + lines + (["Cross-edition check:"] + cross if cross else []))
    print(text)
    OUT.mkdir(parents=True, exist_ok=True)
    (OUT / "summary.txt").write_text(text + "\n", encoding="utf-8")
    if rows:
        fields = list(rows[0])
        for name, subset in (("institutions.csv", rows),
                             ("review.csv", [r for r in rows if r["status"] not in ("agree", "adjudicated", "author")])):
            with (OUT / name).open("w", encoding="utf-8", newline="") as fh:
                writer = csv.DictWriter(fh, fieldnames=fields)
                writer.writeheader()
                writer.writerows(subset)
    return 0
