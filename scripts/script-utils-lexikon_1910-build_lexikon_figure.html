How it works

The docs site is one self-contained HTML file built by utils/docs/build_docs.py, and it must not open a 3.34 GB gitignored PDF to draw a picture. So this script does the expensive half once and writes small tracked artifacts under reference/places/figure/: a handful of JPEGs and one JSON file holding every overlay box, every verbatim string the box covers, and the pilot's headline counts. build_docs.py then only reads that.

Every box is computed, never eyeballed. The Czech page has an ALTO layer whose frame is the scan's own pixel grid, so utils.lexikon_1910.lexikon_1910.render_page renders it 1:1 and the module's existing geometry — body_band, fit_columns, segment_blocks, notes_boxes, the per-line bounds extract_notes already returns — lands directly on the image with no scale factor to carry around. Boxes are then divided by the image size, so the figure survives any later change of render width.

The German side is coarser and says so. The UBW scan has no ALTO and no printed rules — page.rects and page.lines are empty — so its boxes come from pdfplumber word extents clustered into printed lines. That is enough for the one point the German page is on the sheet to make (the same municipality, a different page and a different position in the district, because each edition sorts an okres alphabetically in its own language), and not enough to draw a column grid on.

Run it from the repo root::

python utils/lexikon_1910/build_lexikon_figure.py

Nothing here calls a paid API or a model. It reads two PDFs, one ALTO file, the symbol atlas manifest and — if the pilot has been run — the pilot's check CSVs.

class Zone:
    """One clickable region of a page, in fractions of the image."""

    id: str
    x: float
    y: float
    w: float
    h: float
    label: str
    #: What the text layer actually read inside this box, verbatim. The point of
    #: the figure is the distance between the print and this string, so it is
    #: carried rather than paraphrased.
    read: str = ""
    kind: str = "zone"

    def as_dict(self) -> dict:
        return dict(id=self.id, x=round(self.x, 5), y=round(self.y, 5),
                    w=round(self.w, 5), h=round(self.h, 5),
                    label=self.label, read=self.read, kind=self.kind)

class Boxer:
    """Turns page-pixel boxes into image fractions for one page."""

    def __init__(self, width: float, height: float):
        self.width, self.height = float(width), float(height)
        self.zones: list[Zone] = []

    def add(self, zone_id: str, box, label: str, read: str = "",
            kind: str = "zone") -> Zone:
        left, top, right, bottom = box
        zone = Zone(id=zone_id, x=left / self.width, y=top / self.height,
                    w=(right - left) / self.width, h=(bottom - top) / self.height,
                    label=label, read=read, kind=kind)
        self.zones.append(zone)
        return zone

    def dicts(self) -> list[dict]:
        return [z.as_dict() for z in self.zones]

def german_page(pdf) -> dict:
    """Zones for the German page — word-derived, because there are no rules."""
    page = pdf.pages[DE_SCAN - 1]
    lines = _de_lines(page)

    def match(pattern: str, start: int = 0) -> int | None:
        rx = re.compile(pattern)
        for i in range(start, len(lines)):
            if rx.search(lines[i]["text"]):
                return i
        return None

    head = match(r"B.hmisch|Sk.{1,3}itz\.")
    first = match(r"Brzitz")
    last = match(r"d.ar|/dar|Zd.ar") or (first + 4 if first is not None else None)
    above = match(r"Skalitz, Cesk|Skalitz, Ceskä|Skalitz, Ceskä")
    below = match(r"Cerwe|erveml|Cerveml")
    body = next((i for i, ln in enumerate(lines)
                 if len(re.findall(r"\d", ln["text"])) >= 8), 0)

    b = Boxer(page.width, page.height)
    if head is not None:
        b.add("de-head", (lines[head]["x0"] - 6, lines[head]["top"] - 6,
                          lines[head]["x1"] + 6, lines[head]["bottom"] + 6),
              "A different running head", lines[head]["text"])
    b.add("de-header", (10, lines[max(head or 0, 0) + 1]["top"] - 4,
                        page.width - 10, lines[body]["top"] - 4),
          "The same columns in German")
    if first is not None and last is not None:
        b.add("de-block", (14, lines[first]["top"] - 5, page.width - 14,
                           lines[last]["bottom"] + 5),
              "Brzitz — the identical figures",
              " / ".join(lines[i]["text"] for i in range(first, last + 1)),
              kind="focus")
    for zid, idx, label in (("de-above", above, "The municipality before it"),
                            ("de-below", below, "The municipality after it")):
        if idx is not None:
            b.add(zid, (14, lines[idx]["top"] - 4, page.width - 14,
                        lines[idx]["bottom"] + 4), label, lines[idx]["text"])
    # The apparatus opens on its own heading. Found on the heading rather than on
    # "the last line with a lot of digits", which is the postal legend at the very
    # foot of the sheet — a rule keyed on digits ends the table below the thing it
    # was meant to find.
    tail = next((i for i in range(len(lines) - 1, 0, -1)
                 if re.search(r"Ortsbestandteile|Ämter|Amter|Anstalten", lines[i]["text"])),
                None)
    while tail and tail > 0 and re.search(
            r"Ortsbestandteile|Ämter|Amter|Anstalten", lines[tail - 1]["text"]):
        tail -= 1
    tail = (tail - 1) if tail else len(lines) - 1
    entry_box = None
    if tail + 1 < len(lines):
        band = (lines[tail + 1]["top"] - 6, lines[-1]["bottom"] + 6)
        split = _de_gutter(page, band)
        b.add("de-notes", (14, band[0], split, band[1]),
              "The apparatus, and a gutter found in the whitespace",
              " ".join(ln["text"] for ln in
                       _de_lines(page, band=band, x_range=(0, split))[:3]))
        right = _de_lines(page, band=band, x_range=(split, page.width))
        b.add("de-notes-right", (split, band[0], page.width - 14, band[1]),
              "The Böhmisch Skalitz institutions",
              " ".join(ln["text"] for ln in right[:2]))
        # Not found by its key, because the German text layer misreads it: the
        # printed ``3, 1)`` comes back as ``4, -)``, and the two entries that
        # actually follow are dropped altogether. Found instead by the indent —
        # an entry opens at the section's left margin, a continuation is indented —
        # and then checked against the one thing the line must end with, the ``R``
        # of the Raiffeisen credit union. If that check fails the zone is skipped
        # rather than pointed at whatever was found.
        body = right[1:]
        indent = min((ln["x0"] for ln in body), default=0.0)
        keys = [ln for ln in body if ln["x0"] <= indent + 3]
        hit = keys[1] if len(keys) > 1 else None
        if hit and not hit["text"].rstrip(" .").endswith("R"):
            hit = None
        if hit:
            # Cut one printed line down from the top of the band. Neither the
            # band's centre nor its foot can be used: on this very entry the text
            # layer gives the four symbol tokens a single 11.9 pt box spanning
            # *two* printed lines, while the key beside them sits in a normal
            # 7.1 pt one — so the only trustworthy edge here is the top, and the
            # height comes from the section's own median word box.
            heights = sorted(w["bottom"] - w["top"] for w in page.extract_words()
                             if band[0] <= (w["top"] + w["bottom"]) / 2 <= band[1]
                             and w["x0"] > split)
            lead = heights[len(heights) // 2] if heights else 7.0
            entry_box = (hit["x0"] - 5, hit["top"] - 2.2,
                         hit["x1"] + 9, hit["top"] + lead + 0.2)
            b.add("de-entry", entry_box,
                  "The same four symbols, keyed 3 instead of 1", hit["text"],
                  kind="focus")
    return dict(boxer=b, page=page, lines=lines, entry_box=entry_box,
                head=lines[head]["text"] if head is not None else "")

def pilot_counts() -> dict:
    """Headline counts read from the pilot outputs, or ``{}`` if it has not run.

    Frozen into the JSON rather than read live by the docs build, because
    ``output/`` is gitignored and a fresh clone has none of it. A number on the
    site should be one some run actually produced, and this is the run.
    """
    import pandas as pd

    checks = PILOT / "checks_glmocr.csv"
    if not checks.exists():
        return {}
    df = pd.read_csv(checks)
    out: dict = {"source": checks.name, "checks": {}}
    for name, group in df.groupby("check"):
        counts = group["verdict"].value_counts().to_dict()
        graded = sum(v for k, v in counts.items() if k in ("pass", "fail", "unparseable"))
        if not graded:
            continue
        out["checks"][str(name)] = dict(
            passed=int(counts.get("pass", 0)), failed=int(counts.get("fail", 0)),
            unparseable=int(counts.get("unparseable", 0)),
            na=int(counts.get("na", 0)), graded=int(graded),
            rate=round(100 * counts.get("pass", 0) / graded, 1))

    # The symbol join is not graded pass/fail — there is no arithmetic identity
    # below the rule — so its verdicts are tallied raw.
    join = df[df["check"] == "I11_symbol_join"] if "check" in df else df.iloc[0:0]
    if len(join):
        out["join"] = {str(k): int(v) for k, v in join["verdict"].value_counts().items()}

    agree = PILOT / "symbol_agreement.csv"
    if agree.exists():
        adf = pd.read_csv(agree)
        readers = [c for c in adf.columns if c.startswith("count_")]
        out["agreement"] = dict(
            markers=int(len(adf)),
            readers=len(readers),
            contested=int((adf["verdict"] == "contested").sum())
            if "verdict" in adf else 0,
            unanimous=int((adf["n_claiming"] == adf["n_passes"]).sum())
            if "n_claiming" in adf else 0,
            single_reader=int((adf["n_claiming"] == 1).sum())
            if "n_claiming" in adf else 0,
            by_reader={c.replace("count_", ""): int(adf[c].notna().sum())
                       for c in readers})

    human = PILOT / "inventory_human.csv"
    if human.exists():
        hdf = pd.read_csv(human)
        conf = hdf["confidence"].value_counts().to_dict() if "confidence" in hdf else {}
        out["human"] = dict(symbols=int(len(hdf)),
                            certain=int(conf.get("certain", 0)),
                            unsure=int(conf.get("unsure", 0)),
                            unidentified=int(conf.get("unidentified", 0)))
    return out

def glyphs() -> list[dict]:
    """Re-encode the symbol atlas crops small, and carry their glosses.

    The atlas cuts each symbol out of the printed plate at display size, which is
    the only place in the volume the design is drawn large enough to recognise.
    Every one is inlined on the site, so they are re-encoded to greyscale here
    rather than shipped as the atlas builder left them.
    """
    import csv
    from PIL import Image

    manifest = SYMBOLS / "manifest.csv"
    if not manifest.exists():
        return []
    out_dir = OUT / "glyphs"
    out_dir.mkdir(parents=True, exist_ok=True)
    for stale in out_dir.glob("*.png"):
        stale.unlink()

    records = []
    with manifest.open(encoding="utf-8") as fh:
        for row in csv.DictReader(fh):
            # A plate line the atlas could not cut leaves the file cell empty, and
            # `SYMBOLS / ""` is the directory itself.
            if not (row.get("file") or "").strip():
                continue
            src = SYMBOLS / row["file"]
            if not src.is_file():
                continue
            image = Image.open(src).convert("L")
            if image.height > 56:
                scale = 56 / image.height
                image = image.resize((max(1, round(image.width * scale)), 56),
                                     Image.LANCZOS)
            dest = out_dir / row["file"]
            image.save(dest, optimize=True)
            records.append(dict(token=row["token"], file=f"glyphs/{row['file']}",
                                gloss=row.get("gloss_curated") or row.get("gloss_printed", ""),
                                w=image.width, h=image.height))
    return records

"""Cut the docs-site figure out of the 1910 register — images plus derived geometry.

The docs site is one self-contained HTML file built by ``utils/docs/build_docs.py``,
and it must not open a 3.34 GB gitignored PDF to draw a picture. So this script does
the expensive half **once** and writes small tracked artifacts under
``reference/places/figure/``: a handful of JPEGs and one JSON file holding every
overlay box, every verbatim string the box covers, and the pilot's headline counts.
``build_docs.py`` then only reads that.

**Every box is computed, never eyeballed.** The Czech page has an ALTO layer whose
frame is the scan's own pixel grid, so :func:`~utils.lexikon_1910.lexikon_1910.render_page`
renders it 1:1 and the module's existing geometry — ``body_band``, ``fit_columns``,
``segment_blocks``, ``notes_boxes``, the per-line bounds ``extract_notes`` already
returns — lands directly on the image with no scale factor to carry around. Boxes are
then divided by the image size, so the figure survives any later change of render
width.

**The German side is coarser and says so.** The UBW scan has no ALTO and no printed
rules — ``page.rects`` and ``page.lines`` are empty — so its boxes come from
pdfplumber word extents clustered into printed lines. That is enough for the one
point the German page is on the sheet to make (the same municipality, a different
page and a different position in the district, because each edition sorts an okres
alphabetically in its own language), and not enough to draw a column grid on.

Run it from the repo root::

    python utils/lexikon_1910/build_lexikon_figure.py

Nothing here calls a paid API or a model. It reads two PDFs, one ALTO file, the
symbol atlas manifest and — if the pilot has been run — the pilot's check CSVs.
"""

def _kb(path: Path) -> float:
    return path.stat().st_size / 1024


def _trim(image, pad: int = 40):
    """Crop away the blank paper around the printed block.

    The two plates print about half a page of symbols and leave the rest white.
    Shipping that white costs the same resolution everywhere else, and the plate is
    the one image on the sheet where the reader has to be able to tell two
    six-point glyphs apart.
    """
    from PIL import ImageOps

    grey = image.convert("L")
    mask = grey.point(lambda v: 255 if v < 170 else 0)
    box = mask.getbbox()
    if box is None:
        return ImageOps.autocontrast(grey, cutoff=1)
    left, top, right, bottom = box
    trimmed = grey.crop((max(0, left - pad), max(0, top - pad),
                         min(image.width, right + pad),
                         min(image.height, bottom + pad)))
    return ImageOps.autocontrast(trimmed, cutoff=1)


def _save(image, path: Path, *, quality: int, long_edge: int = 0,
          grey: bool = True) -> Path:
    """Write one figure image, downscaled and (by default) desaturated.

    The scans are ink on paper; the colour channels carry the scanner's cast and
    nothing else, and dropping them is worth roughly a third of the file on a page
    this size. The site inlines every byte of this as base64, so the saving is
    paid straight back in page weight.
    """
    from PIL import Image

    if grey:
        image = image.convert("L")
    if long_edge and max(image.size) > long_edge:
        scale = long_edge / max(image.size)
        image = image.resize((round(image.width * scale), round(image.height * scale)),
                             Image.LANCZOS)
    path.parent.mkdir(parents=True, exist_ok=True)
    if path.suffix == ".png":
        image.save(path, optimize=True)
    else:
        image.save(path, quality=quality, optimize=True, progressive=True)
    return path


@dataclass
class Zone:
    """One clickable region of a page, in fractions of the image."""

    id: str
    x: float
    y: float
    w: float
    h: float
    label: str
    #: What the text layer actually read inside this box, verbatim. The point of
    #: the figure is the distance between the print and this string, so it is
    #: carried rather than paraphrased.
    read: str = ""
    kind: str = "zone"

    def as_dict(self) -> dict:
        return dict(id=self.id, x=round(self.x, 5), y=round(self.y, 5),
                    w=round(self.w, 5), h=round(self.h, 5),
                    label=self.label, read=self.read, kind=self.kind)


class Boxer:
    """Turns page-pixel boxes into image fractions for one page."""

    def __init__(self, width: float, height: float):
        self.width, self.height = float(width), float(height)
        self.zones: list[Zone] = []

    def add(self, zone_id: str, box, label: str, read: str = "",
            kind: str = "zone") -> Zone:
        left, top, right, bottom = box
        zone = Zone(id=zone_id, x=left / self.width, y=top / self.height,
                    w=(right - left) / self.width, h=(bottom - top) / self.height,
                    label=label, read=read, kind=kind)
        self.zones.append(zone)
        return zone

    def dicts(self) -> list[dict]:
        return [z.as_dict() for z in self.zones]

def _find_line(lines, bounds, pattern: str) -> tuple[int, tuple[int, int]] | None:
    """First apparatus line matching ``pattern``, with its printed extent."""
    rx = re.compile(pattern)
    for i, (text, extent) in enumerate(zip(lines, bounds)):
        if rx.search(text):
            return i, extent
    return None

                 and not r.is_of_which]
    notes = nt.extract_notes(page)

    heading_bad = _find_line(notes.right_lines, notes.right_bounds, r"okresu .esk. Skalice:$")
    entry = _find_line(notes.right_lines, notes.right_bounds, r"^1,\s")
    seat = _find_line(notes.right_lines, notes.right_bounds, r"o\.\s*s-?,\s*not")
    legend = _find_line(notes.footer_lines, [(0, 0)] * len(notes.footer_lines),
                        r"Skalice:")

    left, right = page.print_left, page.print_left + (page.print_width or page.width)
    frame_top = min((v["VPOS"] for v in page.verticals
                     if v["HEIGHT"] >= 0.70 * page.height), default=body_top - 300)

    b = Boxer(page.width, page.height)
    b.add("cz-head", (left, 60, right, frame_top), "The running head",
          lx.running_head(page)[:120])
    b.add("cz-header", (cols[0], frame_top, cols[-1], body_top),
          "Seventeen columns, eighteen rules")
    b.add("cz-seq", (cols[0], body_top, cols[1], body_bottom), "Column 1 — two sequences")
    b.add("cz-area", (cols[2], body_top, cols[3], body_bottom),
          "Column 3 — the block delimiter")
    b.add("cz-lasttwo", (cols[15], body_top, cols[17], body_bottom),
          "The last two columns")
    for row in summaries:
        if lx.summary_label(row.get("name"))[0] == "okres":
            b.add("cz-summary", (cols[0], row.top - 4, cols[17], row.bottom + 4),
                  "A summary row is not a municipality", row.get("name"))

          "Brzice — one municipality block", block.name, kind="focus")
    b.add("cz-notes-left", boxes["left"], "The apparatus, left column")
    b.add("cz-notes-right", boxes["right"], "The apparatus, right column")
    if heading_ok:

        b.add("cz-heading-ok", (boxes["left"][0], top - 4, notes.split_x, bottom + 4),
              "A heading that survived", notes.left_lines[i])
    if heading_bad:

              "What Brzice had", notes.right_lines[i], kind="focus")
    if seat:

        bottom = notes.right_bounds[min(i + 3, len(notes.right_bounds) - 1)][1]
        b.add("cz-seat", (notes.split_x, top - 5, boxes["right"][2], bottom + 5),
              "The same field for a district seat",
              " ".join(notes.right_lines[i:i + 4]))

def _de_lines(page, band: tuple[float, float] | None = None,
              x_range: tuple[float, float] | None = None) -> list[dict]:
    """Cluster one German page's words into printed lines, with their extents.

    ``x_range`` restricts the clustering to one column. The Czech page never needs
    that — its divider rule is in the ALTO and the two apparatus columns are cut on
    it. The German scan has no rules at all, so a plain clustering runs straight
    across the gutter and returns lines that are half of one district's entry and
    half of another's; the column has to be guessed before the lines are found.
    """
    words = [w for w in page.extract_words() if -1 <= w["x0"] <= page.width + 1]
    if band:
        words = [w for w in words if band[0] <= (w["top"] + w["bottom"]) / 2 <= band[1]]
    if x_range:
        words = [w for w in words if x_range[0] <= (w["x0"] + w["x1"]) / 2 <= x_range[1]]
    bands: list[list[dict]] = []
    for word in sorted(words, key=lambda w: (w["top"] + w["bottom"]) / 2):
        centre = (word["top"] + word["bottom"]) / 2
        if bands and centre - (bands[-1][-1]["top"] + bands[-1][-1]["bottom"]) / 2 <= _DE_LINE_GAP:
            bands[-1].append(word)
        else:
            bands.append([word])
    out = []
    for band in bands:
        band = sorted(band, key=lambda w: w["x0"])
        # `cy` is the median of the words' own centres, not the middle of the band's
        # extremes. One over-tall glyph box — and the pictograms all have one — drags
        # top or bottom across the neighbouring entry, so the extremes are usable for
        # clustering and useless for cutting a crop.
        centres = sorted((w["top"] + w["bottom"]) / 2 for w in band)
        out.append(dict(text=" ".join(w["text"] for w in band),
                        top=min(w["top"] for w in band),
                        bottom=max(w["bottom"] for w in band),
                        cy=centres[len(centres) // 2],
                        x0=min(w["x0"] for w in band),
                        x1=max(w["x1"] for w in band)))
    return out


def _de_gutter(page, band: tuple[float, float]) -> float:
    """Where the apparatus splits into two columns, found in the whitespace.

    The Czech page hands this over: the divider is a ``GraphicalElement`` in the
    ALTO and ``lexikon_1910_notes.divider_rule`` picks the tallest one below the
    table. The German scan carries the same printed rule as *ink*, not as vector —
    ``page.rects`` and ``page.lines`` are empty — so the only thing left to find it
    with is the column of blank paper beside it: the widest run of x with no word
    over it, taken across the middle half of the sheet.
    """
    words = [w for w in page.extract_words()
             if band[0] <= (w["top"] + w["bottom"]) / 2 <= band[1]]
    if not words:
        return page.width / 2
    step = 1.0
    covered = set()
    for word in words:
        for x in range(int(word["x0"] / step), int(word["x1"] / step) + 1):
            covered.add(x)
    low, high = int(page.width * 0.30 / step), int(page.width * 0.62 / step)
    best, run, start = (0, 0.0), 0, low
    for x in range(low, high + 1):
        if x in covered:
            run, start = 0, x + 1
        else:
            run += 1
            if run > best[0]:
                best = (run, (start + x) / 2 * step)
    return best[1] if best[0] else page.width / 2


def german_page(pdf) -> dict:
    """Zones for the German page — word-derived, because there are no rules."""
    page = pdf.pages[DE_SCAN - 1]
    lines = _de_lines(page)

    def match(pattern: str, start: int = 0) -> int | None:
        rx = re.compile(pattern)
        for i in range(start, len(lines)):
            if rx.search(lines[i]["text"]):
                return i
        return None

    head = match(r"B.hmisch|Sk.{1,3}itz\.")
    first = match(r"Brzitz")
    last = match(r"d.ar|/dar|Zd.ar") or (first + 4 if first is not None else None)
    above = match(r"Skalitz, Cesk|Skalitz, Ceskä|Skalitz, Ceskä")
    below = match(r"Cerwe|erveml|Cerveml")
    body = next((i for i, ln in enumerate(lines)
                 if len(re.findall(r"\d", ln["text"])) >= 8), 0)

    b = Boxer(page.width, page.height)
    if head is not None:
        b.add("de-head", (lines[head]["x0"] - 6, lines[head]["top"] - 6,
                          lines[head]["x1"] + 6, lines[head]["bottom"] + 6),
              "A different running head", lines[head]["text"])
    b.add("de-header", (10, lines[max(head or 0, 0) + 1]["top"] - 4,
                        page.width - 10, lines[body]["top"] - 4),
          "The same columns in German")
    if first is not None and last is not None:
        b.add("de-block", (14, lines[first]["top"] - 5, page.width - 14,
                           lines[last]["bottom"] + 5),
              "Brzitz — the identical figures",
              " / ".join(lines[i]["text"] for i in range(first, last + 1)),
              kind="focus")
    for zid, idx, label in (("de-above", above, "The municipality before it"),
                            ("de-below", below, "The municipality after it")):
        if idx is not None:
            b.add(zid, (14, lines[idx]["top"] - 4, page.width - 14,
                        lines[idx]["bottom"] + 4), label, lines[idx]["text"])
    # The apparatus opens on its own heading. Found on the heading rather than on
    # "the last line with a lot of digits", which is the postal legend at the very
    # foot of the sheet — a rule keyed on digits ends the table below the thing it
    # was meant to find.
    tail = next((i for i in range(len(lines) - 1, 0, -1)
                 if re.search(r"Ortsbestandteile|Ämter|Amter|Anstalten", lines[i]["text"])),
                None)
    while tail and tail > 0 and re.search(
            r"Ortsbestandteile|Ämter|Amter|Anstalten", lines[tail - 1]["text"]):
        tail -= 1
    tail = (tail - 1) if tail else len(lines) - 1
    entry_box = None
    if tail + 1 < len(lines):
        band = (lines[tail + 1]["top"] - 6, lines[-1]["bottom"] + 6)
        split = _de_gutter(page, band)
        b.add("de-notes", (14, band[0], split, band[1]),
              "The apparatus, and a gutter found in the whitespace",
              " ".join(ln["text"] for ln in
                       _de_lines(page, band=band, x_range=(0, split))[:3]))
        right = _de_lines(page, band=band, x_range=(split, page.width))
        b.add("de-notes-right", (split, band[0], page.width - 14, band[1]),
              "The Böhmisch Skalitz institutions",
              " ".join(ln["text"] for ln in right[:2]))
        # Not found by its key, because the German text layer misreads it: the
        # printed ``3, 1)`` comes back as ``4, -)``, and the two entries that
        # actually follow are dropped altogether. Found instead by the indent —
        # an entry opens at the section's left margin, a continuation is indented —
        # and then checked against the one thing the line must end with, the ``R``
        # of the Raiffeisen credit union. If that check fails the zone is skipped
        # rather than pointed at whatever was found.
        body = right[1:]
        indent = min((ln["x0"] for ln in body), default=0.0)
        keys = [ln for ln in body if ln["x0"] <= indent + 3]
        hit = keys[1] if len(keys) > 1 else None
        if hit and not hit["text"].rstrip(" .").endswith("R"):
            hit = None
        if hit:
            # Cut one printed line down from the top of the band. Neither the
            # band's centre nor its foot can be used: on this very entry the text
            # layer gives the four symbol tokens a single 11.9 pt box spanning
            # *two* printed lines, while the key beside them sits in a normal
            # 7.1 pt one — so the only trustworthy edge here is the top, and the
            # height comes from the section's own median word box.
            heights = sorted(w["bottom"] - w["top"] for w in page.extract_words()
                             if band[0] <= (w["top"] + w["bottom"]) / 2 <= band[1]
                             and w["x0"] > split)
            lead = heights[len(heights) // 2] if heights else 7.0
            entry_box = (hit["x0"] - 5, hit["top"] - 2.2,
                         hit["x1"] + 9, hit["top"] + lead + 0.2)
            b.add("de-entry", entry_box,
                  "The same four symbols, keyed 3 instead of 1", hit["text"],
                  kind="focus")
    return dict(boxer=b, page=page, lines=lines, entry_box=entry_box,
                head=lines[head]["text"] if head is not None else "")

def pilot_counts() -> dict:
    """Headline counts read from the pilot outputs, or ``{}`` if it has not run.

    Frozen into the JSON rather than read live by the docs build, because
    ``output/`` is gitignored and a fresh clone has none of it. A number on the
    site should be one some run actually produced, and this is the run.
    """
    import pandas as pd

    checks = PILOT / "checks_glmocr.csv"
    if not checks.exists():
        return {}
    df = pd.read_csv(checks)
    out: dict = {"source": checks.name, "checks": {}}
    for name, group in df.groupby("check"):
        counts = group["verdict"].value_counts().to_dict()
        graded = sum(v for k, v in counts.items() if k in ("pass", "fail", "unparseable"))
        if not graded:
            continue
        out["checks"][str(name)] = dict(
            passed=int(counts.get("pass", 0)), failed=int(counts.get("fail", 0)),
            unparseable=int(counts.get("unparseable", 0)),
            na=int(counts.get("na", 0)), graded=int(graded),
            rate=round(100 * counts.get("pass", 0) / graded, 1))

    # The symbol join is not graded pass/fail — there is no arithmetic identity
    # below the rule — so its verdicts are tallied raw.
    join = df[df["check"] == "I11_symbol_join"] if "check" in df else df.iloc[0:0]
    if len(join):
        out["join"] = {str(k): int(v) for k, v in join["verdict"].value_counts().items()}

    agree = PILOT / "symbol_agreement.csv"
    if agree.exists():
        adf = pd.read_csv(agree)
        readers = [c for c in adf.columns if c.startswith("count_")]
        out["agreement"] = dict(
            markers=int(len(adf)),
            readers=len(readers),
            contested=int((adf["verdict"] == "contested").sum())
            if "verdict" in adf else 0,
            unanimous=int((adf["n_claiming"] == adf["n_passes"]).sum())
            if "n_claiming" in adf else 0,
            single_reader=int((adf["n_claiming"] == 1).sum())
            if "n_claiming" in adf else 0,
            by_reader={c.replace("count_", ""): int(adf[c].notna().sum())
                       for c in readers})

    human = PILOT / "inventory_human.csv"
    if human.exists():
        hdf = pd.read_csv(human)
        conf = hdf["confidence"].value_counts().to_dict() if "confidence" in hdf else {}
        out["human"] = dict(symbols=int(len(hdf)),
                            certain=int(conf.get("certain", 0)),
                            unsure=int(conf.get("unsure", 0)),
                            unidentified=int(conf.get("unidentified", 0)))
    return out

def glyphs() -> list[dict]:
    """Re-encode the symbol atlas crops small, and carry their glosses.

    The atlas cuts each symbol out of the printed plate at display size, which is
    the only place in the volume the design is drawn large enough to recognise.
    Every one is inlined on the site, so they are re-encoded to greyscale here
    rather than shipped as the atlas builder left them.
    """
    import csv
    from PIL import Image

    manifest = SYMBOLS / "manifest.csv"
    if not manifest.exists():
        return []
    out_dir = OUT / "glyphs"
    out_dir.mkdir(parents=True, exist_ok=True)
    for stale in out_dir.glob("*.png"):
        stale.unlink()

    records = []
    with manifest.open(encoding="utf-8") as fh:
        for row in csv.DictReader(fh):
            # A plate line the atlas could not cut leaves the file cell empty, and
            # `SYMBOLS / ""` is the directory itself.
            if not (row.get("file") or "").strip():
                continue
            src = SYMBOLS / row["file"]
            if not src.is_file():
                continue
            image = Image.open(src).convert("L")
            if image.height > 56:
                scale = 56 / image.height
                image = image.resize((max(1, round(image.width * scale)), 56),
                                     Image.LANCZOS)
            dest = out_dir / row["file"]
            image.save(dest, optimize=True)
            records.append(dict(token=row["token"], file=f"glyphs/{row['file']}",
                                gloss=row.get("gloss_curated") or row.get("gloss_printed", ""),
                                w=image.width, h=image.height))
    return records

def build(de_plate: int = DE_PLATE_SCAN) -> Path:
    import pdfplumber

    OUT.mkdir(parents=True, exist_ok=True)
    written: list[Path] = []

    # ── the Czech page ────────────────────────────────────────────────────────
    cz = czech_page((0, 0))
    page = cz["page"]
    image = lx.render_page(CZ_PDF, CZ_SCAN, page.width)
    written.append(_save(image, OUT / "cz_page.jpg", quality=PAGE_QUALITY,
                         long_edge=PAGE_LONG_EDGE))

    # Crops at native resolution — this is the evidence; the page above is a map.
    cols = cz["cols"]
    block = cz["block"]
    written.append(_save(image.crop((cols[0] - 10, block.top - 12, cols[17] + 10,
                                     block.bottom + 2)),
                         OUT / "block_brzice.jpg", quality=82))
    if cz["entry_extent"]:
        top, bottom = cz["entry_extent"]
        # Bounded by the words of that line rather than by the column, so the crop
        # is the entry and not the entry plus a hand's width of blank paper.
        inked = [s for s in page.strings
                 if top - 6 <= s[0] + s[3] // 2 <= bottom + 6 and s[1] > cz["notes"].split_x]
        x1 = max((s[1] + s[2] for s in inked), default=cz["notes_boxes"]["right"][2])
        written.append(_save(image.crop((cz["notes"].split_x - 4, top - 10,
                                         min(x1 + 40, cz["notes_boxes"]["right"][2]),
                                         bottom + 10)),
                             OUT / "entry_brzice.png", quality=90))
    if cz["seat_index"] is not None:
        i = cz["seat_index"]
        bounds = cz["notes"].right_bounds
        top = bounds[i][0]
        bottom = bounds[min(i + 3, len(bounds) - 1)][1]
        written.append(_save(image.crop((cz["notes"].split_x - 4, top - 8,
                                         cz["notes_boxes"]["right"][2], bottom + 8)),
                             OUT / "entry_skalice.jpg", quality=84))
    if cz["legend_index"] is not None:
        fbox = cz["notes_boxes"]["footer"]
        written.append(_save(image.crop((fbox[0], fbox[1], fbox[2], fbox[3])),
                             OUT / "legend_skalice.jpg", quality=82,
                             long_edge=1900))

    # ── the plates ────────────────────────────────────────────────────────────
    plate_cz = lx.render_page(CZ_PDF, CZ_PLATE_SCAN, 2400)
    written.append(_save(_trim(plate_cz), OUT / "cz_plate.jpg", quality=PLATE_QUALITY,
                         long_edge=PAGE_LONG_EDGE))

    # ── the German page and plate ─────────────────────────────────────────────
    with pdfplumber.open(DE_PDF) as pdf:
        de = german_page(pdf)
    de_page_width = float(de["page"].width)
    de_image = lx.render_page(DE_PDF, DE_SCAN, 2200)
    written.append(_save(de_image, OUT / "de_page.jpg", quality=PAGE_QUALITY,
                         long_edge=PAGE_LONG_EDGE))
    if de["entry_box"]:
        # Tighter than the clickable box: the entries are ~8 pt apart, so the pad
        # that makes a zone comfortable to hit puts the neighbour's descenders in
        # the crop, and the crop is the thing being compared.
        #
        # Cut from its own render at twice the page zoom. The German entry is 74 pt
        # of a 516 pt page against the Czech one's 1,290 ALTO pixels, and the two are
        # shown side by side — at the page's own zoom the German half arrives at a
        # quarter of the size and reads as the worse scan, which it is not.
        zoom = DE_CROP_WIDTH / de_page_width
        detail = lx.render_page(DE_PDF, DE_SCAN, DE_CROP_WIDTH)
        left, top, right, bottom = de["entry_box"]
        written.append(_save(detail.crop(((left + 4) * zoom, top * zoom,
                                          (right - 4) * zoom, bottom * zoom)),
                             OUT / "entry_brzitz.png", quality=90))
    plate_de = lx.render_page(DE_PDF, de_plate, 2200)
    written.append(_save(_trim(plate_de), OUT / "de_plate.jpg", quality=PLATE_QUALITY,
                         long_edge=PAGE_LONG_EDGE))

    glyph_records = glyphs()

    payload = {
        "version": __version__,
        "volume": {
            "cz": {"title": "Podrobný seznam míst zemí rakouských — IX. Čechy",
                   "place": "Vídeň / Wien, 1916",
                   "id": "NDK uuid:c2d4a3a0-6fb2-11ec-bc36-005056827e51",
                   "scans": 756},
            "de": {"title": "Spezialortsrepertorium der österreichischen Länder — IX. Böhmen",
                   "place": "Wien, 1915",
                   "id": "UBW urn:nbn:at:at-ubw:g-80326",
                   "scans": 764},
        },
        "pages": {
            "cz": {"image": "cz_page.jpg", "scan": CZ_SCAN, "printed": CZ_PRINTED,
                   "edition": "cz", "head": cz["head"][:160],
                   "rules_observed": cz["observed"],
                   "zones": cz["boxer"].dicts()},
            "de": {"image": "de_page.jpg", "scan": DE_SCAN, "printed": DE_PRINTED,
                   "edition": "de", "head": de["head"][:160],
                   "rules_observed": 0,
                   "zones": de["boxer"].dicts()},
        },
        "plates": {"cz": {"image": "cz_plate.jpg", "scan": CZ_PLATE_SCAN},
                   "de": {"image": "de_plate.jpg", "scan": de_plate}},
        "crops": {"block": "block_brzice.jpg", "entry": "entry_brzice.png",
                  "seat": "entry_skalice.jpg", "legend": "legend_skalice.jpg",
                  "entry_de": "entry_brzitz.png" if de["entry_box"] else ""},
        "block": {"name": block.name, "court": block.court_label,
                  "okres": block.okres_label, "rows": cz["rows"],
                  "columns": lx.COL_NAMES},
        "glyphs": glyph_records,
        "counts": pilot_counts(),
    }

    out_json = OUT / "lexikon_figure.json"
    out_json.write_text(json.dumps(payload, ensure_ascii=False, indent=1),
                        encoding="utf-8")
    written.append(out_json)

    total = sum(p.stat().st_size for p in written)
    glyph_bytes = sum(p.stat().st_size for p in (OUT / "glyphs").glob("*.png"))
    print(f"\n{'file':<28}{'KB':>10}")
    for path in written:
        print(f"{path.name:<28}{_kb(path):>10.0f}")
    print(f"{f'glyphs/ ({len(glyph_records)} files)':<28}{glyph_bytes / 1024:>10.0f}")
    print(f"{'TOTAL':<28}{(total + glyph_bytes) / 1024:>10.0f}  "
          f"(~{(total + glyph_bytes) * 4 / 3 / 1024 / 1024:.1f} MB inlined as base64)")
    return out_json

    ap = argparse.ArgumentParser(description=__doc__.split("\n")[0])
    ap.add_argument("--de-plate", type=int, default=DE_PLATE_SCAN,
