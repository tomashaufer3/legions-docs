How it works

Čechy and Böhmen are the same census table, set twice: once in Czech by the k.k. Statistische Zentralkommission's Prague office, once in German. Neither is a translation of the other's typesetting — the two volumes are ordered differently (Czech by Czech district name, German by German), paginate differently, and were composited and proofed separately. Disagreement between them therefore flags an error with no human in the loop, which is as close to free double-keying as a source like this ever gets.

Two things make the German side harder than the Czech one, and both are handled here rather than assumed away:

  • No ALTO and no printed rules. The UBW scan carries an embedded text layer but page.rects and page.lines are empty, so the grid has to come from the words themselves. The numeric columns are right-aligned, so their right edges pile up into sharp peaks — the same right-edge read lexikon_sk.py uses for the Slovak lexicon.
  • The scans are not consistently cropped (511–541 pt wide across a sample of 80 pages), so absolute positions smear into unusable bands. Normalising each page against its own text extent resolves them back into peaks.

The volumes are located against each other through the running head, which prints Politischer Bezirk: Gerichtsbezirk, and then matched by name within an okres. Not by the printed running number: each volume orders an okres's municipalities alphabetically in its own language, so the same place carries a different number in each — a join on the number pairs unrelated municipalities while looking entirely well-behaved.

def normalise_name(name: str) -> str:
    """Match key for a district name: unaccented, lowercase, alphanumeric only."""
    stripped = unicodedata.normalize("NFKD", str(name))
    stripped = "".join(c for c in stripped if not unicodedata.combining(c))
    return re.sub(r"[^a-z0-9]", "", stripped.lower())

def build_page_index(plaintext_dir: str | Path, head_lines: int = 12) -> dict[str, list[int]]:
    """Map normalised German *politischer Bezirk* name → 1-based page numbers.

    The running head prints ``Politischer Bezirk: Gerichtsbezirk``, sometimes twice
    on one page when a district ends mid-page (``Falkenau: Falkenau. — Friedland:
    Friedland.``). Every district named in the head is indexed.
    """
    index: dict[str, list[int]] = {}
    for path in sorted(Path(plaintext_dir).glob("*.txt")):
        page_number = int(path.stem.split("-")[-1])
        head = path.read_text(encoding="utf-8", errors="replace").split("\n")[:head_lines]
        for line in head:
            for chunk in line.split("—"):
                match = _HEAD_RE.match(chunk)
                if match:
                    key = normalise_name(match.group(1))
                    if key:
                        index.setdefault(key, []).append(page_number)
    return {k: sorted(set(v)) for k, v in index.items()}

def fit_grid(pdf, pages: list[int], bin_width: float = 0.004,
             min_separation: float = 0.022) -> DeGrid:
    """Fit the German column grid from the right edges of numeric words.

    ``pages`` are 1-based page numbers of table pages. One peak is expected for the
    sequence number and one for each of :data:`NUMERIC_ORDER`; sparse columns
    (``conf_israelite``, ``lang_other``, ``foreigners``) contribute weak peaks, so
    the fit pools many pages rather than trusting any single one.
    """
    edges: list[float] = []
    used = 0
    for number in pages:
        page = pdf.pages[number - 1]
        words = _table_words(page)
        if len(words) < 300:            # a part-page or a non-table page
            continue
        low, high = _page_extent(words, page.width)
        span = high - low
        if span <= 0:
            continue
        used += 1
        edges += [(w["x1"] - low) / span for w in words if _INT_RE.match(w["text"])]

    wanted = 1 + len(NUMERIC_ORDER)     # seq + the numeric columns
    centres = _peaks(edges, bin_width, min_separation, wanted)
    if len(centres) < wanted:
        raise ValueError(f"found {len(centres)} column peaks, expected {wanted}")

    # The name column has no right-edge peak — it is left-aligned and ragged — so
    # it must be given its own interval explicitly. Without one, the interval that
    # ought to belong to ``area_ha`` starts immediately after the sequence number
    # and swallows the whole name column along with every figure printed inside it.
    seq_peak, area_peak, *rest = centres
    boundaries = [
        0.0,
        seq_peak + 0.010,                       # end of the sequence number
        max(seq_peak + 0.015, area_peak - 0.050),  # end of the name column
    ]
    # A right-aligned column ends at its own peak, so each remaining boundary sits
    # just right of the peak to its left, bounded by the midpoint to the next.
    for left, right in zip([area_peak, *rest], rest):
        boundaries.append(min(left + 0.010, (left + right) / 2))
    boundaries.append(min(rest[-1] + 0.010, 1.0))
    boundaries.append(1.0 + 1e-6)               # postal key and station note
    columns = ["seq", "name", *NUMERIC_ORDER, "station_note"]
    if len(boundaries) != len(columns) + 1:
        raise ValueError(f"{len(boundaries)} boundaries for {len(columns)} columns")
    return DeGrid(boundaries=boundaries, columns=columns, n_pages=used)

def extract_rows_de(pdf, page_number: int, grid: DeGrid,
                    line_gap: float = 3.0) -> list[Row]:
    """Read one German page into :class:`~utils.lexikon_1910.lexikon_1910.Row` objects.

    Words are assigned by their **right edge**, which is what the typesetting
    aligns. The name column is not a peak — it is everything between the sequence
    number and the area column — so it is reconstructed from the words that fall in
    that interval, in reading order.
    """
    page = pdf.pages[page_number - 1]
    words = _table_words(page)
    if len(words) < 200:
        return []
    low, high = _page_extent(words, page.width)
    span = high - low
    edges = [low + f * span for f in grid.boundaries]

    def column_of(word: dict) -> str | None:
        x = word["x1"]
        for i, (left, right) in enumerate(zip(edges, edges[1:])):
            if left <= x < right:
                return grid.columns[i] if i < len(grid.columns) else None
        return None

    # Cluster into printed lines on the vertical centre of each word.
    ordered = sorted(words, key=lambda w: ((w["top"] + w["bottom"]) / 2, w["x0"]))
    lines: list[list[dict]] = []
    for word in ordered:
        centre = (word["top"] + word["bottom"]) / 2
        if lines and centre - ((lines[-1][-1]["top"] + lines[-1][-1]["bottom"]) / 2) <= line_gap:
            lines[-1].append(word)
        else:
            lines.append([word])

    rows: list[Row] = []
    for line in lines:
        cells: dict[str, list[str]] = {}
        for word in sorted(line, key=lambda w: w["x0"]):
            column = column_of(word)
            if column is None:
                continue
            text = _MARKER_RE.sub("", word["text"])
            # A token with no digit in it landing in a numeric column is the name
            # running long, not a value — the name column has no right-edge peak of
            # its own to anchor it. A token that *does* carry digits stays in the
            # numeric column and is left for the strict parser to judge.
            if column != "seq" and not _HAS_DIGIT_RE.search(text):
                column = "name"
            cells.setdefault(column, []).append(text)
        if not cells:
            continue
        rows.append(Row(
            page=f"de{page_number:04d}",
            top=int(min(w["top"] for w in line)),
            bottom=int(max(w["bottom"] for w in line)),
            cells={k: " ".join(v).strip() for k, v in cells.items() if k in COL_NAMES},
        ))
    return _body_rows(rows)

def match_blocks(czech_blocks, german_blocks,
                 min_similarity: int = NAME_SIMILARITY,
                 tie_margin: int = TIE_MARGIN) -> list[tuple]:
    """Pair Czech and German municipality blocks within **one okres**, by name.

    The printed running number looks like the obvious join key and is not one. Each
    volume lists the municipalities of an okres in **its own language's alphabetical
    order** — Chotěboř runs Bílek, Bořek, Čachotín, Dobkov … in the Czech volume and
    Altmansko, Audlaw, Audolen, Auhrow, Benatek … in the German — so the same
    municipality carries a different number in each. Joining on the number pairs
    unrelated places while looking perfectly well-behaved.

    So the join is on the name, which both volumes print in both languages
    ("Bílek, Bilek" against "Bilek, Bílek"). A pair is accepted only when the best
    candidate clears ``min_similarity`` *and* beats the runner-up by ``tie_margin``;
    a tie is refused rather than broken on order. Each German block is used once.

    Returns ``(score, czech_block, german_block)`` triples.
    """
    from rapidfuzz import fuzz

    candidates = [b for b in german_blocks if not is_heading(b)]
    keys = [normalise_name(b.name) for b in candidates]
    matched: list[tuple] = []
    taken: set[int] = set()

    for czech in czech_blocks:
        if is_heading(czech):
            continue
        target = normalise_name(czech.name)
        if len(target) < 4:
            continue
        scored = sorted(
            ((fuzz.token_set_ratio(target, key), i)
             for i, key in enumerate(keys) if i not in taken and len(key) >= 4),
            reverse=True,
        )
        if not scored:
            continue
        best_score, best_index = scored[0]
        runner_up = scored[1][0] if len(scored) > 1 else 0
        if best_score < min_similarity or best_score - runner_up < tie_margin:
            continue
        taken.add(best_index)
        matched.append((best_score, czech, candidates[best_index]))
    return matched

def compare_cells(czech_block, german_block, columns: list[str] | None = None) -> list[dict]:
    """Cell-by-cell comparison of two matched municipality rows.

    Only the municipality row is compared. The settlement rows beneath it are
    printed in the same order in both volumes, but a single dropped row would shift
    the whole comparison, and there is no independent key to detect that — so the
    comparison stops where it can be trusted.
    """
    from .lexikon_1910 import parse_int_cell

    columns = columns or NUMERIC_ORDER
    findings: list[dict] = []
    for column in columns:
        cz_value, cz_status = parse_int_cell(czech_block.municipality.get(column))
        de_value, de_status = parse_int_cell(german_block.municipality.get(column))
        if cz_status == "bad" or de_status == "bad":
            verdict = "unparseable"
        elif cz_status == "blank" and de_status == "blank":
            verdict = "na"
        elif cz_value == de_value or (cz_value or 0) == (de_value or 0):
            verdict = "pass"
        else:
            verdict = "fail"
        findings.append({
            "block_id": czech_block.block_id, "row_kind": "municipality",
            "row_name": czech_block.name, "check": f"C_crossvolume:{column}",
            "verdict": verdict,
            "detail": "" if verdict in ("pass", "na")
                      else f"cz={czech_block.municipality.get(column)!r} "
                           f"de={german_block.municipality.get(column)!r}",
        })
    return findings

def lexikon_1910_de_report(grid: DeGrid, matched: int, findings: list[dict]) -> str:
    """Coverage and agreement summary for the German channel."""
    counts = Counter(f["verdict"] for f in findings)
    graded = counts["pass"] + counts["fail"] + counts["unparseable"]
    rate = f"{counts['pass'] / graded:.1%}" if graded else "--"
    return "\n".join([
        f"lexikon_1910_de v{__version__} — cross-volume channel",
        f"  grid fitted on {grid.n_pages} German pages, "
        f"{len(grid.columns)} columns",
        f"  municipality blocks matched by name within okres: {matched}",
        f"  cell comparisons: {counts['pass']} agree, {counts['fail']} disagree, "
        f"{counts['unparseable']} unreadable, {counts['na']} both blank",
        f"  agreement on gradeable cells: {rate}",
    ])

"""Spezialortsrepertorium Böhmen 1910 — the German twin of the Czech volume.

Čechy and Böhmen are the *same census table, set twice*: once in Czech by the
k.k. Statistische Zentralkommission's Prague office, once in German. Neither is a
translation of the other's typesetting — the two volumes are ordered differently
(Czech by Czech district name, German by German), paginate differently, and were
composited and proofed separately. Disagreement between them therefore flags an
error with no human in the loop, which is as close to free double-keying as a
source like this ever gets.

Two things make the German side harder than the Czech one, and both are handled
here rather than assumed away:

* **No ALTO and no printed rules.** The UBW scan carries an embedded text layer but
  ``page.rects`` and ``page.lines`` are empty, so the grid has to come from the
  words themselves. The numeric columns are right-aligned, so their right edges
  pile up into sharp peaks — the same right-edge read ``lexikon_sk.py`` uses for the
  Slovak lexicon.
* **The scans are not consistently cropped** (511–541 pt wide across a sample of
  80 pages), so absolute positions smear into unusable bands. Normalising each
  page against its own text extent resolves them back into peaks.

The volumes are located against each other through the running head, which prints
``Politischer Bezirk: Gerichtsbezirk``, and then matched **by name** within an
okres. Not by the printed running number: each volume orders an okres's
municipalities alphabetically in its own language, so the same place carries a
different number in each — a join on the number pairs unrelated municipalities
while looking entirely well-behaved.
"""

from __future__ import annotations

import re
import statistics
import unicodedata
from collections import Counter
from dataclasses import dataclass
from pathlib import Path

from .lexikon_1910 import COL_NAMES, Row

__version__ = "0.1.0"

#: The numeric columns in printed order — what the right-edge peaks map onto.
#: ``seq`` is the leftmost peak; ``name`` is everything between it and ``area_ha``.
NUMERIC_ORDER = [
    "area_ha", "houses", "pop_total", "pop_male", "pop_female",
    "conf_catholic", "conf_evangelical", "conf_israelite", "conf_other",
    "lang_german", "lang_czech", "lang_other", "foreigners",
]

_INT_RE = re.compile(r"^[0-9]+$")
_HAS_DIGIT_RE = re.compile(r"[0-9]")
#: The volume prints reference marks against some figures (``*2089``). They are
#: typography, not OCR noise, so they are stripped before the value is read — but
#: only these, so that genuinely garbled tokens still surface as unreadable.
_MARKER_RE = re.compile(r"^[*†‡•|]+|[*†‡•|]+$")
_HEAD_RE = re.compile(r"^\s*([A-Za-zÄÖÜäöüß][\w.\-’' ]{2,40}?)\s*:\s*([\w.\-’' ]{2,40}?)\s*[.,]?\s*$")


def normalise_name(name: str) -> str:
    """Match key for a district name: unaccented, lowercase, alphanumeric only."""
    stripped = unicodedata.normalize("NFKD", str(name))
    stripped = "".join(c for c in stripped if not unicodedata.combining(c))
    return re.sub(r"[^a-z0-9]", "", stripped.lower())


# --------------------------------------------------------------------------- #
# Locating an okres in the German volume
# --------------------------------------------------------------------------- #

def build_page_index(plaintext_dir: str | Path, head_lines: int = 12) -> dict[str, list[int]]:
    """Map normalised German *politischer Bezirk* name → 1-based page numbers.

    The running head prints ``Politischer Bezirk: Gerichtsbezirk``, sometimes twice
    on one page when a district ends mid-page (``Falkenau: Falkenau. — Friedland:
    Friedland.``). Every district named in the head is indexed.
    """
    index: dict[str, list[int]] = {}
    for path in sorted(Path(plaintext_dir).glob("*.txt")):
        page_number = int(path.stem.split("-")[-1])
        head = path.read_text(encoding="utf-8", errors="replace").split("\n")[:head_lines]
        for line in head:
            for chunk in line.split("—"):
                match = _HEAD_RE.match(chunk)
                if match:
                    key = normalise_name(match.group(1))
                    if key:
                        index.setdefault(key, []).append(page_number)
    return {k: sorted(set(v)) for k, v in index.items()}


def okres_pages(index: dict[str, list[int]], okres_de: str) -> list[int]:
    """Pages of the German volume covering one political district."""
    return index.get(normalise_name(okres_de), [])


# --------------------------------------------------------------------------- #
# The German grid
# --------------------------------------------------------------------------- #

@dataclass
class DeGrid:
    """Normalised column boundaries for the German volume.

    Positions are fractions of each page's own text extent, so they survive the
    inconsistent crop. ``boundaries`` has one more entry than the columns it
    separates.
    """

    boundaries: list[float]
    columns: list[str]
    n_pages: int = 0


def _page_extent(words: list[dict], page_width: float | None = None,
                 quantile: float = 0.01) -> tuple[float, float]:
    """Horizontal extent of the printed table on one page.

    Two guards, both load-bearing. The UBW text layer carries stray words *outside*
    the page box — right edges out to 949 pt on a 527 pt page — so words beyond the
    box are dropped; taken as the denominator they nearly halve every normalised
    position and silently shift the whole grid one column left. And the extent is
    read at the 1st/99th percentile rather than the extremes, so one surviving
    outlier inside the box cannot stretch it either.
    """
    if page_width:
        words = [w for w in words if -1 <= w["x0"] and w["x1"] <= page_width + 1] or words
    lows = sorted(w["x0"] for w in words)
    highs = sorted(w["x1"] for w in words)
    low = lows[int(quantile * (len(lows) - 1))]
    high = highs[int((1 - quantile) * (len(highs) - 1))]
    return low, high


def _table_words(page) -> list[dict]:
    """Words of one page, with anything outside the page box discarded."""
    words = page.extract_words()
    inside = [w for w in words if -1 <= w["x0"] and w["x1"] <= page.width + 1]
    return inside or words


def _peaks(values: list[float], bin_width: float, min_separation: float,
           take: int) -> list[float]:
    """The ``take`` strongest modes of a 1-D distribution, left to right."""
    histogram = Counter(round(v / bin_width) for v in values)
    chosen: list[tuple[int, int]] = []
    for centre, mass in sorted(histogram.items(), key=lambda kv: -kv[1]):
        position = centre * bin_width
        if all(abs(position - c * bin_width) >= min_separation for c, _ in chosen):
            chosen.append((centre, mass))
        if len(chosen) == take:
            break
    return sorted(c * bin_width for c, _ in chosen)


def fit_grid(pdf, pages: list[int], bin_width: float = 0.004,
             min_separation: float = 0.022) -> DeGrid:
    """Fit the German column grid from the right edges of numeric words.

    ``pages`` are 1-based page numbers of table pages. One peak is expected for the
    sequence number and one for each of :data:`NUMERIC_ORDER`; sparse columns
    (``conf_israelite``, ``lang_other``, ``foreigners``) contribute weak peaks, so
    the fit pools many pages rather than trusting any single one.
    """
    edges: list[float] = []
    used = 0
    for number in pages:
        page = pdf.pages[number - 1]
        words = _table_words(page)
        if len(words) < 300:            # a part-page or a non-table page
            continue
        low, high = _page_extent(words, page.width)
        span = high - low
        if span <= 0:
            continue
        used += 1
        edges += [(w["x1"] - low) / span for w in words if _INT_RE.match(w["text"])]

    wanted = 1 + len(NUMERIC_ORDER)     # seq + the numeric columns
    centres = _peaks(edges, bin_width, min_separation, wanted)
    if len(centres) < wanted:
        raise ValueError(f"found {len(centres)} column peaks, expected {wanted}")

    # The name column has no right-edge peak — it is left-aligned and ragged — so
    # it must be given its own interval explicitly. Without one, the interval that
    # ought to belong to ``area_ha`` starts immediately after the sequence number
    # and swallows the whole name column along with every figure printed inside it.
    seq_peak, area_peak, *rest = centres
    boundaries = [
        0.0,
        seq_peak + 0.010,                       # end of the sequence number
        max(seq_peak + 0.015, area_peak - 0.050),  # end of the name column
    ]
    # A right-aligned column ends at its own peak, so each remaining boundary sits
    # just right of the peak to its left, bounded by the midpoint to the next.
    for left, right in zip([area_peak, *rest], rest):
        boundaries.append(min(left + 0.010, (left + right) / 2))
    boundaries.append(min(rest[-1] + 0.010, 1.0))
    boundaries.append(1.0 + 1e-6)               # postal key and station note
    columns = ["seq", "name", *NUMERIC_ORDER, "station_note"]
    if len(boundaries) != len(columns) + 1:
        raise ValueError(f"{len(boundaries)} boundaries for {len(columns)} columns")
    return DeGrid(boundaries=boundaries, columns=columns, n_pages=used)


def extract_rows_de(pdf, page_number: int, grid: DeGrid,
                    line_gap: float = 3.0) -> list[Row]:
    """Read one German page into :class:`~utils.lexikon_1910.lexikon_1910.Row` objects.

    Words are assigned by their **right edge**, which is what the typesetting
    aligns. The name column is not a peak — it is everything between the sequence
    number and the area column — so it is reconstructed from the words that fall in
    that interval, in reading order.
    """
    page = pdf.pages[page_number - 1]
    words = _table_words(page)
    if len(words) < 200:
        return []
    low, high = _page_extent(words, page.width)
    span = high - low
    edges = [low + f * span for f in grid.boundaries]

    def column_of(word: dict) -> str | None:
        x = word["x1"]
        for i, (left, right) in enumerate(zip(edges, edges[1:])):
            if left <= x < right:
                return grid.columns[i] if i < len(grid.columns) else None
        return None

    # Cluster into printed lines on the vertical centre of each word.
    ordered = sorted(words, key=lambda w: ((w["top"] + w["bottom"]) / 2, w["x0"]))
    lines: list[list[dict]] = []
    for word in ordered:
        centre = (word["top"] + word["bottom"]) / 2
        if lines and centre - ((lines[-1][-1]["top"] + lines[-1][-1]["bottom"]) / 2) <= line_gap:
            lines[-1].append(word)
        else:
            lines.append([word])

    rows: list[Row] = []
    for line in lines:
        cells: dict[str, list[str]] = {}
        for word in sorted(line, key=lambda w: w["x0"]):
            column = column_of(word)
            if column is None:
                continue
            text = _MARKER_RE.sub("", word["text"])
            # A token with no digit in it landing in a numeric column is the name
            # running long, not a value — the name column has no right-edge peak of
            # its own to anchor it. A token that *does* carry digits stays in the
            # numeric column and is left for the strict parser to judge.
            if column != "seq" and not _HAS_DIGIT_RE.search(text):
                column = "name"
            cells.setdefault(column, []).append(text)
        if not cells:
            continue
        rows.append(Row(
            page=f"de{page_number:04d}",
            top=int(min(w["top"] for w in line)),
            bottom=int(max(w["bottom"] for w in line)),
            cells={k: " ".join(v).strip() for k, v in cells.items() if k in COL_NAMES},
        ))
    return _body_rows(rows)


def _body_rows(rows: list[Row], min_figures: int = 4) -> list[Row]:
    """Trim the column headings above the table and the footnote block below it.

    The German page has no rules to bound the body, so the body is defined by its
    own content: it runs from the first to the last line carrying several numeric
    cells. Lines between those two are kept whatever they contain, so that wrapped
    names and ``darunter Militär`` sub-rows survive.
    """
    def figures(row: Row) -> int:
        return sum(1 for c in NUMERIC_ORDER if _HAS_DIGIT_RE.search(row.get(c) or ""))

    dense = [i for i, r in enumerate(rows) if figures(r) >= min_figures]
    if not dense:
        return []
    return rows[dense[0]:dense[-1] + 1]


# --------------------------------------------------------------------------- #
# Matching the two volumes
# --------------------------------------------------------------------------- #

#: Minimum name similarity for a cross-volume pair to be accepted.
NAME_SIMILARITY = 82

#: A best match must beat the runner-up by this margin. Where it does not, the pair
#: is refused outright rather than settled on lookup order — the same tie discipline
#: the district matcher uses (``reference/PLACE_MATCHING.md``, D-22/D-23).
TIE_MARGIN = 6

#: Structural lines printed inside the table that are not municipalities.
_HEADING_RE = re.compile(
    r"politischer\s+bezirk|gerichtsbezirk|zusammen|summe|darunter", re.IGNORECASE)


def is_heading(block) -> bool:
    """True for an okres or court-district heading masquerading as a block."""
    return bool(_HEADING_RE.search(block.municipality.get("name") or ""))


def match_blocks(czech_blocks, german_blocks,
                 min_similarity: int = NAME_SIMILARITY,
                 tie_margin: int = TIE_MARGIN) -> list[tuple]:
    """Pair Czech and German municipality blocks within **one okres**, by name.

    The printed running number looks like the obvious join key and is not one. Each
    volume lists the municipalities of an okres in **its own language's alphabetical
    order** — Chotěboř runs Bílek, Bořek, Čachotín, Dobkov … in the Czech volume and
    Altmansko, Audlaw, Audolen, Auhrow, Benatek … in the German — so the same
    municipality carries a different number in each. Joining on the number pairs
    unrelated places while looking perfectly well-behaved.

    So the join is on the name, which both volumes print in both languages
    ("Bílek, Bilek" against "Bilek, Bílek"). A pair is accepted only when the best
    candidate clears ``min_similarity`` *and* beats the runner-up by ``tie_margin``;
    a tie is refused rather than broken on order. Each German block is used once.

    Returns ``(score, czech_block, german_block)`` triples.
    """
    from rapidfuzz import fuzz

    candidates = [b for b in german_blocks if not is_heading(b)]
    keys = [normalise_name(b.name) for b in candidates]
    matched: list[tuple] = []
    taken: set[int] = set()

    for czech in czech_blocks:
        if is_heading(czech):
            continue
        target = normalise_name(czech.name)
        if len(target) < 4:
            continue
        scored = sorted(
            ((fuzz.token_set_ratio(target, key), i)
             for i, key in enumerate(keys) if i not in taken and len(key) >= 4),
            reverse=True,
        )
        if not scored:
            continue
        best_score, best_index = scored[0]
        runner_up = scored[1][0] if len(scored) > 1 else 0
        if best_score < min_similarity or best_score - runner_up < tie_margin:
            continue
        taken.add(best_index)
        matched.append((best_score, czech, candidates[best_index]))
    return matched


def compare_cells(czech_block, german_block, columns: list[str] | None = None) -> list[dict]:
    """Cell-by-cell comparison of two matched municipality rows.

    Only the municipality row is compared. The settlement rows beneath it are
    printed in the same order in both volumes, but a single dropped row would shift
    the whole comparison, and there is no independent key to detect that — so the
    comparison stops where it can be trusted.
    """
    from .lexikon_1910 import parse_int_cell

    columns = columns or NUMERIC_ORDER
    findings: list[dict] = []
    for column in columns:
        cz_value, cz_status = parse_int_cell(czech_block.municipality.get(column))
        de_value, de_status = parse_int_cell(german_block.municipality.get(column))
        if cz_status == "bad" or de_status == "bad":
            verdict = "unparseable"
        elif cz_status == "blank" and de_status == "blank":
            verdict = "na"
        elif cz_value == de_value or (cz_value or 0) == (de_value or 0):
            verdict = "pass"
        else:
            verdict = "fail"
        findings.append({
            "block_id": czech_block.block_id, "row_kind": "municipality",
            "row_name": czech_block.name, "check": f"C_crossvolume:{column}",
            "verdict": verdict,
            "detail": "" if verdict in ("pass", "na")
                      else f"cz={czech_block.municipality.get(column)!r} "
                           f"de={german_block.municipality.get(column)!r}",
        })
    return findings


def lexikon_1910_de_report(grid: DeGrid, matched: int, findings: list[dict]) -> str:
    """Coverage and agreement summary for the German channel."""
    counts = Counter(f["verdict"] for f in findings)
    graded = counts["pass"] + counts["fail"] + counts["unparseable"]
    rate = f"{counts['pass'] / graded:.1%}" if graded else "--"
    return "\n".join([
        f"lexikon_1910_de v{__version__} — cross-volume channel",
        f"  grid fitted on {grid.n_pages} German pages, "
        f"{len(grid.columns)} columns",
        f"  municipality blocks matched by name within okres: {matched}",
        f"  cell comparisons: {counts['pass']} agree, {counts['fail']} disagree, "
        f"{counts['unparseable']} unreadable, {counts['na']} both blank",
        f"  agreement on gradeable cells: {rate}",
    ])
