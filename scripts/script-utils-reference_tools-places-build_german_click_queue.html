How it works

The Czech E3 was checked against 141 glyphs the author clicked (build_glyph_bank.py); the German has no click at all, so nothing can say whether a placed box sits on the right mark (on the German gold, the Czech-trained placement puts boxes on key digits). This writes the same box questions the Token Review page already knows (build_token_review.box_items), but for every token of the German gold sheets and on the section images the route itself cuts (output/frozen/ de-bohemia_gold/sections), so a click is in the coordinates the locator works in and nothing has to be mapped back.

One question per (entry, token): the gold says how many of that mark the entry prints; click each on the entry's crop or on the whole section. Calibration and purposive sheets only (never certification, never the reserve pages). Writes a queue folder of its own, so the Czech queue is untouched; publish it as the Token Review page is published.

    [--kinds offices] [--role calibration,purposive]

def entries_of(block: dict) -> dict:
    """``{muni: {"answer": [...], "settlement": name, "count": Counter(token)}}`` of one sheet block."""
    from build_token_review import _BRACKET_RE
    from read_transcription_md import _key

    entries: dict = {}
    for _lineno, line in block["lines"]:
        parsed = _key(line)
        if not parsed or parsed[0] == "__" or not parsed[-1]:
            continue
        muni, _seq, _name, settlement, _flag, answer = parsed
        entry = entries.setdefault(int(muni), {"answer": [], "settlement": settlement, "count": Counter()})
        entry["answer"].append(answer)
        entry["count"].update(t for t in _BRACKET_RE.findall(answer) if t not in NOT_A_MARK)
    return entries

def build(roles: list[str], kinds: set, out_dir: Path) -> dict:
    from PIL import Image

    from build_token_review import _sheet_sections, _stack, municipality_bands
    from utils.lexikon_1910 import lexikon_1910_notes as lxnt

    if set(roles) & set(SEALED):
        raise SystemExit(f"refused: {sorted(set(roles) & set(SEALED))} pages are sealed (D-55, D-173)")
    with (SECTIONS / "index.csv").open(encoding="utf-8", newline="") as fh:
        index = list(csv.DictReader(fh))
    by_section: dict = {}
    for row in index:
        by_section.setdefault((row["page"], row["district"], row["kind"]), []).append(row)

    (out_dir / "img").mkdir(parents=True, exist_ok=True)
    (out_dir / "glyph").mkdir(parents=True, exist_ok=True)
    items, skipped = [], []
    for role in roles:
        for path in sorted((GOLD_DE / role).glob("[0-9][0-9][0-9][0-9].md")):
            text = path.read_text(encoding="utf-8")
            headings = [l for l in text.splitlines() if re.match(r"^##\s*2[ab]\s*—", l)]
            for n, block in enumerate(_sheet_sections(text)):
                kind = "parts" if block["section"] == "2a" else "offices"
                found = COURT_RE.search(headings[n]) if n < len(headings) else None
                district = lxnt.normalise(found.group(1)) if found else ""
                rows = by_section.get((path.stem, district, kind))
                if kind not in kinds:
                    continue
                if not rows:
                    skipped.append((path.stem, district, kind))
                    continue
                entries = entries_of(block)
                if not any(e["count"] for e in entries.values()):
                    continue
                rows = sorted(rows, key=lambda r: int(r["order"]))
                names = [r["image"] for r in rows]
                images = [Image.open(SECTIONS / name) for name in names]
                bands = municipality_bands(images, list(entries))
                full = []
                for name, image in zip(names, images):
                    jpg = f"img/{Path(name).stem}.jpg"
                    if not (out_dir / jpg).exists():
                        image.convert("RGB").save(out_dir / jpg, quality=80)
                    full.append({"src": jpg, "image": name, "size": list(image.size)})
                for muni, entry in entries.items():
                    pieces = ([(names[i], top, bottom) for i, top, bottom in bands[muni]]
                              if muni in bands else [(nm, 0, im.height) for nm, im in zip(names, images)])
                    for token, count in sorted(entry["count"].items()):
                        item_id = f"{path.stem}-{kind}-{muni}-box-{token.strip('[]')}"
                        crop = f"img/{item_id}.jpg"
                        stacked = _stack([images[names.index(nm)].crop(
                            (0, top, images[names.index(nm)].width, bottom)) for nm, top, bottom in pieces])
                        stacked.save(out_dir / crop, quality=88)
                        glyph = GLYPHS / f"{token.strip('[]')}.png"
                        if glyph.exists():
                            shutil.copy(glyph, out_dir / "glyph" / glyph.name)
                        items.append({
                            "id": item_id, "source": f"gold:de/{role}",
                            "sheet": str(path.relative_to(ROOT)).replace("\\", "/"),
                            "scan": path.stem, "section": kind, "kind": "box",
                            "muni": muni, "seq": "", "settlement": entry["settlement"],
                            "line": 0, "token": token, "expected": count,
                            "answer": "; ".join(entry["answer"]), "suggestion": "",
                            "note": "", "issue": "", "column": "", "candidates": [],
                            "glyph": f"glyph/{glyph.name}" if glyph.exists() else "",
                            "pieces": pieces, "located": muni in bands,
                            "crop": crop, "crop_size": list(stacked.size),
                            "full": [f["src"] for f in full], "sections": full})
    vocabulary = sorted({i["token"] for i in items})
    glyphs = {t: f"glyph/{t.strip('[]')}.png" for t in vocabulary if (out_dir / "glyph" / f"{t.strip('[]')}.png").exists()}
    queue = {"built": date.today().isoformat(), "items": items, "vocabulary": vocabulary, "glyphs": glyphs}
    (out_dir / "queue.json").write_text(json.dumps(queue, ensure_ascii=False, indent=1), encoding="utf-8")
    if TEMPLATE_PAGE.exists():
        shutil.copy(TEMPLATE_PAGE, out_dir / "index.html")
    print(f"{len(items)} questions, {sum(i['expected'] for i in items)} glyphs to click, "
          f"{sum(1 for i in items if i['located'])} entries located on the crop; "
          f"sections with no saved image: {skipped or 'none'} -> {out_dir}")
    return queue
