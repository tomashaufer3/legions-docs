How it works

data/raw/census_hu/tables/8_MO_1865_1910_settlement.accdb (GISta Hungarorum, 12,542 settlements of 1910 Hungary) is an Access database. No Python reader for it is installed here, so this script reads it through the Windows ACE OLEDB provider (Microsoft.ACE.OLEDB.16.0, falling back to 12.0), driven by PowerShell's System.Data.OleDb:

  1. the accdb is copied to a temporary directory (never opened in place, never written next to);
  2. the copy is opened read-only (Mode=Read) and every table in TABLES is dumped as tab-separated text, each value formatted with the invariant culture (so a coordinate is 18.1324304 whatever the machine's locale), NULL written as the empty string;
  3. Python reads each dump as strings and writes data/raw/census_hu/settlements_1910/&lt;table&gt;.csv (UTF-8, LF, every value a string, no type inference) plus manifest.json: the source path, its md5 and size, the provider used, and each table's columns and row count.

The pipeline (utils.parsing.birth_settlement_sk) reads only the CSVs. Re-running on an unchanged accdb rewrites byte-identical CSVs (the manifest's exported time aside).

Run::

python utils/reference_tools/places/export_gista_settlements.py

def export(accdb: Path = ACCDB, out_dir: Path = OUT_DIR, tables=TABLES) -> dict:
    """Copy *accdb* to a temporary directory, dump *tables*, write the CSVs and the manifest."""
    accdb, out_dir = Path(accdb), Path(out_dir)
    src_md5, size = md5(accdb), accdb.stat().st_size
    out_dir.mkdir(parents=True, exist_ok=True)
    manifest = {"script": "utils/reference_tools/places/export_gista_settlements.py", "version": __version__,
                "source": accdb.relative_to(ROOT).as_posix(), "source_md5": src_md5, "source_bytes": size,
                "exported": time.strftime("%Y-%m-%d %H:%M"), "tables": {}}
    with tempfile.TemporaryDirectory(prefix="gista_") as tmp:
        tmp = Path(tmp)
        copy = tmp / "copy.accdb"
        shutil.copyfile(accdb, copy)
        assert md5(copy) == src_md5, "the temporary copy differs from the source"
        ps1 = tmp / "dump.ps1"
        ps1.write_text(_PS, encoding="utf-8-sig")
        for t in tables:
            tsv = tmp / f"{t}.tsv"
            prov, n, fixed = _dump(ps1, copy, t, tsv)
            df = pd.read_csv(tsv, sep="\t", dtype=str, keep_default_na=False, quoting=3, encoding="utf-8")
            assert len(df) == n, f"{t}: {len(df)} rows read back, {n} dumped"
            df.to_csv(out_dir / f"{t}.csv", index=False, encoding="utf-8", lineterminator="\n")
            manifest["provider"] = prov
            manifest["tables"][t] = {"file": f"{t}.csv", "rows": n, "columns": list(df.columns),
                                     "values_with_tab_or_newline": fixed}
        assert md5(accdb) == src_md5, "the source accdb changed during the export"
    (out_dir / "manifest.json").write_text(json.dumps(manifest, ensure_ascii=False, indent=1) + "\n",
                                           encoding="utf-8", newline="\n")
    return manifest

"""Export the GISta Hungarorum settlement database to plain CSV tables the pipeline can read.

``data/raw/census_hu/tables/8_MO_1865_1910_settlement.accdb`` (GISta Hungarorum, 12,542 settlements of
1910 Hungary) is an Access database. No Python reader for it is installed here, so this script reads it
through the Windows **ACE OLEDB** provider (``Microsoft.ACE.OLEDB.16.0``, falling back to 12.0), driven by
PowerShell's ``System.Data.OleDb``:

1. the accdb is **copied** to a temporary directory (never opened in place, never written next to);
2. the copy is opened read-only (``Mode=Read``) and every table in :data:`TABLES` is dumped as
   tab-separated text, each value formatted with the **invariant culture** (so a coordinate is
   ``18.1324304`` whatever the machine's locale), ``NULL`` written as the empty string;
3. Python reads each dump as strings and writes ``data/raw/census_hu/settlements_1910/<table>.csv``
   (UTF-8, LF, every value a string, no type inference) plus ``manifest.json``: the source path, its
   md5 and size, the provider used, and each table's columns and row count.

The pipeline (:mod:`utils.parsing.birth_settlement_sk`) reads only the CSVs. Re-running on an unchanged
accdb rewrites byte-identical CSVs (the manifest's ``exported`` time aside).

Run::

    python utils/reference_tools/places/export_gista_settlements.py

Version 1.0.0 (2026-11-17): first version.
"""

from __future__ import annotations

import hashlib
import json
import shutil
import subprocess
import sys
import tempfile
import time
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[3]
ACCDB = ROOT / "data" / "raw" / "census_hu" / "tables" / "8_MO_1865_1910_settlement.accdb"
OUT_DIR = ROOT / "data" / "raw" / "census_hu" / "settlements_1910"

__version__ = "1.0.0"

# The tables the Slovak settlement gazetteer reads (birth_settlement_sk.build_gazetteer_sk):
TABLES = (
    "MAI_NEV_ES_ORSZAG",                 # IDTel1910 -> 1912 name, járás/county, today's name (MAI NEV) and country
    "MO_1910_ALAPADAT",                  # IDTel1910 -> 1912 name, járás/county names, seat flags
    "MO_1910_TER_LAK_ETN_telepulesek",   # 1910 area, population, mother tongue, religion
    "MO_1900_nevvarians",                # 1900 gazetteer names + the 1904-14 official (BM) names
    "MO_COORDINATES_settlement_CENTER",  # settlement centre lon/lat
    "MO_HAZAK_KOR_IROLVAS_1910_JAV",     # 1910 houses, age, literacy
)
PROVIDERS = ("Microsoft.ACE.OLEDB.16.0", "Microsoft.ACE.OLEDB.12.0")

# PowerShell: open the COPY read-only, dump one table as TSV with invariant-culture values. Tabs and
# newlines inside a value are replaced by a space (none occur in these tables; the count is printed).
_PS = r"""
param([string]$Db, [string]$Table, [string]$Out, [string]$Provider)
$ErrorActionPreference = 'Stop'
$inv = [System.Globalization.CultureInfo]::InvariantCulture
$cn = New-Object System.Data.OleDb.OleDbConnection("Provider=$Provider;Data Source=$Db;Mode=Read;Persist Security Info=False")
$cn.Open()
try {
  $cmd = $cn.CreateCommand(); $cmd.CommandText = "SELECT * FROM [$Table]"
  $rd = $cmd.ExecuteReader()
  $sw = New-Object System.IO.StreamWriter($Out, $false, (New-Object System.Text.UTF8Encoding($false)))
  $sw.NewLine = "`n"
  $names = @(); for ($i = 0; $i -lt $rd.FieldCount; $i++) { $names += $rd.GetName($i) }
  $sw.WriteLine(($names -join "`t"))
  $n = 0; $fixed = 0
  while ($rd.Read()) {
    $vals = New-Object string[] $rd.FieldCount
    for ($i = 0; $i -lt $rd.FieldCount; $i++) {
      $v = $rd.GetValue($i)
      if ($v -is [System.DBNull]) { $s = '' }
      elseif ($v -is [System.IFormattable]) { $s = $v.ToString($null, $inv) }   # a double: 15 significant digits
      else { $s = [string]$v }
      if ($s -match "[`t`r`n]") { $s = $s -replace "[`t`r`n]+", ' '; $fixed++ }
      $vals[$i] = $s
    }
    $sw.WriteLine(($vals -join "`t")); $n++
  }
  $sw.Close(); $rd.Close()
  Write-Output "$n $fixed"
} finally { $cn.Close() }
"""


def md5(path: Path) -> str:
    h = hashlib.md5()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def _dump(ps1: Path, db: Path, table: str, out: Path) -> tuple[str, int, int]:
    """Dump *table* of the copy *db* to *out* (TSV); returns (provider, rows, values with a tab/newline)."""
    last = ""
    for prov in PROVIDERS:
        r = subprocess.run(["powershell.exe", "-NoProfile", "-NonInteractive", "-ExecutionPolicy", "Bypass",
                            "-File", str(ps1), "-Db", str(db), "-Table", table, "-Out", str(out),
                            "-Provider", prov], capture_output=True, text=True)
        if r.returncode == 0 and r.stdout.strip():
            n, fixed = r.stdout.split()[-2:]
            return prov, int(n), int(fixed)
        last = (r.stderr or r.stdout).strip()
    raise RuntimeError(f"could not read {table} through ACE OLEDB: {last[:500]}")

                "exported": time.strftime("%Y-%m-%d %H:%M"), "tables": {}}

if __name__ == "__main__":
    m = export()
    for t, v in m["tables"].items():
        print(f"{t:<36}{v['rows']:>7} rows  {len(v['columns'])} columns")
    print("source md5", m["source_md5"], "| provider", m["provider"], "| ->", OUT_DIR.relative_to(ROOT))
    sys.exit(0)
