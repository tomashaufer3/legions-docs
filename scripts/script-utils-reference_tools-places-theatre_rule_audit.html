How it works

Two samples, one page (Theatre Rule Audit):

  • audit — reference/places/theatre_rule_audit.csv (D-210): 299 of the 24,911 pre-1914 clauses the rules coded, one per text and place, each in the stratum of its rarest flag (no_flag holds the dated clauses the rules made nothing of);
  • gold — reference/places/theatre_model_gold.csv (D-157): 308 of the pre-1914 clauses the rules leave open, stratified by source × kind.

Every clause has a weight (stratum size / drawn), so every share is weighted back to its population, and a calibration and a certification half.

Four questions a clause (D-212). The author found the model gold's 22 event types too many to rule with certainty, so the verdict asks only what reaches the measure, and every question has can't tell:

  1. activity — does the clause put Czech amateur theatre in this place at this date? yes / hedged (yes, but the clause hedges: patrně, snad, ?) / no / german (theatre, but German) / society (a society, no theatre named) / unsure;
  2. company — does it record a company existing over time (founded, listed, said to be active)? yes / no / unsure;
  3. performance — a performance, a named play or a number of plays? same;
  4. date — a dated clause: the rules' reading is right / wrong (with the right years) / unsure; an undated one: the years it dates the event to (given), none, or unsure.

On an audit clause the rules' answers are filled in and the author verifies them (D-210, the author's choice: faster, but anchored); on a gold clause nothing is, since the rules have none.

Scoring (score) turns a verdict into the measure's flags — activity (czech_activity), certain (activity and not hedged), company, performance, and activity_society (the reading in which a society's founding counts) — and compares a reader's flags with them: the rules as drawn, the rules of the current build, or a model's readings run through amateur_theatre.code_clauses exactly as the rules' cues are. Precision, recall and agreement come with stratified bootstrap intervals (King & Lowe 2003; Webber 2013); score_by splits them by land, settlement size, Czech share and source, because error correlated with those is what would bias a coefficient (Johansen, Koschnick & Vedel 2026; Weidmann 2016).

Usage::

python utils/reference_tools/places/theatre_rule_audit.py migrate   # once: verdict columns
python utils/reference_tools/places/theatre_rule_audit.py page OUT.html

def population(clauses: pd.DataFrame) -> pd.DataFrame:
    """The audited clauses: pre-1914 (yes/ambiguous), not puppet, dated, not left to the model,
    both chronicles; one per text and place (the place chronicle's copy kept), with ``stratum``."""
    pop = atm.select_coded(clauses)
    pop["text_key"] = _text_key(pop)
    pop["_order"] = pop["source"].map({"chronicle": 0, "ensemble_chronicle": 1})
    pop = pop.sort_values(["_order", "clause_key"]).drop_duplicates("text_key").drop(columns="_order")
    pop["has_count"] = pop["count_stated"].notna()
    pop["has_titles"] = pop["titles"].fillna("").ne("")
    flags = pop[STRATUM_FLAGS].astype(bool)
    rarest = sorted(STRATUM_FLAGS, key=lambda f: int(flags[f].sum()))
    pop["stratum"] = "no_flag"
    for flag in reversed(rarest):                      # the rarest flag is written last and wins
        pop.loc[flags[flag], "stratum"] = flag
    return pop.reset_index(drop=True)

def draw(pop: pd.DataFrame, seed: int = SEED) -> pd.DataFrame:
    """The stratified sample with ``half``, ``stratum_N``, ``stratum_n`` and ``weight``, shuffled."""
    rng = np.random.default_rng(seed)
    sizes = pop.groupby("stratum").size()
    take = np.minimum(sizes, np.maximum(MIN_PER_STRATUM,
                                        (sizes / sizes.sum() * N_PROPORTIONAL).round().astype(int)))
    parts = []
    for stratum, group in pop.groupby("stratum"):
        n = int(take[stratum])
        drawn = group.iloc[np.sort(rng.permutation(len(group))[:n])].copy()
        drawn["half"] = ["calibration" if i % 2 == 0 else "certification" for i in range(n)]
        drawn["stratum_N"], drawn["stratum_n"] = len(group), n
        drawn["weight"] = len(group) / n
        parts.append(drawn)
    sample = pd.concat(parts, ignore_index=True)
    return sample.iloc[rng.permutation(len(sample))].reset_index(drop=True)

def write_sheet(sample: pd.DataFrame, path: Path = SHEET) -> Path:

        raise FileExistsError(f"{path} already holds verdicts; it is not redrawn")

    sheet.to_csv(path, index=False, encoding="utf-8-sig")

def migrate(clauses: pd.DataFrame | None = None) -> dict:
    """Give both sheets the verdict columns, and the gold sheet what the page and the
    weights need (``stratum``, ``stratum_N``/``_n``, ``weight``, ``glossary``, the rules'
    date reading).  The type-level columns of either sheet must be empty; the gold sheet's
    ``gold_*`` columns are kept as they are.  A sheet that has ``v_activity`` is left alone."""
    clauses = atm.load_clauses() if clauses is None else clauses
    rows = clauses.set_index("clause_key")
    done = {}
    audit = pd.read_csv(SHEET, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    if "v_activity" not in audit:
        assert not (audit[[c for c in _OLD_AUDIT_COLUMNS if c in audit]] != "").any().any(), \
            "audit: the old type-level columns hold rulings"
        audit = audit.drop(columns=[c for c in _OLD_AUDIT_COLUMNS if c in audit])
        drawn = rows.reindex(audit["clause_key"])
        assert drawn["text"].notna().all(), "audit: a drawn clause is missing from the build"
        audit = pd.concat([audit, _drawn_flags(drawn.reset_index(drop=True))], axis=1)
        for column in VERDICT_COLUMNS:
            audit[column] = ""
        audit.to_csv(SHEET, index=False, encoding="utf-8-sig")
        done["audit"] = len(audit)
    gold = pd.read_csv(atm.GOLD_SHEET, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    if "v_activity" not in gold:
        assert not (gold[atm.GOLD_COLUMNS] != "").any().any(), "gold: the type-level columns hold rulings"
        open_ = atm.select_open(clauses, "pre1914")
        sizes = open_.groupby(["source", "open_kind"]).size()
        gold["stratum"] = gold["source"] + "/" + gold["open_kind"]
        gold["stratum_N"] = [int(sizes[(s, k)]) for s, k in zip(gold["source"], gold["open_kind"])]
        gold["stratum_n"] = gold.groupby("stratum")["clause_key"].transform("size")
        gold["weight"] = (gold["stratum_N"] / gold["stratum_n"]).round(4)
        contexted = atm.add_context(open_[open_["clause_key"].isin(gold["clause_key"])])
        gloss = contexted.set_index("clause_key")["glossary"]
        gold["glossary"] = gold["clause_key"].map(gloss).fillna("")
        drawn = rows.reindex(gold["clause_key"])
        gold["drawn_date_form"] = drawn["date_form"].fillna("").values
        for column in ("start_lo", "start_hi", "end_lo", "end_hi"):
            gold["drawn_" + column] = drawn[column].map(_year).values
        for column in VERDICT_COLUMNS:
            gold[column] = ""
        gold.to_csv(atm.GOLD_SHEET, index=False, encoding="utf-8-sig")
        done["gold"] = len(gold)
    return done

def page_clauses(samples: dict[str, pd.DataFrame], seed: int = SEED) -> list[dict]:
    """Both samples as the page reads them, calibration halves first, shuffled within a half.
    An audit clause carries the rules' answers (``prefill``); a gold clause carries none.
    Never the stratum or the weight."""
    out = []
    for name, sheet in samples.items():
        for r in sheet.itertuples(index=False):
            dated = r.date_raw.strip() != ""
            prefill = ({"activity": r.drawn_activity, "company": r.drawn_company,
                        "performance": r.drawn_performance} if name == "audit" else None)
            out.append({"key": r.clause_key, "sample": name, "half": r.half, "source": r.source,
                        "place": r.place, "ensemble": r.ensemble, "context": r.context,
                        "glossary": r.glossary, "clause": r.clause, "date_raw": r.date_raw,
                        "dated": dated, "date_form": r.drawn_date_form,
                        "years": [r.drawn_start_lo, r.drawn_start_hi, r.drawn_end_lo, r.drawn_end_hi],
                        "years_inside": getattr(r, "years_inside", "") or "", "prefill": prefill})
    rng = np.random.default_rng(seed)
    order = rng.permutation(len(out))
    out = [out[i] for i in order]
    out.sort(key=lambda c: c["half"] != "calibration")          # stable: calibration first
    for i, c in enumerate(out):
        c["i"] = i
    return out

def fill_sheets(documents: list[dict]) -> dict:
    """Write the page's documents (``verdicts/<clause_key>``) into the sheets' ``v_*`` columns.
    Only documents marked ``ruled`` are written; every document must name a clause of its

        index = {k: i for i, k in enumerate(sheet["clause_key"])}

                d.get("activity") or "", d.get("company") or "", d.get("performance") or "",

def verdict_rows(samples: tuple[str, ...] = ("audit", "gold"), half: str = "calibration") -> pd.DataFrame:
    """The ruled rows of the samples' ``half``, one frame, with each verdict as a flag
    (1 / 0 / NaN for *can't tell*).  The certification half is sealed until :data:`RULES_FROZEN`."""
    if half == "certification" and not RULES_FROZEN:
        raise PermissionError("the certification half is sealed until RULES_FROZEN is True")
    sheets = load_samples()
    parts = []
    for name in samples:
        s = sheets[name]
        s = s[(s["half"] == half) & s["v_activity"].ne("")].copy()
        s["sample"] = name
        s["stratum"] = name + ":" + s["stratum"]
        # D-218: an answer from the choice page is a second-phase draw — every disagreement,
        # a random share of the agreed clauses — so its weight is divided by its probability
        # and the bootstrap keeps the two groups apart
        if "v_source" in s:
            ab = s["v_source"].eq("ab")
            s.loc[ab, "stratum"] = s.loc[ab, "stratum"] + ":" + s.loc[ab, "ab_group"]
            s["_prob"] = pd.to_numeric(s["ab_prob"], errors="coerce").where(ab, 1.0).fillna(1.0)
        else:
            s["_prob"] = 1.0
        parts.append(s)
    out = pd.concat(parts, ignore_index=True)
    out["weight"] = out["weight"].astype(float) / out.pop("_prob")
    for flag, mapping in _ACTIVITY_MAP.items():
        out["gold_" + flag] = out["v_activity"].map(mapping).astype(float)
    for flag in ("company", "performance"):
        out["gold_" + flag] = out["v_" + flag].map(_YES_NO_MAP).astype(float)
    out["gold_date_ok"] = out["v_date"].map({"right": 1, "wrong": 0}).astype(float)
    return out

def score(rows: pd.DataFrame, flags: pd.DataFrame, draws: int = 1000, seed: int = SEED) -> pd.DataFrame:
    """One reader against the verdicts: per flag, rows known on both sides, *can't tell* rows,
    weighted precision, recall and agreement, each with a 95 % stratified bootstrap interval."""
    out = []
    frame = rows[["stratum", "weight"]].copy()
    for flag in FLAGS:
        frame["g"], frame["p"] = rows["gold_" + flag], flags[flag]
        prec, rec, agree = _metrics(frame["g"].to_numpy(float), frame["p"].to_numpy(float),
                                    frame["weight"].to_numpy(float))
        lo, hi = _interval(frame, "g", "p", draws, seed)
        out.append({"flag": flag, "known": int((frame["g"].notna() & frame["p"].notna()).sum()),
                    "cant_tell": int(frame["g"].isna().sum()),
                    "precision": prec, "precision_ci": (lo[0], hi[0]),
                    "recall": rec, "recall_ci": (lo[1], hi[1]),
                    "agreement": agree, "agreement_ci": (lo[2], hi[2])})
    return pd.DataFrame(out).set_index("flag")

    """A model's interval for the undated clauses against the verdict's: exact (all four
    years, or both say there is no date) and overlapping, weighted, over the undated rows
    with a verdict of ``given`` or ``none`` that the model read."""
    u = rows[rows["v_date"].isin(["given", "none"]) & rows["date_raw"].eq("")].set_index("clause_key")
    m = readings[readings["valid"].astype(bool)].set_index("clause_key").reindex(u.index)
    u, m = u[m["date_form"].notna()], m[m["date_form"].notna()]
    if not len(u):
        return {"undated_scored": 0}
    def years(frame, prefix):
        return frame[[prefix + k for k in ("start_lo", "start_hi", "end_lo", "end_hi")]].apply(pd.to_numeric, errors="coerce")
    g, p = years(u, "v_"), years(m, "")
    g_none, p_none = u["v_date"].eq("none"), m["date_form"].isin(["none", "leading"])
    same = (g_none & p_none) | (~g_none & ~p_none & (g.fillna(-1).to_numpy() == p.fillna(-1).to_numpy()).all(axis=1))
    g_lo, g_hi = g["v_start_lo"].fillna(g["v_end_lo"]), g["v_end_hi"].fillna(g["v_start_hi"])
    p_lo, p_hi = p["start_lo"].fillna(p["end_lo"]), p["end_hi"].fillna(p["start_hi"])
    overlap = (g_none & p_none) | (~g_none & ~p_none & (p_lo <= g_hi) & (g_lo <= p_hi))
    w = u["weight"]
    return {"undated_scored": len(u), "exact": float((same * w).sum() / w.sum()),
            "overlap": float((overlap * w).sum() / w.sum())}

The clause check: the author's verdicts on two samples of chronicle clauses, which score the
rules and every model reader on what enters the theatre measure (D-210, D-212).

Two samples, one page (*Theatre Rule Audit*):

- ``audit`` — ``reference/places/theatre_rule_audit.csv`` (D-210): 299 of the 24,911
  pre-1914 clauses the rules **coded**, one per text and place, each in the stratum of its
  rarest flag (``no_flag`` holds the dated clauses the rules made nothing of);
- ``gold`` — ``reference/places/theatre_model_gold.csv`` (D-157): 308 of the pre-1914
  clauses the rules leave **open**, stratified by source × kind.

Every clause has a weight (stratum size / drawn), so every share is weighted back to its
population, and a calibration and a certification half.

**Four questions a clause** (D-212).  The author found the model gold's 22 event types too
many to rule with certainty, so the verdict asks only what reaches the measure, and every
question has *can't tell*:

1. ``activity`` — does the clause put Czech amateur theatre in this place at this date?
   ``yes`` / ``hedged`` (yes, but the clause hedges: patrně, snad, ?) / ``no`` / ``german``
   (theatre, but German) / ``society`` (a society, no theatre named) / ``unsure``;
2. ``company`` — does it record a company existing over time (founded, listed, said to be
   active)?  ``yes`` / ``no`` / ``unsure``;
3. ``performance`` — a performance, a named play or a number of plays?  same;
4. ``date`` — a dated clause: the rules' reading is ``right`` / ``wrong`` (with the right
   years) / ``unsure``; an undated one: the years it dates the event to (``given``), ``none``,
   or ``unsure``.

author's choice: faster, but anchored); on a gold clause nothing is, since the rules have none.

**Scoring** (:func:`score`) turns a verdict into the measure's flags — ``activity``
(``czech_activity``), ``certain`` (activity and not hedged), ``company``, ``performance``, and

readings run through :func:`amateur_theatre.code_clauses` exactly as the rules' cues are.

Webber 2013); :func:`score_by` splits them by land, settlement size, Czech share and source,

def draw(pop: pd.DataFrame, seed: int = SEED) -> pd.DataFrame:
    """The stratified sample with ``half``, ``stratum_N``, ``stratum_n`` and ``weight``, shuffled."""
    rng = np.random.default_rng(seed)
    sizes = pop.groupby("stratum").size()
    take = np.minimum(sizes, np.maximum(MIN_PER_STRATUM,
                                        (sizes / sizes.sum() * N_PROPORTIONAL).round().astype(int)))
    parts = []
    for stratum, group in pop.groupby("stratum"):
        n = int(take[stratum])
        drawn = group.iloc[np.sort(rng.permutation(len(group))[:n])].copy()
        drawn["half"] = ["calibration" if i % 2 == 0 else "certification" for i in range(n)]
        drawn["stratum_N"], drawn["stratum_n"] = len(group), n
        drawn["weight"] = len(group) / n
        parts.append(drawn)
    sample = pd.concat(parts, ignore_index=True)
    return sample.iloc[rng.permutation(len(sample))].reset_index(drop=True)

                    "place", "ensemble", "context", "glossary", "text", "date_raw"]].rename(columns={"text": "clause"})
    sheet["weight"] = sheet["weight"].round(4)
    sheet["drawn_cues"] = sample["cues"].fillna("")
    sheet["drawn_count"] = sample["count_stated"].map(_year)
    sheet["drawn_titles"] = sample["titles"].fillna("")
    sheet["drawn_date_form"] = sample["date_form"].fillna("")
    for column in ("start_lo", "start_hi", "end_lo", "end_hi"):
        sheet["drawn_" + column] = sample[column].map(_year)
    sheet = pd.concat([sheet, _drawn_flags(sample)], axis=1)
    for column in VERDICT_COLUMNS:
        sheet[column] = ""
    sheet.to_csv(path, index=False, encoding="utf-8-sig")
    return path

def migrate(clauses: pd.DataFrame | None = None) -> dict:
    """Give both sheets the verdict columns, and the gold sheet what the page and the
    weights need (``stratum``, ``stratum_N``/``_n``, ``weight``, ``glossary``, the rules'
    date reading).  The type-level columns of either sheet must be empty; the gold sheet's
    ``gold_*`` columns are kept as they are.  A sheet that has ``v_activity`` is left alone."""
    clauses = atm.load_clauses() if clauses is None else clauses
    rows = clauses.set_index("clause_key")
    done = {}
    audit = pd.read_csv(SHEET, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    if "v_activity" not in audit:
        assert not (audit[[c for c in _OLD_AUDIT_COLUMNS if c in audit]] != "").any().any(), \
            "audit: the old type-level columns hold rulings"
        audit = audit.drop(columns=[c for c in _OLD_AUDIT_COLUMNS if c in audit])
        drawn = rows.reindex(audit["clause_key"])
        assert drawn["text"].notna().all(), "audit: a drawn clause is missing from the build"
        audit = pd.concat([audit, _drawn_flags(drawn.reset_index(drop=True))], axis=1)
        for column in VERDICT_COLUMNS:
            audit[column] = ""
        audit.to_csv(SHEET, index=False, encoding="utf-8-sig")
        done["audit"] = len(audit)
    gold = pd.read_csv(atm.GOLD_SHEET, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    if "v_activity" not in gold:
        assert not (gold[atm.GOLD_COLUMNS] != "").any().any(), "gold: the type-level columns hold rulings"
        open_ = atm.select_open(clauses, "pre1914")
        sizes = open_.groupby(["source", "open_kind"]).size()
        gold["stratum"] = gold["source"] + "/" + gold["open_kind"]
        gold["stratum_N"] = [int(sizes[(s, k)]) for s, k in zip(gold["source"], gold["open_kind"])]
        gold["stratum_n"] = gold.groupby("stratum")["clause_key"].transform("size")
        gold["weight"] = (gold["stratum_N"] / gold["stratum_n"]).round(4)
        contexted = atm.add_context(open_[open_["clause_key"].isin(gold["clause_key"])])
        gloss = contexted.set_index("clause_key")["glossary"]
        gold["glossary"] = gold["clause_key"].map(gloss).fillna("")
        drawn = rows.reindex(gold["clause_key"])
        gold["drawn_date_form"] = drawn["date_form"].fillna("").values
        for column in ("start_lo", "start_hi", "end_lo", "end_hi"):
            gold["drawn_" + column] = drawn[column].map(_year).values
        for column in VERDICT_COLUMNS:
            gold[column] = ""
        gold.to_csv(atm.GOLD_SHEET, index=False, encoding="utf-8-sig")
        done["gold"] = len(gold)
    return done


def load_samples() -> dict[str, pd.DataFrame]:
    return {name: pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
            for name, path in SAMPLES.items()}

def page_clauses(samples: dict[str, pd.DataFrame], seed: int = SEED) -> list[dict]:
    """Both samples as the page reads them, calibration halves first, shuffled within a half.
    An audit clause carries the rules' answers (``prefill``); a gold clause carries none.
    Never the stratum or the weight."""
    out = []
    for name, sheet in samples.items():
        for r in sheet.itertuples(index=False):
            dated = r.date_raw.strip() != ""
            prefill = ({"activity": r.drawn_activity, "company": r.drawn_company,
                        "performance": r.drawn_performance} if name == "audit" else None)
            out.append({"key": r.clause_key, "sample": name, "half": r.half, "source": r.source,
                        "place": r.place, "ensemble": r.ensemble, "context": r.context,
                        "glossary": r.glossary, "clause": r.clause, "date_raw": r.date_raw,
                        "dated": dated, "date_form": r.drawn_date_form,
                        "years": [r.drawn_start_lo, r.drawn_start_hi, r.drawn_end_lo, r.drawn_end_hi],
                        "years_inside": getattr(r, "years_inside", "") or "", "prefill": prefill})
    rng = np.random.default_rng(seed)
    order = rng.permutation(len(out))
    out = [out[i] for i in order]
    out.sort(key=lambda c: c["half"] != "calibration")          # stable: calibration first
    for i, c in enumerate(out):
        c["i"] = i
    return out


def build_page(out: Path) -> Path:
    data = json.dumps(page_clauses(load_samples()), ensure_ascii=False).replace("</", "<\\/")
    template = TEMPLATE.read_text(encoding="utf-8")
    assert template.count(PLACEHOLDER) == 1, "template: one data placeholder"
    Path(out).write_text(template.replace(PLACEHOLDER, data), encoding="utf-8")
    return Path(out)


def fill_sheets(documents: list[dict]) -> dict:
    """Write the page's documents (``verdicts/<clause_key>``) into the sheets' ``v_*`` columns.
    Only documents marked ``ruled`` are written; every document must name a clause of its
    sample's sheet.  Returns counts per sample."""
    counts = {}
    for name, path in SAMPLES.items():
        sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
        index = {k: i for i, k in enumerate(sheet["clause_key"])}
        mine = [d for d in documents if d.get("sample") == name]
        unknown = [d.get("clause_key") for d in mine if d.get("clause_key") not in index]
        assert not unknown, f"{name}: verdicts for clauses not in the sheet: {unknown[:5]}"
        sheet[VERDICT_COLUMNS] = ""
        written = 0
        for d in mine:
            if not d.get("ruled"):
                continue
            sheet.loc[index[d["clause_key"]], VERDICT_COLUMNS] = [
                d.get("activity") or "", d.get("company") or "", d.get("performance") or "",
                d.get("date") or "", *(_year(d.get(k)) if d.get(k) not in (None, "") else ""
                                       for k in ("start_lo", "start_hi", "end_lo", "end_hi")),
                d.get("note") or ""]
            written += 1
        sheet.to_csv(path, index=False, encoding="utf-8-sig")
        counts[name] = {"documents": len(mine), "ruled": written, "rows": len(sheet)}
    return counts

# Scoring

FLAGS = ["activity", "certain", "company", "performance", "activity_society"]
_ACTIVITY_MAP = {"activity": {"yes": 1, "hedged": 1, "no": 0, "german": 0, "society": 0},
                 "certain": {"yes": 1, "hedged": 0, "no": 0, "german": 0, "society": 0},
                 "activity_society": {"yes": 1, "hedged": 1, "no": 0, "german": 0, "society": 1}}
_YES_NO_MAP = {"yes": 1, "no": 0}


def verdict_rows(samples: tuple[str, ...] = ("audit", "gold"), half: str = "calibration") -> pd.DataFrame:
    """The ruled rows of the samples' ``half``, one frame, with each verdict as a flag
    (1 / 0 / NaN for *can't tell*).  The certification half is sealed until :data:`RULES_FROZEN`."""
    if half == "certification" and not RULES_FROZEN:
        raise PermissionError("the certification half is sealed until RULES_FROZEN is True")
    sheets = load_samples()
    parts = []
    for name in samples:
        s = sheets[name]
        s = s[(s["half"] == half) & s["v_activity"].ne("")].copy()
        s["sample"] = name
        s["stratum"] = name + ":" + s["stratum"]
        # D-218: an answer from the choice page is a second-phase draw — every disagreement,
        # a random share of the agreed clauses — so its weight is divided by its probability
        # and the bootstrap keeps the two groups apart
        if "v_source" in s:
            ab = s["v_source"].eq("ab")
            s.loc[ab, "stratum"] = s.loc[ab, "stratum"] + ":" + s.loc[ab, "ab_group"]
            s["_prob"] = pd.to_numeric(s["ab_prob"], errors="coerce").where(ab, 1.0).fillna(1.0)
        else:
            s["_prob"] = 1.0
        parts.append(s)
    out = pd.concat(parts, ignore_index=True)
    out["weight"] = out["weight"].astype(float) / out.pop("_prob")
    for flag, mapping in _ACTIVITY_MAP.items():
        out["gold_" + flag] = out["v_activity"].map(mapping).astype(float)
    for flag in ("company", "performance"):
        out["gold_" + flag] = out["v_" + flag].map(_YES_NO_MAP).astype(float)
    out["gold_date_ok"] = out["v_date"].map({"right": 1, "wrong": 0}).astype(float)
    return out


def _flags_from_coded(coded: pd.DataFrame) -> pd.DataFrame:
    act = coded["czech_activity"].astype(bool)
    return pd.DataFrame({"activity": act.astype(float),
                         "certain": (act & ~coded["hedged"].astype(bool)).astype(float),
                         "company": coded["company"].astype(bool).astype(float),
                         "performance": coded["performance"].astype(bool).astype(float),
                         "activity_society": act.astype(float)}, index=coded.index)


def reader_flags(rows: pd.DataFrame, reader: str = "rules", clauses: pd.DataFrame | None = None,
                 readings: pd.DataFrame | None = None) -> pd.DataFrame:
    """A reader's flags on ``rows`` (from :func:`verdict_rows`), same index.

    ``rules_drawn`` — the rules' answers at the draw (audit rows; a gold row has none, so 0);
    ``rules`` — the current build's coded clauses; any other name — ``readings`` (a model's,
    from :func:`amateur_theatre_model.readings_table`), applied with
    :func:`amateur_theatre_model.apply_readings` so the codebook decides the flags.  A clause
    the model has no valid reading of comes out NaN.
    """
    if reader == "rules_drawn":
        a = rows["drawn_activity"].fillna("")
        out = pd.DataFrame({f: a.map(m).fillna(0).astype(float) for f, m in _ACTIVITY_MAP.items()}, index=rows.index)
        out["activity_society"] = out["activity"]
        for flag in ("company", "performance"):
            out[flag] = rows["drawn_" + flag].fillna("").map(_YES_NO_MAP).fillna(0).astype(float)
        return out[FLAGS]
    clauses = atm.load_clauses() if clauses is None else clauses
    coded = clauses.set_index("clause_key").reindex(rows["clause_key"])
    coded.index = rows.index
    missing = coded["text"].isna()
    if missing.any():
        print(f"{reader}: {int(missing.sum())} checked clauses are not in this build; they score NaN")
    if reader == "rules":
        out = _flags_from_coded(coded[~missing])
        return out.reindex(rows.index)[FLAGS]
    assert readings is not None, "a model reader needs its readings"
    valid = readings[readings["valid"].astype(bool)]
    read = rows["clause_key"].isin(valid["clause_key"]) & ~missing
    parts = []
    for source, group in coded[read].reset_index().groupby("source"):
        applied = atm.apply_readings(group, valid, source)
        applied.index = group["index"].values
        parts.append(applied)
    out = _flags_from_coded(pd.concat(parts)) if parts else pd.DataFrame(columns=FLAGS, dtype=float)
    return out.reindex(rows.index)[FLAGS]


def _metrics(g: np.ndarray, p: np.ndarray, w: np.ndarray) -> tuple[float, float, float]:
    """Weighted precision, recall and agreement over rows where both are known."""
    ok = ~(np.isnan(g) | np.isnan(p))
    g, p, w = g[ok], p[ok], w[ok]
    def share(mask, hit):
        return float((w[mask] * hit[mask]).sum() / w[mask].sum()) if w[mask].sum() else np.nan
    return share(p == 1, g == 1), share(g == 1, p == 1), share(np.ones_like(g, bool), g == p)


def _interval(frame: pd.DataFrame, gcol: str, pcol: str, draws: int, seed: int) -> np.ndarray:
    """95 % stratified bootstrap intervals for the three metrics."""
    rng = np.random.default_rng(seed)
    groups = [g.index.to_numpy() for _, g in frame.groupby("stratum")]
    g, p, w = frame[gcol].to_numpy(float), frame[pcol].to_numpy(float), frame["weight"].to_numpy(float)
    pos = {k: i for i, k in enumerate(frame.index)}
    groups = [np.array([pos[k] for k in grp]) for grp in groups]
    stats = np.array([_metrics(g[idx], p[idx], w[idx]) for idx in
                      (np.concatenate([rng.choice(grp, len(grp)) for grp in groups]) for _ in range(draws))])
    with np.errstate(all="ignore"):
        return np.nanpercentile(stats, [2.5, 97.5], axis=0)


def score(rows: pd.DataFrame, flags: pd.DataFrame, draws: int = 1000, seed: int = SEED) -> pd.DataFrame:
    """One reader against the verdicts: per flag, rows known on both sides, *can't tell* rows,
    weighted precision, recall and agreement, each with a 95 % stratified bootstrap interval."""
    out = []
    frame = rows[["stratum", "weight"]].copy()
    for flag in FLAGS:
        frame["g"], frame["p"] = rows["gold_" + flag], flags[flag]
        prec, rec, agree = _metrics(frame["g"].to_numpy(float), frame["p"].to_numpy(float),
                                    frame["weight"].to_numpy(float))
        lo, hi = _interval(frame, "g", "p", draws, seed)
        out.append({"flag": flag, "known": int((frame["g"].notna() & frame["p"].notna()).sum()),
                    "cant_tell": int(frame["g"].isna().sum()),
                    "precision": prec, "precision_ci": (lo[0], hi[0]),
                    "recall": rec, "recall_ci": (lo[1], hi[1]),
                    "agreement": agree, "agreement_ci": (lo[2], hi[2])})
    return pd.DataFrame(out).set_index("flag")


def undated_date_score(rows: pd.DataFrame, readings: pd.DataFrame) -> dict:
    """A model's interval for the undated clauses against the verdict's: exact (all four
    years, or both say there is no date) and overlapping, weighted, over the undated rows
    with a verdict of ``given`` or ``none`` that the model read."""
    u = rows[rows["v_date"].isin(["given", "none"]) & rows["date_raw"].eq("")].set_index("clause_key")
    m = readings[readings["valid"].astype(bool)].set_index("clause_key").reindex(u.index)
    u, m = u[m["date_form"].notna()], m[m["date_form"].notna()]
    if not len(u):
        return {"undated_scored": 0}
    def years(frame, prefix):
        return frame[[prefix + k for k in ("start_lo", "start_hi", "end_lo", "end_hi")]].apply(pd.to_numeric, errors="coerce")
    g, p = years(u, "v_"), years(m, "")
    g_none, p_none = u["v_date"].eq("none"), m["date_form"].isin(["none", "leading"])
    same = (g_none & p_none) | (~g_none & ~p_none & (g.fillna(-1).to_numpy() == p.fillna(-1).to_numpy()).all(axis=1))
    g_lo, g_hi = g["v_start_lo"].fillna(g["v_end_lo"]), g["v_end_hi"].fillna(g["v_start_hi"])
    p_lo, p_hi = p["start_lo"].fillna(p["end_lo"]), p["end_hi"].fillna(p["start_hi"])
    overlap = (g_none & p_none) | (~g_none & ~p_none & (p_lo <= g_hi) & (g_lo <= p_hi))
    w = u["weight"]
    return {"undated_scored": len(u), "exact": float((same * w).sum() / w.sum()),
            "overlap": float((overlap * w).sum() / w.sum())}


def date_score(rows: pd.DataFrame) -> dict:
    """The rules' leading dates: weighted share read right among dated rows with a verdict."""
    dated = rows[rows["gold_date_ok"].notna()]
    w = dated["weight"]
    return {"dated_ruled": len(dated), "right": float((dated["gold_date_ok"] * w).sum() / w.sum()) if len(dated) else np.nan}

SIZE_BANDS = [0, 500, 2000, np.inf]
SIZE_LABELS = ["under 500", "500–1,999", "2,000 and over"]
CZECH_BANDS = [0, 0.5, 0.9, 1.0001]
CZECH_LABELS = ["Czech under 50 %", "Czech 50–90 %", "Czech 90 % and over"]


def place_covariates() -> pd.DataFrame:
    """Per database place (``geo_id``): its 1910 settlement's land, population band and
    Czech-speaking share band (the rechecked 1910 register), or ``unknown``."""
    places = pd.read_csv(ROOT / "data" / "raw" / "theatre" / "places.csv", usecols=["geo_id", "name", "type", "lat", "lon", "okres"])
    frame = at.settlement_frame(pd.read_parquet(FINAL / "geo" / "settlements_1910.parquet"),
                                pd.read_parquet(FINAL / "geo" / "districts_1910.parquet"))
    tie = at.join_places_to_settlements(places, frame)
    tie = tie.merge(frame[["settlement_row", "settlement_id", "land"]], on="settlement_row", how="left",
                    validate="m:1")
    reg = pd.read_parquet(FINAL / "register_rechecked" / "register_municipalities.parquet",
                          columns=["settlement_id", "pop_total", "lang_czech", "lang_german", "lang_polish", "lang_other"])
    speakers = reg[["lang_czech", "lang_german", "lang_polish", "lang_other"]].sum(axis=1)
    reg["czech_share"] = reg["lang_czech"] / speakers.where(speakers > 0)
    n = len(tie)
    tie = tie.merge(reg[["settlement_id", "pop_total", "czech_share"]], on="settlement_id", how="left", validate="m:1")
    assert len(tie) == n, "covariates: a merge changed the row count"
    tie["size_band"] = pd.cut(tie["pop_total"], SIZE_BANDS, labels=SIZE_LABELS, right=False).astype(object).fillna("unknown")
    tie["czech_band"] = pd.cut(tie["czech_share"], CZECH_BANDS, labels=CZECH_LABELS, right=False).astype(object).fillna("unknown")
    tie["land"] = tie["land"].fillna("unknown")
    return tie[["geo_id", "settlement_id", "land", "size_band", "czech_band"]]


def score_by(rows: pd.DataFrame, flags: pd.DataFrame, by: str, flag: str = "activity",
             covariates: pd.DataFrame | None = None) -> pd.DataFrame:
    """One flag's weighted precision, recall and agreement within each group of ``by``
    (``land``, ``size_band``, ``czech_band`` or ``source``), with the group's row count."""
    if by != "source":
        covariates = place_covariates() if covariates is None else covariates
        geo = rows["clause_key"].str.split(":").str[1].astype(int)
        groups = geo.map(covariates.set_index("geo_id")[by]).fillna("unknown")
    else:
        groups = rows["source"]
    out = []
    for value, idx in groups.groupby(groups).groups.items():
        p, r, a = _metrics(rows.loc[idx, "gold_" + flag].to_numpy(float), flags.loc[idx, flag].to_numpy(float),
                           rows.loc[idx, "weight"].to_numpy(float))
        out.append({by: value, "rows": len(idx), "precision": p, "recall": r, "agreement": a})
    return pd.DataFrame(out).set_index(by)


def report(pop: pd.DataFrame, sample: pd.DataFrame) -> pd.DataFrame:
    """Audit strata: population size, drawn, weight, and the calibration half."""
    out = (sample.groupby("stratum").agg(N=("stratum_N", "first"), drawn=("clause_key", "size"),
                                         calibration=("half", lambda h: int((h == "calibration").sum())),
                                         weight=("weight", "first"))
           .sort_values("N", ascending=False))
    assert out["N"].sum() == len(pop), "strata do not partition the population"
    return out
