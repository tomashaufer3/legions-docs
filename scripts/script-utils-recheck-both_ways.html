How it works

D-190: the correction pass becomes the analysis baseline only if it lowers neither the route's score on the sealed pages nor the D-181 retest. D-196 adds the one check no step was tuned on, German Böhmen against the Czech volume. Two parts, each written to OUT:

  • editions — German Böhmen against the Czech volume (editions_check): the two editions symbol by symbol, route and rechecked, and every change of the merge judged. It fails where the recheck adds more disagreements on a symbol than allowed (two, or LOWER_BY of the symbol's comparisons, whichever is more — one municipality must not sink a rare symbol), or where a step of the correction with STEP_MIN judged changes or more is right on fewer than STEP_RIGHT of them. The Czech volume is not the truth (its own errors, and the few places the editions differ: claims all four readers agree on are borne out 94 % of the time), and adding claims that are right 80 % of the time still raises F1 on a table near 0.97, so the bar catches a step that does harm (§6's box answers, 5 % right), not noise. The author's own steps are listed for review, never gated: his rulings outrank the check (D-190). The bars were set on 2026-11-08 after a first look at this check (D-196). Free, and safe to run as often as needed: the Czech volume is not a sealed page.
  • sealed — the full run's readings of the sealed pages scored twice by score_certification's own metrics (score_sealed_retest.score_run): the route's tables and the rechecked ones. It fails where the recheck lowers claim F1. Run it once, at the end, on the machine of record: the sealed pages are the certification's, and a score looked at while the correction is still being changed would make them in sample for it (D-127).

Nothing is written into a run folder.

def editions_report(r: dict) -> str:
    lines = [f"German Böhmen against the Czech volume (both_ways {__version__}, D-196) — judged by {r['czech']}",
             f"  municipalities both read: {r['both_read'][0]} (route) → {r['both_read'][1]} (rechecked); "
             f"German blind: {r['blind_in_german'][0]} → {r['blind_in_german'][1]}", "",
             f"  {'symbol':32s} {'route':>7s} {'recheck':>8s}   (Czech only / German only)"]
    for s in r["symbols"]:
        lines.append(f"  {s['symbol']:32s} {s['route']:7.3f} {s['rechecked']:8.3f}   "
                     f"({s['route_cz_only']}/{s['route_de_only']} → {s['rechecked_cz_only']}/{s['rechecked_de_only']})")
    lines += ["", f"  {'step':8s} {'action':7s} {'right':>6s} {'wrong':>6s} {'unjudged':>8s}  reason"]
    for c in r["steps"]:
        lines.append(f"  {c['step']:8s} {c['action']:7s} {c['right']:6d} {c['wrong']:6d} {c['unjudged']:8d}  {c['reason']}")
    lines += ["", f"  symbols with more extra disagreements than allowed (2, or {LOWER_BY:.1%}): {', '.join(r['lowered']) or 'none'}",
              f"  correction steps right on under {STEP_RIGHT:.0%} of {STEP_MIN}+ judged changes: {'; '.join(r['weak_steps']) or 'none'}",
              f"  your own rulings the Czech volume does not bear out (to look at, not gated): "
              f"{'; '.join(r.get('author_to_review', [])) or 'none'}",
              f"  THE CZECH-VOLUME CHECK {'PASSES' if r['passes'] else 'DOES NOT PASS'}"]
    return "\n".join(lines)

def sealed(edition: str, recheck: Path = RECHECK, out: Path = OUT, force: bool = False) -> dict | None:
    """The full run's readings of ``edition``'s sealed pages, scored with the route's tables and with
    the rechecked ones (``score_sealed_retest.score_run`` on a temporary copy). Passes where the
    rechecked claim F1 is not lower. Once, at the end, on the machine of record (see the module)."""
    import score_sealed_retest as ssr

    done = Path(out) / f"sealed_{edition}.json"
    if done.exists() and not force:              # scored once (D-127): a second run returns the first (0.1.3)
        print(f"  {edition}: the sealed pages were scored already ({done.name}); pass force=True only on the author's word")
        return json.loads(done.read_text("utf-8"))

    sc = ssr.sc
    paths = sc.sealed_paths(edition)
    if not paths or any(sc.sheet_status(p) != "complete" for p in paths):
        return None
    source = sc.FROZEN / f"{edition}_full"
    inventory = Path(recheck) / edition / "final_inventory_rechecked.csv"
    if not (source / "final_inventory.csv").is_file() or not inventory.exists():
        return None
    route, _sides = ssr.score_run(edition, f"{edition}_full")
    real = sc.FROZEN
    with tempfile.TemporaryDirectory() as tmp:
        folder = Path(tmp) / f"{edition}_rechecked"
        for name in ssr.COPIED:
            if (source / name).is_file():
                (folder / name).parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(source / name, folder / name)
        shutil.copy2(inventory, folder / "final_inventory.csv")
        abbreviations = Path(recheck) / edition / "final_abbreviations_rechecked.csv"
        if abbreviations.exists():
            shutil.copy2(abbreviations, folder / "final_abbreviations.csv")
        sc.FROZEN = Path(tmp)
        try:
            rechecked, _sides = ssr.score_run(edition, f"{edition}_rechecked")
        finally:
            sc.FROZEN = real
    f1 = lambda r: (r.get("as_printed") or {}).get("f1")       # noqa: E731  (score_certification.measure)
    result = {"edition": edition, "check": __version__, "route": route, "rechecked": rechecked,
              "f1": (f1(route), f1(rechecked))}
    result["passes"] = None not in result["f1"] and result["f1"][1] >= result["f1"][0]
    out = Path(out)
    out.mkdir(parents=True, exist_ok=True)
    (out / f"sealed_{edition}.json").write_text(json.dumps(result, indent=1, ensure_ascii=False, default=str),
                                                encoding="utf-8")
    text = "\n".join([f"Sealed pages, route and rechecked — {edition} (both_ways {__version__})",
                      "######## the route's tables", sc.report(route), "",
                      "######## the rechecked tables", sc.report(rechecked), "",
                      f"  F1 {result['f1'][0]} → {result['f1'][1]}: "
                      f"{'PASSES' if result['passes'] else 'DOES NOT PASS'} (D-190: the correction may not lower it)"])
    (out / f"sealed_{edition}.txt").write_text(text, encoding="utf-8")
    return result

def _csv(path: Path) -> list[dict]:
    import csv
    with open(path, encoding="utf-8", newline="") as fh:
        return list(csv.DictReader(fh))
