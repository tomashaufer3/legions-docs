How it works

Produces publication-ready booktabs tables for inclusion in LaTeX documents via \input{tables/name.tex}. Four table types are supported: summary_table, regression_table, series_table, exit_code_table, and a thin wrapper missing_table.

Escaping convention

All user-facing string inputs (caption, notes, col_header, var_labels values, FE/fit row labels, and Series indices) are treated as plain text and escaped by the module. Do not pre-escape them.

If you need raw LaTeX in any of those fields (e.g. math, a command, or \textemdash), wrap it in RawLaTeX::

summary_table(df, var_labels={"y": RawLaTeX(r"Δyt")}, ...)
series_table(s, col_header=RawLaTeX(r"Missing (\%)"), ...)

Hard-coded literals inside the module (booktabs rules, \\, \% formatting in numeric cells) remain raw — only user input is escaped.

def series_table(
    s: pd.Series,
    col_header: Stringy = "Value",
    caption: Stringy = "",
    label: str = "tab:series",
    notes: Stringy = None,
    path: str | Path | None = None,
    fmt: str = "{:.1f}",
    position: str = "htbp",
    font_size: str = "small",
    escape_index: bool = True,
) -> str:
    """Render a pandas Series as a two-column booktabs table.

    Parameters
    ----------
    s : pd.Series
        Index becomes the left column; values the right.
    col_header : str or RawLaTeX
        Header for the value column.  Pass :class:`RawLaTeX` for symbols.
    fmt : str
        Format string for numeric values.  If you want a ``%`` sign in the
        cell, use ``r"{:.1f}\\%"`` (raw string, single backslash).
    escape_index : bool
        Escape the index labels.  Set False only if you've pre-built the
        labels as LaTeX.
    """
    rows = []
    rows.append("& " + _esc(col_header) + r" \\")
    rows.append(r"\midrule")
    for idx, val in s.items():
        idx_str = _esc(str(idx)) if escape_index else str(idx)
        val_str = _fmt(val, fmt)
        rows.append(idx_str + " & " + val_str + r" \\")

    body = "\n".join(rows)
    return _write(path, _wrap_table(
        body, caption, label, notes,
        position, font_size, "lr",
        multicolumn_header=None,
    ))

def summary_table(
    df: pd.DataFrame,
    columns: Sequence[str] | None = None,
    stats: Sequence[str] = ("N", "Mean", "Std. Dev.", "Min", "Max"),
    var_labels: dict[str, Stringy] | None = None,
    caption: Stringy = "Summary Statistics",
    label: str = "tab:summary",
    notes: Stringy = None,
    path: str | Path | None = None,
    fmt_mean: str = "{:.2f}",
    fmt_sd: str = "{:.2f}",
    fmt_min: str = "{:.2f}",
    fmt_max: str = "{:.2f}",
    position: str = "htbp",
    font_size: str = "small",
) -> str:
    """Summary statistics table (N, Mean, Std. Dev., Min, Max).

    ``var_labels`` values are escaped automatically; wrap any LaTeX-bearing
    label in :class:`RawLaTeX` to bypass escaping (e.g. for math).
    """
    if columns is None:
        columns = list(df.select_dtypes("number").columns)

    var_labels = var_labels or {}
    stat_set   = set(stats)

    header_parts = [""]
    for s in ("N", "Mean", "Std. Dev.", "Min", "Max"):
        if s in stat_set:
            header_parts.append(_esc(s))   # "Std. Dev." has no specials but escape for safety

    n_cols = len(header_parts) - 1
    col_format = "l" + "r" * n_cols

    rows = []
    rows.append(" & ".join(header_parts) + r" \\")
    rows.append(r"\midrule")

    for col in columns:
        label_str = _esc(var_labels.get(col, col))
        series    = df[col].dropna()
        parts     = [label_str]
        if "N"         in stat_set: parts.append(_fmt(len(series), "{:.0f}"))
        if "Mean"      in stat_set: parts.append(_fmt(series.mean(), fmt_mean))
        if "Std. Dev." in stat_set: parts.append(_fmt(series.std(),  fmt_sd))
        if "Min"       in stat_set: parts.append(_fmt(series.min(),  fmt_min))
        if "Max"       in stat_set: parts.append(_fmt(series.max(),  fmt_max))
        rows.append(" & ".join(parts) + r" \\")

    body = "\n".join(rows)
    return _write(path, _wrap_table(
        body, caption, label, notes,
        position, font_size, col_format,
        multicolumn_header=None,
    ))

def regression_table(
    results: list[dict],
    dep_var: Stringy = "",
    col_labels: Sequence[Stringy] | None = None,
    fe_rows: dict[str, list] | None = None,
    fit_rows: dict[str, list] | None = None,
    notes: Stringy = None,
    caption: Stringy = "Regression Results",
    label: str = "tab:regression",
    path: str | Path | None = None,
    sig_stars: bool = True,
    position: str = "htbp",
    font_size: str = "small",
) -> str:
    """Coefficient/SE regression table with FE indicators and fit rows.

    Variable names (dict keys in ``results`` / ``fe_rows`` / ``fit_rows``)
    are escaped automatically.  Use :class:`RawLaTeX` for math-bearing
    labels (``RawLaTeX(r"$\\Delta x$")``).
    """
    n_models   = len(results)
    col_labels = col_labels or [f"({i+1})" for i in range(n_models)]

    all_vars: list[str] = []
    seen: set[str] = set()
    for res in results:
        for var in res:
            if var not in seen:
                all_vars.append(var)
                seen.add(var)

    col_format = "l" + "c" * n_models

    rows = []

    if dep_var:
        rows.append(
            r"& \multicolumn{" + str(n_models) + r"}{c}{" +
            _esc(dep_var) + r"} \\"
        )
        rows.append(f"\\cmidrule(lr){{2-{n_models + 1}}}")

    rows.append("& " + " & ".join(_esc(c) for c in col_labels) + r" \\")
    rows.append(r"\midrule")

    def _stars(pval: float) -> str:
        if not sig_stars or pd.isna(pval):
            return ""
        if pval < 0.01: return r"$^{***}$"
        if pval < 0.05: return r"$^{**}$"
        if pval < 0.10: return r"$^{*}$"
        return ""

    for var in all_vars:
        coef_cells, se_cells = [], []
        for res in results:
            entry = res.get(var)
            if entry is None:
                coef_cells.append("")
                se_cells.append("")
            else:
                coef, se = entry[0], entry[1]
                pval     = entry[2] if len(entry) > 2 else float("nan")
                coef_cells.append(f"{_fmt(coef)}{_stars(pval)}")
                se_cells.append(f"({_fmt(se)})")

        rows.append(f"{_esc(var)} & " + " & ".join(coef_cells) + r" \\")
        rows.append("& " + " & ".join(se_cells) + r" \\")

    if fe_rows:
        rows.append(r"\midrule")
        for fe_label, indicators in fe_rows.items():
            cells = []
            for ind in indicators:
                if ind is True:
                    cells.append(r"\checkmark")
                elif ind is False:
                    cells.append("$-$")
                else:
                    cells.append(_esc(ind) if not isinstance(ind, RawLaTeX) else str(ind))
            rows.append(f"{_esc(fe_label)} & " + " & ".join(cells) + r" \\")

    if fit_rows:
        rows.append(r"\midrule")
        for fit_label, values in fit_rows.items():
            cells = []
            for v in values:
                if isinstance(v, bool):
                    cells.append(r"\checkmark" if v else "$-$")
                elif isinstance(v, int):
                    cells.append(f"{v:,}")
                elif isinstance(v, float):
                    cells.append(f"{v:.3f}")
                elif isinstance(v, RawLaTeX):
                    cells.append(str(v))
                else:
                    cells.append(_esc(str(v)))
            rows.append(f"{_esc(fit_label)} & " + " & ".join(cells) + r" \\")

    body = "\n".join(rows)
    return _write(path, _wrap_table(
        body, caption, label, notes,
        position, font_size, col_format,
        multicolumn_header=None,
    ))

def exit_code_table(
    df: pd.DataFrame,
    caption: Stringy = "Exit Code Distribution — Czechoslovak Legions",
    label: str = "tab:exit_codes",
    notes: Stringy = None,
    path: str | Path | None = None,
    position: str = "htbp",
    font_size: str = "small",
) -> str:
    """Render the exit-code frequency table as a booktabs LaTeX table.

    Requires that ``add_exit_columns(df)`` has already been called.
    """
    from collections import Counter

    try:
        from utils.parsing.exit_codes import CODES_CS, CODES_EN
    except ImportError:
        CODES_CS, CODES_EN = {}, {}

    if "exit_codes" not in df.columns:
        raise ValueError("Run add_exit_columns(df) first.")

    counts = Counter(code for codes in df["exit_codes"] for code in codes)
    n = len(df)

    col_format = "llrr"
    rows = []
    # Header — % must be raw \% (single backslash) since this string is hard-coded
    rows.append(r"Code & Description & $n$ & \% \\")
    rows.append(r"\midrule")

    for code, _cs_label in CODES_CS.items():
        en_label = CODES_EN.get(code, "")
        c        = counts.get(code, 0)
        pct      = c / n * 100
        rows.append(
            _esc(code) + " & " + _esc(en_label) + " & " +
            f"{c:,}" + " & " + f"{pct:.2f}" + r"\% \\"
        )

    n_empty = sum(1 for s in df["exit_codes"] if len(s) == 0)
    rows.append(r"\midrule")
    rows.append(
        r"\textemdash & No code parsed & " +
        f"{n_empty:,}" + " & " + f"{n_empty/n*100:.2f}" + r"\% \\"
    )

    body = "\n".join(rows)
    return _write(path, _wrap_table(
        body, caption, label, notes,
        position, font_size, col_format,
        multicolumn_header=None,
    ))

def df_to_latex(df, path, caption, label, column_format=None, header=None,
                index=True, index_name=None, notes=None, position="H"):
    """Save a DataFrame as a booktabs LaTeX table.

    Parameters
    ----------
    df            : DataFrame to render
    path          : output path (str or Path)
    caption       : table caption
    label         : LaTeX label (without \\label{})
    column_format : e.g. "lrrrr"; inferred if None
    header        : list of column header strings; uses df.columns if None
    index         : whether to include the index (default True)
    index_name    : override df.index.name in the header row
    notes         : optional note appended below \\end{tabular}
    position      : float specifier (default "H")
    """
    if index_name is not None:
        df = df.copy()
        df.index.name = index_name

    if column_format is None:
        n_cols = len(df.columns) + (1 if index else 0)
        column_format = "l" + "r" * (n_cols - 1)

    kwargs = dict(
        float_format="{:.2f}".format,
        column_format=column_format,
        caption=caption,
        label=label,
        position=position,
        escape=False,
        index=index,
    )
    if header is not None:
        kwargs["header"] = header

    latex = df.to_latex(**kwargs)

    # booktabs rules
    latex = (
        latex
        .replace("\\hline\n\\hline", "\\toprule")
        .replace("\\hline\n", "\\midrule\n", 1)
        .replace("\\hline", "\\bottomrule")
    )

    # Remove spurious blank index-name row pandas emits when index has a name
    if index and df.index.name:
        spurious = df.index.name + " & " + " & ".join([""] * len(df.columns)) + " \\\\\n"
        latex = latex.replace(spurious, "")

    # Optional notes
    if notes:
        latex = latex.replace(
            "\\end{tabular}",
            "\\end{tabular}\n\\\\[-0.5em]\n{\\footnotesize " + notes + "}",
        )

    # pandas runs with escape=False, so a literal "%" in any header/cell text
    # (e.g. a "Share (%)" column) survives raw and comments out the rest of the
    # line — including the row-ending "\\". df_to_latex never emits a legitimate
    # bare "%", so escape any "%" that isn't already backslash-escaped.
    latex = re.sub(r"(?<!\\)%", r"\\%", latex)

    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w", encoding="utf-8") as f:
        f.write(latex)

#!/usr/bin/env python3
"""
LaTeX table generation utilities.

Produces publication-ready booktabs tables for inclusion in LaTeX documents
via ``\\input{tables/name.tex}``.  Four table types are supported:
``summary_table``, ``regression_table``, ``series_table``, ``exit_code_table``,
and a thin wrapper ``missing_table``.

Escaping convention
-------------------
All user-facing string inputs (``caption``, ``notes``, ``col_header``,
``var_labels`` values, FE/fit row labels, and Series indices) are treated
as **plain text** and escaped by the module.  Do not pre-escape them.

If you need raw LaTeX in any of those fields (e.g. math, a command, or
``\\textemdash``), wrap it in :class:`RawLaTeX`::

    summary_table(df, var_labels={"y": RawLaTeX(r"$\\Delta y_t$")}, ...)
    series_table(s, col_header=RawLaTeX(r"Missing (\\%)"), ...)

Hard-coded literals inside the module (booktabs rules, ``\\\\``, ``\\%``
formatting in numeric cells) remain raw — only user input is escaped.

Author: Tomáš Haufer
"""

from __future__ import annotations

import re
from pathlib import Path
from typing import Sequence, Union

import pandas as pd

__version__ = "1.2.0"


# ---------------------------------------------------------------------------
# RawLaTeX sentinel — opt-out of escaping
# ---------------------------------------------------------------------------

class RawLaTeX(str):
    """Marker subclass: instances bypass ``_esc()``.

    Use for fields where the user genuinely needs LaTeX commands or math::

        caption=RawLaTeX(r"Distribution of $\\beta_i$")
    """
    __slots__ = ()


Stringy = Union[str, RawLaTeX, None]


# ---------------------------------------------------------------------------
# Internal helpers
# ---------------------------------------------------------------------------

def _esc(text: Stringy) -> str:
    """Escape LaTeX special characters in *text*.

    ``None`` becomes ``""``.  :class:`RawLaTeX` instances pass through
    unchanged.  Plain ``str`` is fully escaped via a single regex pass so
    that replacement output (e.g. the ``{}`` in ``\\textbackslash{}``)
    is not itself re-escaped.
    """
    if text is None:
        return ""
    if isinstance(text, RawLaTeX):
        return str(text)
    return _ESC_RE.sub(lambda m: _ESC_MAP[m.group(0)], str(text))


_ESC_MAP = {
    "\\": r"\textbackslash{}",
    "&":  r"\&",
    "%":  r"\%",
    "$":  r"\$",
    "#":  r"\#",
    "_":  r"\_",
    "{":  r"\{",
    "}":  r"\}",
    "~":  r"\textasciitilde{}",
    "^":  r"\textasciicircum{}",
}
_ESC_RE = re.compile("|".join(re.escape(k) for k in _ESC_MAP))


def _write(path: str | Path | None, latex: str) -> str:
    """Write *latex* to *path* (creating parent dirs) and return it."""
    if path is not None:
        p = Path(path)
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(latex, encoding="utf-8")
        print(f"  -> {p}")
    return latex


def _fmt(value, fmt: str = "{:.2f}") -> str:
    """Format a numeric value; return empty string for NaN."""
    if pd.isna(value):
        return ""
    try:
        return fmt.format(float(value))
    except (TypeError, ValueError):
        return str(value)


# ---------------------------------------------------------------------------
# Table wrapper
# ---------------------------------------------------------------------------

def _wrap_table(
    body: str,
    caption: Stringy,
    label: str,
    notes: Stringy,
    position: str,
    font_size: str,
    col_format: str,
    multicolumn_header: tuple[Stringy, int] | None,
) -> str:
    """Wrap a tabular body in a full table environment."""
    lines = []
    lines.append(f"\\begin{{table}}[{position}]")
    lines.append(r"\centering")
    lines.append(f"\\caption{{{_esc(caption)}}}")
    lines.append(f"\\label{{{label}}}")
    if font_size:
        lines.append(f"\\{font_size}")
    lines.append(f"\\begin{{tabular}}{{{col_format}}}")
    lines.append(r"\toprule")
    if multicolumn_header:
        text, n = multicolumn_header
        lines.append(
            r"& \multicolumn{" + str(n) + r"}{c}{" + _esc(text) + r"} \\"
        )
        lines.append(f"\\cmidrule(lr){{2-{n + 1}}}")
    lines.append(body)
    lines.append(r"\bottomrule")
    lines.append(r"\end{tabular}")
    if notes:
        lines.append(
            r"\begin{minipage}{\linewidth}"
            r"\vspace{4pt}"
            r"\footnotesize"
            r"\textit{Notes:} " + _esc(notes) +
            r"\end{minipage}"
        )
    lines.append(r"\end{table}")
    return "\n".join(lines)


# ---------------------------------------------------------------------------
# series_table
# ---------------------------------------------------------------------------

def series_table(
    s: pd.Series,
    col_header: Stringy = "Value",
    caption: Stringy = "",
    label: str = "tab:series",
    notes: Stringy = None,
    path: str | Path | None = None,
    fmt: str = "{:.1f}",
    position: str = "htbp",
    font_size: str = "small",
    escape_index: bool = True,
) -> str:
    """Render a pandas Series as a two-column booktabs table.

    Parameters
    ----------
    s : pd.Series
        Index becomes the left column; values the right.
    col_header : str or RawLaTeX
        Header for the value column.  Pass :class:`RawLaTeX` for symbols.
    fmt : str
        Format string for numeric values.  If you want a ``%`` sign in the
        cell, use ``r"{:.1f}\\%"`` (raw string, single backslash).
    escape_index : bool
        Escape the index labels.  Set False only if you've pre-built the
        labels as LaTeX.
    """
    rows = []
    rows.append("& " + _esc(col_header) + r" \\")
    rows.append(r"\midrule")
    for idx, val in s.items():
        idx_str = _esc(str(idx)) if escape_index else str(idx)
        val_str = _fmt(val, fmt)
        rows.append(idx_str + " & " + val_str + r" \\")

    body = "\n".join(rows)
    return _write(path, _wrap_table(
        body, caption, label, notes,
        position, font_size, "lr",
        multicolumn_header=None,
    ))


# ---------------------------------------------------------------------------
# summary_table
# ---------------------------------------------------------------------------

def summary_table(
    df: pd.DataFrame,
    columns: Sequence[str] | None = None,
    stats: Sequence[str] = ("N", "Mean", "Std. Dev.", "Min", "Max"),
    var_labels: dict[str, Stringy] | None = None,
    caption: Stringy = "Summary Statistics",
    label: str = "tab:summary",
    notes: Stringy = None,
    path: str | Path | None = None,
    fmt_mean: str = "{:.2f}",
    fmt_sd: str = "{:.2f}",
    fmt_min: str = "{:.2f}",
    fmt_max: str = "{:.2f}",
    position: str = "htbp",
    font_size: str = "small",
) -> str:
    """Summary statistics table (N, Mean, Std. Dev., Min, Max).

    ``var_labels`` values are escaped automatically; wrap any LaTeX-bearing
    label in :class:`RawLaTeX` to bypass escaping (e.g. for math).
    """
    if columns is None:
        columns = list(df.select_dtypes("number").columns)

    var_labels = var_labels or {}
    stat_set   = set(stats)

    header_parts = [""]
    for s in ("N", "Mean", "Std. Dev.", "Min", "Max"):
        if s in stat_set:
            header_parts.append(_esc(s))   # "Std. Dev." has no specials but escape for safety

    n_cols = len(header_parts) - 1
    col_format = "l" + "r" * n_cols

    rows = []
    rows.append(" & ".join(header_parts) + r" \\")
    rows.append(r"\midrule")

    for col in columns:
        label_str = _esc(var_labels.get(col, col))
        series    = df[col].dropna()
        parts     = [label_str]
        if "N"         in stat_set: parts.append(_fmt(len(series), "{:.0f}"))
        if "Mean"      in stat_set: parts.append(_fmt(series.mean(), fmt_mean))
        if "Std. Dev." in stat_set: parts.append(_fmt(series.std(),  fmt_sd))
        if "Min"       in stat_set: parts.append(_fmt(series.min(),  fmt_min))
        if "Max"       in stat_set: parts.append(_fmt(series.max(),  fmt_max))
        rows.append(" & ".join(parts) + r" \\")

    body = "\n".join(rows)
    return _write(path, _wrap_table(
        body, caption, label, notes,
        position, font_size, col_format,
        multicolumn_header=None,
    ))

# regression_table
# ---------------------------------------------------------------------------

def regression_table(
    results: list[dict],
    dep_var: Stringy = "",
    col_labels: Sequence[Stringy] | None = None,
    fe_rows: dict[str, list] | None = None,
    fit_rows: dict[str, list] | None = None,
    notes: Stringy = None,
    caption: Stringy = "Regression Results",
    label: str = "tab:regression",
    path: str | Path | None = None,
    sig_stars: bool = True,
    position: str = "htbp",
    font_size: str = "small",
) -> str:
    """Coefficient/SE regression table with FE indicators and fit rows.

    Variable names (dict keys in ``results`` / ``fe_rows`` / ``fit_rows``)
    are escaped automatically.  Use :class:`RawLaTeX` for math-bearing
    labels (``RawLaTeX(r"$\\Delta x$")``).
    """
    n_models   = len(results)
    col_labels = col_labels or [f"({i+1})" for i in range(n_models)]

    all_vars: list[str] = []
    seen: set[str] = set()
    for res in results:
        for var in res:
            if var not in seen:
                all_vars.append(var)
                seen.add(var)

    col_format = "l" + "c" * n_models

    rows = []

    if dep_var:
        rows.append(
            r"& \multicolumn{" + str(n_models) + r"}{c}{" +
            _esc(dep_var) + r"} \\"
        )
        rows.append(f"\\cmidrule(lr){{2-{n_models + 1}}}")

    rows.append("& " + " & ".join(_esc(c) for c in col_labels) + r" \\")
    rows.append(r"\midrule")

    def _stars(pval: float) -> str:
        if not sig_stars or pd.isna(pval):
            return ""
        if pval < 0.01: return r"$^{***}$"
        if pval < 0.05: return r"$^{**}$"
        if pval < 0.10: return r"$^{*}$"
        return ""

    for var in all_vars:
        coef_cells, se_cells = [], []
        for res in results:
            entry = res.get(var)
            if entry is None:
                coef_cells.append("")
                se_cells.append("")
            else:
                coef, se = entry[0], entry[1]
                pval     = entry[2] if len(entry) > 2 else float("nan")
                coef_cells.append(f"{_fmt(coef)}{_stars(pval)}")
                se_cells.append(f"({_fmt(se)})")

        rows.append(f"{_esc(var)} & " + " & ".join(coef_cells) + r" \\")
        rows.append("& " + " & ".join(se_cells) + r" \\")

    if fe_rows:
        rows.append(r"\midrule")
        for fe_label, indicators in fe_rows.items():
            cells = []
            for ind in indicators:
                if ind is True:
                    cells.append(r"\checkmark")
                elif ind is False:
                    cells.append("$-$")
                else:
                    cells.append(_esc(ind) if not isinstance(ind, RawLaTeX) else str(ind))
            rows.append(f"{_esc(fe_label)} & " + " & ".join(cells) + r" \\")

    if fit_rows:
        rows.append(r"\midrule")
        for fit_label, values in fit_rows.items():
            cells = []
            for v in values:
                if isinstance(v, bool):
                    cells.append(r"\checkmark" if v else "$-$")
                elif isinstance(v, int):
                    cells.append(f"{v:,}")
                elif isinstance(v, float):
                    cells.append(f"{v:.3f}")
                elif isinstance(v, RawLaTeX):
                    cells.append(str(v))
                else:
                    cells.append(_esc(str(v)))
            rows.append(f"{_esc(fit_label)} & " + " & ".join(cells) + r" \\")

    body = "\n".join(rows)
    return _write(path, _wrap_table(
        body, caption, label, notes,
        position, font_size, col_format,
        multicolumn_header=None,
    ))

# exit_code_table
# ---------------------------------------------------------------------------

def exit_code_table(
    df: pd.DataFrame,
    caption: Stringy = "Exit Code Distribution — Czechoslovak Legions",
    label: str = "tab:exit_codes",
    notes: Stringy = None,
    path: str | Path | None = None,
    position: str = "htbp",
    font_size: str = "small",
) -> str:
    """Render the exit-code frequency table as a booktabs LaTeX table.

    Requires that ``add_exit_columns(df)`` has already been called.
    """
    from collections import Counter

    try:
        from utils.parsing.exit_codes import CODES_CS, CODES_EN
    except ImportError:
        CODES_CS, CODES_EN = {}, {}

    if "exit_codes" not in df.columns:
        raise ValueError("Run add_exit_columns(df) first.")

    counts = Counter(code for codes in df["exit_codes"] for code in codes)
    n = len(df)

    col_format = "llrr"
    rows = []
    # Header — % must be raw \% (single backslash) since this string is hard-coded
    rows.append(r"Code & Description & $n$ & \% \\")
    rows.append(r"\midrule")

    for code, _cs_label in CODES_CS.items():
        en_label = CODES_EN.get(code, "")
        c        = counts.get(code, 0)
        pct      = c / n * 100
        rows.append(
            _esc(code) + " & " + _esc(en_label) + " & " +
            f"{c:,}" + " & " + f"{pct:.2f}" + r"\% \\"
        )

    n_empty = sum(1 for s in df["exit_codes"] if len(s) == 0)
    rows.append(r"\midrule")
    rows.append(
        r"\textemdash & No code parsed & " +
        f"{n_empty:,}" + " & " + f"{n_empty/n*100:.2f}" + r"\% \\"
    )

    body = "\n".join(rows)
    return _write(path, _wrap_table(
        body, caption, label, notes,
        position, font_size, col_format,
        multicolumn_header=None,
    ))


# ---------------------------------------------------------------------------
# missing_table  (convenience wrapper)
# ---------------------------------------------------------------------------

def missing_table(
    miss: pd.Series,
    caption: Stringy = "Missing Rates — Core Analysis Columns",
    label: str = "tab:missing",
    notes: Stringy = None,
    path: str | Path | None = None,
    position: str = "htbp",
    font_size: str = "small",
) -> str:
    """Render a missing-rate Series as a booktabs table.

    The Series index is treated as plain text and escaped — column names
    like ``birth_year`` will render correctly.
    """
    return series_table(
        miss,
        col_header=RawLaTeX(r"Missing (\%)"),   # symbol → bypass escaping
        caption=caption,
        label=label,
        notes=notes,
        path=path,
        fmt=r"{:.1f}\%",                        # raw string, single backslash
        position=position,
        font_size=font_size,
    )


# ---------------------------------------------------------------------------
# df_to_latex
# ---------------------------------------------------------------------------
def df_to_latex(df, path, caption, label, column_format=None, header=None,
                index=True, index_name=None, notes=None, position="H"):
    """Save a DataFrame as a booktabs LaTeX table.

    Parameters
    ----------
    df            : DataFrame to render
    path          : output path (str or Path)
    caption       : table caption
    label         : LaTeX label (without \\label{})
    column_format : e.g. "lrrrr"; inferred if None
    header        : list of column header strings; uses df.columns if None
    index         : whether to include the index (default True)
    index_name    : override df.index.name in the header row
    notes         : optional note appended below \\end{tabular}
    position      : float specifier (default "H")
    """
    if index_name is not None:
        df = df.copy()
        df.index.name = index_name

    if column_format is None:
        n_cols = len(df.columns) + (1 if index else 0)
        column_format = "l" + "r" * (n_cols - 1)

    kwargs = dict(
        float_format="{:.2f}".format,
        column_format=column_format,
        caption=caption,
        label=label,
        position=position,
        escape=False,
        index=index,
    )
    if header is not None:
        kwargs["header"] = header

    latex = df.to_latex(**kwargs)

    # booktabs rules
    latex = (
        latex
        .replace("\\hline\n\\hline", "\\toprule")
        .replace("\\hline\n", "\\midrule\n", 1)
        .replace("\\hline", "\\bottomrule")
    )

    # Remove spurious blank index-name row pandas emits when index has a name
    if index and df.index.name:
        spurious = df.index.name + " & " + " & ".join([""] * len(df.columns)) + " \\\\\n"
        latex = latex.replace(spurious, "")

    # Optional notes
    if notes:
        latex = latex.replace(
            "\\end{tabular}",
            "\\end{tabular}\n\\\\[-0.5em]\n{\\footnotesize " + notes + "}",
        )

    # pandas runs with escape=False, so a literal "%" in any header/cell text
    # (e.g. a "Share (%)" column) survives raw and comments out the rest of the
    # line — including the row-ending "\\". df_to_latex never emits a legitimate
    # bare "%", so escape any "%" that isn't already backslash-escaped.
    latex = re.sub(r"(?<!\\)%", r"\\%", latex)

    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w", encoding="utf-8") as f:
        f.write(latex)
