How it works

Every stage of the route writes its own file and none overwrites another (D-38, D-55), so nothing is lost — but nothing joins the stages per mark either. This module is that join: an append-only ledger of events, one row per step's reading of one mark, from which the final value of every mark is computed, each mark's trail printed, and each step's changes counted.

A mark is a glyph on the page. Where the locator placed it, its id is the section image and the box (box_mark), so every reader's reading of the same glyph lands on the same mark; where no box was placed, it is the part and the token (unplaced_mark).

The steps, in route order (STEPS):

read each reader's reading — a token, or ABSENT (it read the entry and claimed nothing here) order the printed-order rule flagging a reader's mark as out of place dinov3 the encoder's name for the box (note: agree / disagree / …) cz_de the other edition's reading of the same part opus the verify call's answer, one event per repeat author a ruling by hand; overrides everything before it final the value final_value computes — written, never edited

Nothing in an event is ever changed; a later step adds a row.

A mark's final value is not yet a claim's. A token question ("is T printed for this part?") rules on the claim (part, T) wherever the readers filed it, so its answer sits on the claim's own mark and claim_rulings hands it to the inventory, which applies it over every mark of that part (1.2.0, D-120).

def by_mark(events: list[Event]) -> dict[str, list[Event]]:

    for e in events:

def majority(values: list[str]) -> str | None:
    """The most frequent value, or ``None`` on a tie at the top."""
    if not values:
        return None
    (top, n), *rest = Counter(values).most_common()
    return None if rest and rest[0][1] == n else top

def final_value(events: list[Event], record: str) -> tuple[str, str]:
    """``(value, rule)`` for one mark, by the route of D-72 as amended.

    The author's last ruling wins. Otherwise Opus's majority over its decisive
    answers. Where Opus was asked and settled nothing (:data:`UNSETTLED`: it cannot
    tell, the print is none of the options, the outline sat on another mark), the
    readers' value stands if every reading agrees, and the mark is ``open`` — the
    author's queue — if they do not (D-120). A mark Opus was not asked about takes
    the readers' value when every reading agrees and nothing flags it; otherwise the
    reader of record's reading (``record``, e.g. ``gemini38#1``) — a mark the route
    should have asked about and did not.
    """
    author = [e.read for e in events if e.step == "author"]
    if author:
        return author[-1], "author"
    reads = [e.read for e in events if e.step == "read"]
    unanimous = bool(reads) and len(set(reads)) == 1
    asked = [e.read for e in events if e.step == "opus"]
    if asked:
        decisive = [v for v in asked if v not in UNSETTLED]
        if decisive:                               # a split among its answers is open
            top = majority(decisive)
            return (top, "opus") if top is not None else (OPEN, "opus could not tell")
        if unanimous:
            return reads[0], "opus settled nothing; readers agree"
        return OPEN, "opus could not tell"
    if unanimous and not flagged(events):
        return reads[0], "agreed"
    own = [e.read for e in events if e.step == "read" and e.actor == record]
    return (own[0] if own else ABSENT), "not asked: reader of record"

def with_final(events: list[Event], record: str, version: str = __version__) -> list[Event]:
    """The ledger plus one ``final`` event per mark (any old ``final`` rows dropped)."""
    kept = [e for e in events if e.step != "final"]
    out = list(kept)
    for mark, es in by_mark(kept).items():
        value, rule = final_value(es, record)
        out.append(Event(mark, "final", "route", value, rule, "", version))
    return out

def claim_rulings(events: list[Event], place: dict) -> dict:
    """``{(page, district, muni, part, token): (value, rule)}`` — the claims a token question
    or the author ruled on, by the final value of the claim's own mark (``#token#0``).

    ``value`` is the token (printed), :data:`ABSENT` (not printed), or :data:`OPEN` /
    :data:`UNSURE` (not settled: the author's queue). The inventory applies it over every
    mark of the part that holds the token (D-120).
    """
    out: dict = {}
    for mark, es in by_mark(events).items():
        if not (is_claim_mark(mark) and mark.endswith("#0") and mark in place):
            continue
        if not any(e.step in ("opus", "author") for e in es):
            continue
        final = next((e for e in es if e.step == "final"), None)
        if final is not None:
            out[tuple(map(str, place[mark])) + (mark.split("#")[1],)] = (final.read, final.note)
    return out

def running_values(events: list[Event], record: str) -> list[tuple[str, str, str, str]]:
    """``[(step, actor, before, after)]`` for one mark: the value after each deciding step.

    The value starts as the reader of record's reading. A step *decides* when it
    can set the value — Opus (its majority over the repeats), the author, the
    final — while ``order`` and ``dinov3`` only flag, and a flag is recorded as a
    proposal (``after`` is what it named) without moving the value.
    """
    own = [e.read for e in events if e.step == "read" and e.actor == record]
    value = own[0] if own else ABSENT
    out = [("read", record, "", value)]
    opus = [e.read for e in events if e.step == "opus"]
    for e in events:
        if e.step == "order":                      # a flag names no other token
            out.append(("order", e.actor, value, value))
        elif e.step == "dinov3" and e.read:
            out.append(("dinov3", e.actor, value, e.read))
    if opus:
        decisive = [v for v in opus if v not in UNSETTLED]
        if decisive:
            top = majority(decisive) or OPEN
        else:                                      # settled nothing: as final_value decides
            reads = [e.read for e in events if e.step == "read"]
            top = reads[0] if reads and len(set(reads)) == 1 else OPEN
        out.append(("opus", "opus", value, top))
        value = top
    for e in events:
        if e.step == "author":
            out.append(("author", e.actor, value, e.read))
            value = e.read
    final = [e.read for e in events if e.step == "final"]
    if final:
        out.append(("final", "route", value, final[-1]))
    return out

def step_changes(events: list[Event], record: str, gold: dict | None = None) -> dict:
    """Per step: marks it touched, marks whose value it changed, the transitions, and —
    with ``gold`` (``{mark: value}``) — how many of its changes ended right or wrong."""
    out: dict = {s: {"touched": 0, "changed": 0, "right": 0, "wrong": 0,
                     "transitions": Counter()} for s in STEPS}
    for mark, es in by_mark(events).items():
        for step, _actor, before, after in running_values(es, record):
            row = out[step]
            row["touched"] += 1
            if step == "read" or before == after:
                continue
            row["changed"] += 1
            row["transitions"][(before, after)] += 1
            if gold is not None and step not in ("order", "dinov3"):
                truth = gold.get(mark, ABSENT) or ABSENT
                row["right" if after == truth else "wrong"] += 1
    return out

def trail(mark: str, events: list[Event], gold: str = "", place: str = "") -> list[str]:
    """A mark's history, one line a step, the gold beside it for reference."""
    lines = [f"{mark}" + (f"   {place}" if place else "")

"""The history of every mark of the 1910 register through the reading route (D-109).

Every stage of the route writes its own file and none overwrites another (D-38,
D-55), so nothing is lost — but nothing joins the stages per mark either. This
module is that join: an **append-only ledger** of *events*, one row per step's
reading of one mark, from which the final value of every mark is computed, each
mark's trail printed, and each step's changes counted.

A **mark** is a glyph on the page. Where the locator placed it, its id is the
section image and the box (:func:`box_mark`), so every reader's reading of the
same glyph lands on the same mark; where no box was placed, it is the part and
the token (:func:`unplaced_mark`).

The steps, in route order (:data:`STEPS`):

``read``    each reader's reading — a token, or :data:`ABSENT` (it read the
            entry and claimed nothing here)
``order``   the printed-order rule flagging a reader's mark as out of place
``dinov3``  the encoder's name for the box (``note``: agree / disagree / …)
``cz_de``   the other edition's reading of the same part
``opus``    the verify call's answer, one event per repeat
``author``  a ruling by hand; overrides everything before it
``final``   the value :func:`final_value` computes — written, never edited

Nothing in an event is ever changed; a later step adds a row.

A mark's final value is not yet a claim's. A **token question** ("is T printed for this
part?") rules on the claim (part, T) wherever the readers filed it, so its answer sits on
the claim's own mark and :func:`claim_rulings` hands it to the inventory, which applies it
over every mark of that part (1.2.0, D-120).
"""

from __future__ import annotations

__version__ = "1.2.0"

import csv
from collections import Counter, defaultdict
from dataclasses import asdict, dataclass, fields
from pathlib import Path

STEPS = ("read", "order", "dinov3", "cz_de", "opus", "author", "final")
STEP_NO = {step: n for n, step in enumerate(STEPS, 1)}

#: A reader read the entry and claimed nothing at this mark.
ABSENT = "—"
#: The route could not settle the mark: it waits for the author.
OPEN = "open"
UNSURE = "cannot tell"
#: Opus named, for an outlined box, the token the readers mostly read at *another*
#: mark of the same section and part (D-120): the outline sits on the neighbouring
#: glyph, so the answer rules on nothing at this mark and the readers decide it —
#: when they agree; when they do not, the mark is open (the author's queue).
NO_RULING = "no ruling"
#: Opus answered "none of these" for an outlined box: the print is none of the
#: offered marks, or the outline is on no mark. Like :data:`UNSURE`, the route has
#: not settled the mark; the readers' unanimous value stands, otherwise it is open
#: (D-120; D-118 had let the reader of record decide).
NONE_OF_THESE = "none of these"
#: Opus answers that settle nothing at the mark they were given for.
UNSETTLED = (UNSURE, NO_RULING, NONE_OF_THESE)


@dataclass(frozen=True)
class Event:
    """One step's reading of one mark."""
    mark: str
    step: str
    actor: str
    read: str
    note: str = ""
    evidence: str = ""
    version: str = ""

    @property
    def step_no(self) -> int:
        return STEP_NO[self.step]


FIELDS = ["mark", "step_no", "step", "actor", "read", "note", "evidence", "version"]


def box_mark(image: str, box: str) -> str:
    """``0196_nusle_offices_1.png#1041,383,1093,436``."""

    names = {f.name for f in fields(Event)}
    with path.open(encoding="utf-8", newline="") as fh:
        return [Event(**{k: v for k, v in r.items() if k in names}) for r in csv.DictReader(fh)]


def by_mark(events: list[Event]) -> dict[str, list[Event]]:
    """``{mark: events in route order}``; events of one step keep their order."""
    out: dict = defaultdict(list)
    for e in events:
        out[e.mark].append(e)
    return {m: sorted(es, key=lambda e: e.step_no) for m, es in out.items()}


# --------------------------------------------------------------------------- #
# The value of a mark
# --------------------------------------------------------------------------- #

def majority(values: list[str]) -> str | None:
    """The most frequent value, or ``None`` on a tie at the top."""
    if not values:
        return None
    (top, n), *rest = Counter(values).most_common()
    return None if rest and rest[0][1] == n else top


def flagged(events: list[Event]) -> bool:
    """The order rule flagged it, or DINOv3 disagreed with the reading of record."""
    return any(e.step == "order" for e in events) or any(
        e.step == "dinov3" and e.note == "disagree" for e in events)


def final_value(events: list[Event], record: str) -> tuple[str, str]:
    """``(value, rule)`` for one mark, by the route of D-72 as amended.

    The author's last ruling wins. Otherwise Opus's majority over its decisive
    answers. Where Opus was asked and settled nothing (:data:`UNSETTLED`: it cannot
    tell, the print is none of the options, the outline sat on another mark), the
    readers' value stands if every reading agrees, and the mark is ``open`` — the
    author's queue — if they do not (D-120). A mark Opus was not asked about takes
    the readers' value when every reading agrees and nothing flags it; otherwise the
    reader of record's reading (``record``, e.g. ``gemini38#1``) — a mark the route
    should have asked about and did not.
    """
    author = [e.read for e in events if e.step == "author"]
    if author:
        return author[-1], "author"
    reads = [e.read for e in events if e.step == "read"]
    unanimous = bool(reads) and len(set(reads)) == 1
    asked = [e.read for e in events if e.step == "opus"]
    if asked:
        decisive = [v for v in asked if v not in UNSETTLED]
        if decisive:                               # a split among its answers is open
            top = majority(decisive)
            return (top, "opus") if top is not None else (OPEN, "opus could not tell")
        if unanimous:
            return reads[0], "opus settled nothing; readers agree"
        return OPEN, "opus could not tell"
    if unanimous and not flagged(events):
        return reads[0], "agreed"
    own = [e.read for e in events if e.step == "read" and e.actor == record]
    return (own[0] if own else ABSENT), "not asked: reader of record"


def with_final(events: list[Event], record: str, version: str = __version__) -> list[Event]:
    """The ledger plus one ``final`` event per mark (any old ``final`` rows dropped)."""
    kept = [e for e in events if e.step != "final"]
    out = list(kept)
    for mark, es in by_mark(kept).items():
        value, rule = final_value(es, record)
        out.append(Event(mark, "final", "route", value, rule, "", version))
    return out


def is_claim_mark(mark: str) -> bool:
    """A claim's own mark (:func:`unplaced_mark`, ``page/district/muni/seq#token#k``), not a box."""
    head = mark.split("#")[0]
    return "/" in head and not head.endswith(".png")


def claim_rulings(events: list[Event], place: dict) -> dict:
    """``{(page, district, muni, part, token): (value, rule)}`` — the claims a token question
    or the author ruled on, by the final value of the claim's own mark (``#token#0``).

    ``value`` is the token (printed), :data:`ABSENT` (not printed), or :data:`OPEN` /
    :data:`UNSURE` (not settled: the author's queue). The inventory applies it over every
    mark of the part that holds the token (D-120).
    """
    out: dict = {}
    for mark, es in by_mark(events).items():
        if not (is_claim_mark(mark) and mark.endswith("#0") and mark in place):
            continue
        if not any(e.step in ("opus", "author") for e in es):
            continue
        final = next((e for e in es if e.step == "final"), None)
        if final is not None:
            out[tuple(map(str, place[mark])) + (mark.split("#")[1],)] = (final.read, final.note)
    return out

# What each step changed
# --------------------------------------------------------------------------- #

def running_values(events: list[Event], record: str) -> list[tuple[str, str, str, str]]:
    """``[(step, actor, before, after)]`` for one mark: the value after each deciding step.

    The value starts as the reader of record's reading. A step *decides* when it
    can set the value — Opus (its majority over the repeats), the author, the
    final — while ``order`` and ``dinov3`` only flag, and a flag is recorded as a
    proposal (``after`` is what it named) without moving the value.
    """
    own = [e.read for e in events if e.step == "read" and e.actor == record]
    value = own[0] if own else ABSENT
    out = [("read", record, "", value)]
    opus = [e.read for e in events if e.step == "opus"]
    for e in events:
        if e.step == "order":                      # a flag names no other token
            out.append(("order", e.actor, value, value))
        elif e.step == "dinov3" and e.read:
            out.append(("dinov3", e.actor, value, e.read))
    if opus:
        decisive = [v for v in opus if v not in UNSETTLED]
        if decisive:
            top = majority(decisive) or OPEN
        else:                                      # settled nothing: as final_value decides
            reads = [e.read for e in events if e.step == "read"]
            top = reads[0] if reads and len(set(reads)) == 1 else OPEN
        out.append(("opus", "opus", value, top))
        value = top
    for e in events:
        if e.step == "author":
            out.append(("author", e.actor, value, e.read))
            value = e.read
    final = [e.read for e in events if e.step == "final"]
    if final:
        out.append(("final", "route", value, final[-1]))
    return out


def step_changes(events: list[Event], record: str, gold: dict | None = None) -> dict:
    """Per step: marks it touched, marks whose value it changed, the transitions, and —
    with ``gold`` (``{mark: value}``) — how many of its changes ended right or wrong."""
    out: dict = {s: {"touched": 0, "changed": 0, "right": 0, "wrong": 0,
                     "transitions": Counter()} for s in STEPS}
    for mark, es in by_mark(events).items():
        for step, _actor, before, after in running_values(es, record):
            row = out[step]
            row["touched"] += 1
            if step == "read" or before == after:
                continue
            row["changed"] += 1
            row["transitions"][(before, after)] += 1
            if gold is not None and step not in ("order", "dinov3"):
                truth = gold.get(mark, ABSENT) or ABSENT
                row["right" if after == truth else "wrong"] += 1
    return out


def trail(mark: str, events: list[Event], gold: str = "", place: str = "") -> list[str]:
    """A mark's history, one line a step, the gold beside it for reference."""
    lines = [f"{mark}" + (f"   {place}" if place else "")
             + (f"   gold {gold or ABSENT}" if gold is not None else "")]
    for e in events:
        lines.append(f"    {e.step_no} {e.step:7s} {e.actor:16s} {e.read:22s} {e.note}")
    return lines
