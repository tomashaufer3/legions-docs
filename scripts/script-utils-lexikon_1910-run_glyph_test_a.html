How it works

Castle against poorhouse, ruin against solitary house, railway station against stop: outline against solid, the confusions both language-model readers keep making. If an encoder cannot separate them on clean, hand-placed cuts, there is no whole-volume encoder pass and no box detector is built for one.

Each glyph in output/glyph_bank/glyph_bank.csv (build_glyph_bank.py) is embedded and classified by its nearest neighbours among the glyphs of the other pages — a page never sees its own glyphs, so a page's typesetting cannot carry the answer. Reported for every encoder, cut and feature:

  • each pair on its own (query and bank restricted to the two tokens), 1-NN and 3-NN, with the confusions both ways;
  • all seven tokens together.

Encoders: DINOv3 ViT-S/16 and ViT-B/16, and a plain pixel baseline (32 x 32 grey, mean-removed) — the floor an encoder has to beat. DINOv2 small and base can still be named with --encoders; they are not run by default (D-91). Weights are only loaded from the local Hugging Face cache unless --download is given; DINOv3's are gated on Hugging Face under Meta's licence, which has to be accepted on the account whose token the machine uses.

def pixel_values(model_id: str, download: bool):
    """A function ``[PIL image] -> tensor`` that prepares a batch for the encoder.

    The model's own image processor when it loads; otherwise the same steps by
    hand from its ``preprocessor_config.json`` — resize, scale to 0–1,
    normalise. Under transformers 5.x DINOv3's processor is the "fast" one,
    which needs torchvision; without it the processor raises ImportError while
    the weights load fine (Heavyhorse6, 2026-10-19).
    """
    import json

    import numpy as np
    import torch
    from huggingface_hub import hf_hub_download
    from PIL import Image
    from transformers import AutoImageProcessor

    try:
        processor = AutoImageProcessor.from_pretrained(model_id, local_files_only=not download)
        return lambda batch: processor(images=batch, return_tensors="pt")["pixel_values"]
    except Exception as exc:                                    # noqa: BLE001
        print(f"  {model_id}: image processor unavailable ({type(exc).__name__}: "
              f"{str(exc)[:160]}) — preprocessing by hand from its config")
    path = hf_hub_download(model_id, "preprocessor_config.json", local_files_only=not download)
    cfg = json.loads(Path(path).read_text(encoding="utf-8"))
    size = cfg.get("size") or {}
    side = size.get("shortest_edge") or size.get("height") or 224
    height, width = size.get("height", side), size.get("width", side)
    scale = cfg.get("rescale_factor", 1 / 255)
    mean = np.array(cfg.get("image_mean", [0.485, 0.456, 0.406]), dtype=np.float32)
    std = np.array(cfg.get("image_std", [0.229, 0.224, 0.225]), dtype=np.float32)

    def prepare(batch):
        arrays = [(np.asarray(im.convert("RGB").resize((width, height), Image.BICUBIC),
                              dtype=np.float32) * scale - mean) / std for im in batch]
        return torch.from_numpy(np.stack(arrays).transpose(0, 3, 1, 2).copy())
    return prepare

def encoder_features(model_id: str, images, download: bool) -> dict[str, "np.ndarray"] | None:
    """``{"cls": …, "patch": …}``, or ``None`` when the weights are not available."""
    import numpy as np
    import torch
    from transformers import AutoModel

    device = "cuda" if torch.cuda.is_available() else "cpu"
    try:
        model = AutoModel.from_pretrained(model_id, local_files_only=not download).eval().to(device)
        prepare = pixel_values(model_id, download)
    except Exception as exc:                                    # noqa: BLE001
        print(f"  {model_id}: not available ({type(exc).__name__}: {str(exc)[:200]}) — "
              f"{'check the licence and token' if download else 'run with --download'}")
        return None
    registers = getattr(model.config, "num_register_tokens", 0) or 0
    cls, patch = [], []
    with torch.no_grad():
        for start in range(0, len(images), 32):
            batch = prepare(images[start:start + 32]).to(device)
            hidden = model(pixel_values=batch).last_hidden_state.float().cpu()
            cls.append(hidden[:, 0].numpy())
            patch.append(hidden[:, 1 + registers:].mean(dim=1).numpy())
    return {"cls": np.concatenate(cls), "patch": np.concatenate(patch)}

def leave_page_out(x, tokens: list[str], scans: list[str], keep: set[str], k: int) -> list[tuple]:
    """``[(truth, predicted)]`` for every glyph of a kept token, bank from other pages."""
    import numpy as np

    idx = [i for i, t in enumerate(tokens) if t in keep]
    v = x[idx] / np.linalg.norm(x[idx], axis=1, keepdims=True).clip(1e-9)
    sim = v @ v.T
    out = []
    for a, i in enumerate(idx):
        others = [b for b, j in enumerate(idx) if scans[j] != scans[i]]
        if not others:
            continue
        nearest = sorted(others, key=lambda b: -sim[a, b])[:k]
        votes = Counter(tokens[idx[b]] for b in nearest)
        top = max(votes.values())
        # A tie goes to the single nearest neighbour.
        winner = next(tokens[idx[b]] for b in nearest if votes[tokens[idx[b]]] == top)
        out.append((tokens[i], winner))
    return out

def evaluate(name: str, cut: str, feature: str, x, tokens, scans) -> list[dict]:
    rows = []
    for k in (1, 3):
        for pair in [*PAIRS, tuple(sorted(set(tokens)))]:
            results = leave_page_out(x, tokens, scans, set(pair), k)
            if not results:
                continue
            right = sum(t == p for t, p in results)
            confusions = Counter((t, p) for t, p in results if t != p)
            rows.append({"encoder": name, "cut": cut, "feature": feature, "k": k,
                         "task": " / ".join(pair) if len(pair) == 2 else f"all {len(pair)}",
                         "n": len(results), "accuracy": right / len(results),
                         "confusions": "; ".join(f"{t}->{p} {n}" for (t, p), n
                                                 in confusions.most_common())})
    return rows

        writer.writerows(rows)
    lines = [f"{'encoder':14s} {'cut':6s} {'feat':6s} k  {'task':28s} {'n':>4s} {'acc':>6s}  confusions"]
    for r in rows:
        lines.append(f"{r['encoder']:14s} {r['cut']:6s} {r['feature']:6s} {r['k']}  "
                     f"{r['task'][:28]:28s} {r['n']:4d} {r['accuracy']:6.1%}  {r['confusions']}")
    text = "\n".join(lines)
    print(text)
    out.with_suffix(".txt").write_text(text + "\n", encoding="utf-8")
    print(f"\n  -> {out}")
    return 0
