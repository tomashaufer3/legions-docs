How it works

D-58 makes the German Spezialortsrepertorium von Böhmen (Wien 1915) a second measurement of the register's institutions once its sections can be cut from the image, and D-67 gives it its own gold, matched to the Czech by court and municipality name. This runner reads the German gold pages with the two readers the Czech r5 arms use, each three times:

gemini38   Gemini 3.8 Flash, thinking off       gemini38_r5atlasntgde{1,2,3}
haiku55h   Claude Haiku 5.5, thinking at high   haiku55h_r5atlasgde{1,2,3}

Everything a Czech r5 gold pass holds fixed is held fixed here — one section per call, both apparatus documents and the postal legend, strips cut through blank rows, upscaled, the volume's own symbol key sent with every crop, temperature 0 where the reader takes one — and three things are the German volume's own:

  • the cut. The UBW scan has no text layer to cut on, so the pilot's channel G cannot run. The page is rendered at its native resolution, the geometry is lexikon_1910_image's, and every heading is cut out and read by the pass's reader, then labelled by the German profile's matcher (lexikon_1910_de_notes.cut_page). A section open at the top of a page whose previous scan is not in the run takes its label from a read of that scan's last heading. Strips are 500 px × the page's scale (~375 px), the same amount of print as the Czech 500.
  • what the reader is told. The Czech prompt with the German volume's words, and the German plate's 80 symbols — as the token list (the de rows of institution_abbreviations.csv) and as the key image (symbols/de/atlas_prompt.png).
  • the column header, read once a page and checked against the profile's seventeen labels (D-80, D-83: the German reader must report it).

A German heading names a German court; its entries are filed under the Czech court's slug through the court crosswalk, so they share a key with the German gold sheets and with the Czech edition.

Passes go to bakeoff/&lt;host&gt;/de/ — out of sight of every Czech table — and are scored by score_r5_german_notes.py. Certification sheets are never read.

The frozen route (D-115) calls frozen_pass instead: any pages of any of the three German volumes, into a folder of its own, with the route's Haiku (haiku55nh: the note for a reader that thinks, and the contrast sheet after the key), the reader of record saving its section crops (section_images.py, scaled so the print is the size E3 learned it at) and, with a queue, every uncached crop written as a batch line (reader_batches.py) instead of being sent.

Run it on Heavyhorse6 (D-68):

python utils/lexikon_1910/run_r5_german_notes.py --dry-run   # the plan and a cost estimate; spends nothing
python utils/lexikon_1910/run_r5_german_notes.py             # run what is missing, then score
python utils/lexikon_1910/run_r5_german_notes.py --score     # score only; spends nothing

def cache_key(crop, caption: str, reader, extra: str = "") -> str:
    """The pilot's key — crop bytes, caption, model, prompt, temperature — plus
    the thinking settings, so two settings can never share an answer."""
    h = hashlib.sha256()
    for part in (pilot.encode_png(crop), caption, str(getattr(reader, "model", "")),
                 str(getattr(reader, "system", "")),
                 str(getattr(reader, "temperature", "")),
                 str(getattr(reader, "reasoning_effort", "") or ""),
                 str(getattr(reader, "thinking", "")), str(getattr(reader, "effort", "")),
                 extra):
        h.update(part.encode("utf-8"))
        h.update(bytes(1))
    return h.hexdigest()

def cached_line_reader(reader, cache: Path, caption: str, log: list, usage: Counter):
    """``read(crop, where) -> lines``: one uncached-or-cached read of a small crop.

    For headings and the header. A failure returns ``[]`` and is logged; it is
    not cached, so a rerun asks again.
    """
    cache.mkdir(parents=True, exist_ok=True)

    def read(crop, where: str = "") -> list[str]:
        hit = cache / f"{cache_key(crop, caption, reader)}.json"
        if hit.exists():
            record = json.loads(hit.read_text(encoding="utf-8"))
            log.append({"where": where, "lines": record["lines"], "cached": True})
            return record["lines"]
        try:
            lines, spent = reader([(crop, caption)], "Transcribe this image.")
        except pilot.RateLimited:
            raise
        except Exception as exc:                                # noqa: BLE001
            log.append({"where": where, "lines": [], "error": str(exc)[:200]})
            return []
        lines = [str(line) for line in (lines or [])]
        for key, value in spent.items():
            usage[key] += value or 0
        hit.write_text(json.dumps({"lines": lines, "usage": spent}), encoding="utf-8")
        log.append({"where": where, "lines": lines, "cached": False})
        return lines

    return read

def carry_into(scan: int, profile, read_heading, back: int = 2):
    """``(kind, court)`` of the section open at the top of ``scan``, read off the
    last heading of the nearest earlier scan that has one, or ``None``."""
    for previous in range(scan - 1, max(profile.table_first, scan - back) - 1, -1):
        crop = lxdn.last_heading_crop(lxdn.render_native(profile.source_pdf, previous))
        if crop is None:
            continue
        found = profile.find_heading(read_heading(crop, f"{previous:04d} carry"))
        return (found[0], found[1]) if found else None
    return None

def cut_pages(pages: list[int], profile, read_heading, sections: Path | None = None,
              crosswalk: dict | None = None) -> list[PageCut]:
    """Cut every page in order, carrying the open section from one to the next.
    With ``sections``, also save every section the reader reads (:func:`save_sections`)."""
    from utils.lexikon_1910 import lexikon_1910_image as lximg

    cuts, carry, previous = [], None, None
    for scan in pages:
        stem = f"{scan:04d}"
        image = lxdn.render_native(profile.source_pdf, scan)
        geo = lximg.detect(image, expected_columns=profile.n_columns, full_width_headings=True)
        if previous != scan - 1:
            carry = carry_into(scan, profile, read_heading)
        segments, report = lxdn.cut_page(
            image, stem, profile, carry=carry,
            read_heading=lambda crop, s=stem: read_heading(crop, f"{s} heading"))
        carry = lxnt.trailing_from_segments(
            [s for s in segments if s.kind != "legend"], carry)
        strip = lxdn.strip_height(bo.MAX_HEIGHT, geo.scale)
        crops = lxnt.crop_segments(image, segments, None, upscale=True,
                                   max_height=strip, ink=report.ink)
        if sections is not None:
            save_sections(image, segments, geo.scale, crosswalk or {}, sections)
        printed = ({(s.page, s.region, s.order):
                    pilot.printed_line_count(report.ink, s.box, image.width)
                    for s in segments} if report.ink is not None else {})
        cuts.append(PageCut(stem, image.width, geo.scale, strip, segments, report,
                            crops, lxdn.header_crop(image, geo), printed))
        previous = scan
    return cuts

def read_crops(jobs: list[tuple], readers: dict, atlas, cache_dir: Path,
               workers: int = WORKERS, contrast=None, contrast_digest: str = "",
               queue: Path | None = None) -> tuple[list, dict]:
    """Every crop read once, cached per label, in printed order.

    The pilot's ``read_crop`` rules: a cache hit costs nothing and is served
    even after a quota refusal; a truncated reply doubles the cap and tries
    again; a refusal parks the rest of the pass, which is then marked
    incomplete rather than scored.

    ``contrast`` (an image) is sent after the key, as the pilot's
    ``--notes-contrast`` sends it, and enters the cache key by its digest. With
    ``queue`` a crop not in the cache is written there as a batch line
    (``reader_batches``) and not sent.
    """
    cache_dir.mkdir(parents=True, exist_ok=True)
    extra = "atlas" + (f"+contrast:{contrast_digest}" if contrast is not None else "")
    parked = {"at": None, "error": "", "retry_after": None, "status": None,
              "done_before": 0, "skipped": 0, "from_cache": 0}
    lock = threading.Lock()

    def read_one(job):
        slot, crop, caption, kind = job
        reader = readers[kind]
        hit = cache_dir / f"{cache_key(crop, caption, reader, extra)}.json"
        if hit.exists():
            record = json.loads(hit.read_text(encoding="utf-8"))
            with lock:
                parked["from_cache"] += 1
            return slot, record["lines"], record["usage"]
        with lock:
            if parked["at"] is not None:
                parked["skipped"] += 1
                return slot, None, {"error": "parked"}
        payload = ([(atlas, ATLAS_CAPTION)]
                   + ([(contrast, CONTRAST_CAPTION)] if contrast is not None else [])
                   + [(crop, caption)])
        if queue is not None:
            make = getattr(reader, "params", None) or getattr(reader, "body")
            line = json.dumps({"key": hit.stem, "reader": reader.name,
                               "cache_dir": str(cache_dir), "slot": list(slot),
                               "kind": kind, "model": getattr(reader, "model", ""),
                               "request": make(payload, "Transcribe this image.")})
            with lock:
                with Path(queue).open("a", encoding="utf-8") as fh:
                    fh.write(line + "\n")
                parked["queued"] = parked.get("queued", 0) + 1
            return slot, None, {"error": "batched: queued"}
        for attempt in range(3):
            try:
                lines, usage = reader(payload, "Transcribe this image.")
                tmp = hit.with_suffix(".json.part")
                tmp.write_text(json.dumps({"lines": lines, "usage": usage,
                                           "slot": list(slot), "kind": kind,
                                           "at": time.time()}), encoding="utf-8")
                tmp.replace(hit)
                with lock:
                    parked["done_before"] += 1
                return slot, lines, usage
            except pilot.Truncated:
                reader.max_tokens = min(reader.max_tokens * 2, 32000)
            except pilot.RateLimited as exc:
                with lock:
                    if parked["at"] is None:
                        parked.update(at=time.time(), error=str(exc)[:300],
                                      retry_after=exc.retry_after, status=exc.status)
                    parked["skipped"] += 1
                return slot, None, {"error": f"rate limited: {exc}"}
            except Exception as exc:                            # noqa: BLE001
                if attempt == 2:
                    return slot, None, {"error": str(exc)[:300]}
                time.sleep(2 * (attempt + 1))
        return slot, None, {"error": "truncated"}

    arrived = []
    with ThreadPoolExecutor(max_workers=workers) as pool:
        futures = {pool.submit(read_one, job): k for k, job in enumerate(jobs)}
        for done, future in enumerate(as_completed(futures), 1):
            arrived.append((futures[future], future.result()))
            if done % pilot.PROGRESS_EVERY == 0 or done == len(futures):
                print(f"    {done}/{len(futures)} crops", flush=True)
    return [result for _k, result in sorted(arrived, key=lambda kv: kv[0])], parked

def clean_lines(results: list, allowed: set) -> tuple[dict, dict]:
    """``(lines by slot, counts)`` after the pilot's repeat filter and token guard."""
    by_slot: dict = {}
    counts = Counter()
    for slot, lines, usage in results:
        for key in ("input", "output", "thinking", "cache_write", "cache_read"):
            counts[key] += (usage or {}).get(key, 0) or 0
        if lines is None:
            counts["failed"] += 1
            continue
        if not lines:
            counts["empty"] += 1
            continue
        deduped, repeats = lxv.drop_repeats([str(line) for line in lines])
        clean, rejected = lxv.guard_tokens(deduped, allowed)
        counts["repeats"] += repeats
        counts["rejected"] += sum(rejected.values())
        by_slot.setdefault(slot, []).extend(clean)
    return by_slot, counts

def cache_key(crop, caption: str, reader, extra: str = "") -> str:
    """The pilot's key — crop bytes, caption, model, prompt, temperature — plus
    the thinking settings, so two settings can never share an answer."""
    h = hashlib.sha256()
    for part in (pilot.encode_png(crop), caption, str(getattr(reader, "model", "")),
                 str(getattr(reader, "system", "")),
                 str(getattr(reader, "temperature", "")),
                 str(getattr(reader, "reasoning_effort", "") or ""),
                 str(getattr(reader, "thinking", "")), str(getattr(reader, "effort", "")),
                 extra):
        h.update(part.encode("utf-8"))
        h.update(bytes(1))
    return h.hexdigest()


def cached_line_reader(reader, cache: Path, caption: str, log: list, usage: Counter):
    """``read(crop, where) -> lines``: one uncached-or-cached read of a small crop.

    For headings and the header. A failure returns ``[]`` and is logged; it is
    not cached, so a rerun asks again.
    """
    cache.mkdir(parents=True, exist_ok=True)

    def read(crop, where: str = "") -> list[str]:
        hit = cache / f"{cache_key(crop, caption, reader)}.json"
        if hit.exists():
            record = json.loads(hit.read_text(encoding="utf-8"))
            log.append({"where": where, "lines": record["lines"], "cached": True})
            return record["lines"]
        try:
            lines, spent = reader([(crop, caption)], "Transcribe this image.")
        except pilot.RateLimited:
            raise
        except Exception as exc:                                # noqa: BLE001
            log.append({"where": where, "lines": [], "error": str(exc)[:200]})
            return []
        lines = [str(line) for line in (lines or [])]
        for key, value in spent.items():
            usage[key] += value or 0
        hit.write_text(json.dumps({"lines": lines, "usage": spent}), encoding="utf-8")
        log.append({"where": where, "lines": lines, "cached": False})
        return lines

    return read

    """``(kind, court)`` of the section open at the top of ``scan``, read off the
    last heading of the nearest earlier scan that has one, or ``None``."""
    for previous in range(scan - 1, max(profile.table_first, scan - back) - 1, -1):
        crop = lxdn.last_heading_crop(lxdn.render_native(profile.source_pdf, previous))
        if crop is None:
            continue
        found = profile.find_heading(read_heading(crop, f"{previous:04d} carry"))
        return (found[0], found[1]) if found else None
    return None


def save_sections(image, segments, scale: float, crosswalk: dict, out: Path) -> None:
    """The reader's sections as images, under the Czech court's slug (``section_images``).

    The German scan is rendered at its native resolution, ``scale`` of the Czech
    render; enlarged by ``2 / scale`` its print is the size the locator and the
    bank learned it at.
    """
    import copy

    import section_images as si

    relabelled = []
    for segment in segments:
        czech, _how = lxdn.court_name(segment.district, crosswalk)
        named = copy.copy(segment)
        named.district = czech or segment.district
        relabelled.append(named)
    si.save(image, relabelled, out, enlarge=si.ENLARGE / (scale or 1.0))


def cut_pages(pages: list[int], profile, read_heading, sections: Path | None = None,
              crosswalk: dict | None = None) -> list[PageCut]:
    """Cut every page in order, carrying the open section from one to the next.
    With ``sections``, also save every section the reader reads (:func:`save_sections`)."""
    from utils.lexikon_1910 import lexikon_1910_image as lximg

    cuts, carry, previous = [], None, None
    for scan in pages:
        stem = f"{scan:04d}"
        image = lxdn.render_native(profile.source_pdf, scan)
        geo = lximg.detect(image, expected_columns=profile.n_columns, full_width_headings=True)
        if previous != scan - 1:
            carry = carry_into(scan, profile, read_heading)
        segments, report = lxdn.cut_page(
            image, stem, profile, carry=carry,
            read_heading=lambda crop, s=stem: read_heading(crop, f"{s} heading"))
        carry = lxnt.trailing_from_segments(
            [s for s in segments if s.kind != "legend"], carry)
        strip = lxdn.strip_height(bo.MAX_HEIGHT, geo.scale)
        crops = lxnt.crop_segments(image, segments, None, upscale=True,
                                   max_height=strip, ink=report.ink)
        if sections is not None:
            save_sections(image, segments, geo.scale, crosswalk or {}, sections)
        printed = ({(s.page, s.region, s.order):
                    pilot.printed_line_count(report.ink, s.box, image.width)
                    for s in segments} if report.ink is not None else {})
        cuts.append(PageCut(stem, image.width, geo.scale, strip, segments, report,
                            crops, lxdn.header_crop(image, geo), printed))
        previous = scan
    return cuts


def jobs_of(cuts: list[PageCut]) -> list[tuple]:
    """``(slot, crop, caption, kind)`` in printed order, the pilot's captions."""
    jobs = []
    for cut in cuts:
        for crop, segment, i, n in cut.crops:
            caption = ("This image is the postal and railway legend printed under "
                       "the footnote apparatus." if segment.kind == "legend" else
                       "This image is one section of the footnote apparatus."
                       if n == 1 else
                       f"This image is part {i + 1} of {n} of one section of the "
                       f"footnote apparatus, in printed order.")
            jobs.append(((segment.page, segment.region, segment.order), crop,
                         caption, segment.kind))
    return jobs

def read_crops(jobs: list[tuple], readers: dict, atlas, cache_dir: Path,
               workers: int = WORKERS, contrast=None, contrast_digest: str = "",
               queue: Path | None = None) -> tuple[list, dict]:
    """Every crop read once, cached per label, in printed order.

    The pilot's ``read_crop`` rules: a cache hit costs nothing and is served
    even after a quota refusal; a truncated reply doubles the cap and tries
    again; a refusal parks the rest of the pass, which is then marked
    incomplete rather than scored.

    ``contrast`` (an image) is sent after the key, as the pilot's
    ``--notes-contrast`` sends it, and enters the cache key by its digest. With
    ``queue`` a crop not in the cache is written there as a batch line
    (``reader_batches``) and not sent.
    """
    cache_dir.mkdir(parents=True, exist_ok=True)
    extra = "atlas" + (f"+contrast:{contrast_digest}" if contrast is not None else "")
    parked = {"at": None, "error": "", "retry_after": None, "status": None,
              "done_before": 0, "skipped": 0, "from_cache": 0}
    lock = threading.Lock()

    def read_one(job):
        slot, crop, caption, kind = job
        reader = readers[kind]
        hit = cache_dir / f"{cache_key(crop, caption, reader, extra)}.json"
        if hit.exists():
            record = json.loads(hit.read_text(encoding="utf-8"))
            with lock:
                parked["from_cache"] += 1
            return slot, record["lines"], record["usage"]
        with lock:
            if parked["at"] is not None:
                parked["skipped"] += 1
                return slot, None, {"error": "parked"}
        payload = ([(atlas, ATLAS_CAPTION)]
                   + ([(contrast, CONTRAST_CAPTION)] if contrast is not None else [])
                   + [(crop, caption)])
        if queue is not None:
            make = getattr(reader, "params", None) or getattr(reader, "body")
            line = json.dumps({"key": hit.stem, "reader": reader.name,
                               "cache_dir": str(cache_dir), "slot": list(slot),
                               "kind": kind, "model": getattr(reader, "model", ""),
                               "request": make(payload, "Transcribe this image.")})
            with lock:
                with Path(queue).open("a", encoding="utf-8") as fh:
                    fh.write(line + "\n")
                parked["queued"] = parked.get("queued", 0) + 1
            return slot, None, {"error": "batched: queued"}
        for attempt in range(3):
            try:
                lines, usage = reader(payload, "Transcribe this image.")
                tmp = hit.with_suffix(".json.part")
                tmp.write_text(json.dumps({"lines": lines, "usage": usage,
                                           "slot": list(slot), "kind": kind,
                                           "at": time.time()}), encoding="utf-8")
                tmp.replace(hit)
                with lock:
                    parked["done_before"] += 1
                return slot, lines, usage
            except pilot.Truncated:
                reader.max_tokens = min(reader.max_tokens * 2, 32000)
            except pilot.RateLimited as exc:
                with lock:
                    if parked["at"] is None:
                        parked.update(at=time.time(), error=str(exc)[:300],
                                      retry_after=exc.retry_after, status=exc.status)
                    parked["skipped"] += 1
                return slot, None, {"error": f"rate limited: {exc}"}
            except Exception as exc:                            # noqa: BLE001
                if attempt == 2:
                    return slot, None, {"error": str(exc)[:300]}
                time.sleep(2 * (attempt + 1))
        return slot, None, {"error": "truncated"}

    arrived = []
    with ThreadPoolExecutor(max_workers=workers) as pool:
        futures = {pool.submit(read_one, job): k for k, job in enumerate(jobs)}
        for done, future in enumerate(as_completed(futures), 1):
            arrived.append((futures[future], future.result()))
            if done % pilot.PROGRESS_EVERY == 0 or done == len(futures):
                print(f"    {done}/{len(futures)} crops", flush=True)
    return [result for _k, result in sorted(arrived, key=lambda kv: kv[0])], parked

def clean_lines(results: list, allowed: set) -> tuple[dict, dict]:
    """``(lines by slot, counts)`` after the pilot's repeat filter and token guard."""
    by_slot: dict = {}
    counts = Counter()
    for slot, lines, usage in results:
        for key in ("input", "output", "thinking", "cache_write", "cache_read"):
            counts[key] += (usage or {}).get(key, 0) or 0
        if lines is None:
            counts["failed"] += 1
            continue
        if not lines:
            counts["empty"] += 1
            continue
        deduped, repeats = lxv.drop_repeats([str(line) for line in lines])
        clean, rejected = lxv.guard_tokens(deduped, allowed)
        counts["repeats"] += repeats
        counts["rejected"] += sum(rejected.values())
        by_slot.setdefault(slot, []).extend(clean)
    return by_slot, counts


def legend_rows(by_slot: dict, segments: dict, crosswalk: dict) -> list[dict]:
    """The postal legend as read, keyed by the Czech court slug."""
    rows = []
    for slot, lines in sorted(by_slot.items()):
        if segments[slot].kind != "legend":
            continue
        notes = lxnt.Notes(page=slot[0], top=0, bottom=0, split_x=0,
                           footer_lines=list(lines))
        for court, entries in lxnt.postal_legend([notes]).items():
            czech, how = lxdn.court_name(court, crosswalk)
            for key, entry in sorted(entries.items()):
                rows.append({"page": slot[0], "district": lxnt.normalise(czech or court),
                             "court_de": court, "court_how": how, "key": key,
                             "place": entry.place, "rail_symbol": entry.rail_symbol})
    return rows


def audit_rows(by_slot: dict, segments: dict, printed: dict, profile) -> list[dict]:
    """The pilot's crop audit (D-85) with the German matcher."""
    rows = []
    for slot, segment in sorted(segments.items(), key=lambda kv: (kv[0][0], kv[0][2])):
        lines = by_slot.get(slot, [])
        seen = [h for h in (profile.find_heading(str(line)) for line in lines) if h]
        agree = [h for h in seen if h[0] == segment.kind
                 and lxdn.fold(h[1]) == lxdn.fold(segment.district)]
        n = printed.get(slot)
        share = (len(lines) / n) if n else None
        rows.append({"page": slot[0], "region": slot[1], "order": slot[2],
                     "kind": segment.kind, "court_de": segment.district,
                     "carried": int(segment.carried), "printed_lines": n or "",
                     "returned_lines": len(lines),
                     "line_flag": int(share is not None and (
                         share < pilot.LINE_SHARE_FLAG or share > 1 / pilot.LINE_SHARE_FLAG)),
                     "headings_in_reply": " | ".join(f"{k}:{d}" for k, d, _e in seen),
                     "heading_flag": int(segment.kind in ("parts", "offices")
                                         and bool(seen) and not agree)})
    return rows


def assemble(name: str, cuts: list[PageCut], results: list, abbreviations,
             crosswalk: dict, profile, out: Path | None = None) -> dict:
    """Decode a pass's lines and write its files; return its counts."""
    out = out or bo.DE_PASS_DIR
    segments = {(s.page, s.region, s.order): s for c in cuts for s in c.segments}
    printed = {k: v for c in cuts for k, v in c.printed.items()}
    by_slot, counts = clean_lines(results, lxv.vocabulary(abbreviations))
    read = [(segments[slot], lines) for slot, lines in
            sorted(by_slot.items(), key=lambda kv: (kv[0][0], kv[0][2]))
            if segments[slot].kind != "legend"]
    sections = lxv.sections_from_segments(read, find=profile.find_heading)
    courts = lxdn.relabel_sections(sections, crosswalk)
    records, entries = lxin.decode_sections(sections, abbreviations)
    inventory = lxin.build_inventory(records, [])
    for row in inventory:
        row["channel"] = f"{name}>image"
    pilot.write_csv(out / f"inventory_{name}.csv", inventory)
    # The decoded records as the pilot's ``institutions_<label>.csv`` holds them:
    # what E3 and the order rule read a pass's marks from.
    for record in records:
        record["channel"] = f"{name}:institutions"
    pilot.write_csv(out / f"institutions_{name}.csv", records)
    pilot.write_csv(out / f"parts_{name}.csv", lxnt.parts_records(sections, source=name))
    pilot.write_csv(out / f"attributes_{name}.csv",
                    lxat.attribute_records(lxat.entry_parts(entries), source=name))
    pilot.write_csv(out / f"notes_{name}.csv",
                    [{"page": s[0], "region": s[1], "order": s[2], "kind": segments[s].kind,
                      "court_de": segments[s].district, "line_no": i, "text": line}
                     for s, lines in sorted(by_slot.items()) for i, line in enumerate(lines)])
    pilot.write_csv(out / f"legend_{name}.csv", legend_rows(by_slot, segments, crosswalk))
    pilot.write_csv(out / f"crop_audit_{name}.csv",
                    audit_rows(by_slot, segments, printed, profile))
    pilot.write_csv(out / f"courts_{name}.csv", courts)
    counts["inventory_rows"] = len(inventory)
    counts["courts_unmapped"] = sum(1 for c in courts if not c["court_cs"])
    return dict(counts)

def run_pass(job: dict, pages: list[int], profile, abbreviations, crosswalk,
             out: Path | None = None, sections: Path | None = None,
             queue: Path | None = None, contrast: Path | None = None) -> bool:
    """Read the pages once with one reader. False if a quota stopped it.

    The r5 arm calls it with the defaults. The frozen route (:func:`frozen_pass`)
    gives a folder, a section folder for the reader of record, a batch queue
    (the pass then only queues: nothing is decoded or written but the queue) and
    the contrast sheet for Haiku.
    """
    arm, name = job["arm"], job["label"]
    out = out or bo.DE_PASS_DIR
    out.mkdir(parents=True, exist_ok=True)
    prompts = systems(abbreviations, think=arm.think, contrast=contrast is not None,
                      volume=profile.slug)
    readers = {kind: make_reader(arm, system) for kind, system in prompts.items()}
    for kind in ("offices", "parts", "legend"):
        # the key (and the contrast sheet) lead every crop request
        readers[kind].shared_parts = 2 if contrast is not None else 1
    contrast_image, contrast_digest = None, ""
    if contrast is not None:
        from PIL import Image
        contrast_image = Image.open(contrast).convert("RGB")
        contrast_digest = hashlib.sha256(Path(contrast).read_bytes()).hexdigest()[:16]
    cache = out / "crop_cache" / name
    side_usage, heading_log, header_log = Counter(), [], []
    heading = cached_line_reader(readers["heading"], cache / "headings",
                                 "This image is one line of the footnote apparatus, "
                                 "a section heading.", heading_log, side_usage)
    header = cached_line_reader(readers["header"], cache / "header",
                                "This image is the column header of the table.",
                                header_log, side_usage)
    started = time.time()
    try:
        cuts = cut_pages(pages, profile, lambda crop, where: " ".join(heading(crop, where)),
                         sections=sections, crosswalk=crosswalk)
        header_rows = header_rows_of(cuts, header, profile)
    except pilot.RateLimited as exc:
        return park(name, arm, {"at": time.time(), "error": str(exc), "status": exc.status,
                                "retry_after": exc.retry_after, "done_before": 0,
                                "skipped": 0}, 0, started, out)
    jobs = jobs_of(cuts)
    print(f"  {len(cuts)} pages, {sum(len(c.segments) for c in cuts)} sections, "
          f"{len(jobs)} crops; {len(heading_log)} headings read")
    from PIL import Image
    atlas = Image.open(bo.de_atlas(profile.slug)).convert("RGB")       # its own plate's (D-146)
    results, parked = read_crops(jobs, readers, atlas, cache, contrast=contrast_image,
                                 contrast_digest=contrast_digest, queue=queue)
    if queue is not None:
        print(f"  {parked.get('queued', 0)} crops queued to {Path(queue).name}, "
              f"{parked['from_cache']} already in the cache")
        return True
    counts = assemble(name, cuts, results, abbreviations, crosswalk, profile, out)
    pilot.write_csv(out / f"headings_{name}.csv", heading_rows(heading_log, profile, crosswalk))
    pilot.write_csv(out / f"header_{name}.csv", header_rows)
    for key, value in side_usage.items():
        counts[key] = counts.get(key, 0) + value
    record_timing(name, arm, pages, counts, parked, started, len(jobs), out)
    report_pass(name, arm, counts, cuts, header_rows, parked)
    if parked["at"] is not None:
        return park(name, arm, parked, len(jobs), started, out)
    return True


def frozen_pass(label: str, arm_name: str, pages: list[int], profile_name: str,
                out: Path, sections: Path | None = None, queue: Path | None = None,
                contrast: Path | None = None) -> bool:
    """One pass of the frozen route over any pages of one German volume (D-115).

    Unlike :func:`run`, no page is refused here: the runner that calls it
    (``run_frozen.py``) chooses the pages and keeps certification apart.
    """
    profile = lxvol.load_profile(profile_name)
    abbreviations = lxin.load_abbreviations(profile.abbreviations, lang="de", volume=profile.slug)
    crosswalk = lxdn.load_court_crosswalk()
    job = {"arm": ARMS[arm_name], "label": label, "repeat": 0}
    print("\n" + "=" * 78 + f"\nGERMAN NOTES (frozen)  {label}   ({ARMS[arm_name].model}, "
          f"{profile.title}, {len(pages)} pages)\n" + "=" * 78)
    return run_pass(job, pages, profile, abbreviations, crosswalk, out=out,
                    sections=sections, queue=queue, contrast=contrast)


def header_rows_of(cuts: list[PageCut], header, profile) -> list[dict]:
    rows = []
    for cut in cuts:
        lines = header(cut.header, f"{cut.stem} header") if cut.header is not None else []
        check = lxdn.header_check(lines, profile)
        rows.append({"page": cut.stem, "read": int(bool(lines)),
                     "columns_found": check["found"], "columns": check["columns"],
                     "missing": " ".join(check["missing"]),
                     "text": " | ".join(lines)[:500]})
    return rows


def heading_rows(log: list, profile, crosswalk) -> list[dict]:
    rows = []
    for entry in log:
        text = " ".join(entry.get("lines", []))
        found = profile.find_heading(text)
        czech, how = lxdn.court_name(found[1], crosswalk) if found else ("", "")
        rows.append({"where": entry["where"], "text": text[:200],
                     "kind": found[0] if found else "", "court_de": found[1] if found else "",
                     "court_cs": czech, "court_how": how,
                     "error": entry.get("error", ""), "cached": entry.get("cached", "")})
    return rows


def cost(counts: dict, price) -> float:
    billed_in = (counts.get("input", 0)
                 + pilot.CACHE_WRITE_RATE * counts.get("cache_write", 0)
                 + pilot.CACHE_READ_RATE * counts.get("cache_read", 0))
    return (billed_in * price[0] + counts.get("output", 0) * price[1]) / 1e6


def record_timing(name, arm, pages, counts, parked, started, n_crops, out=None) -> None:
    path = (out or bo.DE_PASS_DIR) / "timings.json"
    timings = json.loads(path.read_text(encoding="utf-8")) if path.exists() else {}
    timings[name] = {"model": arm.model, "route": arm.route, "variant": arm.variant,
                     "pages": ",".join(map(str, pages)), "crops": n_crops,
                     "seconds": round(time.time() - started, 1), "tokens": counts,
                     "usd": round(cost(counts, arm.price), 4), "run_tag": bo.RUN_TAG,
                     "host": bo.HOST, "runner": __version__,
                     "incomplete": parked["at"] is not None,
                     "finished": time.strftime("%Y-%m-%d %H:%M:%S")}
    path.write_text(json.dumps(timings, indent=2), encoding="utf-8")


def report_pass(name, arm, counts, cuts, header_rows, parked) -> None:
    unread = sum(len(c.report.headings_image_unread) for c in cuts)
    full = sum(1 for r in header_rows if r["columns_found"] == r["columns"])
    print(f"\n  [{name}] {counts.get('inventory_rows', 0)} inventory rows; "
          f"{counts.get('failed', 0)} crops failed, {counts.get('empty', 0)} empty, "
          f"{counts.get('rejected', 0)} tokens off the list; {unread} headings found "
          f"and not labelled; {counts.get('courts_unmapped', 0)} sections under a "
          f"court the crosswalk does not know; header: every column label "
          f"found on {full} of {len(header_rows)} pages; "
          f"${cost(counts, arm.price):.3f}"
          + (" — INCOMPLETE, rate limited" if parked["at"] is not None else ""))


def park(name, arm, parked, n_crops, started, out=None) -> bool:
    pilot.write_json((out or bo.DE_PASS_DIR) / f"ratelimit_{name}.json", {
        "label": name, "model": arm.model, "crops_total": n_crops,
        "crops_done_before_limit": parked.get("done_before", 0),
        "crops_skipped": parked.get("skipped", 0),
        "seconds_before_limit": round((parked.get("at") or time.time()) - started, 1),
        "status": parked.get("status"), "error": parked.get("error", ""),
        "retry_after_seconds": parked.get("retry_after"),
        "limit_at": time.strftime("%Y-%m-%d %H:%M:%S")})
    print(f"  !! {name} was refused by the quota — marked incomplete; rerun to resume")
    return False

def image_tokens(width: int, height: int, route: str) -> int:
    """Tokens one image costs: Claude resizes to a 1,568 px long edge and bills
    ``w·h/750``; the Gemini figure is the pilot's own ``w·h/750`` capped."""
    if route == "claude":
        shrink = min(1.0, CLAUDE_LONG_EDGE / max(width, height, 1))
        return int(width * shrink) * int(height * shrink) // 750
    return min(GEMINI_IMAGE_CAP, width * height // 750)


def estimate(arm: Arm, crops: list[tuple[int, int]], headings: list[tuple[int, int]],
             headers: list[tuple[int, int]], system_chars: dict, atlas: tuple[int, int]) -> dict:
    """USD for one pass of one reader, from the crops the geometry cut.

    Input: the image, the key, the prompt (``chars / 4``) and ~60 tokens of
    caption and instruction. On Claude the prompt and the key are cached after
    the first request of each kind, so they bill at the cache-read rate.
    Output: :data:`EST_OUTPUT`, plus the pilot's placeholder thinking budget
    where the reader thinks (``THINKING_TOKENS``) — a guess until the first
    pass measures it.
    """
    thinking = pilot.THINKING_TOKENS.get(arm.effort, 0) if arm.thinking == "adaptive" else 0
    key = image_tokens(*atlas, arm.route)
    cached = arm.route == "claude"
    tokens_in = tokens_out = 0.0
    for kind, sizes, prefix in (("crop", crops, key + system_chars["offices"] // 4),
                                ("heading", headings, system_chars["heading"] // 4),
                                ("header", headers, system_chars["header"] // 4)):
        for i, (w, h) in enumerate(sizes):
            fixed = prefix * (pilot.CACHE_READ_RATE if cached and i else 1.0)
            tokens_in += image_tokens(w, h, arm.route) + fixed + 60
            tokens_out += EST_OUTPUT[kind] + thinking
    return {"crops": len(crops), "headings": len(headings), "headers": len(headers),
            "tokens_in": int(tokens_in), "tokens_out": int(tokens_out),
            "usd": (tokens_in * arm.price[0] + tokens_out * arm.price[1]) / 1e6}


def dry_run(names, repeats: int, pages: list[int]) -> int:
    """Cut the pages as a pass would and price it. No reader is built, no request sent.

    A heading is read in a real pass; here every heading is labelled with a
    placeholder so its boundary is made, which is what decides the crop count.
    """
    profile = lxvol.load_profile(bo.DE_PROFILE)
    abbreviations = lxin.load_abbreviations(profile.abbreviations, lang="de")
    prompts = systems(abbreviations)
    headings: list[tuple[int, int]] = []

    def placeholder(crop, where: str) -> str:
        headings.append(crop.size)
        return "Ortsbestandteile im Gerichtsbezirk Probe:"

    cuts = cut_pages(pages, profile, placeholder)
    crops = [c[0].size for cut in cuts for c in cut.crops]
    headers = [cut.header.size for cut in cuts if cut.header is not None]
    from PIL import Image
    with Image.open(bo.DE_ATLAS) as atlas:
        atlas_size = atlas.size
    print(f"German notes arm {bo.RUN_TAG} — {profile.title} — host {bo.HOST}")
    print(f"  pages {','.join(map(str, pages))}  (calibration + purposive; "
          f"certification never read)")
    print(f"  passes to {bo.DE_PASS_DIR}")
    for cut in cuts:
        kinds = Counter(s.kind for s in cut.segments)
        print(f"    {cut.stem}: {cut.width} px wide (scale {cut.scale:.3f}), "
              f"{dict(kinds)}, {len(cut.crops)} crops at {cut.strip} px strips, "
              f"header {'cut' if cut.header is not None else 'NOT FOUND'}")
    print(f"  per pass: {len(crops)} crops, {len(headings)} heading reads, "
          f"{len(headers)} header reads; key {atlas_size[0]}x{atlas_size[1]}")
    total = 0.0
    print()
    for job in plan(names, repeats):
        arm = job["arm"]
        est = estimate(arm, crops, headings, headers,
                       {k: len(v) for k, v in prompts.items()}, atlas_size)
        total += 0 if job["done"] else est["usd"]
        print(f"  {'done' if job['done'] else 'todo':4s}  {job['label']:26s} "
              f"{arm.model:18s} {arm.route:6s} "
              + (f"reasoning={arm.reasoning}" if arm.route == "gemini"
                 else f"thinking={arm.thinking} effort={arm.effort}")
              + f"  ~{est['tokens_in']:,} in / {est['tokens_out']:,} out  ~${est['usd']:.3f}")
    print(f"\n  still to read: about ${total:.2f} — an estimate (image tokens by w·h/750, "
          f"{EST_OUTPUT['crop']} output tokens a crop, the pilot's placeholder thinking "
          f"budget for Haiku at high). No request was sent.")
    return 0

def ready(arm: Arm) -> tuple[bool, str]:
    """One tiny request as the pass would send it, through the runners that own it."""
    if arm.route == "gemini":
        import run_e1_gemini_version as e1
        ok, why = e1.quota_open(arm.model)
        return ok, why
    import run_e1b_second_reader as e1b
    state, why = e1b.ping(e1b.CANDIDATES["haiku55h"])
    return state == "ok", why


def run(names, repeats: int, pages: list[int], force: bool = False) -> bool:
    profile = lxvol.load_profile(bo.DE_PROFILE)
    abbreviations = lxin.load_abbreviations(profile.abbreviations, lang="de")
    crosswalk = lxdn.load_court_crosswalk()
    for job in plan(names, repeats):
        if job["done"] and not force:
            print(f"  {job['label']:26s} on disk — skipped")
            continue
        ok, why = ready(job["arm"])
        if not ok:
            print(f"\n  {job['label']} not run — the endpoint said not now ({why[:120]}); "
                  f"stopping, rerun to resume")
            return False
        marker = bo.DE_PASS_DIR / f"ratelimit_{job['label']}.json"
        if marker.exists():
            marker.replace(bo.DE_PASS_DIR / f"superseded_ratelimit_{job['label']}_"
                                            f"{time.strftime('%Y%m%d-%H%M%S')}.json")
        print("\n" + "=" * 78 + f"\nGERMAN NOTES  {job['label']}   ({job['arm'].model}, "
              f"repeat {job['repeat']})\n" + "=" * 78)
        if not run_pass(job, pages, profile, abbreviations, crosswalk):
            return False
    return True
