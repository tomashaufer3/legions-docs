How it works

The frozen runner will write the ledger as it goes. Until then this rebuilds it for the twelve gold pages from the files every stage already wrote, so the route can be read mark by mark today:

read the reader of record gemini38_r5atlasntg1 and Haiku's three r5 reads haiku55nh_r5atlasg1–3 (D-108), from E3's placement of their claims (e3_verdicts_r5.csv: each claim's box); a reader that read the entry and claimed nothing at a box reads ABSENT there order the printed-order flags of those four passes (rescore_r5_order_flags.csv) dinov3 the encoder's name for every box, with its verdict on the reading of record (and the glyph-like boxes no reader claimed) opus test B's answers, base arm (output/verify_test_b) — asked about the Gemini × Haiku-1 disputes and DINOv3's doubts, three repeats final computed (lexikon_1910_ledger.final_value)

author the author's rulings on the marks Opus could not tell (author_rulings.csv in the ledger folder, from the Token Review page; 1.2.0), when the frozen runner has written them

No cz_de events are built here: the frozen runner adds them once both editions' final inventories exist (run_frozen.compare_editions). Scan 0449 is the readers' worked example and is left out, as in every gold score. The gold is joined as a reference column, never as a step.

def reader_events(verdicts: list[dict], entries: dict) -> tuple[list, dict, dict]:
    """Reads and DINOv3 names from E3's rows: ``(events, {mark: place}, {mark: gold})``."""
    events, place, gold = [], {}, {}
    rank: dict = {}
    SECTION.clear()
    dino_seen, read_at = set(), defaultdict(set)
    unplaced_k = Counter()
    for r in verdicts:
        if r["pass"] not in READERS or r["scan"] in HELD_OUT:
            continue
        actor, token = READERS[r["pass"]], r["claim"]
        if r["box"]:
            mark = led.box_mark(r["image"], r["box"])
        elif token:
            key = (r["pass"],) + _place(r) + (token,)
            mark = led.unplaced_mark(*_place(r), token, unplaced_k[key])
            unplaced_k[key] += 1
        else:
            continue
        priority = _place_priority(actor, token)
        if mark not in place or priority < rank[mark]:
            place[mark], rank[mark] = _place(r), priority
        if r.get("section"):
            SECTION.setdefault(mark, r["section"])
        if r.get("gold"):
            gold[mark] = r["gold"]
        if token and (actor, mark) not in read_at[mark]:
            events.append(led.Event(mark, "read", actor, token,
                                    "no box found" if not r["box"] else "",
                                    f"e3_verdicts:{r['pass']}", VERSION))
            read_at[mark].add((actor, mark))
        if r["box"] and mark not in dino_seen:
            name = r["dino"] if r["dino"].startswith("[") and "not-in" not in r["dino"] else ""
            note = r["verdict"] if actor == RECORD or not token else ""
            events.append(led.Event(mark, "dinov3", DINO, name, note or r["verdict"],
                                    f"e3_verdicts:{r['pass']}", VERSION))
            dino_seen.add(mark)
    # DINOv3's verdict, relative to the reading of record on that box.
    record_read = {e.mark: e.read for e in events if e.step == "read" and e.actor == RECORD}
    events = [led.Event(e.mark, e.step, e.actor, e.read,
                        ("agree" if e.read == record_read[e.mark] else "disagree")
                        if e.step == "dinov3" and e.read and e.mark in record_read else e.note,
                        e.evidence, e.version) for e in events]
    # A reader that read the entry and put nothing on the mark reads it as absent.
    readers_at = defaultdict(set)
    for e in events:
        if e.step == "read":
            readers_at[e.mark].add(e.actor)
    for mark, (page, district, muni, _seq) in place.items():
        for actor in READERS.values():
            if actor not in readers_at[mark] and (page, district, muni) in entries[actor]:
                events.append(led.Event(mark, "read", actor, led.ABSENT, "no claim here",
                                        "", VERSION))
    return events, place, gold

def order_events(place: dict, events: list) -> list:
    """The order rule's flags, put on the marks of the flagged reading."""
    marks_of = defaultdict(list)                       # (actor, place, token) -> marks
    for e in events:
        if e.step == "read" and e.read != led.ABSENT:
            marks_of[(e.actor, place[e.mark], e.read)].append(e.mark)
    out = []
    if ORDER_FLAGS is None or not Path(ORDER_FLAGS).exists():
        return out
    for r in tb._read_csv(Path(ORDER_FLAGS)):
        if r["pass"] not in READERS or r["verdict"] != "out_of_place":
            continue
        actor = READERS[r["pass"]]
        key = (actor, (r["page"], r["district"], str(r["muni_seq"]),
                       str(r["settlement_seq"] or "0")), r["code"])
        for mark in marks_of.get(key, []):
            out.append(led.Event(mark, "order", "order-rule", r["code"],
                                 f"out of place in {actor}'s reading",
                                 f"order_flags:{r['pass']}", VERSION))
    return out

def opus_events(place: dict, events: list | None=None) -> list

Test B's answers, base arm, as events on the marks they were asked about.

A box question rules on that box. "None of these" is led.NONE_OF_THESE. A token is led.NO_RULING when it is the readers' majority reading at another mark of the same section and part (1.4.0, D-120): E3 then put a reader's claim on the neighbouring glyph — Opus rightly names the glyph inside the outline, but taking that as the box's value would turn a correct claim into a duplicate of its neighbour (three of the Czech gold's seven misses were exactly this). A minority reading elsewhere, or the same token as a part's type in the list of parts, is no such sign: Opus's answer stands. lexikon_1910_ledger.final_value decides a mark Opus settled nothing on.

A token question ("is it printed for this part?") is a ruling on the claim, not on any box, so it goes on the claim's own mark — the unplaced mark of (part, token), created if no reader left one — and lexikon_1910_ledger.claim_rulings applies it to the claim wherever the readers filed it (1.4.0, D-120). Spreading it over the boxes that hold the token (1.3.0) let "T not printed" erase another token a misplaced box also held.

An answer to a question the current items no longer have is skipped and counted, never matched to a different question (D-120).

def opus_events(place: dict, events: list | None = None) -> list:
    """Test B's answers, base arm, as events on the marks they were asked about.

    **A box question** rules on that box. "None of these" is :data:`led.NONE_OF_THESE`.
    A token is :data:`led.NO_RULING` when it is the readers' *majority* reading at
    *another* mark of the same section and part (1.4.0, D-120): E3 then put a reader's
    claim on the neighbouring glyph — Opus rightly names the glyph inside the outline,
    but taking that as the box's value would turn a correct claim into a duplicate of
    its neighbour (three of the Czech gold's seven misses were exactly this). A minority
    reading elsewhere, or the same token as a part's *type* in the list of parts, is
    no such sign: Opus's answer stands. :func:`lexikon_1910_ledger.final_value` decides
    a mark Opus settled nothing on.

    **A token question** ("is it printed for this part?") is a ruling on the *claim*,
    not on any box, so it goes on the claim's own mark — the unplaced mark of (part,
    token), created if no reader left one — and :func:`lexikon_1910_ledger.claim_rulings`
    applies it to the claim wherever the readers filed it (1.4.0, D-120). Spreading it
    over the boxes that hold the token (1.3.0) let "T not printed" erase another token
    a misplaced box also held.

    An answer to a question the current items no longer have is skipped and counted,
    never matched to a different question (D-120).
    """
    reads_at: dict = defaultdict(list)
    for e in events or []:
        if e.step == "read" and e.mark in place:
            reads_at[e.mark].append(e.read)
    holders: dict = defaultdict(set)              # (part, section, token) -> marks it is the majority at
    for mark, reads in reads_at.items():
        for token, n in Counter(r for r in reads if r.startswith("[")).items():
            if n > len(READERS) / 2:
                holders[tuple(map(str, place[mark])) + (section_of(mark), token)].add(mark)
    out_dir = OPUS_DIR
    if out_dir is None or not (Path(out_dir) / "answers_raw.csv").exists():
        return []
    manifest = {m["id"]: m for m in json.loads((out_dir / "items.json").read_text(encoding="utf-8"))}
    out, stale = [], 0
    for r in tb._read_csv(out_dir / "answers_raw.csv"):
        item = manifest.get(r["item"])
        q = next((q for q in item["questions"] if q["n"] == int(r["question"])), None) if item else None
        if q is None:
            stale += 1
            continue
        answer = r["answer"]
        unit = tuple(map(str, item["unit"]))
        if q["kind"] == "mark":
            mark = led.box_mark(q["image"], ",".join(map(str, q["box"])))
            if answer.startswith("["):
                elsewhere = holders.get(unit + (section_of(mark), answer), set()) - {mark}
                read = led.NO_RULING if elsewhere else answer
            else:
                read = led.NONE_OF_THESE if answer == "none of these" else led.UNSURE
        else:
            read = {"printed": q["token"], "not printed": led.ABSENT}.get(answer, led.UNSURE)
            # A place-type question (D-131) rules on the part's type, never on an
            # institution of the same token (`[castle]` is both): a mark of its own,
            # which `claim_rulings` (marks ending `#0`) does not read. A text abbreviation's
            # question (D-134) the same way: `#abbr`, read by `run_frozen.final_abbreviations`.
            kind = "type" if q.get("type_mark") else "abbr" if q.get("abbr_mark") else 0
            mark = led.unplaced_mark(*unit, q["token"], kind)
            place.setdefault(mark, unit)
        out.append(led.Event(mark, "opus", f"opus#{r['repeat']}", read, r.get("note", "")[:120],
                             f"test_b:{r['custom_id']}#q{r['question']}", VERSION))
    if stale:
        print(f"  !! {stale} Opus answers belong to questions the current items do not have: skipped")
    return out

def build() -> tuple[list, dict, dict]:
    verdicts = tb._read_csv(Path(VERDICTS_PATH))
    entries = {READERS[p]: set(lxin._claim_counts(tb.pass_rows(p), "settlement")[0]) for p in READERS}
    entries = {a: {tuple(map(str, e)) for e in es} for a, es in entries.items()}
    events, place, gold = reader_events(verdicts, entries)
    events += order_events(place, events)
    events += opus_events(place, events)
    events += [e for e in author_events() if e.mark in place]
    human = HUMAN_ROWS if HUMAN_ROWS is not None else bo.load_pass(bo.HUMAN)
    truth = {tuple(map(str, c)) for c in lxin._claim_counts(human, "settlement")[2]}
    unplaced_gold(place, gold, truth)
    for mark in place:
        gold.setdefault(mark, led.ABSENT)
    return led.with_final(events, RECORD, VERSION), place, gold

def report(events: list, place: dict, gold: dict) -> str:
    marks = led.by_mark(events)
    final = {m: next(e.read for e in es if e.step == "final") for m, es in marks.items()}
    rules = Counter(next(e.note for e in es if e.step == "final") for es in marks.values())
    record = {m: next((e.read for e in es if e.step == "read" and e.actor == RECORD), led.ABSENT)
              for m, es in marks.items()}
    right = lambda v, m: v == (gold.get(m) or led.ABSENT)                 # noqa: E731
    # A box carries a gold token only where E3 ran on the gold's own section images;
    # on saved crops it carries none, and a mark-by-mark score would count every
    # reading wrong. The claim-level score is the one to read then.
    boxed_gold = any(gold.get(m) not in (None, "", led.ABSENT)
                     for m in marks if m.split("#")[0].endswith(".png"))
    lines = [f"mark ledger {__version__} — {NAME} ({len(marks)} marks, {len(events)} events)", "",
             (f"  reading of record right on {sum(right(record[m], m) for m in marks)} marks; "
              f"the route's final value right on {sum(right(final[m], m) for m in marks)}"
              if boxed_gold else
              "  no gold token on the boxes (saved crops): see the claim-level score"),
             "  how the final value was set: " + ", ".join(f"{k} {n}" for k, n in rules.most_common()), ""]
    changes = led.step_changes(events, RECORD, gold if boxed_gold else None)
    lines.append(f"  {'step':8s} {'touched':>7s} {'changed':>7s} {'right':>6s} {'wrong':>6s}   top transitions")
    for step in led.STEPS:
        c = changes[step]
        if not c["touched"]:
            continue
        top = ", ".join(f"{a} → {b} ×{n}" for (a, b), n in c["transitions"].most_common(4))
        rw = (f"{c['right']:6d} {c['wrong']:6d}" if step not in ("read", "order", "dinov3")
              else f"{'':6s} {'':6s}")
        lines.append(f"  {step:8s} {c['touched']:7d} {c['changed']:7d} {rw}   {top}")
    lines += ["", "  read: the reading of record. order and dinov3 only flag — 'changed' counts the "
                  "marks where DINOv3 named another token, a proposal, not a change of value.",
              "  right/wrong: of the marks a step changed, how many it changed to the gold's value."]
    return "\n".join(lines)

def trails(events: list, place: dict, gold: dict, marks: list | None = None) -> str:
    grouped = led.by_mark(events)
    if marks is None:          # every mark some step disagreed on, or whose final moved
        marks = [m for m, es in grouped.items()
                 if len({e.read for e in es if e.step in ("read", "opus", "final")}) > 1
                 or any(e.step == "dinov3" and e.note == "disagree" for e in es)]
    out = []
    for m in sorted(marks, key=lambda m: (place.get(m, ()), m)):
        p = place.get(m, ("", "", "", ""))
        out += led.trail(m, grouped[m], gold.get(m, ""), f"{p[0]} {p[1]} {p[2]},{p[3]}") + [""]
    return "\n".join(out)

def main(argv=None) -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):
            pass
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--trail", nargs="+", metavar="PAGE DISTRICT MUNI [PART]",
                    help="print every mark of one municipality (or part) and stop")
    args = ap.parse_args(argv)
    events, place, gold = build()
    if args.trail:
        want = tuple(args.trail)
        marks = [m for m, p in place.items() if tuple(p[:len(want)]) == want]
        print(trails(events, place, gold, marks) or "no marks there")
        return 0
    out = paths()
    led.write_ledger(events, out["ledger"])
    with out["marks"].open("w", encoding="utf-8", newline="") as fh:
        w = csv.writer(fh)
        w.writerow(["mark", "page", "district", "muni", "seq", "gold", "final", "rule"])
        for m, es in led.by_mark(events).items():
            fin = next(e for e in es if e.step == "final")
            w.writerow([m, *place.get(m, ("", "", "", "")), gold.get(m, ""), fin.read, fin.note])
    text = report(events, place, gold)
    out["changes"].write_text(text + "\n", encoding="utf-8")
    out["trails"].write_text(trails(events, place, gold) + "\n", encoding="utf-8")
    print(text)
    print(f"\n  wrote {out['ledger'].relative_to(ROOT)}, {out['marks'].name}, "
          f"{out['changes'].name}, {out['trails'].name}")
    return 0

"""Fill the mark ledger (D-109) from what the r5 gold run left on disk, and report it.

The frozen runner will write the ledger as it goes. Until then this rebuilds it
for the twelve gold pages from the files every stage already wrote, so the route
can be read mark by mark today:

``read``    the reader of record ``gemini38_r5atlasntg1`` and Haiku's three r5
            reads ``haiku55nh_r5atlasg1–3`` (D-108), from E3's placement of their
            claims (``e3_verdicts_r5.csv``: each claim's box); a reader that read
            the entry and claimed nothing at a box reads :data:`ABSENT` there
``order``   the printed-order flags of those four passes (``rescore_r5_order_flags.csv``)
``dinov3``  the encoder's name for every box, with its verdict on the reading of
            record (and the glyph-like boxes no reader claimed)
``opus``    test B's answers, base arm (``output/verify_test_b``) — asked about
            the Gemini × Haiku-1 disputes and DINOv3's doubts, three repeats
``final``   computed (:func:`lexikon_1910_ledger.final_value`)

``author``  the author's rulings on the marks Opus could not tell
            (``author_rulings.csv`` in the ledger folder, from the Token Review
            page; 1.2.0), when the frozen runner has written them

No ``cz_de`` events are built here: the frozen runner adds them once both
editions' final inventories exist (``run_frozen.compare_editions``). Scan 0449 is the readers' worked
example and is left out, as in every gold score. The gold is joined as a
reference column, never as a step.

    python utils/lexikon_1910/build_mark_ledger.py                     # build and report
    python utils/lexikon_1910/build_mark_ledger.py --trail 0196 nusle 3 # every mark of a municipality
"""

from __future__ import annotations

__version__ = "1.6.0"

import argparse
import csv
import json
import sys
from collections import Counter, defaultdict
from pathlib import Path

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]
sys.path.insert(0, str(ROOT))
sys.path.insert(0, str(HERE))

import bakeoff as bo                                             # noqa: E402
import run_verify_test_b as tb                                   # noqa: E402
from utils.lexikon_1910 import lexikon_1910_institutions as lxin      # noqa: E402
from utils.lexikon_1910 import lexikon_1910_ledger as led             # noqa: E402

READERS = {"gemini38_r5atlasntg1": "gemini38#1", "haiku55nh_r5atlasg1": "haiku55nh#1",
           "haiku55nh_r5atlasg2": "haiku55nh#2", "haiku55nh_r5atlasg3": "haiku55nh#3"}
RECORD = "gemini38#1"
HELD_OUT = {"0449"}
DINO = "dinov3-vitb16"
OUT = ROOT / "output" / "mark_ledger"
NAME = "cz_gold"
VERDICTS_PATH = tb.VERDICTS
ORDER_FLAGS = bo.BAKEOFF / f"rescore_{bo.RUN_TAG}_order_flags.csv"
OPUS_DIR = tb.ARMS["base"]
VERSION = f"build_mark_ledger {__version__}; ledger {led.__version__}"
#: The gold the marks are joined to, as rows; ``None`` reads ``bakeoff.HUMAN`` (the
#: Czech gold). The frozen runner passes the German gold for a German edition.
HUMAN_ROWS: list | None = None
AUTHOR_FILE = "author_rulings.csv"


def paths() -> dict:
    return {"ledger": OUT / f"ledger_{NAME}.csv", "marks": OUT / f"marks_{NAME}.csv",
            "changes": OUT / f"changes_{NAME}.txt", "trails": OUT / f"trails_changed_{NAME}.txt"}


def configure(readers: dict, record: str, verdicts: Path, order_flags: Path,
              opus_dir: Path | None, out_dir: Path, name: str,
              held_out: set | None = None, human_rows: list | None = None) -> None:
    """Point the builder at another run's files (1.1.0) — the frozen runner's, say.

    ``readers`` maps each pass label to its actor name, the reader of record
    (``record``) among them; ``opus_dir`` holds that run's ``items.json`` and
    ``answers_raw.csv`` (``None``: no Opus step yet). ``held_out`` is empty unless
    given (1.2.0): scan 0449 is left out of a *gold score*, never out of a volume's
    ledger, where dropping it would drop the page from the final inventory.
    ``human_rows`` is the gold to join (``None``: the Czech gold).
    """
    global READERS, RECORD, VERDICTS_PATH, ORDER_FLAGS, OPUS_DIR, OUT, NAME, HELD_OUT, HUMAN_ROWS
    READERS, RECORD, VERDICTS_PATH, ORDER_FLAGS = dict(readers), record, verdicts, order_flags
    OPUS_DIR, OUT, NAME = opus_dir, out_dir, name
    HELD_OUT, HUMAN_ROWS = set(held_out or ()), human_rows


def _place(row: dict) -> tuple:
    return (row["scan"], row["district"], str(row["muni"]), str(row["seq"]))


def _place_priority(actor: str, token: str) -> int:
    """Whose row files a mark under its part (1.4.0, D-120): the reader of record's claim,
    then any reader's claim, and a row that claims nothing (``unclaimed``: E3's guess at
    the part of a box no reader read) only where no reader claimed the box at all."""
    if token:
        return 0 if actor == RECORD else 1
    return 2


#: ``{mark: "offices" | "parts"}`` — the section each mark was read in, filled by
#: :func:`reader_events`: a box answer is compared with marks of its own section only.
SECTION: dict = {}


def section_of(mark: str) -> str:
    """A boxed mark's section from its image; an unplaced one's as its rows recorded it."""
    image = mark.split("#")[0]
    if image.endswith(".png"):
        return "offices" if "_offices_" in image else "parts"
    return SECTION.get(mark, "")


def reader_events(verdicts: list[dict], entries: dict) -> tuple[list, dict, dict]:
    """Reads and DINOv3 names from E3's rows: ``(events, {mark: place}, {mark: gold})``."""
    events, place, gold = [], {}, {}
    rank: dict = {}
    SECTION.clear()
    dino_seen, read_at = set(), defaultdict(set)
    unplaced_k = Counter()
    for r in verdicts:
        if r["pass"] not in READERS or r["scan"] in HELD_OUT:
            continue
        actor, token = READERS[r["pass"]], r["claim"]
        if r["box"]:
            mark = led.box_mark(r["image"], r["box"])
        elif token:
            key = (r["pass"],) + _place(r) + (token,)
            mark = led.unplaced_mark(*_place(r), token, unplaced_k[key])
            unplaced_k[key] += 1
        else:
            continue
        priority = _place_priority(actor, token)
        if mark not in place or priority < rank[mark]:
            place[mark], rank[mark] = _place(r), priority
        if r.get("section"):
            SECTION.setdefault(mark, r["section"])
        if r.get("gold"):
            gold[mark] = r["gold"]
        if token and (actor, mark) not in read_at[mark]:
            events.append(led.Event(mark, "read", actor, token,
                                    "no box found" if not r["box"] else "",
                                    f"e3_verdicts:{r['pass']}", VERSION))
            read_at[mark].add((actor, mark))
        if r["box"] and mark not in dino_seen:
            name = r["dino"] if r["dino"].startswith("[") and "not-in" not in r["dino"] else ""
            note = r["verdict"] if actor == RECORD or not token else ""
            events.append(led.Event(mark, "dinov3", DINO, name, note or r["verdict"],
                                    f"e3_verdicts:{r['pass']}", VERSION))
            dino_seen.add(mark)
    # DINOv3's verdict, relative to the reading of record on that box.
    record_read = {e.mark: e.read for e in events if e.step == "read" and e.actor == RECORD}
    events = [led.Event(e.mark, e.step, e.actor, e.read,
                        ("agree" if e.read == record_read[e.mark] else "disagree")
                        if e.step == "dinov3" and e.read and e.mark in record_read else e.note,
                        e.evidence, e.version) for e in events]
    # A reader that read the entry and put nothing on the mark reads it as absent.
    readers_at = defaultdict(set)
    for e in events:
        if e.step == "read":
            readers_at[e.mark].add(e.actor)
    for mark, (page, district, muni, _seq) in place.items():
        for actor in READERS.values():
            if actor not in readers_at[mark] and (page, district, muni) in entries[actor]:
                events.append(led.Event(mark, "read", actor, led.ABSENT, "no claim here",
                                        "", VERSION))
    return events, place, gold


def order_events(place: dict, events: list) -> list:
    """The order rule's flags, put on the marks of the flagged reading."""
    marks_of = defaultdict(list)                       # (actor, place, token) -> marks
    for e in events:
        if e.step == "read" and e.read != led.ABSENT:
            marks_of[(e.actor, place[e.mark], e.read)].append(e.mark)
    out = []
    if ORDER_FLAGS is None or not Path(ORDER_FLAGS).exists():
        return out
    for r in tb._read_csv(Path(ORDER_FLAGS)):
        if r["pass"] not in READERS or r["verdict"] != "out_of_place":
            continue
        actor = READERS[r["pass"]]
        key = (actor, (r["page"], r["district"], str(r["muni_seq"]),
                       str(r["settlement_seq"] or "0")), r["code"])
        for mark in marks_of.get(key, []):
            out.append(led.Event(mark, "order", "order-rule", r["code"],
                                 f"out of place in {actor}'s reading",
                                 f"order_flags:{r['pass']}", VERSION))
    return out


def opus_events(place: dict, events: list | None = None) -> list:
    """Test B's answers, base arm, as events on the marks they were asked about.

    **A box question** rules on that box. "None of these" is :data:`led.NONE_OF_THESE`.
    A token is :data:`led.NO_RULING` when it is the readers' *majority* reading at
    *another* mark of the same section and part (1.4.0, D-120): E3 then put a reader's
    claim on the neighbouring glyph — Opus rightly names the glyph inside the outline,
    but taking that as the box's value would turn a correct claim into a duplicate of
    its neighbour (three of the Czech gold's seven misses were exactly this). A minority
    reading elsewhere, or the same token as a part's *type* in the list of parts, is
    no such sign: Opus's answer stands. :func:`lexikon_1910_ledger.final_value` decides
    a mark Opus settled nothing on.

    **A token question** ("is it printed for this part?") is a ruling on the *claim*,
    not on any box, so it goes on the claim's own mark — the unplaced mark of (part,
    token), created if no reader left one — and :func:`lexikon_1910_ledger.claim_rulings`
    applies it to the claim wherever the readers filed it (1.4.0, D-120). Spreading it
    over the boxes that hold the token (1.3.0) let "T not printed" erase another token
    a misplaced box also held.

    An answer to a question the current items no longer have is skipped and counted,
    never matched to a different question (D-120).
    """
    reads_at: dict = defaultdict(list)
    for e in events or []:
        if e.step == "read" and e.mark in place:
            reads_at[e.mark].append(e.read)
    holders: dict = defaultdict(set)              # (part, section, token) -> marks it is the majority at
    for mark, reads in reads_at.items():
        for token, n in Counter(r for r in reads if r.startswith("[")).items():
            if n > len(READERS) / 2:
                holders[tuple(map(str, place[mark])) + (section_of(mark), token)].add(mark)
    out_dir = OPUS_DIR
    if out_dir is None or not (Path(out_dir) / "answers_raw.csv").exists():
        return []
    manifest = {m["id"]: m for m in json.loads((out_dir / "items.json").read_text(encoding="utf-8"))}
    out, stale = [], 0
    for r in tb._read_csv(out_dir / "answers_raw.csv"):
        item = manifest.get(r["item"])
        q = next((q for q in item["questions"] if q["n"] == int(r["question"])), None) if item else None
        if q is None:
            stale += 1
            continue
        answer = r["answer"]
        unit = tuple(map(str, item["unit"]))
        if q["kind"] == "mark":
            mark = led.box_mark(q["image"], ",".join(map(str, q["box"])))
            if answer.startswith("["):
                elsewhere = holders.get(unit + (section_of(mark), answer), set()) - {mark}
                read = led.NO_RULING if elsewhere else answer
            else:
                read = led.NONE_OF_THESE if answer == "none of these" else led.UNSURE
        else:
            read = {"printed": q["token"], "not printed": led.ABSENT}.get(answer, led.UNSURE)
            # A place-type question (D-131) rules on the part's type, never on an
            # institution of the same token (`[castle]` is both): a mark of its own,
            # which `claim_rulings` (marks ending `#0`) does not read. A text abbreviation's
            # question (D-134) the same way: `#abbr`, read by `run_frozen.final_abbreviations`.
            kind = "type" if q.get("type_mark") else "abbr" if q.get("abbr_mark") else 0
            mark = led.unplaced_mark(*unit, q["token"], kind)
            place.setdefault(mark, unit)
        out.append(led.Event(mark, "opus", f"opus#{r['repeat']}", read, r.get("note", "")[:120],
                             f"test_b:{r['custom_id']}#q{r['question']}", VERSION))
    if stale:
        print(f"  !! {stale} Opus answers belong to questions the current items do not have: skipped")
    return out


def author_events() -> list:
    """The author's rulings (``author_rulings.csv``: mark, read, note, evidence)."""
    path = Path(OUT) / AUTHOR_FILE
    if not path.exists():
        return []
    return [led.Event(r["mark"], "author", "author", r["read"], r.get("note", "")[:200],
                      r.get("evidence", ""), VERSION) for r in tb._read_csv(path)]


def unplaced_gold(place: dict, gold: dict, truth: set) -> None:
    """An unplaced mark's gold: its token if the gold has it in that part, else absent."""
    for mark, p in place.items():
        if "#" in mark and "/" in mark.split("#")[0] and mark not in gold:
            token = mark.split("#")[1]
            gold[mark] = token if tuple(p) + (token,) in truth else led.ABSENT


def build() -> tuple[list, dict, dict]:
    verdicts = tb._read_csv(Path(VERDICTS_PATH))
    entries = {READERS[p]: set(lxin._claim_counts(tb.pass_rows(p), "settlement")[0]) for p in READERS}
    entries = {a: {tuple(map(str, e)) for e in es} for a, es in entries.items()}
    events, place, gold = reader_events(verdicts, entries)
    events += order_events(place, events)
    events += opus_events(place, events)
    events += [e for e in author_events() if e.mark in place]
    human = HUMAN_ROWS if HUMAN_ROWS is not None else bo.load_pass(bo.HUMAN)
    truth = {tuple(map(str, c)) for c in lxin._claim_counts(human, "settlement")[2]}
    unplaced_gold(place, gold, truth)
    for mark in place:
        gold.setdefault(mark, led.ABSENT)
    return led.with_final(events, RECORD, VERSION), place, gold


def report(events: list, place: dict, gold: dict) -> str:
    marks = led.by_mark(events)
    final = {m: next(e.read for e in es if e.step == "final") for m, es in marks.items()}
    rules = Counter(next(e.note for e in es if e.step == "final") for es in marks.values())
    record = {m: next((e.read for e in es if e.step == "read" and e.actor == RECORD), led.ABSENT)
              for m, es in marks.items()}
    right = lambda v, m: v == (gold.get(m) or led.ABSENT)                 # noqa: E731
    # A box carries a gold token only where E3 ran on the gold's own section images;
    # on saved crops it carries none, and a mark-by-mark score would count every
    # reading wrong. The claim-level score is the one to read then.
    boxed_gold = any(gold.get(m) not in (None, "", led.ABSENT)
                     for m in marks if m.split("#")[0].endswith(".png"))
    lines = [f"mark ledger {__version__} — {NAME} ({len(marks)} marks, {len(events)} events)", "",
             (f"  reading of record right on {sum(right(record[m], m) for m in marks)} marks; "
              f"the route's final value right on {sum(right(final[m], m) for m in marks)}"
              if boxed_gold else
              "  no gold token on the boxes (saved crops): see the claim-level score"),
             "  how the final value was set: " + ", ".join(f"{k} {n}" for k, n in rules.most_common()), ""]
    changes = led.step_changes(events, RECORD, gold if boxed_gold else None)
    lines.append(f"  {'step':8s} {'touched':>7s} {'changed':>7s} {'right':>6s} {'wrong':>6s}   top transitions")
    for step in led.STEPS:
        c = changes[step]
        if not c["touched"]:
            continue
        top = ", ".join(f"{a} → {b} ×{n}" for (a, b), n in c["transitions"].most_common(4))
        rw = (f"{c['right']:6d} {c['wrong']:6d}" if step not in ("read", "order", "dinov3")
              else f"{'':6s} {'':6s}")
        lines.append(f"  {step:8s} {c['touched']:7d} {c['changed']:7d} {rw}   {top}")
    lines += ["", "  read: the reading of record. order and dinov3 only flag — 'changed' counts the "
                  "marks where DINOv3 named another token, a proposal, not a change of value.",
              "  right/wrong: of the marks a step changed, how many it changed to the gold's value."]
    return "\n".join(lines)


def trails(events: list, place: dict, gold: dict, marks: list | None = None) -> str:
    grouped = led.by_mark(events)
    if marks is None:          # every mark some step disagreed on, or whose final moved
        marks = [m for m, es in grouped.items()
                 if len({e.read for e in es if e.step in ("read", "opus", "final")}) > 1
                 or any(e.step == "dinov3" and e.note == "disagree" for e in es)]
    out = []
    for m in sorted(marks, key=lambda m: (place.get(m, ()), m)):
        p = place.get(m, ("", "", "", ""))
        out += led.trail(m, grouped[m], gold.get(m, ""), f"{p[0]} {p[1]} {p[2]},{p[3]}") + [""]
    return "\n".join(out)


def main(argv=None) -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):
            pass
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--trail", nargs="+", metavar="PAGE DISTRICT MUNI [PART]",
                    help="print every mark of one municipality (or part) and stop")
    args = ap.parse_args(argv)
    events, place, gold = build()
    if args.trail:
        want = tuple(args.trail)
        marks = [m for m, p in place.items() if tuple(p[:len(want)]) == want]
        print(trails(events, place, gold, marks) or "no marks there")
        return 0
    out = paths()
    led.write_ledger(events, out["ledger"])
    with out["marks"].open("w", encoding="utf-8", newline="") as fh:
        w = csv.writer(fh)
        w.writerow(["mark", "page", "district", "muni", "seq", "gold", "final", "rule"])
        for m, es in led.by_mark(events).items():
            fin = next(e for e in es if e.step == "final")
            w.writerow([m, *place.get(m, ("", "", "", "")), gold.get(m, ""), fin.read, fin.note])
    text = report(events, place, gold)
    out["changes"].write_text(text + "\n", encoding="utf-8")
    out["trails"].write_text(trails(events, place, gold) + "\n", encoding="utf-8")
    print(text)
    print(f"\n  wrote {out['ledger'].relative_to(ROOT)}, {out['marks'].name}, "
          f"{out['changes'].name}, {out['trails'].name}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
