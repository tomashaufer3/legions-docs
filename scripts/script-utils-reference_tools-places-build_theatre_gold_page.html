#!/usr/bin/env python3
"""
The ruling page for the theatre model pass's gold (D-157).

Reads the blind worksheet ``reference/places/theatre_model_gold.csv`` and the page
template ``theatre_gold_page.html`` beside this script, and writes one self-contained
HTML page with the clauses embedded, for publishing as an artifact whose ``db`` holds the
rulings, one document a clause at ``rulings/<clause_key>``.  The page shows no model reply.

Back the other way, :func:`utils.parsing.amateur_theatre_model.fill_gold_sheet` writes
the ruled documents into the worksheet, which ``read_gold`` and ``score`` read.

Usage::

    python utils/reference_tools/places/build_theatre_gold_page.py OUT.html
"""

from __future__ import annotations

import json
import sys
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

from utils.parsing import amateur_theatre_model as atm                # noqa: E402

TEMPLATE = Path(__file__).with_name("theatre_gold_page.html")
PLACEHOLDER = "/*__DATA__*/"


def page_clauses(sheet: pd.DataFrame) -> list[dict]:
    """The worksheet's rows as the page reads them, in sheet order (``i`` is the row)."""
    return [{"i": i, "key": r.clause_key, "half": r.half, "source": r.source, "kind": r.open_kind,
             "place": r.place, "ensemble": r.ensemble, "context": r.context, "clause": r.clause,
             "date_raw": r.date_raw, "years": r.years_inside, "dated": r.date_raw.strip() != ""}
            for i, r in enumerate(sheet.itertuples(index=False))]


def build(out: Path, sheet_path: Path = atm.GOLD_SHEET) -> Path:
    sheet = pd.read_csv(sheet_path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    data = json.dumps(page_clauses(sheet), ensure_ascii=False).replace("</", "<\\/")
    template = TEMPLATE.read_text(encoding="utf-8")
    assert template.count(PLACEHOLDER) == 1, "template: one data placeholder"
    Path(out).write_text(template.replace(PLACEHOLDER, data), encoding="utf-8")
    return Path(out)


if __name__ == "__main__":
    if len(sys.argv) != 2:
        raise SystemExit(__doc__)
    print("wrote", build(Path(sys.argv[1])))
