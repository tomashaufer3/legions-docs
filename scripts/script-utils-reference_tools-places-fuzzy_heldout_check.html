How it works

N3 links a written name the record's 1910 okres does not hold to the one settlement of the okres whose name is a close spelling of it (rapidfuzz ratio on space-free keys, a lead over the next settlement, a minimum key length). The thresholds come from this check (birth_settlement 0.16.0 ran it as scratch; 0.16.1 moved it here, the 0.16.0 checkers' finding that a constant's justification must be rebuildable from the repo):

  1. pairs — the P1 exact links (p1_register / p1_lexikon) of the legionnaire and the re-parsed VÚA link files output/places/birth_settlement_links_{legion,vua}.parquet: (written name, 1910 okres, settlement), distinct, sorted, shuffled with a fixed seed, the first N_PAIRS;
  2. corruption — each name with one edit of each kind (a dropped letter, two letters swapped, a gender or plural ending, an OCR pair, Sv. written out or abbreviated) and one double edit; a corrupted key that is itself a Czech-lands name is dropped (N3 never runs there, birth_settlement._czech_namesakes);
  3. present — each corrupted name ranked in its own okres (birth_settlement.fuzzy_ranking): a fire on the true settlement is a recovery, a fire on another a wrong link; absent — ranked with the true settlement's names removed, so every fire is a wrong link (the place is not in the okres, as for many real unmatched rows);
  4. mixing — the expected share of wrong links among fires on real unmatched rows, the two tests weighted by the share of N3-eligible unmatched rows whose place the okres holds (MIXING_COUNTS, the sizing's hand-labelled sample of 200 unmatched rows per dataset, 2026-11-17);
  5. choice — the most permissive setting (highest recovery) whose expected wrong-link share is at most WRONG_CAP in both datasets.

Writes output/places/fuzzy_heldout_curve.csv (every setting) and output/places/fuzzy_heldout.json (the inputs, the module's setting and its rates, the setting the rule chooses), LF. Run::

python utils/reference_tools/places/fuzzy_heldout_check.py

Version 1.0.0 (2026-11-18): first version (birth_settlement 0.16.1); the 0.16.0 scratch check (f16/n3val*.py) ported. The 0.16.0 check mixed VÚA at 41/88; the sizing's categories sum to 93 rows (MIXING_COUNTS), so this check reads 41/93 and reports the 0.16.0 shares beside it (mixing_0160).

def mixing_shares(counts: dict = MIXING_COUNTS) -> dict:
    """``{dataset: present / (present + absent)}`` from :data:`MIXING_COUNTS`."""
    out = {}
    for ds, c in counts.items():
        p, a = sum(c["present"].values()), sum(c["absent"].values())
        out[ds] = p / (p + a)
    return out

def p1_pairs(out_dir: Path = OUT) -> list[tuple[str, int, str]]:
    """The distinct ``(written name, 1910 okres, settlement)`` of every P1 exact link of the two link files, sorted."""
    pairs = set()
    for name in LINK_FILES.values():
        d = pd.read_parquet(Path(out_dir) / name, columns=["birth_settlement_match", "birth_settlement_id",
                                                           "birth_okres_id_1910", "birth_settlement_clean"])
        m = (d["birth_settlement_match"].isin(["p1_register", "p1_lexikon"]) & d["birth_settlement_id"].notna()
             & d["birth_okres_id_1910"].notna())
        for c, o, s in zip(d.loc[m, "birth_settlement_clean"], d.loc[m, "birth_okres_id_1910"],
                           d.loc[m, "birth_settlement_id"]):
            if isinstance(c, str):
                pairs.add((c, int(o), s))
    return sorted(pairs)

def corrupt(pairs: list, seed: int = SEED, n: int = N_PAIRS) -> list[tuple]:
    """``[(kind, original, written, okres, settlement)]``: *n* shuffled pairs, each with one edit of each kind and one
    double edit (two random single edits)."""
    rng = random.Random(seed)
    pairs = list(pairs)
    rng.shuffle(pairs)
    pairs = pairs[:n]

    def drop(w):
        i = rng.randrange(len(w))
        return w[:i] + w[i + 1:]

    def swap(w):
        if len(w) < 3:
            return w
        i = rng.randrange(len(w) - 1)
        return w[:i] + w[i + 1] + w[i] + w[i + 2:]

    def ending(w):
        opts = [(a, b) for a, b in ENDINGS if w.endswith(a)]
        if not opts:
            return None
        a, b = rng.choice(opts)
        return w[: len(w) - len(a)] + b

    def sv(w):
        for full in ("Svatý ", "Svatá "):
            if w.startswith(full):
                return "Sv. " + w[6:]
        return "Svatý " + w[4:] if w.startswith("Sv. ") else None

    def ocr(w):
        opts = [(a, b) for a, b in OCR if a in w]
        if not opts:
            return None
        a, b = rng.choice(opts)
        i = w.find(a)
        return w[:i] + b + w[i + len(a):]

    single = {"drop": drop, "swap": swap, "ending": ending, "sv": sv, "ocr": ocr}
    out = []
    for c, o, s in pairs:
        for kind, f in single.items():
            w = f(c)
            if w and w != c:
                out.append((kind, c, w, o, s))
        k1, k2 = rng.sample(["drop", "swap", "ending", "ocr"], 2)
        w = single[k1](c)
        w = single[k2](w) if w else None
        if w and w != c:
            out.append((f"double:{k1}+{k2}", c, w, o, s))
    return out

def rank(samples: list, g) -> pd.DataFrame:
    """Each corrupted name ranked in its okres with the true settlement present (``best``, ``best_sid``, ``second``) and
    removed (``abs_best``, ``abs_second``); names that are a Czech-lands name as written dropped. 0.16.1: a fire on a
    settlement N3's namesake guard (:func:`birth_settlement.fuzzy_namesake_block`) blocks is no fire (``blocked``)."""
    from rapidfuzz import fuzz, process
    rows = []
    for kind, c, w, o, s in samples:
        k = bs.key(w)
        if len(bs.nospace(k)) < 3 or bs._czech_namesakes(k, g):
            continue
        rk = bs.fuzzy_ranking(k, o, g)
        b1 = rk[0] if rk else (0.0, None)
        names, sids = bs._okres_name_list(o, g)
        sc: dict = {}
        for _n, score, i in process.extract(bs.nospace(k), names, scorer=fuzz.ratio, limit=25):
            for x in sids[i]:
                if x != s:
                    sc[x] = max(sc.get(x, 0.0), float(score))
        absent = sorted(sc.items(), key=lambda x: (-x[1], x[0]))
        rows.append({"kind": kind, "orig": c, "written": w, "okres": o, "true": s, "keylen": len(bs.nospace(k)),
                     "best": b1[0], "best_sid": b1[1], "second": rk[1][0] if len(rk) > 1 else 0.0,
                     "blocked": bool(b1[1]) and bs.fuzzy_namesake_block(k, b1[1], g),
                     "abs_best": absent[0][1] if absent else 0.0, "abs_second": absent[1][1] if len(absent) > 1 else 0.0,
                     "abs_blocked": bool(absent) and bs.fuzzy_namesake_block(k, absent[0][0], g)})
    return pd.DataFrame(rows)

def curve(rows: pd.DataFrame, shares: dict, shares_0160: dict | None = None) -> pd.DataFrame:
    """Every setting (:data:`RATIOS` × :data:`LEADS` × :data:`MIN_KEYS`): recovery, the wrong share among fires with the
    target present, the fire rate with it absent, and the expected wrong-link share per dataset (``exp_wrong_<ds>``;
    ``exp_wrong_<ds>_0160`` at the 0.16.0 shares)."""
    R = rows
    out = []
    for ratio in RATIOS:
        for lead in LEADS:
            for mk in MIN_KEYS:
                k = R["keylen"] >= mk
                fp = (R["best"] >= ratio) & ((R["best"] - R["second"]) >= lead) & k & ~R["blocked"]
                ok = fp & (R["best_sid"] == R["true"])
                fa = (R["abs_best"] >= ratio) & ((R["abs_best"] - R["abs_second"]) >= lead) & k & ~R["abs_blocked"]
                firep, wrongp, firea = fp.mean(), (fp & ~ok).mean(), fa.mean()
                row = {"ratio": ratio, "lead": lead, "min_key": mk, "recovery": round(float(ok.mean()), 4),
                       "wrong_present": round(float(wrongp / firep), 4) if firep else 0.0,
                       "fire_absent": round(float(firea), 4)}
                for tag, sh in (("", shares), ("_0160", shares_0160 or {})):
                    for ds, p in sh.items():
                        den = p * firep + (1 - p) * firea
                        row[f"exp_wrong_{ds}{tag}"] = round(float((p * wrongp + (1 - p) * firea) / den), 4) if den else 0.0
                out.append(row)
    return pd.DataFrame(out)

def choose(grid: pd.DataFrame, datasets=("legion", "vua"), cap: float = WRONG_CAP, tag: str = "") -> dict | None:
    """The most permissive setting (highest recovery; then the higher ratio, lead and key) whose expected wrong-link
    share is at most *cap* in every dataset."""
    ok = grid
    for ds in datasets:
        ok = ok[ok[f"exp_wrong_{ds}{tag}"] <= cap]
    if not len(ok):
        return None
    best = ok.sort_values(["recovery", "ratio", "lead", "min_key"], ascending=[False, False, False, False]).iloc[0]
    return {k: (float(v) if isinstance(v, float) else int(v)) for k, v in best.items()}

def main(g=None, out_dir: Path = OUT) -> dict:
    """Run the check on the link files in *out_dir* and write the curve and the summary there."""
    if g is None:
        g = bs.load_gazetteer(bs.FINAL, include_unconfirmed=False)
    pairs = p1_pairs(out_dir)
    samples = corrupt(pairs)
    rows = rank(samples, g)
    shares = mixing_shares()
    grid = curve(rows, shares, MIXING_0160)
    module = {"ratio": bs.FUZZY_RATIO, "lead": bs.FUZZY_LEAD, "min_key": bs.FUZZY_MIN_KEY}
    at = grid[(grid["ratio"] == module["ratio"]) & (grid["lead"] == module["lead"])
              & (grid["min_key"] == module["min_key"])].iloc[0]
    summary = {"check": f"fuzzy_heldout_check {__version__}", "module": f"birth_settlement {bs.__version__}",
               "pairs": len(pairs), "pairs_used": min(len(pairs), N_PAIRS), "samples": len(samples),
               "samples_kept": int(len(rows)), "seed": SEED, "wrong_cap": WRONG_CAP,
               "samples_by_kind": {str(k): int(v) for k, v in rows["kind"].value_counts().sort_index().items()},
               "mixing_counts": MIXING_COUNTS, "mixing": {k: round(v, 4) for k, v in shares.items()},
               "mixing_0160": {k: round(v, 4) for k, v in MIXING_0160.items()},
               "module_setting": module,
               "module_setting_rates": {k: (float(v) if isinstance(v, float) else int(v)) for k, v in at.items()},
               "module_setting_within_cap": bool(at["exp_wrong_legion"] <= WRONG_CAP and at["exp_wrong_vua"] <= WRONG_CAP),
               "chosen": choose(grid), "chosen_at_0160_mixing": choose(grid, tag="_0160"),
               "curve_file": "fuzzy_heldout_curve.csv"}
    out_dir = Path(out_dir)
    grid.to_csv(out_dir / "fuzzy_heldout_curve.csv", index=False, encoding="utf-8", lineterminator="\n")
    with open(out_dir / "fuzzy_heldout.json", "w", encoding="utf-8", newline="\n") as fh:
        json.dump(summary, fh, ensure_ascii=False, indent=1)
    return summary

"""The held-out check behind ``birth_settlement``'s N3 thresholds (:data:`birth_settlement.FUZZY_RATIO`,
:data:`~birth_settlement.FUZZY_LEAD`, :data:`~birth_settlement.FUZZY_MIN_KEY`).

N3 links a written name the record's 1910 okres does not hold to the one settlement of the okres whose name is a close
spelling of it (rapidfuzz ``ratio`` on space-free keys, a lead over the next settlement, a minimum key length). The
thresholds come from this check (birth_settlement 0.16.0 ran it as scratch; 0.16.1 moved it here, the 0.16.0 checkers'
finding that a constant's justification must be rebuildable from the repo):

1. **pairs** — the P1 exact links (``p1_register`` / ``p1_lexikon``) of the legionnaire and the re-parsed VÚA link files

2. **corruption** — each name with one edit of each kind (a dropped letter, two letters swapped, a gender or plural
   ending, an OCR pair, *Sv.* written out or abbreviated) and one double edit; a corrupted key that is itself a

   share of N3-eligible unmatched rows whose place the okres holds (:data:`MIXING_COUNTS`, the sizing's hand-labelled

#: the sizing's hand-labelled sample of unmatched rows (200 per dataset, 2026-11-17; categories of the row's place):
#: *present* — the okres holds the place under another spelling (SPELL; VÚA also VUA_PARSE, a parse slip of a name the
#: okres holds); *absent* — the okres does not hold it (MULTI, NO1910ROW, UNIDENT, PART, OLDNAME, OUTSIDE). The other
#: categories (a wrong okres, a VÚA alternative, a bare head …) are other rules' and do not enter N3's mix.
MIXING_COUNTS = {
    "legion": {"present": {"SPELL": 77},
               "absent": {"MULTI": 18, "NO1910ROW": 16, "UNIDENT": 12, "PART": 5, "OLDNAME": 3, "OUTSIDE": 1}},
    "vua": {"present": {"SPELL": 36, "VUA_PARSE": 5},
            "absent": {"UNIDENT": 24, "NO1910ROW": 13, "OUTSIDE": 7, "PART": 4, "OLDNAME": 2, "MULTI": 2}}}
#: the shares the 0.16.0 scratch check used (legion 0.58 ≈ 77/132; VÚA 41/88 — an arithmetic slip, the VÚA categories
#: above sum to 93), reported beside the corrected ones
MIXING_0160 = {"legion": 0.58, "vua": 41 / 88}
RATIOS = range(70, 97)
LEADS = range(0, 11)
MIN_KEYS = (4, 5, 6)
ENDINGS = [("ice", "ic"), ("ice", "ce"), ("ice", "ica"), ("any", "ana"), ("any", "an"), ("ov", "ova"), ("ov", "ow"),
           ("ec", "ce"), ("ín", "ina"), ("á", "é"), ("é", "á"), ("y", "a"), ("e", "y"), ("a", "e"), ("ky", "ek"),
           ("ná", "ný"), ("ský", "ská"), ("ice", "ic e")]
OCR = [("f", "b"), ("b", "f"), ("oe", "oc"), ("oc", "oe"), ("rn", "m"), ("m", "rn"), ("cl", "d"), ("li", "h"),
       ("n", "u"), ("u", "n"), ("e", "c"), ("c", "e")]


def mixing_shares(counts: dict = MIXING_COUNTS) -> dict:
    """``{dataset: present / (present + absent)}`` from :data:`MIXING_COUNTS`."""
    out = {}
    for ds, c in counts.items():
        p, a = sum(c["present"].values()), sum(c["absent"].values())
        out[ds] = p / (p + a)
    return out


def p1_pairs(out_dir: Path = OUT) -> list[tuple[str, int, str]]:
    """The distinct ``(written name, 1910 okres, settlement)`` of every P1 exact link of the two link files, sorted."""
    pairs = set()
    for name in LINK_FILES.values():
        d = pd.read_parquet(Path(out_dir) / name, columns=["birth_settlement_match", "birth_settlement_id",
                                                           "birth_okres_id_1910", "birth_settlement_clean"])
        m = (d["birth_settlement_match"].isin(["p1_register", "p1_lexikon"]) & d["birth_settlement_id"].notna()
             & d["birth_okres_id_1910"].notna())
        for c, o, s in zip(d.loc[m, "birth_settlement_clean"], d.loc[m, "birth_okres_id_1910"],
                           d.loc[m, "birth_settlement_id"]):
            if isinstance(c, str):
                pairs.add((c, int(o), s))
    return sorted(pairs)


def corrupt(pairs: list, seed: int = SEED, n: int = N_PAIRS) -> list[tuple]:
    """``[(kind, original, written, okres, settlement)]``: *n* shuffled pairs, each with one edit of each kind and one
    double edit (two random single edits)."""
    rng = random.Random(seed)
    pairs = list(pairs)
    rng.shuffle(pairs)
    pairs = pairs[:n]

    def drop(w):
        i = rng.randrange(len(w))
        return w[:i] + w[i + 1:]

    def swap(w):
        if len(w) < 3:
            return w
        i = rng.randrange(len(w) - 1)
        return w[:i] + w[i + 1] + w[i] + w[i + 2:]

    def ending(w):
        opts = [(a, b) for a, b in ENDINGS if w.endswith(a)]
        if not opts:
            return None
        a, b = rng.choice(opts)
        return w[: len(w) - len(a)] + b

    def sv(w):
        for full in ("Svatý ", "Svatá "):
            if w.startswith(full):
                return "Sv. " + w[6:]
        return "Svatý " + w[4:] if w.startswith("Sv. ") else None

    def ocr(w):
        opts = [(a, b) for a, b in OCR if a in w]
        if not opts:
            return None
        a, b = rng.choice(opts)
        i = w.find(a)
        return w[:i] + b + w[i + len(a):]

    single = {"drop": drop, "swap": swap, "ending": ending, "sv": sv, "ocr": ocr}
    out = []
    for c, o, s in pairs:
        for kind, f in single.items():
            w = f(c)
            if w and w != c:
                out.append((kind, c, w, o, s))
        k1, k2 = rng.sample(["drop", "swap", "ending", "ocr"], 2)
        w = single[k1](c)
        w = single[k2](w) if w else None
        if w and w != c:
            out.append((f"double:{k1}+{k2}", c, w, o, s))
    return out


def rank(samples: list, g) -> pd.DataFrame:
    """Each corrupted name ranked in its okres with the true settlement present (``best``, ``best_sid``, ``second``) and
    removed (``abs_best``, ``abs_second``); names that are a Czech-lands name as written dropped. 0.16.1: a fire on a
    settlement N3's namesake guard (:func:`birth_settlement.fuzzy_namesake_block`) blocks is no fire (``blocked``)."""
    from rapidfuzz import fuzz, process
    rows = []
    for kind, c, w, o, s in samples:
        k = bs.key(w)
        if len(bs.nospace(k)) < 3 or bs._czech_namesakes(k, g):
            continue
        rk = bs.fuzzy_ranking(k, o, g)
        b1 = rk[0] if rk else (0.0, None)
        names, sids = bs._okres_name_list(o, g)
        sc: dict = {}
        for _n, score, i in process.extract(bs.nospace(k), names, scorer=fuzz.ratio, limit=25):
            for x in sids[i]:
                if x != s:
                    sc[x] = max(sc.get(x, 0.0), float(score))
        absent = sorted(sc.items(), key=lambda x: (-x[1], x[0]))
        rows.append({"kind": kind, "orig": c, "written": w, "okres": o, "true": s, "keylen": len(bs.nospace(k)),
                     "best": b1[0], "best_sid": b1[1], "second": rk[1][0] if len(rk) > 1 else 0.0,
                     "blocked": bool(b1[1]) and bs.fuzzy_namesake_block(k, b1[1], g),
                     "abs_best": absent[0][1] if absent else 0.0, "abs_second": absent[1][1] if len(absent) > 1 else 0.0,
                     "abs_blocked": bool(absent) and bs.fuzzy_namesake_block(k, absent[0][0], g)})
    return pd.DataFrame(rows)


def curve(rows: pd.DataFrame, shares: dict, shares_0160: dict | None = None) -> pd.DataFrame:
    """Every setting (:data:`RATIOS` × :data:`LEADS` × :data:`MIN_KEYS`): recovery, the wrong share among fires with the
    target present, the fire rate with it absent, and the expected wrong-link share per dataset (``exp_wrong_<ds>``;
    ``exp_wrong_<ds>_0160`` at the 0.16.0 shares)."""
    R = rows
    out = []
    for ratio in RATIOS:
        for lead in LEADS:
            for mk in MIN_KEYS:
                k = R["keylen"] >= mk
                fp = (R["best"] >= ratio) & ((R["best"] - R["second"]) >= lead) & k & ~R["blocked"]
                ok = fp & (R["best_sid"] == R["true"])
                fa = (R["abs_best"] >= ratio) & ((R["abs_best"] - R["abs_second"]) >= lead) & k & ~R["abs_blocked"]
                firep, wrongp, firea = fp.mean(), (fp & ~ok).mean(), fa.mean()
                row = {"ratio": ratio, "lead": lead, "min_key": mk, "recovery": round(float(ok.mean()), 4),
                       "wrong_present": round(float(wrongp / firep), 4) if firep else 0.0,
                       "fire_absent": round(float(firea), 4)}
                for tag, sh in (("", shares), ("_0160", shares_0160 or {})):
                    for ds, p in sh.items():
                        den = p * firep + (1 - p) * firea
                        row[f"exp_wrong_{ds}{tag}"] = round(float((p * wrongp + (1 - p) * firea) / den), 4) if den else 0.0
                out.append(row)
    return pd.DataFrame(out)


def choose(grid: pd.DataFrame, datasets=("legion", "vua"), cap: float = WRONG_CAP, tag: str = "") -> dict | None:
    """The most permissive setting (highest recovery; then the higher ratio, lead and key) whose expected wrong-link
    share is at most *cap* in every dataset."""
    ok = grid
    for ds in datasets:
        ok = ok[ok[f"exp_wrong_{ds}{tag}"] <= cap]
    if not len(ok):
        return None
    best = ok.sort_values(["recovery", "ratio", "lead", "min_key"], ascending=[False, False, False, False]).iloc[0]
    return {k: (float(v) if isinstance(v, float) else int(v)) for k, v in best.items()}


def main(g=None, out_dir: Path = OUT) -> dict:
    """Run the check on the link files in *out_dir* and write the curve and the summary there."""
    if g is None:
        g = bs.load_gazetteer(bs.FINAL, include_unconfirmed=False)
    pairs = p1_pairs(out_dir)
    samples = corrupt(pairs)
    rows = rank(samples, g)
    shares = mixing_shares()
    grid = curve(rows, shares, MIXING_0160)
    module = {"ratio": bs.FUZZY_RATIO, "lead": bs.FUZZY_LEAD, "min_key": bs.FUZZY_MIN_KEY}
    at = grid[(grid["ratio"] == module["ratio"]) & (grid["lead"] == module["lead"])
              & (grid["min_key"] == module["min_key"])].iloc[0]
    summary = {"check": f"fuzzy_heldout_check {__version__}", "module": f"birth_settlement {bs.__version__}",
               "pairs": len(pairs), "pairs_used": min(len(pairs), N_PAIRS), "samples": len(samples),
               "samples_kept": int(len(rows)), "seed": SEED, "wrong_cap": WRONG_CAP,
               "samples_by_kind": {str(k): int(v) for k, v in rows["kind"].value_counts().sort_index().items()},
               "mixing_counts": MIXING_COUNTS, "mixing": {k: round(v, 4) for k, v in shares.items()},
               "mixing_0160": {k: round(v, 4) for k, v in MIXING_0160.items()},
               "module_setting": module,
               "module_setting_rates": {k: (float(v) if isinstance(v, float) else int(v)) for k, v in at.items()},
               "module_setting_within_cap": bool(at["exp_wrong_legion"] <= WRONG_CAP and at["exp_wrong_vua"] <= WRONG_CAP),
               "chosen": choose(grid), "chosen_at_0160_mixing": choose(grid, tag="_0160"),
               "curve_file": "fuzzy_heldout_curve.csv"}
    out_dir = Path(out_dir)
    grid.to_csv(out_dir / "fuzzy_heldout_curve.csv", index=False, encoding="utf-8", lineterminator="\n")
    with open(out_dir / "fuzzy_heldout.json", "w", encoding="utf-8", newline="\n") as fh:
        json.dump(summary, fh, ensure_ascii=False, indent=1)
    return summary


if __name__ == "__main__":
    print(json.dumps(main(), ensure_ascii=False, indent=1))
