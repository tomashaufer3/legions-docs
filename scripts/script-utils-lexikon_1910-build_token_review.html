How it works

A token that is not on the atlas ([figre-brigade], [bath]) is either a typo, a symbol the atlas lacks, or a mark nobody can read — and only someone looking at the print can say which. This script gathers every such token, cuts the lines of the scan it came from, ranks the atlas tokens it most resembles, and writes a queue the Token Review artifact page reads. The page stores each decision in its own database; --apply writes the decisions back into the sheets.

The queue is source-agnostic on purpose. The same shape will carry the tokens a model pass emits that lexikon_1910_vision.guard_tokens rejects, and a decision that a spelling always means one token becomes a rule the full run can apply without asking again.

What counts as suspect

unknown a bracketed token not on the atlas, in either section bare a word standing where a mark belongs, in section 2a (& scattered 4-49 — the brackets were dropped) challenge a valid token a second reader questions (--challenges) heading a section heading the text layer missed, so its section is filed under the one above (--headings; D-76)

The page times every question while it is on screen and saves the seconds with the decision, so the cost of ruling by hand can be set against a verify call (D-89).

def suspects_in_sheet(path: Path, allowed: set) -> list[dict]:
    """Every suspect token on one sheet, with where it sits."""
    text = path.read_text(encoding="utf-8")
    out: list[dict] = []
    for block in _sheet_sections(text):
        printed: list[int] = []      # municipalities with an answer, in order
        for lineno, line in block["lines"]:
            parsed = _key(line)
            if not parsed or parsed[0] == "__" or not parsed[-1]:
                continue
            muni, seq, _muni_name, settlement, _flag, answer = parsed
            if int(muni) not in printed:
                printed.append(int(muni))
            found = [("unknown", m.group(0), m.start())
                     for m in _BRACKET_RE.finditer(answer)
                     if m.group(0) not in (UNREADABLE, OFF_ATLAS)
                     and lxv.canonical(m.group(0)) not in allowed]
            if block["section"] == "2a":
                found += [("bare", m.group(1), m.start(1))
                          for m in _BARE_RE.finditer(answer)]
            for n, (kind, token, at) in enumerate(found):
                out.append({
                    "kind": kind, "token": token, "at": at,
                    "section": "parts" if block["section"] == "2a" else "offices",
                    "muni": int(muni), "seq": seq, "settlement": settlement,
                    "answer": answer, "line": lineno, "n": n,
                    "images": block["images"], "printed": printed,
                })
        # `printed` is complete only once the block is read; the dicts share it.
    return out

def candidates(token: str, vocabulary: list[str], n: int = 5) -> list[dict]:
    """The atlas tokens a misspelling most resembles, best first."""
    bare = token.strip("[]").lower()
    scored = sorted(((difflib.SequenceMatcher(None, bare, v.strip("[]")).ratio(), v)
                     for v in vocabulary), reverse=True)
    return [{"token": v, "score": round(s, 2)} for s, v in scored[:n]]

def text_lines(image) -> list[tuple[int, int, int]]:
    """``[(top, bottom, first_ink_x)]`` for each printed line of a section crop.

    Read off the ink, not the text layer: the crops on disk were cut by whatever
    geometry was current when the sheet was generated, and the image is the one
    thing the transcriber and this queue are guaranteed to share. Columns inked
    over most of the height are the printed frame and are ignored.
    """
    import numpy as np

    ink = _text_ink(image)
    height, width = ink.shape
    rows = ink.sum(axis=1) >= 3
    lines, start, gap = [], None, 0
    for y, on in enumerate(rows):
        if on:
            start = y if start is None else start
            gap = 0
        elif start is not None:
            gap += 1
            if gap >= 2:
                lines.append((start, y - gap))
                start, gap = None, 0
    if start is not None:
        lines.append((start, height - 1))
    def is_rule(top: int, bottom: int) -> bool:
        """A printed rule: one row of it covers nearly every inked column."""
        band = ink[top:bottom + 1]
        return band.sum(axis=1).max() > 0.8 * max(1, (band.sum(axis=0) > 0).sum())

    lines = [(top, bottom) for top, bottom in lines if not is_rule(top, bottom)]
    if not lines:
        return []
    median = sorted(b - t for t, b in lines)[len(lines) // 2]
    out = []
    for top, bottom in lines:
        if bottom - top < 0.45 * median:       # a stray dot or a descender
            continue
        cols = np.where(ink[top:bottom + 1].sum(axis=0) >= 2)[0]
        if len(cols):
            out.append((top, bottom, int(cols[0])))
    return out

def municipality_bands(images: list, printed: list[int]) -> dict:
    """``muni -> [(image_index, top, bottom), …]``, or ``{}`` when the count is off.

    A printed line whose first ink is in the key column opens a municipality.
    The sheet already says which municipalities are printed and in what order,
    so the k-th such line is the k-th of them — and when the two
    counts disagree (a table spills into the crop, a key the eye cannot place)
    nothing is guessed: the page shows the whole section instead. A count that
    agrees by coincidence still mislocates, which is why the page always keeps
    the whole section one click away and the crop shows its own key number. An
    entry runs
    on into the next image until that image's first key, so a municipality
    split by the column break comes back as two pieces.
    """
    events = []                                  # (index, y, opens_a_key)
    for index, image in enumerate(images):
        lines = text_lines(image)
        if lines:
            ink = _text_ink(image)
            left = min(x for _t, _b, x in lines)
            cut = left + KEY_COLUMN_SHARE * image.width
            span = int(KEY_INK_SHARE * image.width)
            # A key is digits: several inked columns where the line starts. A
            # sliver of the frame beside a centred heading starts in the key
            # column too, but is a column or two wide (scan 0136, where the
            # heading counted as the first key and shifted every entry by one).
            events += [(index, top, True) for top, bottom, x in lines
                       if x <= cut
                       and (ink[top:bottom + 1, x:x + span].sum(axis=0) > 0).sum() >= KEY_MIN_COLUMNS]
        events.append((index, image.height, False))      # foot of this image
    if sum(1 for e in events if e[2]) != len(printed):
        return {}
    bands: dict = {}
    k = -1
    for i, (index, top, opens) in enumerate(events):
        if not opens:
            continue
        k += 1
        pieces, start = [], (index, max(0, top - 10))
        for later_index, later_top, later_opens in events[i + 1:]:
            pieces.append((start[0], start[1],
                           max(start[1], later_top - (10 if later_opens else 0))))
            if later_opens:
                break
            start = (later_index + 1, 0)         # carry on at the next image
            if later_index + 1 >= len(images):
                break
        bands[printed[k]] = [p for p in pieces if p[2] - p[1] > 12]
    return bands

def challenges(spec: Path) -> list[tuple]:
    """Readings a second reader questions although every token is on the atlas.

    ``spec`` is a JSON list of ``{sheet, section, muni, seq, token, suggestion,
    issue, note}``. ``token`` is the exact text to change — a span such as
    `[school] exp.`, a misspelt name, a number — and ``""`` for an answer that
    is missing altogether; ``suggestion`` is what replaces it, ``""`` to delete.
    ``section`` may be ``table``, with the row id (`r12`) in the note. A line
    that no longer carries the token is reported and skipped: the question has
    already been settled.
    """
    out = []
    for k, c in enumerate(json.loads(spec.read_text(encoding="utf-8"))):
        path = ROOT / c["sheet"]
        text = path.read_text(encoding="utf-8")
        found = None
        base = {"kind": "challenge", "token": c["token"], "section": c["section"],
                "issue": c.get("issue", ""), "n": 0, "k": k,
                "column": c.get("column", ""), "bbox": c.get("bbox")}
        if c["section"] == "table":
            # A numbered row is found by its id (`r12`, in the note); an
            # unnumbered header or total row by the text of its name cell.
            row = re.search(r"\br\d+\b", c.get("note", ""))
            label = row.group(0) if row else c.get("row_name", "")
            # A sheet with two courts numbers each table from r1, so an id can
            # occur twice: take the row that carries the token and the number.
            printed_no = str(c["muni"] if str(c["seq"]) == "0" else c["seq"])
            hit = (_table_row(text, label, c["token"], printed_no) if row
                   else _named_row(text, label) if label else None)
            if hit and c["token"] in hit[1]:
                found = {**base, "muni": c["muni"], "seq": str(c["seq"]),
                         "settlement": f"table row {label}",
                         "answer": hit[1], "line": hit[0],
                         "images": [f"{path.stem}_table.png"], "printed": []}
        for block in ([] if c["section"] == "table" else _sheet_sections(text)):
            if block["section"] != ("2a" if c["section"] == "parts" else "2b"):
                continue
            printed = []
            for lineno, line in block["lines"]:
                parsed = _key(line)
                if not parsed or parsed[0] == "__":
                    continue
                muni, seq, _m, settlement, _f, answer = parsed
                if answer and int(muni) not in printed:
                    printed.append(int(muni))
                if ((int(muni), seq) == (c["muni"], str(c["seq"]))
                        and c["token"] in answer
                        and (answer or not c["token"])):
                    found = {**base, "muni": int(muni), "seq": seq,
                             "settlement": settlement, "answer": answer,
                             "line": lineno, "images": block["images"],
                             "printed": printed}
        if found is None:
            print(f"  !! {c['sheet']} {c['section']} {c['muni']},{c['seq']}: "
                  f"{c['token']!r} is no longer there - skipped")
            continue
        out.append((path, found, c["suggestion"], c.get("note", "")))
    return out

def box_items(paths: list[Path], out_dir: Path, tokens=BOX_TOKENS) -> list[dict]:
    """One question per entry and glyph: click every such mark the entry prints.

    The gold says how many of each mark an entry holds; the reviewer clicks
    them on the entry's crop, or on the whole section, and
    ``build_glyph_bank.py`` cuts each glyph from the section image at the click.
    ``pieces`` maps the stacked crop back to the section images: ``[image, top,
    bottom]`` in stacking order; ``sections`` are the whole images, a click on
    one of which is stored as ``[x, y, image]``.
    """
    from PIL import Image

    wanted = set(tokens)
    out = []
    for path in paths:
        text = path.read_text(encoding="utf-8")
        for block in _sheet_sections(text):
            section = "parts" if block["section"] == "2a" else "offices"
            entries: dict[int, dict] = {}
            for _lineno, line in block["lines"]:
                parsed = _key(line)
                if not parsed or parsed[0] == "__" or not parsed[-1]:
                    continue
                muni, _seq, _name, settlement, _flag, answer = parsed
                entry = entries.setdefault(int(muni), {"answer": [], "settlement": settlement,
                                                       "count": Counter()})
                entry["answer"].append(answer)
                entry["count"].update(t for t in _BRACKET_RE.findall(answer) if t in wanted)
            if not any(e["count"] for e in entries.values()):
                continue
            names = [n for n in block["images"] if (path.parent / n).exists()]
            images = [Image.open(path.parent / n) for n in names]
            bands = municipality_bands(images, list(entries))
            # The whole section, one click away and clickable too: a crop the
            # count happened to place on the wrong entry is caught on the page.
            full = []
            for name, image in zip(names, images):
                jpg = f"img/{Path(name).stem}.jpg"
                if not (out_dir / jpg).exists():
                    image.convert("RGB").save(out_dir / jpg, quality=80)
                full.append({"src": jpg, "image": name, "size": list(image.size)})
            for muni, entry in entries.items():
                if not entry["count"]:
                    continue
                pieces = ([(names[i], top, bottom) for i, top, bottom in bands[muni]]
                          if muni in bands else [(n, 0, im.height) for n, im in zip(names, images)])
                for token, n in sorted(entry["count"].items()):
                    item_id = f"{path.stem}-{section}-{muni}-box-{token.strip('[]')}"
                    crop = f"img/{item_id}.jpg"
                    stacked = _stack([images[names.index(name)].crop(
                        (0, top, images[names.index(name)].width, bottom))
                        for name, top, bottom in pieces])
                    stacked.save(out_dir / crop, quality=88)
                    glyph = SYMBOLS / f"{token.strip('[]')}.png"
                    if glyph.exists():
                        shutil.copy(glyph, out_dir / "glyph" / glyph.name)
                    out.append({
                        "id": item_id, "source": f"gold:{path.parent.name}",
                        "sheet": str(path.relative_to(ROOT)).replace("\\", "/"),
                        "scan": path.stem, "section": section, "kind": "box",
                        "muni": muni, "seq": "", "settlement": entry["settlement"],
                        "line": 0, "token": token, "expected": n,
                        "answer": "; ".join(entry["answer"]), "suggestion": "",
                        "note": "", "issue": "", "column": "", "candidates": [],
                        "glyph": f"glyph/{glyph.name}" if glyph.exists() else "",
                        "pieces": pieces, "located": muni in bands,
                        "crop": crop, "crop_size": list(stacked.size),
                        "full": [f["src"] for f in full], "sections": full,
                    })
    print(f"  {len(out)} glyph-box questions, "
          f"{sum(i['expected'] for i in out)} glyphs")
    return out

def build(paths: list[Path], out_dir: Path, challenge_spec: Path | None = None,
          heading_spec: Path | None = None, box_paths: list[Path] | None = None) -> dict:
    """Queue every suspect token, plus any challenges, keeping earlier items.

    Items already in ``queue.json`` stay: once a decision is applied its token
    is gone from the sheet, but the page keeps showing what was decided.
    """
    abbreviations = lxin.load_abbreviations(ABBREVIATIONS, lang="cs")
    allowed = lxv.vocabulary(abbreviations) | TOPOGRAPHIC
    vocabulary = sorted(v for v in allowed if v.startswith("["))
    (out_dir / "img").mkdir(parents=True, exist_ok=True)
    (out_dir / "glyph").mkdir(parents=True, exist_ok=True)

    fresh = [_item(path, s, out_dir, vocabulary)
             for path in paths for s in suspects_in_sheet(path, allowed)]
    if challenge_spec:
        fresh += [_item(path, s, out_dir, vocabulary, suggestion, note)
                  for path, s, suggestion, note in challenges(challenge_spec)]
    if heading_spec:
        fresh += headings(heading_spec, out_dir)
    if box_paths:
        fresh += box_items(box_paths, out_dir)
    queue_path = out_dir / "queue.json"
    earlier = (json.loads(queue_path.read_text("utf-8"))["items"]
               if queue_path.exists() else [])
    ids = {i["id"] for i in fresh}
    items = [i for i in earlier if i["id"] not in ids] + fresh
    # Every atlas glyph travels with the queue, so a reviewer who picks a token
    # outside the ranked five still sees what it looks like.

def apply(decisions_path: Path, queue_path: Path) -> int:
    """Rewrite each decided token in its sheet, refusing a line that moved.

    ``decisions.json`` is the page's ``decisions`` collection as a list of
    ``{id, decision, replacement}``; ``replace`` writes ``replacement``,
    ``unreadable`` writes ``[?]``, ``off_atlas`` writes ``[!]`` and ``keep``
    leaves the token alone.
    """
    queue = {i["id"]: i for i in json.loads(queue_path.read_text("utf-8"))["items"]}
    decisions = json.loads(decisions_path.read_text("utf-8"))
    heading_rows = [_heading_row(queue[d["id"]], d) for d in decisions
                    if d["id"] in queue and queue[d["id"]]["kind"] == "heading"
                    and d.get("decision")]
    if heading_rows:
        write_heading_rulings(heading_rows)
    by_sheet: dict[str, list] = {}
    for d in decisions:
        item = queue.get(d["id"])
        if item is not None and item["kind"] == "heading":
            continue
        if item is None or d.get("decision") in (None, "", "keep"):
            continue
        if d["decision"] == "replace":
            new = d.get("replacement", "")       # "" deletes, for a challenge
        else:
            new = {"unreadable": UNREADABLE, "off_atlas": OFF_ATLAS}.get(
                d["decision"], None)
        if new is None or (new == "" and item["kind"] != "challenge"):
            continue
        if item["section"] == "table" and not item.get("column"):
            # A table cell is not unique on its row (`.` and `—` repeat), so
            # without its column it has to be applied by a person.
            print(f"  .. {item['id']}: table cell {item['token']!r} -> {new!r} "
                  f"— apply by hand ({item['settlement']})")
            continue
        by_sheet.setdefault(item["sheet"], []).append((item, new))
    for sheet, edits in by_sheet.items():
        path = ROOT / sheet
        raw = path.read_bytes().decode("utf-8")
        newline = "\r\n" if "\r\n" in raw else "\n"
        lines = raw.split(newline)
        for item, new in edits:
            line = lines[item["line"] - 1]
            old = item["token"]
            if item["section"] == "table":
                lines[item["line"] - 1] = _set_cell(line, item, new)
                continue
            if item["answer"] not in line:
                # An earlier decision on the same line already changed it; the
                # token itself can still be placed if it occurs exactly once.
                if old and line.count(old) == 1:
                    lines[item["line"] - 1] = line.replace(old, new, 1)
                    print(f"  {item['id']}: {old!r} -> {new!r}")
                else:
                    print(f"  !! {item['id']}: line {item['line']} changed since "
                          f"the queue was built — skipped")
                continue
            if not item["answer"]:                       # a missing answer
                lines[item["line"] - 1] = line.rstrip() + " " + new
            else:

                lines[item["line"] - 1] = line.replace(item["answer"], answer, 1)
            print(f"  {item['id']}: {old!r} -> {new!r}")
        path.write_bytes(newline.join(lines).encode("utf-8"))
    return 0

A token that is not on the atlas (`[figre-brigade]`, `[bath]`) is either a typo,
a symbol the atlas lacks, or a mark nobody can read — and only someone looking
at the print can say which. This script gathers every such token, cuts the lines
of the scan it came from, ranks the atlas tokens it most resembles, and writes a
queue the *Token Review* artifact page reads. The page stores each decision in
its own database; ``--apply`` writes the decisions back into the sheets.

The queue is source-agnostic on purpose. The same shape will carry the tokens a
model pass emits that ``lexikon_1910_vision.guard_tokens`` rejects, and a
decision that a spelling *always* means one token becomes a rule the full run

``unknown``     a bracketed token not on the atlas, in either section
``bare``        a word standing where a mark belongs, in section 2a

def _sheet_sections(text: str) -> list[dict]:
    """``[{section, images, lines: [(lineno, line)]}]`` for the 2a/2b ledgers."""
    out: list[dict] = []
    current = None
    in_ledger = False
    # The German sheets name one image for the whole lower half (`image:
    # `0479_lower.png``) above 2a and 2b, where the Czech ones list each
    # section's own images; a section that lists none takes that one.
    part_images: list[str] = []
    for lineno, line in enumerate(text.splitlines(), 1):
        heading = _SECTION_RE.match(line)
        if heading:
            current = {"section": heading.group(1), "images": list(part_images),
                       "lines": []}
            out.append(current)
            continue
        # Ledger and HTML comments open `#   …` too; only a numbered top-level
        # heading (`# 3 — …`) closes the section.
        if not in_ledger and re.match(r"^# \d+ —", line):
            current = None
            part_images = []
        single = _PART_IMAGE_RE.match(line)
        if single and current is None:
            part_images = [single.group(1)]
        if current is None:
            continue
        images = _IMAGES_RE.match(line)
        if images:
            current["images"] = re.findall(r"`([^`]+)`", images.group(1))
        if line.startswith("```"):
            in_ledger = line.startswith("```ledger")
            continue
        if in_ledger:
            current["lines"].append((lineno, line))
    return out


def suspects_in_sheet(path: Path, allowed: set) -> list[dict]:
    """Every suspect token on one sheet, with where it sits."""
    text = path.read_text(encoding="utf-8")
    out: list[dict] = []
    for block in _sheet_sections(text):
        printed: list[int] = []      # municipalities with an answer, in order
        for lineno, line in block["lines"]:
            parsed = _key(line)
            if not parsed or parsed[0] == "__" or not parsed[-1]:
                continue
            muni, seq, _muni_name, settlement, _flag, answer = parsed
            if int(muni) not in printed:
                printed.append(int(muni))
            found = [("unknown", m.group(0), m.start())
                     for m in _BRACKET_RE.finditer(answer)
                     if m.group(0) not in (UNREADABLE, OFF_ATLAS)
                     and lxv.canonical(m.group(0)) not in allowed]
            if block["section"] == "2a":
                found += [("bare", m.group(1), m.start(1))
                          for m in _BARE_RE.finditer(answer)]
            for n, (kind, token, at) in enumerate(found):
                out.append({
                    "kind": kind, "token": token, "at": at,
                    "section": "parts" if block["section"] == "2a" else "offices",
                    "muni": int(muni), "seq": seq, "settlement": settlement,
                    "answer": answer, "line": lineno, "n": n,
                    "images": block["images"], "printed": printed,
                })
        # `printed` is complete only once the block is read; the dicts share it.
    return out


def candidates(token: str, vocabulary: list[str], n: int = 5) -> list[dict]:
    """The atlas tokens a misspelling most resembles, best first."""

def _longest_run(column) -> int:
    """Length of the longest unbroken run of True in a boolean column."""
    import numpy as np

    padded = np.concatenate(([0], column.astype(np.int8), [0]))
    edges = np.flatnonzero(np.diff(padded))
    return int((edges[1::2] - edges[::2]).max()) if len(edges) else 0


def _text_ink(image):
    """Dark pixels of a crop, with the frame, rules and gutter blanked out."""
    import numpy as np

    ink = np.asarray(image.convert("L")) < 100
    height, width = ink.shape
    # A vertical rule or the dark band of the gutter: any column with an
    # unbroken ink run taller than ~two printed lines. Partial ones count too —
    # the frame often shows over only part of a crop.
    rule = np.array([_longest_run(ink[:, x]) > 120 for x in range(width)])
    for shift in range(-4, 5):                      # the rule's anti-aliased edge
        ink[:, np.roll(rule, shift)] = False
    # The outer 1.8 % is gutter and frame on every crop; the key column starts
    # beyond it (2.4 % on the tightest crop measured, scan 0286).
    ink[:, :int(0.018 * width)] = False
    return ink


def text_lines(image) -> list[tuple[int, int, int]]:
    """``[(top, bottom, first_ink_x)]`` for each printed line of a section crop.

    Read off the ink, not the text layer: the crops on disk were cut by whatever
    geometry was current when the sheet was generated, and the image is the one
    thing the transcriber and this queue are guaranteed to share. Columns inked
    over most of the height are the printed frame and are ignored.
    """
    import numpy as np

    ink = _text_ink(image)
    height, width = ink.shape
    rows = ink.sum(axis=1) >= 3
    lines, start, gap = [], None, 0
    for y, on in enumerate(rows):
        if on:
            start = y if start is None else start
            gap = 0
        elif start is not None:
            gap += 1
            if gap >= 2:
                lines.append((start, y - gap))
                start, gap = None, 0
    if start is not None:
        lines.append((start, height - 1))
    def is_rule(top: int, bottom: int) -> bool:
        """A printed rule: one row of it covers nearly every inked column."""
        band = ink[top:bottom + 1]
        return band.sum(axis=1).max() > 0.8 * max(1, (band.sum(axis=0) > 0).sum())

    lines = [(top, bottom) for top, bottom in lines if not is_rule(top, bottom)]
    if not lines:
        return []
    median = sorted(b - t for t, b in lines)[len(lines) // 2]
    out = []
    for top, bottom in lines:
        if bottom - top < 0.45 * median:       # a stray dot or a descender
            continue
        cols = np.where(ink[top:bottom + 1].sum(axis=0) >= 2)[0]
        if len(cols):
            out.append((top, bottom, int(cols[0])))
    return out


def municipality_bands(images: list, printed: list[int]) -> dict:
    """``muni -> [(image_index, top, bottom), …]``, or ``{}`` when the count is off.

    A printed line whose first ink is in the key column opens a municipality.
    The sheet already says which municipalities are printed and in what order,
    so the k-th such line is the k-th of them — and when the two
    counts disagree (a table spills into the crop, a key the eye cannot place)
    nothing is guessed: the page shows the whole section instead. A count that
    agrees by coincidence still mislocates, which is why the page always keeps
    the whole section one click away and the crop shows its own key number. An
    entry runs
    on into the next image until that image's first key, so a municipality
    split by the column break comes back as two pieces.
    """
    events = []                                  # (index, y, opens_a_key)
    for index, image in enumerate(images):
        lines = text_lines(image)
        if lines:
            ink = _text_ink(image)
            left = min(x for _t, _b, x in lines)
            cut = left + KEY_COLUMN_SHARE * image.width
            span = int(KEY_INK_SHARE * image.width)
            # A key is digits: several inked columns where the line starts. A
            # sliver of the frame beside a centred heading starts in the key
            # column too, but is a column or two wide (scan 0136, where the
            # heading counted as the first key and shifted every entry by one).
            events += [(index, top, True) for top, bottom, x in lines
                       if x <= cut
                       and (ink[top:bottom + 1, x:x + span].sum(axis=0) > 0).sum() >= KEY_MIN_COLUMNS]
        events.append((index, image.height, False))      # foot of this image
    if sum(1 for e in events if e[2]) != len(printed):
        return {}
    bands: dict = {}
    k = -1
    for i, (index, top, opens) in enumerate(events):
        if not opens:
            continue
        k += 1
        pieces, start = [], (index, max(0, top - 10))
        for later_index, later_top, later_opens in events[i + 1:]:
            pieces.append((start[0], start[1],
                           max(start[1], later_top - (10 if later_opens else 0))))
            if later_opens:
                break
            start = (later_index + 1, 0)         # carry on at the next image
            if later_index + 1 >= len(images):
                break
        bands[printed[k]] = [p for p in pieces if p[2] - p[1] > 12]
    return bands


def _stack(pieces: list):
    """The pieces of one entry, one under the other, on the paper's own tone."""
    from PIL import Image

    width = max(p.width for p in pieces)
    out = Image.new("RGB", (width, sum(p.height for p in pieces)), (214, 208, 196))
    y = 0
    for piece in pieces:
        out.paste(piece.convert("RGB"), (0, y))
        y += piece.height
    return out

def _item(path: Path, s: dict, out_dir: Path, vocabulary: list[str],
          suggestion: str = "", note: str = "") -> dict:
    """One queue entry: the sheet line, its scan crop and the ranked tokens."""
    from PIL import Image

    scan = path.stem
    images = [Image.open(path.parent / name) for name in s["images"]
              if (path.parent / name).exists()]
    item_id = (f"{scan}-{s['section']}-{s['muni']}-{s['seq']}-{s['n']}"
               + (f"-c{s['k']}" if s["kind"] == "challenge" else "")
               ).replace("?", "q").replace("+", "p")
    band = municipality_bands(images, s["printed"]).get(s["muni"])
    crop = ""
    if s.get("bbox") and images:
        # A table cell: the reader gives the box (row band, with the column
        # headings it needs to be read) in the table image's own pixels.
        crop = f"img/{item_id}.jpg"
        images[0].convert("RGB").crop(tuple(s["bbox"])).save(out_dir / crop,
                                                             quality=88)
    elif band:
        crop = f"img/{item_id}.jpg"
        _stack([images[i].crop((0, top, images[i].width, bottom))
                for i, top, bottom in band]).save(out_dir / crop, quality=88)
    full = []
    # JPEG, not the sheets' PNG: an enlarged section is 2-4 MB as PNG and the
    # page has to stay well under its 16 MB budget.
    for name in s["images"]:
        if (path.parent / name).exists():
            jpg = f"img/{Path(name).stem}.jpg"
            if not (out_dir / jpg).exists():
                Image.open(path.parent / name).convert("RGB").save(
                    out_dir / jpg, quality=80)
            full.append(jpg)
    if s["kind"] == "challenge":
        # The reviewer chooses between two readings, so the proposal leads
        # ("Keep as written" is the other); a token proposal brings its
        # look-alikes, a name or a number only itself.
        ranked = [{"token": suggestion, "score": 1.0}]
        if re.fullmatch(r"\[[^\]]+\]", suggestion):
            ranked += [c for c in candidates(suggestion, vocabulary, 4)
                       if c["token"] != suggestion]
    else:
        probe = s["token"] if s["kind"] == "unknown" else f"[{s['token']}]"
        ranked = candidates(probe, vocabulary)
        suggestion = (ranked[0]["token"]
                      if ranked and ranked[0]["score"] >= 0.6 else "")
    for c in ranked:
        glyph = SYMBOLS / f"{c['token'].strip('[]')}.png"
        if glyph.exists():
            shutil.copy(glyph, out_dir / "glyph" / glyph.name)
            c["glyph"] = f"glyph/{glyph.name}"
    print(f"  {item_id:30s} {s['token']:18s} -> {suggestion or '?':18s} "
          f"{'line crop' if crop else 'whole section'}")
    return {
        "id": item_id, "source": f"gold:{path.parent.name}",
        "sheet": str(path.relative_to(ROOT)).replace("\\", "/"),
        "scan": scan, "section": s["section"], "kind": s["kind"],
        "muni": s["muni"], "seq": s["seq"], "settlement": s["settlement"],
        "line": s["line"], "token": s["token"], "answer": s["answer"],
        "suggestion": suggestion, "note": note, "issue": s.get("issue", ""),
        "column": s.get("column", ""),
        "candidates": ranked, "crop": crop, "full": full,
    }


def _table_row(text: str, row: str, token: str = "",
               printed_no: str = "") -> tuple[int, str] | None:
    """``(line number, row text)`` of table row ``r12`` on a sheet.

    Where the id occurs in more than one court's table, the row whose printed
    number is ``printed_no`` and which still carries ``token`` wins.
    """
    hits = [(lineno, line.strip())
            for lineno, line in enumerate(text.splitlines(), 1)
            if re.match(rf"^\|\s*{re.escape(row)}\s*\|", line.strip())]
    for lineno, line in hits:
        cells = [cell.strip() for cell in line.split("|")]
        if token in line and (not printed_no or cells[2] == printed_no):
            return lineno, line
    return hits[0] if len(hits) == 1 else None


def _named_row(text: str, name: str) -> tuple[int, str] | None:
    """``(line number, row text)`` of the table row whose name cell holds ``name``."""
    for lineno, line in enumerate(text.splitlines(), 1):
        cells = line.strip().split("|")
        if line.strip().startswith("|") and len(cells) > 3 and name in cells[3]:
            return lineno, line.strip()
    return None


def challenges(spec: Path) -> list[tuple]:
    """Readings a second reader questions although every token is on the atlas.

    ``spec`` is a JSON list of ``{sheet, section, muni, seq, token, suggestion,
    issue, note}``. ``token`` is the exact text to change — a span such as
    `[school] exp.`, a misspelt name, a number — and ``""`` for an answer that
    is missing altogether; ``suggestion`` is what replaces it, ``""`` to delete.
    ``section`` may be ``table``, with the row id (`r12`) in the note. A line
    that no longer carries the token is reported and skipped: the question has
    already been settled.
    """
    out = []
    for k, c in enumerate(json.loads(spec.read_text(encoding="utf-8"))):
        path = ROOT / c["sheet"]
        text = path.read_text(encoding="utf-8")
        found = None
        base = {"kind": "challenge", "token": c["token"], "section": c["section"],
                "issue": c.get("issue", ""), "n": 0, "k": k,
                "column": c.get("column", ""), "bbox": c.get("bbox")}
        if c["section"] == "table":
            # A numbered row is found by its id (`r12`, in the note); an
            # unnumbered header or total row by the text of its name cell.
            row = re.search(r"\br\d+\b", c.get("note", ""))
            label = row.group(0) if row else c.get("row_name", "")
            # A sheet with two courts numbers each table from r1, so an id can
            # occur twice: take the row that carries the token and the number.
            printed_no = str(c["muni"] if str(c["seq"]) == "0" else c["seq"])
            hit = (_table_row(text, label, c["token"], printed_no) if row
                   else _named_row(text, label) if label else None)
            if hit and c["token"] in hit[1]:
                found = {**base, "muni": c["muni"], "seq": str(c["seq"]),
                         "settlement": f"table row {label}",
                         "answer": hit[1], "line": hit[0],
                         "images": [f"{path.stem}_table.png"], "printed": []}
        for block in ([] if c["section"] == "table" else _sheet_sections(text)):
            if block["section"] != ("2a" if c["section"] == "parts" else "2b"):
                continue
            printed = []
            for lineno, line in block["lines"]:
                parsed = _key(line)
                if not parsed or parsed[0] == "__":
                    continue
                muni, seq, _m, settlement, _f, answer = parsed
                if answer and int(muni) not in printed:
                    printed.append(int(muni))
                if ((int(muni), seq) == (c["muni"], str(c["seq"]))
                        and c["token"] in answer
                        and (answer or not c["token"])):
                    found = {**base, "muni": int(muni), "seq": seq,
                             "settlement": settlement, "answer": answer,
                             "line": lineno, "images": block["images"],
                             "printed": printed}
        if found is None:
            print(f"  !! {c['sheet']} {c['section']} {c['muni']},{c['seq']}: "
                  f"{c['token']!r} is no longer there - skipped")
            continue
        out.append((path, found, c["suggestion"], c.get("note", "")))
    return out


def headings(spec: Path, out_dir: Path) -> list[dict]:
    """Headings the text layer missed, each cut from the scan for a ruling.

    ``spec`` is a JSON list of ``{scan, region, top, bottom, filed_kind,
    filed_court, court, note}``: the gap above a carried section's first line
    (``measure_apparatus_coverage.py``), what the pipeline files that section
    as now, and Claude's reading of the court, which the reviewer confirms or
    corrects. The crop is the column's half of the page, so no column
    geometry is needed.
    """
    from utils.lexikon_1910 import lexikon_1910 as lx

    out = []
    for h in json.loads(spec.read_text(encoding="utf-8")):
        stem = h["scan"]
        page = lx.read_alto(CZ_PAGES / f"{stem}.alto.xml")
        image = lx.render_page(CZ_PDF, int(stem), page.width).convert("RGB")
        # The binding gutter makes the columns unequal, so each side takes a
        # share past the middle rather than an exact half.
        left, right = ((0, round(image.width * COLUMN_SHARE)) if h["region"] == "left"
                       else (round(image.width * (1 - COLUMN_SHARE)), image.width))
        item_id = f"{stem}-heading-{h['region']}-{h['top']}"
        crop, full = f"img/{item_id}.jpg", f"img/{item_id}-context.jpg"
        image.crop((left, max(0, h["top"] - HEADING_PAD[0]), right,
                    min(image.height, h["bottom"] + HEADING_PAD[1]))
                   ).save(out_dir / crop, quality=90)
        image.crop((left, max(0, h["top"] - CONTEXT_PAD), right,
                    min(image.height, h["bottom"] + CONTEXT_PAD))
                   ).save(out_dir / full, quality=82)
        print(f"  {item_id:30s} heading -> {h['court']}")
        out.append({
            "id": item_id, "source": "geometry:missed-heading", "sheet": "",
            "scan": stem, "section": "heading", "kind": "heading",
            "muni": "", "seq": "", "settlement": "", "line": h["top"],
            "region": h["region"], "token": h["court"],
            "answer": f"Části osad v soudním okresu {h['court']}:",
            "suggestion": h["court"], "note": h.get("note", ""), "issue": "",
            "column": "", "filed_kind": h["filed_kind"],
            "filed_court": h["filed_court"], "candidates": [],
            "crop": crop, "full": [full],
        })
    return out


#: The glyphs test A asks an image encoder to tell apart: the fill pairs and
#: their neighbours (SYMBOL_ACCURACY_PLAN.md, "Two tests come first").
BOX_TOKENS = ("[castle]", "[poorhouse]", "[ruin]", "[solitary]",
              "[rail-station]", "[rail-stop]", "[gamekeeper]")


def box_items(paths: list[Path], out_dir: Path, tokens=BOX_TOKENS) -> list[dict]:
    """One question per entry and glyph: click every such mark the entry prints.

    The gold says how many of each mark an entry holds; the reviewer clicks
    them on the entry's crop, or on the whole section, and
    ``build_glyph_bank.py`` cuts each glyph from the section image at the click.
    ``pieces`` maps the stacked crop back to the section images: ``[image, top,
    bottom]`` in stacking order; ``sections`` are the whole images, a click on
    one of which is stored as ``[x, y, image]``.
    """
    from PIL import Image

    wanted = set(tokens)
    out = []
    for path in paths:
        text = path.read_text(encoding="utf-8")
        for block in _sheet_sections(text):
            section = "parts" if block["section"] == "2a" else "offices"
            entries: dict[int, dict] = {}
            for _lineno, line in block["lines"]:
                parsed = _key(line)
                if not parsed or parsed[0] == "__" or not parsed[-1]:
                    continue
                muni, _seq, _name, settlement, _flag, answer = parsed
                entry = entries.setdefault(int(muni), {"answer": [], "settlement": settlement,
                                                       "count": Counter()})
                entry["answer"].append(answer)
                entry["count"].update(t for t in _BRACKET_RE.findall(answer) if t in wanted)
            if not any(e["count"] for e in entries.values()):
                continue
            names = [n for n in block["images"] if (path.parent / n).exists()]
            images = [Image.open(path.parent / n) for n in names]
            bands = municipality_bands(images, list(entries))
            # The whole section, one click away and clickable too: a crop the
            # count happened to place on the wrong entry is caught on the page.
            full = []
            for name, image in zip(names, images):
                jpg = f"img/{Path(name).stem}.jpg"
                if not (out_dir / jpg).exists():
                    image.convert("RGB").save(out_dir / jpg, quality=80)
                full.append({"src": jpg, "image": name, "size": list(image.size)})
            for muni, entry in entries.items():
                if not entry["count"]:
                    continue
                pieces = ([(names[i], top, bottom) for i, top, bottom in bands[muni]]
                          if muni in bands else [(n, 0, im.height) for n, im in zip(names, images)])
                for token, n in sorted(entry["count"].items()):
                    item_id = f"{path.stem}-{section}-{muni}-box-{token.strip('[]')}"
                    crop = f"img/{item_id}.jpg"
                    stacked = _stack([images[names.index(name)].crop(
                        (0, top, images[names.index(name)].width, bottom))
                        for name, top, bottom in pieces])
                    stacked.save(out_dir / crop, quality=88)
                    glyph = SYMBOLS / f"{token.strip('[]')}.png"
                    if glyph.exists():
                        shutil.copy(glyph, out_dir / "glyph" / glyph.name)
                    out.append({
                        "id": item_id, "source": f"gold:{path.parent.name}",
                        "sheet": str(path.relative_to(ROOT)).replace("\\", "/"),
                        "scan": path.stem, "section": section, "kind": "box",
                        "muni": muni, "seq": "", "settlement": entry["settlement"],
                        "line": 0, "token": token, "expected": n,
                        "answer": "; ".join(entry["answer"]), "suggestion": "",
                        "note": "", "issue": "", "column": "", "candidates": [],
                        "glyph": f"glyph/{glyph.name}" if glyph.exists() else "",
                        "pieces": pieces, "located": muni in bands,
                        "crop": crop, "crop_size": list(stacked.size),
                        "full": [f["src"] for f in full], "sections": full,
                    })
    print(f"  {len(out)} glyph-box questions, "
          f"{sum(i['expected'] for i in out)} glyphs")
    return out


def build(paths: list[Path], out_dir: Path, challenge_spec: Path | None = None,
          heading_spec: Path | None = None, box_paths: list[Path] | None = None) -> dict:
    """Queue every suspect token, plus any challenges, keeping earlier items.

    Items already in ``queue.json`` stay: once a decision is applied its token
    is gone from the sheet, but the page keeps showing what was decided.
    """
    abbreviations = lxin.load_abbreviations(ABBREVIATIONS, lang="cs")
    allowed = lxv.vocabulary(abbreviations) | TOPOGRAPHIC
    vocabulary = sorted(v for v in allowed if v.startswith("["))
    (out_dir / "img").mkdir(parents=True, exist_ok=True)
    (out_dir / "glyph").mkdir(parents=True, exist_ok=True)

    fresh = [_item(path, s, out_dir, vocabulary)
             for path in paths for s in suspects_in_sheet(path, allowed)]
    if challenge_spec:
        fresh += [_item(path, s, out_dir, vocabulary, suggestion, note)
                  for path, s, suggestion, note in challenges(challenge_spec)]
    if heading_spec:
        fresh += headings(heading_spec, out_dir)
    if box_paths:
        fresh += box_items(box_paths, out_dir)
    queue_path = out_dir / "queue.json"
    earlier = (json.loads(queue_path.read_text("utf-8"))["items"]
               if queue_path.exists() else [])
    ids = {i["id"] for i in fresh}
    items = [i for i in earlier if i["id"] not in ids] + fresh
    # Every atlas glyph travels with the queue, so a reviewer who picks a token
    # outside the ranked five still sees what it looks like.
    glyphs = {}
    for token in vocabulary:
        glyph = SYMBOLS / f"{token.strip('[]')}.png"
        if glyph.exists():
            shutil.copy(glyph, out_dir / "glyph" / glyph.name)
            glyphs[token] = f"glyph/{glyph.name}"
    queue = {"built": date.today().isoformat(), "items": items,
             "vocabulary": vocabulary, "glyphs": glyphs}
    queue_path.write_text(json.dumps(queue, ensure_ascii=False, indent=1),
                          encoding="utf-8")
    print(f"\n  {len(fresh)} new, {len(items)} in the queue -> {queue_path}")
    return queue

def _set_cell(line: str, item: dict, new: str) -> str:
    """Rewrite one cell of a table row, found by its column name.

    The cell must still read what the queue recorded; otherwise the row is
    left alone and reported, since someone has changed it since.
    """
    from utils.lexikon_1910 import lexikon_1910 as lx

    cells = line.split("|")
    index = 2 + [name for name, _kind in lx.COLUMNS].index(item["column"])
    if cells[index].strip() != item["token"]:
        print(f"  !! {item['id']}: {item['column']} reads "
              f"{cells[index].strip()!r}, not {item['token']!r} — skipped")
        return line
    cells[index] = f" {new} "
    print(f"  {item['id']}: {item['column']} {item['token']!r} -> {new!r}")
    return "|".join(cells)


def _heading_row(item: dict, d: dict) -> dict:
    """A heading ruling: ``replace`` confirms a heading naming ``replacement``."""
    verdict = {"replace": "heading"}.get(d["decision"], d["decision"])
    return {"scan": item["scan"], "region": item["region"], "top": item["line"],
            "verdict": verdict,
            "kind": d.get("heading_kind", "parts") if verdict == "heading" else "",
            "court": d.get("replacement", "") if verdict == "heading" else "",
            "filed_kind": item["filed_kind"], "filed_court": item["filed_court"],
            "seconds": d.get("seconds", ""), "comment": d.get("comment", ""),
            "decided_at": d.get("decided_at", "")}


def write_heading_rulings(rows: list[dict]) -> None:
    """Merge rulings into ``heading_rulings.csv``, a later ruling replacing one."""
    import csv

    key = lambda r: (r["scan"], r["region"], str(r["top"]))   # noqa: E731
    merged = {}
    if HEADING_RULINGS.exists():
        with HEADING_RULINGS.open(encoding="utf-8", newline="") as f:
            merged = {key(r): r for r in csv.DictReader(f)}
    merged.update({key(r): r for r in rows})
    with HEADING_RULINGS.open("w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(sorted(merged.values(), key=key))
    for r in rows:
        print(f"  heading {r['scan']} {r['region']}: {r['verdict']} {r['court']}")
    print(f"  -> {HEADING_RULINGS.relative_to(ROOT)}")


def apply(decisions_path: Path, queue_path: Path) -> int:
    """Rewrite each decided token in its sheet, refusing a line that moved.

    ``decisions.json`` is the page's ``decisions`` collection as a list of
    ``{id, decision, replacement}``; ``replace`` writes ``replacement``,
    ``unreadable`` writes ``[?]``, ``off_atlas`` writes ``[!]`` and ``keep``
    leaves the token alone.
    """
    queue = {i["id"]: i for i in json.loads(queue_path.read_text("utf-8"))["items"]}
    decisions = json.loads(decisions_path.read_text("utf-8"))
    heading_rows = [_heading_row(queue[d["id"]], d) for d in decisions
                    if d["id"] in queue and queue[d["id"]]["kind"] == "heading"
                    and d.get("decision")]
    if heading_rows:
        write_heading_rulings(heading_rows)
    by_sheet: dict[str, list] = {}
    for d in decisions:
        item = queue.get(d["id"])
        if item is not None and item["kind"] == "heading":
            continue
        if item is None or d.get("decision") in (None, "", "keep"):
            continue
        if d["decision"] == "replace":
            new = d.get("replacement", "")       # "" deletes, for a challenge
        else:
            new = {"unreadable": UNREADABLE, "off_atlas": OFF_ATLAS}.get(
                d["decision"], None)
        if new is None or (new == "" and item["kind"] != "challenge"):
            continue
        if item["section"] == "table" and not item.get("column"):
            # A table cell is not unique on its row (`.` and `—` repeat), so
            # without its column it has to be applied by a person.
            print(f"  .. {item['id']}: table cell {item['token']!r} -> {new!r} "
                  f"— apply by hand ({item['settlement']})")
            continue
        by_sheet.setdefault(item["sheet"], []).append((item, new))
    for sheet, edits in by_sheet.items():
        path = ROOT / sheet
        raw = path.read_bytes().decode("utf-8")
        newline = "\r\n" if "\r\n" in raw else "\n"
        lines = raw.split(newline)
        for item, new in edits:
            line = lines[item["line"] - 1]
            old = item["token"]
            if item["section"] == "table":
                lines[item["line"] - 1] = _set_cell(line, item, new)
                continue
            if item["answer"] not in line:
                # An earlier decision on the same line already changed it; the
                # token itself can still be placed if it occurs exactly once.
                if old and line.count(old) == 1:
                    lines[item["line"] - 1] = line.replace(old, new, 1)
                    print(f"  {item['id']}: {old!r} -> {new!r}")
                else:
                    print(f"  !! {item['id']}: line {item['line']} changed since "
                          f"the queue was built — skipped")
                continue
            if not item["answer"]:                       # a missing answer
                lines[item["line"] - 1] = line.rstrip() + " " + new
            else:
                answer = item["answer"].replace(old, new, 1)
                lines[item["line"] - 1] = line.replace(item["answer"], answer, 1)
            print(f"  {item['id']}: {old!r} -> {new!r}")
        path.write_bytes(newline.join(lines).encode("utf-8"))
    return 0


def main() -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):
            pass
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--role", default="certification",
                    choices=("calibration", "certification", "purposive", "none"),
                    help="the sheets to scan for suspects; none scans no sheet")
    ap.add_argument("--out", default=str(OUT_DIR))
    ap.add_argument("--apply", default="", help="decisions.json to write back")
    ap.add_argument("--challenges", default="",
                    help="JSON list of valid-token readings to question")
    ap.add_argument("--headings", default="",
                    help="JSON list of missed headings to rule on")
    ap.add_argument("--boxes", default="",
                    help="comma-separated roles whose fill-pair glyphs to queue for "
                         "boxing (test A), e.g. calibration,purposive; never certification")
    args = ap.parse_args()
    out_dir = Path(args.out)
    if args.apply:
        return apply(Path(args.apply), out_dir / "queue.json")
    paths = ([] if args.role == "none" else
             sorted((GOLD_ROOT / args.role).glob("[0-9][0-9][0-9][0-9].md")))
    box_roles = [r for r in args.boxes.split(",") if r]
    if "certification" in box_roles:
        raise SystemExit("certification is sealed until the pipeline is frozen (D-55)")
    box_paths = [p for role in box_roles
                 for p in sorted((GOLD_ROOT / role).glob("[0-9][0-9][0-9][0-9].md"))]
    build(paths, out_dir, Path(args.challenges) if args.challenges else None,
          Path(args.headings) if args.headings else None, box_paths)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
