How it works

The four pages read so far (118, 270, 450, 451) were chosen because they were interesting. That makes them a fine development set and an invalid basis for a claim about the volume: an accuracy measured on pages picked for difficulty estimates nothing, and one measured on pages picked for legibility estimates it upward. A number that certifies the symbol channel has to come from a probability sample of the volume, drawn before anything is read.

Systematic, not simple random. A simple random draw of sixteen pages from 480 can land nine of them in the same hundred, and the error this design most fears - a reading that degrades with print or scan quality, which varies down the volume and therefore across okresy - is exactly the error a clustered sample hides. A systematic draw with a random start spreads the sample over the whole book by construction while keeping the start random.

Two disjoint halves, assigned by alternating along the sequence so that each half keeps the spread:

CALIBRATION   thresholds, cluster labels, prompt changes are fitted here
CERTIFICATION sealed; read once, at the end, and reported

They must not be the same pages. A threshold chosen on the pages it is then scored against is not a measurement, and 95% obtained that way is a number about the fitting, not about the volume.

def draw_volume(slug: str, seed: int, n: int = VOLUME_N, check=has_apparatus) -> list[dict]:
    """The systematic draw on one German volume's own frame.

    The frame is the profile's table pages less its prose pages (the statutory cities: a
    small table between paragraphs, no apparatus to speak of) and its other pages. A drawn
    page that is not in the frame, already drawn, or whose image shows no apparatus divider
    moves forward to the next that is, and the move is recorded — the same rule as the
    Czech draw. ``check`` is the apparatus test, a parameter so the rule can be tested.
    """
    from utils.lexikon_1910 import lexikon_1910_volume as lxvol

    profile = lxvol.load_profile(slug)
    first, last = profile.table_first, profile.table_last
    out_of_frame = set(profile.prose_pages) | set(profile.other_pages)
    step = (last - first + 1) // n
    start = random.Random(seed).randrange(step)
    rows, taken = [], set()
    for k in range(n):
        page, replaced, why = first + start + k * step, "", ""
        while page <= last and (page in out_of_frame or page in taken or not check(profile, page)):
            replaced = replaced or str(page)
            why = why or ("prose or other page" if page in out_of_frame else
                          "drawn already" if page in taken else "no apparatus divider")
            page += 1
        if page > last:
            raise RuntimeError(f"{slug}: draw {k + 1} ran off the end of the table")
        taken.add(page)
        rows.append({"scan": page, "role": "calibration" if k % 2 == 0 else "certification",
                     "draw_order": k + 1, "replaced": replaced, "replaced_why": why,
                     "transcribed": "", "note": ""})
    return rows

def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--seed", type=int, default=1910,
                    help="fixed so the draw can be re-run and checked")
    ap.add_argument("--n", type=int, default=16)
    ap.add_argument("--out", default=str(OUT))
    ap.add_argument("--volume", choices=("maehren_de", "schlesien_de"),
                    help="draw a German volume's own sample instead (D-133); --n defaults to 8")
    args = ap.parse_args()

    if args.volume:
        out = Path(str(VOLUME_OUT).format(slug=args.volume))
        if out.exists():
            print(f"{out.relative_to(ROOT)} exists — a draw is made once; delete it to redraw")
            return 1
        rows = draw_volume(args.volume, args.seed, VOLUME_N if args.n == 16 else args.n)
        # The scans only: what is printed on them is not read here, for the sealed half.
        for role in ("calibration", "certification"):
            print(f"{role:13s} {[r['scan'] for r in rows if r['role'] == role]}")
        for r in rows:
            if r["replaced"]:
                print(f"  draw {r['draw_order']}: {r['replaced']} -> {r['scan']} ({r['replaced_why']})")
        with out.open("w", encoding="utf-8", newline="") as fh:
            writer = csv.DictWriter(fh, fieldnames=list(rows[0]))
            writer.writeheader()
            writer.writerows(rows)
        print(f"written to {out.relative_to(ROOT)}")
        return 0

    rows = draw(args.seed, args.n)
    span = TABLE_LAST - TABLE_FIRST + 1

    print(f"frame        scans {TABLE_FIRST}-{TABLE_LAST}  ({span} table pages)")
    print(f"design       systematic, step {span // args.n}, random start, "
          f"seed {args.seed}")
    print(f"already read {', '.join(str(p) for p in ALREADY_READ)}  "
          f"(development set - excluded)")
    print()
    print(f"  {'scan':>5s} {'print':>6s}  {'role':13s} {'note'}")
    for row in rows:
        note = f"replaced {row['replaced']}" if row["replaced"] else ""
        print(f"  {row['scan']:5d} {row['printed_page']:6d}  "
              f"{row['role']:13s} {note}")

    for role in ("calibration", "certification"):
        pages = [r["scan"] for r in rows if r["role"] == role]
        print(f"\n{role:13s} {len(pages):2d} pages   --pages "
              f"{','.join(str(p) for p in pages)}")

    with Path(args.out).open("w", encoding="utf-8", newline="") as fh:
        writer = csv.DictWriter(fh, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)
    # A path outside the repo is legitimate — checking the draw against the
    # committed CSV means writing it somewhere else — and `relative_to` raises on
    # one, which crashed the run AFTER the file had been written.
    written = Path(args.out)
    try:
        written = written.relative_to(ROOT)
    except ValueError:
        pass
    print(f"\nwritten to {written}")
    return 0

from __future__ import annotations

import argparse
import csv
import random
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

#: The table, and only the table: scans 15-494 of the 756-scan Czech volume.
#: The rest is front matter, the Prehled zeme tables, the appendix, the changes
#: lists and the index, and none of it carries an offices section.
TABLE_FIRST, TABLE_LAST = 15, 494

#: Read already, and purposively - kept as a development set, never counted
#: toward the certification. If the draw lands on one it is replaced, and the
#: replacement is recorded rather than quietly substituted.
#:
#: 449 joined them on 2026-08-29 and is the one that was never a dev page. Trutnov
#: spans 449-451, and a soudni okres read WHOLE is the only configuration in which
#: the postal legend can be gold-read at all: its numbering is district-wide, which
#: is why 85 of 535 pilot keys come back `legend truncated`. It sits outside both
#: halves like the other three - none of them is ever substituted for a drawn page,
#: which would break the certification interval. Adding it leaves the realised draw
#: byte-identical, because the systematic candidates are 16, 46, 76 ... 466 and 449
#: is not among them; the frame shrinks by one page out of 480.
ALREADY_READ = (118, 270, 449, 450, 451)

#: Table pages that print no footnote apparatus at all, so there is no offices
#: section to read. Measured 2026-08-14: scans 15-18 have no divider rule and
#: `extract_notes` returns nothing for any of them, while 19 onward are normal.
#: The volume opens its table before it starts printing the apparatus beneath it.
#: They are excluded from the frame rather than drawn and then abandoned, which
#: would leave a hole nobody could see afterwards.
NO_APPARATUS = (15, 16, 17, 18)

OUT = ROOT / "reference" / "places" / "gold_sample_pages.csv"
