How it works

The section cutter missed an Ämter, Anstalten heading on some pages, so the offices lines below it were filed as parts and read under the parts prompt, which does not show the institution glyphs: the re-split of the saved notes (utils.recheck.offices_resplit, notebook 17 §3) recovers them precisely but short — telegraph 0.35 and gendarme 0.40 against the Czech volume, where the pairs both editions read agree at 0.97–0.99.

This module reads each region of output/validation/offices_coverage_regions.csv again, as the route reads an offices section:

  • the crop is cut from the page as the route rendered and cut it (render_native, the page's ink mask, strips at the route's ceiling), from the region's top — the heading the coverage check located, or the section's top where it continues one — to the next heading or the section's foot, mapped from the section image (sections/index.csv) back to the page;
  • the readers are the run's own: Gemini, the reader of record, once, and Haiku three times with the run's contrast sheet, each with its volume's key and the offices prompt (run_r5_german_notes.systems / make_reader), each label with a crop cache of its own under output/recheck/&lt;edition&gt;/reread/;
  • the lines are cleaned, split at any heading they carry (offices_resplit.resplit_sections) and decoded by the route's decoder (offices_resplit.decoded_inventory);
  • the claims are settled as §3 settles them: one every reader makes stands, one only some make is disputed (for Opus, §6), and an entry not every reader reached is the reader of record's.

Nothing is sent unless spend is true, and then only on the machine of record (D-68): without it the regions are cut, priced and checked against the cache, and whatever the cache already holds is decoded. The Czech volume's three regions want the Czech readers and are listed, not read.

python -m utils.recheck.offices_reread --edition de-bohemia            # price, decode the cache
python -m utils.recheck.offices_reread --edition de-bohemia --spend    # read (Heavyhorse6 only)

def page_box(index_row: dict, y_from: str, y_to: str) -> tuple[int, int, int, int]:
    """A region's box on the page: the section's box, its top and foot moved to the region's
    rows of the section image, which is the section enlarged by ``enlarge``."""
    x0, y0, x1, y1 = (int(float(index_row[k])) for k in ("x0", "y0", "x1", "y1"))
    enlarge = float(index_row["enlarge"])
    top = min(y1 - 1, max(y0, y0 + round(float(y_from or 0) / enlarge)))
    bottom = y0 + round(float(y_to) / enlarge) if str(y_to).strip() else y1
    return x0, top, x1, min(y1, max(top + 1, bottom))

def cut(edition: str, pages: str = "full") -> tuple[list[tuple], dict]:
    """``(jobs, segments)``: every region cut as the route cuts an offices section — jobs are
    ``(slot, crop, caption, "offices")`` in printed order, segments by slot. The court of a
    region is the court the reader of record read for its crop (its ``notes`` file), so the
    decoder files its claims as the route files them; a heading inside the crop still splits."""
    profile = lxvol.load_profile(rf.GERMAN[edition])
    index = {r["image"]: r for r in _read(rf.out_dir(edition, pages) / "sections" / "index.csv")}
    record = rs.readers(edition, pages)[0]
    court_de = {(r["page"], r["region"], str(r["order"])): r["court_de"] for r in rs._rows("notes", record)}
    by_page = defaultdict(list)
    for r in regions(edition):
        row = index[r["section"] + ".png"]
        by_page[row["page"]].append((r, row))
    jobs, segments = [], {}
    for page in sorted(by_page):
        image = lxdn.render_native(profile.source_pdf, int(page))
        geo = lximg.detect(image, expected_columns=profile.n_columns, full_width_headings=True)
        ink = lxnt.page_ink(image, geo)
        strip = lxdn.strip_height(bo.MAX_HEIGHT, geo.scale)
        for r, row in sorted(by_page[page], key=lambda pair: int(pair[1]["order"])):
            slot = (row["page"], row["region"], int(row["order"]))
            segment = lxnt.Segment(kind="offices", page=row["page"], region=row["region"],
                                   district=court_de.get((row["page"], row["region"], row["order"]),
                                                         row["district"]),
                                   box=page_box(row, r["y_from"], r["y_to"]), order=int(row["order"]))
            segments[slot] = segment
            for crop, _s, i, n in lxnt.crop_segments(image, [segment], None, upscale=True,
                                                     max_height=strip, ink=ink):
                jobs.append((slot, crop, _caption(i, n), "offices"))
    return jobs, segments

def readers(edition: str, pages: str = "full") -> dict:
    """``{label: {"arm", "reader", "contrast", "contrast_digest"}}`` — the run's four readers
    with the offices prompt of the volume's own key, Haiku with the run's contrast sheet."""

    profile = lxvol.load_profile(rf.GERMAN[edition])
    abbreviations = lxin.load_abbreviations(profile.abbreviations, lang="de", volume=profile.slug)
    sheet = rf.out_dir(edition, pages) / "freeze" / "contrast_sheet.png"
    contrast = Image.open(sheet).convert("RGB")
    digest = hashlib.sha256(sheet.read_bytes()).hexdigest()[:16]
    out = {}
    with _placeholder_keys(("ANTHROPIC_API_KEY", pilot.READER_ENDPOINTS["gemini"][1])):
        for label in rs.readers(edition, pages):

def price(jobs: list[tuple], spec: dict, atlas, system_chars: int) -> float:
    """USD to read the jobs live with one reader (``run_r5_german_notes.estimate``), the
    contrast sheet added for Haiku (cached after the first request, as the key is)."""
    sizes = [(crop.width, crop.height) for _s, crop, _c, _k in jobs]
    est = gn.estimate(spec["arm"], sizes, [], [], {"offices": system_chars, "heading": 0, "header": 0},
                      atlas.size)
    if spec["contrast"] is not None and sizes:
        tokens = gn.image_tokens(*spec["contrast"].size, "claude")
        est["usd"] += tokens * (1 + (len(sizes) - 1) * pilot.CACHE_READ_RATE) * spec["arm"].price[0] / 1e6
    return est["usd"]

def read(edition: str, pages: str = "full", spend: bool = False, out: Path = OUT,
         allow_other_machine: bool = False) -> dict:
    """Cut, price, and read every region with every reader; decode what is read.

    Without ``spend`` nothing is sent: a reader whose crops are all in its cache is decoded,
    the others are priced and left. Returns the summary also written to ``summary.json``."""
    if edition not in GERMAN:
        raise ValueError(f"{edition}: the re-read is for the German volumes; the Czech regions "
                         "want the Czech readers")
    if spend and bo.HOST != MACHINE_OF_RECORD and not allow_other_machine:
        raise RuntimeError(f"the re-read spends on {MACHINE_OF_RECORD} only (D-68): this is {bo.HOST}")
    from PIL import Image

    profile = lxvol.load_profile(rf.GERMAN[edition])
    abbreviations = lxin.load_abbreviations(profile.abbreviations, lang="de", volume=profile.slug)
    atlas = Image.open(bo.de_atlas(profile.slug)).convert("RGB")
    jobs, segments = cut(edition, pages)
    folder = Path(out) / edition / SUB
    folder.mkdir(parents=True, exist_ok=True)
    if spend:
        from utils.lexikon_1910 import lexikon_1910_pages as lxp
        lxp.load_credentials()
    summary = {"module": __version__, "edition": edition, "pages": pages, "host": bo.HOST,
               "run": time.strftime("%Y-%m-%d %H:%M:%S"), "regions": len(segments),
               "crops": len(jobs), "readers": {}}
    for label, spec in readers(edition, pages).items():
        cache = _cache(edition, label, atlas, out)
        missing = [j for j in jobs if not _cached(j, spec, cache)]
        usd = price(missing, spec, atlas, len(spec["system"]))
        entry = {"cached": len(jobs) - len(missing), "to_read": len(missing), "usd_live": round(usd, 2)}
        if missing and not spend:
            entry["state"] = "not read (spend is off)"
            summary["readers"][label] = entry
            continue
        results, parked = gn.read_crops(jobs, {"offices": spec["reader"]}, atlas, cache,
                                        contrast=spec["contrast"],
                                        contrast_digest=spec["contrast_digest"])
        by_slot, counts = gn.clean_lines(results, lxv.vocabulary(abbreviations))
        entry.update({k: counts.get(k, 0) for k in ("failed", "empty", "rejected", "input", "output")})
        if parked.get("at") is not None or counts.get("failed"):
            entry["state"] = "incomplete: run again to finish (the cache keeps what was read)"
        else:
            entry["state"] = "read"
        decode(edition, label, segments, by_slot, folder, profile)
        entry["rows"] = len(_read(folder / f"inventory_{label}.csv"))
        summary["readers"][label] = entry
        print(f"  {label}: {entry}")
    (folder / "summary.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return summary

def decode(edition: str, label: str, segments: dict, by_slot: dict, folder: Path, profile) -> list[dict]:
    """One reader's lines into its inventory: split at any heading the lines carry, decoded by
    the route's decoder. Writes ``notes_<label>.csv``, ``splits_<label>.csv`` and
    ``inventory_<label>.csv``."""
    read_ = [(segments[slot], by_slot.get(slot, []))
             for slot in sorted(segments, key=lambda s: (s[0], s[2]))]
    sections, splits = rs.resplit_sections(read_, profile.find_heading,
                                           rs.other_court(lxdn.load_court_crosswalk()), mend=True)
    inventory = rs.decoded_inventory(sections, edition)
    rs._write(folder / f"notes_{label}.csv",
              [{"page": s.page, "region": s.region, "order": s.order, "court_de": s.district,
                "line_no": i, "text": line} for s, lines in read_ for i, line in enumerate(lines)])
    rs._write(folder / f"splits_{label}.csv", splits)
    rs._write(folder / f"inventory_{label}.csv", inventory)
    return inventory

def settle(claims: dict, record: str, route: set = frozenset()) -> list[dict]:
    """``claims`` — ``{label: {(page, district, muni, part, code)}}`` — settled: on the entries
    every reader reached, a claim all make is ``agreed`` and one only some make ``disputed``;
    on an entry not every reader reached, the reader of record's claim stands alone."""
    touched = set().union(*({c[:4] for c in cs} for cs in claims.values()))
    entries = set.intersection(*({c[:4] for c in cs} for cs in claims.values()))
    shared = {l: {c for c in cs if c[:4] in entries} for l, cs in claims.items()}
    union, agreed = set().union(*shared.values()), set.intersection(*shared.values())
    rows = [dict(zip(CLAIM, c), status="agreed" if c in agreed else "disputed",
                 support=sum(c in s for s in shared.values()), in_route=c in route)
            for c in sorted(union)]
    alone = {c for c in claims[record] if c[:4] in touched - entries}
    rows += [dict(zip(CLAIM, c), status="reader of record", support=1, in_route=c in route)
             for c in sorted(alone)]
    return rows

def consensus(edition: str, pages: str = "full", out: Path = OUT) -> dict:
    """The re-read claims settled as §3 settles the re-split's (``offices_resplit.consensus``):
    on entries every reader's re-read reached, a claim all make stands and one only some make is
    disputed; elsewhere the reader of record's claim stands alone. ``in_route``: the route's
    final inventory already holds the claim. Writes ``consensus.csv``."""
    folder = Path(out) / edition / SUB
    labels = rs.readers(edition, pages)
    claims = {}
    for label in labels:
        path = folder / f"inventory_{label}.csv"
        if not path.exists():
            raise FileNotFoundError(f"{path.name}: {label} has not read the regions yet")
        claims[label] = {rs._key(r) for r in _read(path) if r.get("code")}
    route = {rs._key(r) for r in _read(rf.out_dir(edition, pages) / "final_inventory.csv")}
    rows = settle(claims, labels[0], route)
    rs._write(folder / "consensus.csv", rows)
    status = Counter(r["status"] for r in rows)
    touched = set().union(*({c[:4] for c in cs} for cs in claims.values()))
    entries = set.intersection(*({c[:4] for c in cs} for cs in claims.values()))
    return {"entries": len(touched), "entries_every_reader": len(entries),
            "agreed": status["agreed"], "disputed": status["disputed"],
            "reader_of_record": status["reader of record"],
            "new_to_route": sum(not r["in_route"] for r in rows)}

def report(summary: dict) -> str:
    lines = [f"{summary['edition']}: {summary['regions']} regions, {summary['crops']} crops"]
    total = 0.0
    for label, r in summary["readers"].items():
        total += r["usd_live"]
        lines.append(f"  {label}: {r['state']}; {r['cached']} cached, {r['to_read']} to read "
                     f"(≈ ${r['usd_live']:.2f} live)" + (f", {r['rows']} rows" if "rows" in r else ""))
    lines.append(f"  to read: ≈ ${total:.2f} live")
    return "\n".join(lines)

"""The offices regions the route filed as parts, read again with the offices prompt (D-192).

The section cutter missed an *Ämter, Anstalten* heading on some pages, so the offices lines below
it were filed as parts and read under the parts prompt, which does not show the institution
glyphs: the re-split of the saved notes (:mod:`utils.recheck.offices_resplit`, notebook 17 §3)
recovers them precisely but short — telegraph 0.35 and gendarme 0.40 against the Czech volume,
where the pairs both editions read agree at 0.97–0.99.

This module reads each region of ``output/validation/offices_coverage_regions.csv`` again, as the
route reads an offices section:

* **the crop** is cut from the page as the route rendered and cut it (``render_native``, the
  page's ink mask, strips at the route's ceiling), from the region's top — the heading the
  coverage check located, or the section's top where it continues one — to the next heading or
  the section's foot, mapped from the section image (``sections/index.csv``) back to the page;
* **the readers** are the run's own: Gemini, the reader of record, once, and Haiku three times
  with the run's contrast sheet, each with its volume's key and the **offices** prompt
  (``run_r5_german_notes.systems`` / ``make_reader``), each label with a crop cache of its own
  under ``output/recheck/<edition>/reread/``;
* **the lines** are cleaned, split at any heading they carry (``offices_resplit.resplit_sections``)
  and decoded by the route's decoder (``offices_resplit.decoded_inventory``);
* **the claims** are settled as §3 settles them: one every reader makes stands, one only some make
  is disputed (for Opus, §6), and an entry not every reader reached is the reader of record's.

Nothing is sent unless ``spend`` is true, and then only on the machine of record (D-68): without it
the regions are cut, priced and checked against the cache, and whatever the cache already holds is
decoded. The Czech volume's three regions want the Czech readers and are listed, not read.

    python -m utils.recheck.offices_reread --edition de-bohemia            # price, decode the cache
    python -m utils.recheck.offices_reread --edition de-bohemia --spend    # read (Heavyhorse6 only)
"""
from __future__ import annotations

__version__ = "0.2.0"   # 0.2.0 an entry key read with a full stop is a key (D-201: Haiku on 0083)

import argparse
import contextlib
import csv
import hashlib

from collections import Counter, defaultdict
from pathlib import Path

sys.path[:0] = [str(ROOT), str(ROOT / "utils" / "lexikon_1910")]

import bakeoff as bo                                                   # noqa: E402
import run_frozen as rf                                                # noqa: E402
import run_lexikon_1910_pilot as pilot                                 # noqa: E402
import run_r5_german_notes as gn                                       # noqa: E402

from utils.lexikon_1910 import lexikon_1910_image as lximg             # noqa: E402
from utils.lexikon_1910 import lexikon_1910_institutions as lxin       # noqa: E402
from utils.lexikon_1910 import lexikon_1910_notes as lxnt              # noqa: E402

from utils.lexikon_1910 import lexikon_1910_volume as lxvol            # noqa: E402

OUT = ROOT / "output" / "recheck"

def _read(path: Path) -> list[dict]:
    with open(path, encoding="utf-8", newline="") as fh:
        return list(csv.DictReader(fh))


def regions(edition: str) -> list[dict]:
    """The edition's rows of the coverage check's regions table."""
    return [r for r in _read(REGIONS) if r["edition"] == edition]


def _caption(i: int, n: int) -> str:
    """The route's caption for a crop (``run_r5_german_notes.jobs_of``)."""
    return ("This image is one section of the footnote apparatus." if n == 1 else
            f"This image is part {i + 1} of {n} of one section of the footnote apparatus, "
            "in printed order.")


def page_box(index_row: dict, y_from: str, y_to: str) -> tuple[int, int, int, int]:
    """A region's box on the page: the section's box, its top and foot moved to the region's
    rows of the section image, which is the section enlarged by ``enlarge``."""
    x0, y0, x1, y1 = (int(float(index_row[k])) for k in ("x0", "y0", "x1", "y1"))
    enlarge = float(index_row["enlarge"])
    top = min(y1 - 1, max(y0, y0 + round(float(y_from or 0) / enlarge)))
    bottom = y0 + round(float(y_to) / enlarge) if str(y_to).strip() else y1
    return x0, top, x1, min(y1, max(top + 1, bottom))


def cut(edition: str, pages: str = "full") -> tuple[list[tuple], dict]:
    """``(jobs, segments)``: every region cut as the route cuts an offices section — jobs are
    ``(slot, crop, caption, "offices")`` in printed order, segments by slot. The court of a
    region is the court the reader of record read for its crop (its ``notes`` file), so the
    decoder files its claims as the route files them; a heading inside the crop still splits."""
    profile = lxvol.load_profile(rf.GERMAN[edition])
    index = {r["image"]: r for r in _read(rf.out_dir(edition, pages) / "sections" / "index.csv")}
    record = rs.readers(edition, pages)[0]
    court_de = {(r["page"], r["region"], str(r["order"])): r["court_de"] for r in rs._rows("notes", record)}
    by_page = defaultdict(list)
    for r in regions(edition):
        row = index[r["section"] + ".png"]
        by_page[row["page"]].append((r, row))
    jobs, segments = [], {}
    for page in sorted(by_page):
        image = lxdn.render_native(profile.source_pdf, int(page))
        geo = lximg.detect(image, expected_columns=profile.n_columns, full_width_headings=True)
        ink = lxnt.page_ink(image, geo)
        strip = lxdn.strip_height(bo.MAX_HEIGHT, geo.scale)
        for r, row in sorted(by_page[page], key=lambda pair: int(pair[1]["order"])):
            slot = (row["page"], row["region"], int(row["order"]))
            segment = lxnt.Segment(kind="offices", page=row["page"], region=row["region"],
                                   district=court_de.get((row["page"], row["region"], row["order"]),
                                                         row["district"]),
                                   box=page_box(row, r["y_from"], r["y_to"]), order=int(row["order"]))
            segments[slot] = segment
            for crop, _s, i, n in lxnt.crop_segments(image, [segment], None, upscale=True,
                                                     max_height=strip, ink=ink):
                jobs.append((slot, crop, _caption(i, n), "offices"))
    return jobs, segments


@contextlib.contextmanager
def _placeholder_keys(env_names):
    """Readers are built to compute cache keys and prices; a client wants a key to be built
    at all. Where none is set, a placeholder stands in while they are built and is removed
    after — nothing is sent with it (:func:`read` sends only with ``spend``)."""
    added = [k for k in env_names if not os.environ.get(k)]
    for k in added:
        os.environ[k] = "placeholder-not-sent"
    try:
        yield
    finally:
        for k in added:
            os.environ.pop(k, None)


def readers(edition: str, pages: str = "full") -> dict:
    """``{label: {"arm", "reader", "contrast", "contrast_digest"}}`` — the run's four readers
    with the offices prompt of the volume's own key, Haiku with the run's contrast sheet."""
    from PIL import Image

    profile = lxvol.load_profile(rf.GERMAN[edition])
    abbreviations = lxin.load_abbreviations(profile.abbreviations, lang="de", volume=profile.slug)
    sheet = rf.out_dir(edition, pages) / "freeze" / "contrast_sheet.png"
    contrast = Image.open(sheet).convert("RGB")
    digest = hashlib.sha256(sheet.read_bytes()).hexdigest()[:16]
    out = {}
    with _placeholder_keys(("ANTHROPIC_API_KEY", pilot.READER_ENDPOINTS["gemini"][1])):
        for label in rs.readers(edition, pages):
            arm = gn.ARMS["gemini38" if label.startswith("gemini38") else "haiku55nh"]
            claude = arm.route == "claude"
            system = gn.systems(abbreviations, think=arm.think, contrast=claude,
                                volume=profile.slug)["offices"]
            reader = gn.make_reader(arm, system)
            reader.shared_parts = 2 if claude else 1
            out[label] = {"arm": arm, "reader": reader, "system": system,
                          "contrast": contrast if claude else None,
                          "contrast_digest": digest if claude else ""}
    return out


def _cache(edition: str, label: str, atlas, out: Path = OUT) -> Path:
    return pilot.key_image_cache(Path(out) / edition / SUB / "crop_cache" / label,
                                 pilot.key_image_digest(atlas))


def _cached(job, spec: dict, cache: Path) -> bool:
    _slot, crop, caption, _kind = job
    extra = "atlas" + (f"+contrast:{spec['contrast_digest']}" if spec["contrast"] is not None else "")
    return (cache / f"{gn.cache_key(crop, caption, spec['reader'], extra)}.json").exists()


def price(jobs: list[tuple], spec: dict, atlas, system_chars: int) -> float:
    """USD to read the jobs live with one reader (``run_r5_german_notes.estimate``), the
    contrast sheet added for Haiku (cached after the first request, as the key is)."""
    sizes = [(crop.width, crop.height) for _s, crop, _c, _k in jobs]
    est = gn.estimate(spec["arm"], sizes, [], [], {"offices": system_chars, "heading": 0, "header": 0},
                      atlas.size)
    if spec["contrast"] is not None and sizes:
        tokens = gn.image_tokens(*spec["contrast"].size, "claude")
        est["usd"] += tokens * (1 + (len(sizes) - 1) * pilot.CACHE_READ_RATE) * spec["arm"].price[0] / 1e6
    return est["usd"]


def read(edition: str, pages: str = "full", spend: bool = False, out: Path = OUT,
         allow_other_machine: bool = False) -> dict:
    """Cut, price, and read every region with every reader; decode what is read.

    Without ``spend`` nothing is sent: a reader whose crops are all in its cache is decoded,
    the others are priced and left. Returns the summary also written to ``summary.json``."""
    if edition not in GERMAN:
        raise ValueError(f"{edition}: the re-read is for the German volumes; the Czech regions "
                         "want the Czech readers")
    if spend and bo.HOST != MACHINE_OF_RECORD and not allow_other_machine:
        raise RuntimeError(f"the re-read spends on {MACHINE_OF_RECORD} only (D-68): this is {bo.HOST}")
    from PIL import Image

    profile = lxvol.load_profile(rf.GERMAN[edition])
    abbreviations = lxin.load_abbreviations(profile.abbreviations, lang="de", volume=profile.slug)
    atlas = Image.open(bo.de_atlas(profile.slug)).convert("RGB")
    jobs, segments = cut(edition, pages)
    folder = Path(out) / edition / SUB
    folder.mkdir(parents=True, exist_ok=True)
    if spend:
        from utils.lexikon_1910 import lexikon_1910_pages as lxp
        lxp.load_credentials()
    summary = {"module": __version__, "edition": edition, "pages": pages, "host": bo.HOST,
               "run": time.strftime("%Y-%m-%d %H:%M:%S"), "regions": len(segments),
               "crops": len(jobs), "readers": {}}
    for label, spec in readers(edition, pages).items():
        cache = _cache(edition, label, atlas, out)
        missing = [j for j in jobs if not _cached(j, spec, cache)]
        usd = price(missing, spec, atlas, len(spec["system"]))
        entry = {"cached": len(jobs) - len(missing), "to_read": len(missing), "usd_live": round(usd, 2)}
        if missing and not spend:
            entry["state"] = "not read (spend is off)"
            summary["readers"][label] = entry
            continue
        results, parked = gn.read_crops(jobs, {"offices": spec["reader"]}, atlas, cache,
                                        contrast=spec["contrast"],
                                        contrast_digest=spec["contrast_digest"])
        by_slot, counts = gn.clean_lines(results, lxv.vocabulary(abbreviations))
        entry.update({k: counts.get(k, 0) for k in ("failed", "empty", "rejected", "input", "output")})
        if parked.get("at") is not None or counts.get("failed"):
            entry["state"] = "incomplete: run again to finish (the cache keeps what was read)"
        else:
            entry["state"] = "read"
        decode(edition, label, segments, by_slot, folder, profile)
        entry["rows"] = len(_read(folder / f"inventory_{label}.csv"))
        summary["readers"][label] = entry
        print(f"  {label}: {entry}")
    (folder / "summary.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return summary


def decode(edition: str, label: str, segments: dict, by_slot: dict, folder: Path, profile) -> list[dict]:
    """One reader's lines into its inventory: split at any heading the lines carry, decoded by
    the route's decoder. Writes ``notes_<label>.csv``, ``splits_<label>.csv`` and
    ``inventory_<label>.csv``."""
    read_ = [(segments[slot], by_slot.get(slot, []))
             for slot in sorted(segments, key=lambda s: (s[0], s[2]))]
    sections, splits = rs.resplit_sections(read_, profile.find_heading,
                                           rs.other_court(lxdn.load_court_crosswalk()), mend=True)
    inventory = rs.decoded_inventory(sections, edition)
    rs._write(folder / f"notes_{label}.csv",
              [{"page": s.page, "region": s.region, "order": s.order, "court_de": s.district,
                "line_no": i, "text": line} for s, lines in read_ for i, line in enumerate(lines)])
    rs._write(folder / f"splits_{label}.csv", splits)
    rs._write(folder / f"inventory_{label}.csv", inventory)
    return inventory


def settle(claims: dict, record: str, route: set = frozenset()) -> list[dict]:
    """``claims`` — ``{label: {(page, district, muni, part, code)}}`` — settled: on the entries
    every reader reached, a claim all make is ``agreed`` and one only some make ``disputed``;
    on an entry not every reader reached, the reader of record's claim stands alone."""
    touched = set().union(*({c[:4] for c in cs} for cs in claims.values()))
    entries = set.intersection(*({c[:4] for c in cs} for cs in claims.values()))
    shared = {l: {c for c in cs if c[:4] in entries} for l, cs in claims.items()}
    union, agreed = set().union(*shared.values()), set.intersection(*shared.values())
    rows = [dict(zip(CLAIM, c), status="agreed" if c in agreed else "disputed",
                 support=sum(c in s for s in shared.values()), in_route=c in route)
            for c in sorted(union)]
    alone = {c for c in claims[record] if c[:4] in touched - entries}
    rows += [dict(zip(CLAIM, c), status="reader of record", support=1, in_route=c in route)
             for c in sorted(alone)]
    return rows


def consensus(edition: str, pages: str = "full", out: Path = OUT) -> dict:
    """The re-read claims settled as §3 settles the re-split's (``offices_resplit.consensus``):
    on entries every reader's re-read reached, a claim all make stands and one only some make is
    disputed; elsewhere the reader of record's claim stands alone. ``in_route``: the route's
    final inventory already holds the claim. Writes ``consensus.csv``."""
    folder = Path(out) / edition / SUB
    labels = rs.readers(edition, pages)
    claims = {}
    for label in labels:
        path = folder / f"inventory_{label}.csv"
        if not path.exists():
            raise FileNotFoundError(f"{path.name}: {label} has not read the regions yet")
        claims[label] = {rs._key(r) for r in _read(path) if r.get("code")}
    route = {rs._key(r) for r in _read(rf.out_dir(edition, pages) / "final_inventory.csv")}
    rows = settle(claims, labels[0], route)
    rs._write(folder / "consensus.csv", rows)
    status = Counter(r["status"] for r in rows)
    touched = set().union(*({c[:4] for c in cs} for cs in claims.values()))
    entries = set.intersection(*({c[:4] for c in cs} for cs in claims.values()))
    return {"entries": len(touched), "entries_every_reader": len(entries),
            "agreed": status["agreed"], "disputed": status["disputed"],
            "reader_of_record": status["reader of record"],
            "new_to_route": sum(not r["in_route"] for r in rows)}


def report(summary: dict) -> str:
    lines = [f"{summary['edition']}: {summary['regions']} regions, {summary['crops']} crops"]
    total = 0.0
    for label, r in summary["readers"].items():
        total += r["usd_live"]
        lines.append(f"  {label}: {r['state']}; {r['cached']} cached, {r['to_read']} to read "
                     f"(≈ ${r['usd_live']:.2f} live)" + (f", {r['rows']} rows" if "rows" in r else ""))
    lines.append(f"  to read: ≈ ${total:.2f} live")
    return "\n".join(lines)


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--edition", default="de-bohemia", choices=GERMAN)
    ap.add_argument("--spend", action="store_true")
    ap.add_argument("--out", default=str(OUT))
    args = ap.parse_args(argv)
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    summary = read(args.edition, spend=args.spend, out=Path(args.out))
    print(report(summary))
    if all(r["state"] == "read" for r in summary["readers"].values()):
        print("  consensus:", consensus(args.edition, out=Path(args.out)))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
