How it works

Both scraper_kramerius.py (v5) and scraper_kramerius7.py (v7) walk a page list and fetch two or three streams per page. That loop is the only thing worth watching in a long download, so this module gives them one bar each — and a runner that drives several documents at once (run_lexicon_downloads.py) can stack those bars by passing distinct position values.

The bar is optional. tqdm is declared in requirements.txt, but if it is missing (or --no-progress was passed) page_bar() hands back a no-op stub with the same surface, and the callers fall back to their old every-25-pages log line. Nothing in the download depends on the bar existing.

Usage::

from progress import page_bar, redirect_logging

with redirect_logging(), page_bar(n, "Schlesien", position=0) as bar:
    for page in pages:
        ...
        bar.update(1)
        bar.set_postfix(fetched=n_fetched)

redirect_logging() routes logging output through tqdm.write for the duration, so a warning printed mid-download does not shred the bar.

#!/usr/bin/env python3
"""
Progress-bar helper shared by the Kramerius downloaders.

Both ``scraper_kramerius.py`` (v5) and ``scraper_kramerius7.py`` (v7) walk a page
list and fetch two or three streams per page.  That loop is the only thing worth
watching in a long download, so this module gives them one bar each — and a
runner that drives several documents at once (``run_lexicon_downloads.py``) can
stack those bars by passing distinct ``position`` values.

The bar is optional.  ``tqdm`` is declared in ``requirements.txt``, but if it is
missing (or ``--no-progress`` was passed) ``page_bar()`` hands back a no-op stub
with the same surface, and the callers fall back to their old
every-25-pages log line.  Nothing in the download depends on the bar existing.

Usage::

    from progress import page_bar, redirect_logging

    with redirect_logging(), page_bar(n, "Schlesien", position=0) as bar:
        for page in pages:
            ...
            bar.update(1)
            bar.set_postfix(fetched=n_fetched)

``redirect_logging()`` routes ``logging`` output through ``tqdm.write`` for the
duration, so a warning printed mid-download does not shred the bar.
"""

from __future__ import annotations

import contextlib

__version__ = "1.0.0"

try:  # optional dependency
    from tqdm.auto import tqdm as _tqdm
except ImportError:  # pragma: no cover - exercised only without tqdm installed
    _tqdm = None

# desc / bar / 96/122 / [elapsed<remaining, rate] / postfix
BAR_FORMAT = "{desc} {bar} {n_fmt}/{total_fmt} [{elapsed}<{remaining}, {rate_fmt}]{postfix}"


class _NullBar:
    """Stand-in with tqdm's surface, used when tqdm is absent or disabled."""

    live = False

    def __init__(self, total: int = 0, desc: str = "") -> None:
        self.total = total
        self.desc = desc
        self.n = 0

    def update(self, n: int = 1) -> None:
        self.n += n

    def set_postfix(self, **kwargs) -> None:
        pass

    def set_description(self, desc: str) -> None:
        self.desc = desc

    def write(self, msg: str) -> None:
        print(msg)

    def close(self) -> None:
        pass

    def __enter__(self) -> "_NullBar":
        return self

    def __exit__(self, *exc) -> bool:
        self.close()
        return False


def has_tqdm() -> bool:
    """True when a real progress bar can be drawn."""
    return _tqdm is not None


def page_bar(
    total: int,
    desc: str,
    *,
    enabled: bool = True,
    position: int | None = None,
    unit: str = "p",
):
    """Return a page-loop progress bar (tqdm) or a silent stub.

    ``position`` stacks bars when several documents download concurrently:
    position 0 is the top line, 1 the next, and so on.
    """
    if _tqdm is None or not enabled:
        return _NullBar(total, desc)
    bar = _tqdm(
        total=total,
        desc=desc,
        unit=unit,
        position=position,
        leave=True,
        dynamic_ncols=True,
        bar_format=BAR_FORMAT,
    )
    bar.live = True  # callers use this to suppress their fallback logging
    return bar


def redirect_logging():
    """Context manager routing ``logging`` through ``tqdm.write`` (no-op without tqdm)."""
    if _tqdm is None:
        return contextlib.nullcontext()
    try:
        from tqdm.contrib.logging import logging_redirect_tqdm

        return logging_redirect_tqdm()
    except ImportError:  # very old tqdm
        return contextlib.nullcontext()
