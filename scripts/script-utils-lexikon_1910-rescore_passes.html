How it works

SYMBOL_ACCURACY_PLAN.md (revision of 2026-09-22, D-56) moves the headline from the (entry, code) claim to (entry, settlement, code) — the part of a municipality an institution is filed under is what a birthplace joins to — and asks for three things beside it before any new model run: an interval that treats a page as one cluster, sensitivity and specificity per glyph, and a diagnosis of the claims that name the right institution under the wrong part. This script applies lexikon_1910_institutions 0.6.0 to the inventories already on disk and writes all four.

It reads inventory_human.csv and every inventory_&lt;label&gt;.csv under output/lexikon_1910_pilot/bakeoff/&lt;host&gt;/. The inventories are the decoded passes, so nothing is re-decoded here: a pass written before the 0.5.0 decoder fix still carries its misfiled parts, and its row says so in false_misfiled. Rate-limited passes are excluded, as everywhere (bakeoff.passes_on_disk).

What it does not cover: the whole-page (v2) repeats of 2026-09-10, which are rebuilt from raw payloads by run_e1_gemini_version.py and whose pipeline the plan has stopped.

Outputs, all under output/lexikon_1910_pilot/bakeoff/, named for the run tag (r4 here) so an earlier run's report is not overwritten by a later one:

rescore_r4_accuracy.csv           one row per pass: both units, counts, interval
rescore_r4_symbol_rates.csv       one row per (pass, glyph): tp fp fn tn, rates
rescore_r4_part_attribution.csv   one row per misfiled institution
rescore_r4_named_parts.csv        one row per pass: the Části osad parts
rescore_r4_parts_symbols.csv      one row per named part whose symbol differs
rescore_r4_attributes.csv         one row per (pass, attribute): schools,
                                  burgher schools, cooperatives
rescore_r4_order.csv              one row per pass: what the printed order
                                  of the marks flags (I22)
rescore_r4_order_flags.csv        one row per flagged mark, on every page
rescore_r4_order_disputes.csv     one row per pair of passes: the flags on
                                  the claims the two disagree on
rescore_r4_report.txt             the ledgers below, as printed

The last three need the two files a pass writes since 2026-10-06 — its named parts (parts_&lt;label&gt;.csv) and what it read beside the school and cooperative glyphs (attributes_&lt;label&gt;.csv) — and the gold's own, parts_human.csv and attributes_human.csv. A pass read before that date has neither until it is replayed from its crop cache; it is then scored on symbols only, and says so.

The three order files are the printed-order check of lexikon_1910_institutions 0.9.0 (D-73). It needs each mark's place in its line, which an inventory does not keep, so it reads the pass's decoded records, institutions_&lt;label&gt;.csv. The order itself is learned from the gold sheets of the purposive and calibration roles, read directly (read_transcription_md.read_order_reference) and never from the sealed certification sheets; each page is judged on the order of the other pages. The check flags and routes. It changes no claim, and no figure above it moves.

Usage

python utils/lexikon_1910/rescore_passes.py                 # every host's passes
python utils/lexikon_1910/rescore_passes.py --hosts ""      # this machine's only
python utils/lexikon_1910/rescore_passes.py --legacy        # + pre-split passes
python utils/lexikon_1910/rescore_passes.py --top 5         # detail for the best 5
python utils/lexikon_1910/rescore_passes.py --run-tag r3    # an earlier run's passes

def write_csv(path: Path, rows: list[dict]) -> None:
    """Write ``rows``; with none, write nothing rather than a headerless file."""
    if not rows:
        path.unlink(missing_ok=True)
        return
    with path.open("w", encoding="utf-8", newline="") as fh:
        writer = csv.DictWriter(fh, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)

def score_sidecars(passes: dict, human: list[dict]) -> dict:
    """The named parts and the attributes of every pass that wrote them.

    Returns ``named_parts``, ``parts_symbols`` and ``attributes`` as row lists and
    ``reports`` as ``{label: ledger text}``. Both gold frames are held to the pages
    ``inventory_human.csv`` covers, so the scored gold is one set of pages for
    every figure in the report; the attributes are also held to the entries the
    pass and the gold both read, as the symbols are.
    """
    pages = {str(row["page"]) for row in human}
    gold_parts = bo.load_pass(PARTS_HUMAN) if PARTS_HUMAN.exists() else []
    gold_attributes = [row for row in (bo.load_pass(ATTRIBUTES_HUMAN)
                                       if ATTRIBUTES_HUMAN.exists() else [])
                       if str(row["page"]) in pages]
    read_by_gold = lxin._claim_counts(human)[0]
    out = {"named_parts": [], "parts_symbols": [], "attributes": [], "reports": {}}
    for label, rows in passes.items():
        ledgers = []
        parts = sidecar(label, "parts")
        if gold_parts and parts:
            summary, differ = lxnt.score_parts(gold_parts, parts, pages=pages)
            out["named_parts"].append({"pass": label, **summary})
            out["parts_symbols"] += [{"pass": label, **pair} for pair in differ]
            ledgers.append(lxnt.lexikon_1910_parts_score_report(summary, differ, label))
        attributes = sidecar(label, "attributes")
        if gold_attributes and attributes:
            shared = {tuple(map(str, entry))
                      for entry in read_by_gold & lxin._claim_counts(rows)[0]}
            scored = lxat.score_attributes(gold_attributes, attributes, entries=shared)
            out["attributes"] += [{"pass": label, **row} for row in scored]
            ledgers.append(lxat.lexikon_1910_attributes_report(scored, label))
        if ledgers:
            out["reports"][label] = "\n\n".join(ledgers)
    return out

def order_frames(passes: dict, human: list[dict]) -> dict:
    """The printed-order check (I22) on every pass whose decoded records are on disk.

    Returns ``reference`` (the runs of marks the gold sheets print), ``flags``
    as ``{label: rows}``, and three row lists: ``scores`` (one per pass),
    ``flag_rows`` (one per flagged mark, with ``in_reference`` saying whether
    the scored gold has the claim, blank where it has no such entry, and
    ``in_sheets`` the same against the line of whichever gold sheet transcribes
    the part) and ``disputes`` (one per pair of passes).

    The pass's entry text is not on disk, so a named part inside an entry is
    seen only where the name stands directly before a mark, and parentheses are
    not seen at all (:func:`lexikon_1910_institutions.order_runs`).
    """
    reference = gold_sheets.read_order_reference()
    flags = {}
    for label in passes:
        records = sidecar(label, "institutions")
        if records:
            flags[label] = lxin.check_order(records, reference, leave_page_out=True)
    scored = {"human": human, **passes}
    gold_entries, _units, gold_claims = lxin._claim_counts(human)
    _entries, sheet_units, sheet_claims = lxin._claim_counts(
        lxin.order_sheet_rows(reference))
    flag_rows = []
    for label, rows in flags.items():
        for row in rows:
            (claim,) = lxin._claim_counts([row])[2]
            flag_rows.append({"pass": label, **row, "in_reference": (
                "" if claim[:3] not in gold_entries else claim in gold_claims),
                "in_sheets": ("" if claim[:-1] not in sheet_units
                              else claim in sheet_claims)})
    labels = sorted(flags)
    disputes = [lxin.order_disputes(scored, flags, first, second)
                for k, first in enumerate(labels) for second in labels[k + 1:]]
    return {"reference": reference, "flags": flags, "flag_rows": flag_rows,
            "scores": lxin.order_scores(scored, flags, sheets=reference),
            "disputes": disputes}

def dispute_means(disputes: list[dict]) -> list[dict]:
    """The pair rows averaged over the repeats of each two readers.

    One row per (first reader, second reader) as the rows have them — a reader
    with itself is its repeats against each other — with every count averaged
    and ``pairs`` saying how many pairs of passes went in.
    """
    groups: dict[tuple, list[dict]] = {}
    for row in disputes:
        if row.get("disputed"):
            key = (family_of(row["first"]), family_of(row["second"]))
            groups.setdefault(key, []).append(row)
    return [{**{key: (round(sum(row[key] for row in rows) / len(rows), 2)
                      if isinstance(value, int) else value)
                for key, value in rows[0].items()},
             "first": first, "second": second, "pairs": len(rows)}
            for (first, second), rows in sorted(groups.items())]

def order_ledger(frames: dict) -> str:
    """The I22 block of the report: the reference, each pass, each two readers."""
    reference = frames["reference"]
    return "\n".join([
        lxin.lexikon_1910_order_report(
            lxin.order_summary(reference), frames["scores"],

        "", "  mean over the pairs of repeats of each two readers:", "",
        lxin.lexikon_1910_order_disputes_report(dispute_means(frames["disputes"]))])

def rescore(hosts: str = "all", legacy: bool = False, top: int = 3,
            draws: int = 2000, run_tag: str = "") -> str:
    """Score every pass on disk, write the outputs, return the report.

    ``run_tag`` names an earlier run to score (``r3``); empty means the current
    one. :func:`bakeoff.passes_on_disk` only reads labels carrying its run tag, so
    without this the passes of every earlier run would drop out of the report the
    day the tag is bumped.
    """
    if not bo.HUMAN.exists():
        return f"no gold at {bo.HUMAN} — nothing to score against"
    human = bo.load_pass(bo.HUMAN)
    if run_tag:
        bo.RUN_TAG = run_tag
    passes = bo.passes_on_disk(legacy=legacy, hosts=hosts)
    if not passes:
        return (f"no {bo.RUN_TAG} passes under {bo.BAKEOFF} (hosts={hosts!r}, "
                f"legacy={legacy}); an earlier run is scored with --run-tag")

    scored = lxin.score_against_reference({"human": human, **passes},
                                          reference="human")
    intervals, rates, misfiled = {}, [], []
    for label, rows in passes.items():
        pair = {"human": human, label: rows}
        intervals[label] = lxin.clustered_interval(pair, label, draws=draws)
        rates.extend(lxin.symbol_rates(pair, label))
        misfiled.extend(lxin.part_attribution(pair, label))
    accuracy = [{**row, **{key: intervals[row["pass"]][key]
                           for key in INTERVAL_KEYS}} for row in scored]
    beside = score_sidecars(passes, human)
    order = order_frames(passes, human)

    lines = [f"rescore_passes v{__version__} — {time.strftime('%Y-%m-%d %H:%M')}"
             f" — run {bo.RUN_TAG}",
             gold_line(human), "",
             lxin.lexikon_1910_accuracy_report(scored), "",
             lxin.lexikon_1910_clustered_interval_report(
                 [intervals[row["pass"]] for row in scored]),
             "", "", order_ledger(order)]
    for row in scored[:top]:
        label = row["pass"]
        lines += ["", "", lxin.lexikon_1910_symbol_rates_report(
                      [r for r in rates if r["pass"] == label]),
                  "", lxin.lexikon_1910_part_attribution_report(
                      [r for r in misfiled if r["pass"] == label])]
        lines += ["", beside["reports"].get(
            label, f"  {label}: no named-parts or attributes file — a pass read "
                   f"before 2026-10-06 has neither until it is replayed")]
    report = "\n".join(lines)

    bo.BAKEOFF.mkdir(parents=True, exist_ok=True)
    out = out_paths(bo.RUN_TAG)
    write_csv(out["accuracy"], accuracy)
    write_csv(out["rates"], rates)
    write_csv(out["parts"], misfiled)
    for name in ("named_parts", "parts_symbols", "attributes"):
        write_csv(out[name], beside[name])
    write_csv(out["order"], order["scores"])
    write_csv(out["order_flags"], order["flag_rows"])
    write_csv(out["order_disputes"], order["disputes"])
    out["report"].write_text(report + "\n", encoding="utf-8")
    return report

def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--hosts", default="all",
                    help='"all" pools every machine\'s passes (default); '
                         '"" reads this machine\'s only')
    ap.add_argument("--legacy", action="store_true",
                    help="also read passes written before the per-host split")
    ap.add_argument("--top", type=int, default=3,
                    help="passes to print glyph rates and misfilings for")
    ap.add_argument("--bootstrap", type=int, default=2000,
                    help="page-bootstrap draws for the interval")
    ap.add_argument("--run-tag", default="",
                    help="score an earlier run's passes (e.g. r3) instead of the "
                         "current one")
    args = ap.parse_args(argv)
    print(rescore(hosts=args.hosts, legacy=args.legacy, top=args.top,
                  draws=args.bootstrap, run_tag=args.run_tag))
    return 0

asks for three things beside it before any new model run: an interval that treats
a page as one cluster, sensitivity and specificity per glyph, and a diagnosis of
the claims that name the right institution under the wrong part. This script
applies ``lexikon_1910_institutions`` 0.6.0 to the inventories already on disk
and writes all four.

It reads ``inventory_human.csv`` and every ``inventory_<label>.csv`` under
``output/lexikon_1910_pilot/bakeoff/<host>/``. The inventories are the decoded
passes, so nothing is re-decoded here: a pass written before the 0.5.0 decoder
fix still carries its misfiled parts, and its row says so in ``false_misfiled``.
Rate-limited passes are excluded, as everywhere (:func:`bakeoff.passes_on_disk`).

What it does not cover: the whole-page (v2) repeats of 2026-09-10, which are
rebuilt from raw payloads by ``run_e1_gemini_version.py`` and whose pipeline the
plan has stopped.

Outputs, all under ``output/lexikon_1910_pilot/bakeoff/``, named for the run tag
(``r4`` here) so an earlier run's report is not overwritten by a later one:

    rescore_r4_accuracy.csv           one row per pass: both units, counts, interval
    rescore_r4_symbol_rates.csv       one row per (pass, glyph): tp fp fn tn, rates
    rescore_r4_part_attribution.csv   one row per misfiled institution
    rescore_r4_named_parts.csv        one row per pass: the Části osad parts
    rescore_r4_parts_symbols.csv      one row per named part whose symbol differs
    rescore_r4_attributes.csv         one row per (pass, attribute): schools,
                                      burgher schools, cooperatives
    rescore_r4_order.csv              one row per pass: what the printed order
                                      of the marks flags (I22)
    rescore_r4_order_flags.csv        one row per flagged mark, on every page
    rescore_r4_order_disputes.csv     one row per pair of passes: the flags on
                                      the claims the two disagree on
    rescore_r4_report.txt             the ledgers below, as printed

The last three need the two files a pass writes since 2026-10-06 — its named
parts (``parts_<label>.csv``) and what it read beside the school and cooperative
glyphs (``attributes_<label>.csv``) — and the gold's own, ``parts_human.csv`` and
``attributes_human.csv``. A pass read before that date has neither until it is
replayed from its crop cache; it is then scored on symbols only, and says so.

The three ``order`` files are the printed-order check of
``lexikon_1910_institutions`` 0.9.0 (D-73). It needs each mark's place in its
line, which an inventory does not keep, so it reads the pass's decoded records,
``institutions_<label>.csv``. The order itself is learned from the gold sheets
of the purposive and calibration roles, read directly
(:func:`read_transcription_md.read_order_reference`) and never from the sealed
certification sheets; each page is judged on the order of the other pages. The
check flags and routes. It changes no claim, and no figure above it moves.

Usage
-----
    python utils/lexikon_1910/rescore_passes.py                 # every host's passes
    python utils/lexikon_1910/rescore_passes.py --hosts ""      # this machine's only
    python utils/lexikon_1910/rescore_passes.py --legacy        # + pre-split passes
    python utils/lexikon_1910/rescore_passes.py --top 5         # detail for the best 5
    python utils/lexikon_1910/rescore_passes.py --run-tag r3    # an earlier run's passes
"""

from __future__ import annotations

__version__ = "1.4.0"

import argparse
import csv
import sys
import time
from pathlib import Path

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]
sys.path.insert(0, str(ROOT))
sys.path.insert(0, str(HERE))

import bakeoff as bo                                             # noqa: E402
import read_transcription_md as gold_sheets                      # noqa: E402
from utils.lexikon_1910 import lexikon_1910_attributes as lxat        # noqa: E402
from utils.lexikon_1910 import lexikon_1910_institutions as lxin      # noqa: E402
from utils.lexikon_1910 import lexikon_1910_notes as lxnt             # noqa: E402

#: The frames the gold sheets yield beside the symbol inventory.
PARTS_HUMAN = bo.OUT_DIR / "parts_human.csv"
ATTRIBUTES_HUMAN = bo.OUT_DIR / "attributes_human.csv"


def out_paths(run_tag: str) -> dict[str, Path]:
    """The outputs of one run, named for its tag.

    Tagged since 1.1.0, so the report of an earlier run is not overwritten the
    first time a later one is scored: ``rescore_r3_report.txt`` and
    ``rescore_r4_report.txt`` sit side by side.
    """
    stem = f"rescore_{run_tag}"
    return {"accuracy": bo.BAKEOFF / f"{stem}_accuracy.csv",
            "rates": bo.BAKEOFF / f"{stem}_symbol_rates.csv",
            "parts": bo.BAKEOFF / f"{stem}_part_attribution.csv",
            "named_parts": bo.BAKEOFF / f"{stem}_named_parts.csv",
            "parts_symbols": bo.BAKEOFF / f"{stem}_parts_symbols.csv",
            "attributes": bo.BAKEOFF / f"{stem}_attributes.csv",
            "order": bo.BAKEOFF / f"{stem}_order.csv",
            "order_flags": bo.BAKEOFF / f"{stem}_order_flags.csv",
            "order_disputes": bo.BAKEOFF / f"{stem}_order_disputes.csv",
            "report": bo.BAKEOFF / f"{stem}_report.txt"}


#: The interval's columns, copied onto each pass's accuracy row.
INTERVAL_KEYS = ("pages", "f1_lo", "f1_hi", "f1_by_page")


def write_csv(path: Path, rows: list[dict]) -> None:
    """Write ``rows``; with none, write nothing rather than a headerless file."""
    if not rows:
        path.unlink(missing_ok=True)
        return
    with path.open("w", encoding="utf-8", newline="") as fh:
        writer = csv.DictWriter(fh, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)


def gold_line(human: list[dict]) -> str:
    """What the reference is, counted from the file rather than typed."""
    return (f"gold: {bo.HUMAN.name} — {lxin.describe_reference(human)} "
            f"(development gold; certifies nothing)")


def sidecar(label: str, stem: str) -> list[dict]:
    """A pass's ``<stem>_<label>.csv``, from whichever machine's folder holds it."""
    path = bo.find_pass_file(stem, label)
    return bo.load_pass(path) if path else []


def score_sidecars(passes: dict, human: list[dict]) -> dict:
    """The named parts and the attributes of every pass that wrote them.

    Returns ``named_parts``, ``parts_symbols`` and ``attributes`` as row lists and
    ``reports`` as ``{label: ledger text}``. Both gold frames are held to the pages
    ``inventory_human.csv`` covers, so the scored gold is one set of pages for
    every figure in the report; the attributes are also held to the entries the
    pass and the gold both read, as the symbols are.
    """
    pages = {str(row["page"]) for row in human}
    gold_parts = bo.load_pass(PARTS_HUMAN) if PARTS_HUMAN.exists() else []
    gold_attributes = [row for row in (bo.load_pass(ATTRIBUTES_HUMAN)
                                       if ATTRIBUTES_HUMAN.exists() else [])
                       if str(row["page"]) in pages]
    read_by_gold = lxin._claim_counts(human)[0]
    out = {"named_parts": [], "parts_symbols": [], "attributes": [], "reports": {}}
    for label, rows in passes.items():
        ledgers = []
        parts = sidecar(label, "parts")
        if gold_parts and parts:
            summary, differ = lxnt.score_parts(gold_parts, parts, pages=pages)
            out["named_parts"].append({"pass": label, **summary})
            out["parts_symbols"] += [{"pass": label, **pair} for pair in differ]
            ledgers.append(lxnt.lexikon_1910_parts_score_report(summary, differ, label))
        attributes = sidecar(label, "attributes")
        if gold_attributes and attributes:
            shared = {tuple(map(str, entry))
                      for entry in read_by_gold & lxin._claim_counts(rows)[0]}
            scored = lxat.score_attributes(gold_attributes, attributes, entries=shared)
            out["attributes"] += [{"pass": label, **row} for row in scored]
            ledgers.append(lxat.lexikon_1910_attributes_report(scored, label))
        if ledgers:
            out["reports"][label] = "\n\n".join(ledgers)
    return out


def family_of(label: str) -> str:
    """The reader a pass label belongs to: ``gemini38_r4atlasntg2`` -> ``gemini38``."""
    return label.split(f"_{bo.RUN_TAG}")[0]


def order_frames(passes: dict, human: list[dict]) -> dict:
    """The printed-order check (I22) on every pass whose decoded records are on disk.

    Returns ``reference`` (the runs of marks the gold sheets print), ``flags``
    as ``{label: rows}``, and three row lists: ``scores`` (one per pass),
    ``flag_rows`` (one per flagged mark, with ``in_reference`` saying whether
    the scored gold has the claim, blank where it has no such entry, and
    ``in_sheets`` the same against the line of whichever gold sheet transcribes
    the part) and ``disputes`` (one per pair of passes).

    The pass's entry text is not on disk, so a named part inside an entry is
    seen only where the name stands directly before a mark, and parentheses are
    not seen at all (:func:`lexikon_1910_institutions.order_runs`).
    """
    reference = gold_sheets.read_order_reference()
    flags = {}
    for label in passes:
        records = sidecar(label, "institutions")
        if records:
            flags[label] = lxin.check_order(records, reference, leave_page_out=True)
    scored = {"human": human, **passes}
    gold_entries, _units, gold_claims = lxin._claim_counts(human)
    _entries, sheet_units, sheet_claims = lxin._claim_counts(
        lxin.order_sheet_rows(reference))
    flag_rows = []
    for label, rows in flags.items():
        for row in rows:
            (claim,) = lxin._claim_counts([row])[2]
            flag_rows.append({"pass": label, **row, "in_reference": (
                "" if claim[:3] not in gold_entries else claim in gold_claims),
                "in_sheets": ("" if claim[:-1] not in sheet_units
                              else claim in sheet_claims)})
    labels = sorted(flags)
    disputes = [lxin.order_disputes(scored, flags, first, second)
                for k, first in enumerate(labels) for second in labels[k + 1:]]
    return {"reference": reference, "flags": flags, "flag_rows": flag_rows,
            "scores": lxin.order_scores(scored, flags, sheets=reference),
            "disputes": disputes}


def dispute_means(disputes: list[dict]) -> list[dict]:
    """The pair rows averaged over the repeats of each two readers.

    One row per (first reader, second reader) as the rows have them — a reader
    with itself is its repeats against each other — with every count averaged
    and ``pairs`` saying how many pairs of passes went in.
    """
    groups: dict[tuple, list[dict]] = {}
    for row in disputes:
        if row.get("disputed"):
            key = (family_of(row["first"]), family_of(row["second"]))
            groups.setdefault(key, []).append(row)
    return [{**{key: (round(sum(row[key] for row in rows) / len(rows), 2)
                      if isinstance(value, int) else value)
                for key, value in rows[0].items()},
             "first": first, "second": second, "pairs": len(rows)}
            for (first, second), rows in sorted(groups.items())]


def order_ledger(frames: dict) -> str:
    """The I22 block of the report: the reference, each pass, each two readers."""
    reference = frames["reference"]
    return "\n".join([
        lxin.lexikon_1910_order_report(
            lxin.order_summary(reference), frames["scores"],
            sheet_flags=lxin.order_reference_flags(reference),
            joins=lxin.order_joins(reference)),
        "", "  mean over the pairs of repeats of each two readers:", "",
        lxin.lexikon_1910_order_disputes_report(dispute_means(frames["disputes"]))])


def rescore(hosts: str = "all", legacy: bool = False, top: int = 3,
            draws: int = 2000, run_tag: str = "") -> str:
    """Score every pass on disk, write the outputs, return the report.

    ``run_tag`` names an earlier run to score (``r3``); empty means the current
    one. :func:`bakeoff.passes_on_disk` only reads labels carrying its run tag, so
    without this the passes of every earlier run would drop out of the report the
    day the tag is bumped.
    """
    if not bo.HUMAN.exists():
        return f"no gold at {bo.HUMAN} — nothing to score against"
    human = bo.load_pass(bo.HUMAN)
    if run_tag:
        bo.RUN_TAG = run_tag
    passes = bo.passes_on_disk(legacy=legacy, hosts=hosts)
    if not passes:
        return (f"no {bo.RUN_TAG} passes under {bo.BAKEOFF} (hosts={hosts!r}, "
                f"legacy={legacy}); an earlier run is scored with --run-tag")

    scored = lxin.score_against_reference({"human": human, **passes},
                                          reference="human")
    intervals, rates, misfiled = {}, [], []
    for label, rows in passes.items():
        pair = {"human": human, label: rows}
        intervals[label] = lxin.clustered_interval(pair, label, draws=draws)
        rates.extend(lxin.symbol_rates(pair, label))
        misfiled.extend(lxin.part_attribution(pair, label))
    accuracy = [{**row, **{key: intervals[row["pass"]][key]
                           for key in INTERVAL_KEYS}} for row in scored]
    beside = score_sidecars(passes, human)
    order = order_frames(passes, human)

    lines = [f"rescore_passes v{__version__} — {time.strftime('%Y-%m-%d %H:%M')}"
             f" — run {bo.RUN_TAG}",
             gold_line(human), "",
             lxin.lexikon_1910_accuracy_report(scored), "",
             lxin.lexikon_1910_clustered_interval_report(
                 [intervals[row["pass"]] for row in scored]),
             "", "", order_ledger(order)]
    for row in scored[:top]:
        label = row["pass"]
        lines += ["", "", lxin.lexikon_1910_symbol_rates_report(
                      [r for r in rates if r["pass"] == label]),
                  "", lxin.lexikon_1910_part_attribution_report(
                      [r for r in misfiled if r["pass"] == label])]
        lines += ["", beside["reports"].get(
            label, f"  {label}: no named-parts or attributes file — a pass read "
                   f"before 2026-10-06 has neither until it is replayed")]
    report = "\n".join(lines)

    bo.BAKEOFF.mkdir(parents=True, exist_ok=True)
    out = out_paths(bo.RUN_TAG)
    write_csv(out["accuracy"], accuracy)
    write_csv(out["rates"], rates)
    write_csv(out["parts"], misfiled)
    for name in ("named_parts", "parts_symbols", "attributes"):
        write_csv(out[name], beside[name])
    write_csv(out["order"], order["scores"])
    write_csv(out["order_flags"], order["flag_rows"])
    write_csv(out["order_disputes"], order["disputes"])
    out["report"].write_text(report + "\n", encoding="utf-8")
    return report


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--hosts", default="all",
                    help='"all" pools every machine\'s passes (default); '
                         '"" reads this machine\'s only')
    ap.add_argument("--legacy", action="store_true",
                    help="also read passes written before the per-host split")
    ap.add_argument("--top", type=int, default=3,
                    help="passes to print glyph rates and misfilings for")
    ap.add_argument("--bootstrap", type=int, default=2000,
                    help="page-bootstrap draws for the interval")
    ap.add_argument("--run-tag", default="",
                    help="score an earlier run's passes (e.g. r3) instead of the "
                         "current one")
    args = ap.parse_args(argv)
    print(rescore(hosts=args.hosts, legacy=args.legacy, top=args.top,
                  draws=args.bootstrap, run_tag=args.run_tag))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
