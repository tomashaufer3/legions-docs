How it works

The certification drew 8 German pages; the full run reads 641. A page whose geometry the route cannot read either stops the run (the apparatus cut raised on Böhmen 21, the first page of the range, and on 73 0088 0232 …) or loses its table without a word (no blocks on 48 Böhmen and 9 Mähren table pages before lexikon_1910_image 0.3.0). This cuts every page the way the route does — the page geometry, the apparatus sections (run_r5_german_notes.cut_pages, headings left unread), the table blocks (lexikon_1910_de_table.page_blocks) — and lists what is odd:

cut fails the apparatus cut raised: the full run would stop here no table no table block: the page's municipalities are not read no apparatus no apparatus divider found: the page's symbols are not read fallback a page read through a 0.3.0 rule (header by position, column template)

No model is called and nothing is written but the audit. Six workers; about 20 minutes.

python utils/reference_tools/places/audit_full_run_pages.py [--editions de-moravia]

Writes output/frozen/page_audit.{csv,txt}. Outside utils/lexikon_1910.

def audit_page(job: tuple[str, int]) -> dict:
    """One page's geometry, apparatus cut and table blocks."""
    _paths()
    import run_frozen as rf
    import run_r5_german_notes as r5
    from utils.lexikon_1910 import lexikon_1910_de_notes as lxdn
    from utils.lexikon_1910 import lexikon_1910_de_table as dt

    edition, scan = job
    profile = rf._profile(edition)
    row = {"edition": edition, "scan": scan}
    image = lxdn.render_native(profile.source_pdf, scan)
    geo, blocks = dt.page_blocks(image, profile, f"{scan:04d}")
    row.update(kind=geo.kind, columns=geo.n_columns, expected=len(profile.columns),
               divider=geo.divider is not None, table_blocks=len(blocks), notes=" | ".join(geo.notes))
    try:
        cut = r5.cut_pages([scan], profile, lambda crop, caption: "")[0]
        row.update(cut="ok", sections=sum(s.kind != "legend" for s in cut.segments))
    except Exception as exc:                                   # the full run would stop here
        row.update(cut=f"{type(exc).__name__}: {exc}"[:160], sections=0)
    return row

"""Every German full-run page, cut offline before anything is read: does the route's cut hold?

The certification drew 8 German pages; the full run reads 641. A page whose geometry the route
cannot read either stops the run (the apparatus cut raised on Böhmen 21, the first page of the
range, and on 73 0088 0232 …) or loses its table without a word (no blocks on 48 Böhmen and 9
Mähren table pages before ``lexikon_1910_image`` 0.3.0). This cuts every page the way the route
does — the page geometry, the apparatus sections (``run_r5_german_notes.cut_pages``, headings
left unread), the table blocks (``lexikon_1910_de_table.page_blocks``) — and lists what is odd:

  cut fails      the apparatus cut raised: the full run would stop here
  no table       no table block: the page's municipalities are not read
  no apparatus   no apparatus divider found: the page's symbols are not read
  fallback       a page read through a 0.3.0 rule (header by position, column template)

No model is called and nothing is written but the audit. Six workers; about 20 minutes.

    python utils/reference_tools/places/audit_full_run_pages.py [--editions de-moravia]

Writes ``output/frozen/page_audit.{csv,txt}``. Outside ``utils/lexikon_1910``.
"""
from __future__ import annotations

__version__ = "0.1.0"

import argparse
import csv
import sys
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
OUT = ROOT / "output" / "frozen"
EDITIONS = ("de-bohemia", "de-moravia", "de-silesia")


def _paths() -> None:
    for p in (str(ROOT), str(ROOT / "utils" / "lexikon_1910")):
        if p not in sys.path:
            sys.path.insert(0, p)


def audit_page(job: tuple[str, int]) -> dict:
    """One page's geometry, apparatus cut and table blocks."""
    _paths()
    import run_frozen as rf
    import run_r5_german_notes as r5
    from utils.lexikon_1910 import lexikon_1910_de_notes as lxdn
    from utils.lexikon_1910 import lexikon_1910_de_table as dt

    edition, scan = job
    profile = rf._profile(edition)
    row = {"edition": edition, "scan": scan}
    image = lxdn.render_native(profile.source_pdf, scan)
    geo, blocks = dt.page_blocks(image, profile, f"{scan:04d}")
    row.update(kind=geo.kind, columns=geo.n_columns, expected=len(profile.columns),
               divider=geo.divider is not None, table_blocks=len(blocks), notes=" | ".join(geo.notes))
    try:
        cut = r5.cut_pages([scan], profile, lambda crop, caption: "")[0]
        row.update(cut="ok", sections=sum(s.kind != "legend" for s in cut.segments))
    except Exception as exc:                                   # the full run would stop here
        row.update(cut=f"{type(exc).__name__}: {exc}"[:160], sections=0)
    return row


def flags(row: dict) -> list[str]:
    out = []
    if row["cut"] != "ok":
        out.append("cut fails")
    if not row["table_blocks"]:
        out.append("no table")
    if not row["divider"]:                 # sections themselves open only on a heading's read, which this skips
        out.append("no apparatus")
    if "by position" in row["notes"] or "template" in row["notes"]:
        out.append("fallback")
    return out
