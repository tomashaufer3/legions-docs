How it works

Every ruling in that table is ruled_by = "evidence" until the author confirms it, and utils.recheck.anchor_overrides.apply uses only author rulings by default. This page puts the rulings in front of the author, one card per ruling group, and its readback (as_author_rulings) writes the answers into the table.

Groups. Rulings that share a settlement id (old or new) are one card, and so are a de-bohemia row and the cz row the register links it to (cz_muni_seq / cz_settlement_seq): a coupled set — Vejprty cz 2/3 + 3/2; Kostomlaty cz 14/0 + 15/0 and the parts; Velká Úpa II/III in both volumes; Blatiny 27/0 + 27/1 with Lhota 302024049/31/0 — is confirmed or rejected as a whole, since apply refuses one settlement on two municipalities of an edition. A twin pair (cz + de-bohemia) is one card.

The card shows for every row: edition, court, muni_seq/settlement_seq, the printed cell (the register's printed_name and, on a Czech page, the text layer's line), the register's 1910 population and houses, the frozen anchor and any generated override, the proposed settlement with its lexikon entry (name_1910, status, okres_all, parent), the ČSÚ lexicon 1869–2005 díl I population of both settlements, the settlement's point against the register's court, and the ruling's basis. Beside it, the register's page (rendered from the volume's PDF, opened at the row on a Czech page) and the court's parts sections the frozen run cut on that page — never from a page excluded keeps out (the savings-bank census's exclusion, savings_bank_census.skipped_pages: sealed, reserve, calibration and gold). A row on such a page shows text only, and the card says so.

Answers: confirm / reject / cannot tell, plus a note, stored in the page's db collection cards (doc id = card id) with the fingerprint of the rows the card showed. The main session reads the collection back into output/review_readback/anchor_rulings/rulings.json; as_author_rulings then sets ruled_by = "author" on every row of a confirmed card (its basis gains the date and the note), moves a rejected card's rows to reference/places/anchor_rulings_rejected.csv, and leaves cannot tell and unanswered cards alone. A card whose rows changed since the page was built is refused. dry_run (the default) writes nothing.

def groups(rows: list[dict], places=None) -> list[list[dict]]:
    """The rulings in groups: two rows are one group when they share a settlement id (old or new) or the
    register links one as the other's twin. Groups in the order of their first row; rows in file order."""
    parent = list(range(len(rows)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        parent[find(i)] = find(j)

    holder = {}
    for i, r in enumerate(rows):
        for sid in (r.get("old_settlement_id"), r.get("settlement_id")):
            if sid:
                union(i, holder.setdefault(sid, i))
    for i, j in _twin_links(rows, places):
        union(i, j)
    out: dict[int, list[dict]] = {}
    for i, r in enumerate(rows):
        out.setdefault(find(i), []).append(r)
    return list(out.values())

def cz_line(page: str, pop, houses, name: str) -> dict | None:
    """The Czech text layer's line printing ``pop`` on ``page``: ``{text, y}`` (``y`` a share of the page
    height), the line also printing ``houses`` and nearest the printed name preferred; None if no line."""
    import run_lexikon_1910_pilot as pilot
    from utils.lexikon_1910 import lexikon_1910 as lx
    path = pilot.CZ_PAGES / f"{str(page).zfill(4)}.alto.xml"
    if pop is None or not path.exists():
        return None
    alto = lx.read_alto(path)
    best = None
    for hit in (s for s in alto.strings if s[4] == str(pop)):
        mid = hit[0] + hit[3] / 2
        line = sorted((s for s in alto.strings if abs(s[0] + s[3] / 2 - mid) < hit[3] * 0.6), key=lambda s: s[1])
        words = [s[4] for s in line]
        head = " ".join(s[4] for s in line if s[1] < hit[1])
        score = (houses is not None and str(houses) in words,
                 difflib.SequenceMatcher(None, head.lower(), str(name or "").lower()).ratio())
        if best is None or score > best[0]:
            best = (score, {"text": " ".join(words), "y": round(mid / alto.height, 4)})
    return best[1] if best else None

def card_images(facts: list[dict], out: Path, inline: bool = False) -> tuple[list[dict], list[str]]:
    """``(images, withheld)`` for one card: per (edition, page) the register page and the court's parts
    sections, each once; a page :func:`excluded` keeps out gives no image, only its line in ``withheld``.
    An image is a file under ``out/img`` (``src`` relative) or, with ``inline``, a data URI."""
    from PIL import Image
    images, withheld, seen = [], [], set()
    for f in facts:
        for pg in f.get("pages", []):
            if (f["edition"], pg) in seen:
                continue
            seen.add((f["edition"], pg))
            if f["excluded"].get(pg):
                withheld.append(f"{f['edition']} scan {pg}: {f['excluded'][pg]}; no image of it is shown")
                continue
            shots = [("page", f"{f['edition']}_{pg}_page.jpg", lambda e=f["edition"], p=pg: render_page(e, p),
                      PAGE_WIDTH, f"{f['edition']} scan {pg}, the register page")]
            for path in parts_sections(f["edition"], pg, f.get("court_id")):
                shots.append(("parts", f"{f['edition']}_{path.stem}.jpg", lambda q=path: Image.open(q),
                              SECTION_WIDTH, f"{f['edition']} scan {pg}, parts section {path.stem.split('_', 1)[1]}"))
            for kind, name, load, width, caption in shots:
                target = Path(out) / "img" / name
                if not target.exists():
                    target.parent.mkdir(parents=True, exist_ok=True)
                    with load() as im:
                        target.write_bytes(_jpeg_bytes(im, width))
                src = (f"data:image/jpeg;base64,{base64.b64encode(target.read_bytes()).decode()}"
                       if inline else f"img/{name}")
                ys = [x["line"]["y"] for x in facts if kind == "page" and x["edition"] == f["edition"]
                      and pg in x.get("pages", []) and x.get("line")]
                images.append({"kind": kind, "edition": f["edition"], "page": pg, "src": src,
                               "caption": caption, "y": ys[0] if ys else None, "file": f"img/{name}"})
    return images, withheld

def _twin_links(rows: list[dict], places=None) -> list[tuple[int, int]]:
    """Index pairs ``(de-bohemia row, cz row)`` the register links (``cz_muni_seq``/``cz_settlement_seq``)."""
    if places is None:
        return []
    at = {row_key(r): i for i, r in enumerate(rows)}
    p = places[places["edition"] == "de-bohemia"]
    link = {f"de-bohemia|{c}|{m}|{s}": f"cz|{c}|{int(cm)}|{int(cs)}"
            for c, m, s, cm, cs in zip(p["court_key"].astype(str), p["muni_seq"], p["settlement_seq"],
                                       p["cz_muni_seq"], p["cz_settlement_seq"])
            if cm is not None and cs is not None and str(cm) != "<NA>" and str(cs) != "<NA>"}
    return [(at[k], at[link[k]]) for k in at if k in link and link[k] in at]


def groups(rows: list[dict], places=None) -> list[list[dict]]:
    """The rulings in groups: two rows are one group when they share a settlement id (old or new) or the
    register links one as the other's twin. Groups in the order of their first row; rows in file order."""
    parent = list(range(len(rows)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        parent[find(i)] = find(j)

    holder = {}
    for i, r in enumerate(rows):
        for sid in (r.get("old_settlement_id"), r.get("settlement_id")):
            if sid:
                union(i, holder.setdefault(sid, i))
    for i, j in _twin_links(rows, places):
        union(i, j)
    out: dict[int, list[dict]] = {}
    for i, r in enumerate(rows):
        out.setdefault(find(i), []).append(r)
    return list(out.values())

    return "ar-" + re.sub(r"[^a-z0-9]+", "-", ids[0].lower()).strip("-")

    for g in groups(rows, places):
        cid = card_id(g)
        if cid in out:
            raise ValueError(f"two ruling groups share the card id {cid}")
        out[cid] = g
    return out

def _val(v):
    """A plain JSON value: None for a missing one, int for an integral number."""
    import pandas as pd
    if v is None or (not isinstance(v, str) and pd.isna(v)):
        return None
    if isinstance(v, float) and v.is_integer():
        return int(v)
    if hasattr(v, "item"):
        return v.item()
    return v


def lexikon_entry(sid: str, frames: dict) -> dict:
    """The settlement's lexikon row, díl I populations and point (no court yet)."""
    if not sid:
        return {}
    s, d, g = frames["settlements"], frames["dil1"], frames["geo"]
    out = {"settlement_id": sid}
    row = s[s["settlement_id"] == sid]
    if len(row):
        r = row.iloc[0]
        out.update({k: _val(r[k]) for k in ("cs_name", "name_1910", "status_1910", "okres_id", "okres_all",
                                             "parent_1910", "de_name")})
    else:
        out["missing"] = True
    dr = d[d["settlement_id"] == sid]
    if len(dr):
        out.update({k: _val(dr.iloc[0][k]) for k in ("dil1_obec_pop", "dil1_part_pop", "dil1_parts")})
    gr = g[g["settlement_id"] == sid]
    if len(gr):
        r = gr.iloc[0]
        out.update({"lat": _val(r["lat"]), "lon": _val(r["lon"]), "coord_source": _val(r["coord_source"]),
                    "dil1_obec_name": _val(r["obec_name"]), "dil1_okres_name": _val(r["okres_name"])})
    return out


def _locate(entry: dict, register_court, courts, court_names: dict) -> None:
    """Add the court the point lies in (plain containment) and its agreement with the register court."""
    if not entry or entry.get("lat") is None or courts is None:
        return
    cid = int(courts.locate([entry["lon"]], [entry["lat"]])[0])
    entry["point_court_id"] = cid or None
    entry["point_court_cs"] = court_names.get(cid)
    if register_court is not None:
        entry["point_in_register_court"] = cid == int(register_court)
        if cid != int(register_court) and int(register_court) in courts.courts:
            entry["point_km_from_register_court"] = round(
                courts.distance_m(int(register_court), entry["lon"], entry["lat"]) / 1000, 1)


def holders(sid: str, frames: dict, skip: set) -> list[dict]:
    """Every register place (not in ``skip``) whose frozen anchor or generated override is ``sid``."""
    p, o = frames["places"], frames["overrides"]
    out = []
    for r in p[p["settlement_id"] == sid].itertuples(index=False):
        k = f"{r.edition}|{r.court_key}|{r.muni_seq}|{r.settlement_seq}"
        if k not in skip:
            out.append({"key": k, "printed_name": r.printed_name, "pop_total": _val(r.pop_total),
                        "place_level": r.place_level, "how": f"frozen ({r.anchor_status})"})
    if len(o):
        for r in o[o["settlement_id"].astype(str) == sid].itertuples(index=False):
            k = f"{r.edition}|{r.court_key}|{r.muni_seq}|{r.settlement_seq}"
            if k not in skip:
                out.append({"key": k, "printed_name": r.printed_name, "pop_total": _val(r.pop_total),
                            "place_level": r.place_level, "how": f"generated override ({r.rule})"})
    return out


def row_facts(r: dict, frames: dict, courts=None, court_names: dict | None = None) -> dict:
    """Everything a card shows for one ruling row (text only; images are added by :func:`card_images`)."""
    p = frames["places"]
    hit = p[(p["edition"] == r["edition"]) & (p["court_key"] == str(r["court_key"]))
            & (p["muni_seq"] == int(r["muni_seq"])) & (p["settlement_seq"] == int(r["settlement_seq"]))]
    reg = hit.iloc[0] if len(hit) else None
    o = frames["overrides"]
    gen = (o[(o["edition"] == r["edition"]) & (o["court_key"].astype(str) == str(r["court_key"]))
             & (o["muni_seq"] == int(r["muni_seq"])) & (o["settlement_seq"] == int(r["settlement_seq"]))]
           if len(o) else o)
    court_id = _val(reg["court_id"]) if reg is not None else None
    out = {**fingerprint(r), "edition": r["edition"], "court_key": r["court_key"], "muni_seq": r["muni_seq"],
           "settlement_seq": r["settlement_seq"], "place_level": r["place_level"], "basis": r["basis"],
           "ruled_by": r["ruled_by"], "ruling_pop": r["pop_total"], "in_register": reg is not None}
    if reg is not None:
        out.update({"court_cs": reg["court_cs"], "court_id": court_id, "pages": _pages(reg["pages"]),
                    "register_name": reg["printed_name"], "pop_total": _val(reg["pop_total"]),
                    "houses": _val(reg["houses"]), "frozen_anchor": _val(reg["settlement_id"]),
                    "anchor_status": reg["anchor_status"], "district": reg["district"],
                    "components": _val(reg["components"])})
    out["generated"] = ({"settlement_id": str(gen.iloc[0]["settlement_id"]), "rule": gen.iloc[0]["rule"]}
                        if len(gen) else None)
    out["proposed"] = lexikon_entry(r.get("settlement_id") or "", frames)
    out["current"] = lexikon_entry(r.get("old_settlement_id") or out.get("frozen_anchor") or "", frames)
    for side in ("proposed", "current"):
        _locate(out[side], court_id, courts, court_names or {})
    out["excluded"] = {pg: why for pg in out.get("pages", []) if (why := excluded(r["edition"], pg))}
    return out

def _jpeg_bytes(image, width: int) -> bytes:
    image = image.convert("RGB")
    if image.width > width:
        image = image.resize((width, round(image.height * width / image.width)))
    buf = io.BytesIO()
    image.save(buf, "JPEG", quality=JPEG_QUALITY, optimize=True)
    return buf.getvalue()


def cz_line(page: str, pop, houses, name: str) -> dict | None:
    """The Czech text layer's line printing ``pop`` on ``page``: ``{text, y}`` (``y`` a share of the page
    height), the line also printing ``houses`` and nearest the printed name preferred; None if no line."""
    import run_lexikon_1910_pilot as pilot
    from utils.lexikon_1910 import lexikon_1910 as lx
    path = pilot.CZ_PAGES / f"{str(page).zfill(4)}.alto.xml"
    if pop is None or not path.exists():
        return None
    alto = lx.read_alto(path)
    best = None
    for hit in (s for s in alto.strings if s[4] == str(pop)):
        mid = hit[0] + hit[3] / 2
        line = sorted((s for s in alto.strings if abs(s[0] + s[3] / 2 - mid) < hit[3] * 0.6), key=lambda s: s[1])
        words = [s[4] for s in line]
        head = " ".join(s[4] for s in line if s[1] < hit[1])
        score = (houses is not None and str(houses) in words,
                 difflib.SequenceMatcher(None, head.lower(), str(name or "").lower()).ratio())
        if best is None or score > best[0]:
            best = (score, {"text": " ".join(words), "y": round(mid / alto.height, 4)})
    return best[1] if best else None

def card_images(facts: list[dict], out: Path, inline: bool = False) -> tuple[list[dict], list[str]]:
    """``(images, withheld)`` for one card: per (edition, page) the register page and the court's parts
    sections, each once; a page :func:`excluded` keeps out gives no image, only its line in ``withheld``.
    An image is a file under ``out/img`` (``src`` relative) or, with ``inline``, a data URI."""
    from PIL import Image
    images, withheld, seen = [], [], set()
    for f in facts:
        for pg in f.get("pages", []):
            if (f["edition"], pg) in seen:
                continue
            seen.add((f["edition"], pg))
            if f["excluded"].get(pg):
                withheld.append(f"{f['edition']} scan {pg}: {f['excluded'][pg]}; no image of it is shown")
                continue
            shots = [("page", f"{f['edition']}_{pg}_page.jpg", lambda e=f["edition"], p=pg: render_page(e, p),
                      PAGE_WIDTH, f"{f['edition']} scan {pg}, the register page")]
            for path in parts_sections(f["edition"], pg, f.get("court_id")):
                shots.append(("parts", f"{f['edition']}_{path.stem}.jpg", lambda q=path: Image.open(q),
                              SECTION_WIDTH, f"{f['edition']} scan {pg}, parts section {path.stem.split('_', 1)[1]}"))
            for kind, name, load, width, caption in shots:
                target = Path(out) / "img" / name
                if not target.exists():
                    target.parent.mkdir(parents=True, exist_ok=True)
                    with load() as im:
                        target.write_bytes(_jpeg_bytes(im, width))
                src = (f"data:image/jpeg;base64,{base64.b64encode(target.read_bytes()).decode()}"
                       if inline else f"img/{name}")
                ys = [x["line"]["y"] for x in facts if kind == "page" and x["edition"] == f["edition"]
                      and pg in x.get("pages", []) and x.get("line")]
                images.append({"kind": kind, "edition": f["edition"], "page": pg, "src": src,
                               "caption": caption, "y": ys[0] if ys else None, "file": f"img/{name}"})
    return images, withheld


# ── the page ──────────────────────────────────────────────────────────────────
_LOWER = {"nad", "pod", "u", "na", "v", "ve", "pri", "při", "an", "der", "in", "im"}
_ROMAN = re.compile(r"^(?:i|ii|iii|iv|v|vi|vii|viii|ix|x)$", re.IGNORECASE)


def name_case(name: str) -> str:
    """A lexikon headword (``PEC POD SNĚŽKOU``, ``VELKÁ ÚPA II``) in ordinary case: ``Pec pod Sněžkou``."""
    words = str(name or "").split()
    out = []
    for k, w in enumerate(words):
        low = w.lower()
        out.append(w.upper() if _ROMAN.match(w) and k else low if k and low in _LOWER else low[:1].upper() + low[1:])
    return " ".join(out)


def card_title(group_facts: list[dict]) -> str:
    names = []
    for f in group_facts:
        e = f["proposed"] or f["current"]
        n = name_case(e.get("cs_name") or f["printed_name"])
        if n not in names:
            names.append(n)
    return " / ".join(names)


def build_items(rulings: Path = RULINGS, out: Path = OUT, final_dir: Path = FINAL, inline: bool = False,
                with_images: bool = True) -> list[dict]:
    """The cards, in table order, with every row's facts and the card's images."""
    from utils.recheck import anchor_overrides as ao
    frames = load_frames(final_dir)
    _fields, rows = read_rulings(rulings)
    courts = ao.Courts()
    p = frames["places"]
    court_names = {int(c): n for c, n in zip(p["court_id"], p["court_cs"]) if c is not None and str(c) != "<NA>"}
    items = []
    for cid, group in cards(rows, frames["places"]).items():
        facts = [row_facts(r, frames, courts, court_names) for r in group]
        for f in facts:
            f["line"] = (cz_line(f["pages"][0], f.get("pop_total"), f.get("houses"), f.get("register_name"))
                         if f["edition"] == "cz" and f.get("pages") and not f["excluded"] else None)
        skip = {f["key"] for f in facts}
        sids = sorted({s for f in facts for s in (f["old"], f["new"], f.get("frozen_anchor") or "") if s})
        images, withheld = card_images(facts, out, inline) if with_images else ([], [])
        items.append({"id": cid, "title": card_title(facts), "rows": facts,
                      "fingerprint": [fingerprint(r) for r in group],
                      "holders": {s: holders(s, frames, skip) for s in sids},
                      "places": len({(f["edition"], f["court_key"], f["muni_seq"]) for f in facts}),
                      "images": images, "withheld": withheld})
    return items


def page(items: list[dict], out: Path = OUT) -> Path:
    html = TEMPLATE.read_text(encoding="utf-8").replace(

def build(rulings: Path = RULINGS, out: Path = OUT, final_dir: Path = FINAL, inline: bool = False) -> dict:
    items = build_items(rulings, out, final_dir, inline)
    path = page(items, out)
    (Path(out) / "card_key.json").write_bytes(json.dumps(
        {i["id"]: i["fingerprint"] for i in items}, ensure_ascii=False, indent=1).encode("utf-8") + b"\n")
    files = sorted({im["file"] for i in items for im in i["images"]})
    return {"cards": len(items), "rows": sum(len(i["rows"]) for i in items),
            "images": sum(len(i["images"]) for i in items), "image_files": files,
            "withheld": {i["id"]: i["withheld"] for i in items if i["withheld"]},
            "page": str(path), "page_bytes": path.stat().st_size,
            "image_bytes": sum((Path(out) / f).stat().st_size for f in files)}

def _stale(card_rows: list[dict], shown) -> str:
    """'' when the answer's fingerprint (if it carries one) is the card's rows now, else what differs."""
    if not shown:
        return ""
    now = sorted((json.dumps(fingerprint(r), sort_keys=True) for r in card_rows))
    then = sorted((json.dumps({k: s.get(k, "") for k in ("key", "printed_name", "action", "old", "new")},
                              sort_keys=True) for s in shown))
    return "" if now == then else "the card's rows changed since the page was built"


def as_author_rulings(answers: dict, rulings: Path = RULINGS, rejected: Path = REJECTED,
                      dry_run: bool = True, on: str | None = None, places=None) -> dict:
    """Write the answered cards into the rulings table: *confirm* sets ``ruled_by = "author"`` on all the
    card's rows (``basis`` gains ``author: confirmed <on> on the anchor-rulings page`` and the note);
    *reject* moves them to ``rejected`` with the date and the note; *cannot tell* and unanswered cards
    change nothing. A card id no longer in the table, an unknown answer, or rows changed since the page
    was built (the answer's ``rows`` fingerprint) are refused, listed in ``problems``. The result is
    checked by ``anchor_overrides.check_rulings`` and, given ``places``, by ``apply`` in its default mode.
    Nothing is written when ``dry_run``."""
    from utils.recheck import anchor_overrides as ao
    import pandas as pd
    answers = answers.get("rulings", answers) if isinstance(answers, dict) else {}
    on = on or dt.date.today().isoformat()
    fields, rows = read_rulings(rulings)
    by_card = cards(rows, places)
    report = {"confirmed": [], "rejected": [], "cannot_tell": [], "unanswered": [], "problems": [],
              "rows_confirmed": [], "rows_rejected": [], "already_author": [], "dry_run": bool(dry_run)}
    confirm, reject, notes = set(), set(), {}
    for cid, a in answers.items():
        value = str((a or {}).get("value") or "")
        if cid not in by_card:
            report["problems"].append({"card": cid, "problem": "no such card in the rulings table"})
        elif value not in ANSWERS:
            report["problems"].append({"card": cid, "problem": f"answer {value!r}"})
        elif (why := _stale(by_card[cid], (a or {}).get("rows"))):
            report["problems"].append({"card": cid, "problem": why})
        elif value == "cannot-tell":
            report["cannot_tell"].append(cid)
        else:
            keys = {row_key(r) for r in by_card[cid]}
            (confirm if value == "confirm" else reject).update(keys)
            report["confirmed" if value == "confirm" else "rejected"].append(cid)
            notes.update({k: str((a or {}).get("note") or "").strip() for k in keys})
    report["unanswered"] = [c for c in by_card if c not in answers]
    kept, gone = [], []
    for r in rows:
        k = row_key(r)
        if k in reject:
            gone.append({**r, "rejected_on": on, "rejected_note": notes.get(k, "")})
            report["rows_rejected"].append(k)
            continue
        if k in confirm:
            if ao.is_confirmed(r["ruled_by"]):
                report["already_author"].append(k)
            else:
                note = notes.get(k, "")
                r = {**r, "ruled_by": "author",
                     "basis": f"{r['basis']} | author: confirmed {on} on the anchor-rulings page"
                              + (f": {note}" if note else "")}
                report["rows_confirmed"].append(k)
        kept.append(r)
    new_text = _csv_text(fields, kept)
    frame = pd.read_csv(io.StringIO(new_text), dtype=str, keep_default_na=False, na_values=[""])
    ao.check_rulings(frame)
    if places is not None:
        applied = ao.apply(places, rulings=frame)
        report["applied_by_default"] = len(applied.attrs["anchor_rulings_applied"])
    old_text = Path(rulings).read_text(encoding="utf-8")
    report["diff"] = "".join(difflib.unified_diff(old_text.splitlines(True), new_text.splitlines(True),
                                                  str(rulings.name), str(rulings.name) + " (after)"))
    report["rows_after"] = len(kept)
    if not dry_run:
        if new_text != old_text:
            Path(rulings).write_bytes(new_text.encode("utf-8"))
        if gone:
            _append_rejected(gone, fields, rejected)
    return report


def _csv_text(fields: list[str], rows: list[dict]) -> str:
    buf = io.StringIO()
    w = csv.DictWriter(buf, fieldnames=fields, lineterminator="\n", extrasaction="ignore")
    w.writeheader()
    w.writerows(rows)
    return buf.getvalue()


def _append_rejected(gone: list[dict], fields: list[str], path: Path) -> None:
    cols = fields + [c for c in REJECTED_EXTRA if c not in fields]
    before = read_rulings(path)[1] if Path(path).exists() else []
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    Path(path).write_bytes(_csv_text(cols, before + gone).encode("utf-8"))
