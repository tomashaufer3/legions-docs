How it works

The German table reader labels every row with its political district (P. B. Boskowitz), not its court, and the register files a row under that label wherever courts.csv knows it on the row's page (lexikon_1910_register._table_court). Where a political district holds two or three courts, a page that carries one court and opens the next files the first court's rows under the second, or under the page's only court:

  • Moravian 0031: Blansko's nos. 26–49 (Oberlhotta … Žernovník), printed above II. Gerichtsbezirk Boskowitz, filed as Boskovice 26–49; 0143: Rožnov's nos. 5–12 as Valašské Meziříčí 5–12. The harm is double: the claims of the real municipalities find no table row (no name, no population), and the other court's own nos. 26–44 or 5–12 take the carried rows' names and populations, since the register keeps the first row that names a number.
  • Silesian 0027: Vidnava's nos. 1–2 (Alt-Rothwasser, Domsdorf), printed under III. GB. Weidenau, given Javorník, the one court the apparatus read on that page, whose nos. 1–2 it already had.

The table prints the courts itself: a heading row before each court's municipalities — I. Gerichtsbezirk Blansko 49 …, II. GB. Wallachisch Meseritsch …, or the court's bare name with its counts (Rožnov pod Radhoštěm ) 12 …). court_rows walks the table in printed order and gives each row the court of the heading above it (resolved through the route's court crosswalk, or by the court's Czech name); a political district's heading (P. B.) closes the court. The walk alone is not trusted — a heading it misses lets a court run on (Kunštát read to 80 against its printed 60) — so a row moves to the heading's court only on evidence (evidence):

  • its name — the table row's printed name matches the heading court's list of parts for that number (components) and not the list of the court the register gave it;
  • where neither list names it, the page — the apparatus has that number of the heading's court on the row's page and not of the register's (the register's own fallback, which the reader's label overrode: Blansko 26–48 on 0031 has offices there, Boskovice 26–48 none).

A row with no evidence moves only in a segment (a page × the two courts) whose other rows all point to the heading's court; a row whose evidence points to the register's court never moves. Over the Mähren and Schlesien tables (2026-11-10) every segment pointed one way: Blansko 0031 20 to 0, Rožnov 0143 8 to 0, and the walk's overruns (Kunštát on 0036–0039, Bohumín on 0023–0026) 0 to 113.

Where the row's own label and the heading name the same court but the page's courts.csv lacks it (the apparatus missed the heading: Vidnava on Silesian 0027, whose parts the frozen parts list still files under Javorník, so its names point the wrong way), the court is added to courts.csv for that page: two readings of the table agree. Each heading's printed count is set beside the numbers below it (count_check), a report, not a gate.

Nothing in the frozen register changes: mend hands register_rechecked a run with the table's labels and courts mended, and the route's own register keeps the rows as they were read.

def heading_court(name: str, known: dict, crosswalk: dict) -> Court | None:
    """The court a table heading row names, or ``None`` (not a court heading)."""
    text = str(name or "")
    if POLITICAL.match(text):
        return None
    rest = PREFIX.sub("", text, count=1)
    cut = NAME_END.search(rest)
    head = (rest[:cut.start()] if cut else rest).strip(" ,.;:")
    count = re.search(r"\d+", rest[cut.start():]) if cut else None
    for piece in [head, *head.split(",")]:
        piece = piece.strip()
        if len(piece) < 3:
            continue
        czech = lxdn.court_name(piece, crosswalk)[0] or piece
        hit = known.get(reg._label(czech))
        if hit:
            return Court(hit[0], hit[1], int(count.group()) if count else None)
    return None

def court_rows(table: pd.DataFrame, courts: pd.DataFrame) -> pd.DataFrame:
    """``[index, page, seq, row, old, court_id, court_cs, heading]`` for every channel-A row under a
    court heading: the court the table prints above it."""
    cols = ["index", "page", "seq", "row", "old", "court_id", "court_cs", "heading"]
    if table.empty or "channel" not in table:
        return pd.DataFrame(columns=cols)
    known, crosswalk = known_courts(courts), lxdn.load_court_crosswalk()
    out, current, heading = [], None, None
    for i, r in table[table["channel"] == "A"].iterrows():
        seq = str(r["seq"]).strip()
        if not seq.isdigit():
            if POLITICAL.match(str(r["name"])):
                current = heading = None
            else:
                found = heading_court(r["name"], known, crosswalk)
                if found:
                    current, heading = found, i
            continue
        if current is not None:
            out.append({"index": i, "page": str(r["page"]).zfill(4), "seq": seq, "row": str(r["row"]),
                        "old": str(r.get("court_label", "")), "court_id": current.court_id,
                        "court_cs": current.court_cs, "heading": heading})
    return pd.DataFrame(out, columns=cols)

def count_check(table: pd.DataFrame, rows: pd.DataFrame, courts: pd.DataFrame) -> pd.DataFrame:
    """Per court heading: its printed count against the highest municipality number below it."""
    if rows.empty:
        return pd.DataFrame(columns=["heading", "court_cs", "printed", "read", "agrees"])
    known, crosswalk = known_courts(courts), lxdn.load_court_crosswalk()
    out = []
    for h, g in rows.groupby("heading"):
        court = heading_court(table.loc[h, "name"], known, crosswalk)
        top = max(int(s) for s, row in zip(g["seq"], g["row"]) if row == "0") if (g["row"] == "0").any() else None
        out.append({"heading": h, "court_cs": g["court_cs"].iloc[0], "printed": court.count if court else None,
                    "read": top, "agrees": court is not None and court.count == top})
    return pd.DataFrame(out)

def apparatus(run: reg.RunInput) -> tuple:
    """The register's units from the claims and the parts list (``build_register``'s own order, the
    table not yet in) and ``{(court_key, muni): name keys}`` off each municipality's parts."""
    units = reg._Units(run)
    reg._add_claim_units(units, run.inventory)
    reg._add_claim_units(units, run.abbreviations)
    reg._add_parts_list(units, run.parts)
    names = defaultdict(set)
    for (ck, muni, _seq), place in units.places.items():
        for _pos, name, _topo in place["components"]:
            names[(ck, muni)] |= _pieces(name)
    return units, names

def evidence(units, names, page: str, muni: int, table_name: str, heading: str, given: str | None) -> str:
    """``heading``, ``given``, ``both`` or ``none``: which court the row's name — or, where neither
    list of parts names it, the apparatus on its page — points to."""
    mine = _pieces(table_name)
    h, g = _named(mine, names[(heading, muni)]), bool(given) and _named(mine, names[(given, muni)])
    if not (h or g):
        def on_page(ck):
            return any(page in pl["pages"] for (c, m, _s), pl in units.places.items() if c == ck and m == muni)
        h, g = on_page(heading), bool(given) and on_page(given)
    return "both" if h and g else "heading" if h else "given" if g else "none"

def decide(run: reg.RunInput) -> pd.DataFrame:
    """Every channel-A row the table's headings put under another court than the register gives it,
    with the evidence and whether it moves (the rule in the module's docstring)."""
    rows = court_rows(run.table, run.courts)
    units, names = apparatus(run)
    out, last = [], None
    for r in rows.itertuples():
        t = run.table.loc[r.index]
        if r.row == "0":
            given = reg._table_court(units, r.page, int(r.seq), t.get("court_label", ""))
            last = None if given == r.court_id else {
                "index": r.index, "page": r.page, "seq": r.seq, "name": reg.clean_display(t["name"]),
                "label": r.old, "given": given or "", "court_id": r.court_id, "court_cs": r.court_cs,
                "evidence": evidence(units, names, r.page, int(r.seq), t["name"], r.court_id, given),
                "parts_rows": []}
            if last:
                out.append(last)
        elif last is not None:
            last["parts_rows"].append(r.index)
    frame = pd.DataFrame(out, columns=["index", "page", "seq", "name", "label", "given", "court_id",
                                       "court_cs", "evidence", "parts_rows"])
    seg = frame.groupby(["page", "court_id", "given"]).evidence.agg(
        lambda e: (e == "heading").sum() > 0 and (e == "given").sum() == 0)
    frame["moves"] = [e == "heading" or (e in ("none", "both") and seg[(p, c, g)])
                      for e, p, c, g in zip(frame["evidence"], frame["page"], frame["court_id"], frame["given"])]
    return frame

def missing_courts(run: reg.RunInput, units=None) -> pd.DataFrame:
    """Courts to add to ``courts.csv``: a page whose table rows the reader labelled with the very
    court the heading above them names, which the page's ``courts.csv`` lacks — **and whose numbers
    the apparatus holds under that court on that page** (at least half of them; 0.1.1). Without
    that, a heading walk that runs on past its court files another court's rows under it: 0.1.0 put
    Fryštát's nos. 15–24 on Silesian 0023 under Bohumín, whose nos. 15–24 hold no claim, and
    Kutná Hora's on German Böhmen 0228 the same way. Vidnava 1–2 on 0027 hold their claims there
    (moved by §3), so their court is added."""
    rows = court_rows(run.table, run.courts)
    have = {(str(r["page"]).zfill(4), r["district"]) for _i, r in run.courts.iterrows()}
    agree = _where(rows, [reg._label(o) == reg._label(c) and r == "0"
                          for o, c, r in zip(rows["old"], rows["court_cs"], rows["row"])])
    agree = agree.assign(district=[reg._label(c) for c in agree["court_cs"]])
    agree = _where(agree, [(p, d) not in have for p, d in zip(agree["page"], agree["district"])])
    if agree.empty:
        return pd.DataFrame(columns=["page", "district", "court_id", "court_cs", "court_source"])
    units = units or apparatus(run)[0]
    held = {(ck, m, page) for (ck, m, _s), pl in units.places.items() for page in pl["pages"]}
    keep = []
    for (page, district), g in agree.groupby(["page", "district"]):
        on_page = sum((g["court_id"].iloc[0], int(s), page) in held for s in g["seq"])
        if on_page * 2 >= len(g):
            keep.append(g.iloc[0])
    keep = pd.DataFrame(keep)
    if keep.empty:
        return pd.DataFrame(columns=["page", "district", "court_id", "court_cs", "court_source"])
    return pd.DataFrame({"page": keep["page"], "district": keep["district"], "court_id": keep["court_id"],
                         "court_cs": keep["court_cs"], "court_source": "table heading and label, held on the page (D-202)"})

def alias_courts(run: reg.RunInput, path: Path = ALIASES) -> tuple[reg.RunInput, list[dict]]:
    """The run with every ``courts.csv`` row whose label the route left without a court given the
    court ``court_aliases_recheck.csv`` names for it (D-205): a garbled heading its real court, a
    court set up after 1910 a court id of its own inside its 1910 okres. Returns the rows filled."""
    aliases = pd.read_csv(path, dtype=str, keep_default_na=False) if Path(path).exists() else pd.DataFrame()
    mine = {r.label: r for r in aliases.itertuples() if r.edition == run.edition} if len(aliases) else {}
    if not mine:
        return run, []
    courts = run.courts.copy()
    filled = []
    for i, r in courts.iterrows():
        a = mine.get(r["district"])
        if a is not None and not str(r.get("court_id", "")).isdigit():
            courts.loc[i, ["court_id", "court_cs"]] = [a.court_id, a.court_cs]
            if "court_source" in courts.columns:
                courts.loc[i, "court_source"] = f"recheck alias, {a.kind} (D-205)"
            filled.append({"page": str(r["page"]), "label": r["district"], "court_id": a.court_id, "kind": a.kind})
    return replace(run, courts=courts), filled

def mend(run: reg.RunInput) -> tuple[reg.RunInput, dict]:
    """The run with its unresolved court labels aliased (:func:`alias_courts`), the table rows that
    move relabelled with their printed court and the courts its pages lacked added, and what changed.
    German volumes only: the Czech table reader files by court (its heading rows name no German
    *Gerichtsbezirk*)."""
    if run.edition not in GERMAN:
        return run, {"edition": run.edition, "rows_moved": 0}
    run, aliased = alias_courts(run)
    frame = decide(run)
    moving = _where(frame, frame["moves"].tolist())
    table = run.table.copy()
    for r in moving.itertuples():
        table.loc[[r.index, *r.parts_rows], "court_label"] = r.court_cs
    have = {(str(r["page"]).zfill(4), r["district"]) for _i, r in run.courts.iterrows()}
    moved_to = moving.assign(district=[reg._label(c) for c in moving["court_cs"]]).drop_duplicates(["page", "district"])
    moved_to = _where(moved_to, [(p, d) not in have for p, d in zip(moved_to["page"], moved_to["district"])])
    added = pd.concat([missing_courts(run), pd.DataFrame({
        "page": moved_to["page"], "district": moved_to["district"], "court_id": moved_to["court_id"],
        "court_cs": moved_to["court_cs"], "court_source": "table heading, on evidence (D-202)"})],
        ignore_index=True).drop_duplicates(["page", "district"])
    courts = pd.concat([run.courts, added], ignore_index=True) if len(added) else run.courts
    checks = count_check(run.table, court_rows(run.table, run.courts), run.courts)
    report = {"edition": run.edition, "courts_aliased": aliased, "municipality_rows_disputed": len(frame),
              "evidence": frame["evidence"].value_counts().to_dict(), "rows_moved": int(frame["moves"].sum()),
              "moves": (moving.groupby(["page", "label", "court_cs"]).size().reset_index(name="municipalities")
                        .to_dict("records") if len(moving) else []),
              "courts_added": added[["page", "district", "court_id", "court_source"]].to_dict("records"),
              "headings": len(checks), "counts_disagree": int((~checks["agrees"]).sum()) if len(checks) else 0}
    return replace(run, table=table, courts=courts), report

The German table reader labels every row with its **political district** (*P. B. Boskowitz*), not
its court, and the register files a row under that label wherever ``courts.csv`` knows it on the
row's page (``lexikon_1910_register._table_court``). Where a political district holds two or three
courts, a page that carries one court and opens the next files the first court's rows under the
second, or under the page's only court:

* Moravian 0031: Blansko's nos. 26–49 (Oberlhotta … Žernovník), printed above *II. Gerichtsbezirk
  Boskowitz*, filed as Boskovice 26–49; 0143: Rožnov's nos. 5–12 as Valašské Meziříčí 5–12. The harm
  is double: the claims of the real municipalities find no table row (no name, no population), and
  the other court's own nos. 26–44 or 5–12 take the carried rows' names and populations, since the
  register keeps the first row that names a number.
* Silesian 0027: Vidnava's nos. 1–2 (Alt-Rothwasser, Domsdorf), printed under *III. GB. Weidenau*,
  given Javorník, the one court the apparatus read on that page, whose nos. 1–2 it already had.

The table prints the courts itself: a heading row before each court's municipalities — *I.
Gerichtsbezirk Blansko 49 …*, *II. GB. Wallachisch Meseritsch …*, or the court's bare name with its
counts (*Rožnov pod Radhoštěm **) 12 …*). :func:`court_rows` walks the table in printed order
and gives each row the court of the heading above it (resolved through the route's court crosswalk,
or by the court's Czech name); a political district's heading (*P. B.*) closes the court. The walk
alone is not trusted — a heading it misses lets a court run on (Kunštát read to 80 against its
printed 60) — so a row moves to the heading's court only on evidence (:func:`evidence`):

* **its name** — the table row's printed name matches the heading court's list of parts for that
  number (``components``) and not the list of the court the register gave it;
* where neither list names it, **the page** — the apparatus has that number of the heading's court
  on the row's page and not of the register's (the register's own fallback, which the reader's
  label overrode: Blansko 26–48 on 0031 has offices there, Boskovice 26–48 none).

A row with no evidence moves only in a segment (a page × the two courts) whose other rows all point
to the heading's court; a row whose evidence points to the register's court never moves. Over the
Mähren and Schlesien tables (2026-11-10) every segment pointed one way: Blansko 0031 20 to 0,
Rožnov 0143 8 to 0, and the walk's overruns (Kunštát on 0036–0039, Bohumín on 0023–0026) 0 to 113.

Where the row's own label and the heading name the same court but the page's ``courts.csv`` lacks
it (the apparatus missed the heading: Vidnava on Silesian 0027, whose parts the frozen parts list
still files under Javorník, so its names point the wrong way), the court is added to ``courts.csv``
for that page: two readings of the table agree. Each heading's printed count is set beside the
numbers below it (:func:`count_check`), a report, not a gate.

Nothing in the frozen register changes: :func:`mend` hands ``register_rechecked`` a run with the
table's labels and courts mended, and the route's own register keeps the rows as they were read.
"""

import sys
from collections import defaultdict
from dataclasses import dataclass, replace
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT), str(ROOT / "utils" / "lexikon_1910")]

from utils.lexikon_1910 import lexikon_1910_de_notes as lxdn          # noqa: E402
from utils.lexikon_1910 import lexikon_1910_notes as lxnt             # noqa: E402
from utils.lexikon_1910 import lexikon_1910_register as reg           # noqa: E402

GERMAN = ("de-bohemia", "de-moravia", "de-silesia")
#: Court labels the route's courts.csv left without a court: garbled headings, and the courts set up
#: after 1910 that the 1910 backbone lacks, each with its evidence (D-205).
ALIASES = ROOT / "reference" / "places" / "court_aliases_recheck.csv"
POLITICAL = re.compile(r"^\s*p\.\s*b\.", re.I)                        # a political district's heading
PREFIX = re.compile(r"^\s*(?:[IVX]+\.\s*)?(?:gerichtsbezirk|g\.\s?b\.|gb\.)?\s*", re.I)
NAME_END = re.compile(r"[\d\[\(*]")                                   # the counts, a gloss or a footnote


@dataclass(frozen=True)
class Court:
    court_id: str
    court_cs: str
    count: int | None          # the heading's own number of municipalities


def known_courts(courts: pd.DataFrame) -> dict[str, tuple[str, str]]:
    """``{normalised Czech court name or label: (court_id, court_cs)}`` over the whole run."""
    out = {}
    for _i, r in courts.iterrows():
        if str(r.get("court_id", "")).isdigit():
            for key in (reg._label(r.get("court_cs", "")), r["district"]):
                if key:
                    out.setdefault(key, (str(r["court_id"]), str(r.get("court_cs", ""))))
    return out


def heading_court(name: str, known: dict, crosswalk: dict) -> Court | None:
    """The court a table heading row names, or ``None`` (not a court heading)."""
    text = str(name or "")
    if POLITICAL.match(text):
        return None
    rest = PREFIX.sub("", text, count=1)
    cut = NAME_END.search(rest)
    head = (rest[:cut.start()] if cut else rest).strip(" ,.;:")
    count = re.search(r"\d+", rest[cut.start():]) if cut else None
    for piece in [head, *head.split(",")]:
        piece = piece.strip()
        if len(piece) < 3:
            continue
        czech = lxdn.court_name(piece, crosswalk)[0] or piece
        hit = known.get(reg._label(czech))
        if hit:
            return Court(hit[0], hit[1], int(count.group()) if count else None)
    return None


def court_rows(table: pd.DataFrame, courts: pd.DataFrame) -> pd.DataFrame:
    """``[index, page, seq, row, old, court_id, court_cs, heading]`` for every channel-A row under a
    court heading: the court the table prints above it."""
    cols = ["index", "page", "seq", "row", "old", "court_id", "court_cs", "heading"]
    if table.empty or "channel" not in table:
        return pd.DataFrame(columns=cols)
    known, crosswalk = known_courts(courts), lxdn.load_court_crosswalk()
    out, current, heading = [], None, None
    for i, r in table[table["channel"] == "A"].iterrows():
        seq = str(r["seq"]).strip()
        if not seq.isdigit():
            if POLITICAL.match(str(r["name"])):
                current = heading = None
            else:
                found = heading_court(r["name"], known, crosswalk)
                if found:
                    current, heading = found, i
            continue
        if current is not None:
            out.append({"index": i, "page": str(r["page"]).zfill(4), "seq": seq, "row": str(r["row"]),
                        "old": str(r.get("court_label", "")), "court_id": current.court_id,
                        "court_cs": current.court_cs, "heading": heading})
    return pd.DataFrame(out, columns=cols)


def count_check(table: pd.DataFrame, rows: pd.DataFrame, courts: pd.DataFrame) -> pd.DataFrame:
    """Per court heading: its printed count against the highest municipality number below it."""
    if rows.empty:
        return pd.DataFrame(columns=["heading", "court_cs", "printed", "read", "agrees"])
    known, crosswalk = known_courts(courts), lxdn.load_court_crosswalk()
    out = []
    for h, g in rows.groupby("heading"):
        court = heading_court(table.loc[h, "name"], known, crosswalk)
        top = max(int(s) for s, row in zip(g["seq"], g["row"]) if row == "0") if (g["row"] == "0").any() else None
        out.append({"heading": h, "court_cs": g["court_cs"].iloc[0], "printed": court.count if court else None,
                    "read": top, "agrees": court is not None and court.count == top})
    return pd.DataFrame(out)


def _where(frame: pd.DataFrame, mask) -> pd.DataFrame:
    """Rows by a boolean list — never columns, which an empty list would select."""
    return frame.loc[pd.Series(list(mask), index=frame.index, dtype=bool)]


def _pieces(name) -> set[str]:
    """A printed name's parts as match keys: each comma- or bracket-separated name, letters only."""
    text = reg.clean_display(name).replace("(", ",").replace(")", ",")
    return {k for k in (lxnt.normalise(p) for p in text.split(",")) if len(k) >= 4}


def _named(a: set, b: set) -> bool:
    return any(x == y or (min(len(x), len(y)) >= 6 and (x.startswith(y) or y.startswith(x))) for x in a for y in b)


def apparatus(run: reg.RunInput) -> tuple:
    """The register's units from the claims and the parts list (``build_register``'s own order, the
    table not yet in) and ``{(court_key, muni): name keys}`` off each municipality's parts."""
    units = reg._Units(run)
    reg._add_claim_units(units, run.inventory)
    reg._add_claim_units(units, run.abbreviations)
    reg._add_parts_list(units, run.parts)
    names = defaultdict(set)
    for (ck, muni, _seq), place in units.places.items():
        for _pos, name, _topo in place["components"]:
            names[(ck, muni)] |= _pieces(name)
    return units, names


def evidence(units, names, page: str, muni: int, table_name: str, heading: str, given: str | None) -> str:
    """``heading``, ``given``, ``both`` or ``none``: which court the row's name — or, where neither
    list of parts names it, the apparatus on its page — points to."""
    mine = _pieces(table_name)
    h, g = _named(mine, names[(heading, muni)]), bool(given) and _named(mine, names[(given, muni)])
    if not (h or g):
        def on_page(ck):
            return any(page in pl["pages"] for (c, m, _s), pl in units.places.items() if c == ck and m == muni)
        h, g = on_page(heading), bool(given) and on_page(given)
    return "both" if h and g else "heading" if h else "given" if g else "none"


def decide(run: reg.RunInput) -> pd.DataFrame:
    """Every channel-A row the table's headings put under another court than the register gives it,
    with the evidence and whether it moves (the rule in the module's docstring)."""
    rows = court_rows(run.table, run.courts)
    units, names = apparatus(run)
    out, last = [], None
    for r in rows.itertuples():
        t = run.table.loc[r.index]
        if r.row == "0":
            given = reg._table_court(units, r.page, int(r.seq), t.get("court_label", ""))
            last = None if given == r.court_id else {
                "index": r.index, "page": r.page, "seq": r.seq, "name": reg.clean_display(t["name"]),
                "label": r.old, "given": given or "", "court_id": r.court_id, "court_cs": r.court_cs,
                "evidence": evidence(units, names, r.page, int(r.seq), t["name"], r.court_id, given),
                "parts_rows": []}
            if last:
                out.append(last)
        elif last is not None:
            last["parts_rows"].append(r.index)
    frame = pd.DataFrame(out, columns=["index", "page", "seq", "name", "label", "given", "court_id",
                                       "court_cs", "evidence", "parts_rows"])
    seg = frame.groupby(["page", "court_id", "given"]).evidence.agg(
        lambda e: (e == "heading").sum() > 0 and (e == "given").sum() == 0)
    frame["moves"] = [e == "heading" or (e in ("none", "both") and seg[(p, c, g)])
                      for e, p, c, g in zip(frame["evidence"], frame["page"], frame["court_id"], frame["given"])]
    return frame


def missing_courts(run: reg.RunInput, units=None) -> pd.DataFrame:
    """Courts to add to ``courts.csv``: a page whose table rows the reader labelled with the very
    court the heading above them names, which the page's ``courts.csv`` lacks — **and whose numbers
    the apparatus holds under that court on that page** (at least half of them; 0.1.1). Without
    that, a heading walk that runs on past its court files another court's rows under it: 0.1.0 put
    Fryštát's nos. 15–24 on Silesian 0023 under Bohumín, whose nos. 15–24 hold no claim, and
    Kutná Hora's on German Böhmen 0228 the same way. Vidnava 1–2 on 0027 hold their claims there
    (moved by §3), so their court is added."""
    rows = court_rows(run.table, run.courts)
    have = {(str(r["page"]).zfill(4), r["district"]) for _i, r in run.courts.iterrows()}
    agree = _where(rows, [reg._label(o) == reg._label(c) and r == "0"
                          for o, c, r in zip(rows["old"], rows["court_cs"], rows["row"])])
    agree = agree.assign(district=[reg._label(c) for c in agree["court_cs"]])
    agree = _where(agree, [(p, d) not in have for p, d in zip(agree["page"], agree["district"])])
    if agree.empty:
        return pd.DataFrame(columns=["page", "district", "court_id", "court_cs", "court_source"])
    units = units or apparatus(run)[0]
    held = {(ck, m, page) for (ck, m, _s), pl in units.places.items() for page in pl["pages"]}
    keep = []
    for (page, district), g in agree.groupby(["page", "district"]):
        on_page = sum((g["court_id"].iloc[0], int(s), page) in held for s in g["seq"])
        if on_page * 2 >= len(g):
            keep.append(g.iloc[0])
    keep = pd.DataFrame(keep)
    if keep.empty:
        return pd.DataFrame(columns=["page", "district", "court_id", "court_cs", "court_source"])
    return pd.DataFrame({"page": keep["page"], "district": keep["district"], "court_id": keep["court_id"],
                         "court_cs": keep["court_cs"], "court_source": "table heading and label, held on the page (D-202)"})


def alias_courts(run: reg.RunInput, path: Path = ALIASES) -> tuple[reg.RunInput, list[dict]]:
    """The run with every ``courts.csv`` row whose label the route left without a court given the
    court ``court_aliases_recheck.csv`` names for it (D-205): a garbled heading its real court, a
    court set up after 1910 a court id of its own inside its 1910 okres. Returns the rows filled."""
    aliases = pd.read_csv(path, dtype=str, keep_default_na=False) if Path(path).exists() else pd.DataFrame()
    mine = {r.label: r for r in aliases.itertuples() if r.edition == run.edition} if len(aliases) else {}
    if not mine:
        return run, []
    courts = run.courts.copy()
    filled = []
    for i, r in courts.iterrows():
        a = mine.get(r["district"])
        if a is not None and not str(r.get("court_id", "")).isdigit():
            courts.loc[i, ["court_id", "court_cs"]] = [a.court_id, a.court_cs]
            if "court_source" in courts.columns:
                courts.loc[i, "court_source"] = f"recheck alias, {a.kind} (D-205)"
            filled.append({"page": str(r["page"]), "label": r["district"], "court_id": a.court_id, "kind": a.kind})
    return replace(run, courts=courts), filled


def mend(run: reg.RunInput) -> tuple[reg.RunInput, dict]:
    """The run with its unresolved court labels aliased (:func:`alias_courts`), the table rows that
    move relabelled with their printed court and the courts its pages lacked added, and what changed.
    German volumes only: the Czech table reader files by court (its heading rows name no German
    *Gerichtsbezirk*)."""
    if run.edition not in GERMAN:
        return run, {"edition": run.edition, "rows_moved": 0}
    run, aliased = alias_courts(run)
    frame = decide(run)
    moving = _where(frame, frame["moves"].tolist())
    table = run.table.copy()
    for r in moving.itertuples():
        table.loc[[r.index, *r.parts_rows], "court_label"] = r.court_cs
    have = {(str(r["page"]).zfill(4), r["district"]) for _i, r in run.courts.iterrows()}
    moved_to = moving.assign(district=[reg._label(c) for c in moving["court_cs"]]).drop_duplicates(["page", "district"])
    moved_to = _where(moved_to, [(p, d) not in have for p, d in zip(moved_to["page"], moved_to["district"])])
    added = pd.concat([missing_courts(run), pd.DataFrame({
        "page": moved_to["page"], "district": moved_to["district"], "court_id": moved_to["court_id"],
        "court_cs": moved_to["court_cs"], "court_source": "table heading, on evidence (D-202)"})],
        ignore_index=True).drop_duplicates(["page", "district"])
    courts = pd.concat([run.courts, added], ignore_index=True) if len(added) else run.courts
    checks = count_check(run.table, court_rows(run.table, run.courts), run.courts)
    report = {"edition": run.edition, "courts_aliased": aliased, "municipality_rows_disputed": len(frame),
              "evidence": frame["evidence"].value_counts().to_dict(), "rows_moved": int(frame["moves"].sum()),
              "moves": (moving.groupby(["page", "label", "court_cs"]).size().reset_index(name="municipalities")
                        .to_dict("records") if len(moving) else []),
              "courts_added": added[["page", "district", "court_id", "court_source"]].to_dict("records"),
              "headings": len(checks), "counts_disagree": int((~checks["agrees"]).sum()) if len(checks) else 0}
    return replace(run, table=table, courts=courts), report
