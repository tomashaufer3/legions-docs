How it works

The eleven Städte mit eigenem Statut print their institutions as running prose — every office spelled out, no keyed apparatus, no pictogram (D-40, D-82) — so the symbol route never reads them and the full run skips their pages (run_frozen 0.21.0). Without this, the largest cities of the three crown lands have no institutions at all. Stage 1, here, makes an exact transcription; stage 2 (to come) maps it onto institution_abbreviations.csv.

pages Czech 15-18 (Praha, Liberec), German Böhmen 21-24 (Prag, Reichenberg), Mähren 19-24 (Brünn, Iglau, Kremsier, Olmütz, Ungarisch Hradisch, Znaim), Schlesien 13-15 (Troppau, Bielitz, Friedek): 17 pages, the full run's STATUTORY pages. strips each page cut at its widest white gaps between lines into strips of at most MAX_STRIP_LINES printed lines, so no line is split and no text is read twice. read every strip transcribed verbatim by Gemini 3.8 Flash (thinking off) and by Opus 5.5, independently, with one prompt; tables are marked [table] and not transcribed. Cached by image, prompt and model; paid only with --spend, on Heavyhorse6 (D-68). align the two readings token by token. Where they agree, that is the text. Where they differ, the text layer (ALTO for the Czech volume, the libraries' OCR for the German ones, noisy on this type) votes, and a reading it matches wins as "2 of 3". What is left goes to the author, against the strip image, in disagreements.csv.

Writes output/statutory_prose/&lt;edition&gt;_&lt;scan&gt;/ (strips, readings) and output/statutory_prose/{disagreements.csv,summary.txt}; the verified text, once the author's column is filled, is reference/places/statutory_prose/&lt;edition&gt;_&lt;scan&gt;.txt (stage 1b).

python utils/reference_tools/places/statutory_prose.py                 # dry run: strips + estimate
python utils/reference_tools/places/statutory_prose.py --spend         # Heavyhorse6: read, align

Outside utils/lexikon_1910 so as not to change the route's code digests.

def ink_mask(image):
    """Print, at a threshold halfway between the page's paper (its median grey) and its darkest
    strokes (2nd percentile): the Czech scans are cream, the German print faint (~147 on ~178)."""
    import numpy as np

    grey = np.asarray(image.convert("L"))
    paper, stroke = np.percentile(grey, 50), np.percentile(grey, 2)
    return grey < (paper + stroke) / 2

def line_bands(image) -> list[tuple[int, int]]:
    """``(top, bottom)`` of each run of rows holding print, top to bottom."""
    ink = ink_mask(image)
    ink = ink[:, ink.mean(axis=0) <= RULE_COLUMN]         # the frame, the rules and the scanner bed are not print
    inked = ink.sum(axis=1) > max(3, 0.003 * ink.shape[1])
    bands, start = [], None
    for y, on in enumerate(inked):
        if on and start is None:
            start = y
        elif not on and start is not None:
            bands.append((start, y))
            start = None
    if start is not None:
        bands.append((start, len(inked)))
    return [b for b in bands if b[1] - b[0] >= 4]

def print_columns(image) -> tuple[int, int]:
    """``(x0, x1)`` of the print, rules, frame and scanner bed left out, padded a little."""
    import numpy as np

    share = ink_mask(image).mean(axis=0)
    cols = np.nonzero((share > 0.002) & (share <= RULE_COLUMN))[0]
    if not len(cols):
        return 0, image.width
    return max(0, int(cols[0]) - 15), min(image.width, int(cols[-1]) + 16)

def strips(image, aspect: float = STRIP_ASPECT) -> list[tuple[int, int]]:
    """Cut the page at white gaps between lines: ``(y0, y1)`` per strip, each at most ``aspect``
    times the print's width tall, cut at the widest gap among the last third of its lines (a gap
    inside a line is narrower than the gap between two lines)."""
    bands = line_bands(image)
    if not bands:
        return []
    x0, x1 = print_columns(image)
    max_height = aspect * (x1 - x0)
    out, first = [], 0
    while first < len(bands):
        last = first
        while last + 1 < len(bands) and bands[last + 1][1] - bands[first][0] <= max_height:
            last += 1
        if last < len(bands) - 1:
            span = last - first + 1
            window = range(max(first, last - span // 3), last + 1)
            last = max(window, key=lambda j: bands[j + 1][0] - bands[j][1])
        y0 = bands[first][0] - (bands[first][0] - bands[first - 1][1]) // 2 if first else max(0, bands[0][0] - 8)
        y1 = (bands[last][1] + (bands[last + 1][0] - bands[last][1]) // 2 if last + 1 < len(bands)
              else min(image.height, bands[last][1] + 8))
        out.append((y0, y1))
        first = last + 1
    return out

def read_strip(png: bytes, model: str, cache: Path, spend: bool) -> dict | None:
    """One verbatim transcription, cached by image, prompt and model."""
    key = hashlib.sha256(png + PROMPT.encode() + model.encode()).hexdigest()[:16]
    path = cache / f"{model}_{key}.json"
    if path.exists():
        return json.loads(path.read_text(encoding="utf-8"))
    if not spend:
        return None
    from utils.lexikon_1910 import lexikon_1910_pages as lxp

    lxp.load_credentials()
    b64 = base64.b64encode(png).decode()
    if model == GEMINI:
        from openai import OpenAI

        client = OpenAI(base_url=GEMINI_URL, api_key=os.environ["GEMINI_API_KEY"])
        reply = client.chat.completions.create(
            model=model, temperature=0.0, reasoning_effort="none",
            messages=[{"role": "user", "content": [
                {"type": "text", "text": PROMPT},
                {"type": "image_url", "image_url": {"url": "data:image/png;base64," + b64}}]}])
        text = reply.choices[0].message.content or ""
        tokens_in, tokens_out = reply.usage.prompt_tokens, reply.usage.total_tokens - reply.usage.prompt_tokens
    else:
        import anthropic

        # Opus 5.5 takes no sampling parameters; effort as the route's Opus (run_verify_test_b.EFFORT).
        # gzip only, as every route client: httpx2 asks for brotli and the installed brotlicffi
        # cannot decode it (lexikon_1910_pages.AnthropicPageReader.ACCEPT_ENCODING).
        client = anthropic.Anthropic(default_headers={"Accept-Encoding": "gzip, deflate"})
        reply = client.messages.create(
            model=model, max_tokens=8000, output_config={"effort": "low"},
            messages=[{"role": "user", "content": [
                {"type": "image", "source": {"type": "base64", "media_type": "image/png", "data": b64}},
                {"type": "text", "text": PROMPT}]}])
        text = "".join(b.text for b in reply.content if getattr(b, "type", "") == "text")
        tokens_in, tokens_out = reply.usage.input_tokens, reply.usage.output_tokens
    p_in, p_out = PRICE[model]
    record = {"model": model, "text": text, "tokens_in": tokens_in, "tokens_out": tokens_out,
              "cost_usd": round((tokens_in * p_in + tokens_out * p_out) / 1e6, 5)}
    cache.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(record, ensure_ascii=False, indent=1), encoding="utf-8")
    return record

def normalise(text: str) -> str:
    """NFC, one kind of quote, dash and space; ``[table]`` lines kept."""
    text = unicodedata.normalize("NFC", text)
    for a, b in (("’", "'"), ("‘", "'"), ("„", '"'), ("“", '"'), ("”", '"'), ("–", "-"), ("—", "-"),
                 *zip("⁰¹²³⁴⁵⁶⁷⁸⁹", "0123456789")):          # a footnote mark read raised or not
        text = text.replace(a, b)
    text = re.sub(r"\n[ \t]*\n\s*", " ¶ ", text.strip())          # a printed paragraph break
    return re.sub(r"\s+", " ", text).strip()

def align(gemini: str, opus: str, layer: str, rulings: dict | None = None) -> tuple[list[str], list[dict]]:
    """``(text tokens, disagreements)``: the agreed tokens, a "2 of 3" choice where the text layer
    matches one reading, a ruling where one is recorded (``rulings``: ``(gemini, opus) -> (text,
    ruled_by)``), and an ``[[A|B]]`` slot for the author where none of these settles it."""
    rulings = rulings or {}
    g, o = tokens(gemini), tokens(opus)
    layer_folded = " " + " ".join(_fold(t) for t in tokens(layer)) + " "
    out, rows = [], []
    for op, i1, i2, j1, j2 in difflib.SequenceMatcher(None, g, o, autojunk=False).get_opcodes():
        if op == "equal":
            out += g[i1:i2]
            continue
        a, b = " ".join(g[i1:i2]), " ".join(o[j1:j2])
        left = " ".join(g[max(0, i1 - 4):i1])
        right = " ".join(g[i2:i2 + 4])
        in_layer = {name: bool(s) and f" {' '.join(_fold(t) for t in s.split())} " in layer_folded
                    for name, s in (("gemini", a), ("opus", b))}
        kind = disagreement_kind(a, b)
        if kind != "word":
            # Immaterial to which institutions a city has: a comma for a full stop between two
            # list items, or whether a district's place-type pictogram was seen. Gemini's kept.
            how = f"immaterial ({kind})"
            out += (a or b).split()
        elif in_layer["gemini"] != in_layer["opus"] and not re.search(r"\d", a + b):
            # Never on a figure: the OCR misreads 4 as 1 as a reader does (cz 0017, "49 pracovních
            # škol": Gemini and the text layer 19, the print 49), so the vote is not independent there.
            choice, how = (a, "gemini (2 of 3)") if in_layer["gemini"] else (b, "opus (2 of 3)")
            out += choice.split()
        elif (a, b) in rulings:
            text, who = rulings[(a, b)]
            how = f"ruled ({who})"
            out += tokens(text)
        else:
            how = "author"
            out.append(f"[[{a}|{b}]]")
        rows.append({"before": left, "gemini": a, "opus": b, "after": right, "settled": how,
                     "kind": kind, "number": bool(re.search(r"\d", a + b))})
    return out, rows

def disagreement_kind(a: str, b: str) -> str:
    """``punctuation`` (both sides punctuation only), ``symbol`` (pictograms, paragraph marks and
    punctuation only, on either side), else ``word``."""
    def bare(s: str) -> str:
        return s.replace("[symbol]", "").replace("¶", "").replace(" ", "")
    if not re.search(r"\w", bare(a)) and not re.search(r"\w", bare(b)):
        if "[symbol]" in a + b or "¶" in a + b or not (bare(a) and bare(b)):
            return "symbol"
        return "punctuation"
    return "word"

def detokenise(toks: list[str]) -> str:
    """Tokens back to text: no space before closing punctuation or after an opening bracket."""
    text = " ".join(toks)
    text = re.sub(r" ([,.;:)\]!?])", r"\1", text)
    text = re.sub(r"([(\[]) ", r"\1", text)
    return re.sub(r"\s*¶\s*", "\n\n", text).strip()

def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--spend", action="store_true", help="read what is not cached (Heavyhorse6)")
    parser.add_argument("--allow-other-machine", action="store_true", help="break D-68 on purpose")
    parser.add_argument("--editions", default=",".join(PAGES))
    args = parser.parse_args(argv)
    if args.spend and _host() != MACHINE_OF_RECORD and not args.allow_other_machine:
        raise SystemExit(f"this is {_host()}, not {MACHINE_OF_RECORD} (D-68); nothing was sent")

    OUT.mkdir(parents=True, exist_ok=True)
    all_rows, lines, spent, n_strips, missing = [], [], 0.0, 0, 0
    rulings = load_rulings()
    for edition in args.editions.split(","):
        for scan in PAGES[edition]:
            folder = OUT / f"{edition}_{scan:04d}"
            folder.mkdir(parents=True, exist_ok=True)
            image = render(edition, scan)
            layer = text_layer(edition, scan)
            (folder / "text_layer.txt").write_text(layer, encoding="utf-8")
            cuts = strips(image)
            x0, x1 = print_columns(image)
            n_strips += len(cuts)
            page_text, page_rows, complete = [], [], True
            for k, (y0, y1) in enumerate(cuts):
                crop = image.crop((x0, y0, x1, y1))
                crop.save(folder / f"strip_{k:02d}.png")
                buffer = io.BytesIO()
                crop.save(buffer, format="PNG")
                png = buffer.getvalue()
                got = {m: read_strip(png, m, folder / "cache", args.spend) for m in (GEMINI, OPUS)}
                if any(v is None for v in got.values()):
                    complete, missing = False, missing + 1
                    continue
                spent += sum(v["cost_usd"] for v in got.values())
                here = {(g, o): v for (e, sc, st, g, o), v in rulings.items() if (e, sc, st) == (edition, scan, k)}
                toks, rows = align(clean_reply(got[GEMINI]["text"]), clean_reply(got[OPUS]["text"]), layer, here)
                page_text.append(detokenise(toks))
                page_rows += [{"edition": edition, "scan": scan, "strip": k, **r} for r in rows]
            n_author = sum(r["settled"] == "author" for r in page_rows)
            lines.append(f"  {edition} {scan:04d}: {len(cuts)} strips, text layer {len(layer.split())} words"
                         + (f"; {len(page_rows)} disagreements, {n_author} for the author "
                            f"({sum(r['number'] and r['settled'] == 'author' for r in page_rows)} with a number)"
                            if complete else "; not read"))
            if complete:
                (folder / "aligned.txt").write_text("\n\n".join(page_text) + "\n", encoding="utf-8")
            all_rows += page_rows
    head = [f"Statutory-city prose, stage 1 ({__version__}): {sum(len(v) for v in PAGES.values())} pages, "
            f"{n_strips} strips, readers {GEMINI} + {OPUS}, text layer as the third vote"]
    if missing:
        head.append(f"  {missing} strips not cached: run with --spend on {MACHINE_OF_RECORD}; "
                    f"ceiling for every strip ≈ ${estimate(n_strips):.2f}")
    else:
        head.append(f"  spent (from the cache records): ${spent:.2f}")
    text = "\n".join(head + lines)
    print(text)
    (OUT / "summary.txt").write_text(text + "\n", encoding="utf-8")
    if all_rows:
        with (OUT / "disagreements.csv").open("w", encoding="utf-8", newline="") as fh:
            writer = csv.DictWriter(fh, fieldnames=[*all_rows[0], "author"])
            writer.writeheader()
            writer.writerows({**r, "author": ""} for r in all_rows)
    return 0

"""The statutory cities' prose, transcribed precisely: two vision readers and the text layer.

The eleven *Städte mit eigenem Statut* print their institutions as running prose — every office
spelled out, no keyed apparatus, no pictogram (D-40, D-82) — so the symbol route never reads them
and the full run skips their pages (``run_frozen`` 0.21.0). Without this, the largest cities of the
three crown lands have no institutions at all. Stage 1, here, makes an exact transcription; stage 2
(to come) maps it onto ``institution_abbreviations.csv``.

  pages   Czech 15-18 (Praha, Liberec), German Böhmen 21-24 (Prag, Reichenberg), Mähren 19-24
          (Brünn, Iglau, Kremsier, Olmütz, Ungarisch Hradisch, Znaim), Schlesien 13-15 (Troppau,
          Bielitz, Friedek): 17 pages, the full run's ``STATUTORY`` pages.
  strips  each page cut at its widest white gaps between lines into strips of at most
          ``MAX_STRIP_LINES`` printed lines, so no line is split and no text is read twice.
  read    every strip transcribed verbatim by Gemini 3.8 Flash (thinking off) and by Opus 5.5,
          independently, with one prompt; tables are marked ``[table]`` and not transcribed.
          Cached by image, prompt and model; paid only with ``--spend``, on Heavyhorse6 (D-68).
  align   the two readings token by token. Where they agree, that is the text. Where they differ,
          the text layer (ALTO for the Czech volume, the libraries' OCR for the German ones, noisy
          on this type) votes, and a reading it matches wins as "2 of 3". What is left goes to the
          author, against the strip image, in ``disagreements.csv``.

Writes ``output/statutory_prose/<edition>_<scan>/`` (strips, readings) and
``output/statutory_prose/{disagreements.csv,summary.txt}``; the verified text, once the author's
column is filled, is ``reference/places/statutory_prose/<edition>_<scan>.txt`` (stage 1b).

    python utils/reference_tools/places/statutory_prose.py                 # dry run: strips + estimate
    python utils/reference_tools/places/statutory_prose.py --spend         # Heavyhorse6: read, align

Outside ``utils/lexikon_1910`` so as not to change the route's code digests.
"""

def text_layer(edition: str, scan: int) -> str:

def ink_mask(image):
    """Print, at a threshold halfway between the page's paper (its median grey) and its darkest
    strokes (2nd percentile): the Czech scans are cream, the German print faint (~147 on ~178)."""
    import numpy as np

    grey = np.asarray(image.convert("L"))
    paper, stroke = np.percentile(grey, 50), np.percentile(grey, 2)
    return grey < (paper + stroke) / 2


def line_bands(image) -> list[tuple[int, int]]:
    """``(top, bottom)`` of each run of rows holding print, top to bottom."""
    ink = ink_mask(image)
    ink = ink[:, ink.mean(axis=0) <= RULE_COLUMN]         # the frame, the rules and the scanner bed are not print
    inked = ink.sum(axis=1) > max(3, 0.003 * ink.shape[1])
    bands, start = [], None
    for y, on in enumerate(inked):
        if on and start is None:
            start = y
        elif not on and start is not None:
            bands.append((start, y))
            start = None
    if start is not None:
        bands.append((start, len(inked)))
    return [b for b in bands if b[1] - b[0] >= 4]


def print_columns(image) -> tuple[int, int]:
    """``(x0, x1)`` of the print, rules, frame and scanner bed left out, padded a little."""
    import numpy as np

    share = ink_mask(image).mean(axis=0)
    cols = np.nonzero((share > 0.002) & (share <= RULE_COLUMN))[0]
    if not len(cols):
        return 0, image.width
    return max(0, int(cols[0]) - 15), min(image.width, int(cols[-1]) + 16)


def strips(image, aspect: float = STRIP_ASPECT) -> list[tuple[int, int]]:
    """Cut the page at white gaps between lines: ``(y0, y1)`` per strip, each at most ``aspect``
    times the print's width tall, cut at the widest gap among the last third of its lines (a gap
    inside a line is narrower than the gap between two lines)."""
    bands = line_bands(image)
    if not bands:
        return []
    x0, x1 = print_columns(image)
    max_height = aspect * (x1 - x0)
    out, first = [], 0
    while first < len(bands):
        last = first
        while last + 1 < len(bands) and bands[last + 1][1] - bands[first][0] <= max_height:
            last += 1
        if last < len(bands) - 1:
            span = last - first + 1
            window = range(max(first, last - span // 3), last + 1)
            last = max(window, key=lambda j: bands[j + 1][0] - bands[j][1])
        y0 = bands[first][0] - (bands[first][0] - bands[first - 1][1]) // 2 if first else max(0, bands[0][0] - 8)
        y1 = (bands[last][1] + (bands[last + 1][0] - bands[last][1]) // 2 if last + 1 < len(bands)
              else min(image.height, bands[last][1] + 8))
        out.append((y0, y1))
        first = last + 1
    return out

def _host() -> str:

def read_strip(png: bytes, model: str, cache: Path, spend: bool) -> dict | None:
    """One verbatim transcription, cached by image, prompt and model."""
    key = hashlib.sha256(png + PROMPT.encode() + model.encode()).hexdigest()[:16]
    path = cache / f"{model}_{key}.json"
    if path.exists():
        return json.loads(path.read_text(encoding="utf-8"))
    if not spend:
        return None
    from utils.lexikon_1910 import lexikon_1910_pages as lxp

    lxp.load_credentials()
    b64 = base64.b64encode(png).decode()
    if model == GEMINI:
        from openai import OpenAI

        client = OpenAI(base_url=GEMINI_URL, api_key=os.environ["GEMINI_API_KEY"])
        reply = client.chat.completions.create(
            model=model, temperature=0.0, reasoning_effort="none",
            messages=[{"role": "user", "content": [
                {"type": "text", "text": PROMPT},
                {"type": "image_url", "image_url": {"url": "data:image/png;base64," + b64}}]}])
        text = reply.choices[0].message.content or ""
        tokens_in, tokens_out = reply.usage.prompt_tokens, reply.usage.total_tokens - reply.usage.prompt_tokens
    else:
        import anthropic

        # Opus 5.5 takes no sampling parameters; effort as the route's Opus (run_verify_test_b.EFFORT).
        # gzip only, as every route client: httpx2 asks for brotli and the installed brotlicffi
        # cannot decode it (lexikon_1910_pages.AnthropicPageReader.ACCEPT_ENCODING).
        client = anthropic.Anthropic(default_headers={"Accept-Encoding": "gzip, deflate"})
        reply = client.messages.create(
            model=model, max_tokens=8000, output_config={"effort": "low"},
            messages=[{"role": "user", "content": [
                {"type": "image", "source": {"type": "base64", "media_type": "image/png", "data": b64}},
                {"type": "text", "text": PROMPT}]}])
        text = "".join(b.text for b in reply.content if getattr(b, "type", "") == "text")
        tokens_in, tokens_out = reply.usage.input_tokens, reply.usage.output_tokens
    p_in, p_out = PRICE[model]
    record = {"model": model, "text": text, "tokens_in": tokens_in, "tokens_out": tokens_out,
              "cost_usd": round((tokens_in * p_in + tokens_out * p_out) / 1e6, 5)}
    cache.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(record, ensure_ascii=False, indent=1), encoding="utf-8")
    return record


def estimate(n_strips: int) -> float:
    """A ceiling, USD: ~1,600 image tokens and ~1,200 output tokens a strip, both models live."""
    return sum(n_strips * (1_600 * p_in + 1_200 * p_out) / 1e6 for p_in, p_out in PRICE.values())

#: A reply that only says the strip is blank (both readers do it on a strip of bare paper).
BLANK_REPLY = re.compile(r"^\W*((the|this|there)\b.{0,80}\b)?(no (printed |visible )?text|blank|nothing to transcribe|does not contain any text)",
                         re.I | re.S)
#: The running head, which one reader transcribes and the other leaves out.
RUNNING_HEAD = re.compile(r"^\W*\d*\W*(Städte mit eigenem Statut|Města s vlastním řádem obecním)[^\n]*\n?", re.M)
#: A printed pictogram (the city districts' list marks each district's place type): any single
#: character that is neither a letter, a digit nor ordinary punctuation, and the readers' stand-ins.
SYMBOL = re.compile(r"\[ ?symbol ?\]|\[\?\]|[^\w\s,.;:!?()\[\]\-'\"/&*%§+=¶‚]")


def clean_reply(text: str) -> str:
    """A reply as text to align: empty for a blank-strip remark, the running head left out, every
    pictogram written ``[symbol]``."""
    if BLANK_REPLY.search(text.strip()[:200]) and len(text) < 300:
        return ""
    return SYMBOL.sub(" [symbol] ", RUNNING_HEAD.sub("", text))


def normalise(text: str) -> str:
    """NFC, one kind of quote, dash and space; ``[table]`` lines kept."""
    text = unicodedata.normalize("NFC", text)
    for a, b in (("’", "'"), ("‘", "'"), ("„", '"'), ("“", '"'), ("”", '"'), ("–", "-"), ("—", "-"),
                 *zip("⁰¹²³⁴⁵⁶⁷⁸⁹", "0123456789")):          # a footnote mark read raised or not
        text = text.replace(a, b)
    text = re.sub(r"\n[ \t]*\n\s*", " ¶ ", text.strip())          # a printed paragraph break
    return re.sub(r"\s+", " ", text).strip()


TOKEN = re.compile(r"\[symbol\]|\[\?\]|\[table\]|¶|\d+|\w+|[^\w\s]", re.UNICODE)


def tokens(text: str) -> list[str]:
    return TOKEN.findall(normalise(text))


def _fold(token: str) -> str:
    """A text-layer token compared loosely: lower case, no diacritics (the OCR loses them)."""
    return "".join(c for c in unicodedata.normalize("NFD", token.lower()) if not unicodedata.combining(c))


#: Rulings on what the readers and the text layer left open, each checked against the scan:
#: ``(edition, scan, strip, gemini, opus) -> (text, ruled_by)``. Kept apart from the run's
#: ``disagreements.csv``, which every run rewrites.
RULINGS = ROOT / "reference" / "places" / "statutory_prose_rulings.csv"


def load_rulings(path: Path = RULINGS) -> dict:
    if not path.exists():
        return {}
    with path.open(encoding="utf-8", newline="") as fh:
        return {(r["edition"], int(r["scan"]), int(r["strip"]), r["gemini"], r["opus"]): (r["text"], r["ruled_by"])
                for r in csv.DictReader(fh)}


def align(gemini: str, opus: str, layer: str, rulings: dict | None = None) -> tuple[list[str], list[dict]]:
    """``(text tokens, disagreements)``: the agreed tokens, a "2 of 3" choice where the text layer
    matches one reading, a ruling where one is recorded (``rulings``: ``(gemini, opus) -> (text,
    ruled_by)``), and an ``[[A|B]]`` slot for the author where none of these settles it."""
    rulings = rulings or {}
    g, o = tokens(gemini), tokens(opus)
    layer_folded = " " + " ".join(_fold(t) for t in tokens(layer)) + " "
    out, rows = [], []
    for op, i1, i2, j1, j2 in difflib.SequenceMatcher(None, g, o, autojunk=False).get_opcodes():
        if op == "equal":
            out += g[i1:i2]
            continue
        a, b = " ".join(g[i1:i2]), " ".join(o[j1:j2])
        left = " ".join(g[max(0, i1 - 4):i1])
        right = " ".join(g[i2:i2 + 4])
        in_layer = {name: bool(s) and f" {' '.join(_fold(t) for t in s.split())} " in layer_folded
                    for name, s in (("gemini", a), ("opus", b))}
        kind = disagreement_kind(a, b)
        if kind != "word":
            # Immaterial to which institutions a city has: a comma for a full stop between two
            # list items, or whether a district's place-type pictogram was seen. Gemini's kept.
            how = f"immaterial ({kind})"
            out += (a or b).split()
        elif in_layer["gemini"] != in_layer["opus"] and not re.search(r"\d", a + b):
            # Never on a figure: the OCR misreads 4 as 1 as a reader does (cz 0017, "49 pracovních
            # škol": Gemini and the text layer 19, the print 49), so the vote is not independent there.
            choice, how = (a, "gemini (2 of 3)") if in_layer["gemini"] else (b, "opus (2 of 3)")
            out += choice.split()
        elif (a, b) in rulings:
            text, who = rulings[(a, b)]
            how = f"ruled ({who})"
            out += tokens(text)
        else:
            how = "author"
            out.append(f"[[{a}|{b}]]")
        rows.append({"before": left, "gemini": a, "opus": b, "after": right, "settled": how,
                     "kind": kind, "number": bool(re.search(r"\d", a + b))})
    return out, rows


def disagreement_kind(a: str, b: str) -> str:
    """``punctuation`` (both sides punctuation only), ``symbol`` (pictograms, paragraph marks and
    punctuation only, on either side), else ``word``."""
    def bare(s: str) -> str:
        return s.replace("[symbol]", "").replace("¶", "").replace(" ", "")
    if not re.search(r"\w", bare(a)) and not re.search(r"\w", bare(b)):
        if "[symbol]" in a + b or "¶" in a + b or not (bare(a) and bare(b)):
            return "symbol"
        return "punctuation"
    return "word"


def detokenise(toks: list[str]) -> str:
    """Tokens back to text: no space before closing punctuation or after an opening bracket."""
    text = " ".join(toks)
    text = re.sub(r" ([,.;:)\]!?])", r"\1", text)
    text = re.sub(r"([(\[]) ", r"\1", text)
    return re.sub(r"\s*¶\s*", "\n\n", text).strip()

def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--spend", action="store_true", help="read what is not cached (Heavyhorse6)")
    parser.add_argument("--allow-other-machine", action="store_true", help="break D-68 on purpose")
    parser.add_argument("--editions", default=",".join(PAGES))
    args = parser.parse_args(argv)
    if args.spend and _host() != MACHINE_OF_RECORD and not args.allow_other_machine:
        raise SystemExit(f"this is {_host()}, not {MACHINE_OF_RECORD} (D-68); nothing was sent")

    OUT.mkdir(parents=True, exist_ok=True)
    all_rows, lines, spent, n_strips, missing = [], [], 0.0, 0, 0
    rulings = load_rulings()
    for edition in args.editions.split(","):
        for scan in PAGES[edition]:
            folder = OUT / f"{edition}_{scan:04d}"
            folder.mkdir(parents=True, exist_ok=True)
            image = render(edition, scan)
            layer = text_layer(edition, scan)
            (folder / "text_layer.txt").write_text(layer, encoding="utf-8")
            cuts = strips(image)
            x0, x1 = print_columns(image)
            n_strips += len(cuts)
            page_text, page_rows, complete = [], [], True
            for k, (y0, y1) in enumerate(cuts):
                crop = image.crop((x0, y0, x1, y1))
                crop.save(folder / f"strip_{k:02d}.png")
                buffer = io.BytesIO()
                crop.save(buffer, format="PNG")
                png = buffer.getvalue()
                got = {m: read_strip(png, m, folder / "cache", args.spend) for m in (GEMINI, OPUS)}
                if any(v is None for v in got.values()):
                    complete, missing = False, missing + 1
                    continue
                spent += sum(v["cost_usd"] for v in got.values())
                here = {(g, o): v for (e, sc, st, g, o), v in rulings.items() if (e, sc, st) == (edition, scan, k)}
                toks, rows = align(clean_reply(got[GEMINI]["text"]), clean_reply(got[OPUS]["text"]), layer, here)
                page_text.append(detokenise(toks))
                page_rows += [{"edition": edition, "scan": scan, "strip": k, **r} for r in rows]
            n_author = sum(r["settled"] == "author" for r in page_rows)
            lines.append(f"  {edition} {scan:04d}: {len(cuts)} strips, text layer {len(layer.split())} words"
                         + (f"; {len(page_rows)} disagreements, {n_author} for the author "
                            f"({sum(r['number'] and r['settled'] == 'author' for r in page_rows)} with a number)"
                            if complete else "; not read"))
            if complete:
                (folder / "aligned.txt").write_text("\n\n".join(page_text) + "\n", encoding="utf-8")
            all_rows += page_rows
    head = [f"Statutory-city prose, stage 1 ({__version__}): {sum(len(v) for v in PAGES.values())} pages, "
            f"{n_strips} strips, readers {GEMINI} + {OPUS}, text layer as the third vote"]
    if missing:
        head.append(f"  {missing} strips not cached: run with --spend on {MACHINE_OF_RECORD}; "
                    f"ceiling for every strip ≈ ${estimate(n_strips):.2f}")
    else:
        head.append(f"  spent (from the cache records): ${spent:.2f}")
    text = "\n".join(head + lines)
    print(text)
    (OUT / "summary.txt").write_text(text + "\n", encoding="utf-8")
    if all_rows:
        with (OUT / "disagreements.csv").open("w", encoding="utf-8", newline="") as fh:
            writer = csv.DictWriter(fh, fieldnames=[*all_rows[0], "author"])
            writer.writeheader()
            writer.writerows({**r, "author": ""} for r in all_rows)
    return 0
