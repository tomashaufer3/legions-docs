How it works

After §9's court-heading fix (D-202, table_courts) the rechecked register still held municipalities with institutions and no table row: 88 in the Czech volume, 105 in German Böhmen, 8 in Mähren, 6 in Schlesien. They have no printed name from the table, no population, no figures, and the claims hang on a number alone. Read against the table rows, most are two faults of the table reader, which this module mends the way table_courts mends the German headings — on the run handed to register_rechecked, never in the frozen route:

  • A seat town's row read without its number (the Czech volume). The court's seat is printed in a larger type with its parts below it, and the reader caught the town's short name and lost the number: "" Gabel between Německé Jablonné's 12 and 14, "" Münchengrätz between Mnichovo Hradiště's 23 and 25, "" Skalitz between Česká Skalice's 2 and 4. The register skips a row with no number, so the court's largest place had none. number_seats gives such a block the number between its neighbours — only where the blocks either side on the page carry the same label and read n − 1 and n + 1. A court total, a mezi tím vojsko line or a political district's total never sits between two consecutive numbers of one court, so none is numbered. Czech volume only: the German readers lose a seat differently — the row is missing (Pohořelice 12, Mähren 0110) or read as its darunter Militär line (Česká Lípa 2, Böhmen 0055) — and nothing is left to number.
  • A court carried from the page before, filed under the next court's label (all volumes). A page that ends one court and opens the next is labelled with the court it opens: Lomnice nad Popelkou's 17–26 on Czech 0390 as Semily (whose own 1–2 follow), Postoloprty's 18–22 on 0485 as Žatec (whose 1–19 follow), and Kladno's 3–7 on 0172 left unlabelled above Unhošť's 1–10, which the register could not place because both courts hold a no. 4 on the page. carried_runs moves a run — the rows of one page and label whose numbers rise without a break — to another court of the page only where all three hold:
  1. its shape is a court change the label missed: the label's numbering restarts lower after the run on the page (a court carried in), or the run is the restart (a court opened under the label of the one before it: Nová Bystřice's 1–2 on Czech 0148 as Jindřichův Hradec), or it has no label a court of the page answers to (Jindřichov's 1–8 on Silesian 0035, read under the political district's total line);
  2. the evidence (table_courts.evidence, per row: the row's printed name in the court's list of parts, else the apparatus holding that number of the court on that page) points to exactly one other court of the page for at least half the run, and to the label's court for none;
  3. no duplicate: that court has no table row of its own with those numbers within a page.

Rows the reader skipped outright (Přísečnice and Vejprty on Czech 0360, three rows read of a full page) and keys the apparatus misread are not mended here: they want the table read again, or an eye.

def number_seats(table: pd.DataFrame) -> tuple[pd.DataFrame, list[dict]]:
    """The table with each unnumbered block that sits between *n − 1* and *n + 1* of one label on its
    page given *n*, and what was numbered."""
    blocks = _blocks(table)
    table, numbered = table.copy(), []
    for _page, g in blocks.groupby("page", sort=False):
        g = g.reset_index(drop=True)
        for k in range(1, len(g) - 1):
            prev, this, nxt = g.iloc[k - 1], g.iloc[k], g.iloc[k + 1]
            if (this["n"] is None or pd.isna(this["n"])) and this["kind"] == "municipality" \
                    and pd.notna(prev["n"]) and pd.notna(nxt["n"]) and int(prev["n"]) + 2 == int(nxt["n"]) \
                    and prev["label"] == this["label"] == nxt["label"]:
                n = int(prev["n"]) + 1
                table.loc[this["rows"], "seq"] = str(n)
                numbered.append({"page": this["page"], "label": this["label"], "seq": n, "name": this["name"]})
    return table, numbered

def carried_runs(run: reg.RunInput) -> pd.DataFrame:
    """Every run of the table that is a carried court's shape, with the court the evidence names and
    whether it moves (the three conditions of the module's docstring)."""
    blocks = _blocks(run.table)
    units, names = tc.apparatus(run)
    runs = _runs(blocks)
    filed = {}                                    # (court_id, n) -> {(page, block)} its table rows are on
    now = {}                                      # block -> the court the register files it under
    for b in blocks[blocks["n"].notna()].itertuples():
        ck = reg._table_court(units, b.page, int(b.n), b.label)
        now[b.block_id] = ck
        if ck:
            filed.setdefault((ck, int(b.n)), set()).add((int(b.page), b.block_id))
    out = []
    for k, r in enumerate(runs):
        page, label = r["page"].iloc[0], r["label"].iloc[0]
        before = runs[k - 1] if k > 0 else None
        after = runs[k + 1] if k + 1 < len(runs) else None
        opened = before is not None and _restarts(before, r)
        # a label no court of the page answers to is no label: Jindřichov's 1–8 on Silesian 0035 read
        # as "Jägerndorf darunter Militär 4 GB", the political district's total line above them
        known = bool(label) and (page, reg._label(label)) in units.court_of
        if known and not (_restarts(r, after) or opened):
            continue
        given = reg.court_key(units.court_of[(page, reg._label(label))], label) if known else None
        others = {reg.court_key(c, d) for (p, d), c in units.court_of.items() if p == page} - {given}
        counts = {}
        for c in sorted(others):
            ev = [tc.evidence(units, names, page, int(b["n"]), b["name"], c, given) for _i, b in r.iterrows()]
            counts[c] = (ev.count("heading") + ev.count("both"), ev.count("heading"), ev.count("given"))
        need = math.ceil(len(r) / 2)
        named = [c for c, (_hb, h, g) in counts.items() if g == 0 and h >= 1 and counts[c][0] >= need]
        own = set(r["block_id"])                  # the run's own rows are no duplicate of themselves
        clash = {c: sorted(int(n) for n in r["n"] if any(abs(p - int(page)) <= NEARBY and blk not in own
                                                           for p, blk in filed.get((c, int(n)), set())))
                 for c in named}
        free = [c for c in named if not clash[c]]
        if len(free) == 1 and all(now.get(blk) == free[0] for blk in own):
            free = []                             # the register already files every row there
        out.append({"page": page, "label": label, "given": given or "", "first": int(r["n"].iloc[0]),
                    "last": int(r["n"].iloc[-1]), "rows": len(r),
                    "shape": ("unlabelled" if not label else "unknown label" if not known
                              else "carried" if _restarts(r, after) else "opened"),
                    "named": ";".join(named), "clash": "; ".join(f"{c}: {v}" for c, v in clash.items() if v),
                    "court_id": free[0] if len(free) == 1 else "", "moves": len(free) == 1,
                    "court_cs": units.court_cs.get(free[0], "") if len(free) == 1 else "",
                    "block_ids": list(r["block_id"])})
    return pd.DataFrame(out, columns=["page", "label", "given", "first", "last", "rows", "shape", "named",
                                      "clash", "court_id", "moves", "court_cs", "block_ids"])

def mend(run: reg.RunInput) -> tuple[reg.RunInput, dict]:
    """The run with its seat rows numbered (Czech volume) and its carried runs filed under their
    court, and a report of both."""
    table, numbered = (number_seats(run.table) if run.edition in SEAT_EDITIONS else (run.table, []))
    run = replace(run, table=table)
    frame = carried_runs(run)
    moving = frame[frame["moves"]] if len(frame) else frame
    if len(moving):
        table = run.table.copy()
        for r in moving.itertuples():
            rows = table["block_id"].isin(r.block_ids) & (table["channel"] == "A")
            table.loc[rows, "court_label"] = r.court_cs
        run = replace(run, table=table)
    report = {"edition": run.edition, "seats_numbered": numbered, "runs_checked": len(frame),
              "runs_moved": int(frame["moves"].sum()) if len(frame) else 0,
              "rows_moved": int(moving["rows"].sum()) if len(moving) else 0,
              "moves": moving.drop(columns=["block_ids"]).to_dict("records") if len(moving) else [],
              "not_moved": (frame[~frame["moves"]].drop(columns=["block_ids"]).to_dict("records")
                            if len(frame) else [])}
    return run, report

def _blocks(table: pd.DataFrame) -> pd.DataFrame:
    """One row per channel-A block in printed order: page, block, label, its number (row 0's, or
    None), and the indices of its row-0 rows."""
    a = table[(table["channel"] == "A") & (table["row"].astype(str) == "0")]
    out = []
    for block_id, g in a.groupby("block_id", sort=False):
        seqs = [int(s) for s in g["seq"].astype(str).str.strip() if s.isdigit()]
        kind = str(g["block_kind"].iloc[0]) if "block_kind" in g else "municipality"
        out.append({"page": str(g["page"].iloc[0]).zfill(4), "block_id": block_id, "kind": kind,
                    "label": str(g["court_label"].iloc[0]), "n": seqs[0] if seqs else None,
                    "name": reg.clean_display(g["name"].iloc[-1]), "rows": list(g.index)})
    return pd.DataFrame(out, columns=["page", "block_id", "kind", "label", "n", "name", "rows"])

def _runs(blocks: pd.DataFrame) -> list[pd.DataFrame]:
    """The numbered blocks cut into runs: one page, one label, numbers rising without a break."""
    runs = []
    for _page, g in blocks[blocks["n"].notna()].groupby("page", sort=False):
        current = []
        for _i, b in g.iterrows():
            if current and (b["label"] != current[-1]["label"] or int(b["n"]) <= int(current[-1]["n"])):
                runs.append(pd.DataFrame(current))
                current = []
            current.append(b)
        if current:
            runs.append(pd.DataFrame(current))
    return runs


def _restarts(run: pd.DataFrame, after: pd.DataFrame | None) -> bool:
    """The label's numbering starts lower in the run printed next on the page."""
    return (after is not None and after["page"].iloc[0] == run["page"].iloc[0]
            and after["label"].iloc[0] == run["label"].iloc[0]
            and int(after["n"].iloc[0]) < int(run["n"].iloc[0]))
