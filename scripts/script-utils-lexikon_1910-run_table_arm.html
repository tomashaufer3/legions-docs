How it works

Everything the pipeline reads from the table today comes from the NDK ALTO text layer (channel B). On the 13 gold table pages (403 rows) that layer gets 91.4 % of the number cells right but only 60.6 % of the names word for word, and 15 % of rows whole (score_table_passes.py on a channel-B-only run, 2026-10-19). This arm asks the same question of the two readers of the notes bake-off:

gemini38   Gemini 3.8 Flash, thinking off       3 repeats
haiku55    Claude Haiku 5.5, thinking at high   3 repeats

Each pass is channel A of run_lexikon_1910_pilot.py on the gold pages: one call per municipality block, the crop cut where channel B found the block, so the two channels are compared on the same crop and differ only in who reads it. The crop's edges come from the page geometry; nothing the text layer read reaches the model. A gold row no crop holds is missed by both, and the scorer reports it as recall.

Repeats, because Gemini is not deterministic (run_e1_gemini_version.py). Channel A has no response cache, so a repeat is a fresh, billed reading.

Run it on Heavyhorse6 (D-68):

python utils/lexikon_1910/run_table_arm.py --dry-run    # the plan and the pilot's estimate
python utils/lexikon_1910/run_table_arm.py              # run what is missing, then score
python utils/lexikon_1910/run_table_arm.py --score      # score only; spend nothing

def flags(job: dict, dry_run: bool = False) -> list[str]:
    """``bakeoff.flags_for`` for the table kind, with the hosted reader swapped in."""
    argv = bo.flags_for(job["model_label"], job["label"], bo.gold_pages(),
                        bo.VARIANT[VARIANT], {}, kind=KIND)
    out, skip = [], False
    for arg in argv:
        if skip:
            skip = False
            continue
        if arg in ("--reader", "--reader-model", "--workers"):
            skip = True
            continue
        out.append(arg)
    out += ["--workers", str(WORKERS), *READERS[job["model_label"]]]
    if dry_run:
        # Away from the pass directory: a dry run still writes the text layer's
        # rows, and the scorer's default sweep must not find them.
        out[out.index("--out") + 1] = str(bo.PASS_DIR / "dry_run")
        out.append("--dry-run")
    return out

from __future__ import annotations

import argparse
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))

import bakeoff as bo                                            # noqa: E402
import score_table_passes as stp                                # noqa: E402

REPEATS = 3
WORKERS = 4                 # a hosted endpoint
KIND = "table"
VARIANT = "base"            # the table has no pictograms, so no atlas (bakeoff.TABLE_VARIANTS)

#: label -> the pilot flags that pick and price the reader. Prices are USD per
#: million tokens: Gemini's is the introductory rate in ``lexikon_1910_pages``
#: (doubles 2027-01-01); Haiku's is the pilot's own ``PRICING`` entry.
READERS = {
    "gemini38": ["--reader", "gemini", "--reader-model", "gemini-3.8-flash",
                 "--reader-reasoning-effort", "none", "--reader-price", "0.75,3.75"],
    "haiku55":  ["--reader", "claude", "--model", "claude-haiku-5-5",
                 "--thinking", "adaptive", "--effort", "high"],
}


def label(model_label: str, repeat: int) -> str:
    """``gemini38_r5baset2``: the bake-off's own label for a table pass, plus the repeat."""
    return bo.label_for(model_label, VARIANT, KIND) + str(repeat)


def rows_file(name: str) -> Path:
    return bo.PASS_DIR / f"table_rows_{name}.csv"


def read_by_model(name: str) -> bool:
    """A pass is done when its rows file holds channel A rows, not the text layer alone."""
    path = rows_file(name)
    if not path.exists():
        return False
    with path.open(encoding="utf-8") as fh:
        return any(line.startswith("A,") for line in fh)


def plan() -> list[dict]:
    """Interleaved by repeat, so a run cut short leaves both readers comparable."""

             "done": read_by_model(label(m, r))}
            for r in range(1, REPEATS + 1) for m in READERS]

def run(force: bool = False, dry_run: bool = False) -> bool:
