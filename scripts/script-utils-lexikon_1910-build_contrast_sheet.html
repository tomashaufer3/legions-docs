How it works

The key (atlas_prompt.png) draws every symbol once, larger than the volume prints it, and the r5 readers' errors are swaps between symbols that differ by a fill or a letter at six-point size: castle and poorhouse, gamekeeper and huntsman, the customs and the revenue office (D-97). This sheet shows each such pair as the page prints it — a row per pair, each token labelled and cut from real pages — so the reader can decide by the difference instead of by the name.

Which pairs. Every pair the readers swapped at least PAIR_MIN times against the gold, counted over all r5 passes on the pages of --roles (e3_verdicts_r5.csv: a reader's token on a box the gold names otherwise), at most MAX_PAIRS, most swapped first. A box swap can also be the locator putting a claim on the glyph beside its own (fire brigade / savings bank); such a row is kept, since a rule that picks pairs by eye would be a choice made on the pages being scored.

Which cuts. Up to CUTS per token, from different scans where it can: the glyphs the author clicked for test A first (output/glyph_bank), then E3's alignment of the gold (boxes_gold.csv) — but an aligned box only where DINOv3, the encoder of record classifying it leave-page-out, names the gold's token on it (e3_predictions.csv). The locator puts some gold claims on the glyph or the word beside them, and a sheet that showed a reader the wrong drawing under a token would teach the swap it is meant to stop. The plate's own glyph only when no page offers a cut.

Two layouts. first (v1, contrast_sheet.png, the D-99/D-102 run) takes each token's cuts in page order. weight (v2, contrast_sheet_v2.png, D-103, the default) answers why v1 misled Gemini: ink varies by page, and v1's three castles all came from light pages, so a heavily printed castle looked like its poorhouses. v2 builds each pair's cuts together: up to two columns from pages that print both tokens — the lightest and the heaviest such page, so the difference shown is shape at equal ink — then each token's heaviest remaining cut; every cut carries its page number.

Which pages. --roles calibration (the default) while the purposive pages are the development test: the sheet never sees the pages it is scored on. At the freeze it is rebuilt with --roles calibration,purposive. Certification is refused (D-55).

def swaps(verdicts: list[dict], scans: set) -> Counter:
    """``{frozenset((a, b)): n}``: a reader's token on a box the gold names otherwise."""
    out: Counter = Counter()
    for r in verdicts:
        claim, gold = r.get("claim", ""), r.get("gold", "")
        if (r["scan"] in scans and claim.startswith("[") and gold.startswith("[")
                and claim != gold):
            out[frozenset((claim, gold))] += 1
    return out

def pick_pairs(counts: Counter, minimum: int = PAIR_MIN,
               most: int = MAX_PAIRS) -> list[tuple[str, str, int]]:
    """The pairs worth a row, most swapped first; ties by name, so the sheet is stable."""
    rows = sorted(((n, *sorted(pair)) for pair, n in counts.items() if n >= minimum),
                  key=lambda r: (-r[0], r[1], r[2]))
    return [(a, b, n) for n, a, b in rows[:most]]

def candidate_cuts(scans: set, confirmed: set | None = None) -> dict:
    """``{token: [(scan, image, box, line_height, source)]}``, clicked glyphs first.

    ``confirmed`` (:func:`confirmed_boxes`) keeps an aligned box only where the
    encoder agrees with the gold on it; ``None`` keeps them all.
    """
    out: dict = {}
    if CLICKED.exists():
        for r in _read(CLICKED):
            box = _box(r.get("tight_box", ""))
            if r["scan"] in scans and box:
                out.setdefault(f"[{r['token']}]", []).append(
                    (r["scan"], r["image"], box, int(float(r["line_height"])), "clicked"))
    for r in sorted(_read(BOXES_GOLD), key=lambda r: (r["scan"], int(r["muni"]), int(r["order"]))):
        box = _box(r.get("composite_box", ""))
        if confirmed is not None and (r["scan"], r["section"], r["muni"], r["order"]) not in confirmed:
            continue
        if r["scan"] in scans and box:
            out.setdefault(r["token"], []).append(
                (r["scan"], r["image"], box, int(float(r["line_height"])), "aligned"))
    return out

def choose(cands: list, n: int = CUTS) -> list:
    """Up to ``n`` cuts, one scan each first, in the order given."""
    picked, seen = [], set()
    for c in cands:
        if c[0] not in seen and len(picked) < n:
            picked.append(c)
            seen.add(c[0])
    for c in cands:
        if c not in picked and len(picked) < n:
            picked.append(c)
    return picked

def ink_share(cut: tuple, roles: dict) -> float:
    """Share of dark pixels inside a cut's box: how heavily that copy is printed."""
    from PIL import Image

    path = _image_path(cut[1], roles)
    if path is None:
        return 0.0
    x0, y0, x1, y1 = cut[2]
    box = Image.open(path).convert("L").crop((x0, y0, x1, y1))
    dark = sum(1 for v in box.getdata() if v < INK)
    return dark / max(1, box.width * box.height)

def choose_weighted(a: list, b: list, ink, n: int = CUTS) -> tuple[list, list]:
    """v2: the two tokens' cuts for one row, chosen together (D-103).

    First up to :data:`SAME_PAGE` columns cut from pages that print both tokens,
    the lightest and the heaviest such page by the first token's ink; then each
    token's heaviest remaining cuts, one a page where it can. ``ink`` maps a cut
    to its ink share.
    """
    first_on = lambda cands: {c[0]: c for c in reversed(cands)}       # noqa: E731
    on_a, on_b = first_on(a), first_on(b)
    shared = sorted(set(on_a) & set(on_b), key=lambda scan: (ink(on_a[scan]), scan))
    pages = shared[:1] + shared[-1:] if len(shared) > 1 else shared
    pages = pages[:SAME_PAGE]
    picked_a, picked_b = [on_a[s] for s in pages], [on_b[s] for s in pages]

    def fill(picked: list, cands: list) -> list:
        rest = sorted((c for c in cands if c not in picked), key=lambda c: (-ink(c), c[0]))
        used = {c[0] for c in picked}
        for c in rest:                           # heaviest first, one a page
            if len(picked) < n and c[0] not in used:
                picked.append(c)
                used.add(c[0])
        for c in rest:                           # then any
            if len(picked) < n and c not in picked:
                picked.append(c)
        return picked

    return fill(picked_a, a), fill(picked_b, b)

def render(pairs: list, cuts: dict, roles: dict, page_labels: bool = False):
    """``cuts`` is ``{(a, b): (cuts_a, cuts_b)}``: a token's cuts can differ by row."""
    from PIL import Image, ImageDraw, ImageFont

    try:
        font = ImageFont.load_default(size=28)
        small = ImageFont.load_default(size=18) if page_labels else None
    except TypeError:                                   # Pillow < 10.1
        font = ImageFont.load_default()
        small = ImageFont.load_default() if page_labels else None
    rows = [(_cell(a, cuts[(a, b)][0], roles, font, small),
             _cell(b, cuts[(a, b)][1], roles, font, small)) for a, b, _n in pairs]
    left = max(r[0].width for r in rows)
    width = left + 3 * GAP + max(r[1].width for r in rows)
    height = sum(max(r[0].height, r[1].height) for r in rows)
    sheet = Image.new("RGB", (width, height), "white")
    draw = ImageDraw.Draw(sheet)
    y = 0
    for a, b in rows:
        sheet.paste(a, (0, y))
        sheet.paste(b, (left + 3 * GAP, y))
        h = max(a.height, b.height)
        draw.line((left + GAP, y + 6, left + GAP, y + h - 6), fill=(150, 150, 150), width=2)
        draw.line((0, y + h - 1, width, y + h - 1), fill=(200, 200, 200), width=1)
        y += h
    return sheet

def build(role_names: tuple = ("calibration",), layout: str = "weight",
          out_dir: Path | None = None) -> tuple[Path, list]:
    """``out_dir`` writes the sheet and its pairs there instead (the frozen run's own folder)."""
    sheet_path, pairs_path = LAYOUTS[layout]
    if out_dir is not None:
        out_dir.mkdir(parents=True, exist_ok=True)
        sheet_path, pairs_path = out_dir / sheet_path.name, out_dir / pairs_path.name
    roles = {scan: role for scan, role in roles_of().items() if role in role_names}
    scans = set(roles)
    pairs = pick_pairs(swaps(_read(VERDICTS), scans))
    pool = candidate_cuts(scans, confirmed_boxes())
    if layout == "first":
        cuts = {(a, b): (choose(pool.get(a, [])), choose(pool.get(b, []))) for a, b, _n in pairs}
    else:
        memo: dict = {}
        ink = lambda c: memo.setdefault(c, ink_share(c, roles))    # noqa: E731
        cuts = {(a, b): choose_weighted(pool.get(a, []), pool.get(b, []), ink)
                for a, b, _n in pairs}
    sheet_path.parent.mkdir(parents=True, exist_ok=True)
    render(pairs, cuts, roles, page_labels=layout == "weight").save(sheet_path)
    with pairs_path.open("w", encoding="utf-8", newline="") as fh:
        w = csv.writer(fh)
        w.writerow(["token_a", "token_b", "swaps", "roles", "layout", "cuts_a", "cuts_b"])
        for a, b, n in pairs:
            w.writerow([a, b, n, "+".join(role_names), layout,
                        *(" ".join(f"{c[0]}:{c[4]}" for c in side) or "plate"
                          for side in cuts[(a, b)])])
    return sheet_path, pairs

def main(argv=None) -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):            # Jupyter's stream has no reconfigure
            pass
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--roles", default="calibration",
                    help="gold roles to learn from: calibration (default) or "
                         "calibration,purposive at the freeze")
    ap.add_argument("--layout", default="weight", choices=sorted(LAYOUTS),
                    help="weight: v2, cuts spanning print weight, same-page pairs (D-103); "
                         "first: v1, as run in D-102")
    args = ap.parse_args(argv)
    names = tuple(r.strip() for r in args.roles.split(",") if r.strip())
    if not set(names) <= set(ROLES):
        raise SystemExit(f"--roles takes {', '.join(ROLES)} only; certification is sealed (D-55)")
    path, pairs = build(names, args.layout)
    print(f"contrast sheet {__version__} from {'+'.join(names)} pages: {len(pairs)} pairs -> "
          f"{path.relative_to(ROOT)}")
    for a, b, n in pairs:
        print(f"  {n:4d}  {a} / {b}")
    return 0

SHEET = OUT / "contrast_sheet.png"                # v1, layout "first"
PAIRS = OUT / "contrast_pairs.csv"
SHEET_V2 = OUT / "contrast_sheet_v2.png"          # v2, layout "weight" (D-103)
PAIRS_V2 = OUT / "contrast_pairs_v2.csv"
LAYOUTS = {"first": (SHEET, PAIRS), "weight": (SHEET_V2, PAIRS_V2)}
SAME_PAGE = 2                                # same-page columns a pair gets in v2
INK = 128                                    # darker than this is ink (the bank's threshold)

ROLES = ("calibration", "purposive")         # never certification (D-55)
#: Three, not five: ruin / solitary house is swapped three times on calibration
#: and is half of the claims no reader reached on r5 (D-99).
PAIR_MIN = 3
MAX_PAIRS = 24
CUTS = 3
CUT_HEIGHT = 64                              # px; the scans print a glyph ~45-65 px tall
LABEL_WIDTH = 270
GAP = 16
PAPER = (238, 233, 222)


def roles_of() -> dict:
    """``{scan: role}`` for every gold sheet that is not sealed."""
    return {p.stem: role for role in ROLES for p in (GOLD_DIR / role).glob("*.md")}


def _read(path: Path) -> list[dict]:
    with path.open(encoding="utf-8", newline="") as fh:
        return list(csv.DictReader(fh))


def swaps(verdicts: list[dict], scans: set) -> Counter:
    """``{frozenset((a, b)): n}``: a reader's token on a box the gold names otherwise."""
    out: Counter = Counter()
    for r in verdicts:
        claim, gold = r.get("claim", ""), r.get("gold", "")
        if (r["scan"] in scans and claim.startswith("[") and gold.startswith("[")
                and claim != gold):
            out[frozenset((claim, gold))] += 1
    return out


def pick_pairs(counts: Counter, minimum: int = PAIR_MIN,
               most: int = MAX_PAIRS) -> list[tuple[str, str, int]]:
    """The pairs worth a row, most swapped first; ties by name, so the sheet is stable."""
    rows = sorted(((n, *sorted(pair)) for pair, n in counts.items() if n >= minimum),
                  key=lambda r: (-r[0], r[1], r[2]))
    return [(a, b, n) for n, a, b in rows[:most]]


def _box(text: str):
    try:
        return tuple(int(float(v)) for v in str(text).split(","))
    except ValueError:
        return None


def confirmed_boxes() -> set:
    """Gold boxes the encoder of record names as their own token: ``(scan, section, muni, order)``."""
    import json

    encoder, cut, feature, k = json.loads(E3_CONFIG.read_text(encoding="utf-8"))["of_record"]
    return {(r["scan"], r["section"], r["muni"], r["order"])
            for r in _read(PREDICTIONS)
            if (r["encoder"], r["cut"], r["feature"], r["k"]) == (encoder, cut, feature, str(k))
            and r["named"] == r["token"]}


def candidate_cuts(scans: set, confirmed: set | None = None) -> dict:
    """``{token: [(scan, image, box, line_height, source)]}``, clicked glyphs first.

    ``confirmed`` (:func:`confirmed_boxes`) keeps an aligned box only where the
    encoder agrees with the gold on it; ``None`` keeps them all.
    """
    out: dict = {}
    if CLICKED.exists():
        for r in _read(CLICKED):
            box = _box(r.get("tight_box", ""))
            if r["scan"] in scans and box:
                out.setdefault(f"[{r['token']}]", []).append(
                    (r["scan"], r["image"], box, int(float(r["line_height"])), "clicked"))
    for r in sorted(_read(BOXES_GOLD), key=lambda r: (r["scan"], int(r["muni"]), int(r["order"]))):
        box = _box(r.get("composite_box", ""))
        if confirmed is not None and (r["scan"], r["section"], r["muni"], r["order"]) not in confirmed:
            continue
        if r["scan"] in scans and box:
            out.setdefault(r["token"], []).append(
                (r["scan"], r["image"], box, int(float(r["line_height"])), "aligned"))
    return out


def choose(cands: list, n: int = CUTS) -> list:
    """Up to ``n`` cuts, one scan each first, in the order given."""
    picked, seen = [], set()
    for c in cands:
        if c[0] not in seen and len(picked) < n:
            picked.append(c)
            seen.add(c[0])
    for c in cands:
        if c not in picked and len(picked) < n:
            picked.append(c)
    return picked


def ink_share(cut: tuple, roles: dict) -> float:
    """Share of dark pixels inside a cut's box: how heavily that copy is printed."""
    from PIL import Image

    path = _image_path(cut[1], roles)
    if path is None:
        return 0.0
    x0, y0, x1, y1 = cut[2]
    box = Image.open(path).convert("L").crop((x0, y0, x1, y1))
    dark = sum(1 for v in box.getdata() if v < INK)
    return dark / max(1, box.width * box.height)


def choose_weighted(a: list, b: list, ink, n: int = CUTS) -> tuple[list, list]:
    """v2: the two tokens' cuts for one row, chosen together (D-103).

    First up to :data:`SAME_PAGE` columns cut from pages that print both tokens,
    the lightest and the heaviest such page by the first token's ink; then each
    token's heaviest remaining cuts, one a page where it can. ``ink`` maps a cut
    to its ink share.
    """
    first_on = lambda cands: {c[0]: c for c in reversed(cands)}       # noqa: E731
    on_a, on_b = first_on(a), first_on(b)
    shared = sorted(set(on_a) & set(on_b), key=lambda scan: (ink(on_a[scan]), scan))
    pages = shared[:1] + shared[-1:] if len(shared) > 1 else shared
    pages = pages[:SAME_PAGE]
    picked_a, picked_b = [on_a[s] for s in pages], [on_b[s] for s in pages]

    def fill(picked: list, cands: list) -> list:
        rest = sorted((c for c in cands if c not in picked), key=lambda c: (-ink(c), c[0]))
        used = {c[0] for c in picked}
        for c in rest:                           # heaviest first, one a page
            if len(picked) < n and c[0] not in used:
                picked.append(c)
                used.add(c[0])
        for c in rest:                           # then any
            if len(picked) < n and c not in picked:
                picked.append(c)
        return picked

    return fill(picked_a, a), fill(picked_b, b)


def _image_path(name: str, roles: dict) -> Path | None:
    for role in set(roles.values()):
        path = GOLD_DIR / role / name
        if path.exists():
            return path
    return None


def _cell(token: str, cuts: list, roles: dict, font, small=None):
    """A labelled strip: the token, then its cuts at a common height.

    ``small`` (a font) writes each cut's page number under it — v2 only, so the
    v1 sheet rebuilds byte for byte.
    """
    from PIL import Image, ImageDraw

    pieces, pages = [], []
    for scan, image, (x0, y0, x1, y1), lh, _src in cuts:
        path = _image_path(image, roles)
        if path is None:
            continue
        pad = max(3, lh // 5)
        cut = Image.open(path).convert("RGB").crop((x0 - pad, y0 - pad, x1 + pad, y1 + pad))
        pieces.append(cut.resize((max(1, round(cut.width * CUT_HEIGHT / cut.height)), CUT_HEIGHT)))
        pages.append(scan)
    if not pieces and (SYMBOLS / f"{token.strip('[]')}.png").exists():
        plate = Image.open(SYMBOLS / f"{token.strip('[]')}.png").convert("RGB")
        # Trimmed to its ink: the plate files keep a wide margin, which at a
        # common height would draw the glyph a third of the size of the cuts.
        ink = plate.convert("L").point(lambda v: 255 if v < 128 else 0).getbbox()
        if ink:
            plate = plate.crop((max(0, ink[0] - 4), max(0, ink[1] - 4),
                                min(plate.width, ink[2] + 4), min(plate.height, ink[3] + 4)))
        pieces.append(plate.resize((max(1, round(plate.width * CUT_HEIGHT / plate.height)), CUT_HEIGHT)))
        pages.append("key")
    under = 22 if small is not None else 0
    width = LABEL_WIDTH + sum(p.width + GAP for p in pieces)
    cell = Image.new("RGB", (width, CUT_HEIGHT + 2 * GAP + under), PAPER)
    draw = ImageDraw.Draw(cell)
    draw.text((GAP, GAP + CUT_HEIGHT // 3), token, fill=(20, 20, 20), font=font)
    x = LABEL_WIDTH
    for piece, page in zip(pieces, pages):
        cell.paste(piece, (x, GAP))
        if small is not None:
            draw.text((x, GAP + CUT_HEIGHT + 2), page, fill=(90, 90, 90), font=small)
        x += piece.width + GAP
    return cell


def render(pairs: list, cuts: dict, roles: dict, page_labels: bool = False):
    """``cuts`` is ``{(a, b): (cuts_a, cuts_b)}``: a token's cuts can differ by row."""
    from PIL import Image, ImageDraw, ImageFont

    try:
        font = ImageFont.load_default(size=28)
        small = ImageFont.load_default(size=18) if page_labels else None
    except TypeError:                                   # Pillow < 10.1
        font = ImageFont.load_default()
        small = ImageFont.load_default() if page_labels else None
    rows = [(_cell(a, cuts[(a, b)][0], roles, font, small),
             _cell(b, cuts[(a, b)][1], roles, font, small)) for a, b, _n in pairs]
    left = max(r[0].width for r in rows)
    width = left + 3 * GAP + max(r[1].width for r in rows)
    height = sum(max(r[0].height, r[1].height) for r in rows)
    sheet = Image.new("RGB", (width, height), "white")
    draw = ImageDraw.Draw(sheet)
    y = 0
    for a, b in rows:
        sheet.paste(a, (0, y))
        sheet.paste(b, (left + 3 * GAP, y))
        h = max(a.height, b.height)
        draw.line((left + GAP, y + 6, left + GAP, y + h - 6), fill=(150, 150, 150), width=2)
        draw.line((0, y + h - 1, width, y + h - 1), fill=(200, 200, 200), width=1)
        y += h
    return sheet


def build(role_names: tuple = ("calibration",), layout: str = "weight",
          out_dir: Path | None = None) -> tuple[Path, list]:
    """``out_dir`` writes the sheet and its pairs there instead (the frozen run's own folder)."""
    sheet_path, pairs_path = LAYOUTS[layout]
    if out_dir is not None:
        out_dir.mkdir(parents=True, exist_ok=True)
        sheet_path, pairs_path = out_dir / sheet_path.name, out_dir / pairs_path.name
    roles = {scan: role for scan, role in roles_of().items() if role in role_names}
    scans = set(roles)
    pairs = pick_pairs(swaps(_read(VERDICTS), scans))
    pool = candidate_cuts(scans, confirmed_boxes())
    if layout == "first":
        cuts = {(a, b): (choose(pool.get(a, [])), choose(pool.get(b, []))) for a, b, _n in pairs}
    else:
        memo: dict = {}
        ink = lambda c: memo.setdefault(c, ink_share(c, roles))    # noqa: E731
        cuts = {(a, b): choose_weighted(pool.get(a, []), pool.get(b, []), ink)
                for a, b, _n in pairs}
    sheet_path.parent.mkdir(parents=True, exist_ok=True)
    render(pairs, cuts, roles, page_labels=layout == "weight").save(sheet_path)
    with pairs_path.open("w", encoding="utf-8", newline="") as fh:
        w = csv.writer(fh)
        w.writerow(["token_a", "token_b", "swaps", "roles", "layout", "cuts_a", "cuts_b"])
        for a, b, n in pairs:
            w.writerow([a, b, n, "+".join(role_names), layout,
                        *(" ".join(f"{c[0]}:{c[4]}" for c in side) or "plate"
                          for side in cuts[(a, b)])])
    return sheet_path, pairs


def main(argv=None) -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):            # Jupyter's stream has no reconfigure
            pass
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--roles", default="calibration",
                    help="gold roles to learn from: calibration (default) or "
                         "calibration,purposive at the freeze")
    ap.add_argument("--layout", default="weight", choices=sorted(LAYOUTS),
                    help="weight: v2, cuts spanning print weight, same-page pairs (D-103); "
                         "first: v1, as run in D-102")
    args = ap.parse_args(argv)
    names = tuple(r.strip() for r in args.roles.split(",") if r.strip())
    if not set(names) <= set(ROLES):
        raise SystemExit(f"--roles takes {', '.join(ROLES)} only; certification is sealed (D-55)")
    path, pairs = build(names, args.layout)
    print(f"contrast sheet {__version__} from {'+'.join(names)} pages: {len(pairs)} pairs -> "
          f"{path.relative_to(ROOT)}")
    for a, b, n in pairs:
        print(f"  {n:4d}  {a} / {b}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
