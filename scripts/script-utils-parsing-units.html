How it works

The raw field has the structure: NUMBER.TYPE.pl. ( Formation ) e.g. 1.stř.pl. ( Čs. legie v Rusku )

Named support units (inv.rota, prac.rota, zál.pr., etc.) are preserved as-is. Entries with fewer than MIN_COUNT records across the whole dataset are collapsed to "Other (THEATRE)" to keep visualisations readable.

The parenthetical is an army, not just a theatre

It names the formation the man served in — Čs. legie v Rusku but also Cizinecké legie (French Foreign Legion), Francouská armáda, Srbská legie, Ruská armáda, Britská armáda. Regiment numbering restarts in each of them, so regiment alone is not a unit identifier: the 4,347 records (3.3 %) on a foreign army include a 1.pl. that is the Foreign Legion's first regiment — and rota Nazdar with it — not the Czechoslovak 1st Rifle. extract_regiment deliberately strips the parenthetical, so before v1.1.0 that distinction was discarded at parse time and could not be recovered downstream.

add_unit_columns therefore also emits legion_army (the parenthetical, normalised) and unit_key = "regiment @ army". Use unit_key, not regiment, for anything that treats a unit as an entity — unit fixed effects, clustering, size denominators. Cross-checked against Fidler (2025); see reference/VALIDATION_FIDLER.md.

def extract_army(unit: str) -> str | None:
    """Canonical army/formation named in the **last** parenthetical (else ``None``).

    The army is the trailing parenthetical csol.cz appends to every unit; a unit's own name
    can carry one before it — ``Royal Fusiliers (R.F.) ( Britská armáda )``,
    ``223rd (Scandinavians) Battalion ( Kanadská armáda )``, ``1.pl. (rota Nazdar) ( Cizinecké
    legie )``. Reading the *first* one (until v1.3.0) labelled 14 Royal Fusiliers "French
    Army" (``r.f.`` is the République française key) and gave ~150 British soldiers a
    regimental nickname as their army (audit 2026-09-22 §B1). An unclosed trailing
    parenthetical counts, and a trailing ``?`` (``LR?``) is an uncertainty mark, not part of
    the name.
    """
    if pd.isna(unit):
        return None
    parts = re.findall(r"\(([^()]*)(?:\)|$)", str(unit))
    if not parts:
        return None
    raw = " ".join(parts[-1].split()).rstrip("?").strip()
    if raw.lower() in _ARMY_NULL:
        return None
    return ARMY_MAP.get(raw.lower(), raw)

def extract_unit_type(regiment: str | None) -> str | None:
    """Map regiment label to unit type string."""
    if regiment is None:
        return None
    for abbrev, label in UNIT_TYPE_MAP.items():
        if abbrev in regiment:
            return label
    return "Other"

def translate_unit(label) -> str | None:
    """English rendering of a Czech unit label, e.g. ``6.stř.pl.`` → ``6th Rifle
    Regiment``, ``stráž.pr.`` → ``Guard Battalion``.

    Built from the leading number plus the English :data:`UNIT_TYPE_MAP` type. Labels
    whose type is unmapped (``"Other"`` — foreign, ad-hoc or garbled units) **pass
    through unchanged**: there is no confident English form, so the raw label is kept
    rather than a wrong guess.
    """
    if label is None or (not isinstance(label, str) and pd.isna(label)):
        return None
    core = re.sub(r"\s*\([^)]*\)", "", str(label)).strip()   # tolerate a parenthetical
    if not core:
        return None
    kind = extract_unit_type(core)
    if kind in (None, "Other"):
        return core                                          # keep the raw label verbatim
    kind_en = kind.title()                                   # "Rifle regiment" → "Rifle Regiment"
    m = re.match(r"^\s*(\d+)\.", core)
    return f"{_ordinal_en(int(m.group(1)))} {kind_en}" if m else kind_en

def add_unit_columns(df: pd.DataFrame,
                     source_col: str = "Jednotka v době zařazení",
                     min_count: int = MIN_COUNT,
                     ) -> pd.DataFrame:
    """Append ``regiment``, ``legion_army``, ``unit_key``, ``unit_type``,
    ``regiment_en``, ``unit_key_en`` and ``regiment_plot`` columns.

    ``regiment_en`` / ``unit_key_en`` are the English value renderings (``6.stř.pl.``
    → ``6th Rifle Regiment``); unmapped/foreign units pass through verbatim.

    ``regiment_plot`` collapses regiments below ``min_count`` records
    into ``"Other (Russia)"``, ``"Other (Italy)"``, etc., using the
    ``theater`` column if present, otherwise ``"Other"``.

    ``unit_key`` is the unit identity to analyse on; ``regiment`` is the display
    label. They differ wherever two armies numbered a regiment the same way.

    Args:
        df:         input DataFrame (not modified in place)
        source_col: raw unit column name
        min_count:  minimum records for individual regiment in plots

    Returns:
        DataFrame with unit columns appended.
    """
    df = df.copy()
    df["regiment"]    = df[source_col].apply(extract_regiment)
    df["legion_army"] = df[source_col].apply(extract_army)
    df["unit_key"]    = [make_unit_key(r, a)
                         for r, a in zip(df["regiment"], df["legion_army"])]
    df["unit_type"]   = df["regiment"].apply(extract_unit_type)
    # English value renderings (regiment label + the analysable identity)
    df["regiment_en"] = df["regiment"].apply(translate_unit)
    df["unit_key_en"] = [make_unit_key(r, a)
                         for r, a in zip(df["regiment_en"], df["legion_army"])]

    counts = df["regiment"].value_counts()
    small  = set(counts[counts < min_count].index)

    def plot_label(row):
        if row["regiment"] is None:
            return "Unknown"
        if row["regiment"] in small:
            theater = row.get("theater", "Unknown") if isinstance(row, dict) else \
                      row["theater"] if "theater" in row.index else "Unknown"
            return f"Other ({theater})"
        return row["regiment"]

    df["regiment_plot"] = df.apply(plot_label, axis=1)
    return df

def unit_report(df: pd.DataFrame, top_n: int = 20) -> None:
    """Print top regiments by size with desertion rates."""
    if "regiment" not in df.columns:
        print("No regiment column. Run add_unit_columns() first.")
        return
    # Grouped on unit_key, not regiment: the same label names different units in different
    # armies (D-02), and a size table is exactly where that must not be merged.
    key = "unit_key" if "unit_key" in df.columns else "regiment"
    summary = (
        df.groupby(key)
        .agg(n=("ID", "count"), deserted=("deserted", "sum"))
        .assign(desertion_rate=lambda x: x["deserted"] / x["n"] * 100)
        .sort_values("n", ascending=False)
        .head(top_n)
    )
    print(f"Top {top_n} regiments by size:")
    print(summary.to_string(float_format="{:.2f}".format))

========================================
Extracts a clean regiment label and unit type from
``Jednotka v době zařazení``.

    ``NUMBER.TYPE.pl. ( Formation )``
e.g. ``1.stř.pl. ( Čs. legie v Rusku )``

the whole dataset are collapsed to ``"Other (THEATRE)"`` to keep
visualisations readable.

-----------------------------------------------------

    "čs. legie ve francii": "CS Legion (France)",
    "cizinecké legie": "French Foreign Legion",
    "cl": "French Foreign Legion",
    "rota nazdar": "French Foreign Legion",
    "francouská armáda": "French Army",
    "francouzská armáda": "French Army",
    "r.f.": "French Army",
    "ruská armáda": "Russian Army",
    "ra": "Russian Army",
    "kornilovova armáda": "Russian Army (Kornilov)",
    "srbská legie": "Serbian Legion",
    "polská legie": "Polish Legion",
    "rumunská armáda": "Romanian Army",
    "britská armáda": "British Army",
    "k.o.y.l.i.": "British Army",
    "lincolns": "British Army",
    "americká armáda": "US Army",
    "kanadská armáda": "Canadian Army",
    "australská armáda": "Australian Army",

# Parentheticals that carry no formation information.
_ARMY_NULL = {"x", "?", "-", ""}


def extract_regiment(unit: str) -> str | None:
    """Strip the army parenthetical and return the clean regiment label.

    NOTE the label alone is ambiguous across armies (``1.pl.`` is both a Foreign
    Legion and a Legion regiment) — pair it with :func:`extract_army` via
    ``unit_key``.
    """
    if pd.isna(unit):
        return None
    label = re.sub(r"\s*\([^)]*\)", "", str(unit)).strip()
    return label if label else None


def extract_army(unit: str) -> str | None:
    """Canonical army/formation named in the **last** parenthetical (else ``None``).

    The army is the trailing parenthetical csol.cz appends to every unit; a unit's own name
    can carry one before it — ``Royal Fusiliers (R.F.) ( Britská armáda )``,
    ``223rd (Scandinavians) Battalion ( Kanadská armáda )``, ``1.pl. (rota Nazdar) ( Cizinecké
    legie )``. Reading the *first* one (until v1.3.0) labelled 14 Royal Fusiliers "French
    Army" (``r.f.`` is the République française key) and gave ~150 British soldiers a
    regimental nickname as their army (audit 2026-09-22 §B1). An unclosed trailing
    parenthetical counts, and a trailing ``?`` (``LR?``) is an uncertainty mark, not part of
    the name.
    """
    if pd.isna(unit):
        return None
    parts = re.findall(r"\(([^()]*)(?:\)|$)", str(unit))
    if not parts:
        return None
    raw = " ".join(parts[-1].split()).rstrip("?").strip()
    if raw.lower() in _ARMY_NULL:
        return None
    return ARMY_MAP.get(raw.lower(), raw)


def make_unit_key(regiment: str | None, army: str | None) -> str | None:

    if regiment is None:
        return None
    return f"{regiment} @ {army or 'unknown'}"


def extract_unit_type(regiment: str | None) -> str | None:
    """Map regiment label to unit type string."""
    if regiment is None:
        return None
    for abbrev, label in UNIT_TYPE_MAP.items():
        if abbrev in regiment:
            return label
    return "Other"


def _ordinal_en(n: int) -> str:
    """1→'1st', 2→'2nd', 3→'3rd', 11→'11th', 21→'21st'."""
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


def translate_unit(label) -> str | None:
    """English rendering of a Czech unit label, e.g. ``6.stř.pl.`` → ``6th Rifle
    Regiment``, ``stráž.pr.`` → ``Guard Battalion``.

    Built from the leading number plus the English :data:`UNIT_TYPE_MAP` type. Labels
    whose type is unmapped (``"Other"`` — foreign, ad-hoc or garbled units) **pass
    through unchanged**: there is no confident English form, so the raw label is kept
    rather than a wrong guess.
    """
    if label is None or (not isinstance(label, str) and pd.isna(label)):
        return None
    core = re.sub(r"\s*\([^)]*\)", "", str(label)).strip()   # tolerate a parenthetical
    if not core:
        return None
    kind = extract_unit_type(core)
    if kind in (None, "Other"):
        return core                                          # keep the raw label verbatim
    kind_en = kind.title()                                   # "Rifle regiment" → "Rifle Regiment"
    m = re.match(r"^\s*(\d+)\.", core)
    return f"{_ordinal_en(int(m.group(1)))} {kind_en}" if m else kind_en


def add_unit_columns(df: pd.DataFrame,
                     source_col: str = "Jednotka v době zařazení",
                     min_count: int = MIN_COUNT,
                     ) -> pd.DataFrame:
    """Append ``regiment``, ``legion_army``, ``unit_key``, ``unit_type``,
    ``regiment_en``, ``unit_key_en`` and ``regiment_plot`` columns.

    ``regiment_en`` / ``unit_key_en`` are the English value renderings (``6.stř.pl.``
    → ``6th Rifle Regiment``); unmapped/foreign units pass through verbatim.

    ``regiment_plot`` collapses regiments below ``min_count`` records
    into ``"Other (Russia)"``, ``"Other (Italy)"``, etc., using the
    ``theater`` column if present, otherwise ``"Other"``.

    ``unit_key`` is the unit identity to analyse on; ``regiment`` is the display
    label. They differ wherever two armies numbered a regiment the same way.

    Args:
        df:         input DataFrame (not modified in place)
        source_col: raw unit column name
        min_count:  minimum records for individual regiment in plots

    Returns:
        DataFrame with unit columns appended.
    """
    df = df.copy()
    df["regiment"]    = df[source_col].apply(extract_regiment)
    df["legion_army"] = df[source_col].apply(extract_army)
    df["unit_key"]    = [make_unit_key(r, a)
                         for r, a in zip(df["regiment"], df["legion_army"])]
    df["unit_type"]   = df["regiment"].apply(extract_unit_type)
    # English value renderings (regiment label + the analysable identity)
    df["regiment_en"] = df["regiment"].apply(translate_unit)
    df["unit_key_en"] = [make_unit_key(r, a)
                         for r, a in zip(df["regiment_en"], df["legion_army"])]

    counts = df["regiment"].value_counts()
    small  = set(counts[counts < min_count].index)

    def plot_label(row):
        if row["regiment"] is None:
            return "Unknown"
        if row["regiment"] in small:
            theater = row.get("theater", "Unknown") if isinstance(row, dict) else \
                      row["theater"] if "theater" in row.index else "Unknown"
            return f"Other ({theater})"
        return row["regiment"]

    df["regiment_plot"] = df.apply(plot_label, axis=1)
    return df


def unit_report(df: pd.DataFrame, top_n: int = 20) -> None:
    """Print top regiments by size with desertion rates."""
    if "regiment" not in df.columns:
        print("No regiment column. Run add_unit_columns() first.")
        return
    # Grouped on unit_key, not regiment: the same label names different units in different
    # armies (D-02), and a size table is exactly where that must not be merged.
    key = "unit_key" if "unit_key" in df.columns else "regiment"
    summary = (
        df.groupby(key)
        .agg(n=("ID", "count"), deserted=("deserted", "sum"))
        .assign(desertion_rate=lambda x: x["deserted"] / x["n"] * 100)
        .sort_values("n", ascending=False)
        .head(top_n)
    )
    print(f"Top {top_n} regiments by size:")
    print(summary.to_string(float_format="{:.2f}".format))
