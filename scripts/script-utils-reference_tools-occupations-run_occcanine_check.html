How it works

Why the German gloss and not the Czech string: OccCANINE is trained on 13 languages — en, da, se, nl, ca, fr, no, is, pt, ge, es, it, gr — and Czech is not one of them. Feeding it Czech would measure the model's out-of-distribution behaviour, not our coding. Feeding it the same German gloss the link bridge used keeps the comparison honest: both coders see identical input, and any disagreement is about the code, not the translation.

This is a validation script, not a pipeline input — the same standing as utils/reference_tools/units/build_unit_authority.py. Its dependency (torch + transformers) is deliberately kept out of requirements.txt; install it once, run it once, record the result in reference/VALIDATION_OCCUPATIONS.md.

def check(occcanine_dir: Path, batch: int = 256) -> pd.DataFrame:
    cur = pd.read_csv(CURATION, dtype=str).fillna("")
    todo = cur[(cur["de_gloss"].str.strip() != "") &
               (cur["hisco"].str.strip() != "")].copy()
    if todo.empty:
        raise SystemExit("No German glosses in the curation — run run_link_bridge.py first.")
    print(f"  {len(todo):,} glossed and coded keys to check")

    model = load_model(occcanine_dir, batch=batch)
    # Returns one row per input, in input order, as columns
    # ``occ1`` (echoed as "ge[SEP]<gloss>"), ``hisco_1``, ``desc_1``, ``conf``.
    # Align by position — ``occ1`` is not the input string verbatim.
    preds = model.predict(todo["de_gloss"].tolist(), lang="ge",
                          what="pred").reset_index(drop=True)
    if "hisco_1" not in preds.columns:
        raise SystemExit(f"Unexpected OccCANINE output columns: {list(preds.columns)}")

    todo["occcanine_hisco"] = preds["hisco_1"].astype(str).str.strip().values
    todo["occcanine_desc"] = preds["desc_1"].astype(str).values
    todo["occcanine_p"] = preds["conf"].astype(float).round(3).astype(str).values

    # Compare on the zero-stripped code: HISCO is 5-digit but the two sources differ on
    # whether the leading zero is written.
    same = (todo["occcanine_hisco"].str.lstrip("0") == todo["hisco"].str.lstrip("0"))
    todo["agree"] = same.map({True: "1", False: "0"})
    todo.loc[~same, "review"] = "1"

    for col in ("occcanine_hisco", "occcanine_p", "agree", "review"):
        if col not in cur.columns:
            cur[col] = ""
        cur.loc[todo.index, col] = todo[col]

    n = len(todo)
    todo["n_i"] = pd.to_numeric(todo["n"])
    w = todo["n_i"].sum()

    # Exact 5-digit agreement is the wrong yardstick on its own. Most disagreements are
    # *neighbouring* codes inside the same trade — kovář 83110 vs 83190 are both
    # blacksmiths, holič 57030 vs 57025 both hairdressers. What the analysis consumes is
    # HISCLASS and the major group, so report agreement at every level and let the reader
    # see which one they need.
    auth = pd.read_csv(ROOT / "reference" / "occupations" / "hisco_authority.csv",
                       dtype={"hisco": str})
    base = auth[(auth[["status", "relation", "product"]].astype(str) == "-9").all(axis=1)]
    hisclass_of = dict(zip(base["hisco"], base["hisclass"]))

    def _pad(c):
        c = str(c).strip()
        return c.zfill(5) if c and not c.startswith("-") and c.isdigit() else c

    ours = todo["hisco"].map(_pad)
    theirs = todo["occcanine_hisco"].map(_pad)
    levels = {
        "exact 5-digit code": same,
        "same unit group (3-digit)": ours.str[:3] == theirs.str[:3],
        "same minor group (2-digit)": ours.str[:2] == theirs.str[:2],
        "same major group (1-digit)": ours.str[:1] == theirs.str[:1],
        "same HISCLASS": (ours.map(hisclass_of).astype("Float64")
                          == theirs.map(hisclass_of).astype("Float64")),
    }
    print(f"\n  agreement with OccCANINE ({n:,} keys, {int(w):,} records)")
    print(f"  {'level':<28} {'keys':>14}   {'records':>16}")
    for lab, mask in levels.items():
        m = mask.fillna(False)
        wa = todo.loc[m, "n_i"].sum()
        print(f"  {lab:<28} {int(m.sum()):>5,} ({m.mean():>5.1%})   "
              f"{int(wa):>9,} ({wa / w:>5.1%})")
    print(f"\n  mean confidence — agreeing {todo.loc[same, 'occcanine_p'].astype(float).mean():.2f}, "
          f"disagreeing {todo.loc[~same, 'occcanine_p'].astype(float).mean():.2f}")

    dis = todo[~same]
    print("\n  largest disagreements (ours vs OccCANINE):")
    for r in dis.sort_values("n_i", ascending=False).head(25).itertuples():
        print(f"    {int(r.n_i):>6,}  {r.key:<22} {r.de_gloss:<20} "
              f"ours={r.hisco:<7} occ={r.occcanine_hisco:<7} p={r.occcanine_p:<6} "
              f"{r.occcanine_desc[:40]}")
    return cur

from __future__ import annotations

import argparse
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

import pandas as pd

__version__ = "1.0.0"

CURATION = ROOT / "reference" / "occupations" / "occupation_hisco_curation.csv"


def _relax_tls() -> None:
    """Let huggingface_hub fetch the model weights on a TLS-intercepted machine.

    This project's Windows box sits behind a TLS-intercepting proxy whose root CA is
    malformed — OpenSSL 3 rejects it with "Basic Constraints of CA cert not marked
    critical", the same defect ``utils/scraping/scraper_vuapraha.py`` documents for
    vuapraha.cz. Exporting the Windows trust store does not help: the CA is found and
    *then* refused. ``set_client_factory`` is huggingface_hub's supported hook for
    supplying your own client, so the relaxation is scoped to this one download rather
    than applied process-wide.
    """
    import httpx
    from huggingface_hub.utils import _http as hf_http

    hf_http.set_client_factory(
        lambda: httpx.Client(verify=False, follow_redirects=True, timeout=60.0))
    print("  note: TLS verification relaxed for the HuggingFace download "
          "(local trust store rejects this machine's intercepting CA)")
