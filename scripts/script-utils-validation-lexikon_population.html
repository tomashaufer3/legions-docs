How it works

Volume I prints, for every obec of 2005 and every one of its parts (části obce), the population and houses at each census from 1869, recomputed onto the territory of 2005 (its foreword: sestaven podle správního rozdělení státu k 1. 1. 2005). Its notes say where that recomputation joins or moves places: V tom též X — the figure includes X as well; viz Z — the figure is counted under Z; each scoped by year (Rok 1869 až 1880, obyvatelstvo v roce 1921) or limited to houses (domy v roce 1961).

The PDF's data pages have no usable text layer: its font maps every glyph to a code 29 below the character it draws, and several accented letters share the code of a plain one (á is drawn on i's code, í on t's, é on p's). Text is decoded by the shift (decode) and a modern name is compared by folding it into the same alphabet (fold). Columns are assigned by the right edge of the year headers, which is where the numbers are aligned.

Validation only (check_ostafin_silesia.py): nothing here feeds data/final.

def touches_1910(note: str) -> list[str]:
    """The actions of a decoded note that reach the 1910 population, as ``includes: …`` or
    ``counted under: …``. A scope set by a year range, a single year or ``domy`` applies to
    the next action only; an action with no scope applies to every census."""
    out, years, houses = [], set(), False
    tokens = list(_NOTE.finditer(note))
    for i, t in enumerate(tokens):
        if t["range"]:
            years |= set(range(int(t["a"]), int(t["b"] or t["a"]) + 1))
            houses = False
        elif t["pop"]:
            years.add(int(t["y"]))
            houses = False
        elif t["houses"]:
            years, houses = set(), True
        elif t["act"]:
            end = tokens[i + 1].start() if i + 1 < len(tokens) else len(note)
            if not houses and (not years or 1910 in years):
                what = "includes" if "tom" in t["act"] else "counted under"
                out.append(f"{what}: {note[t.end():end].strip(' ,')}")
            years, houses = set(), False
    return out

def note_rows(page) -> list[tuple[list[str], str]]:
    """``(keys, text)`` per note on a notes page: two columns, continuation lines joined."""
    columns = collections.defaultdict(list)
    for block in page.get_text("rawdict")["blocks"]:
        for line in block.get("lines", []):
            text = "".join(decode(ch["c"]) for span in line["spans"] for ch in span["chars"])
            x0, y0 = line["bbox"][:2]
            if y0 > 60 and text.strip():
                columns[x0 >= 300].append((round(y0), text))
    out = []
    for column in columns.values():
        for _, text in sorted(column):
            m = re.match(r"\s*((?:\d+(?:\.\d+)?\.,?\s*)+)(.*)", text)
            if m and re.match(r"\d", text.strip()):
                out.append((re.findall(r"\d+(?:\.\d+)?", m.group(1)), m.group(2)))
            elif out:
                out[-1] = (out[-1][0], out[-1][1] + " " + text.strip())
    return out

from __future__ import annotations

__version__ = "0.1.0"

import collections
import re
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
PDF = ROOT / "library" / "lexicons" / "CSU_Historicky-lexikon-obci-CR_1869-2005_dil-1_obyvatele-domy.pdf"
#: The 2005 okresy that cover former Austrian Silesia, and their pages (0-based) in díl I:
#: data pages, the facing pages of later years and houses, then the notes.
OKRESY = {"Jeseník": (649, 652), "Bruntál": (705, 712), "Frýdek-Místek": (713, 716),
          "Karviná": (717, 720), "Nový Jičín": (721, 726), "Opava": (727, 734),
          "Ostrava-město": (735, 738)}
#: The font's code offset: a code below this is a shifted character (0x60 = 96).
SHIFT, SHIFTED_BELOW = 29, 0x60
#: A modern spelling folded into the volume's alphabet: each accented letter onto the code the
#: font draws it on (read off the decoded names, e.g. ``Bruntil``, ``Jesentk``, ``Novê Jiþtn``).
FOLD = str.maketrans({"á": "i", "í": "t", "é": "p", "ý": "ê", "č": "þ", "š": "ã", "ř": "Ĝ",
                      "ě": "Č", "ů": "Ĥ", "ň": "Ė", "ť": "Ģ", "ď": "ć", "ž": "å", "Č": "ý",
                      "Š": "â", "Ž": "ä", "Ú": "Ò", "Ř": "ě", "ú": "u", "ó": "o"})
#: Abbreviations in modern names, spelled out as the volume prints them.
ABBREVIATIONS = [(r"\bn\.\s*", "nad "), (r"\bp\.\s*", "pod "), (r"\bu\.\s*", "u ")]
_NOTE = re.compile(r"(?P<range>[Rr]ok (?P<a>1[89]\d\d|20\d\d)(?: aå (?P<b>1[89]\d\d|20\d\d))?)"
                   r"|(?P<pop>obyvatelstvo v roce (?P<y>1[89]\d\d|20\d\d))"
                   r"|(?P<houses>[Dd]omy v roce (?P<h>1[89]\d\d|20\d\d))"
                   r"|(?P<act>[Vv] tom(?: tpå)?|viz)")


def decode(text: str) -> str:
    """The volume's text from its glyph codes."""
    return "".join(chr(ord(c) + SHIFT) if ord(c) < SHIFTED_BELOW else c for c in text)


def fold(name: str) -> str:
    """A modern name in the volume's alphabet, lower-cased, abbreviations spelled out."""
    for short, full in ABBREVIATIONS:
        name = re.sub(short, full, name)
    return re.sub(r"\s+", " ", name.translate(FOLD)).strip().lower()

def _spans(page) -> list[tuple[int, float, float, str]]:
    out = []
    for block in page.get_text("rawdict")["blocks"]:
        for line in block.get("lines", []):
            for span in line["spans"]:
                text = "".join(decode(ch["c"]) for ch in span["chars"]).strip()
                if text:
                    x0, y0, x1, _ = span["bbox"]
                    out.append((round(y0), x0, x1, text))
    return out


def _years(spans) -> list[tuple[float, int]]:
    """The census-year headers of a page as ``(right edge, year)``; none on a notes page."""
    return [(x1, int(t)) for y, x0, x1, t in spans if y < 90 and re.fullmatch(r"1[89]\d\d|20\d\d", t)]
