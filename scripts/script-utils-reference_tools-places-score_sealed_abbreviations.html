How it works

The sealed scorer (utils/lexikon_1910/score_certification.py) scores the pictograms and, for German Böhmen, the cooperative letters; the other text abbreviations (a school printed Volksschule, a fire brigade F.W., a course) had no sealed-page score, and the only measure of them was the gold's, which counts the entries every reader reached (run_frozen.score_abbreviations). They feed the register (cooperative counts by language, school classes), so this scores them the way the pictograms are scored. Written 2026-10-31, before it was run on any edition.

The sheets are the same eight sealed pages, already transcribed; nothing is read again and no paid call is made. It lives outside utils/lexikon_1910 on purpose: every module in that folder is digested into the route's freeze, so a new one there would make the full run read as another route (run_frozen._code_digests).

What it reports (all on the claim (page, court, municipality, part, code))

  1. Claim-level P / R / F1 over every entry the sheets print, with the 95 % page-clustered interval (score_certification.page_clustered_interval: 10,000 draws, seed 1910).
  2. The same by kind: institution, cooperative, course.
  3. Exact upper bounds (the upper end of the two-sided 95 % Clopper-Pearson interval, claims taken as independent, which understates the uncertainty) on the false share of the route's claims and on the missed share of the gold's: the bootstrap over eight pages is anti-conservative when errors are few, and this is the figure to quote beside it.
  4. Count accuracy: among true claims, the share whose printed count the route reproduces.
  5. For cooperatives, language accuracy: among true cooperative claims, the share whose count by printed language letter ({none: 1, d: 1}) the route reproduces exactly.
  6. Per code: gold instances, hit, missed, false.

Both decoders are the route's own (run_frozen.sheet_abbreviations for the sheets), so the score measures reading, not decoding. A run folder is read, never written to except its own abbreviation_score.{json,txt}.

def upper_bound(k: int, n: int, alpha: float = ALPHA) -> float | None:
    """Exact (Clopper-Pearson) ``1 - alpha`` upper bound on a proportion: the
    largest ``p`` with ``P(X <= k) >= alpha`` for ``X ~ Binomial(n, p)``. ``None`` for ``n = 0``."""
    if n <= 0:
        return None
    if k >= n:
        return 1.0
    lo, hi = k / n, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        cdf = sum(comb(n, i) * mid**i * (1 - mid) ** (n - i) for i in range(k + 1))
        lo, hi = (mid, hi) if cdf >= alpha else (lo, mid)
    return (lo + hi) / 2

def parse_language_counts(text: str) -> dict:
    """``"none:1|d:1"`` -> ``{"none": 1, "d": 1}`` (the route's ``language_counts``)."""
    out = {}
    for part in (text or "").split("|"):
        if ":" in part:
            key, _, value = part.rpartition(":")
            out[key.strip()] = out.get(key.strip(), 0) + int(value)
    return out

def route_side(run: Path) -> tuple[dict, dict, dict, set]:
    """``({claim: count}, {claim: kind}, {claim: languages}, entries)`` from a run's
    ``final_abbreviations.csv``, keyed as the sealed scorer keys a claim."""
    counts, kind_of, languages, entries = {}, {}, {}, set()
    with (run / "final_abbreviations.csv").open(encoding="utf-8") as fh:
        for row in csv.DictReader(fh):
            place = (row.get("court_id") or "").strip() or row["district"]
            claim = (row["page"], place, row["muni_seq"], row["settlement_seq"], row["code"])
            counts[claim] = (row.get("count") or "").strip()
            kind_of[claim] = row["kind"]
            languages[claim] = parse_language_counts(row.get("language_counts", ""))
            entries.add(claim[:3])
    return counts, kind_of, languages, entries

def gold_side(edition: str) -> dict:
    """The sealed sheets' abbreviations: ``counts``, ``kinds``, ``languages`` per claim, the
    printed entries (the population) and the sealed pages."""
    import run_frozen as rf
    import score_certification as sc

    court = sc.court_of(edition)
    _entries, claims = rf.sheet_abbreviations(sc.sealed_paths(edition), edition)
    table = rf.abbreviation_table(edition)
    key = lambda t: (str(t[0]), court(str(t[1])), *map(str, t[2:]))               # noqa: E731
    counts = {key(c): str(v["count"]) for c, v in claims.items()}
    kinds = {key(c): table[c[4]].kind for c in claims}
    languages = {key(c): dict(v["languages"]) for c, v in claims.items()}
    sheets = sc.sealed_sheets(edition)
    return {"counts": counts, "kinds": kinds, "languages": languages,
            "entries": sc.printed_entries(sheets, court),
            "pages": {str(sheet["scan"]) for sheet in sheets}}

def sides(edition: str, run: Path, tag: str = "") -> dict:
    """Both sides of one edition, the pages prefixed with ``tag`` when editions are pooled."""
    gold = gold_side(edition)
    counts, kinds, languages, entries = route_side(run)
    out = {"gold_counts": gold["counts"], "gold_kinds": gold["kinds"],
           "gold_languages": gold["languages"], "route_counts": counts, "route_kinds": kinds,
           "route_languages": languages, "entries": gold["entries"] | entries,
           "printed": gold["entries"], "pages": gold["pages"]}
    if tag:
        page = lambda c: (f"{tag}:{c[0]}", *c[1:])                                # noqa: E731
        for name in ("gold_counts", "gold_kinds", "gold_languages", "route_counts",
                     "route_kinds", "route_languages"):
            out[name] = {page(c): v for c, v in out[name].items()}
        for name in ("entries", "printed"):
            out[name] = {page(e) for e in out[name]}
        out["pages"] = {f"{tag}:{p}" for p in out["pages"]}
    return out

            if isinstance(value, set):

def block(s: dict, keep) -> dict:
    """The metrics for the claims ``keep`` selects (a kind, or all)."""
    import score_certification as sc

    pages = s["pages"]
    truth = {c for c in s["gold_counts"] if c[0] in pages and keep(s["gold_kinds"][c])}
    got = {c for c in s["route_counts"] if c[0] in pages and keep(s["route_kinds"][c])}
    result = sc.score_sets(truth, got)
    result["interval"] = sc.page_clustered_interval(truth, got)
    result["false_upper"] = upper_bound(result["false"], len(got))
    result["missed_upper"] = upper_bound(result["missed"], len(truth))
    right = got & truth
    result["count"] = sc.count_accuracy(s["gold_counts"], s["route_counts"], right)
    languages = [c for c in right if s["gold_kinds"][c] == "cooperative"]
    same = sum(1 for c in languages if s["gold_languages"][c] == s["route_languages"].get(c))
    result["language"] = {"claims": len(languages), "language_right": same,
                          "share": same / len(languages) if languages else None}
    result["symbols"] = sc.per_symbol(truth, got, s["printed"])
    return result

def report(title: str, result: dict) -> str:
    lines = [title, "  claims on the sealed pages, over every entry the sheets print",
             "  (the interval is 95 % over the pages resampled; the exact bounds treat claims as independent)"]
    for name in ("all", *KINDS):
        b = result[name]
        if not (b["claims_gold"] or b["claims_route"]):
            continue
        lines.append(_line(name, b))
        lines.append(f"  {'':12s} exact 95 % (two-sided) upper bound: false share of the route's claims "
                     f"{_bound(b['false_upper'])}, missed share of the sheets' "
                     f"{_bound(b['missed_upper'])}")
        c = b["count"]
        lines.append(f"  {'':12s} printed count right on {c['count_right']} of {c['claims']} true claims")
        if name == "cooperative" and b["language"]["claims"]:
            lg = b["language"]
            lines.append(f"  {'':12s} language letters right on {lg['language_right']} of "
                         f"{lg['claims']} true cooperative claims")
    lines += ["", "  per code (as printed)", f"    {'code':14s} {'gold':>5s} {'hit':>4s} {'miss':>5s} {'false':>6s}"]
    for row in result["all"]["symbols"]:
        lines.append(f"    {row['token']:14s} {row['gold']:5d} {row['hit']:4d} {row['missed']:5d} {row['false']:6d}")
    return "\n".join(lines)

def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--edition", default="de-bohemia")
    parser.add_argument("--run", default=None, help="run folder under output/frozen")
    parser.add_argument("--german", action="store_true", help="the three German editions, then pooled")
    args = parser.parse_args(argv)
    frozen = ROOT / "output" / "frozen"

    if not args.german:
        run = frozen / (args.run or f"{args.edition}_certification")
        result = measure(sides(args.edition, run))
        title = f"Sealed-page abbreviations — {args.edition}, {run.name}"
        print(report(title, result))
        write(run, title, result)
        return 0

    suffix = "" if not args.run else args.run.replace("de-bohemia_certification", "")
    parts, pooled = [], {}
    for edition in GERMAN:
        run = frozen / f"{edition}_certification{suffix}"
        side = sides(edition, run)
        result = measure(side)
        title = f"Sealed-page abbreviations — {edition}, {run.name}"
        print(report(title, result), "\n")
        write(run, title, result)
        parts.append(sides(edition, run, tag=edition))
    result = measure(merge(parts))
    title = f"Sealed-page abbreviations — German pooled, {suffix or 'round 2'}"
    print(report(title, result))
    write(frozen / f"de-pooled_certification{suffix}", title, result)
    return 0

def upper_bound(k: int, n: int, alpha: float = ALPHA) -> float | None:
    """Exact (Clopper-Pearson) ``1 - alpha`` upper bound on a proportion: the
    largest ``p`` with ``P(X <= k) >= alpha`` for ``X ~ Binomial(n, p)``. ``None`` for ``n = 0``."""
    if n <= 0:
        return None
    if k >= n:
        return 1.0
    lo, hi = k / n, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        cdf = sum(comb(n, i) * mid**i * (1 - mid) ** (n - i) for i in range(k + 1))
        lo, hi = (mid, hi) if cdf >= alpha else (lo, mid)
    return (lo + hi) / 2

def route_side(run: Path) -> tuple[dict, dict, dict, set]:
    """``({claim: count}, {claim: kind}, {claim: languages}, entries)`` from a run's
    ``final_abbreviations.csv``, keyed as the sealed scorer keys a claim."""
    counts, kind_of, languages, entries = {}, {}, {}, set()
    with (run / "final_abbreviations.csv").open(encoding="utf-8") as fh:
        for row in csv.DictReader(fh):
            place = (row.get("court_id") or "").strip() or row["district"]
            claim = (row["page"], place, row["muni_seq"], row["settlement_seq"], row["code"])
            counts[claim] = (row.get("count") or "").strip()
            kind_of[claim] = row["kind"]
            languages[claim] = parse_language_counts(row.get("language_counts", ""))
            entries.add(claim[:3])
    return counts, kind_of, languages, entries


def gold_side(edition: str) -> dict:
    """The sealed sheets' abbreviations: ``counts``, ``kinds``, ``languages`` per claim, the
    printed entries (the population) and the sealed pages."""
    import run_frozen as rf
    import score_certification as sc

    court = sc.court_of(edition)
    _entries, claims = rf.sheet_abbreviations(sc.sealed_paths(edition), edition)
    table = rf.abbreviation_table(edition)
    key = lambda t: (str(t[0]), court(str(t[1])), *map(str, t[2:]))               # noqa: E731
    counts = {key(c): str(v["count"]) for c, v in claims.items()}
    kinds = {key(c): table[c[4]].kind for c in claims}
    languages = {key(c): dict(v["languages"]) for c, v in claims.items()}
    sheets = sc.sealed_sheets(edition)
    return {"counts": counts, "kinds": kinds, "languages": languages,
            "entries": sc.printed_entries(sheets, court),
            "pages": {str(sheet["scan"]) for sheet in sheets}}


def sides(edition: str, run: Path, tag: str = "") -> dict:
    """Both sides of one edition, the pages prefixed with ``tag`` when editions are pooled."""
    gold = gold_side(edition)
    counts, kinds, languages, entries = route_side(run)
    out = {"gold_counts": gold["counts"], "gold_kinds": gold["kinds"],
           "gold_languages": gold["languages"], "route_counts": counts, "route_kinds": kinds,
           "route_languages": languages, "entries": gold["entries"] | entries,
           "printed": gold["entries"], "pages": gold["pages"]}
    if tag:
        page = lambda c: (f"{tag}:{c[0]}", *c[1:])                                # noqa: E731
        for name in ("gold_counts", "gold_kinds", "gold_languages", "route_counts",
                     "route_kinds", "route_languages"):
            out[name] = {page(c): v for c, v in out[name].items()}
        for name in ("entries", "printed"):
            out[name] = {page(e) for e in out[name]}
        out["pages"] = {f"{tag}:{p}" for p in out["pages"]}
    return out


def merge(parts: list[dict]) -> dict:
    out: dict = {}
    for part in parts:
        for name, value in part.items():
            if isinstance(value, set):
                out.setdefault(name, set()).update(value)
            else:
                out.setdefault(name, {}).update(value)
    return out


def block(s: dict, keep) -> dict:
    """The metrics for the claims ``keep`` selects (a kind, or all)."""
    import score_certification as sc

    pages = s["pages"]
    truth = {c for c in s["gold_counts"] if c[0] in pages and keep(s["gold_kinds"][c])}
    got = {c for c in s["route_counts"] if c[0] in pages and keep(s["route_kinds"][c])}
    result = sc.score_sets(truth, got)
    result["interval"] = sc.page_clustered_interval(truth, got)
    result["false_upper"] = upper_bound(result["false"], len(got))
    result["missed_upper"] = upper_bound(result["missed"], len(truth))
    right = got & truth
    result["count"] = sc.count_accuracy(s["gold_counts"], s["route_counts"], right)
    languages = [c for c in right if s["gold_kinds"][c] == "cooperative"]
    same = sum(1 for c in languages if s["gold_languages"][c] == s["route_languages"].get(c))
    result["language"] = {"claims": len(languages), "language_right": same,
                          "share": same / len(languages) if languages else None}
    result["symbols"] = sc.per_symbol(truth, got, s["printed"])
    return result


def measure(s: dict) -> dict:
    result = {"all": block(s, lambda kind: True)}
    for kind in KINDS:
        result[kind] = block(s, lambda k, kind=kind: k == kind)
    return result


def _line(name: str, b: dict) -> str:
    lo, hi = b["interval"]["f1"]
    return (f"  {name:12s} {b['claims_gold']:4d} printed {b['claims_route']:4d} by the route "
            f"({b['hit']} hit, {b['false']} false, {b['missed']} missed)  F1 {b['f1']:.3f} "
            f"[{lo:.3f}, {hi:.3f}]  precision {b['precision']:.3f}  recall {b['recall']:.3f}")


def _bound(value) -> str:
    return "—" if value is None else f"{value:.3f}"


def report(title: str, result: dict) -> str:
    lines = [title, "  claims on the sealed pages, over every entry the sheets print",
             "  (the interval is 95 % over the pages resampled; the exact bounds treat claims as independent)"]
    for name in ("all", *KINDS):
        b = result[name]
        if not (b["claims_gold"] or b["claims_route"]):
            continue
        lines.append(_line(name, b))
        lines.append(f"  {'':12s} exact 95 % (two-sided) upper bound: false share of the route's claims "
                     f"{_bound(b['false_upper'])}, missed share of the sheets' "
                     f"{_bound(b['missed_upper'])}")
        c = b["count"]
        lines.append(f"  {'':12s} printed count right on {c['count_right']} of {c['claims']} true claims")
        if name == "cooperative" and b["language"]["claims"]:
            lg = b["language"]
            lines.append(f"  {'':12s} language letters right on {lg['language_right']} of "
                         f"{lg['claims']} true cooperative claims")
    lines += ["", "  per code (as printed)", f"    {'code':14s} {'gold':>5s} {'hit':>4s} {'miss':>5s} {'false':>6s}"]
    for row in result["all"]["symbols"]:
        lines.append(f"    {row['token']:14s} {row['gold']:5d} {row['hit']:4d} {row['missed']:5d} {row['false']:6d}")
    return "\n".join(lines)


def write(run: Path, title: str, result: dict) -> None:
    run.mkdir(parents=True, exist_ok=True)
    (run / "abbreviation_score.json").write_text(
        json.dumps({"scorer": __version__, "title": title, **result}, indent=1, default=list),
        encoding="utf-8")
    (run / "abbreviation_score.txt").write_text(report(title, result) + "\n", encoding="utf-8")


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--edition", default="de-bohemia")
    parser.add_argument("--run", default=None, help="run folder under output/frozen")
    parser.add_argument("--german", action="store_true", help="the three German editions, then pooled")
    args = parser.parse_args(argv)
    frozen = ROOT / "output" / "frozen"

    if not args.german:
        run = frozen / (args.run or f"{args.edition}_certification")
        result = measure(sides(args.edition, run))
        title = f"Sealed-page abbreviations — {args.edition}, {run.name}"
        print(report(title, result))
        write(run, title, result)
        return 0

    suffix = "" if not args.run else args.run.replace("de-bohemia_certification", "")
    parts, pooled = [], {}
    for edition in GERMAN:
        run = frozen / f"{edition}_certification{suffix}"
        side = sides(edition, run)
        result = measure(side)
        title = f"Sealed-page abbreviations — {edition}, {run.name}"
        print(report(title, result), "\n")
        write(run, title, result)
        parts.append(sides(edition, run, tag=edition))
    result = measure(merge(parts))
    title = f"Sealed-page abbreviations — German pooled, {suffix or 'round 2'}"
    print(report(title, result))
    write(frozen / f"de-pooled_certification{suffix}", title, result)
    return 0
