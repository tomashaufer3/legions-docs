How it works

Why MPIDR is the spine

The MPIDR shapefile (MPIDR & CGG 2012, after Rumpler & Seger 2010) is a clean, complete polygon layer of the 1910 census administrative units with a stable 3-level hierarchy encoded in ID_1 = REGION·10^6 + DISTRICT·10^3 + LOCAL_:

LAND (crown land)  →  DISTRICT (political okres)  →  LOCAL_ (Gerichtsbezirk / soudní okres)

so it authoritatively fixes which court districts belong to which political okres, carries stable geo-IDs + polygons (the census-merge key and the spatial backbone), and its German names are clean apart from six transliterations repaired by GERMAN_NAME_FIX (Sziszkow = Žižkow).

Czech names — provenance

Each district's canonical Czech name (*_cs) is resolved, best-first:

  1. curated (district_cs_curation.csv) — the accented canonical form; all 330 court and 158 okres names are curated, so this is the only tier in use;
  2. MPIDR de-accented Czech (the NAME field's post-slash part) — a fallback that is the right name minus diacritics, flagged needs_review.

The OCR Ortsverzeichnis gazetteer once supplied a third tier; its Czech district names proved unusable ("Bude jo vice") and it is archived (_archive/gazetteer.py). attach_gazetteer_names is kept only for that history.

Note on the city/rural twins: both members of a pair carry the same okres_de (the seat court's German name, e.g. Olmuetz); okres_cs is what tells them apart (Olomouc / Olomouc (město)).

def read_dbf(path, encodings: tuple[str, ...] = ("cp1250", "latin-1", "utf-8")) -> list[dict]:
    """Read a .dbf attribute table into a list of ``{field: value}`` dicts."""
    raw = Path(path).read_bytes()
    n_rec = struct.unpack("<I", raw[4:8])[0]
    hdr_len = struct.unpack("<H", raw[8:10])[0]
    rec_len = struct.unpack("<H", raw[10:12])[0]

    fields: list[tuple[str, int]] = []
    off = 32
    while raw[off] != 0x0D:                                   # 0x0D terminates the descriptor array
        desc = raw[off:off + 32]
        name = desc[0:11].split(b"\x00")[0].decode("latin-1")
        fields.append((name, desc[16]))                      # byte 16 = field length
        off += 32

    def _decode(b: bytes) -> str:
        for enc in encodings:
            try:
                return b.decode(enc).strip()
            except UnicodeDecodeError:
                continue
        return b.decode("latin-1", "replace").strip()

    records: list[dict] = []
    for i in range(n_rec):
        row = raw[hdr_len + i * rec_len: hdr_len + (i + 1) * rec_len]
        if not row or row[0:1] == b"*":                      # deletion flag
            continue
        pos, rec = 1, {}
        for name, flen in fields:
            rec[name] = _decode(row[pos:pos + flen])
            pos += flen
        records.append(rec)
    return records

def parse_mpidr(dbf_path, lands: dict[str, str] = LAND_MAP) -> pd.DataFrame:
    """Parse the MPIDR .dbf into a court-district backbone for the Czech lands.

    One row per Gerichtsbezirk (LOCAL_ polygon). Columns:
        ``land`` · ``okres_code`` (REGION-DISTRICT) · ``court_id`` (ID_1, unique) ·
        ``mpidr_oid`` (OID_ → .shp geometry) · ``gb_de`` (German court-district name) ·
        ``gb_cs_mpidr`` (MPIDR de-accented Czech, post-slash) · ``gb_de_norm`` (match key).
    """
    rows = []
    for r in read_dbf(dbf_path):
        if r["LAND"] not in lands:
            continue
        de, _, cs = r["NAME"].partition("/")
        rows.append({
            "land":        lands[r["LAND"]],
            "okres_code":  f"{r['REGION']}-{r['DISTRICT']}",
            "court_id":    r["ID_1"],
            "mpidr_oid":   r["OID_"],
            "gb_de":       de.strip(),
            "gb_cs_mpidr": cs.strip() or None,
            "gb_de_norm":  _norm_de(de),
        })
    return pd.DataFrame(rows)

def attach_gazetteer_names(backbone: pd.DataFrame, gazetteer: pd.DataFrame,
                           threshold: int = _FUZZY_THRESHOLD) -> pd.DataFrame:
    """Fuzzy-join gazetteer Czech names onto the MPIDR backbone via the German name.

    Brings ``gz_gb_cs`` (court-district Czech), ``gz_polit_de`` / ``gz_polit_cs``
    (political-okres German/Czech) from the gazetteer, matched on the normalised German
    Gerichtsbezirk name (exact first, then rapidfuzz ratio ≥ *threshold*). Rows are
    preferred that already carry an accented Czech court name.
    """
    from rapidfuzz import process, fuzz

    gb = gazetteer[["gerichtsbezirk_de", "gerichtsbezirk_cs",
                    "polit_bezirk_de", "polit_bezirk_cs"]].dropna(subset=["gerichtsbezirk_de"]).copy()
    gb["k"] = gb["gerichtsbezirk_de"].map(_norm_de)
    gb["_cs_ok"] = gb["gerichtsbezirk_cs"].map(_is_accented_cs)
    gb = gb.sort_values("_cs_ok", ascending=False).drop_duplicates("k").set_index("k")
    choices = list(gb.index)

    def _match(k: str) -> str | None:
        if not k:
            return None
        if k in gb.index:
            return k
        m = process.extractOne(k, choices, scorer=fuzz.ratio)
        return m[0] if m and m[1] >= threshold else None

    out = backbone.copy()
    out["_mk"] = out["gb_de_norm"].map(_match)
    out["gz_gb_cs"]    = out["_mk"].map(lambda k: gb.loc[k, "gerichtsbezirk_cs"] if k in gb.index else None)
    out["gz_polit_de"] = out["_mk"].map(lambda k: gb.loc[k, "polit_bezirk_de"] if k in gb.index else None)
    out["gz_polit_cs"] = out["_mk"].map(lambda k: gb.loc[k, "polit_bezirk_cs"] if k in gb.index else None)
    return out.drop(columns="_mk")

def build_district_backbone(dbf_path, curation_csv=None) -> pd.DataFrame:
    """Assemble the district backbone: MPIDR spine + curated Czech names.

    One row per court district (Gerichtsbezirk). Czech names come from the curation file
    (authoritative), falling back to MPIDR's de-accented Czech (flagged ``needs_review``)
    when a unit is not yet curated. The political-okres Czech name is taken from the
    curation ``okres`` map; the okres seat (court whose Czech name equals the okres name)
    is flagged ``is_okres_seat``.
    """
    df = parse_mpidr(dbf_path)
    court_cs, okres_cs, court_modern, okres_modern = _load_curation(curation_csv)

    res = df.apply(lambda r: _resolve_cs(None, r["gb_cs_mpidr"], court_cs.get(r["gb_de"])), axis=1)
    df["gerichtsbezirk_de"]         = df["gb_de"]
    df["gerichtsbezirk_cs"]         = [x[0] for x in res]
    df["gerichtsbezirk_cs_modern"]  = df["gb_de"].map(court_modern)   # today's name if renamed
    df["court_cs_source"]           = [x[1] for x in res]
    df["court_needs_review"]        = [x[2] for x in res]

    # okres Czech: curated by code; fallback = de-accented seat court (any court sharing
    # the curated okres name), else the first court's de-accented name.
    okres_res = {}
    for code, sub in df.groupby("okres_code"):
        curated = okres_cs.get(code)
        fallback = sub.iloc[0]["gb_cs_mpidr"]
        okres_res[code] = _resolve_cs(None, fallback, curated)
    df["okres_cs"]           = df["okres_code"].map(lambda c: okres_res[c][0])
    df["okres_cs_modern"]    = df["okres_code"].map(okres_modern)
    df["okres_cs_source"]    = df["okres_code"].map(lambda c: okres_res[c][1])
    df["okres_needs_review"] = df["okres_code"].map(lambda c: okres_res[c][2])

    # seat = court whose Czech name equals the okres name (ignoring the "(město)" marker
    # that distinguishes the statutory-city okres from its rural twin)
    okres_base = df["okres_cs"].str.replace(r"\s*\(město\)\s*$", "", regex=True)
    df["is_okres_seat"] = df["gerichtsbezirk_cs"].eq(okres_base) & okres_base.notna()
    seat_de = (df[df["is_okres_seat"]].drop_duplicates("okres_code")
                 .set_index("okres_code")["gerichtsbezirk_de"])
    df["okres_de"] = df["okres_code"].map(seat_de)
    # fill okresy whose seat is not a court district in the spine (e.g. 302-2 Bärn)
    df["okres_de"] = df["okres_de"].fillna(df["okres_code"].map(OKRES_DE_OVERRIDE))
    # repair the MPIDR transliterations of German names (Sziszkow = Žižkow)
    df["gerichtsbezirk_de"] = df["gerichtsbezirk_de"].replace(GERMAN_NAME_FIX)
    df["okres_de"] = df["okres_de"].replace(GERMAN_NAME_FIX)

    cols = ["land", "okres_code", "okres_de", "okres_cs", "okres_cs_modern",
            "okres_cs_source", "okres_needs_review",
            "court_id", "mpidr_oid", "gerichtsbezirk_de", "gerichtsbezirk_cs",
            "gerichtsbezirk_cs_modern", "court_cs_source", "court_needs_review",
            "is_okres_seat", "gb_cs_mpidr"]
    return df[cols].sort_values(["land", "okres_code", "court_id"]).reset_index(drop=True)

def write_curation_template(dbf_path, path) -> pd.DataFrame:
    """Write the blank Czech-name curation skeleton (schema consumed by :func:`_load_curation`).

    One ``court`` row per court district (key = German name) and one ``okres`` row per
    ``okres_code``, each with the MPIDR de-accented Czech as an edit seed and an empty
    ``cs`` to fill. Columns beyond ``level/key/cs`` (``land``, ``de``,
    ``mpidr_cs_deaccented``) are human aids and ignored by the loader.
    """
    mp = parse_mpidr(dbf_path)
    rows = []
    for gb_de, sub in mp.drop_duplicates("gb_de").groupby("gb_de", sort=False):
        r = sub.iloc[0]
        rows.append({"level": "court", "key": gb_de, "land": r["land"],
                     "de": gb_de, "mpidr_cs_deaccented": r["gb_cs_mpidr"], "cs": ""})
    for code, sub in mp.groupby("okres_code"):
        r = sub.iloc[0]
        rows.append({"level": "okres", "key": code, "land": r["land"],
                     "de": "", "mpidr_cs_deaccented": "", "cs": ""})
    tmpl = pd.DataFrame(rows, columns=["level", "key", "land", "de", "mpidr_cs_deaccented", "cs"])
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    tmpl.to_csv(p, index=False, encoding="utf-8")
    return tmpl

def districts_report(backbone: pd.DataFrame) -> None:
    """Print coverage of the district backbone (units, Czech-name provenance, review load)."""
    n_court = len(backbone)
    n_okres = backbone["okres_code"].nunique()
    print(f"District backbone  (court districts = {n_court}, political okresy = {n_okres})\n")
    for land, sub in backbone.groupby("land"):
        print(f"  {land:<10} court={len(sub):>4}  okres={sub['okres_code'].nunique():>3}")
    print("\n  court-district Czech name source:")
    for src, n in backbone["court_cs_source"].value_counts().items():
        print(f"      {src:<18} {n:>4}")
    okres = backbone.drop_duplicates("okres_code")
    print("  political-okres Czech name source:")
    for src, n in okres["okres_cs_source"].value_counts().items():
        print(f"      {src:<18} {n:>4}")
    print(f"\n  needs review: {int(backbone['court_needs_review'].sum())} court / "
          f"{int(okres['okres_needs_review'].sum())} okres")

def validate_against_code(backbone: pd.DataFrame, code_path) -> dict[str, pd.DataFrame]:
    """Audit the backbone's German names against the census CODE-file (Rumpler-Seger codebook).

    The census codebook (``CODE file… Coding IDs`` sheet) is the same project as the MPIDR spine,
    so ``IDCourtdistr`` equals ``court_id`` exactly — an independent authority for the German unit
    names. Joins on that key and compares (accent/spelling-normalised) the court name
    ``gerichtsbezirk_de`` ↔ ``NameCourtdistr`` and the okres name ``okres_de`` ↔ ``NamePoldistrict``.
    Prints a report and returns ``{"court": …, "okres": …}`` frames of the genuine mismatches.
    """
    code = (pd.read_excel(code_path, sheet_name="Coding IDs")
              .drop_duplicates("IDCourtdistr")
            [["IDCourtdistr", "NameCourtdistr", "IDPoldistrict", "NamePoldistrict"]]).copy()
    # court_id comes off the .dbf reader as a string but IDCourtdistr reads as int64 —
    # normalise both to a canonical string id so the exact-code merge doesn't dtype-clash.
    code["IDCourtdistr"] = code["IDCourtdistr"].astype("Int64").astype(str)
    bb = backbone.copy()
    bb["court_id"] = bb["court_id"].astype("Int64").astype(str)
    m = bb.merge(code, left_on="court_id", right_on="IDCourtdistr", how="left")

    # the codebook names are bilingual "German/Czech" (e.g. "Asch/As", "Trautenau/Trutnov");
    # compare only the German part (before the slash) against our German-only names.
    def _de(s):
        return str(s).split("/")[0] if pd.notna(s) else s

    m["_gb_ok"] = [_gnorm_de(a) == _gnorm_de(_de(b))
                   for a, b in zip(m["gerichtsbezirk_de"], m["NameCourtdistr"])]
    court_mis = (m.loc[~m["_gb_ok"], ["okres_code", "court_id", "gerichtsbezirk_de", "NameCourtdistr"]]
                 .reset_index(drop=True))
    ok = m.drop_duplicates("okres_code").copy()
    ok["_ok"] = [_gnorm_de(a) == _gnorm_de(_de(b)) for a, b in zip(ok["okres_de"], ok["NamePoldistrict"])]
    okres_mis = (ok.loc[~ok["_ok"], ["okres_code", "okres_cs", "okres_de", "NamePoldistrict"]]
                 .reset_index(drop=True))

    joined = int(m["IDCourtdistr"].notna().sum())
    print(f"CODE-file audit: {joined}/{len(m)} courts joined on court_id == IDCourtdistr")
    print(f"  court German-name mismatches: {len(court_mis)}")
    if len(court_mis):
        print(court_mis.to_string(index=False))
    print(f"  okres German-name mismatches:  {len(okres_mis)}")
    if len(okres_mis):
        print(okres_mis.to_string(index=False))
    return {"court": court_mis, "okres": okres_mis}

"""
districts.py — 1910 administrative-district backbone (okres + soudní okres), in Czech
=====================================================================================
Builds the **authoritative** 1910 district crosswalk for the Czech crown lands from the
**MPIDR / Rumpler-Seger** GIS boundary file (``reference/places/mpidr/``) — the spine — with
curated Czech names from ``reference/places/district_cs_curation.csv``.

Why MPIDR is the spine
----------------------
The MPIDR shapefile (MPIDR & CGG 2012, after Rumpler & Seger 2010) is a clean, complete
polygon layer of the 1910 census administrative units with a stable 3-level hierarchy
encoded in ``ID_1 = REGION·10^6 + DISTRICT·10^3 + LOCAL_``:

    LAND (crown land)  →  DISTRICT (political okres)  →  LOCAL_ (Gerichtsbezirk / soudní okres)

so it authoritatively fixes **which court districts belong to which political okres**,
carries **stable geo-IDs + polygons** (the census-merge key and the spatial backbone), and
its German names are clean apart from six transliterations repaired by
:data:`GERMAN_NAME_FIX` (``Sziszkow`` = Žižkow).

Czech names — provenance
------------------------
Each district's canonical Czech name (``*_cs``) is resolved, best-first:

  1. **curated** (``district_cs_curation.csv``) — the accented canonical form; all 330
     court and 158 okres names are curated, so this is the only tier in use;
  2. **MPIDR de-accented** Czech (the ``NAME`` field's post-slash part) — a fallback that
     is the right name minus diacritics, flagged ``needs_review``.

The OCR *Ortsverzeichnis* gazetteer once supplied a third tier; its Czech district names
proved unusable ("Bude jo vice") and it is archived (``_archive/gazetteer.py``).
:func:`attach_gazetteer_names` is kept only for that history.

Note on the city/rural twins: both members of a pair carry the same ``okres_de`` (the
seat court's German name, e.g. ``Olmuetz``); ``okres_cs`` is what tells them apart
(``Olomouc`` / ``Olomouc (město)``).

Usage
-----
    from utils.parsing import districts as dz
    backbone = dz.build_district_backbone(DBF_PATH, curation_csv=CURATION)   # 339 court districts
    dz.districts_report(backbone)
    dz.write_curation_template(backbone, "reference/places/district_cs_curation.csv")
"""

from __future__ import annotations

import re
import struct
import unicodedata
from pathlib import Path

import pandas as pd

__version__ = "1.2.0"   # MPIDR German-name repairs; docstring (audit 2026-09-22)

# German crown-land code → Czech-lands label (the three we keep; empire-wide DBF is filtered).
LAND_MAP: dict[str, str] = {"BOEHMEN": "Böhmen", "MAEHREN": "Mähren", "SCHLESIEN": "Schlesien"}

_FUZZY_THRESHOLD = 82  # rapidfuzz ratio on normalised German names for the gazetteer join


# ═══════════════════════════════════════════════════════════════════════════
# Minimal dBASE III/IV reader (stdlib only — avoids a geopandas/pyshp dependency;
# geometry is not needed here, the MPIDR IDs link back to the .shp)
# ═══════════════════════════════════════════════════════════════════════════
def read_dbf(path, encodings: tuple[str, ...] = ("cp1250", "latin-1", "utf-8")) -> list[dict]:
    """Read a .dbf attribute table into a list of ``{field: value}`` dicts."""
    raw = Path(path).read_bytes()
    n_rec = struct.unpack("<I", raw[4:8])[0]
    hdr_len = struct.unpack("<H", raw[8:10])[0]
    rec_len = struct.unpack("<H", raw[10:12])[0]

    fields: list[tuple[str, int]] = []
    off = 32
    while raw[off] != 0x0D:                                   # 0x0D terminates the descriptor array
        desc = raw[off:off + 32]
        name = desc[0:11].split(b"\x00")[0].decode("latin-1")
        fields.append((name, desc[16]))                      # byte 16 = field length
        off += 32

    def _decode(b: bytes) -> str:
        for enc in encodings:
            try:
                return b.decode(enc).strip()
            except UnicodeDecodeError:
                continue
        return b.decode("latin-1", "replace").strip()

    records: list[dict] = []
    for i in range(n_rec):
        row = raw[hdr_len + i * rec_len: hdr_len + (i + 1) * rec_len]
        if not row or row[0:1] == b"*":                      # deletion flag
            continue
        pos, rec = 1, {}
        for name, flen in fields:
            rec[name] = _decode(row[pos:pos + flen])
            pos += flen
        records.append(rec)
    return records


# ═══════════════════════════════════════════════════════════════════════════
# Name normalisation
# ═══════════════════════════════════════════════════════════════════════════
def _norm_de(s: str | None) -> str:
    """Match key for a German place name (umlauts + oe/ae/ue digraphs → base, alnum only)."""
    if not s:
        return ""
    s = s.strip().lower()
    s = s.replace("ö", "o").replace("ä", "a").replace("ü", "u").replace("ß", "ss")
    s = s.replace("oe", "o").replace("ae", "a").replace("ue", "u")
    s = "".join(c for c in unicodedata.normalize("NFKD", s) if not unicodedata.combining(c))
    return re.sub(r"[^a-z0-9]", "", s)


def _is_accented_cs(s) -> bool:
    """True if *s* looks like a real accented/mixed-case Czech name (not de-accented junk)."""
    if not isinstance(s, str) or not s.strip():
        return False
    if any(c in s for c in "áčďéěíňóřšťúůýžÁČĎÉĚÍŇÓŘŠŤÚŮÝŽ"):
        return True
    return s[:1].isupper() and s.lower() != s  # mixed-case multi-word (e.g. "Karlovy Vary")


def _strip_stadt(cs: str | None) -> str | None:
    """Drop trailing 'St.'/'Stadt'/'Land'/'Umgebung' role markers from a truncated CS name."""
    if not isinstance(cs, str):
        return cs
    return re.sub(r"\s+(st\.?|stadt|land|okoli|umgebung|mesto|mest.?)\s*$", "", cs.strip(),
                  flags=re.IGNORECASE).strip() or None


# ═══════════════════════════════════════════════════════════════════════════
# MPIDR backbone
# ═══════════════════════════════════════════════════════════════════════════
def parse_mpidr(dbf_path, lands: dict[str, str] = LAND_MAP) -> pd.DataFrame:
    """Parse the MPIDR .dbf into a court-district backbone for the Czech lands.

    One row per Gerichtsbezirk (LOCAL_ polygon). Columns:
        ``land`` · ``okres_code`` (REGION-DISTRICT) · ``court_id`` (ID_1, unique) ·
        ``mpidr_oid`` (OID_ → .shp geometry) · ``gb_de`` (German court-district name) ·
        ``gb_cs_mpidr`` (MPIDR de-accented Czech, post-slash) · ``gb_de_norm`` (match key).
    """
    rows = []
    for r in read_dbf(dbf_path):
        if r["LAND"] not in lands:
            continue
        de, _, cs = r["NAME"].partition("/")
        rows.append({
            "land":        lands[r["LAND"]],
            "okres_code":  f"{r['REGION']}-{r['DISTRICT']}",
            "court_id":    r["ID_1"],
            "mpidr_oid":   r["OID_"],
            "gb_de":       de.strip(),
            "gb_cs_mpidr": cs.strip() or None,
            "gb_de_norm":  _norm_de(de),
        })
    return pd.DataFrame(rows)


def attach_gazetteer_names(backbone: pd.DataFrame, gazetteer: pd.DataFrame,
                           threshold: int = _FUZZY_THRESHOLD) -> pd.DataFrame:
    """Fuzzy-join gazetteer Czech names onto the MPIDR backbone via the German name.

    Brings ``gz_gb_cs`` (court-district Czech), ``gz_polit_de`` / ``gz_polit_cs``
    (political-okres German/Czech) from the gazetteer, matched on the normalised German
    Gerichtsbezirk name (exact first, then rapidfuzz ratio ≥ *threshold*). Rows are
    preferred that already carry an accented Czech court name.
    """
    from rapidfuzz import process, fuzz

    gb = gazetteer[["gerichtsbezirk_de", "gerichtsbezirk_cs",
                    "polit_bezirk_de", "polit_bezirk_cs"]].dropna(subset=["gerichtsbezirk_de"]).copy()
    gb["k"] = gb["gerichtsbezirk_de"].map(_norm_de)
    gb["_cs_ok"] = gb["gerichtsbezirk_cs"].map(_is_accented_cs)
    gb = gb.sort_values("_cs_ok", ascending=False).drop_duplicates("k").set_index("k")
    choices = list(gb.index)

    def _match(k: str) -> str | None:
        if not k:
            return None
        if k in gb.index:
            return k
        m = process.extractOne(k, choices, scorer=fuzz.ratio)
        return m[0] if m and m[1] >= threshold else None

    out = backbone.copy()
    out["_mk"] = out["gb_de_norm"].map(_match)
    out["gz_gb_cs"]    = out["_mk"].map(lambda k: gb.loc[k, "gerichtsbezirk_cs"] if k in gb.index else None)
    out["gz_polit_de"] = out["_mk"].map(lambda k: gb.loc[k, "polit_bezirk_de"] if k in gb.index else None)
    out["gz_polit_cs"] = out["_mk"].map(lambda k: gb.loc[k, "polit_bezirk_cs"] if k in gb.index else None)
    return out.drop(columns="_mk")


def _resolve_cs(row_gz_cs, row_mpidr_cs, curated: str | None) -> tuple[str | None, str, bool]:
    """Best-first Czech name → (name, source, needs_review).

    Ordering reflects a hard lesson from the data: the OCR gazetteer's *accented* Czech
    district names are badly mangled ("Bude jo vice" = Budějovice, "Suchy Did"), so they
    are **not** trusted over MPIDR. MPIDR's de-accented Czech has the right letters (just
    no diacritics) and is the reliable base; the curated override is authoritative.
        1. curated  → final (accents verified)
        2. MPIDR de-accented → base, but needs_review (diacritics unverified)
        3. gazetteer accented → last resort when MPIDR has no Czech at all
    """
    if isinstance(curated, str) and curated.strip():
        return curated.strip(), "curated", False
    if isinstance(row_mpidr_cs, str) and row_mpidr_cs.strip():
        return _strip_stadt(row_mpidr_cs), "mpidr_deaccented", True
    if _is_accented_cs(row_gz_cs):
        return _strip_stadt(row_gz_cs), "gazetteer", True
    return None, "none", True


def _load_curation(curation_csv) -> tuple[dict, dict, dict, dict]:
    """Read the curated names → (court_cs, okres_cs, court_modern, okres_modern).

    The gazetteer proved unusable for district names, so curation is the *only* accented
    source. Columns: ``level`` ("court"/"okres"), ``key`` (German court name / ``okres_code``),
    ``cs`` (canonical 1910 Czech), optional ``cs_modern`` (today's name if renamed).
    """
    court, okres, court_m, okres_m = {}, {}, {}, {}
    if curation_csv and Path(curation_csv).exists():
        cur = pd.read_csv(curation_csv, dtype=str).fillna("")
        has_modern = "cs_modern" in cur.columns
        for _, r in cur.iterrows():
            key, cs = r["key"].strip(), r["cs"].strip()
            modern = r["cs_modern"].strip() if has_modern else ""
            if not cs:
                continue
            if r["level"] == "court":
                court[key] = cs
                if modern:
                    court_m[key] = modern
            elif r["level"] == "okres":
                okres[key] = cs
                if modern:
                    okres_m[key] = modern
    return court, okres, court_m, okres_m


# Political-okres German seat name where it cannot be derived from a court district (the okres
# seat is not itself a Gerichtsbezirk in the MPIDR spine). 302-2 Moravský Beroun = Bärn, whose
# court sits elsewhere; confirmed against the census CODE-file NamePoldistrict.
OKRES_DE_OVERRIDE: dict[str, str] = {"302-2": "Bärn"}

# German names the MPIDR ``NAME`` field transliterates with ``sz`` for ``ž`` or otherwise
# mangles; corrected against the census CODE-file (``NameCourtdistr`` / ``NamePoldistrict``,
# the same Rumpler-Seger project), which :func:`validate_against_code` lists as mismatches.
# Applied to the delivered ``gerichtsbezirk_de`` / ``okres_de`` only — the Czech-name
# curation stays keyed on the raw MPIDR spelling (audit 2026-09-22 §B25).
GERMAN_NAME_FIX: dict[str, str] = {
    "Sziszkow": "Žižkow",
    "Neubydszow": "Neubydžow",
    "Rosznau am Radhost": "Rosenau am Radhost",
    "Reichenau an der Kneszna": "Reichenau an der Kněžna",
    "Lomnitz an der Lusznitz": "Lomnitz an der Lužnitz",
    "Weseli an der Lusznitz": "Weseli an der Lužnitz",
}


def build_district_backbone(dbf_path, curation_csv=None) -> pd.DataFrame:
    """Assemble the district backbone: MPIDR spine + curated Czech names.

    One row per court district (Gerichtsbezirk). Czech names come from the curation file
    (authoritative), falling back to MPIDR's de-accented Czech (flagged ``needs_review``)
    when a unit is not yet curated. The political-okres Czech name is taken from the
    curation ``okres`` map; the okres seat (court whose Czech name equals the okres name)
    is flagged ``is_okres_seat``.
    """
    df = parse_mpidr(dbf_path)
    court_cs, okres_cs, court_modern, okres_modern = _load_curation(curation_csv)

    res = df.apply(lambda r: _resolve_cs(None, r["gb_cs_mpidr"], court_cs.get(r["gb_de"])), axis=1)
    df["gerichtsbezirk_de"]         = df["gb_de"]
    df["gerichtsbezirk_cs"]         = [x[0] for x in res]
    df["gerichtsbezirk_cs_modern"]  = df["gb_de"].map(court_modern)   # today's name if renamed
    df["court_cs_source"]           = [x[1] for x in res]
    df["court_needs_review"]        = [x[2] for x in res]

    # okres Czech: curated by code; fallback = de-accented seat court (any court sharing
    # the curated okres name), else the first court's de-accented name.
    okres_res = {}
    for code, sub in df.groupby("okres_code"):
        curated = okres_cs.get(code)
        fallback = sub.iloc[0]["gb_cs_mpidr"]
        okres_res[code] = _resolve_cs(None, fallback, curated)
    df["okres_cs"]           = df["okres_code"].map(lambda c: okres_res[c][0])
    df["okres_cs_modern"]    = df["okres_code"].map(okres_modern)
    df["okres_cs_source"]    = df["okres_code"].map(lambda c: okres_res[c][1])
    df["okres_needs_review"] = df["okres_code"].map(lambda c: okres_res[c][2])

    # seat = court whose Czech name equals the okres name (ignoring the "(město)" marker
    # that distinguishes the statutory-city okres from its rural twin)
    okres_base = df["okres_cs"].str.replace(r"\s*\(město\)\s*$", "", regex=True)
    df["is_okres_seat"] = df["gerichtsbezirk_cs"].eq(okres_base) & okres_base.notna()
    seat_de = (df[df["is_okres_seat"]].drop_duplicates("okres_code")
                 .set_index("okres_code")["gerichtsbezirk_de"])
    df["okres_de"] = df["okres_code"].map(seat_de)
    # fill okresy whose seat is not a court district in the spine (e.g. 302-2 Bärn)
    df["okres_de"] = df["okres_de"].fillna(df["okres_code"].map(OKRES_DE_OVERRIDE))
    # repair the MPIDR transliterations of German names (Sziszkow = Žižkow)
    df["gerichtsbezirk_de"] = df["gerichtsbezirk_de"].replace(GERMAN_NAME_FIX)
    df["okres_de"] = df["okres_de"].replace(GERMAN_NAME_FIX)

    cols = ["land", "okres_code", "okres_de", "okres_cs", "okres_cs_modern",
            "okres_cs_source", "okres_needs_review",
            "court_id", "mpidr_oid", "gerichtsbezirk_de", "gerichtsbezirk_cs",
            "gerichtsbezirk_cs_modern", "court_cs_source", "court_needs_review",
            "is_okres_seat", "gb_cs_mpidr"]
    return df[cols].sort_values(["land", "okres_code", "court_id"]).reset_index(drop=True)


def write_curation_template(dbf_path, path) -> pd.DataFrame:
    """Write the blank Czech-name curation skeleton (schema consumed by :func:`_load_curation`).

    One ``court`` row per court district (key = German name) and one ``okres`` row per
    ``okres_code``, each with the MPIDR de-accented Czech as an edit seed and an empty
    ``cs`` to fill. Columns beyond ``level/key/cs`` (``land``, ``de``,
    ``mpidr_cs_deaccented``) are human aids and ignored by the loader.
    """
    mp = parse_mpidr(dbf_path)
    rows = []
    for gb_de, sub in mp.drop_duplicates("gb_de").groupby("gb_de", sort=False):
        r = sub.iloc[0]
        rows.append({"level": "court", "key": gb_de, "land": r["land"],
                     "de": gb_de, "mpidr_cs_deaccented": r["gb_cs_mpidr"], "cs": ""})
    for code, sub in mp.groupby("okres_code"):
        r = sub.iloc[0]
        rows.append({"level": "okres", "key": code, "land": r["land"],
                     "de": "", "mpidr_cs_deaccented": "", "cs": ""})
    tmpl = pd.DataFrame(rows, columns=["level", "key", "land", "de", "mpidr_cs_deaccented", "cs"])
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    tmpl.to_csv(p, index=False, encoding="utf-8")
    return tmpl


def districts_report(backbone: pd.DataFrame) -> None:
    """Print coverage of the district backbone (units, Czech-name provenance, review load)."""
    n_court = len(backbone)
    n_okres = backbone["okres_code"].nunique()
    print(f"District backbone  (court districts = {n_court}, political okresy = {n_okres})\n")
    for land, sub in backbone.groupby("land"):
        print(f"  {land:<10} court={len(sub):>4}  okres={sub['okres_code'].nunique():>3}")
    print("\n  court-district Czech name source:")
    for src, n in backbone["court_cs_source"].value_counts().items():
        print(f"      {src:<18} {n:>4}")
    okres = backbone.drop_duplicates("okres_code")
    print("  political-okres Czech name source:")
    for src, n in okres["okres_cs_source"].value_counts().items():
        print(f"      {src:<18} {n:>4}")
    print(f"\n  needs review: {int(backbone['court_needs_review'].sum())} court / "
          f"{int(okres['okres_needs_review'].sum())} okres")


def _gnorm_de(s) -> str:
    """German name → comparison key (ä/ö/ü/ß → ae/oe/ue/ss, drop accents/punct/case)."""
    import unicodedata
    if pd.isna(s):
        return ""
    s = (str(s).strip().lower()
         .replace("ä", "ae").replace("ö", "oe").replace("ü", "ue").replace("ß", "ss"))
    s = "".join(c for c in unicodedata.normalize("NFKD", s) if not unicodedata.combining(c))
    return re.sub(r"[^a-z0-9]", "", s)


def validate_against_code(backbone: pd.DataFrame, code_path) -> dict[str, pd.DataFrame]:
    """Audit the backbone's German names against the census CODE-file (Rumpler-Seger codebook).

    The census codebook (``CODE file… Coding IDs`` sheet) is the same project as the MPIDR spine,
    so ``IDCourtdistr`` equals ``court_id`` exactly — an independent authority for the German unit
    names. Joins on that key and compares (accent/spelling-normalised) the court name
    ``gerichtsbezirk_de`` ↔ ``NameCourtdistr`` and the okres name ``okres_de`` ↔ ``NamePoldistrict``.
    Prints a report and returns ``{"court": …, "okres": …}`` frames of the genuine mismatches.
    """
    code = (pd.read_excel(code_path, sheet_name="Coding IDs")
              .drop_duplicates("IDCourtdistr")
            [["IDCourtdistr", "NameCourtdistr", "IDPoldistrict", "NamePoldistrict"]]).copy()
    # court_id comes off the .dbf reader as a string but IDCourtdistr reads as int64 —
    # normalise both to a canonical string id so the exact-code merge doesn't dtype-clash.
    code["IDCourtdistr"] = code["IDCourtdistr"].astype("Int64").astype(str)
    bb = backbone.copy()
    bb["court_id"] = bb["court_id"].astype("Int64").astype(str)
    m = bb.merge(code, left_on="court_id", right_on="IDCourtdistr", how="left")

    # the codebook names are bilingual "German/Czech" (e.g. "Asch/As", "Trautenau/Trutnov");
    # compare only the German part (before the slash) against our German-only names.
    def _de(s):
        return str(s).split("/")[0] if pd.notna(s) else s

    m["_gb_ok"] = [_gnorm_de(a) == _gnorm_de(_de(b))
                   for a, b in zip(m["gerichtsbezirk_de"], m["NameCourtdistr"])]
    court_mis = (m.loc[~m["_gb_ok"], ["okres_code", "court_id", "gerichtsbezirk_de", "NameCourtdistr"]]
                 .reset_index(drop=True))
    ok = m.drop_duplicates("okres_code").copy()
    ok["_ok"] = [_gnorm_de(a) == _gnorm_de(_de(b)) for a, b in zip(ok["okres_de"], ok["NamePoldistrict"])]
    okres_mis = (ok.loc[~ok["_ok"], ["okres_code", "okres_cs", "okres_de", "NamePoldistrict"]]
                 .reset_index(drop=True))

    joined = int(m["IDCourtdistr"].notna().sum())
    print(f"CODE-file audit: {joined}/{len(m)} courts joined on court_id == IDCourtdistr")
    print(f"  court German-name mismatches: {len(court_mis)}")
    if len(court_mis):
        print(court_mis.to_string(index=False))
    print(f"  okres German-name mismatches:  {len(okres_mis)}")
    if len(okres_mis):
        print(okres_mis.to_string(index=False))
    return {"court": court_mis, "okres": okres_mis}
