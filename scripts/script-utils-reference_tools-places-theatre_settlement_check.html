How it works

The clause check (theatre_rule_audit) asks whether each clause was read right. It cannot see what goes wrong only when clauses are put together: a company counted twice across the place chronicle and an ensemble page, a lineage split into two companies, a place tied to the wrong 1910 settlement, a decade left empty between two listed decades. And the corrections for a text-built variable in a regression (Egami et al. 2023; Ludwig, Mullainathan & Rambachan 2026) need the expert value at the level of the variable that enters the regression: here, a 1910 settlement in a decade. This is that sample.

Population: every 1910 settlement of the Czech lands tied to at least one database place (6,668 on the build of 2026-11-03). A settlement tied to none is a zero of the measure with no text to read; a place the tie missed is outside this check.

Strata by what the measure says of 1850–1913: certain (Czech activity certain in some year), possible (possible only) and none. TAKE per stratum, alternated into a calibration and a certification half; weight = stratum size / drawn.

The question, for each decade 1850s … 1910s (1910–1913): was Czech amateur theatre active in this place at some time in the decade? yes / possible (the text allows it but does not pin it) / no / unsure. The measure's answer is filled in (certain in some year of the decade → yes, possible → possible, else no), and the author corrects it.

def decade_measure(panel: pd.DataFrame | None = None) -> pd.DataFrame:
    """Per settlement and decade 1850–1913: the measure's answer (``yes`` / ``possible`` / ``no``)."""

def population(ties: pd.DataFrame, measure: pd.DataFrame) -> pd.DataFrame:
    """Settlements tied to a database place and carried by the measure (a 1910 settlement with
    a resolved okres and status), with ``stratum`` from the measure 1850–1913.  A place tied

def draw(pop: pd.DataFrame, seed: int = SEED) -> pd.DataFrame:
    rng = np.random.default_rng(seed)
    parts = []
    for stratum, group in pop.groupby("stratum"):
        n = min(TAKE[stratum], len(group))
        drawn = group.iloc[np.sort(rng.permutation(len(group))[:n])].copy()
        drawn["half"] = ["calibration" if i % 2 == 0 else "certification" for i in range(n)]
        drawn["stratum_N"], drawn["stratum_n"] = len(group), n
        drawn["weight"] = round(len(group) / n, 4)
        parts.append(drawn)
    sample = pd.concat(parts, ignore_index=True)
    return sample.iloc[rng.permutation(len(sample))].reset_index(drop=True)

def write_sheet(sample: pd.DataFrame, path: Path = SHEET) -> Path:
    """The worksheet: the measure's answer per decade and an empty verdict column per decade.
    Refuses to overwrite a sheet that holds any verdict."""
    path = Path(path)
    verdicts = [f"v_{d}" for d in DECADES] + ["v_note"]
    if path.exists():
        old = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
        if (old[[c for c in verdicts if c in old]] != "").any().any():
            raise FileExistsError(f"{path} already holds verdicts; it is not redrawn")
    sheet = sample.copy()
    for column in verdicts:
        sheet[column] = ""
    sheet.to_csv(path, index=False, encoding="utf-8-sig")
    return path

def settlement_texts(sheet: pd.DataFrame, ties: pd.DataFrame) -> dict[str, list[dict]]:
    """For each settlement: its database places, each with the first report, the chronicle,
    the ensemble list and the ensemble pages of companies starting before 1914."""
    places = pd.read_csv(RAW / "places.csv", low_memory=False,
                         usecols=["geo_id", "name", "type", "first_record_raw", "chronicle_text"]).set_index("geo_id")
    ensembles = pd.read_csv(RAW / "ensembles.csv", usecols=["ensemble_id", "geo_id", "name", "span_raw", "year_from"])
    pages = pd.read_csv(RAW / "ensemble_pages.csv", usecols=["ensemble_id", "active_raw", "chronicle_text"]).set_index("ensemble_id")
    wanted = ties[ties["settlement_id"].isin(sheet["settlement_id"])]
    out: dict[str, list[dict]] = {}
    for sid, group in wanted.groupby("settlement_id"):
        entries = []
        for geo in group["geo_id"]:
            p = places.loc[geo]
            ens = ensembles[ensembles["geo_id"] == geo].sort_values("year_from")
            listed, own = [], []
            for e in ens.itertuples(index=False):
                listed.append({"name": _clean(e.name), "span": _clean(e.span_raw).strip("()")})
                if e.ensemble_id in pages.index and (pd.isna(e.year_from) or e.year_from < 1914):
                    pg = pages.loc[e.ensemble_id]
                    if _clean(pg["chronicle_text"]) or _clean(pg["active_raw"]):
                        own.append({"name": _clean(e.name), "span": _clean(e.span_raw).strip("()"),
                                    "active": _clean(pg["active_raw"]), "chronicle": _clean(pg["chronicle_text"])})
            entries.append({"name": _clean(p["name"]), "type": _clean(p["type"]),
                            "first": _clean(p["first_record_raw"]), "chronicle": _clean(p["chronicle_text"]),
                            "ensembles": listed, "pages": own})
        out[str(sid)] = entries
    return out

def build_page(out: Path, sheet_path: Path = SHEET) -> Path:
    sheet = pd.read_csv(sheet_path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    texts = settlement_texts(sheet, settlement_ties())
    sheet = sheet.sort_values("half", key=lambda h: h != "calibration", kind="stable").reset_index(drop=True)
    data = [{"i": i, "key": str(r.settlement_id), "half": r.half, "name": r.cs_name.title(), "okres": r.okres_1910_cs,
             "land": r.land, "measure": {str(d): getattr(r, f"measure_{d}") for d in DECADES},
             "places": texts.get(str(r.settlement_id), [])}
            for i, r in enumerate(sheet.itertuples(index=False))]
    payload = json.dumps(data, ensure_ascii=False).replace("</", "<\\/")
    template = TEMPLATE.read_text(encoding="utf-8")
    assert template.count(PLACEHOLDER) == 1, "template: one data placeholder"
    Path(out).write_text(template.replace(PLACEHOLDER, payload), encoding="utf-8")
    return Path(out)

def fill_sheet(documents: list[dict], path: Path = SHEET) -> dict:
    """Write the page's documents (``verdicts/<settlement_id>``) into the sheet."""
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    sheet["_key"] = sheet["settlement_id"].astype(str)
    index = {k: i for i, k in enumerate(sheet["_key"])}
    unknown = [d.get("settlement_id") for d in documents if str(d.get("settlement_id")) not in index]
    assert not unknown, f"verdicts for settlements not in the sheet: {unknown[:5]}"
    columns = [f"v_{d}" for d in DECADES] + ["v_note"]
    sheet[columns] = ""
    written = 0
    for d in documents:
        if not d.get("ruled"):
            continue
        answers = d.get("decades") or {}
        sheet.loc[index[str(d["settlement_id"])], columns] = [answers.get(str(x), "") for x in DECADES] + [d.get("note") or ""]
        written += 1
    sheet.drop(columns="_key").to_csv(path, index=False, encoding="utf-8-sig")
    return {"documents": len(documents), "ruled": written, "rows": len(sheet)}

def score(path: Path = SHEET, half: str = "calibration", measure: pd.DataFrame | None = None) -> pd.DataFrame:
    """The measure against the author per settlement-decade, weighted: for ``certain`` (the
    measure's yes against the author's yes) and ``possible`` (the measure's yes or possible
    against the author's yes or possible), precision, recall and agreement over the cells
    the author could tell.  ``measure`` (from :func:`decade_measure`) scores a later build;
    without it, the measure as drawn (the answers the page showed)."""
    if half == "certification" and not RULES_FROZEN:
        raise PermissionError("the certification half is sealed until RULES_FROZEN is True")
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    sheet = sheet[(sheet["half"] == half) & (sheet[[f"v_{d}" for d in DECADES]] != "").all(axis=1)]
    cells = []
    for r in sheet.itertuples(index=False):
        for d in DECADES:
            said = getattr(r, f"measure_{d}") if measure is None else (
                measure.at[r.settlement_id, d] if r.settlement_id in measure.index else "no")
            cells.append({"weight": float(r.weight), "measure": said, "author": getattr(r, f"v_{d}")})
    c = pd.DataFrame(cells, columns=["weight", "measure", "author"])
    c = c[c["author"] != "unsure"]
    out = []
    for name, levels in (("certain", {"yes"}), ("possible", {"yes", "possible"})):
        m, a, w = c["measure"].isin(levels), c["author"].isin(levels), c["weight"]
        share = lambda mask, hit: float((w[mask] * hit[mask]).sum() / w[mask].sum()) if w[mask].sum() else np.nan
        out.append({"level": name, "cells": len(c), "precision": share(m, a), "recall": share(a, m),
                    "agreement": share(pd.Series(True, index=c.index), m == a)})
    return pd.DataFrame(out).set_index("level")

    parts = []
    for stratum, group in pop.groupby("stratum"):
        n = min(TAKE[stratum], len(group))
        drawn = group.iloc[np.sort(rng.permutation(len(group))[:n])].copy()
        drawn["half"] = ["calibration" if i % 2 == 0 else "certification" for i in range(n)]
        drawn["stratum_N"], drawn["stratum_n"] = len(group), n
        drawn["weight"] = round(len(group) / n, 4)
        parts.append(drawn)
    sample = pd.concat(parts, ignore_index=True)
    return sample.iloc[rng.permutation(len(sample))].reset_index(drop=True)


def write_sheet(sample: pd.DataFrame, path: Path = SHEET) -> Path:
    """The worksheet: the measure's answer per decade and an empty verdict column per decade.
    Refuses to overwrite a sheet that holds any verdict."""
    path = Path(path)
    verdicts = [f"v_{d}" for d in DECADES] + ["v_note"]
    if path.exists():
        old = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
        if (old[[c for c in verdicts if c in old]] != "").any().any():
            raise FileExistsError(f"{path} already holds verdicts; it is not redrawn")
    sheet = sample.copy()
    for column in verdicts:
        sheet[column] = ""
    sheet.to_csv(path, index=False, encoding="utf-8-sig")
    return path


def _clean(value) -> str:
    return value if isinstance(value, str) else ""


def settlement_texts(sheet: pd.DataFrame, ties: pd.DataFrame) -> dict[str, list[dict]]:
    """For each settlement: its database places, each with the first report, the chronicle,
    the ensemble list and the ensemble pages of companies starting before 1914."""
    places = pd.read_csv(RAW / "places.csv", low_memory=False,
                         usecols=["geo_id", "name", "type", "first_record_raw", "chronicle_text"]).set_index("geo_id")
    ensembles = pd.read_csv(RAW / "ensembles.csv", usecols=["ensemble_id", "geo_id", "name", "span_raw", "year_from"])
    pages = pd.read_csv(RAW / "ensemble_pages.csv", usecols=["ensemble_id", "active_raw", "chronicle_text"]).set_index("ensemble_id")
    wanted = ties[ties["settlement_id"].isin(sheet["settlement_id"])]
    out: dict[str, list[dict]] = {}
    for sid, group in wanted.groupby("settlement_id"):
        entries = []
        for geo in group["geo_id"]:
            p = places.loc[geo]
            ens = ensembles[ensembles["geo_id"] == geo].sort_values("year_from")
            listed, own = [], []
            for e in ens.itertuples(index=False):
                listed.append({"name": _clean(e.name), "span": _clean(e.span_raw).strip("()")})
                if e.ensemble_id in pages.index and (pd.isna(e.year_from) or e.year_from < 1914):
                    pg = pages.loc[e.ensemble_id]
                    if _clean(pg["chronicle_text"]) or _clean(pg["active_raw"]):
                        own.append({"name": _clean(e.name), "span": _clean(e.span_raw).strip("()"),
                                    "active": _clean(pg["active_raw"]), "chronicle": _clean(pg["chronicle_text"])})
            entries.append({"name": _clean(p["name"]), "type": _clean(p["type"]),
                            "first": _clean(p["first_record_raw"]), "chronicle": _clean(p["chronicle_text"]),
                            "ensembles": listed, "pages": own})
        out[str(sid)] = entries
    return out


def build_page(out: Path, sheet_path: Path = SHEET) -> Path:
    sheet = pd.read_csv(sheet_path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    texts = settlement_texts(sheet, settlement_ties())
    sheet = sheet.sort_values("half", key=lambda h: h != "calibration", kind="stable").reset_index(drop=True)
    data = [{"i": i, "key": str(r.settlement_id), "half": r.half, "name": r.cs_name.title(), "okres": r.okres_1910_cs,
             "land": r.land, "measure": {str(d): getattr(r, f"measure_{d}") for d in DECADES},
             "places": texts.get(str(r.settlement_id), [])}
            for i, r in enumerate(sheet.itertuples(index=False))]
    payload = json.dumps(data, ensure_ascii=False).replace("</", "<\\/")
    template = TEMPLATE.read_text(encoding="utf-8")
    assert template.count(PLACEHOLDER) == 1, "template: one data placeholder"
    Path(out).write_text(template.replace(PLACEHOLDER, payload), encoding="utf-8")
    return Path(out)


def fill_sheet(documents: list[dict], path: Path = SHEET) -> dict:
    """Write the page's documents (``verdicts/<settlement_id>``) into the sheet."""
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    sheet["_key"] = sheet["settlement_id"].astype(str)
    index = {k: i for i, k in enumerate(sheet["_key"])}
    unknown = [d.get("settlement_id") for d in documents if str(d.get("settlement_id")) not in index]
    assert not unknown, f"verdicts for settlements not in the sheet: {unknown[:5]}"
    columns = [f"v_{d}" for d in DECADES] + ["v_note"]
    sheet[columns] = ""
    written = 0
    for d in documents:
        if not d.get("ruled"):
            continue
        answers = d.get("decades") or {}
        sheet.loc[index[str(d["settlement_id"])], columns] = [answers.get(str(x), "") for x in DECADES] + [d.get("note") or ""]
        written += 1
    sheet.drop(columns="_key").to_csv(path, index=False, encoding="utf-8-sig")
    return {"documents": len(documents), "ruled": written, "rows": len(sheet)}


def score(path: Path = SHEET, half: str = "calibration", measure: pd.DataFrame | None = None) -> pd.DataFrame:
    """The measure against the author per settlement-decade, weighted: for ``certain`` (the
    measure's yes against the author's yes) and ``possible`` (the measure's yes or possible
    against the author's yes or possible), precision, recall and agreement over the cells
    the author could tell.  ``measure`` (from :func:`decade_measure`) scores a later build;
    without it, the measure as drawn (the answers the page showed)."""
    if half == "certification" and not RULES_FROZEN:
        raise PermissionError("the certification half is sealed until RULES_FROZEN is True")
    sheet = pd.read_csv(path, encoding="utf-8-sig", dtype=str, keep_default_na=False)
    sheet = sheet[(sheet["half"] == half) & (sheet[[f"v_{d}" for d in DECADES]] != "").all(axis=1)]
    cells = []
    for r in sheet.itertuples(index=False):
        for d in DECADES:
            said = getattr(r, f"measure_{d}") if measure is None else (
                measure.at[r.settlement_id, d] if r.settlement_id in measure.index else "no")
            cells.append({"weight": float(r.weight), "measure": said, "author": getattr(r, f"v_{d}")})
    c = pd.DataFrame(cells, columns=["weight", "measure", "author"])
    c = c[c["author"] != "unsure"]
    out = []
    for name, levels in (("certain", {"yes"}), ("possible", {"yes", "possible"})):
        m, a, w = c["measure"].isin(levels), c["author"].isin(levels), c["weight"]
        share = lambda mask, hit: float((w[mask] * hit[mask]).sum() / w[mask].sum()) if w[mask].sum() else np.nan
        out.append({"level": name, "cells": len(c), "precision": share(m, a), "recall": share(a, m),
                    "agreement": share(pd.Series(True, index=c.index), m == a)})
    return pd.DataFrame(out).set_index("level")
