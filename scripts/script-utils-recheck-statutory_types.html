How it works

The plate vocabulary was made for the short entries of ordinary municipalities. The eleven statutory cities print prose instead, in Czech and in German, and much of it names things the plate vocabulary has no code for. This module keeps one row per distinct printed string (reference/places/statutory_lookup.csv, the leading count stripped), and the pipeline labels every prose row by looking its string up. Nothing else decides a label.

How a lookup row gets its values:

  • Pre-label. Ordered regex rules (statutory_type_rules.csv, first match wins) and the route's plate code give a first type; sponsor comes from the type (statutory_type_table.csv), read off the name only for the types whose sponsor varies (a museum may be municipal or Land); nationality from the printed language split, the words deutsch / německý, or a short hand list of named bodies, and is left for the author wherever the name says only český / böhmisch, which names the Land as often as the nation; orders lists every order the string names.
  • Author. A row the author has ruled on the review page (status = author) takes the ruling; --rulings builds the pre-labels afresh and lays every ruling read back over them, so the lookup is a function of the rules and the rulings.
  • Frozen (FROZEN, D-240). Notebook 17 §9 only reads the lookup (add): it writes register_statutory_types (one row per printed body) and the per-city n_type_*, n_person_*, n_sponsor_*, n_nation_* and order columns onto register_statutory_counts and register_okres.

What the string alone cannot say is read per printed row: the count (two brigade numbers in one line are two brigades), whether a row is a total (statutory_counts.totals, which also catches the unflagged cooperative totals), the printed language split, the language sections of a Land board (a count on the row before them, not an institution), and a seat held elsewhere for now (Archipresbyterat (dz. in Winař)), which is not counted for the city.

The Czech and German prints of Prague and Liberec are paired row by row (twins); a pair whose labels disagree is an error on at least one side, and Cohen's kappa over the pairs is the edition agreement.

def fold(quote: str) -> str:
    """Lower case, one space, ``a - b`` joined as ``a-b``, the pictogram placeholders dropped."""
    q = re.sub(r"\[symbol\]", " ", str(quote))
    q = re.sub(r"\s*-\s*", "-", q)
    return re.sub(r"\s+", " ", q).strip().lower()

def load_types(path: Path = TYPES, merges: Path | None = MERGES) -> pd.DataFrame:
    """The closed type table, with the author's approved merges (``statutory_type_merges.csv``, decision ``merge``)
    laid over it: each merged type gets a row of its own (its members' domain, aggregate and sponsor) and each member
    ``merged_into`` it. The member rows stay, so the rules, the lookup and the rulings keep naming the specific type;
    :func:`label` reports the merged one as ``type`` and the specific one as ``type_detail``."""
    types = pd.read_csv(path, dtype=str, keep_default_na=False).set_index("type")
    types["merged_into"] = ""
    if merges is None:
        return types
    approved = load_merges(merges)
    for new, g in approved[approved["decision"] == "merge"].groupby("merged_type"):
        olds = [t for t in g["old_type"] if t in types.index]
        if not olds:
            continue
        if new not in types.index:
            first = types.loc[olds[0]]
            types.loc[new] = {**first.to_dict(), "merged_into": "",
                              "sponsor_from_name": "yes" if (types.loc[olds, "sponsor_from_name"] == "yes").any() else "no"}
        for old in olds:
            if old != new:
                types.loc[old, "merged_into"] = new
    return types

def prelabel(q: str, edition: str, category: str, code: str, rules: pd.DataFrame, plate: dict) -> dict:
    """A first label for one folded string: ``{type, rule, plate_english, branch}`` (a division-and-brigade command
    gives ``type = "division command + brigade command"``)."""
    p_eng, _ = plate.get(code, ("", ""))
    if SUBUNIT.search(q):                              # reading rule: a sub-unit line
        if SECTION.search(q) and not code:
            return {"type": "language section", "rule": "R3", "plate_english": "", "branch": False}
        typ = SUBUNIT_TYPE.get(p_eng, ALIASES.get(p_eng, p_eng)) or "branch of the body before"
        return {"type": typ, "rule": "R3", "plate_english": typ if p_eng else "", "branch": True}
    military = category == "military" or bool(MILITARY.search(q))
    if p_eng and p_eng not in REFINE:                  # a plate code stands unless a rule names it wrong
        rules = rules[rules["overrides_plate"] == "yes"]
    elif p_eng:
        rules = rules[rules["domain"].isin(REFINE[p_eng])]
    branch = is_branch(q, edition)
    q_type = re.sub(r"^(filiale|filiálka|zweiganstalt|nebenstelle|zweigniederlassung|expositur) (der|des) ", "", q) \
        if branch else q
    hits = _match(rules, q_type, military)
    if hits:
        h = hits[0]
        pe = h["plate_english"]
        if h["rule_id"] == "C08" and not re.search(r"městsk|städt|stadt", q):
            pe = ""                                    # only a municipal archive is the plate's town archive
        if h["rule_id"] in ("M04", "M13", "M14") and LANDWEHR.search(q) and pe:
            pe = "Landwehr " + pe.replace("military ", "")
        return {"type": SPLIT.join(x["type"] for x in hits), "rule": "+".join(x["rule_id"] for x in hits),
                "plate_english": pe, "branch": branch and h["domain"] == "finance"}
    if p_eng:
        typ = ALIASES.get(p_eng, p_eng)
        if typ.startswith("Evangelical parish") and re.search(r"a\.\s*(und|a)\s*h\.", q):
            typ = "Evangelical parish, A.C. and H.C."
        return {"type": typ, "rule": "plate", "plate_english": p_eng, "branch": False}
    return {"type": "UNLABELLED", "rule": "", "plate_english": "", "branch": False}

def nationality_of(q: str, typ: str, sponsor: str) -> tuple[str, str]:
    """``(nationality, basis)`` from the name alone; a printed language split overrides it per row (:func:`label`).
    *český / böhmisch* names the Land as often as the nation, so a name that says only that is left for the author."""
    if NUMBERED_SPLIT.search(q):                       # counts per language: bodies of each nation, read per row
        return "split", "printed language split"
    for pattern, value, sure in NATIONALITY_HAND:
        if re.search(pattern, q):
            return value, "named body" if sure else "suggested: rule it"
    if sponsor == "state":                             # a state body names a Land or a country, never a nation
        return "none", "state body, no language printed"
    if BOTH.search(q):
        return "both", "name prints both"
    if NATION_DE.search(q) and not TEUTONIC.search(q):
        return "german", "name says deutsch / německý"
    if TERRITORIAL.search(q):
        return "territorial", "name says the Land"
    if AMBIGUOUS.search(q):
        return "unknown", "name says český / böhmisch: Land or nation? rule it"
    if sponsor in ("Land", "district"):
        return "none", "public body, no language printed"
    return "not stated", ""

def build_lookup(statutory: pd.DataFrame, rules: pd.DataFrame, plate: dict, types: pd.DataFrame,
                 existing: pd.DataFrame | None = None) -> pd.DataFrame:
    """One row per distinct string. Author-ruled rows are kept as they are; the others are pre-labelled afresh."""
    keep = {} if existing is None or existing.empty else \
        {r["key"]: r for r in existing.to_dict("records") if r.get("status") == "author"}
    groups: dict[str, list] = {}
    for _, s in statutory.iterrows():
        groups.setdefault(key_of(s["quote"]), []).append(s)
    out = []
    for key, rows in groups.items():
        editions = ",".join(sorted({r["edition"] for r in rows}))
        if key in keep:
            out.append({**keep[key], "n_rows": len(rows), "editions": editions})
            continue
        labels = [prelabel(fold(r["quote"]), r["edition"], r["category"], str(r.get("code") or ""), rules, plate)
                  for r in rows]
        best = Counter(lab["type"] for lab in labels).most_common(1)[0][0]
        lab = next(x for x in labels if x["type"] == best)
        q = key
        orders = orders_of(q)
        p_eng = plate.get(str(rows[0].get("code") or ""), ("", ""))[0]
        if not orders and best == "religious house" and p_eng in HOUSE_SEX:
            orders = [("order not named", HOUSE_SEX[p_eng])]
        sponsor = sponsor_of(best, q, types)
        nat, basis = nationality_of(q, best, sponsor)
        # "2 Universitäten (deutsch u. böhmisch)": both nations named on a line of several bodies is one body of each,
        # as the rows count it (D-251's check found the string label "none" or "both" on eight of these)
        if BOTH.search(q) and min(_count(r.get("count")) for r in rows) >= 2:
            nat, basis = "split", "both nations named on a line of several bodies"
        out.append({"key": key, "example": rows[0]["quote"], "n_rows": len(rows), "editions": editions,
                    "type": best, "rule": lab["rule"], "plate_english": lab["plate_english"],
                    "branch": lab["branch"], "sponsor": sponsor, "nationality": nat, "nationality_basis": basis,
                    "orders": "; ".join(o for o, _ in orders), "order_sexes": "; ".join(s for _, s in orders),
                    "house_kind": next((k for k, p in HOUSE if re.search(p, q)), "convent or monastery")
                    if best == "religious house" else "",
                    "legal_form": ("joint-stock" if re.search(r"akciov|aktien", q) else
                                   "limited liability" if re.search(r"obm.*ruč|beschr", q) else ""),
                    "seat_elsewhere": bool(SEAT_ELSEWHERE.search(q)), "status": "prelabel", "note": ""})
    for key, r in keep.items():                        # an author row whose string has left the readings
        if key not in groups:
            out.append({**r, "n_rows": 0})
    return pd.DataFrame(out, columns=LOOKUP_FIELDS).sort_values("key").reset_index(drop=True)

def language_split(languages: str) -> dict[str, int]:
    """``"czech 55, german 9"`` → ``{"czech": 55, "german": 9}`` (stage 2's printed split; ``{}`` where none)."""
    out: dict[str, int] = {}
    for part in str(languages or "").split(","):
        if part.strip():
            lang, n = part.strip().rsplit(" ", 1)
            out[lang] = out.get(lang, 0) + int(n)
    return out

def kappa(x: pd.Series, y: pd.Series) -> float:
    """Cohen's kappa of two labelings of the same items."""
    n = len(x)
    if n == 0:
        return float("nan")
    po = (x.values == y.values).mean()
    px, py = x.value_counts(normalize=True), y.value_counts(normalize=True)
    pe = sum(px.get(k, 0) * py.get(k, 0) for k in set(px.index) | set(py.index))
    return float((po - pe) / (1 - pe)) if pe < 1 else 1.0

    """The institutions by nation, a split line broken into its nations (the remainder ``not stated``)."""
    whole = inst[inst["nationality"] != "split"][["edition", "city", "nationality", "count"]]

def city_columns(lab: pd.DataFrame, types: pd.DataFrame | None = None) -> pd.DataFrame:
    """Per city (``edition``, ``city``): ``n_type_<aggregate>`` (the institutions, each printed count), every aggregate
    of the type table present so a 0 is a 0; ``n_person_<type>`` (Q1); ``n_sponsor_<sponsor>``, ``n_nation_<value>``
    over the same institutions; ``n_order_house_{male,female}`` and ``orders_present`` (the orders with a house)."""
    types = load_types() if types is None else types
    inst = lab[lab["institution"]]
    persons = lab[lab["counted"] & lab["is_person"]]
    owner = ["edition", "city"]
    cities = lab[owner].drop_duplicates().set_index(owner).index

    def block(frame, col, prefix, universe):
        wide = frame.pivot_table(index=owner, columns=col, values="count", aggfunc="sum", fill_value=0)
        wide = wide.reindex(index=cities, columns=sorted(set(universe) | set(wide.columns)), fill_value=0)
        return wide.rename(columns=lambda c: prefix + (slug(c) or "not_given"))

    aggregates = sorted(set(types.loc[(types["person"] != "yes") & ~types["domain"].isin(("not counted", LEGAL_FORM)),
                                      "aggregate"]))
    person_types = sorted(types.index[types["person"] == "yes"])
    legal_types = sorted(types.index[types["domain"] == LEGAL_FORM])
    legal = lab[lab["counted"] & lab["domain"].eq(LEGAL_FORM)]
    out = pd.concat([block(inst, "aggregate", "n_type_", aggregates),
                     block(persons, "type", "n_person_", person_types),
                     block(legal, "type", "n_legal_form_", legal_types),
                     block(inst, "sponsor", "n_sponsor_", []),
                     block(nation_rows(inst), "nationality", "n_nation_", NATIONS)], axis=1)
    houses = inst[inst["type"].eq("religious house")]
    for sex in ("male", "female"):
        hit = houses[houses["order_sexes"].str.contains(rf"\b{sex}\b", na=False)]     # "female" holds "male"
        out[f"n_order_house_{sex}"] = hit.groupby(owner)["count"].sum().reindex(cities).fillna(0)
    out["orders_present"] = (houses.groupby(owner)["orders"]
                             .agg(lambda s: "; ".join(sorted({o.strip() for v in s for o in v.split(";") if o.strip()})))
                             .reindex(cities).fillna(""))
    ints = [c for c in out.columns if c != "orders_present"]
    return out.astype({c: "Int64" for c in ints}).reset_index()

def apply_rulings(lookup: pd.DataFrame, folder: Path = READBACK) -> tuple[pd.DataFrame, dict]:
    """The page's rulings (read back as JSON, one file per document, under ``folder/rulings``, ``folder/types``)
    into the lookup: a ruled string takes the fields the author set and becomes ``status = author``; every string of
    an accepted type keeps its pre-label as the author's. Returns the lookup and a count of what changed."""
    look = lookup.copy().set_index("key")
    report = {"strings ruled": 0, "fields set": 0, "types accepted": 0, "strings by accepted type": 0}
    for doc in sorted((folder / "rulings").glob("*.json")):
        d = json.loads(doc.read_text(encoding="utf-8"))
        d = d.get("data", d)
        key = d.get("key")
        if key not in look.index:
            continue
        for f in ("type", "sponsor", "nationality", "note"):
            if d.get(f):
                look.loc[key, f] = d[f]
                report["fields set"] += 1
        if d.get("nationality"):
            look.loc[key, "nationality_basis"] = "author"
        elif d.get("sponsor") or d.get("type"):
            # The pre-label's nationality was read with the pre-label's sponsor ("none" for a Land body); a ruled
            # sponsor or type reads it again, so a ruling never leaves a nationality its own reading would not give.
            first = look.loc[key, "type"].split(SPLIT)[0]
            look.loc[key, ["nationality", "nationality_basis"]] = list(nationality_of(key, first, look.loc[key, "sponsor"]))
        look.loc[key, "status"] = "author"
        report["strings ruled"] += 1
    for doc in sorted((folder / "types").glob("*.json")):
        d = json.loads(doc.read_text(encoding="utf-8"))
        d = d.get("data", d)
        if not d.get("accepted"):
            continue
        report["types accepted"] += 1
        hit = look["type"].str.split(SPLIT).map(lambda parts: d.get("type") in parts) & (look["status"] != "author")
        look.loc[hit, "status"] = "author"
        report["strings by accepted type"] += int(hit.sum())
    return look.reset_index()[LOOKUP_FIELDS], report

"""The statutory cities' prose institutions as a string lookup table: type, sponsor, nationality, orders (D-240).

The plate vocabulary was made for the short entries of ordinary municipalities. The eleven statutory cities print prose
instead, in Czech and in German, and much of it names things the plate vocabulary has no code for. This module keeps
**one row per distinct printed string** (``reference/places/statutory_lookup.csv``, the leading count stripped), and the
pipeline labels every prose row by looking its string up. Nothing else decides a label.

How a lookup row gets its values:

* **Pre-label.** Ordered regex rules (``statutory_type_rules.csv``, first match wins) and the route's plate code give a
  first type; ``sponsor`` comes from the type (``statutory_type_table.csv``), read off the name only for the types
  whose sponsor varies (a museum may be municipal or Land); ``nationality`` from the printed language split, the
  words *deutsch / německý*, or a short hand list of named bodies, and is left for the author wherever the name says
  only *český / böhmisch*, which names the Land as often as the nation; ``orders`` lists every order the string names.
* **Author.** A row the author has ruled on the review page (``status = author``) takes the ruling; ``--rulings``
  builds the pre-labels afresh and lays every ruling read back over them, so the lookup is a function of the rules
  and the rulings.
* **Frozen** (:data:`FROZEN`, D-240). Notebook 17 §9 only reads the lookup (:func:`add`): it writes
  ``register_statutory_types`` (one row per printed body) and the per-city ``n_type_*``, ``n_person_*``,
  ``n_sponsor_*``, ``n_nation_*`` and order columns onto ``register_statutory_counts`` and ``register_okres``.

What the string alone cannot say is read per printed row: the count (two brigade numbers in one line are two brigades),
whether a row is a total (``statutory_counts.totals``, which also catches the unflagged cooperative totals), the
printed language split, the language sections of a Land board (a count on the row before them, not an institution),
and a seat held elsewhere for now (``Archipresbyterat (dz. in Winař)``), which is not counted for the city.

The Czech and German prints of Prague and Liberec are paired row by row (``twins``); a pair whose labels disagree is
an error on at least one side, and Cohen's kappa over the pairs is the edition agreement.

    python -m utils.recheck.statutory_types        # refresh the pre-labels, label, pair, write the review page
"""

def fold(quote: str) -> str:
    """Lower case, one space, ``a - b`` joined as ``a-b``, the pictogram placeholders dropped."""
    q = re.sub(r"\[symbol\]", " ", str(quote))
    q = re.sub(r"\s*-\s*", "-", q)
    return re.sub(r"\s+", " ", q).strip().lower()


def key_of(quote: str) -> str:
    """The lookup key: the folded string without its leading count ("5 Konsulate" and "2 Konsulate" are one)."""
    return re.sub(r"^\d+\s+", "", fold(quote))


def load_rules(path: Path = RULES) -> pd.DataFrame:
    rules = pd.read_csv(path, dtype=str, keep_default_na=False)
    rules["regex"] = [re.compile(p) for p in rules["pattern"]]
    return rules


def load_types(path: Path = TYPES, merges: Path | None = MERGES) -> pd.DataFrame:
    """The closed type table, with the author's approved merges (``statutory_type_merges.csv``, decision ``merge``)
    laid over it: each merged type gets a row of its own (its members' domain, aggregate and sponsor) and each member
    ``merged_into`` it. The member rows stay, so the rules, the lookup and the rulings keep naming the specific type;
    :func:`label` reports the merged one as ``type`` and the specific one as ``type_detail``."""
    types = pd.read_csv(path, dtype=str, keep_default_na=False).set_index("type")
    types["merged_into"] = ""
    if merges is None:
        return types
    approved = load_merges(merges)
    for new, g in approved[approved["decision"] == "merge"].groupby("merged_type"):
        olds = [t for t in g["old_type"] if t in types.index]
        if not olds:
            continue
        if new not in types.index:
            first = types.loc[olds[0]]
            types.loc[new] = {**first.to_dict(), "merged_into": "",
                              "sponsor_from_name": "yes" if (types.loc[olds, "sponsor_from_name"] == "yes").any() else "no"}
        for old in olds:
            if old != new:
                types.loc[old, "merged_into"] = new
    return types


def canonical(typ: str, types: pd.DataFrame) -> str:
    """The type a specific type is counted under: its merge target, or itself."""
    return (types.at[typ, "merged_into"] or typ) if typ in types.index else typ

def _match(rules: pd.DataFrame, q: str, military: bool) -> list[pd.Series]:
    """The rules that label one folded quote: the first match, or a division and a brigade command together."""
    hits = []
    for _, r in rules.iterrows():
        if r["domain"] == "military" and not military:
            continue
        if r["regex"].search(q):
            hits.append(r)
            if r["rule_id"] != "M05":            # only a division command may be followed by a brigade command
                break
    if len(hits) == 2 and [h["rule_id"] for h in hits] != ["M05", "M06"]:
        hits = hits[:1]
    return hits


def is_branch(q: str, edition: str) -> bool:
    if re.search(r"seelsorge|^der [a-zäöü]+innen$", q):
        return False
    if BRANCH.search(q):
        return True
    return edition == "cz" and not re.match(r"^\d", q) and bool(BRANCH_CS.search(q))


def prelabel(q: str, edition: str, category: str, code: str, rules: pd.DataFrame, plate: dict) -> dict:
    """A first label for one folded string: ``{type, rule, plate_english, branch}`` (a division-and-brigade command
    gives ``type = "division command + brigade command"``)."""
    p_eng, _ = plate.get(code, ("", ""))
    if SUBUNIT.search(q):                              # reading rule: a sub-unit line
        if SECTION.search(q) and not code:
            return {"type": "language section", "rule": "R3", "plate_english": "", "branch": False}
        typ = SUBUNIT_TYPE.get(p_eng, ALIASES.get(p_eng, p_eng)) or "branch of the body before"
        return {"type": typ, "rule": "R3", "plate_english": typ if p_eng else "", "branch": True}
    military = category == "military" or bool(MILITARY.search(q))
    if p_eng and p_eng not in REFINE:                  # a plate code stands unless a rule names it wrong
        rules = rules[rules["overrides_plate"] == "yes"]
    elif p_eng:
        rules = rules[rules["domain"].isin(REFINE[p_eng])]
    branch = is_branch(q, edition)
    q_type = re.sub(r"^(filiale|filiálka|zweiganstalt|nebenstelle|zweigniederlassung|expositur) (der|des) ", "", q) \
        if branch else q
    hits = _match(rules, q_type, military)
    if hits:
        h = hits[0]
        pe = h["plate_english"]
        if h["rule_id"] == "C08" and not re.search(r"městsk|städt|stadt", q):
            pe = ""                                    # only a municipal archive is the plate's town archive
        if h["rule_id"] in ("M04", "M13", "M14") and LANDWEHR.search(q) and pe:
            pe = "Landwehr " + pe.replace("military ", "")
        return {"type": SPLIT.join(x["type"] for x in hits), "rule": "+".join(x["rule_id"] for x in hits),
                "plate_english": pe, "branch": branch and h["domain"] == "finance"}
    if p_eng:
        typ = ALIASES.get(p_eng, p_eng)
        if typ.startswith("Evangelical parish") and re.search(r"a\.\s*(und|a)\s*h\.", q):
            typ = "Evangelical parish, A.C. and H.C."
        return {"type": typ, "rule": "plate", "plate_english": p_eng, "branch": False}
    return {"type": "UNLABELLED", "rule": "", "plate_english": "", "branch": False}


def orders_of(q: str) -> list[tuple[str, str]]:
    """Every religious order a string names, in printed order: ``[(order, sex), …]``."""
    found = []
    for order, sex, pattern in ORDERS:
        m = re.search(pattern, q)
        if m:
            found.append((m.start(), order, sex))
    return [(o, s) for _, o, s in sorted(found)]


def sponsor_of(typ: str, q: str, types: pd.DataFrame) -> str:
    """Who ran it: the type's sponsor, or for a type whose sponsor varies, the first cue the name prints."""
    first = typ.split(SPLIT)[0]
    if first not in types.index:
        return ""
    t = types.loc[first]
    if t["sponsor_from_name"] == "yes":
        for value, pattern in SPONSOR_CUES:
            if re.search(pattern, q):
                return value
    return t["sponsor"]


def nationality_of(q: str, typ: str, sponsor: str) -> tuple[str, str]:
    """``(nationality, basis)`` from the name alone; a printed language split overrides it per row (:func:`label`).
    *český / böhmisch* names the Land as often as the nation, so a name that says only that is left for the author."""
    if NUMBERED_SPLIT.search(q):                       # counts per language: bodies of each nation, read per row
        return "split", "printed language split"
    for pattern, value, sure in NATIONALITY_HAND:
        if re.search(pattern, q):
            return value, "named body" if sure else "suggested: rule it"
    if sponsor == "state":                             # a state body names a Land or a country, never a nation
        return "none", "state body, no language printed"
    if BOTH.search(q):
        return "both", "name prints both"
    if NATION_DE.search(q) and not TEUTONIC.search(q):
        return "german", "name says deutsch / německý"
    if TERRITORIAL.search(q):
        return "territorial", "name says the Land"
    if AMBIGUOUS.search(q):
        return "unknown", "name says český / böhmisch: Land or nation? rule it"
    if sponsor in ("Land", "district"):
        return "none", "public body, no language printed"
    return "not stated", ""


def build_lookup(statutory: pd.DataFrame, rules: pd.DataFrame, plate: dict, types: pd.DataFrame,
                 existing: pd.DataFrame | None = None) -> pd.DataFrame:
    """One row per distinct string. Author-ruled rows are kept as they are; the others are pre-labelled afresh."""
    keep = {} if existing is None or existing.empty else \
        {r["key"]: r for r in existing.to_dict("records") if r.get("status") == "author"}
    groups: dict[str, list] = {}
    for _, s in statutory.iterrows():
        groups.setdefault(key_of(s["quote"]), []).append(s)
    out = []
    for key, rows in groups.items():
        editions = ",".join(sorted({r["edition"] for r in rows}))
        if key in keep:
            out.append({**keep[key], "n_rows": len(rows), "editions": editions})
            continue
        labels = [prelabel(fold(r["quote"]), r["edition"], r["category"], str(r.get("code") or ""), rules, plate)
                  for r in rows]
        best = Counter(lab["type"] for lab in labels).most_common(1)[0][0]
        lab = next(x for x in labels if x["type"] == best)
        q = key
        orders = orders_of(q)
        p_eng = plate.get(str(rows[0].get("code") or ""), ("", ""))[0]
        if not orders and best == "religious house" and p_eng in HOUSE_SEX:
            orders = [("order not named", HOUSE_SEX[p_eng])]
        sponsor = sponsor_of(best, q, types)
        nat, basis = nationality_of(q, best, sponsor)
        # "2 Universitäten (deutsch u. böhmisch)": both nations named on a line of several bodies is one body of each,
        # as the rows count it (D-251's check found the string label "none" or "both" on eight of these)
        if BOTH.search(q) and min(_count(r.get("count")) for r in rows) >= 2:
            nat, basis = "split", "both nations named on a line of several bodies"
        out.append({"key": key, "example": rows[0]["quote"], "n_rows": len(rows), "editions": editions,
                    "type": best, "rule": lab["rule"], "plate_english": lab["plate_english"],
                    "branch": lab["branch"], "sponsor": sponsor, "nationality": nat, "nationality_basis": basis,
                    "orders": "; ".join(o for o, _ in orders), "order_sexes": "; ".join(s for _, s in orders),
                    "house_kind": next((k for k, p in HOUSE if re.search(p, q)), "convent or monastery")
                    if best == "religious house" else "",
                    "legal_form": ("joint-stock" if re.search(r"akciov|aktien", q) else
                                   "limited liability" if re.search(r"obm.*ruč|beschr", q) else ""),
                    "seat_elsewhere": bool(SEAT_ELSEWHERE.search(q)), "status": "prelabel", "note": ""})
    for key, r in keep.items():                        # an author row whose string has left the readings
        if key not in groups:
            out.append({**r, "n_rows": 0})
    return pd.DataFrame(out, columns=LOOKUP_FIELDS).sort_values("key").reset_index(drop=True)

def _numbers_for(q: str, echelon: str, split: bool = False) -> str:
    """The unit numbers printed for one echelon: of a line naming a division and a brigade command, each command's
    own number (the ordinal before its word, up to two words between); otherwise every number after the count."""
    pattern = {"division command": r"(\d+)\.\s*(?:\S+\s+){0,2}?\S*divis",
               "brigade command": r"(\d+)\.\s*(?:\S+\s+){0,2}?\S*brig"}.get(echelon)
    if split and pattern:
        return ",".join(re.findall(pattern, q))
    rest = re.sub(r"^\d+\s", "", q)
    return ",".join(re.findall(r"\b([ivx]+|\d+)\b", rest))


def language_split(languages: str) -> dict[str, int]:
    """``"czech 55, german 9"`` → ``{"czech": 55, "german": 9}`` (stage 2's printed split; ``{}`` where none)."""
    out: dict[str, int] = {}
    for part in str(languages or "").split(","):
        if part.strip():
            lang, n = part.strip().rsplit(" ", 1)
            out[lang] = out.get(lang, 0) + int(n)
    return out


def _as_bool(v) -> bool:
    return v is True or str(v).lower() == "true"


def label(statutory: pd.DataFrame, lookup: pd.DataFrame, types: pd.DataFrame,
          totals: pd.Series | None = None) -> pd.DataFrame:
    """Every printed row labelled from its string's lookup row, with the per-row reading facts."""
    look = lookup.set_index("key")
    totals = statutory["is_total"].astype(bool) if totals is None else totals
    out = []
    for idx, s in statutory.iterrows():
        q, key = fold(s["quote"]), key_of(s["quote"])
        entry = look.loc[key] if key in look.index else None
        typ = entry["type"] if entry is not None else "UNLABELLED"
        parts = typ.split(SPLIT)
        for detail in parts:
            part = canonical(detail, types)                  # counted under the merged type; the specific one kept
            t = types.loc[part] if part in types.index else None
            count = s["count"]
            numbers = ""
            domain = t["domain"] if t is not None else "UNLABELLED"
            if domain == "military":
                numbers = _numbers_for(q, detail, len(parts) > 1)
                if len(parts) > 1:
                    count = max(1, len(numbers.split(","))) if numbers else 1
                elif str(count) in ("1", "1.0") and re.search(r"\d+\.\s*(a|und|u\.)\s*\d+\.", q) and numbers:
                    count = len(numbers.split(","))
            langs = str(s.get("languages") or "")
            nat, basis = (entry["nationality"], entry["nationality_basis"]) if entry is not None else ("", "")
            split = language_split(langs)
            if split:
                # A printed split counts bodies of each nation (64 schools: 55 Czech, 9 German), so a line of more than
                # one nation is "split" and its numbers go to each; "both" stays for one body that is both (the split
                # sums past the count: a German and a Czech department of one academy).
                total, n = sum(split.values()), _count(count)
                nat = ("both" if len(split) > 1 and total > n else
                       next(iter(split)) if len(split) == 1 and total >= n else "split")
                basis = "printed language split"
            out.append({
                "edition": s["edition"], "city": s["city"], "row": int(s["row"]), "quote": s["quote"], "key": key,
                "count": count, "languages": langs, "reader_category": s["category"],
                "plate_code": str(s.get("code") or ""), "is_total": bool(totals.loc[idx]),
                "type": part, "type_detail": detail, "domain": domain,
                "aggregate": t["aggregate"] if t is not None else "",
                "is_person": t is not None and t["person"] == "yes",
                "sponsor": entry["sponsor"] if entry is not None else "",
                "nationality": nat, "nationality_basis": basis,
                "n_czech": split.get("czech", 0) if nat == "split" else pd.NA,
                "n_german": split.get("german", 0) if nat == "split" else pd.NA,
                "plate_english": entry["plate_english"] if entry is not None else "",
                "branch": _as_bool(entry["branch"]) if entry is not None else False,
                "orders": entry["orders"] if entry is not None else "",
                "order_sexes": entry["order_sexes"] if entry is not None else "",
                "house_kind": entry["house_kind"] if entry is not None else "",
                "legal_form": entry["legal_form"] if entry is not None else "",
                "army": ("Landwehr" if LANDWEHR.search(q) else "common army") if domain == "military" else "",
                "arm": next((a for a, p in ARM if re.search(p, q)), "") if domain == "military" else "",
                "unit_numbers": numbers,
                "seat_elsewhere": _as_bool(entry["seat_elsewhere"]) if entry is not None else False,
                "temporary_here": bool(TEMPORARY_HERE.search(q)),
                "housed": bool(re.match(r"^mit\s", q)) or bool(HOUSED.fullmatch(q)),
                "status": entry["status"] if entry is not None else "", "national_sections": 0})
    lab = pd.DataFrame(out)
    # A language-section line is a count on the row before it (stage order keeps them together), not a body.
    for i in lab.index[lab["type"] == "language section"]:
        e, c, r = lab.loc[i, ["edition", "city", "row"]]
        parent = lab.index[(lab["edition"] == e) & (lab["city"] == c) & (lab["row"] == r - 1)]
        if len(parent):
            lab.loc[parent, "national_sections"] = int(float(lab.loc[i, "count"]))
    lab["counted"] = ~lab["is_total"] & ~lab["seat_elsewhere"] & (lab["domain"] != "not counted")
    return lab

def _count(v) -> float:
    try:
        return float(v)
    except (TypeError, ValueError):
        return 1.0


def _pair_cost(a: pd.Series, b: pd.Series, types: pd.DataFrame) -> float:
    # The type is part of the cost: without a translation nothing else pairs a Czech line with its German twin (a
    # label-free pairing matched a weather station with a homeless shelter). So agreement over the pairs is reported
    # only for what the pairing does not use (nationality, branch, orders); the type is checked by the per-type
    # counts of the two prints and by the rows left without a twin (:func:`agreement`).
    cost = 0.0
    cost -= 3 if _count(a["count"]) == _count(b["count"]) else 0
    cost -= 3 if a["type"] == b["type"] else 0
    cost -= 1 if a["aggregate"] == b["aggregate"] else 0
    cost -= 1 if a["reader_category"] == b["reader_category"] else 0
    na, nb = set(re.findall(r"\d+", a["quote"])[1:]), set(re.findall(r"\d+", b["quote"])[1:])
    cost += (-2 if na == nb else 2) if (na or nb) else 0
    pa = {w for w in PROPER.findall(a["quote"]) if w not in ("Prag", "Praha")}
    pb = {w for w in PROPER.findall(b["quote"]) if w not in ("Prag", "Praha")}
    cost -= 2 * len(pa & pb)
    # A nation or Land word is printed content, and the one word every translation keeps: a line that names it and
    # one that does not are rarely the same body. Without this, two same-type, same-count lines in one city (the
    # Czech academy and the German learned society in Prague) paired across. It overlaps the nationality pre-label's
    # cue, so :func:`agreement` also reports nationality on the pairs that print no such word.
    for cz, de in NATION_WORDS:
        ha, hb = bool(cz.search(a["key"])), bool(de.search(b["key"]))
        cost += -1.5 if ha and hb else (1.5 if ha != hb else 0)
    cost -= 2 * sum(1 for cz, de in GLOSS if cz.search(a["key"]) and de.search(b["key"]))
    return cost


def twins(lab: pd.DataFrame, types: pd.DataFrame, threshold: float = -4.0) -> pd.DataFrame:
    """Row pairs of the same institution in the Czech and the German print: an assignment on a cost built from what
    the print says, never from the labels (the count, the reader's category, the numbers and proper names printed,
    the language split, the place in the reading order); a pair must reach ``threshold`` to count."""
    from scipy.optimize import linear_sum_assignment

    out = []
    for (e1, c1), (e2, c2) in EDITION_PAIRS:
        a = lab[(lab["edition"] == e1) & (lab["city"] == c1)].reset_index(drop=True)
        b = lab[(lab["edition"] == e2) & (lab["city"] == c2)].reset_index(drop=True)
        if a.empty or b.empty:
            continue
        rank_a, rank_b = a["row"].rank(pct=True), b["row"].rank(pct=True)
        cost = [[_pair_cost(a.loc[i], b.loc[j], types) + 4 * abs(rank_a[i] - rank_b[j]) for j in b.index]
                for i in a.index]
        for i, j in zip(*linear_sum_assignment(cost)):
            if cost[i][j] <= threshold:
                out.append({"pair": f"{c1} / {c2}", "cost": round(cost[i][j], 2),
                            **{f"{k}_cz": a.loc[i, k] for k in ("quote", "key", "type", "sponsor", "nationality",
                                                               "branch", "orders", "count")},
                            **{f"{k}_de": b.loc[j, k] for k in ("quote", "key", "type", "sponsor", "nationality",
                                                               "branch", "orders", "count")}})
    return pd.DataFrame(out)


def kappa(x: pd.Series, y: pd.Series) -> float:
    """Cohen's kappa of two labelings of the same items."""
    n = len(x)
    if n == 0:
        return float("nan")
    po = (x.values == y.values).mean()
    px, py = x.value_counts(normalize=True), y.value_counts(normalize=True)
    pe = sum(px.get(k, 0) * py.get(k, 0) for k in set(px.index) | set(py.index))
    return float((po - pe) / (1 - pe)) if pe < 1 else 1.0


def agreement(pairs: pd.DataFrame, lab: pd.DataFrame) -> dict:
    """Per attribute: share of twin pairs that agree, and kappa; plus how many rows found a twin."""
    out = {"pairs": int(len(pairs))}
    for (e1, c1), (e2, c2) in EDITION_PAIRS:
        out[f"rows {c1}"] = int(((lab["edition"] == e1) & (lab["city"] == c1)).sum())
        out[f"rows {c2}"] = int(((lab["edition"] == e2) & (lab["city"] == c2)).sum())
    for attr in ("nationality", "branch", "orders"):
        if pairs.empty:
            break
        x, y = pairs[f"{attr}_cz"].astype(str), pairs[f"{attr}_de"].astype(str)
        out[attr] = {"agree": round(float((x == y).mean()), 3), "kappa": round(kappa(x, y), 3)}
    if not pairs.empty:
        # The pairing uses the printed nation words; nationality on the pairs that print none is the clean figure.
        clean = ~pairs.apply(lambda r: any(cz.search(r["key_cz"]) or de.search(r["key_de"])
                                           for cz, de in NATION_WORDS), axis=1)
        x, y = pairs.loc[clean, "nationality_cz"].astype(str), pairs.loc[clean, "nationality_de"].astype(str)
        out["nationality, pairs printing no nation word"] = {"pairs": int(clean.sum()),
                                                             "agree": round(float((x == y).mean()), 3),
                                                             "kappa": round(kappa(x, y), 3)}
    counts = type_counts(lab)
    out["type counts equal"] = {p: f"{int((g.cz == g.de).sum())} of {len(g)}" for p, g in counts.groupby("pair")}
    paired = set(pairs["key_cz"]) | set(pairs["key_de"]) if not pairs.empty else set()
    out["rows without a twin"] = int(sum(1 for (e, c), k in zip(zip(lab["edition"], lab["city"]), lab["key"])
                                         if (e, c) in {x for pr in EDITION_PAIRS for x in pr} and k not in paired))
    return out


def type_counts(lab: pd.DataFrame) -> pd.DataFrame:
    """Per city pair and type: the printed count in the Czech and in the German print (rows counted, not totals)."""
    rows = []
    for (e1, c1), (e2, c2) in EDITION_PAIRS:
        sums = {}
        for (e, c), side in (((e1, c1), "cz"), ((e2, c2), "de")):
            sub = lab[(lab["edition"] == e) & (lab["city"] == c) & lab["counted"]]
            for typ, g in sub.groupby("type"):
                sums.setdefault(typ, {"cz": 0.0, "de": 0.0})[side] = sum(_count(v) for v in g["count"])
        rows += [{"pair": f"{c1} / {c2}", "type": t, **s} for t, s in sorted(sums.items())]
    return pd.DataFrame(rows)

#: Rows that look like the members of a Land body printed under its seat city (the Silesian agricultural society's
#: "111 Ortsvereinen, Klubs und Kasinos im Lande", the Moravian societies' "107 landwirtschaftlichen Bezirksvereinen"):
#: suspects for the author, never a label. Ruled "Land-wide member tally", a row is not counted for the city.
LAND_MEMBERS = re.compile(r"im lande\b|v zemi\b|bezirksvereinen\b|okresních (?:hospodářských )?spolků|kasinos und lesevereinen",
                          re.IGNORECASE)


def issues(lookup: pd.DataFrame, lab: pd.DataFrame, hidden: set) -> dict:
    """What is still open after the rulings: Land-wide member tallies counted for a city, the strings that weigh most in
    the counts and stand on a pre-label (the rule-next queue), and the strings still ``unknown`` on nationality."""
    counted = lab[lab["counted"]].assign(n=pd.to_numeric(lab["count"], errors="coerce").fillna(1))
    weight = counted.groupby("key")["n"].sum()
    status = lookup.set_index("key")["status"]
    land = counted[counted["quote"].str.contains(LAND_MEMBERS) & (counted["n"] > 2) & ~counted["key"].isin(hidden)]
    land = [{"key": r.key, "city": r.city, "count": int(r.n), "ruled": status.get(r.key) == "author"}
            for r in land.itertuples()]
    pre = lookup[(lookup["status"] != "author") & ~lookup["key"].isin(hidden) & (lookup["n_rows"] > 0)]
    queue = (pre.assign(weight=pre["key"].map(weight).fillna(0))
             .sort_values(["weight", "n_rows"], ascending=False).head(60))
    return {"land_wide": land,
            "rule_next": [{"key": r.key, "weight": int(r.weight), "from_plate": r.rule == "plate"}
                          for r in queue.itertuples()],
            "unknown": [k for k in lookup.loc[lookup["nationality"].eq("unknown"), "key"] if k not in hidden],
            "unknown_held_back": int(sum(k in hidden for k in lookup.loc[lookup["nationality"].eq("unknown"), "key"]))}


def load_merges(path: Path = MERGES) -> pd.DataFrame:
    """The proposed type merges: ``group``, ``merged_type``, ``old_type``, ``decision`` (proposed / merge / keep), ``note``."""
    return pd.read_csv(path, dtype=str, keep_default_na=False) if Path(path).exists() else \
        pd.DataFrame(columns=["group", "merged_type", "old_type", "decision", "note"])


def merge_groups(merges: pd.DataFrame, types: pd.DataFrame, lookup: pd.DataFrame | None = None) -> list[dict]:
    """Each proposed merge with its types, their strings and plate types. A merge may join only types of one aggregate
    and one default sponsor (so no analysis column and no sponsor changes); one that does not raises."""
    out = []
    for gid, g in merges.groupby("group", sort=True):
        old = list(g["old_type"])
        missing = [t for t in old if t not in types.index]
        if missing:
            raise ValueError(f"type merge {gid}: {missing} not in the type table")
        aggs, sponsors = set(types.loc[old, "aggregate"]), set(types.loc[old, "sponsor"])
        if len(aggs) > 1 or len(sponsors) > 1:
            raise ValueError(f"type merge {gid}: joins aggregates {sorted(aggs)} or sponsors {sorted(sponsors)}")
        parts = []
        for t in old:
            s = lookup[lookup["type"].str.split(SPLIT).map(lambda p: t in p)] if lookup is not None else pd.DataFrame()
            plates = sorted(set(s["plate_english"]) - {""}) if not s.empty else []
            parts.append({"type": t, "n_strings": int(len(s)), "plates": plates,
                          "examples": list(s["example"].head(6)) if not s.empty else []})
        plate_sets = [frozenset(p["plates"]) for p in parts if p["plates"]]
        out.append({"group": gid, "merged_type": g["merged_type"].iloc[0], "aggregate": next(iter(aggs)),
                    "sponsor": next(iter(sponsors)), "note": next((n for n in g["note"] if n), ""),
                    "decision": g["decision"].iloc[0], "plates_differ": len(set(plate_sets)) > 1, "types": parts})
    return out


def apply_merge_decisions(folder: Path | None = None, path: Path = MERGES) -> dict:
    """The page's merge decisions (``folder/merges/<group>.json``: ``decision`` merge or keep, ``name`` if renamed) into
    ``statutory_type_merges.csv``. Records the decision only: an approved merge is carried into the type table, the
    rules and the lookup by a separate step, checked to leave every ``n_type_*`` column as it was."""
    merges = load_merges(path)
    seen = {}
    for doc in sorted((Path(READBACK if folder is None else folder) / "merges").glob("*.json")):
        d = json.loads(doc.read_text(encoding="utf-8"))
        d = d.get("data", d)
        if d.get("group") in set(merges["group"]) and d.get("decision") in ("merge", "keep"):
            seen[d["group"]] = d
    for gid, d in seen.items():
        hit = merges["group"] == gid
        merges.loc[hit, "decision"] = d["decision"]
        if d.get("name"):
            merges.loc[hit, "merged_type"] = d["name"]
    if seen:
        merges.to_csv(path, index=False, encoding="utf-8")
    return {"merges decided": len(seen), "merge": sum(d["decision"] == "merge" for d in seen.values()),
            "keep apart": sum(d["decision"] == "keep" for d in seen.values())}


def page_data(lookup: pd.DataFrame, lab: pd.DataFrame, pairs: pd.DataFrame, types: pd.DataFrame) -> dict:
    """Strings to rule (those of the blind sample held back until that check is done), twin conflicts, nationality
    questions, the type table, the agreement, and what is still open (:func:`issues`)."""
    hidden_rows = held_out()
    hidden = {k for e, c, r, k in zip(lab["edition"], lab["city"], lab["row"], lab["key"]) if (e, c, r) in hidden_rows}
    look = lookup[~lookup["key"].isin(hidden) & (lookup["n_rows"] > 0)].copy()
    cities = lab.groupby("key")["city"].agg(lambda s: ", ".join(sorted(set(s))))
    look["cities"] = look["key"].map(cities).fillna("")
    look["counted_as"] = look["type"].map(lambda t: SPLIT.join(canonical(p, types) for p in str(t).split(SPLIT)))
    shown_pairs = pairs[~pairs["key_cz"].isin(hidden) & ~pairs["key_de"].isin(hidden)] if not pairs.empty else pairs
    conflicts = shown_pairs[(shown_pairs["type_cz"] != shown_pairs["type_de"]) |
                            (shown_pairs["sponsor_cz"] != shown_pairs["sponsor_de"]) |
                            (shown_pairs["nationality_cz"] != shown_pairs["nationality_de"])] \
        if not shown_pairs.empty else shown_pairs
    if not conflicts.empty:
        conflicts = conflicts.assign(differs=[", ".join(f for f in ("type", "sponsor", "nationality")
                                                        if r[f"{f}_cz"] != r[f"{f}_de"]) for _, r in conflicts.iterrows()])
    tt = types.reset_index()
    clean = lambda df: df.astype(object).where(df.notna(), "").to_dict("records")   # noqa: E731
    paired = set(pairs["key_cz"]) | set(pairs["key_de"]) if not pairs.empty else set()
    in_pair = pd.Series([(e, c) in {x for pr in EDITION_PAIRS for x in pr}
                         for e, c in zip(lab["edition"], lab["city"])], index=lab.index)
    unpaired = lab[in_pair & ~lab["key"].isin(paired) & ~lab["key"].isin(hidden)][
        ["edition", "city", "quote", "key", "type", "count"]]
    counts = type_counts(lab)
    return {"version": __version__, "n_rows": int(len(lab)), "n_keys": int((lookup["n_rows"] > 0).sum()),
            "frozen": FROZEN, "n_ruled": int((lookup["status"] == "author").sum()), "issues": issues(lookup, lab, hidden),
            "merges": merge_groups(load_merges(), types, lookup[~lookup["key"].isin(hidden)]),
            "n_hidden_keys": len(hidden), "agreement": agreement(pairs, lab),
            "lookup": clean(look), "conflicts": clean(conflicts), "types": clean(tt),
            "unpaired": clean(unpaired), "count_diffs": clean(counts[counts["cz"] != counts["de"]]),
            "sponsors": ["state", "Land", "district", "municipal", "church", "voluntary", "commercial", "chamber",
                         "foundation"],
            "nationalities": ["czech", "german", "both", "utraquist", "territorial", "none", "not stated"]}


def page(data: dict, out: Path = OUT) -> Path:
    blob = json.dumps(data, ensure_ascii=False, default=str).replace("</", r"<\/")
    html = PAGE_TEMPLATE.read_text(encoding="utf-8").replace("/*__DATA__*/null", blob)
    out.mkdir(parents=True, exist_ok=True)
    (out / "index.html").write_text(html, encoding="utf-8")
    return out / "index.html"


def run(write_lookup: bool = True, rulings: Path | None = None) -> dict:
    """Build the lookup, label the rows, pair the twins, write the page. With ``rulings`` (the page's read-back
    folder) the lookup is built afresh from the rules and the rulings laid over it, so the table is a function of
    the two and a ruling that merely confirms a pre-label takes the pre-label of the rules as they are now."""
    from utils.recheck import statutory_counts as sc

    statutory = pd.read_parquet(STATUTORY)
    rules, types = load_rules(), load_types()
    vocab = sc.vocabulary()
    v = vocab.drop_duplicates("code")
    plate = {c: (e, k) for c, e, k in zip(v["code"], v["english"], v["category"])}
    report = None
    if rulings is not None:                  # the only way the frozen lookup changes: pre-labels afresh, rulings over them
        lookup, report = apply_rulings(build_lookup(statutory, rules, plate, types), rulings)
    elif LOOKUP.exists():                    # frozen (D-240): read as §9 reads it, never re-labelled here
        lookup = frozen_lookup().astype({"n_rows": int})
        write_lookup = False
    else:
        lookup = build_lookup(statutory, rules, plate, types)
    if write_lookup:
        lookup.to_csv(LOOKUP, index=False, encoding="utf-8")
    flagged, _ = sc.totals(statutory, vocab)
    lab = label(statutory, lookup, types, flagged)
    pairs = twins(lab, types)
    OUT.mkdir(parents=True, exist_ok=True)
    lab.to_csv(OUT / "labelled.csv", index=False, encoding="utf-8")

def slug(name: str) -> str:
    """A column-name piece: ``savings banks and credit associations`` → ``savings_banks_and_credit_associations``."""
    return re.sub(r"[^0-9a-z]+", "_", str(name).lower().replace("'", "")).strip("_")


def frozen_lookup(path: Path = LOOKUP) -> pd.DataFrame:
    """The frozen lookup, as written by the last ``--rulings`` run."""
    return pd.read_csv(path, dtype=str, keep_default_na=False)


def digest(*paths: Path) -> str:
    """sha256 over the files (the lookup and the type table), the first twelve hex digits: what a build was labelled by."""
    import hashlib

    h = hashlib.sha256()
    for p in paths:
        h.update(Path(p).read_bytes())
    return h.hexdigest()[:12]


def mentions(statutory: pd.DataFrame, lookup: pd.DataFrame | None = None, types: pd.DataFrame | None = None,
             vocab: pd.DataFrame | None = None) -> pd.DataFrame:
    """``register_statutory_types``: every printed row of ``register_statutory`` labelled from the frozen lookup, one
    row per body (``part``), with ``institution`` = counted as an institution of the city. Raises if a printed string
    is not in the lookup: the lookup is frozen, so a new reading needs a ruling, not a guess."""
    from utils.recheck import statutory_counts as sc

    lookup = frozen_lookup() if lookup is None else lookup
    types = load_types() if types is None else types
    stat = statutory.assign(is_total=statutory["is_total"].astype(bool), count=statutory["count"].astype(int))
    missing = sorted(set(stat["quote"].map(key_of)) - set(lookup["key"]))
    if missing:
        raise ValueError(f"statutory lookup (frozen {FROZEN}): {len(missing)} printed string(s) it does not hold, e.g. "
                         f"{missing[:3]} — rule them on the page and read back with --rulings")
    left_out, _ = sc.totals(stat, sc.vocabulary() if vocab is None else vocab)
    lab = label(stat, lookup, types, left_out)
    lab.insert(3, "part", lab.groupby(["edition", "city", "row"]).cumcount())
    lab["count"] = pd.to_numeric(lab["count"], errors="coerce").fillna(1).astype(int)
    lab = lab.astype({"n_czech": "Int64", "n_german": "Int64"})
    lab["institution"] = (lab["counted"] & ~lab["is_person"] & ~lab["domain"].eq(LEGAL_FORM)
                          & ~(lab["housed"] & lab["aggregate"].eq(HOUSED_COLLECTION)))
    out = lab.drop(columns=["quote", "plate_code", "reader_category", "languages"])
    if out.duplicated(TABLE_KEYS["register_statutory_types"]).any():
        raise AssertionError("register_statutory_types: key (edition, city, row, part) duplicated")
    return out.reset_index(drop=True)


#: The nations a city's institutions are counted under (``n_nation_*``); ``split`` is never one: a split line's
#: numbers go to each nation, what its split leaves over to ``not stated``.
NATIONS = ["czech", "german", "both", "utraquist", "territorial", "none", "not stated", "unknown"]


def nation_rows(inst: pd.DataFrame) -> pd.DataFrame:
    """The institutions by nation, a split line broken into its nations (the remainder ``not stated``)."""
    whole = inst[inst["nationality"] != "split"][["edition", "city", "nationality", "count"]]
    parts = []
    for r in inst[inst["nationality"] == "split"].itertuples():
        # a split the row's own reading did not number (stage 2 gave no languages) is not stated as a whole
        cz, de = (0, 0) if pd.isna(r.n_czech) else (int(r.n_czech), int(r.n_german))
        rest = int(r.count) - cz - de
        parts += [(r.edition, r.city, n, k) for n, k in (("czech", cz), ("german", de), ("not stated", rest)) if k > 0]
    return pd.concat([whole, pd.DataFrame(parts, columns=whole.columns)], ignore_index=True)


def city_columns(lab: pd.DataFrame, types: pd.DataFrame | None = None) -> pd.DataFrame:
    """Per city (``edition``, ``city``): ``n_type_<aggregate>`` (the institutions, each printed count), every aggregate
    of the type table present so a 0 is a 0; ``n_person_<type>`` (Q1); ``n_sponsor_<sponsor>``, ``n_nation_<value>``
    over the same institutions; ``n_order_house_{male,female}`` and ``orders_present`` (the orders with a house)."""
    types = load_types() if types is None else types
    inst = lab[lab["institution"]]
    persons = lab[lab["counted"] & lab["is_person"]]
    owner = ["edition", "city"]
    cities = lab[owner].drop_duplicates().set_index(owner).index

    def block(frame, col, prefix, universe):
        wide = frame.pivot_table(index=owner, columns=col, values="count", aggfunc="sum", fill_value=0)
        wide = wide.reindex(index=cities, columns=sorted(set(universe) | set(wide.columns)), fill_value=0)
        return wide.rename(columns=lambda c: prefix + (slug(c) or "not_given"))

    aggregates = sorted(set(types.loc[(types["person"] != "yes") & ~types["domain"].isin(("not counted", LEGAL_FORM)),
                                      "aggregate"]))
    person_types = sorted(types.index[types["person"] == "yes"])
    legal_types = sorted(types.index[types["domain"] == LEGAL_FORM])
    legal = lab[lab["counted"] & lab["domain"].eq(LEGAL_FORM)]
    out = pd.concat([block(inst, "aggregate", "n_type_", aggregates),
                     block(persons, "type", "n_person_", person_types),
                     block(legal, "type", "n_legal_form_", legal_types),
                     block(inst, "sponsor", "n_sponsor_", []),
                     block(nation_rows(inst), "nationality", "n_nation_", NATIONS)], axis=1)
    houses = inst[inst["type"].eq("religious house")]
    for sex in ("male", "female"):
        hit = houses[houses["order_sexes"].str.contains(rf"\b{sex}\b", na=False)]     # "female" holds "male"
        out[f"n_order_house_{sex}"] = hit.groupby(owner)["count"].sum().reindex(cities).fillna(0)
    out["orders_present"] = (houses.groupby(owner)["orders"]
                             .agg(lambda s: "; ".join(sorted({o.strip() for v in s for o in v.split(";") if o.strip()})))
                             .reindex(cities).fillna(""))
    ints = [c for c in out.columns if c != "orders_present"]
    return out.astype({c: "Int64" for c in ints}).reset_index()


def add(tables: dict) -> dict:
    """``tables`` with ``register_statutory_types`` and the per-city type columns joined onto
    ``register_statutory_counts`` (the edition that table keeps) and onto ``register_okres``'s prose rows (``<NA>``
    on the plate rows: the plates have no such types). Run after ``statutory_counts.add``."""
    lab = mentions(tables["register_statutory"])
    cols = city_columns(lab)
    counts = tables["register_statutory_counts"]
    before = len(counts)
    counts = counts.merge(cols, on=["edition", "city"], how="left", validate="1:1")
    if len(counts) != before or counts[[c for c in cols.columns if c.startswith("n_type_")]].isna().any().any():
        raise AssertionError("register_statutory_counts: a city found no type columns")
    okres = tables["register_okres"]
    extra = [c for c in cols.columns if c.startswith(TYPE_COLUMNS) or c == "orders_present"]
    okres = okres.merge(counts[["okres_id", *extra]], on="okres_id", how="left", validate="1:1")
    if len(okres) != len(tables["register_okres"]):
        raise AssertionError("register_okres: the type columns changed its rows")
    report = {"frozen": FROZEN, "lookup_digest": digest(LOOKUP, TYPES), "mentions": int(len(lab)),
              "institutions": int(lab.loc[lab["institution"], "count"].sum()),
              "persons": int(lab.loc[lab["counted"] & lab["is_person"], "count"].sum()),
              "housed_collections_left_out": int((lab["counted"] & lab["housed"]
                                                  & lab["aggregate"].eq(HOUSED_COLLECTION)).sum()),
              "ruled_strings": int((frozen_lookup()["status"] == "author").sum())}
    out = {**tables, "register_statutory_types": lab, "register_statutory_counts": counts, "register_okres": okres}
    out["_meta"] = {**tables.get("_meta", {}), "statutory_types": report}
    return out


def write(tables: dict, out_dir: Path) -> None:
    """``register_statutory_types`` as the register writes its tables (parquet and csv); the two count tables it
    widened are written by ``statutory_counts.write``."""
    for name in TABLE_KEYS:
        tables[name].to_parquet(Path(out_dir) / f"{name}.parquet", index=False)
        tables[name].to_csv(Path(out_dir) / f"{name}.csv", index=False, encoding="utf-8")

def apply_rulings(lookup: pd.DataFrame, folder: Path = READBACK) -> tuple[pd.DataFrame, dict]:
    """The page's rulings (read back as JSON, one file per document, under ``folder/rulings``, ``folder/types``)
    into the lookup: a ruled string takes the fields the author set and becomes ``status = author``; every string of
    an accepted type keeps its pre-label as the author's. Returns the lookup and a count of what changed."""
    look = lookup.copy().set_index("key")
    report = {"strings ruled": 0, "fields set": 0, "types accepted": 0, "strings by accepted type": 0}
    for doc in sorted((folder / "rulings").glob("*.json")):
        d = json.loads(doc.read_text(encoding="utf-8"))
        d = d.get("data", d)
        key = d.get("key")
        if key not in look.index:
            continue
        for f in ("type", "sponsor", "nationality", "note"):
            if d.get(f):
                look.loc[key, f] = d[f]
                report["fields set"] += 1
        if d.get("nationality"):
            look.loc[key, "nationality_basis"] = "author"
        elif d.get("sponsor") or d.get("type"):
            # The pre-label's nationality was read with the pre-label's sponsor ("none" for a Land body); a ruled
            # sponsor or type reads it again, so a ruling never leaves a nationality its own reading would not give.
            first = look.loc[key, "type"].split(SPLIT)[0]
            look.loc[key, ["nationality", "nationality_basis"]] = list(nationality_of(key, first, look.loc[key, "sponsor"]))
        look.loc[key, "status"] = "author"
        report["strings ruled"] += 1
    for doc in sorted((folder / "types").glob("*.json")):
        d = json.loads(doc.read_text(encoding="utf-8"))
        d = d.get("data", d)
        if not d.get("accepted"):
            continue
        report["types accepted"] += 1
        hit = look["type"].str.split(SPLIT).map(lambda parts: d.get("type") in parts) & (look["status"] != "author")
        look.loc[hit, "status"] = "author"
        report["strings by accepted type"] += int(hit.sum())
    return look.reset_index()[LOOKUP_FIELDS], report
