How it works

Ground truth: the author's named clicks (german_click_page, collection german_clicks, saved as JSON under output/german_click_page/clicks/german_clicks), in section-image coordinates. Three measures, each against a click and not against "a box was placed":

  1. candidate recall — the share of clicks that fall inside some candidate cluster the locator considers (locate_glyphs.section_candidates). Below 1 the candidate step itself loses glyphs (merged with a letter, split, or never found), whatever is trained on top of it.
  2. placement accuracy — of the gold claims the locator places, the share whose box holds a click of any token (the box is on a pictogram) and the share whose box holds a click of the claimed token (the box is on the right pictogram), for the Czech-trained locator and for a German-trained one placing each page by the others (german_locator_trial).
  3. where the false boxes go — the placed boxes holding no click, by what they are (key digits left of the text column, others).

Offices sections of gold pages 0142, 0276, 0480 only. Reads the section images and clicks; writes nothing. Outside utils/lexikon_1910 so as not to change the route's code digests.

def candidate_recall(data, clicks: dict, pad_lines: float) -> dict:
    """Clicks inside at least one / exactly one candidate cluster, over every click."""
    stats, missed = Counter(), []
    for section, cands in data:
        for name in section.images:
            mine = [c for c in cands if c.image == name]
            for x, y, token, page, strip in clicks.get(name, []):
                hits = [c for c in mine if inside(c.box, x, y, pad_lines * c.lh)]
                stats["clicks"] += 1
                stats["in_any"] += bool(hits)
                stats["in_one"] += len(hits) == 1
                if not hits:
                    missed.append((page, strip, token, x, y))
    return {**stats, "missed": missed}

def placement_accuracy(placed: dict, clicks: dict, pad_lines: float, text_columns: dict) -> dict:
    """Of the placed claims: box on any click, on a click of the claimed token; what the others are."""
    stats, other = Counter(), Counter()
    for section, cands, where in placed.values():
        for claim in section.claims:
            i = where[claim.order]
            stats["claims"] += 1
            if i is None:
                stats["unplaced"] += 1
                continue
            c = cands[i]
            stats["placed"] += 1
            near = [tok for x, y, tok, _p, _s in clicks.get(c.image, [])
                    if inside(c.box, x, y, pad_lines * c.lh)]
            on_any = bool(near)
            on_token = claim.token in near
            stats["on_any_click"] += on_any
            stats["on_claimed_token"] += on_token
            if not on_any:
                left = c.box[0] < text_columns.get(c.image, 0.0) - 0.3 * c.lh
                other["key_column" if left else "elsewhere"] += 1
    return {**stats, "false_boxes": dict(other)}

def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--rounds", type=int, default=4)
    parser.add_argument("--pad", type=float, default=0.25, help="slack around a box, in line heights")
    args = parser.parse_args(argv)

    import locate_glyphs as lg
    import german_locator_trial as trial

    clicks = load_clicks()
    pages = sorted({p for v in clicks.values() for _x, _y, _t, p, _s in v})
    sections, _unmatched = trial.german_sections()
    sections = [s for s in sections if s.kind == "offices" and s.scan in pages]
    data = [(s, lg.section_candidates(s)) for s in sections]
    columns = {}
    for section, _c in data:
        columns.update(lg.section_text_columns(section))
    print(f"{sum(len(v) for v in clicks.values())} clicks on pages {pages}; {len(sections)} offices sections, "
          f"{sum(len(s.claims) for s in sections)} gold claims, {sum(len(c) for _s, c in data)} candidate clusters")

    r = candidate_recall(data, clicks, args.pad)
    print(f"\ncandidate recall: {r['in_any']} of {r['clicks']} clicks inside some candidate "
          f"({r['in_any'] / r['clicks']:.3f}); inside exactly one {r['in_one']}")
    print("  clicks in no candidate (page, strip, token, x, y):", r["missed"][:12], "..." if len(r["missed"]) > 12 else "")

    print("\nCzech-trained locator:")
    czech = lg.load_sections()
    locator = lg.train_locator(czech, {s.scan for s, _ in czech if s.role == "calibration"}, args.rounds)
    placed = {}
    for section, cands in data:
        scores = locator.glyph.score(cands) if cands else []
        for c, v in zip(cands, scores):
            c.glyphness = float(v)
        placed[section.key] = (section, cands, locator.place(cands, section.claims))
    report("Czech-trained", placement_accuracy(placed, clicks, args.pad, columns))

    print("\nGerman-trained locator, each page placed by locators trained on the other German gold pages:")
    all_sections, _u = trial.german_sections()
    all_data = [(s, lg.section_candidates(s)) for s in all_sections]
    calibration = sorted({s.scan for s, _ in all_data if s.role == "calibration"})
    seeds = trial.safe_plate_templates(ROOT / "reference" / "places" / "symbols" / "de")
    trained, placed = {}, {}
    for section, cands in all_data:
        if section.kind != "offices" or section.scan not in pages:
            continue
        key = tuple(sorted(set(calibration) - {section.scan}))
        if key not in trained:
            trained[key] = lg.train_locator(all_data, set(key), args.rounds, seeds)
        loc = trained[key]
        scores = loc.glyph.score(cands) if cands else []
        for c, v in zip(cands, scores):
            c.glyphness = float(v)
        placed[section.key] = (section, cands, loc.place(cands, section.claims))
    report("German-trained", placement_accuracy(placed, clicks, args.pad, columns))
    return 0
