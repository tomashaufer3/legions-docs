How it works

The Czech table is read block by block (run_lexikon_1910_pilot.py, channel A): the NDK ALTO says where each municipality's block sits on the page, the block is cropped there and a vision model reads its seventeen columns. Nothing the text layer read reaches the model, only where it found things. The German volumes — Böhmen, Mähren, Schlesien — have no ALTO, and the UBW / MSVK word layers are OCR whose names are garbage (Brocnai for Brocná), so here both halves come from the page itself:

  • where — lexikon_1910_image finds the frame, the column rules, the header and the table foot on the rendered scan (measured against the ALTO on the Czech volume, D-80). The printed rows are the ink bands of the figure columns (houses … pop_female), which every row fills and whose digits leave clean white between rows. A row with ink in the area column opens a block, as the area cell marks the municipality line on the Czech side (lexikon_1910.segment_blocks); one with ink in the area column but none in the running-number column is a district total (the German print sets the political and judicial districts' totals as a bold unnumbered line), read as a block of its own so that its name labels the blocks under it. Rows above the page's first opener continue the last municipality of the page before.
  • what — every block is read whole by the model: the names off the image, with the prompt's rules for the two forms the German volume prints (Krasnowes, Krásnoves) and their diacritics, and the figures, which the volume's own identities then check (male + female = total, confessions, languages + foreigners; Schlesien's languages include polnisch) and, where a block has parts, the parts against their municipality.

The columns, their printed labels and the identities are the volume profile's (reference/places/volumes/&lt;slug&gt;.toml), so one code reads the three German volumes: Schlesien prints an eighteenth column (lang_polish) and gets it in its prompt, its schema and its language identity.

def table_ink(image, geo: lximg.PageGeometry):
    """The dark pixels that are print: every pixel column that is dark down most of the
    table body (a column rule, single or double) taken out, and nothing else. Not the image
    module's rule-free ink: there a column rule and the digits set against it merge into one
    wide "rule" (de 0380: x 214-243 across a number column of 160-228), and taking it out
    wiped the running numbers. A digit never fills half the body's rows."""
    grey = lximg.to_grey(image)
    mask = grey < geo.threshold
    if geo.header_bottom is None or geo.table_foot is None:
        return mask
    # Found at the image module's lenient cut: the UBW and MSVK scans print the interior
    # rules faint, and at the type's cut they cover only a quarter to a half of the body.
    faint = grey < lximg.ink_threshold(grey, lximg.RULE_SHARE)
    slab = max(20, int(RULE_SLAB * geo.scale))
    for y0 in range(geo.header_bottom, geo.table_foot, slab):
        y1 = min(geo.table_foot, y0 + slab)
        rules = faint[y0:y1].mean(axis=0) > RULE_COVERAGE
        for shift in (1, 2):                                # a rule's anti-aliased edge
            rules |= np.roll(rules, shift) | np.roll(rules, -shift)
        mask[y0:y1, rules] = False
    return mask

def printed_rows(ink, geo: lximg.PageGeometry, profile) -> list[PrintedRow]:
    """The table body's rows, top to bottom (header and foot excluded): the figure columns'
    ink bands, each with the name lines that belong to it (:func:`_join_name_lines`)."""
    s = geo.scale
    if geo.kind != "table" or geo.header_bottom is None or geo.table_foot is None \
            or geo.n_columns != len(profile.columns):
        return []
    top, foot = geo.header_bottom + int(4 * s), geo.table_foot - int(2 * s)
    first, last = (column_index(profile, c) for c in (ROW_COLUMNS[0], ROW_COLUMNS[-1]))
    area, seq, name = (column_index(profile, c) for c in ("area_ha", "seq", "name"))
    inset = int(CELL_INSET * s)
    x0, x1 = geo.columns[first] + inset, geo.columns[last + 1] - inset
    figures = [r for r in _column_bands(ink, geo, first, last, top, foot)
               if ROW_COVERAGE <= ink[r[0]:r[1] + 1, x0:x1].any(axis=0).mean() < HORIZONTAL_RULE]

    def per_row(column: int):
        return ink[:, geo.columns[column] + inset:geo.columns[column + 1] - inset].sum(axis=1)

    has_area = lambda r: _cell_ink(ink, geo, area, r[0], r[1]) > AREA_INK * s  # noqa: E731
    rows = _join_name_lines(figures, per_row(name) + per_row(seq), per_row(seq), has_area, top, foot)
    return [PrintedRow(a, b, has_area((a, b)), _cell_ink(ink, geo, seq, a, b) > SEQ_INK * s)
            for a, b in rows]

def cut_blocks(page: str, rows: list[PrintedRow]) -> list[TableBlock]:
    """Rows into blocks: an area row opens one (a district total if it has no running
    number), the rows under it are its parts; rows above the first opener continue the
    previous page's last municipality."""
    blocks: list[TableBlock] = []
    for row in rows:
        if row.area:
            kind = "municipality" if row.seq else "summary"
            opened = sum(b.kind != "continued" for b in blocks)        # the continued block is #00
            blocks.append(TableBlock(page, opened + 1, kind, [row]))
        elif blocks:                            # a part, or a total's "darunter Militär" line
            blocks[-1].rows.append(row)
        else:
            blocks.append(TableBlock(page, 0, "continued", [row]))
    return blocks

def page_blocks(image, profile, page: str) -> tuple[lximg.PageGeometry, list[TableBlock]]:
    """``(geometry, blocks)`` of one rendered page; no blocks on a page that is not a table
    or whose column rules do not match the profile."""
    geo = detect(image, profile)
    if geo.kind != "table":
        return geo, []
    return geo, cut_blocks(page, printed_rows(table_ink(image, geo), geo, profile))

def crop_block(image, geo: lximg.PageGeometry, block: TableBlock,
               target_width: int = lx.VISION_LONG_EDGE):
    """The block across the whole table, frame to frame, scaled to the vision ceiling (the
    Czech ``lx.tile_block`` with one tile). The pad stays under half a row, so no figure of a
    neighbouring block is cut into the image."""
    pad = int(CROP_PAD * geo.scale)
    left, right = geo.frame
    crop = image.crop((max(0, left - 2 * pad), max(0, block.top - pad),
                       min(image.width, right + 2 * pad), min(image.height, block.bottom + pad)))
    return lx.scale_to_width(crop, target_width)

def build_system(profile) -> str:
    """The Czech channel-A prompt (``run_lexikon_1910_pilot.TRANSCRIBE_SYSTEM``) with the
    German volume's title, its own columns and their printed labels, and the name rules."""
    polish = "lang_polish" in [c.name for c in profile.columns]
    columns = "\n".join(f"{i + 1:2d}  {c.name:16s} {c.label}" for i, c in enumerate(profile.columns))
    names = NAME_RULES.format(other="Czech or Polish" if polish else "Czech",
                              polish="; Polish ą ć ę ł ń ó ś ź ż" if polish else "")
    return f"""\
You are transcribing one block of a printed statistical table from the 1910 Austrian census: \
*{profile.title}*, printed in German.

The image shows one municipality (Ortsgemeinde) and the places (Ortschaften) printed beneath \
it, or one district total line, or rows that continue a municipality from the previous page — \
the line under the image says which. Vertical rules separate the columns. The table has \
exactly {len(profile.columns)} columns, in this left-to-right order (field name, printed \
header):

{columns}

{names}
Figures:
- Transcribe exactly what is printed. Do not correct, complete, infer, or compute any value
  from any other value. If a figure is genuinely illegible, put "?" — do not guess.
- A cell holding a centred dot "." is the printed nil marker. Return "" for it. Return "" for
  a truly empty cell too. Never write "0" or "." for either.
- area_ha is printed only on the municipality line (and on a district total).
- The image is cropped to one block, but a sliver of the row above or below may be visible at
  the very top or bottom edge. Ignore any partial row that is cut off.
- Every printed line gets one object, in printed order, including a "darunter Militär" line.
- Long place names wrap onto a second printed line; join them into one object.
- Return the figures as digit strings, without thousands separators.
"""

def grade_block(rows: list[dict], profile, kind: str) -> list[dict]:
    """The volume's identities on every row, and on a municipality block with parts the parts
    against their municipality, column by column (``lx.grade_row``, ``lx.grade_block_sum``)."""
    parsed = [lx.Row(page="", top=0, bottom=0, cells={k: str(v or "") for k, v in r.items()})
              for r in rows]
    parsed = [r for r in parsed if not lx.OF_WHICH_RE.search(r.get("name"))
              and not _MILITARY_RE.search(r.get("name"))]
    out = []
    for n, row in enumerate(parsed):
        for ident in profile.identities:
            inv = lx.Invariant(ident.key, ident.label, ident.parts, ident.whole)
            verdict, detail = lx.grade_row(row, inv)
            out.append({"row": n, "check": ident.key, "verdict": verdict, "detail": detail})
    if kind == "municipality" and len(parsed) > 1:
        block = lx.Block(page="", index=0, municipality=parsed[0], settlements=parsed[1:])
        for col, (verdict, detail) in lx.grade_block_sum(block).items():
            out.append({"row": 0, "check": f"I4_parts:{col}", "verdict": verdict, "detail": detail})
    return out

def summary_label(rows: list[dict]) -> tuple[str, str]:
    """``(kind, district)`` of a total: ``okres`` or ``court`` from the row that carries the
    district's mark (a count line such as ``3 GB., 67 ⊕`` may be read first), and that row's
    first name form without leaders or marks. ``("", "")`` for a total with no mark — a
    footnote's ``Einschließlich der Stadt …`` total — which labels nothing."""
    for row in rows:
        text = str(row.get("name", ""))
        kind = "okres" if _OKRES_MARK_RE.match(text) else "court" if _COURT_MARK_RE.match(text) else ""
        if kind:
            text = _COURT_MARK_RE.sub("", _OKRES_MARK_RE.sub("", text))
            text = re.sub(r"\.{2,}|\*\)|\[[^\]]*\]|\(\d+\)", " ", text)
            return kind, " ".join(text.split(",")[0].split()).strip(" .")
    return "", ""

def court_labels(blocks_read: list[tuple[TableBlock, list[dict]]],
                 carry: tuple[str, str] = ("", "")) -> list[tuple[str, str]]:
    """``(okres, court)`` each block was printed under, from the last total lines above it,
    carried across pages in reading order as ``lx.segment_blocks`` carries the Czech summary
    rows. A political district's line sets both (an okres that is its own court prints one
    line); a judicial district's line sets the court."""
    labels, (okres, court) = [], carry
    for block, rows in blocks_read:
        if block.kind == "summary" and rows:
            kind, name = summary_label(rows)
            if name and kind == "okres":
                okres, court = name, name
            elif name and kind == "court":
                court = name
        labels.append((okres, court))
    return labels

def table_rows(page_reads: list[tuple[TableBlock, list[dict], str, str, str]], profile) -> list[dict]:
    """Every row each block was read with, one per printed line, keyed by its block, in the
    frame ``run_lexikon_1910_pilot.table_rows`` writes (channel ``A``), plus the block's kind,

"""The German volumes' table, cut into blocks from the image and read by a model (D-160).

The Czech table is read block by block (``run_lexikon_1910_pilot.py``, channel A): the NDK
ALTO says where each municipality's block sits on the page, the block is cropped there and a
vision model reads its seventeen columns. Nothing the text layer *read* reaches the model, only
where it found things. The German volumes — Böhmen, Mähren, Schlesien — have no ALTO, and the
UBW / MSVK word layers are OCR whose names are garbage (``Brocnai`` for *Brocná*), so here both
halves come from the page itself:

* **where** — :mod:`lexikon_1910_image` finds the frame, the column rules, the header and the
  table foot on the rendered scan (measured against the ALTO on the Czech volume, D-80). The
  printed rows are the ink bands of the figure columns (``houses`` … ``pop_female``), which
  every row fills and whose digits leave clean white between rows. A row with ink in the
  **area** column opens a block, as the area cell marks the municipality line on the Czech
  side (``lexikon_1910.segment_blocks``); one with ink in the area column but none in the
  running-number column is a **district total** (the German print sets the political and
  judicial districts' totals as a bold unnumbered line), read as a block of its own so that
  its name labels the blocks under it. Rows above the page's first opener **continue** the
  last municipality of the page before.
* **what** — every block is read whole by the model: the names off the image, with the
  prompt's rules for the two forms the German volume prints (``Krasnowes, Krásnoves``) and
  their diacritics, and the figures, which the volume's own identities then check
  (male + female = total, confessions, languages + foreigners; Schlesien's languages include
  ``polnisch``) and, where a block has parts, the parts against their municipality.

The columns, their printed labels and the identities are the volume profile's
(``reference/places/volumes/<slug>.toml``), so one code reads the three German volumes:
Schlesien prints an eighteenth column (``lang_polish``) and gets it in its prompt, its schema
and its language identity.
"""

from __future__ import annotations

import re
from dataclasses import dataclass, field

import numpy as np

from . import lexikon_1910 as lx
from . import lexikon_1910_image as lximg

__version__ = "0.1.0"

#: The figure columns whose ink gives the printed rows: every row — municipality, part,
#: district total — prints houses and population, and digits stand clear of the rows around.
ROW_COLUMNS = ("houses", "pop_total", "pop_male", "pop_female")
#: At the Czech reference scale (``lexikon_1910_image.REFERENCE_FRAME``): a row is at least
#: this tall, a gap of up to this many blank pixel rows stays inside one row, cells are read
#: this far inside their rules, and a crop takes this much white above and below its rows.
ROW_MIN_HEIGHT, ROW_HOLE, CELL_INSET, CROP_PAD = 10, 4, 6, 6
#: Dark pixels a pixel row of the name or number column needs to count as print in a gap
#: between two rows of figures, so a speck does not stretch a row.
LINE_INK = 3
#: Dark pixels in a cell, at the reference scale, that make it printed: an area figure or a
#: running number is two to five digits (hundreds of pixels); a speck or a dot leader is not.
AREA_INK, SEQ_INK = 30, 15
#: Measured 2026-10-29 on the seven Böhmen gold pages: rows 25-31 px at the German scans'
#: scale (~0.75), and on 0142 0276 0454 0479 exactly as many area rows as the gold's
#: municipalities (19, 15, 11, 18).

KINDS = ("municipality", "summary", "continued")


@dataclass
class PrintedRow:
    """One printed table row: its band and which cells carry ink."""

    top: int
    bottom: int
    area: bool
    seq: bool


@dataclass
class TableBlock:
    """What one model call reads: a municipality and its parts, a district total, or the rows
    that continue the previous page's last municipality."""

    page: str
    index: int
    kind: str
    rows: list[PrintedRow] = field(default_factory=list)

    @property
    def top(self) -> int:
        return min(r.top for r in self.rows)

    @property
    def bottom(self) -> int:
        return max(r.bottom for r in self.rows)

    @property
    def block_id(self) -> str:
        return f"{self.page}#{self.index:02d}"


def column_index(profile, name: str) -> int:
    return [c.name for c in profile.columns].index(name)


def detect(image, profile) -> lximg.PageGeometry:
    """The page's geometry; ``kind == "table"`` with the profile's column count to cut on."""
    return lximg.detect(image, expected_columns=len(profile.columns), full_width_headings=True)


#: A pixel column dark on more than this share of a slab's rows is a rule there. The slab is
#: short because the scans are skewed: a rule drifts 4 px down the body of de 0142 and 10 px
#: down de 0479, so over the whole body no single pixel column holds it.
RULE_COVERAGE, RULE_SLAB = 0.8, 120


def table_ink(image, geo: lximg.PageGeometry):
    """The dark pixels that are print: every pixel column that is dark down most of the
    table body (a column rule, single or double) taken out, and nothing else. Not the image
    module's rule-free ink: there a column rule and the digits set against it merge into one
    wide "rule" (de 0380: x 214-243 across a number column of 160-228), and taking it out
    wiped the running numbers. A digit never fills half the body's rows."""
    grey = lximg.to_grey(image)
    mask = grey < geo.threshold
    if geo.header_bottom is None or geo.table_foot is None:
        return mask
    # Found at the image module's lenient cut: the UBW and MSVK scans print the interior
    # rules faint, and at the type's cut they cover only a quarter to a half of the body.
    faint = grey < lximg.ink_threshold(grey, lximg.RULE_SHARE)
    slab = max(20, int(RULE_SLAB * geo.scale))
    for y0 in range(geo.header_bottom, geo.table_foot, slab):
        y1 = min(geo.table_foot, y0 + slab)
        rules = faint[y0:y1].mean(axis=0) > RULE_COVERAGE
        for shift in (1, 2):                                # a rule's anti-aliased edge
            rules |= np.roll(rules, shift) | np.roll(rules, -shift)
        mask[y0:y1, rules] = False
    return mask


def _bands(profile_1d, hole: int) -> list[tuple[int, int]]:
    """Runs of rows with any ink, bridging gaps of up to ``hole`` blank rows."""
    out, start, end, gap = [], None, 0, 0
    for y, dark in enumerate(profile_1d > 0):
        if dark:
            start = y if start is None else start
            end, gap = y, 0
        elif start is not None:
            gap += 1
            if gap > hole:
                out.append((start, end))
                start = None
    if start is not None:
        out.append((start, end))
    return out


def _cell_ink(ink, geo, column: int, top: int, bottom: int) -> int:
    inset = int(CELL_INSET * geo.scale)
    return int(ink[top:bottom + 1, geo.columns[column] + inset:geo.columns[column + 1] - inset].sum())


def _column_bands(ink, geo, first: int, last: int, top: int, foot: int,
                  hole: float = ROW_HOLE) -> list[list[int]]:
    """``[[top, bottom], …]`` of the ink in columns ``first``..``last``, page coordinates.
    Each column is read inside its rules, so a rule running down the page joins nothing."""
    s, inset = geo.scale, int(CELL_INSET * geo.scale)
    profile = sum(ink[top:foot, geo.columns[c] + inset:geo.columns[c + 1] - inset].sum(axis=1)
                  for c in range(first, last + 1))
    bands = [(a, b) for a, b in _bands(profile, int(hole * s)) if b - a >= int(ROW_MIN_HEIGHT * s)]
    return [[a + top, b + top] for a, b in _split_tall(bands, profile)]


#: A band whose ink reaches this share of the figure columns' width is a horizontal rule, and
#: one whose ink reaches less than ``ROW_COVERAGE`` of it is specks. Measured 2026-10-29 on
#: de 0380, 0479, maehren 0025, schlesien 0016: specks and the closing rule's remnants reach
#: 0.2-1.7 %, a ``darunter Militär`` line of single figures 4.8-8.1 %, a row of figures 14-55 %.
HORIZONTAL_RULE, ROW_COVERAGE = 0.85, 0.03

#: A band taller than this many median rows holds rows set so close that their ink touches
#: (de-maehren 0025: Rakwitz, Saitz and Schakwitz read as one band).
TALL_ROW = 1.6


def _split_tall(bands: list[tuple[int, int]], profile) -> list[tuple[int, int]]:
    """Each over-tall band cut into as many rows as its height holds, at the thinnest ink
    within a third of a row of where each cut is expected."""
    if not bands:
        return bands
    median = float(np.median([b - a for a, b in bands]))
    out = []
    for a, b in bands:
        k = round((b - a) / median) if b - a > TALL_ROW * median else 1
        cuts, reach = [a], max(1, int(median / 3))
        for i in range(1, k):
            want = a + round(i * (b - a) / k)
            lo, hi = max(cuts[-1] + 1, want - reach), min(b - 1, want + reach)
            cuts.append(lo + int(np.argmin(profile[lo:hi + 1])) if hi > lo else want)
        cuts.append(b)
        out += [(cuts[i] if i == 0 else cuts[i] + 1, cuts[i + 1]) for i in range(len(cuts) - 1)]
    return out


def _join_name_lines(rows: list[list[int]], name_ink, seq_ink, area_of, top: int,
                     foot: int) -> list[list[int]]:
    """Give every wrapped name line to its row, as ``lx.extract_rows`` joins a wrapped name.

    Decided per gap between two rows of figures, not per line: the name column's lines stand
    too close to separate reliably. The German print sets a wrapped name's figures on its
    **last** line — ``…, Lhota``, the three-line ``…, auch Ústrašice``, the total ``P. B.
    Reichenau an der / Kněžna, …`` — so a gap's name ink is the head of the row below. The one
    exception is the line under a district total (``23 ⊕, 57 ◐, dar. 3 ◎.``, its count of
    places), which is the total's tail. Where the gap prints a running number, the ink from it
    down is the row below's and the ink above it the row above's. Below the last row, the ink
    is the last row's.

    ``name_ink`` and ``seq_ink`` are per-pixel-row ink counts of the page (index = y);
    ``area_of(row)`` says whether a row of figures has an area."""
    rows = [list(r) for r in rows]
    in_total = False
    for k in range(len(rows) + 1):
        gap_top = rows[k - 1][1] + 1 if k else top
        gap_bottom = rows[k][0] - 1 if k < len(rows) else foot - 1
        ys = [y for y in range(gap_top, gap_bottom + 1) if name_ink[y] >= LINE_INK]
        numbered = [y for y in ys if seq_ink[y] >= LINE_INK]
        if k == len(rows) or (not numbered and in_total):
            split = gap_bottom + 1                       # all of it is the row above's tail
        else:
            split = numbered[0] if numbered else gap_top
        upper, lower = [y for y in ys if y < split], [y for y in ys if y >= split]
        if upper and k:
            rows[k - 1][1] = max(rows[k - 1][1], upper[-1])
        if lower and k < len(rows):
            rows[k][0] = min(rows[k][0], lower[0])
        if k < len(rows) and area_of(rows[k]):
            in_total = not any(seq_ink[y] >= LINE_INK for y in range(rows[k][0], rows[k][1] + 1))
    return rows


def printed_rows(ink, geo: lximg.PageGeometry, profile) -> list[PrintedRow]:
    """The table body's rows, top to bottom (header and foot excluded): the figure columns'
    ink bands, each with the name lines that belong to it (:func:`_join_name_lines`)."""
    s = geo.scale
    if geo.kind != "table" or geo.header_bottom is None or geo.table_foot is None \
            or geo.n_columns != len(profile.columns):
        return []
    top, foot = geo.header_bottom + int(4 * s), geo.table_foot - int(2 * s)
    first, last = (column_index(profile, c) for c in (ROW_COLUMNS[0], ROW_COLUMNS[-1]))
    area, seq, name = (column_index(profile, c) for c in ("area_ha", "seq", "name"))
    inset = int(CELL_INSET * s)
    x0, x1 = geo.columns[first] + inset, geo.columns[last + 1] - inset
    figures = [r for r in _column_bands(ink, geo, first, last, top, foot)
               if ROW_COVERAGE <= ink[r[0]:r[1] + 1, x0:x1].any(axis=0).mean() < HORIZONTAL_RULE]

    def per_row(column: int):
        return ink[:, geo.columns[column] + inset:geo.columns[column + 1] - inset].sum(axis=1)

    has_area = lambda r: _cell_ink(ink, geo, area, r[0], r[1]) > AREA_INK * s  # noqa: E731
    rows = _join_name_lines(figures, per_row(name) + per_row(seq), per_row(seq), has_area, top, foot)
    return [PrintedRow(a, b, has_area((a, b)), _cell_ink(ink, geo, seq, a, b) > SEQ_INK * s)
            for a, b in rows]


def cut_blocks(page: str, rows: list[PrintedRow]) -> list[TableBlock]:
    """Rows into blocks: an area row opens one (a district total if it has no running
    number), the rows under it are its parts; rows above the first opener continue the
    previous page's last municipality."""
    blocks: list[TableBlock] = []
    for row in rows:
        if row.area:
            kind = "municipality" if row.seq else "summary"
            opened = sum(b.kind != "continued" for b in blocks)        # the continued block is #00
            blocks.append(TableBlock(page, opened + 1, kind, [row]))
        elif blocks:                            # a part, or a total's "darunter Militär" line
            blocks[-1].rows.append(row)
        else:
            blocks.append(TableBlock(page, 0, "continued", [row]))
    return blocks


def page_blocks(image, profile, page: str) -> tuple[lximg.PageGeometry, list[TableBlock]]:
    """``(geometry, blocks)`` of one rendered page; no blocks on a page that is not a table
    or whose column rules do not match the profile."""
    geo = detect(image, profile)
    if geo.kind != "table":
        return geo, []
    return geo, cut_blocks(page, printed_rows(table_ink(image, geo), geo, profile))


def crop_block(image, geo: lximg.PageGeometry, block: TableBlock,
               target_width: int = lx.VISION_LONG_EDGE):
    """The block across the whole table, frame to frame, scaled to the vision ceiling (the
    Czech ``lx.tile_block`` with one tile). The pad stays under half a row, so no figure of a
    neighbouring block is cut into the image."""
    pad = int(CROP_PAD * geo.scale)
    left, right = geo.frame
    crop = image.crop((max(0, left - 2 * pad), max(0, block.top - pad),
                       min(image.width, right + 2 * pad), min(image.height, block.bottom + pad)))
    return lx.scale_to_width(crop, target_width)


# ── the reader ────────────────────────────────────────────────────────────────
#: What each kind of block is said to be, before the reader transcribes it.
INSTRUCTIONS = {
    "municipality": "Transcribe this block.",
    "summary": "This image is one district total line (a bold line with no running number). "
               "Transcribe it as one row; its name goes in the name field.",
    "continued": "These rows continue a municipality whose first line is on the previous page. "
                 "Transcribe them; none of them is the municipality's own line.",
}

NAME_RULES = """\
Names — the most important cell, and the one the text layer gets wrong:
- Copy the name exactly as printed, letter for letter. The volume prints the German name
  first and the {other} name after a comma ("Krasnowes, Krásnoves"); keep both, in that
  order, with the comma. A third form after "auch" or "also" is kept too.
- Keep every diacritic: German ä ö ü ß; Czech á č ď é ě í ň ó ř š ť ú ů ý ž{polish}. Do not
  normalise, modernise, translate or complete a name, and do not swap the two forms.
- Leave out the dot leaders that run from the name to the figures. Keep a footnote mark
  printed against a name ("*)") and a part's own number in brackets ("(4)").
- A printed symbol after a name (a place-type pictogram) is written as "[?]".
"""


def build_system(profile) -> str:
    """The Czech channel-A prompt (``run_lexikon_1910_pilot.TRANSCRIBE_SYSTEM``) with the
    German volume's title, its own columns and their printed labels, and the name rules."""
    polish = "lang_polish" in [c.name for c in profile.columns]
    columns = "\n".join(f"{i + 1:2d}  {c.name:16s} {c.label}" for i, c in enumerate(profile.columns))
    names = NAME_RULES.format(other="Czech or Polish" if polish else "Czech",
                              polish="; Polish ą ć ę ł ń ó ś ź ż" if polish else "")
    return f"""\
You are transcribing one block of a printed statistical table from the 1910 Austrian census: \
*{profile.title}*, printed in German.

The image shows one municipality (Ortsgemeinde) and the places (Ortschaften) printed beneath \
it, or one district total line, or rows that continue a municipality from the previous page — \
the line under the image says which. Vertical rules separate the columns. The table has \
exactly {len(profile.columns)} columns, in this left-to-right order (field name, printed \
header):

{columns}

{names}
Figures:
- Transcribe exactly what is printed. Do not correct, complete, infer, or compute any value
  from any other value. If a figure is genuinely illegible, put "?" — do not guess.
- A cell holding a centred dot "." is the printed nil marker. Return "" for it. Return "" for
  a truly empty cell too. Never write "0" or "." for either.
- area_ha is printed only on the municipality line (and on a district total).
- The image is cropped to one block, but a sliver of the row above or below may be visible at
  the very top or bottom edge. Ignore any partial row that is cut off.
- Every printed line gets one object, in printed order, including a "darunter Militär" line.
- Long place names wrap onto a second printed line; join them into one object.
- Return the figures as digit strings, without thousands separators.
"""


def row_schema(profile) -> dict:
    names = [c.name for c in profile.columns]
    return {"type": "object", "additionalProperties": False, "required": ["rows"],
            "properties": {"rows": {"type": "array", "items": {
                "type": "object", "additionalProperties": False, "required": names,
                "properties": {n: {"type": "string"} for n in names}}}}}


# ── what the reading is checked against ───────────────────────────────────────
def grade_block(rows: list[dict], profile, kind: str) -> list[dict]:
    """The volume's identities on every row, and on a municipality block with parts the parts
    against their municipality, column by column (``lx.grade_row``, ``lx.grade_block_sum``)."""
    parsed = [lx.Row(page="", top=0, bottom=0, cells={k: str(v or "") for k, v in r.items()})
              for r in rows]
    parsed = [r for r in parsed if not lx.OF_WHICH_RE.search(r.get("name"))
              and not _MILITARY_RE.search(r.get("name"))]
    out = []
    for n, row in enumerate(parsed):
        for ident in profile.identities:
            inv = lx.Invariant(ident.key, ident.label, ident.parts, ident.whole)
            verdict, detail = lx.grade_row(row, inv)
            out.append({"row": n, "check": ident.key, "verdict": verdict, "detail": detail})
    if kind == "municipality" and len(parsed) > 1:
        block = lx.Block(page="", index=0, municipality=parsed[0], settlements=parsed[1:])
        for col, (verdict, detail) in lx.grade_block_sum(block).items():
            out.append({"row": 0, "check": f"I4_parts:{col}", "verdict": verdict, "detail": detail})
    return out


#: The garrison sub-row, the German print's *mezi tím vojsko*: no identity applies to it alone.
_MILITARY_RE = re.compile(r"darunter\s+Milit", re.IGNORECASE)


#: The German total lines' own labels: ``P. B. Turnau, Turnov`` (Böhmen) or ``Pol. Bez.
#: Auspitz, Hustopeč`` (Mähren, Schlesien, with ``(Landbezirk)`` where a city is apart) is a
#: political district; ``III. GB. Schatzlar, Žacléř`` a judicial district, numbered in it.
_OKRES_MARK_RE = re.compile(r"^\s*(?:[IVXL]+\.\s*)?(?:P\.\s*B\.|Pol\.\s*Bez\.)\s*(?:\([^)]*\)\s*)?",
                            re.IGNORECASE)
_COURT_MARK_RE = re.compile(r"^\s*(?:[IVXL]+\.\s*)?G\.?\s*B\.\s*(?=[^\W\d_])", re.IGNORECASE)


def summary_label(rows: list[dict]) -> tuple[str, str]:
    """``(kind, district)`` of a total: ``okres`` or ``court`` from the row that carries the
    district's mark (a count line such as ``3 GB., 67 ⊕`` may be read first), and that row's
    first name form without leaders or marks. ``("", "")`` for a total with no mark — a
    footnote's ``Einschließlich der Stadt …`` total — which labels nothing."""
    for row in rows:
        text = str(row.get("name", ""))
        kind = "okres" if _OKRES_MARK_RE.match(text) else "court" if _COURT_MARK_RE.match(text) else ""
        if kind:
            text = _COURT_MARK_RE.sub("", _OKRES_MARK_RE.sub("", text))
            text = re.sub(r"\.{2,}|\*\)|\[[^\]]*\]|\(\d+\)", " ", text)
            return kind, " ".join(text.split(",")[0].split()).strip(" .")
    return "", ""


def court_labels(blocks_read: list[tuple[TableBlock, list[dict]]],
                 carry: tuple[str, str] = ("", "")) -> list[tuple[str, str]]:
    """``(okres, court)`` each block was printed under, from the last total lines above it,
    carried across pages in reading order as ``lx.segment_blocks`` carries the Czech summary
    rows. A political district's line sets both (an okres that is its own court prints one
    line); a judicial district's line sets the court."""
    labels, (okres, court) = [], carry
    for block, rows in blocks_read:
        if block.kind == "summary" and rows:
            kind, name = summary_label(rows)
            if name and kind == "okres":
                okres, court = name, name
            elif name and kind == "court":
                court = name
        labels.append((okres, court))
    return labels


def table_rows(page_reads: list[tuple[TableBlock, list[dict], str, str, str]], profile) -> list[dict]:
    """Every row each block was read with, one per printed line, keyed by its block, in the
    frame ``run_lexikon_1910_pilot.table_rows`` writes (channel ``A``), plus the block's kind,
    the German district names its total lines gave and the Czech court they resolve to
    (``court_label``, the key the Czech table rows and ``courts.csv`` share)."""
    names = [c.name for c in profile.columns]
    out = []
    for block, rows, okres_de, court_de, court_cs in page_reads:
        base = {"channel": "A", "page": block.page, "block_id": block.block_id,
                "block_index": block.index, "block_kind": block.kind,
                "okres_label": okres_de, "court_label": court_cs or court_de, "court_de": court_de,
                "printed_rows": len(block.rows)}
        for n, row in enumerate(rows):
            out.append({**base, "row": n, **{c: str(row.get(c) or "").strip() for c in names}})
    return out
