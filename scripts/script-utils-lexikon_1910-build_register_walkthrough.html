How it works

The Digitising the 1910 register sheet follows one municipality — Brzice, No. 1 of court Česká Skalice (okres Náchod) in the Czech volume, Brzitz, No. 3 in the German one — through the page. This script carries it on through the step after the frozen run: the register build (lexikon_1910_register) on the Czech gold run, which holds the page, and the crosswalk (lexikon_1910_crosswalk) run on the court's German numbering. Everything it writes is computed by that code; the one transcribed input is the German page's own numbering, read off the sheet's figure (GERMAN_PRINTED).

Writes reference/places/figure/register_walkthrough.json; utils/docs/build_docs.py only reads it. Run from the repo root (no API, no model; reads output/frozen and data/final, writes nothing else)::

python utils/lexikon_1910/build_register_walkthrough.py

                         k: (None if pd.isna(s[k]) else (float(s[k]) if k in ("lat", "lon") else str(s[k])))

                                   "okres_1910", "lat", "lon")}})
    own = inst[(inst["edition"] == "cz") & (inst["court_key"] == str(COURT)) & (inst["muni_seq"] == MUNI)]

    return {

"""The docs site's Brzice walkthrough: from the printed line to the place authority (D-148, D-150).

The *Digitising the 1910 register* sheet follows one municipality — Brzice, No. 1 of court
Česká Skalice (okres Náchod) in the Czech volume, Brzitz, No. 3 in the German one — through
the page. This script carries it on through the step after the frozen run: the register build
(:mod:`lexikon_1910_register`) on the Czech gold run, which holds the page, and the crosswalk
(:mod:`lexikon_1910_crosswalk`) run on the court's German numbering. Everything it writes is
computed by that code; the one transcribed input is the German page's own numbering, read off
the sheet's figure (:data:`GERMAN_PRINTED`).

Writes ``reference/places/figure/register_walkthrough.json``; ``utils/docs/build_docs.py``
only reads it. Run from the repo root (no API, no model; reads ``output/frozen`` and
``data/final``, writes nothing else)::

    python utils/lexikon_1910/build_register_walkthrough.py
"""

from __future__ import annotations

import datetime as dt
import json
import sys
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
for _p in (ROOT, Path(__file__).resolve().parent):
    if str(_p) not in sys.path:
        sys.path.insert(0, str(_p))

from utils.lexikon_1910 import lexikon_1910_crosswalk as lxcw     # noqa: E402
from utils.lexikon_1910 import lexikon_1910_names as lxn          # noqa: E402
from utils.lexikon_1910 import lexikon_1910_register as reg       # noqa: E402

__version__ = "1.0.0"

OUT = ROOT / "reference" / "places" / "figure" / "register_walkthrough.json"
RUNS = [reg.FROZEN / "cz_gold", reg.FROZEN / "de-bohemia_gold"]
COURT, MUNI = 301057121, 1                                 # Česká Skalice; Brzice
#: The German volume's own numbering of the court's first municipalities, transcribed from scan
#: 0283 of the UBW Böhmen volume — the figure's `de-notes` box (`1. Aujezdetz, Oujezdec … 2.
#: Böhmisch Skalitz`), `de-above`, `de-block` and `de-below`. The German gold run does not hold
#: this page, so it is the one input here that is read off the print rather than off a run.
GERMAN_PRINTED = [(1, "Aujezdetz, Oujezdec"), (2, "Böhmisch Skalitz, Česká Skalice"),
                  (3, "Brzitz, Brzice"), (4, "Cerwena Hora, Červená Hora")]
#: The Czech municipality the gold page lacks: Újezdec sorts late in the Czech alphabet and is
#: printed on a later page. Its cell is how the Czech table prints it (Czech, German).
LATER_CZECH = (99, "Újezdec, Aujezdetz")


def _candidates(cell: str, authority: lxn.Authority, okres: int, n: int = 3) -> list[dict]:
    """The best ``n`` settlements of the okres for a name cell, scored as the anchor scores."""
    from rapidfuzz import fuzz

    forms = [lxn.normalise(f) for f in lxn.parse_name_cell(cell)]
    rows = []
    for o in authority.okres(okres):
        if o.status != "obec" or not o.keys:
            continue
        best = max(fuzz.ratio(f, k) for f in forms for k in o.keys)
        rows.append({"settlement_id": o.settlement_id, "cs_name": o.cs_name, "score": int(best)})
    return sorted(rows, key=lambda r: -r["score"])[:n]


def _german_order(places: pd.DataFrame) -> dict:
    """The court's Czech cells sorted by German name, and the crosswalk run on the German page."""
    cz = places[(places["edition"] == "cz") & (places["court_id"] == COURT)
                & (places["settlement_seq"] == 0)][["muni_seq", "printed_name", "settlement_id"]]
    cz = cz.assign(german_key=cz["printed_name"].map(lxcw.german_key)).sort_values("german_key")
    de = pd.DataFrame([{"muni_seq": n, "printed_name": name if n != 1 else "",
                        "settlement_id": pd.NA, "settlement_seq": 0} for n, name in GERMAN_PRINTED])
    de["settlement_id"] = de["settlement_id"].astype("string")
    czf = cz.drop(columns="german_key").assign(settlement_seq=0)

    def run(czech: pd.DataFrame, complete: bool) -> dict:
        pairs, status = lxcw.pair_court(de, czech.reset_index(drop=True), complete)
        czech = czech.reset_index(drop=True)
        return {"status": status, "pairs": [
            {"de_muni": int(de.at[d, "muni_seq"]), "cz_muni": int(czech.at[c, "muni_seq"]),
             "cz_name": czech.at[c, "printed_name"], "how": how} for d, (c, how) in sorted(pairs.items())]}

    later = pd.DataFrame([{"muni_seq": LATER_CZECH[0], "printed_name": LATER_CZECH[1],
                           "settlement_id": pd.NA, "settlement_seq": 0}])
    later["settlement_id"] = later["settlement_id"].astype("string")
    return {"czech_by_german_name": [{"muni_seq": int(r.muni_seq), "printed_name": r.printed_name,
                                      "german_key": r.german_key} for r in cz.itertuples()],
            "german_printed": [{"muni_seq": n, "name": name} for n, name in GERMAN_PRINTED],
            "gold_page_only": run(czf, complete=False),
            "with_the_later_page": run(pd.concat([czf, later], ignore_index=True), complete=True)}

    muni = tables["register_municipalities"].set_index("settlement_id")

        rows.append({"settlement_seq": int(r.settlement_seq), "printed_name": r.printed_name,
                     "name_source": r.name_source, "settlement_id": r.settlement_id,
                     "score": int(r.anchor_score), "runner_up": int(r.anchor_runner_up),
                     "status": r.anchor_status,
                     "authority": None if s is None else {
                         k: (None if pd.isna(s[k]) else (float(s[k]) if k in ("lat", "lon") else str(s[k])))
                         for k in ("cs_name", "name_1910", "de_name", "status_1910", "parent_1910",
                                   "okres_1910", "lat", "lon")}})
    own = inst[(inst["edition"] == "cz") & (inst["court_key"] == str(COURT)) & (inst["muni_seq"] == MUNI)]
    sid = block.loc[block["settlement_seq"] == 0, "settlement_id"].iloc[0]

        "court": {"court_id": COURT, "court_cs": block["court_cs"].iloc[0], "okres_id": COURT // 1000},
        "places": rows,
        "candidates": _candidates(block["printed_name"].iloc[0], authority, COURT // 1000),
        "institutions": [{"settlement_seq": int(r.settlement_seq), "mark": r.mark, "count": int(r.count),
                          "rule": r.rule, "settled_by": r.settled_by,

def main() -> None:
    payload = build()
    OUT.write_text(json.dumps(payload, ensure_ascii=False, indent=1, default=str), encoding="utf-8")
    print(f"wrote {OUT.relative_to(ROOT).as_posix()}")


if __name__ == "__main__":
    main()
