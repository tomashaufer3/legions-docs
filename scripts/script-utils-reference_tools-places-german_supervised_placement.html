How it works

Placement is what fails on German (the Czech-trained locator puts the box on the right glyph for 27 % of claims; german_click_recall). Two suspects are tried here, each against the clicks:

  • the entry mask. locate_glyphs.entry_mask forbids every candidate that has no entry (muni) for every claim, and the entry finder tags only 23 % of the German clusters (70 % of the Czech): where tagging is partial, most candidates can hold no claim at all. --mask off places without it;
  • the glyphness model. The Czech locator learns "what a glyph looks like" with no hand label; here locate_glyphs.fit_glyph_model is fitted on the clicks instead (a candidate that holds a click is a positive, every other candidate a negative), each page placed by a model trained on the other two.

Measure, as in german_click_recall: of the gold claims, the share whose placed box holds a click of the claimed token (on_claimed_token over all claims). Optionally --czech also runs the Czech-trained locator on the same candidates for the baseline. Three pages are three folds: a direction, not a rate; these are the gold pages the route was tuned on.

def label_candidates(cands: list, clicks: dict, pad: float = 0.1, narrow: float | None = None) -> list[int]:
    """1 where a candidate holds a click (and, with ``narrow``, is no wider than that many line heights)."""
    import locate_glyphs as lg

    out = []
    for c in cands:
        held = any(lg.inside(c.box, x, y, pad * c.lh) for x, y, *_ in clicks.get(c.image, []))
        wide = narrow is not None and (c.box[2] - c.box[0]) > narrow * c.lh
        out.append(int(held and not c.bracket and not wide))
    return out

def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--ink", type=int, default=140, help="ink threshold for the candidate step (route: 128)")
    parser.add_argument("--rounds", type=int, default=4)
    parser.add_argument("--czech", action="store_true", help="also run the Czech-trained locator as the baseline")
    parser.add_argument("--pad", type=float, default=0.25)
    args = parser.parse_args(argv)

    import numpy as np

    import german_click_recall as rec
    import german_locator_trial as trial
    import locate_glyphs as lg

    orig = lg.ink_mask
    lg.ink_mask = lambda image, threshold=args.ink: orig(image, threshold)
    clicks = rec.load_clicks()
    pages = sorted({p for v in clicks.values() for _x, _y, _t, p, _s in v})
    sections, _ = trial.german_sections()
    sections = [s for s in sections if s.kind == "offices" and s.scan in pages]
    data = [(s, lg.section_candidates(s)) for s in sections]
    columns = {}
    for section, _c in data:
        columns.update(lg.section_text_columns(section))
    n_claims = sum(len(s.claims) for s, _ in data)
    tagged = sum(c.muni is not None for _s, cs in data for c in cs) / sum(len(cs) for _s, cs in data)
    print(f"ink < {args.ink}: {len(data)} sections, {n_claims} claims, {sum(len(c) for _s, c in data)} candidates "
          f"({tagged:.2f} tagged with an entry)")
    r = rec.candidate_recall(data, clicks, 0.1)
    print(f"candidate recall at 0.1 line heights: {r['in_any']} of {r['clicks']} ({r['in_any'] / r['clicks']:.3f})\n")

    def show(name: str, placed: dict) -> None:
        stats = rec.placement_accuracy(placed, clicks, args.pad, columns)
        print(f"  {name:46s} placed {stats['placed']:3d}/{n_claims}  right glyph {stats['on_claimed_token']:3d}/{n_claims} "
              f"= {stats['on_claimed_token'] / n_claims:.3f}   any click {stats['on_any_click']:3d}/{stats['placed']}")

    if args.czech:
        czech = lg.load_sections()
        locator = lg.train_locator(czech, {s.scan for s, _ in czech if s.role == "calibration"}, args.rounds)
        models = {s.scan: locator.glyph for s, _ in data}
        for mask in (True, False):
            show(f"Czech-trained glyphness, mask {'on ' if mask else 'off'}", place_all(_fresh(data), models, mask))

    for narrow in (None, 1.3):
        trained = {}
        for held in pages:
            xs, ys = [], []
            for section, cands in data:
                if section.scan == held or not cands:
                    continue
                xs.append(lg.feature_matrix(cands))
                ys.append(np.array(label_candidates(cands, clicks, narrow=narrow)))
            trained[held] = lg.fit_glyph_model(np.concatenate(xs), np.concatenate(ys), [p for p in pages if p != held])
        label = "any candidate holding a click" if narrow is None else f"narrow (<= {narrow} lh) candidates only"
        for mask in (True, False):
            show(f"supervised on clicks ({label}), mask {'on ' if mask else 'off'}",
                 place_all(_fresh(data), trained, mask))
    lg.ink_mask = orig
    return 0

def _fresh(data):
    """The same sections and candidates with their entry tags as the entry finder gave them."""
    import copy

    return [(s, copy.deepcopy(c)) for s, c in data]
