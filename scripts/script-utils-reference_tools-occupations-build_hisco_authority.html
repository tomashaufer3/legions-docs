How it works

Why that key and not the HISCO code alone: in the HSN release the four-field key determines HISCLASS, HISCLASS_5, HISCAM and SOCPO exactly — 5,762 keys, zero conflicts — whereas the HISCO code on its own does not: 806 of 1,346 codes (59.9 %) take more than one HISCLASS depending on the modifiers. The release documentation says why: status can move an occupation to another class, any value on relation excludes it from HISCLASS entirely, and status 41 (student) is likewise excluded. Keying on the code alone would silently pick one of several classes. The script asserts the dependency rather than trusting it, so a future release that breaks it fails loudly.

Two sources, both under data/raw/hisco/ (see reference/DATA_SOURCES.md §3a): - History of Work (druid, CC-BY-SA 4.0) — the hierarchy and the modifier code lists. - HSN release 2020.02 — the HISCO -> HISCLASS / HISCLASS_5 / HISCAM / SOCPO crosswalk. Its Dutch title strings are not read; only the code columns are.

Codes present in the hierarchy but never coded by HSN are kept with default modifiers and empty scales, so a Czech string can still resolve to a HISCO code and an English label even where no class exists for it.

def load_hierarchy() -> pd.DataFrame:
    """HISCO micro code -> English label + its major/minor/unit ancestry."""
    h = _read_csv_any(HIER)
    out = pd.DataFrame({
        "hisco": h["hisco_full_id"].map(_pad),
        "hisco_pretty": h["hisco_full_id_pretty"].str.strip(),
        "label_en": h["label45"].str.strip(),
        "major": h["hisco_1_id"].str.strip(),
        "major_label": h["label1"].str.strip(),
        "minor": h["hisco_2_id"].str.strip(),
        "minor_label": h["label2"].str.strip(),
        "unit": h["hisco_3_id"].str.strip(),
        "unit_label": h["label3"].str.strip(),
    })
    dupes = out["hisco"].duplicated().sum()
    if dupes:
        print(f"  note: {dupes} duplicate hisco rows in the hierarchy — keeping the first")
        out = out.drop_duplicates("hisco")
    return out

def load_code_lists() -> dict[str, dict[str, str]]:
    """status/relation/product code -> label, from the druid modifier code lists
    (``<name>_id, label, country_code, preferred``)."""
    lists: dict[str, dict[str, str]] = {}
    for name, path in CODE_LISTS.items():
        df = _read_csv_any(path)
        code_col = f"{name}_id" if f"{name}_id" in df.columns else df.columns[0]
        label_col = "label" if "label" in df.columns else df.columns[1]
        if "preferred" in df.columns:      # one preferred label per code
            df = df[df["preferred"].str.strip() == "1"]
        df = df.drop_duplicates(code_col)
        lists[name] = dict(zip(df[code_col].str.strip(), df[label_col].str.strip()))
        print(f"  {name:<9} {len(lists[name]):>3} codes ({code_col} -> {label_col})")
    return lists

def load_hsn_scales() -> pd.DataFrame:
    """(hisco, status, relation, product) -> HISCLASS / HISCLASS_5 / HISCAM / SOCPO.

    Asserts the functional dependency instead of assuming it: if a future release makes
    the key ambiguous, this raises rather than silently picking a class.
    """
    cols = ["HISCO", "STATUS", "RELATION", "PRODUCT",
            "HISCLASS", "HISCLASS_5", "HISCAM_U1", "SOCPO"]
    df = pd.read_csv(HSN, dtype=str, keep_default_na=False, usecols=cols)
    df["HISCO"] = df["HISCO"].map(_pad)

    key = ["HISCO", "STATUS", "RELATION", "PRODUCT"]
    scales = ["HISCLASS", "HISCLASS_5", "HISCAM_U1", "SOCPO"]
    ambiguous = df.groupby(key)[scales].nunique().gt(1).any(axis=1)
    if ambiguous.any():
        bad = ambiguous[ambiguous].index.tolist()[:10]
        raise AssertionError(
            f"{int(ambiguous.sum())} (hisco,status,relation,product) keys carry more than "
            f"one scale value — the crosswalk is no longer a function. First: {bad}"
        )

    n_titles = df.groupby(key).size().rename("n_hsn")
    out = df.drop_duplicates(key).set_index(key).join(n_titles).reset_index()
    out = out.rename(columns={
        "HISCO": "hisco", "STATUS": "status", "RELATION": "relation",
        "PRODUCT": "product", "HISCLASS": "hisclass", "HISCLASS_5": "hisclass_5",
        "HISCAM_U1": "hiscam_u1", "SOCPO": "socpo",
    })
    for c in ["hisclass", "hisclass_5", "hiscam_u1", "socpo"]:
        out[c] = _to_num(out[c])

    # The 12->5 aggregation is documented; check the file agrees with the documentation.
    both = out.dropna(subset=["hisclass", "hisclass_5"])
    expected = both["hisclass"].astype(int).map(HISCLASS_5_FROM_12)
    mismatch = int((expected != both["hisclass_5"].astype(int)).sum())
    if mismatch:
        raise AssertionError(
            f"{mismatch} rows where HISCLASS_5 disagrees with the documented "
            f"12->5 aggregation (§4.1)"
        )
    return out

def build_authority() -> pd.DataFrame:
    print("Reading inputs …")
    hier = load_hierarchy()
    print(f"  hierarchy {len(hier):,} HISCO micro codes")
    lists = load_code_lists()
    hsn = load_hsn_scales()
    print(f"  HSN       {len(hsn):,} (hisco,status,relation,product) keys "
          f"from {hsn['n_hsn'].sum():,} titles")

    # Codes the hierarchy knows but HSN never coded: keep them with default modifiers so
    # a Czech string can still reach a code + label (scales stay <NA>, honestly empty).
    coded = set(hsn["hisco"])
    uncoded = hier.loc[~hier["hisco"].isin(coded), ["hisco"]].copy()
    for c, v in [("status", "-9"), ("relation", "-9"), ("product", "-9")]:
        uncoded[c] = v
    for c in ["hisclass", "hisclass_5", "socpo"]:
        uncoded[c] = pd.Series(pd.NA, index=uncoded.index, dtype="Int64")
    uncoded["hiscam_u1"] = pd.Series(pd.NA, index=uncoded.index, dtype="Float64")
    uncoded["n_hsn"] = 0
    print(f"  {len(uncoded):,} hierarchy codes carry no HSN class (kept, scales empty)")

    df = pd.concat([hsn, uncoded], ignore_index=True)
    df["in_hsn"] = df["n_hsn"].gt(0)
    df = df.merge(hier, on="hisco", how="left")

    # HSN codes with no micro-hierarchy row: the sentinels (-1/-2/-3) plus codes HSN
    # placed at a coarser level. Label those from their ancestor rather than shipping a
    # blank, and record which level the label came from.
    anc = _ancestor_labels(hier)
    df["label_level"] = pd.NA
    df.loc[df["label_en"].notna() & (df["label_en"] != ""), "label_level"] = "micro"
    for level, width, label_col in [("unit", 3, "unit_label"),
                                    ("minor", 2, "minor_label"),
                                    ("major", 1, "major_label")]:
        gap = df["label_level"].isna() & ~df["hisco"].str.startswith("-")
        if not gap.any():
            break
        fill = df.loc[gap, "hisco"].str[:width].map(anc[level])
        hit = fill.notna()
        df.loc[gap & hit.reindex(df.index, fill_value=False), "label_en"] = fill[hit]
        df.loc[gap & hit.reindex(df.index, fill_value=False), label_col] = fill[hit]
        df.loc[gap & hit.reindex(df.index, fill_value=False), "label_level"] = level

    # The label is only half of what the ancestor supplies. Those same codes had blank
    # `major`/`minor`/`unit` because they carry no micro-hierarchy row to merge, which left
    # `occupation_hisco_major` <NA> for 553 records that do carry a HISCO code *and* a
    # HISCLASS — `brusič skla` -> 89125 alone is 304 of them. HISCO is positional: across
    # all 1,675 micro rows `major == hisco[0]`, `minor == hisco[1]` and `unit == hisco[2]`
    # exactly, so the ancestry is derivable from the code itself. Fill every level, not
    # just the one the label came from; sentinels (-1/-2/-3) are left alone.
    real = ~df["hisco"].str.startswith("-")
    blank = lambda c: df[c].isna() | df[c].astype("string").fillna("").str.strip().eq("")
    n_filled = 0
    for level, pos, width, code_col, label_col in [
        ("major", 0, 1, "major", "major_label"),
        ("minor", 1, 2, "minor", "minor_label"),
        ("unit", 2, 3, "unit", "unit_label"),
    ]:
        gap = real & blank(code_col)
        n_filled += int(gap.sum())
        df.loc[gap, code_col] = df.loc[gap, "hisco"].str[pos]
        lab_gap = real & blank(label_col)
        df.loc[lab_gap, label_col] = df.loc[lab_gap, "hisco"].str[:width].map(anc[level])
    if n_filled:
        print(f"  {n_filled:,} ancestry cells derived from the code's own digits")

    coarse = df.loc[df["label_level"].isin(["unit", "minor", "major"]), "hisco"].unique()
    if len(coarse):
        print(f"  {len(coarse)} codes labelled from an ancestor group "
              f"(coarser than micro): {sorted(coarse)[:12]}")
    blank = sorted(df.loc[df["label_level"].isna(), "hisco"].unique())
    if blank:
        print(f"  {len(blank)} codes with no label at all: {blank}")

    # Labels
    df["hisclass_label"] = df["hisclass"].map(
        lambda v: HISCLASS_LABELS.get(int(v)) if pd.notna(v) else pd.NA).astype("string")
    df["hisclass_5_label"] = df["hisclass_5"].map(
        lambda v: HISCLASS_5_LABELS.get(int(v)) if pd.notna(v) else pd.NA).astype("string")
    df["socpo_label"] = df["socpo"].map(
        lambda v: SOCPO_LABELS.get(int(v)) if pd.notna(v) else pd.NA).astype("string")
    for name in CODE_LISTS:
        df[f"{name}_label"] = df[name].map(lists[name]).astype("string")

    df["hisclass"] = df["hisclass"].astype("Int64")
    df["hisclass_5"] = df["hisclass_5"].astype("Int64")
    df["socpo"] = df["socpo"].astype("Int64")

    order = ["hisco", "status", "relation", "product",
             "label_en", "label_level", "hisco_pretty",
             "major", "major_label", "minor", "minor_label", "unit", "unit_label",
             "hisclass", "hisclass_label", "hisclass_5", "hisclass_5_label",
             "hiscam_u1", "socpo", "socpo_label",
             "status_label", "relation_label", "product_label",
             "n_hsn", "in_hsn"]
    return df[order].sort_values(["hisco", "status", "relation", "product"],
                                 ignore_index=True)

def authority_report(df: pd.DataFrame) -> None:
    print("\nAuthority table")
    print(f"  rows                    {len(df):,}")
    print(f"  distinct HISCO codes    {df['hisco'].nunique():,}")
    print(f"  default-modifier rows   {int(((df[['status','relation','product']] == '-9').all(axis=1)).sum()):,}")
    print(f"  with an English label   {int(df['label_en'].notna().sum()):,} "
          f"({int((df['label_level'] == 'micro').sum()):,} at micro level)")
    print(f"  with a HISCLASS         {int(df['hisclass'].notna().sum()):,}")
    print(f"  with a HISCAM           {int(df['hiscam_u1'].notna().sum()):,}")
    print("\n  HISCLASS distribution (rows, not people):")
    vc = df["hisclass"].value_counts().sort_index()
    for k, v in vc.items():
        print(f"    {int(k):>3}  {v:>6,}  {HISCLASS_LABELS.get(int(k), '?')}")

    bad = df["hisclass"].dropna()
    assert bad.between(1, 13).all(), "HISCLASS outside 1..13"
    h5 = df["hisclass_5"].dropna()
    assert h5.between(1, 5).all(), "HISCLASS_5 outside 1..5"
    assert not df.duplicated(["hisco", "status", "relation", "product"]).any(), \
        "duplicate (hisco,status,relation,product) rows"
    print("\n  assertions passed: HISCLASS 1..13, HISCLASS_5 1..5, key unique")

"""
build_hisco_authority.py — generate reference/occupations/hisco_authority.csv
=============================================================================
One row per **(hisco, status, relation, product)**, carrying the English label, the
position in the HISCO hierarchy, and the derived stratification scales.

Why that key and not the HISCO code alone: in the HSN release the four-field key
determines HISCLASS, HISCLASS_5, HISCAM and SOCPO exactly — 5,762 keys, **zero**
conflicts — whereas the HISCO code on its own does not: 806 of 1,346 codes (59.9 %) take
more than one HISCLASS depending on the modifiers. The release documentation says why:
status can move an occupation to another class, any value on *relation* excludes it from
HISCLASS entirely, and status 41 (student) is likewise excluded. Keying on the code alone
would silently pick one of several classes. The script asserts the dependency rather than
trusting it, so a future release that breaks it fails loudly.

Two sources, both under data/raw/hisco/ (see reference/DATA_SOURCES.md §3a):
  * History of Work (druid, CC-BY-SA 4.0) — the hierarchy and the modifier code lists.
  * HSN release 2020.02 — the HISCO -> HISCLASS / HISCLASS_5 / HISCAM / SOCPO crosswalk.
    Its Dutch title strings are not read; only the code columns are.

Codes present in the hierarchy but never coded by HSN are kept with default modifiers and
empty scales, so a Czech string can still resolve to a HISCO code and an English label
even where no class exists for it.

Run:  python utils/reference_tools/occupations/build_hisco_authority.py
Emits: reference/occupations/hisco_authority.csv
Prerequisite: python utils/reference_tools/occupations/download_hisco_sources.py
Consumed by utils.parsing.occupations.load_hisco_authority.
"""
from __future__ import annotations

import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

import pandas as pd

__version__ = "1.1.0"   # + ancestry codes backfilled for codes labelled above micro level

RAW = ROOT / "data" / "raw" / "hisco"
OUT = ROOT / "reference" / "occupations" / "hisco_authority.csv"

HIER = RAW / "hisco_45_full_03.csv"
HSN = RAW / "HSN_HISCO_release_2020_02.csv"
CODE_LISTS = {"status": RAW / "status.csv",
              "relation": RAW / "relation.csv",
              "product": RAW / "product.csv"}

# Sentinels, per the HSN release documentation (§2, §4.1–4.3). Note that -9 on the three
# modifier fields means "no modifier", which is the *normal* case and part of the key —
# it is kept as-is; only the scale sentinels below become <NA>.
SCALE_SENTINELS = {"-1", "-9", "-9.00", "999", ""}

# HISCLASS, verbatim from HSN release 2020.02 §4.1. Class 13 is an HSN addition for
# HISCO 99900/99920, whose split into 11 vs 12 is context-dependent.
HISCLASS_LABELS = {
    1: "Higher managers",
    2: "Higher professionals",
    3: "Lower managers",
    4: "Lower professionals, higher and middle clerical and sales personnel",
    5: "Lower clerical and sales personnel",
    6: "Foremen",
    7: "Medium skilled workers",
    8: "Farmers and fishermen",
    9: "Lower skilled workers",
    10: "Lower skilled farm workers",
    11: "Unskilled workers",
    12: "Unskilled farm workers",
    13: "Unskilled workers not specified",
}

# HISCLASS_5, verbatim from §4.1. Also the aggregation the file itself uses — asserted
# below against the observed HISCLASS x HISCLASS_5 crosstab.
HISCLASS_5_LABELS = {
    1: "Elite",
    2: "Lower middle class",
    3: "Skilled workers",
    4: "Self-employed farmers and fishermen",
    5: "Unskilled workers and farm workers",
}
HISCLASS_5_FROM_12 = {1: 1, 2: 1, 3: 2, 4: 2, 5: 2, 6: 2, 7: 3, 9: 3, 8: 4,
                      10: 5, 11: 5, 12: 5, 13: 5}

# SOCPO, §4.3. Levels 5 / 41 / 42 / 2 / 1 are verbatim; level 3 is the "Skilled workers"
# rung of the same five-level ladder (the page break in the release PDF drops the line).
SOCPO_LABELS = {
    1: "Unskilled workers",
    2: "Semi-skilled workers",
    3: "Skilled workers",
    41: "Middle class — farmers",
    42: "Middle class — supervisors of skilled workers",
    5: "Elite",
}


def _require_inputs() -> None:
    missing = [p for p in [HIER, HSN, *CODE_LISTS.values()] if not p.exists()]
    if missing:
        print("Missing input files:")
        for p in missing:
            print(f"  {p.relative_to(ROOT)}")
        print("\nRun:  python utils/reference_tools/occupations/download_hisco_sources.py")
        raise SystemExit(1)


def _read_csv_any(path: Path, **kw) -> pd.DataFrame:
    """Read a druid CSV. The separator is not consistent across the assets — the
    hierarchy is semicolon-separated, the modifier code lists are comma-separated and
    fully quoted — and the encoding is not declared, so sniff both."""
    last: Exception | None = None
    for enc in ("utf-8-sig", "cp1252", "latin-1"):
        for sep in (";", ","):
            try:
                df = pd.read_csv(path, sep=sep, dtype=str, encoding=enc,
                                 keep_default_na=False, **kw)
            except UnicodeDecodeError as e:
                last = e
                break                       # encoding is wrong; try the next one
            except Exception as e:          # noqa: BLE001 — parser errors mean wrong sep
                last = e
                continue
            if df.shape[1] > 1:
                return df
    raise ValueError(f"could not parse {path.name} as CSV ({last})")


def _pad(code: str) -> str:
    """HISCO codes are 5 digits; HSN stores them numerically, so 0-11.10 arrives as
    '1110'. Sentinels (-1/-2/-3) are left alone."""
    code = str(code).strip()
    if not code or code.startswith("-"):
        return code
    return code.zfill(5)


def _to_num(s: pd.Series) -> pd.Series:
    """Scale column -> nullable numeric, sentinels to <NA>."""
    return pd.to_numeric(s.where(~s.isin(SCALE_SENTINELS)), errors="coerce").astype("Float64")


def load_hierarchy() -> pd.DataFrame:
    """HISCO micro code -> English label + its major/minor/unit ancestry."""
    h = _read_csv_any(HIER)
    out = pd.DataFrame({
        "hisco": h["hisco_full_id"].map(_pad),
        "hisco_pretty": h["hisco_full_id_pretty"].str.strip(),
        "label_en": h["label45"].str.strip(),
        "major": h["hisco_1_id"].str.strip(),
        "major_label": h["label1"].str.strip(),
        "minor": h["hisco_2_id"].str.strip(),
        "minor_label": h["label2"].str.strip(),
        "unit": h["hisco_3_id"].str.strip(),
        "unit_label": h["label3"].str.strip(),
    })
    dupes = out["hisco"].duplicated().sum()
    if dupes:
        print(f"  note: {dupes} duplicate hisco rows in the hierarchy — keeping the first")
        out = out.drop_duplicates("hisco")
    return out


def _ancestor_labels(hier: pd.DataFrame) -> dict[str, dict[str, str]]:
    """Prefix -> label at the unit / minor / major level.

    HSN sometimes codes at a level coarser than the micro group — ``39000`` (minor group
    3-90), ``02800`` (unit group 0-28), ``99999``. Those codes are valid HISCO but have no
    row in the 1,675-line micro hierarchy, so their label has to come from the ancestor.
    """
    return {
        "unit": dict(zip(hier["hisco"].str[:3], hier["unit_label"])),
        "minor": dict(zip(hier["hisco"].str[:2], hier["minor_label"])),
        "major": dict(zip(hier["hisco"].str[:1], hier["major_label"])),
    }


def load_code_lists() -> dict[str, dict[str, str]]:
    """status/relation/product code -> label, from the druid modifier code lists
    (``<name>_id, label, country_code, preferred``)."""
    lists: dict[str, dict[str, str]] = {}
    for name, path in CODE_LISTS.items():
        df = _read_csv_any(path)
        code_col = f"{name}_id" if f"{name}_id" in df.columns else df.columns[0]
        label_col = "label" if "label" in df.columns else df.columns[1]
        if "preferred" in df.columns:      # one preferred label per code
            df = df[df["preferred"].str.strip() == "1"]
        df = df.drop_duplicates(code_col)
        lists[name] = dict(zip(df[code_col].str.strip(), df[label_col].str.strip()))
        print(f"  {name:<9} {len(lists[name]):>3} codes ({code_col} -> {label_col})")
    return lists


def load_hsn_scales() -> pd.DataFrame:
    """(hisco, status, relation, product) -> HISCLASS / HISCLASS_5 / HISCAM / SOCPO.

    Asserts the functional dependency instead of assuming it: if a future release makes
    the key ambiguous, this raises rather than silently picking a class.
    """
    cols = ["HISCO", "STATUS", "RELATION", "PRODUCT",
            "HISCLASS", "HISCLASS_5", "HISCAM_U1", "SOCPO"]
    df = pd.read_csv(HSN, dtype=str, keep_default_na=False, usecols=cols)
    df["HISCO"] = df["HISCO"].map(_pad)

    key = ["HISCO", "STATUS", "RELATION", "PRODUCT"]
    scales = ["HISCLASS", "HISCLASS_5", "HISCAM_U1", "SOCPO"]
    ambiguous = df.groupby(key)[scales].nunique().gt(1).any(axis=1)
    if ambiguous.any():
        bad = ambiguous[ambiguous].index.tolist()[:10]
        raise AssertionError(
            f"{int(ambiguous.sum())} (hisco,status,relation,product) keys carry more than "
            f"one scale value — the crosswalk is no longer a function. First: {bad}"
        )

    n_titles = df.groupby(key).size().rename("n_hsn")
    out = df.drop_duplicates(key).set_index(key).join(n_titles).reset_index()
    out = out.rename(columns={
        "HISCO": "hisco", "STATUS": "status", "RELATION": "relation",
        "PRODUCT": "product", "HISCLASS": "hisclass", "HISCLASS_5": "hisclass_5",
        "HISCAM_U1": "hiscam_u1", "SOCPO": "socpo",
    })
    for c in ["hisclass", "hisclass_5", "hiscam_u1", "socpo"]:
        out[c] = _to_num(out[c])

    # The 12->5 aggregation is documented; check the file agrees with the documentation.
    both = out.dropna(subset=["hisclass", "hisclass_5"])
    expected = both["hisclass"].astype(int).map(HISCLASS_5_FROM_12)
    mismatch = int((expected != both["hisclass_5"].astype(int)).sum())
    if mismatch:
        raise AssertionError(
            f"{mismatch} rows where HISCLASS_5 disagrees with the documented "
            f"12->5 aggregation (§4.1)"
        )
    return out


def build_authority() -> pd.DataFrame:
    print("Reading inputs …")
    hier = load_hierarchy()
    print(f"  hierarchy {len(hier):,} HISCO micro codes")
    lists = load_code_lists()
    hsn = load_hsn_scales()
    print(f"  HSN       {len(hsn):,} (hisco,status,relation,product) keys "
          f"from {hsn['n_hsn'].sum():,} titles")

    # Codes the hierarchy knows but HSN never coded: keep them with default modifiers so
    # a Czech string can still reach a code + label (scales stay <NA>, honestly empty).
    coded = set(hsn["hisco"])
    uncoded = hier.loc[~hier["hisco"].isin(coded), ["hisco"]].copy()
    for c, v in [("status", "-9"), ("relation", "-9"), ("product", "-9")]:
        uncoded[c] = v
    for c in ["hisclass", "hisclass_5", "socpo"]:
        uncoded[c] = pd.Series(pd.NA, index=uncoded.index, dtype="Int64")
    uncoded["hiscam_u1"] = pd.Series(pd.NA, index=uncoded.index, dtype="Float64")
    uncoded["n_hsn"] = 0
    print(f"  {len(uncoded):,} hierarchy codes carry no HSN class (kept, scales empty)")

    df = pd.concat([hsn, uncoded], ignore_index=True)
    df["in_hsn"] = df["n_hsn"].gt(0)
    df = df.merge(hier, on="hisco", how="left")

    # HSN codes with no micro-hierarchy row: the sentinels (-1/-2/-3) plus codes HSN
    # placed at a coarser level. Label those from their ancestor rather than shipping a
    # blank, and record which level the label came from.
    anc = _ancestor_labels(hier)
    df["label_level"] = pd.NA
    df.loc[df["label_en"].notna() & (df["label_en"] != ""), "label_level"] = "micro"
    for level, width, label_col in [("unit", 3, "unit_label"),
                                    ("minor", 2, "minor_label"),
                                    ("major", 1, "major_label")]:
        gap = df["label_level"].isna() & ~df["hisco"].str.startswith("-")
        if not gap.any():
            break
        fill = df.loc[gap, "hisco"].str[:width].map(anc[level])
        hit = fill.notna()
        df.loc[gap & hit.reindex(df.index, fill_value=False), "label_en"] = fill[hit]
        df.loc[gap & hit.reindex(df.index, fill_value=False), label_col] = fill[hit]
        df.loc[gap & hit.reindex(df.index, fill_value=False), "label_level"] = level

    # The label is only half of what the ancestor supplies. Those same codes had blank
    # `major`/`minor`/`unit` because they carry no micro-hierarchy row to merge, which left
    # `occupation_hisco_major` <NA> for 553 records that do carry a HISCO code *and* a
    # HISCLASS — `brusič skla` -> 89125 alone is 304 of them. HISCO is positional: across
    # all 1,675 micro rows `major == hisco[0]`, `minor == hisco[1]` and `unit == hisco[2]`
    # exactly, so the ancestry is derivable from the code itself. Fill every level, not
    # just the one the label came from; sentinels (-1/-2/-3) are left alone.
    real = ~df["hisco"].str.startswith("-")
    blank = lambda c: df[c].isna() | df[c].astype("string").fillna("").str.strip().eq("")
    n_filled = 0
    for level, pos, width, code_col, label_col in [
        ("major", 0, 1, "major", "major_label"),
        ("minor", 1, 2, "minor", "minor_label"),
        ("unit", 2, 3, "unit", "unit_label"),
    ]:
        gap = real & blank(code_col)
        n_filled += int(gap.sum())
        df.loc[gap, code_col] = df.loc[gap, "hisco"].str[pos]
        lab_gap = real & blank(label_col)
        df.loc[lab_gap, label_col] = df.loc[lab_gap, "hisco"].str[:width].map(anc[level])
    if n_filled:
        print(f"  {n_filled:,} ancestry cells derived from the code's own digits")

    coarse = df.loc[df["label_level"].isin(["unit", "minor", "major"]), "hisco"].unique()
    if len(coarse):
        print(f"  {len(coarse)} codes labelled from an ancestor group "
              f"(coarser than micro): {sorted(coarse)[:12]}")
    blank = sorted(df.loc[df["label_level"].isna(), "hisco"].unique())
    if blank:
        print(f"  {len(blank)} codes with no label at all: {blank}")

    # Labels
    df["hisclass_label"] = df["hisclass"].map(
        lambda v: HISCLASS_LABELS.get(int(v)) if pd.notna(v) else pd.NA).astype("string")
    df["hisclass_5_label"] = df["hisclass_5"].map(
        lambda v: HISCLASS_5_LABELS.get(int(v)) if pd.notna(v) else pd.NA).astype("string")
    df["socpo_label"] = df["socpo"].map(
        lambda v: SOCPO_LABELS.get(int(v)) if pd.notna(v) else pd.NA).astype("string")
    for name in CODE_LISTS:
        df[f"{name}_label"] = df[name].map(lists[name]).astype("string")

    df["hisclass"] = df["hisclass"].astype("Int64")
    df["hisclass_5"] = df["hisclass_5"].astype("Int64")
    df["socpo"] = df["socpo"].astype("Int64")

    order = ["hisco", "status", "relation", "product",
             "label_en", "label_level", "hisco_pretty",
             "major", "major_label", "minor", "minor_label", "unit", "unit_label",
             "hisclass", "hisclass_label", "hisclass_5", "hisclass_5_label",
             "hiscam_u1", "socpo", "socpo_label",
             "status_label", "relation_label", "product_label",
             "n_hsn", "in_hsn"]
    return df[order].sort_values(["hisco", "status", "relation", "product"],
                                 ignore_index=True)


def authority_report(df: pd.DataFrame) -> None:
    print("\nAuthority table")
    print(f"  rows                    {len(df):,}")
    print(f"  distinct HISCO codes    {df['hisco'].nunique():,}")
    print(f"  default-modifier rows   {int(((df[['status','relation','product']] == '-9').all(axis=1)).sum()):,}")
    print(f"  with an English label   {int(df['label_en'].notna().sum()):,} "
          f"({int((df['label_level'] == 'micro').sum()):,} at micro level)")
    print(f"  with a HISCLASS         {int(df['hisclass'].notna().sum()):,}")
    print(f"  with a HISCAM           {int(df['hiscam_u1'].notna().sum()):,}")
    print("\n  HISCLASS distribution (rows, not people):")
    vc = df["hisclass"].value_counts().sort_index()
    for k, v in vc.items():
        print(f"    {int(k):>3}  {v:>6,}  {HISCLASS_LABELS.get(int(k), '?')}")

    bad = df["hisclass"].dropna()
    assert bad.between(1, 13).all(), "HISCLASS outside 1..13"
    h5 = df["hisclass_5"].dropna()
    assert h5.between(1, 5).all(), "HISCLASS_5 outside 1..5"
    assert not df.duplicated(["hisco", "status", "relation", "product"]).any(), \
        "duplicate (hisco,status,relation,product) rows"
    print("\n  assertions passed: HISCLASS 1..13, HISCLASS_5 1..5, key unique")


def main() -> int:
    _require_inputs()
    df = build_authority()
    authority_report(df)
    OUT.parent.mkdir(parents=True, exist_ok=True)
    df.to_csv(OUT, index=False, encoding="utf-8")
    print(f"\nWrote {OUT.relative_to(ROOT)}  ({OUT.stat().st_size:,} B)")
    return 0


if __name__ == "__main__":
    sys.exit(main())
