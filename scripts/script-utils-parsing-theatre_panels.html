TL;DR

The theatre runner (utils/reference_tools/places/run_amateur_theatre_pilot.py, about eight minutes, the model pass's readings applied) builds the measure in data/interim/theatre/. This module copies the tables the analysis can use into data/final/theatre/ — nothing is recomputed — after checking each one; notebook 02 §3f calls promote, and utils/validation/data_checks.py runs the same checks on what was written.

How it works

Six tables, each on its own declared key, at its own level (kept normalised: none is merged into the person files, which happens at the estimation step):

========================== ============================== ================================= table key joins a person on ========================== ============================== ================================= theatre_places geo_id — (the database's place → its 1910 settlement, court and okres) theatre_place_year geo_id × year — (the base every level is summed from) theatre_settlement_year settlement_id × year birth_settlement_id (§4a) theatre_court_year court_id × year birth_court_id (1910 courts only: amateur_theatre 0.13.1 folds the two post-1910 courts into theirs, D-255) theatre_okres_year okres_id × year birth_okres_id theatre_census_okres_year census_okres_id × year birth_census_okres_id (the census baseline unit, D-18) ========================== ============================== =================================

Every panel is balanced on 1780–1913 (amateur_theatre.PANEL_FIRST_YEAR / PANEL_LAST_YEAR). The evidence and clause tables stay in data/interim/theatre/: they are the measure's audit trail, not analysis data. Parquet only: the settlement panel alone is 2.3 million rows.

manifest.json beside the tables records when the runner built them and with which code, so the gate can tell a theatre build older than the 1910 tables it was tied to.

def load(src: Path = INTERIM) -> dict[str, pd.DataFrame]:
    """The six tables as the runner wrote them, ``theatre_places`` with integer unit ids."""
    missing = [f for f, _ in TABLES.values() if not (Path(src) / f).is_file()]
    if missing:
        raise FileNotFoundError(f"{src} lacks {missing}: run "
                                "utils/reference_tools/places/run_amateur_theatre_pilot.py --all first")
    out = {}
    for name, (file, _) in TABLES.items():
        path = Path(src) / file
        if path.suffix == ".csv":
            d = pd.read_csv(path, dtype={"settlement_id": "string"})
            for col in ("okres_id", "court_id"):
                d[col] = d[col].astype("Int64")
        else:
            d = pd.read_parquet(path)
        out[name] = d
    return out

def checks(tables: dict[str, pd.DataFrame], final: Path = FINAL,
           raw_places: Path | None = RAW_PLACES) -> list[tuple[str, bool, str, bool]]:
    """Every invariant of the theatre tables, as ``(label, ok, detail_if_failed, hard)``.

    Keys unique and never missing; every panel balanced on the window; certain never above
    possible; every unit id a real 1910 unit (settlements_1910, the 1910 courts, the census okresy,
    pooled and not); the places table and the place panel on the same places; and — where the
    scraper's place list is at hand — a build on every place, not a pilot's three okresy.
    """
    first, last = _years()
    n_years = last - first + 1
    out = []
    for name, (_, key) in TABLES.items():
        d = tables[name]
        na, dup = int(d[key].isna().any(axis=1).sum()), int(d.duplicated(key).sum())
        out.append((f"{name}: key {key} unique and never missing", na == 0 and dup == 0,
                    f"{na} row(s) with a missing key, {dup} duplicate key(s)", True))
    for name in PANELS:
        d, unit = tables[name], TABLES[name][1][0]
        span = (int(d["year"].min()), int(d["year"].max())) if len(d) else (None, None)
        per_unit = d.groupby(unit)["year"].nunique()
        out.append((f"{name}: balanced, every unit in every year {first}–{last}",
                    span == (first, last) and bool((per_unit == n_years).all()),
                    f"years {span}; {int((per_unit != n_years).sum())} unit(s) lack a year", True))
        over = {b: int((d[f"{b}_certain"] > d[f"{b}_possible"]).sum())
                for b in CERTAIN_WITHIN_POSSIBLE if f"{b}_certain" in d.columns}
        out.append((f"{name}: certain never above possible", not any(over.values()),
                    f"rows with certain > possible: {over}", True))
    out += _unit_checks(tables, final)
    places, panel = set(tables["theatre_places"]["geo_id"]), set(tables["theatre_place_year"]["geo_id"])
    out.append(("theatre_places and theatre_place_year hold the same places", places == panel,
                f"{len(places - panel)} place(s) only in the places table, {len(panel - places)} only in the panel",
                True))
    if raw_places is not None and Path(raw_places).is_file():
        scraped = set(pd.read_csv(raw_places, usecols=["geo_id"])["geo_id"])
        out.append((f"Theatre tables built on every scraped place ({len(scraped):,})", panel == scraped,
                    f"{len(scraped - panel):,} scraped place(s) missing — a pilot build (--okres); "
                    "run the theatre runner with --all", True))
    return out

def built_at(src: Path = INTERIM) -> str:
    """When the runner last wrote the tables: the newest of their files, ISO, to the second."""
    stamp = max((Path(src) / f).stat().st_mtime for f, _ in TABLES.values())
    return _dt.datetime.fromtimestamp(stamp).isoformat(timespec="seconds")

def promote(src: Path = INTERIM, dst: Path = FINAL_DIR, final: Path = FINAL,
            raw_places: Path | None = RAW_PLACES) -> dict[str, pd.DataFrame]:
    """Check the runner's tables and copy them to ``dst`` as parquet, with ``manifest.json``.

    Raises ``ValueError`` listing every broken invariant, before anything is written, so a pilot
    build or a table tied to an older 1910 frame never reaches data/final.
    """
    from utils.parsing import amateur_theatre as at
    from utils.parsing import amateur_theatre_model as atm

    tables = load(src)
    broken = [f"{label}: {detail}" for label, ok, detail, hard in checks(tables, final, raw_places)
              if hard and not ok]
    if broken:
        raise ValueError("theatre tables not promoted:\n  " + "\n  ".join(broken))
    manifest = {"built": built_at(src), "promoted": _dt.datetime.now().isoformat(timespec="seconds"),
                "theatre_panels": __version__, "amateur_theatre": at.__version__,
                "amateur_theatre_model": atm.__version__,
                "rows": {name: len(d) for name, d in tables.items()}}
    # written beside dst and swapped in whole, so a write that fails part-way leaves the old build
    dst = Path(dst)
    staging, old = dst.with_name(dst.name + ".new"), dst.with_name(dst.name + ".old")
    for leftover in (staging, old):
        shutil.rmtree(leftover, ignore_errors=True)
    staging.mkdir(parents=True)
    for name, d in tables.items():
        d.to_parquet(staging / f"{name}.parquet", index=False)
    (staging / MANIFEST).write_text(json.dumps(manifest, indent=1), encoding="utf-8")
    if dst.exists():
        _rename(dst, old)
    try:
        _rename(staging, dst)
    except OSError:
        if old.exists() and not dst.exists():
            _rename(old, dst)            # put the previous build back; the new one stays in staging
        raise
    shutil.rmtree(old, ignore_errors=True)
    return tables

def panels_report(tables: dict[str, pd.DataFrame]) -> pd.DataFrame:
    """One row per table: rows, units, and the place-years (unit-years) with certain activity."""
    rows = []
    for name, (_, key) in TABLES.items():
        d = tables[name]
        row = {"table": name, "key": " × ".join(key), "rows": len(d), "units": d[key[0]].nunique()}
        if "any_activity_certain" in d.columns:
            row["unit_years_certain"] = int((d["any_activity_certain"] > 0).sum())
            row["unit_years_possible"] = int((d["any_activity_possible"] > 0).sum())
        rows.append(row)
    report = pd.DataFrame(rows)
    print(report.to_string(index=False))
    return report

"""
theatre_panels.py — the amateur-theatre measure, promoted to ``data/final`` (D-255)
==================================================================================

The theatre runner (``utils/reference_tools/places/run_amateur_theatre_pilot.py``, about eight
minutes, the model pass's readings applied) builds the measure in ``data/interim/theatre/``. This
module copies the tables the analysis can use into ``data/final/theatre/`` — **nothing is
recomputed** — after checking each one; notebook 02 §3f calls :func:`promote`, and
``utils/validation/data_checks.py`` runs the same :func:`checks` on what was written.

Six tables, each on its own declared key, at its own level (kept normalised: none is merged into
the person files, which happens at the estimation step):

==========================  ==============================  =================================
table                       key                             joins a person on
==========================  ==============================  =================================
``theatre_places``          ``geo_id``                      — (the database's place → its 1910
                                                            settlement, court and okres)
``theatre_place_year``      ``geo_id`` × ``year``           — (the base every level is summed from)
``theatre_settlement_year`` ``settlement_id`` × ``year``    ``birth_settlement_id`` (§4a)
``theatre_court_year``      ``court_id`` × ``year``         ``birth_court_id`` (1910 courts only:
                                                            ``amateur_theatre`` 0.13.1 folds the two
                                                            post-1910 courts into theirs, D-255)
``theatre_okres_year``      ``okres_id`` × ``year``         ``birth_okres_id``
``theatre_census_okres_year`` ``census_okres_id`` × ``year`` ``birth_census_okres_id`` (the
                                                            census baseline unit, D-18)
==========================  ==============================  =================================

Every panel is balanced on 1780–1913 (``amateur_theatre.PANEL_FIRST_YEAR`` /
``PANEL_LAST_YEAR``). The evidence and clause tables stay in ``data/interim/theatre/``: they are
the measure's audit trail, not analysis data. Parquet only: the settlement panel alone is
2.3 million rows.

``manifest.json`` beside the tables records when the runner built them and with which code, so
the gate can tell a theatre build older than the 1910 tables it was tied to.

INTERIM = ROOT / "data" / "interim" / "theatre"
FINAL = ROOT / "data" / "final"

def _unit_checks(tables: dict[str, pd.DataFrame], final: Path) -> list[tuple[str, bool, str, bool]]:
    """Every unit id in the panels and the places table is a 1910 unit of data/final."""
    final = Path(final)
    known = {
        "settlement_id": (set(pd.read_parquet(final / "geo" / "settlements_1910.parquet",
                                              columns=["settlement_id"])["settlement_id"]), "settlements_1910"),
        # a 1910 court of the backbone: a person's birth_court_id can name nothing else, so a
        # register court set up after 1910 (Slavonice, Třešť, D-205) here would reach no person
        "court_id": (_ids(pd.read_parquet(final / "geo" / "districts_1910.parquet",
                                          columns=["court_id"])["court_id"]), "districts_1910"),
        "okres_id": (_ids(pd.read_parquet(final / "census" / "census_okres_1910.parquet",
                                          columns=["okres_id"])["okres_id"]), "census_okres_1910"),
        "census_okres_id": (_ids(pd.read_parquet(final / "census" / "census_okres_1910_pooled.parquet",
                                                 columns=["census_okres_id"])["census_okres_id"]),
                            "census_okres_1910_pooled"),
    }
    out = []
    for name in ["theatre_places", *PANELS[1:]]:
        d = tables[name]
        for col, (ids, table) in known.items():
            if col not in d.columns:
                continue
            have = set(d[col].dropna()) if col == "settlement_id" else _ids(d[col])
            stray = have - ids
            out.append((f"{name}: every {col} is in {table}", not stray,
                        f"{len(stray)} unknown id(s), e.g. {sorted(stray)[:5]}", True))
    return out


def built_at(src: Path = INTERIM) -> str:
    """When the runner last wrote the tables: the newest of their files, ISO, to the second."""
    stamp = max((Path(src) / f).stat().st_mtime for f, _ in TABLES.values())
    return _dt.datetime.fromtimestamp(stamp).isoformat(timespec="seconds")


def promote(src: Path = INTERIM, dst: Path = FINAL_DIR, final: Path = FINAL,
            raw_places: Path | None = RAW_PLACES) -> dict[str, pd.DataFrame]:
    """Check the runner's tables and copy them to ``dst`` as parquet, with ``manifest.json``.

    Raises ``ValueError`` listing every broken invariant, before anything is written, so a pilot
    build or a table tied to an older 1910 frame never reaches data/final.
    """
    from utils.parsing import amateur_theatre as at
    from utils.parsing import amateur_theatre_model as atm

    tables = load(src)
    broken = [f"{label}: {detail}" for label, ok, detail, hard in checks(tables, final, raw_places)
              if hard and not ok]
    if broken:
        raise ValueError("theatre tables not promoted:\n  " + "\n  ".join(broken))
    manifest = {"built": built_at(src), "promoted": _dt.datetime.now().isoformat(timespec="seconds"),
                "theatre_panels": __version__, "amateur_theatre": at.__version__,
                "amateur_theatre_model": atm.__version__,
                "rows": {name: len(d) for name, d in tables.items()}}
    # written beside dst and swapped in whole, so a write that fails part-way leaves the old build
    dst = Path(dst)
    staging, old = dst.with_name(dst.name + ".new"), dst.with_name(dst.name + ".old")
    for leftover in (staging, old):
        shutil.rmtree(leftover, ignore_errors=True)
    staging.mkdir(parents=True)
    for name, d in tables.items():
        d.to_parquet(staging / f"{name}.parquet", index=False)
    (staging / MANIFEST).write_text(json.dumps(manifest, indent=1), encoding="utf-8")
    if dst.exists():
        _rename(dst, old)
    try:
        _rename(staging, dst)
    except OSError:
        if old.exists() and not dst.exists():
            _rename(old, dst)            # put the previous build back; the new one stays in staging
        raise
    shutil.rmtree(old, ignore_errors=True)
    return tables


def _rename(src: Path, dst: Path, tries: int = 10, wait: float = 1.0) -> None:
    """``src.rename(dst)``, retried: Dropbox briefly locks a folder it is syncing (WinError 5)."""
    import time
    for attempt in range(tries):
        try:
            src.rename(dst)
            return
        except PermissionError:
            if attempt == tries - 1:
                raise
            time.sleep(wait)


def panels_report(tables: dict[str, pd.DataFrame]) -> pd.DataFrame:
    """One row per table: rows, units, and the place-years (unit-years) with certain activity."""
    rows = []
    for name, (_, key) in TABLES.items():
        d = tables[name]
        row = {"table": name, "key": " × ".join(key), "rows": len(d), "units": d[key[0]].nunique()}
        if "any_activity_certain" in d.columns:
            row["unit_years_certain"] = int((d["any_activity_certain"] > 0).sum())
            row["unit_years_possible"] = int((d["any_activity_possible"] > 0).sum())
        rows.append(row)
    report = pd.DataFrame(rows)
    print(report.to_string(index=False))
    return report
