How it works

Source: GEONAMES-5514-gpkg-&lt;date&gt;.zip from the ČÚZK Atom feed https://atom.cuzk.gov.cz/GEONAMES-GPKG/GEONAMES-GPKG.xml — one GeoPackage for the whole state, S-JTSK (EPSG:5514), open data under CC-BY 4.0 (the feed's rights: žádné podmínky neplatí).

Read with the standard-library sqlite3 (a GeoPackage is an SQLite file); the point geometry is a GeoPackage blob — an 8-byte GP header, an optional envelope, then ISO WKB — decoded by gpkg_point. Only the settlement-like layers are kept (LAYERS); rivers, hills, forests and roads are not places a person was born in.

Output: geonames_places [geonames_id] — name, layer, kind (druh), the modern obec and okres it is filed under (parsed from lokalizace), S-JTSK x/y and WGS 84 lat/lon.

def gpkg_point(blob: bytes) -> tuple[float, float]:
    """Decode a GeoPackage point blob → ``(x, y)``; ``(nan, nan)`` for an empty geometry."""
    if blob is None or blob[:2] != b"GP":
        return (np.nan, np.nan)
    flags = blob[3]
    env = (flags >> 1) & 0b111                 # envelope indicator: 0, 1 (4 dbl), 2/3 (6), 4 (8)
    env_len = {0: 0, 1: 32, 2: 48, 3: 48, 4: 64}[env]
    wkb = blob[8 + env_len:]
    order = "<" if wkb[0] == 1 else ">"
    gtype = struct.unpack(order + "I", wkb[1:5])[0]
    if gtype % 1000 != 1:                      # 1 = Point (also 1001/2001/3001 with Z/M)
        return (np.nan, np.nan)
    x, y = struct.unpack(order + "dd", wkb[5:21])
    return (x, y)

def read_geonames(gpkg_path, layers: dict[str, str] = LAYERS) -> pd.DataFrame:
    """Read the settlement-like layers of the Geonames GeoPackage (or its zip) into one table."""
    if str(gpkg_path).endswith(".zip"):
        gpkg_path = _gpkg_from_zip(gpkg_path)
    con = sqlite3.connect(f"file:{Path(gpkg_path).as_posix()}?mode=ro", uri=True)
    frames = []
    try:
        for table, layer in layers.items():
            rows = con.execute(
                f'SELECT OBJECTID, fid_jmeno, jmeno, druh, lokalizace, Shape FROM "{table}"'
            ).fetchall()
            df = pd.DataFrame(rows, columns=["objectid", "fid_jmeno", "name", "kind",
                                             "lokalizace", "shape"])
            df.insert(0, "layer", layer)
            frames.append(df)
    finally:
        con.close()
    out = pd.concat(frames, ignore_index=True)
    out["name"] = out["name"].str.strip()
    xy = np.array([gpkg_point(b) for b in out.pop("shape")])
    out["x"], out["y"] = xy[:, 0], xy[:, 1]
    loc = out["lokalizace"].map(_split_lokalizace)
    out["obec_name"] = [a for a, _b in loc]
    out["okres_name"] = [b for _a, b in loc]
    lon, lat = ru.sjtsk_to_wgs84(out["x"].to_numpy(), out["y"].to_numpy())
    out["lat"], out["lon"] = lat, lon
    # OBJECTID is unique within a layer only; the name feature id repeats across a feature's
    # several points (42 MistniCasti rows). The layer-qualified OBJECTID is the key.
    out.insert(0, "geonames_id", out["layer"] + ":" + out["objectid"].astype(str))
    assert out["geonames_id"].is_unique
    return out.drop(columns=["objectid"])

"""
geonames.py — named places from the ČÚZK Geonames database (Databáze geografických jmen ČR)
===========================================================================================
The RÚIAN tables (:mod:`utils.parsing.ruian`) stop at the official část obce and the ZSJ. The
1910 register's smallest parts — a mill, a gamekeeper's lodge, a solitary farm, a glassworks —
are below that, and many of them survive only as a **name on the map**. ČÚZK's Geonames
database is that map's name layer: every named settlement part (*místní část*), down to the
single house, with a point, maintained from ZABAGED.

Source: ``GEONAMES-5514-gpkg-<date>.zip`` from the ČÚZK Atom feed
https://atom.cuzk.gov.cz/GEONAMES-GPKG/GEONAMES-GPKG.xml — one GeoPackage for the whole state,
S-JTSK (EPSG:5514), open data under CC-BY 4.0 (the feed's ``rights``: *žádné podmínky neplatí*).

Read with the standard-library ``sqlite3`` (a GeoPackage is an SQLite file); the point geometry
is a GeoPackage blob — an 8-byte ``GP`` header, an optional envelope, then ISO WKB — decoded
by :func:`gpkg_point`. Only the settlement-like layers are kept (:data:`LAYERS`); rivers,
hills, forests and roads are not places a person was born in.

Output: ``geonames_places`` [``geonames_id``] — name, layer, kind (``druh``), the modern obec and
okres it is filed under (parsed from ``lokalizace``), S-JTSK x/y and WGS 84 lat/lon.
"""

from __future__ import annotations

import sqlite3
import struct
from pathlib import Path

import numpy as np
import pandas as pd

from utils.parsing import ruian as ru

__version__ = "1.0.0"   # first build (2026-10-29)

# Settlement-like layers, and why each is in. A 1910 *Bestandteil* can be any of these.
LAYERS: dict[str, str] = {
    "MestaObce": "obec",                        # municipalities (6,265)
    "CastiMestObci": "cast_obce",               # official parts (15,126)
    "MistniCasti": "mistni_cast",               # named local parts: mills, lodges, solitary houses
    "HradyZamkyTvrze": "castle",                # Schloss / Burg / Meierhof seats
    "VyrobniATezebniZarizeni": "works",         # mines, mills, breweries, glassworks
    "UbytovaciAStravovaciZarizeni": "lodge",    # mountain boudy — inhabited in 1910
}


def gpkg_point(blob: bytes) -> tuple[float, float]:
    """Decode a GeoPackage point blob → ``(x, y)``; ``(nan, nan)`` for an empty geometry."""
    if blob is None or blob[:2] != b"GP":
        return (np.nan, np.nan)
    flags = blob[3]
    env = (flags >> 1) & 0b111                 # envelope indicator: 0, 1 (4 dbl), 2/3 (6), 4 (8)
    env_len = {0: 0, 1: 32, 2: 48, 3: 48, 4: 64}[env]
    wkb = blob[8 + env_len:]
    order = "<" if wkb[0] == 1 else ">"
    gtype = struct.unpack(order + "I", wkb[1:5])[0]
    if gtype % 1000 != 1:                      # 1 = Point (also 1001/2001/3001 with Z/M)
        return (np.nan, np.nan)
    x, y = struct.unpack(order + "dd", wkb[5:21])
    return (x, y)


def _split_lokalizace(s: str) -> tuple[str | None, str | None]:
    """``'Žihle; okr. Plzeň-sever; Plzeňský kraj'`` → ``('Žihle', 'Plzeň-sever')``."""
    if not isinstance(s, str):
        return (None, None)
    parts = [p.strip() for p in s.split(";")]
    obec = parts[0] or None
    okres = next((p[5:].strip() for p in parts if p.startswith("okr.")), None)
    if okres is None and obec == "Praha":          # the capital is in no okres; RÚIAN's label
        okres = "Hlavní město Praha"
    return (obec, okres)


def _gpkg_from_zip(zip_path) -> Path:
    """The GeoPackage inside the downloaded zip, unpacked beside it on first use (SQLite needs a
    file on disk). The zip is the source; the unpacked file is a cache."""
    import zipfile

    zip_path = Path(zip_path)
    with zipfile.ZipFile(zip_path) as zf:
        member = next(n for n in zf.namelist() if n.endswith(".gpkg"))
        target = zip_path.parent / Path(member).name
        if not target.exists() or target.stat().st_size != zf.getinfo(member).file_size:
            zf.extract(member, zip_path.parent)
    return target
