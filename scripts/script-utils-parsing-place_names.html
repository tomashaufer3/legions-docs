How it works

  • a stray space inside a word — the headword ÚSTÍ NAD ORL ICÍ, the 1910 name Vykýřovic e, the German Po mmerndorf, the parent Te tín;
  • a status word leaking into the name — the entry reads pod názvem Kosmanosy o bec v okr. …; the parser's pod názvem (.+?) (obec|osada|část|město|v okr.) cannot see the split o bec, so the name runs on to v okr. and comes out Kosmanosy o bec, and the status test \b(obec|město)\b fails too, leaving status_1910 empty. The same happens with a hamlet's osad a obce Hodkov (name, status and parent in one cell, parent_1910 empty);
  • a též alternate the parser did not split — t. printed t . (Chrastice t . Krastice) and every parent_1910 of the form Svrčeves t. Cvrčoves (the parser splits the alternate off name_1910 only).

Healing is evidence-led, never a guess:

  • a status phrase is removed only at the end of a name (&lt;name&gt; &lt;status&gt; or &lt;name&gt; &lt;osada|část&gt; obce &lt;parent&gt;), split or not — the lexicon's own grammar, which says what the removed words were, so the recovered status and parent are kept with the heal;
  • a t. alternate is split off at the lexicon's own t. marker (the name keeps the first form, the rest is kept as alt);
  • a space-joined form is accepted only when it equals an official RÚIAN name (obec, část obce, katastrální území or ZSJ, data/final/geo/ruian_*) or a name the 1910 register prints in the same okres — and the unjoined form equals neither. Two joinable forms that both have evidence leave the name alone. Only a space between two letters is ever removed, so 1. díl and Děčín VII -Chrochvice (spacing conventions, not defects) are never touched.

Nothing here changes settlement_id or settlements_1910: heal_frame returns one row per healed cell (settlement_id, column, original, healed, recovered status/parent/alt, rule, evidence) and name_index the long table of every name form a matcher may use, healed forms carrying the same settlement_id.

def name_key(text) -> str:
    """Comparison key for evidence: NFC, case-folded, whitespace collapsed; accents kept."""
    text = unicodedata.normalize("NFC", str(text or ""))
    return re.sub(r"\s+", " ", text).strip().casefold()

def strip_status_tail(text: str) -> tuple[str, str | None, str | None]:
    """``(name, status, parent)`` with a trailing status phrase removed; unchanged if none.

    ``Kosmanosy o bec`` → ``("Kosmanosy", "obec", None)``; ``Želívec osad a obce Hodkov`` →
    ``("Želívec", "osada", "Hodkov")``. A word in the middle of a name is never touched.
    """
    m = _STATUS_TAIL_RE.match(str(text or ""))
    if not m:
        return str(text or ""), None, None
    word = next(w for i, w in enumerate(STATUS_WORDS) if m.group(f"w{i}"))
    parent = m.group("parent")
    if parent is not None and STATUS_WORDS[word] not in ("osada", "cast"):
        return str(text or ""), None, None          # only a dependent place names a parent
    return m.group("name").strip(), STATUS_WORDS[word], (parent.strip() if parent else None)

def split_alternate(text: str) -> tuple[str, str | None]:

def joined_forms(text: str, max_spaces: int = MAX_SPACES) -> list[str]:
    """Every form of ``text`` with one or more letter-to-letter spaces removed."""
    text = str(text or "")
    capitals = text == text.upper()
    spaces = [m.start() for m in _LETTER_SPACE_RE.finditer(text)
              if capitals or text[m.start() + 1].islower()]
    if not spaces or len(spaces) > max_spaces:
        return []
    out = []
    for drop in itertools.product((False, True), repeat=len(spaces)):
        if not any(drop):
            continue
        cut = {pos for pos, d in zip(spaces, drop) if d}
        out.append("".join(c for i, c in enumerate(text) if i not in cut))
    return out

def join_spaces(text: str, official: set[str] | frozenset = frozenset(),
                register: set[str] | frozenset = frozenset()) -> tuple[str, str]:
    """``(name, evidence)``: the unique joined form an evidence set holds, else ``(text, "")``.

    ``official`` and ``register`` hold :func:`name_key` keys. Nothing is joined when the text as
    written is already in either set, or when two joined forms are. An all-capitals headword
    (whose spaces carry no case clue) is joined on RÚIAN evidence alone.
    """
    key = name_key(text)
    if key in official or key in register:
        return text, ""
    if text == text.upper():
        register = frozenset()     # a capitals headword may join any space: RÚIAN evidence only
    hits = []
    for form in joined_forms(text):
        fk = name_key(form)
        if fk in official:
            hits.append((form, "ruian"))
        elif fk in register:
            hits.append((form, "register"))
    if len({name_key(f) for f, _ in hits}) != 1:
        return text, ""
    return hits[0]

def heal_name(text, official: set[str] | frozenset = frozenset(),
              register: set[str] | frozenset = frozenset()) -> Healed:
    """Heal one lexicon name cell; see the module docstring for the rules.

    Order: the status tail first (it is what blocks the other two), then the ``t.`` alternate,
    then the evidence-led space join on the name that is left (the alternate is joined too).
    """
    raw = "" if text is None or (isinstance(text, float) and pd.isna(text)) else str(text)
    rules, evidence = [], []
    name, status, parent = strip_status_tail(raw)
    if status is not None:
        rules.append("status_tail")
    name, alt = split_alternate(name)
    if alt is not None:
        rules.append("alt_split")
    joined, why = join_spaces(name, official, register)
    if why:
        rules.append("space_join")
        evidence.append(f"{why}:{joined}")
        name = joined
    if alt is not None:
        alt_joined, alt_why = join_spaces(alt, official, register)
        if alt_why:
            rules.append("alt_space_join")
            evidence.append(f"{alt_why}:{alt_joined}")
            alt = alt_joined
    if parent is not None:
        parent, _ = split_alternate(parent)
    return Healed(name=name, status=status, parent=parent, alt=alt,
                  rule="+".join(rules), evidence="; ".join(evidence))

def official_names(geo_dir) -> frozenset:

    keys: set[str] = set()
    for table, col in RUIAN_NAME_COLUMNS.items():

def heal_frame(settlements: pd.DataFrame, official: frozenset = frozenset(),
               register: dict[int, frozenset] | None = None,
               columns=NAME_COLUMNS) -> pd.DataFrame:
    """One row per healed cell of ``settlements`` (``settlements_1910``); nothing else changes.

    Columns: ``settlement_id, okres_id, column, original, healed, status_recovered,
    parent_recovered, alt, rule, evidence``. A register form counts as evidence only in the
    settlement's own okres.
    """
    register = register or {}
    rows = []
    for row in settlements.itertuples(index=False):
        okres = getattr(row, "okres_id")
        reg = register.get(int(okres), frozenset()) if pd.notna(okres) else frozenset()
        for col in columns:
            value = getattr(row, col)
            if not isinstance(value, str) or not value.strip():
                continue
            h = heal_name(value, official, reg)
            if not h.changed:
                continue
            rows.append({"settlement_id": row.settlement_id, "okres_id": okres, "column": col,
                         "original": value, "healed": h.name, "status_recovered": h.status,
                         "parent_recovered": h.parent, "alt": h.alt, "rule": h.rule,
                         "evidence": h.evidence})
    cols = ["settlement_id", "okres_id", "column", "original", "healed", "status_recovered",
            "parent_recovered", "alt", "rule", "evidence"]
    return pd.DataFrame(rows, columns=cols)

def unhealed_splits(settlements: pd.DataFrame, heals: pd.DataFrame,
                    columns=NAME_COLUMNS) -> pd.DataFrame:
    """Cells that *look* split (a 1–2 letter fragment glued to nothing) but were left alone.

    Reported, never changed: no evidence named the joined form. ``u``, ``v``, ``na``, ``nad`` …
    and Roman numerals are words, not fragments.
    """
    words = {"za", "x", "xv", "xx", "u", "v", "na", "nad", "pod", "při", "pri", "an", "am", "im", "in", "bei", "ob",
             "a", "i", "z", "ve", "ze", "se", "od", "do", "dil", "díl", "st", "sv", "t", "b",
             "ii", "iii", "iv", "vi", "vii", "viii", "ix", "xi", "xii", "der", "die", "zu"}
    done = set(zip(heals["settlement_id"], heals["column"]))
    rows = []
    for row in settlements.itertuples(index=False):
        for col in columns:
            value = getattr(row, col)
            if not isinstance(value, str) or (row.settlement_id, col) in done:
                continue
            frags = [t for t in re.findall(r"[^\W\d_]+", value)
                     if len(t) <= 2 and t.casefold() not in words]
            if frags and _LETTER_SPACE_RE.search(value):
                rows.append({"settlement_id": row.settlement_id, "column": col, "value": value,
                             "fragments": " ".join(frags)})
    return pd.DataFrame(rows, columns=["settlement_id", "column", "value", "fragments"])

def healed_attributes(settlements: pd.DataFrame, heals: pd.DataFrame) -> pd.DataFrame:
    """``settlements`` with its name cells, empty status and empty parent healed — an overlay copy.

    ``settlement_id`` and row order are untouched; a recovered status or parent fills only an
    empty cell; ``healed_rule`` lists the rules that touched the row.
    """
    out = settlements.copy()
    pos = {sid: i for i, sid in enumerate(out["settlement_id"])}
    rule = [""] * len(out)
    for h in heals.itertuples(index=False):
        i = pos[h.settlement_id]
        out.iat[i, out.columns.get_loc(h.column)] = h.healed
        if h.column in ("name_1910", "de_name"):
            if h.status_recovered and pd.isna(out.iat[i, out.columns.get_loc("status_1910")]):
                out.iat[i, out.columns.get_loc("status_1910")] = h.status_recovered
            if h.parent_recovered and pd.isna(out.iat[i, out.columns.get_loc("parent_1910")]):
                out.iat[i, out.columns.get_loc("parent_1910")] = h.parent_recovered
        rule[i] = "+".join(filter(None, [rule[i], f"{h.column}:{h.rule}"]))

"""place_names.py — heal the ČSÚ lexicon's text-layer defects in settlement names, as an overlay
===============================================================================================
``settlements_1910`` is parsed (:mod:`utils.parsing.lexikon`) from the text layer of the ČSÚ
*Historický lexikon obcí 1869–2011*, alphabetical part. The PDF is modern and was not OCR'd, but its
text layer still breaks words, and the parse keeps the breaks:

* **a stray space inside a word** — the headword ``ÚSTÍ NAD ORL ICÍ``, the 1910 name ``Vykýřovic e``,
  the German ``Po mmerndorf``, the parent ``Te tín``;
* **a status word leaking into the name** — the entry reads ``pod názvem Kosmanosy o bec v okr. …``;
  the parser's ``pod názvem (.+?) (obec|osada|část|město|v okr.)`` cannot see the split ``o bec``, so
  the name runs on to ``v okr.`` and comes out ``Kosmanosy o bec``, and the status test
  ``\\b(obec|město)\\b`` fails too, leaving ``status_1910`` empty. The same happens with a hamlet's
  ``osad a obce Hodkov`` (name, status *and* parent in one cell, ``parent_1910`` empty);
* **a ``též`` alternate the parser did not split** — ``t.`` printed ``t .`` (``Chrastice t . Krastice``)
  and every ``parent_1910`` of the form ``Svrčeves t. Cvrčoves`` (the parser splits the alternate off
  ``name_1910`` only).

Healing is **evidence-led, never a guess**:

* a status phrase is removed only at the **end** of a name (``<name> <status>`` or
  ``<name> <osada|část> obce <parent>``), split or not — the lexicon's own grammar, which says what
  the removed words were, so the recovered status and parent are kept with the heal;
* a ``t.`` alternate is split off at the lexicon's own ``t.`` marker (the name keeps the first form,
  the rest is kept as ``alt``);
* a **space-joined** form is accepted only when it equals an official RÚIAN name (obec, část obce,
  katastrální území or ZSJ, ``data/final/geo/ruian_*``) or a name the 1910 register prints **in the
  same okres** — and the unjoined form equals neither. Two joinable forms that both have evidence
  leave the name alone. Only a space *between two letters* is ever removed, so ``1. díl`` and
  ``Děčín VII -Chrochvice`` (spacing conventions, not defects) are never touched.

Nothing here changes ``settlement_id`` or ``settlements_1910``: :func:`heal_frame` returns one row per
healed cell (``settlement_id``, column, original, healed, recovered status/parent/alt, rule, evidence)
and :func:`name_index` the long table of every name form a matcher may use, healed forms carrying the
**same** ``settlement_id``.

Usage
-----
    from utils.parsing import place_names as pn
    official = pn.official_names(FINAL / "geo")
    register = pn.register_names(places)                  # {okres_id: {casefolded printed forms}}
    heals = pn.heal_frame(settlements, official, register)
    index = pn.name_index(settlements, heals)
    pn.heal_report(heals)
"""

from __future__ import annotations

import itertools
import re
import unicodedata
from dataclasses import dataclass
from pathlib import Path

import pandas as pd

__version__ = "1.0.0"   # 2026-11-16: status-tail strip, t.-alternate split, evidence-led space joins

#: The settlement columns whose text came off the lexicon's text layer.
NAME_COLUMNS = ("cs_name", "name_1910", "de_name", "parent_1910")

#: The 1910 status words of the lexicon's grammar and the ``status_1910`` each one means
#: (``lexikon._parse_entry``: ``obec``/``město`` → obec; ``osada`` → osada; ``část`` → cast).
STATUS_WORDS = {"obec": "obec", "město": "obec", "osada": "osada", "část": "cast"}

#: Most spaces a name may have before the join search gives up (2**10 forms).
MAX_SPACES = 10

#: The RÚIAN tables and name columns that count as official modern names.
RUIAN_NAME_COLUMNS = {"ruian_obce": "obec_name", "ruian_casti_obce": "cast_name",
                      "ruian_ku": "ku_name", "ruian_zsj": "zsj_name"}


def _split_word(word: str) -> str:
    """A regex for ``word`` with at most one stray space between any two of its letters."""
    letters = [re.escape(c) for c in word]
    return "".join(c + r"\s?" for c in letters[:-1]) + letters[-1]


_STATUS_ALT = "|".join(f"(?P<w{i}>{_split_word(w)})" for i, w in enumerate(STATUS_WORDS))
#: ``<name> <status>`` or ``<name> <osada|část> obce <parent>``, the status word split or not,
#: only at the end of the cell. **Case-sensitive**: the grammar's status words are lower-case, a
#: name's words capitalised, so ``Staré Město`` or ``Nová Osada`` is never cut.
_STATUS_TAIL_RE = re.compile(
    rf"^(?P<name>.*?\S)\s+(?:{_STATUS_ALT})(?:\s+{_split_word('obce')}\s+(?P<parent>\S.*))?\s*$")
#: The lexicon's ``t.`` (*též*, "also") before an alternate form, ``t .`` included.
_ALT_RE = re.compile(r"\s+t\s?\.\s+(?=\S)")
#: A space between two letters: the only kind a join may remove. In mixed-case text the letter
#: after it must be lower-case — a capital starts a word (``Deutsch Brodek`` is two words, and a
#: register reading that dropped the space must not make it one); an all-capitals headword gives no
#: such clue, so any letter-to-letter space is a candidate there.
_LETTER_SPACE_RE = re.compile(r"(?<=[^\W\d_]) (?=[^\W\d_])")


@dataclass(frozen=True)
class Healed:
    """One name after healing: the name, what the removed words said, and which rules fired."""

    name: str
    status: str | None = None       # recovered from a leaked status word
    parent: str | None = None       # recovered from a leaked ``obce <parent>``
    alt: str | None = None          # a ``t.`` alternate split off
    rule: str = ""                  # "+"-joined rules that fired; "" = unchanged
    evidence: str = ""              # what a join was checked against

    @property
    def changed(self) -> bool:
        return bool(self.rule)


def name_key(text) -> str:
    """Comparison key for evidence: NFC, case-folded, whitespace collapsed; accents kept."""
    text = unicodedata.normalize("NFC", str(text or ""))
    return re.sub(r"\s+", " ", text).strip().casefold()


def strip_status_tail(text: str) -> tuple[str, str | None, str | None]:
    """``(name, status, parent)`` with a trailing status phrase removed; unchanged if none.

    ``Kosmanosy o bec`` → ``("Kosmanosy", "obec", None)``; ``Želívec osad a obce Hodkov`` →
    ``("Želívec", "osada", "Hodkov")``. A word in the middle of a name is never touched.
    """
    m = _STATUS_TAIL_RE.match(str(text or ""))
    if not m:
        return str(text or ""), None, None
    word = next(w for i, w in enumerate(STATUS_WORDS) if m.group(f"w{i}"))
    parent = m.group("parent")
    if parent is not None and STATUS_WORDS[word] not in ("osada", "cast"):
        return str(text or ""), None, None          # only a dependent place names a parent
    return m.group("name").strip(), STATUS_WORDS[word], (parent.strip() if parent else None)


def split_alternate(text: str) -> tuple[str, str | None]:
    """``(name, alternate)`` split at the lexicon's ``t.`` marker; ``(text, None)`` if none."""
    parts = _ALT_RE.split(str(text or ""), maxsplit=1)
    if len(parts) == 1:
        return str(text or ""), None
    return parts[0].strip(), parts[1].strip() or None


def joined_forms(text: str, max_spaces: int = MAX_SPACES) -> list[str]:
    """Every form of ``text`` with one or more letter-to-letter spaces removed."""
    text = str(text or "")
    capitals = text == text.upper()
    spaces = [m.start() for m in _LETTER_SPACE_RE.finditer(text)
              if capitals or text[m.start() + 1].islower()]
    if not spaces or len(spaces) > max_spaces:
        return []
    out = []
    for drop in itertools.product((False, True), repeat=len(spaces)):
        if not any(drop):
            continue
        cut = {pos for pos, d in zip(spaces, drop) if d}
        out.append("".join(c for i, c in enumerate(text) if i not in cut))
    return out


def join_spaces(text: str, official: set[str] | frozenset = frozenset(),
                register: set[str] | frozenset = frozenset()) -> tuple[str, str]:
    """``(name, evidence)``: the unique joined form an evidence set holds, else ``(text, "")``.

    ``official`` and ``register`` hold :func:`name_key` keys. Nothing is joined when the text as
    written is already in either set, or when two joined forms are. An all-capitals headword
    (whose spaces carry no case clue) is joined on RÚIAN evidence alone.
    """
    key = name_key(text)
    if key in official or key in register:
        return text, ""
    if text == text.upper():
        register = frozenset()     # a capitals headword may join any space: RÚIAN evidence only
    hits = []
    for form in joined_forms(text):
        fk = name_key(form)
        if fk in official:
            hits.append((form, "ruian"))
        elif fk in register:
            hits.append((form, "register"))
    if len({name_key(f) for f, _ in hits}) != 1:
        return text, ""
    return hits[0]


def heal_name(text, official: set[str] | frozenset = frozenset(),
              register: set[str] | frozenset = frozenset()) -> Healed:
    """Heal one lexicon name cell; see the module docstring for the rules.

    Order: the status tail first (it is what blocks the other two), then the ``t.`` alternate,
    then the evidence-led space join on the name that is left (the alternate is joined too).
    """
    raw = "" if text is None or (isinstance(text, float) and pd.isna(text)) else str(text)
    rules, evidence = [], []
    name, status, parent = strip_status_tail(raw)
    if status is not None:
        rules.append("status_tail")
    name, alt = split_alternate(name)
    if alt is not None:
        rules.append("alt_split")
    joined, why = join_spaces(name, official, register)
    if why:
        rules.append("space_join")
        evidence.append(f"{why}:{joined}")
        name = joined
    if alt is not None:
        alt_joined, alt_why = join_spaces(alt, official, register)
        if alt_why:
            rules.append("alt_space_join")
            evidence.append(f"{alt_why}:{alt_joined}")
            alt = alt_joined
    if parent is not None:
        parent, _ = split_alternate(parent)
    return Healed(name=name, status=status, parent=parent, alt=alt,
                  rule="+".join(rules), evidence="; ".join(evidence))


# ── evidence ──────────────────────────────────────────────────────────────────
def official_names(geo_dir) -> frozenset:
    """Every RÚIAN obec / část obce / KÚ / ZSJ name as a :func:`name_key`."""
    keys: set[str] = set()
    for table, col in RUIAN_NAME_COLUMNS.items():
        path = Path(geo_dir) / f"{table}.parquet"
        if path.exists():
            keys |= {name_key(v) for v in pd.read_parquet(path, columns=[col])[col].dropna()}
    return frozenset(keys)


#: The register cell's own separators and the words it prints that are no part of a name.
_CELL_SPLIT_RE = re.compile(r"[,;()]|\.(?=\s|$)")
_CELL_NOISE_RE = re.compile(r"\b(také|auch|nyní|jetzt|dříve|früher|resp|siehe|viz|betr|ant|odd)\b\.?",
                            re.IGNORECASE)


#: A cross-reference: what follows names *another* place (``Hasentanz . odd. viz Dvůr Králové``).
_CELL_XREF_RE = re.compile(r"\b(?:viz|siehe)\b", re.IGNORECASE)


def cell_forms(cell) -> list[str]:
    """The name forms a register name cell prints (``"Aš, Asch"`` → ``["Aš", "Asch"]``).

    Text after a ``viz`` / ``siehe`` cross-reference is dropped: it names the place the section is
    filed under, not this one.
    """
    text = _CELL_XREF_RE.split(str(cell or ""), maxsplit=1)[0]
    text = _CELL_NOISE_RE.sub(",", text)
    forms = [re.sub(r"\s+", " ", p).strip(" .-") for p in _CELL_SPLIT_RE.split(text)]
    return list(dict.fromkeys(f for f in forms if f))


def register_names(places: pd.DataFrame) -> dict[int, frozenset]:
    """``{okres_id: {name_key of every form printed in that okres}}`` from ``register_places``."""
    out: dict[int, set[str]] = {}
    for okres, cell in zip(places["okres_id"], places["printed_name"]):
        if pd.isna(okres):
            continue
        out.setdefault(int(okres), set()).update(name_key(f) for f in cell_forms(cell))
    return {k: frozenset(v) for k, v in out.items()}


# ── whole frame ───────────────────────────────────────────────────────────────
def heal_frame(settlements: pd.DataFrame, official: frozenset = frozenset(),
               register: dict[int, frozenset] | None = None,
               columns=NAME_COLUMNS) -> pd.DataFrame:
    """One row per healed cell of ``settlements`` (``settlements_1910``); nothing else changes.

    Columns: ``settlement_id, okres_id, column, original, healed, status_recovered,
    parent_recovered, alt, rule, evidence``. A register form counts as evidence only in the
    settlement's own okres.
    """
    register = register or {}
    rows = []
    for row in settlements.itertuples(index=False):
        okres = getattr(row, "okres_id")
        reg = register.get(int(okres), frozenset()) if pd.notna(okres) else frozenset()
        for col in columns:
            value = getattr(row, col)
            if not isinstance(value, str) or not value.strip():
                continue
            h = heal_name(value, official, reg)
            if not h.changed:
                continue
            rows.append({"settlement_id": row.settlement_id, "okres_id": okres, "column": col,
                         "original": value, "healed": h.name, "status_recovered": h.status,
                         "parent_recovered": h.parent, "alt": h.alt, "rule": h.rule,
                         "evidence": h.evidence})
    cols = ["settlement_id", "okres_id", "column", "original", "healed", "status_recovered",
            "parent_recovered", "alt", "rule", "evidence"]
    return pd.DataFrame(rows, columns=cols)


def unhealed_splits(settlements: pd.DataFrame, heals: pd.DataFrame,
                    columns=NAME_COLUMNS) -> pd.DataFrame:
    """Cells that *look* split (a 1–2 letter fragment glued to nothing) but were left alone.

    Reported, never changed: no evidence named the joined form. ``u``, ``v``, ``na``, ``nad`` …
    and Roman numerals are words, not fragments.
    """
    words = {"za", "x", "xv", "xx", "u", "v", "na", "nad", "pod", "při", "pri", "an", "am", "im", "in", "bei", "ob",
             "a", "i", "z", "ve", "ze", "se", "od", "do", "dil", "díl", "st", "sv", "t", "b",
             "ii", "iii", "iv", "vi", "vii", "viii", "ix", "xi", "xii", "der", "die", "zu"}
    done = set(zip(heals["settlement_id"], heals["column"]))
    rows = []
    for row in settlements.itertuples(index=False):
        for col in columns:
            value = getattr(row, col)
            if not isinstance(value, str) or (row.settlement_id, col) in done:
                continue
            frags = [t for t in re.findall(r"[^\W\d_]+", value)
                     if len(t) <= 2 and t.casefold() not in words]
            if frags and _LETTER_SPACE_RE.search(value):
                rows.append({"settlement_id": row.settlement_id, "column": col, "value": value,
                             "fragments": " ".join(frags)})
    return pd.DataFrame(rows, columns=["settlement_id", "column", "value", "fragments"])


def healed_attributes(settlements: pd.DataFrame, heals: pd.DataFrame) -> pd.DataFrame:
    """``settlements`` with its name cells, empty status and empty parent healed — an overlay copy.

    ``settlement_id`` and row order are untouched; a recovered status or parent fills only an
    empty cell; ``healed_rule`` lists the rules that touched the row.
    """
    out = settlements.copy()
    pos = {sid: i for i, sid in enumerate(out["settlement_id"])}
    rule = [""] * len(out)
    for h in heals.itertuples(index=False):
        i = pos[h.settlement_id]
        out.iat[i, out.columns.get_loc(h.column)] = h.healed
        if h.column in ("name_1910", "de_name"):
            if h.status_recovered and pd.isna(out.iat[i, out.columns.get_loc("status_1910")]):
                out.iat[i, out.columns.get_loc("status_1910")] = h.status_recovered
            if h.parent_recovered and pd.isna(out.iat[i, out.columns.get_loc("parent_1910")]):
                out.iat[i, out.columns.get_loc("parent_1910")] = h.parent_recovered
        rule[i] = "+".join(filter(None, [rule[i], f"{h.column}:{h.rule}"]))
    out["healed_rule"] = rule
    return out


def name_index(settlements: pd.DataFrame, heals: pd.DataFrame) -> pd.DataFrame:
    """Every name form a matcher may use: ``settlement_id, okres_id, form, source, healed``.

    The forms as parsed **and** their healed forms (a ``t.`` alternate as its own form), all under
    the settlement's own ``settlement_id``. ``source`` is the column; ``healed`` marks a form only
    the overlay supplies.
    """
    rows = []
    for row in settlements.itertuples(index=False):
        for col in ("cs_name", "name_1910", "de_name"):
            value = getattr(row, col)
            if isinstance(value, str) and value.strip():
                rows.append((row.settlement_id, row.okres_id, value.strip(), col, False))
    for h in heals.itertuples(index=False):
        if h.column == "parent_1910":
            continue
        rows.append((h.settlement_id, h.okres_id, h.healed, h.column, True))
        if isinstance(h.alt, str) and h.alt:
            rows.append((h.settlement_id, h.okres_id, h.alt, f"{h.column}_alt", True))
    out = pd.DataFrame(rows, columns=["settlement_id", "okres_id", "form", "source", "healed"])
    out["key"] = out["form"].map(name_key)
    return (out.sort_values("healed").drop_duplicates(["settlement_id", "key"])
            .sort_values(["settlement_id", "source"]).reset_index(drop=True))


def heal_report(heals: pd.DataFrame, unhealed: pd.DataFrame | None = None) -> dict:
    """Counts by column × rule, plus examples; printed and returned."""
    out = {"healed_cells": int(len(heals)),
           "settlements_touched": int(heals["settlement_id"].nunique()) if len(heals) else 0,
           "status_recovered": int(heals["status_recovered"].notna().sum()) if len(heals) else 0,
           "parent_recovered": int(heals["parent_recovered"].notna().sum()) if len(heals) else 0,
           "by_column_rule": {}, "examples": {}}
    for (col, rule), grp in heals.groupby(["column", "rule"]):
        out["by_column_rule"][f"{col}|{rule}"] = int(len(grp))
        out["examples"][f"{col}|{rule}"] = [f"{o} -> {h}" for o, h in
                                            zip(grp["original"].head(5), grp["healed"].head(5))]
    if unhealed is not None:
        out["left_alone_looks_split"] = int(len(unhealed))
        out["left_alone_examples"] = unhealed["value"].head(15).tolist()
    print(f"  place_names {__version__}: {out['healed_cells']} cells healed on "
          f"{out['settlements_touched']} settlements; status recovered {out['status_recovered']}, "
          f"parent {out['parent_recovered']}")
    for k, v in out["by_column_rule"].items():
        print(f"    {k:40s} {v:5d}  e.g. {out['examples'][k][:2]}")
    return out
