How it works

  1. settlement — the two places anchored to the same settlement_id (each id once per side);
  2. name — a German place's printed forms against the Czech place's cell, through lexikon_1910_names.anchor_blocks (≥ 84, margin 5, ties refused, one to one);
  3. order — the German volume numbers a court's municipalities in alphabetical order of their German names, accents folded (Altsedlowitz 3, Bausnitz 4, Burkersdorf 6, Döberle 8 … on Trutnov, where the Czech page holds all 33). Between two paired municipalities the German numbers form a closed run of slots; the Czech municipalities whose German name sorts between the two pairs' names fill it, in order — only if there are exactly as many as there are slots, which is what proves no municipality is missing from the Czech pages read. Ends of a court are closed only on a full run (complete). A court whose paired names do not rise with the German numbers is refused the order step whole (a collation the rule does not know).

A German place that its own names left unanchored takes the Czech place's settlement_id (via_czech_name / via_czech_order). Where both anchored on their own, crosswalk_agrees records whether they agree (never on a settlement pair, which agrees by construction). order_holdout tests the order rule itself: each known pair is dropped in turn and the order step asked to find it again. Parts are paired by settlement and by name only: their German order is not yet verified on any page.

def crosswalk_report(places: pd.DataFrame, status: dict) -> str:
    """How many German places paired, by step, how many inherited, and how often the two agree."""
    de = places[places["edition"] == GERMAN]
    if de.empty or "crosswalk" not in de:
        return "  crosswalk: no German Böhmen places in this build"
    by = de.groupby("place_level")
    lines = ["  crosswalk to the Czech volume (German Böhmen places):"]
    for level, g in by:
        how = g["crosswalk"].value_counts().to_dict()
        agree = g["crosswalk_agrees"].dropna()
        lines.append(f"    {level:12s} {len(g):4d} places, paired {int(g['crosswalk'].notna().sum())} {how}; "
                     f"inherited {int(g['anchor_status'].isin(INHERITED.values()).sum())}; "
                     f"both anchored {len(agree)}, agree {int(agree.sum())}")
    hold = order_holdout(places)
    if len(hold):
        lines.append(f"    order rule, leave-one-out on {len(hold)} known pairs: predicted "
                     f"{int(hold['predicted'].sum())}, of those correct {int(hold['correct'].sum())}")
    refused = sorted(k for k, s in status.items() if s == "not monotone")
    lines.append(f"    order step by court: {pd.Series(status).value_counts().to_dict() if status else {}}"
                 + (f"; refused (not monotone): {refused}" if refused else ""))
    return "\n".join(lines)

"""lexikon_1910_crosswalk — the German Böhmen volume's places paired with the Czech volume's (D-150)
=================================================================================================
Both volumes print the same 1910 municipalities of Bohemia, each under both names — the Czech one
``Dílce, Diletz``, the German one ``Diletz, Dílce`` — but each numbers them in its own alphabet, so
Dílce is No. 7 of court Jičín in the Czech volume and No. 9 in the German one. A German place the
route read but could not name (no line on the reader's parts list, and the German table is never
read) can still be found through the Czech volume, whose table prints every municipality's German
name. Three steps per court, each refusing what it cannot settle:

1. **settlement** — the two places anchored to the same ``settlement_id`` (each id once per side);
2. **name** — a German place's printed forms against the Czech place's cell, through
   :func:`lexikon_1910_names.anchor_blocks` (≥ 84, margin 5, ties refused, one to one);
3. **order** — the German volume numbers a court's municipalities in alphabetical order of their
   German names, accents folded (Altsedlowitz 3, Bausnitz 4, Burkersdorf 6, Döberle 8 … on
   Trutnov, where the Czech page holds all 33). Between two paired municipalities the German
   numbers form a closed run of slots; the Czech municipalities whose German name sorts between the
   two pairs' names fill it, in order — **only if there are exactly as many as there are slots**,
   which is what proves no municipality is missing from the Czech pages read. Ends of a court are
   closed only on a full run (``complete``). A court whose paired names do not rise with the German
   numbers is refused the order step whole (a collation the rule does not know).

A German place that its own names left unanchored takes the Czech place's ``settlement_id``
(``via_czech_name`` / ``via_czech_order``). Where both anchored on their own, ``crosswalk_agrees``
records whether they agree (never on a settlement pair, which agrees by construction).
:func:`order_holdout` tests the order rule itself: each known pair is dropped in turn and the
order step asked to find it again. Parts are paired by settlement
and by name only: their German order is not yet verified on any page.
"""

from __future__ import annotations

import re
from types import SimpleNamespace

import pandas as pd

from utils.lexikon_1910 import lexikon_1910_names as lxn

__version__ = "1.0.0"   # first crosswalk, 2026-10-29 (D-150)

CZECH, GERMAN = "cz", "de-bohemia"
INHERITED = {"name": "via_czech_name", "order": "via_czech_order"}
_ALTERNATE_RE = re.compile(r"^(také|auch)\s", re.IGNORECASE)


def german_key(cell) -> str:
    """The sort key of a Czech name cell's German name: its last form that is not a Czech
    alternate (``také …``), accents folded, ``ß`` as ``ss``; a cell with one name is its own
    (a Czech-only place prints its Czech name in the German volume too)."""
    forms = [f.strip() for f in str(cell or "").split(",") if f.strip()]
    forms = [f for f in forms if not _ALTERNATE_RE.match(f)]
    name = forms[-1] if len(forms) >= 2 else (forms[0] if forms else "")
    return lxn.normalise(name.replace("ß", "ss"))


def _sid(value) -> str | None:
    return None if value is None or pd.isna(value) else str(value)


def _settlement_pairs(de: pd.DataFrame, cz: pd.DataFrame) -> dict[int, int]:
    """German index → Czech index, where each side anchored the id exactly once."""
    de_ids, cz_ids = de["settlement_id"].dropna(), cz["settlement_id"].dropna()
    once_de = de_ids[~de_ids.duplicated(keep=False)]
    once_cz = cz_ids[~cz_ids.duplicated(keep=False)]
    by_id = {sid: i for i, sid in once_cz.items()}
    return {i: by_id[sid] for i, sid in once_de.items() if sid in by_id}


def _name_pairs(de: pd.DataFrame, cz: pd.DataFrame) -> dict[int, int]:
    """German index → Czech index by printed names (one to one, ties refused)."""
    named = de[de["printed_name"].fillna("") != ""]
    if named.empty or cz.empty:
        return {}
    cands = [lxn.Obec(cs_name=str(i), okres_id=0, okres_1910="",
                      variants=set(lxn.parse_name_cell(cell)))
             for i, cell in cz["printed_name"].fillna("").items()]
    blocks = [SimpleNamespace(block_id=str(i), name=cell, municipality={"name": cell})
              for i, cell in named["printed_name"].items()]
    return {int(a.block_id): int(a.obec.cs_name) for a in lxn.anchor_blocks(blocks, cands)
            if a.verdict == "matched"}


def _order_pairs(de: pd.DataFrame, cz: pd.DataFrame, pairs: dict[int, int],
                 complete: bool) -> tuple[dict[int, int], str]:
    """German index → Czech index by German alphabetical order between pairs (module docstring).

    Returns the new pairs and the court's order status: ``filled``, ``nothing to fill``,
    ``not monotone`` (refused) or ``no anchors``."""
    key = cz["printed_name"].map(german_key)
    anchors = sorted((int(de.at[d, "muni_seq"]), key[c]) for d, c in pairs.items())
    if any(k1 >= k2 for (_, k1), (_, k2) in zip(anchors, anchors[1:])):
        return {}, "not monotone"
    if complete:
        anchors = [(0, ""), *anchors, (len(cz) + 1, "￿")]
    if len(anchors) < 2:
        return {}, "no anchors"
    used = set(pairs.values())
    free = key[[i for i in cz.index if i not in used]]
    de_by_num = {int(n): i for i, n in de["muni_seq"].items()}
    out: dict[int, int] = {}
    for (a, ka), (b, kb) in zip(anchors, anchors[1:]):
        slots = list(range(a + 1, b))
        between = free[(free > ka) & (free < kb)].sort_values()
        if not slots or len(between) != len(slots) or between.duplicated().any():
            continue
        for num, cz_index in zip(slots, between.index):
            if num in de_by_num:
                out[de_by_num[num]] = cz_index
    return out, "filled" if out else "nothing to fill"


def pair_court(de: pd.DataFrame, cz: pd.DataFrame, complete: bool = False
               ) -> tuple[dict[int, tuple[int, str]], str]:

    pairs.update({d: (c, "order") for d, c in order.items()})

def _link(out: pd.DataFrame, d: int, c: int, how: str, anchored: set[str]) -> None:
    """Record one pair on the German row; inherit the Czech settlement where the German has none."""
    out.at[d, "cz_muni_seq"] = int(out.at[c, "muni_seq"])
    out.at[d, "cz_settlement_seq"] = int(out.at[c, "settlement_seq"])
    out.at[d, "crosswalk"] = how
    de_sid, cz_sid = _sid(out.at[d, "settlement_id"]), _sid(out.at[c, "settlement_id"])
    if de_sid and cz_sid and how != "settlement":       # a settlement pair agrees by construction
        out.at[d, "crosswalk_agrees"] = de_sid == cz_sid
    elif cz_sid and out.at[d, "anchor_status"] not in anchored and how in INHERITED:
        out.at[d, "settlement_id"] = cz_sid
        out.at[d, "anchor_status"] = INHERITED[how]


def crosswalk_report(places: pd.DataFrame, status: dict) -> str:
    """How many German places paired, by step, how many inherited, and how often the two agree."""
    de = places[places["edition"] == GERMAN]
    if de.empty or "crosswalk" not in de:
        return "  crosswalk: no German Böhmen places in this build"
    by = de.groupby("place_level")
    lines = ["  crosswalk to the Czech volume (German Böhmen places):"]
    for level, g in by:
        how = g["crosswalk"].value_counts().to_dict()
        agree = g["crosswalk_agrees"].dropna()
        lines.append(f"    {level:12s} {len(g):4d} places, paired {int(g['crosswalk'].notna().sum())} {how}; "
                     f"inherited {int(g['anchor_status'].isin(INHERITED.values()).sum())}; "
                     f"both anchored {len(agree)}, agree {int(agree.sum())}")
    hold = order_holdout(places)
    if len(hold):
        lines.append(f"    order rule, leave-one-out on {len(hold)} known pairs: predicted "
                     f"{int(hold['predicted'].sum())}, of those correct {int(hold['correct'].sum())}")
    refused = sorted(k for k, s in status.items() if s == "not monotone")
    lines.append(f"    order step by court: {pd.Series(status).value_counts().to_dict() if status else {}}"
                 + (f"; refused (not monotone): {refused}" if refused else ""))
    return "\n".join(lines)
