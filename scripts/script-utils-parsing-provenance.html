How it works

  1. Archival personnel document — which of VÚA-VHA's two standard legionnaire records backs the entry: the legionářský poslužný spis (service file) and/or the osobní karta legionáře (personal card). → source_spis / source_karta
  2. Additional cited source — anything csol's editors added beyond those two: a published okres volume, a regimental memorial book, a family archive, a museum fond. This is a marker of modern editorial attention, not of a different wartime document. → source_extra / source_extra_ref

Keeping them apart matters: a record citing Legionáři okresu Pardubice (2006) was worked over by a local historian ninety years after the fact, which is a statement about recording completeness, not about the archive.

Matching is anchored at the start of the string and never a substring test — the two document names are stripped in sequence from the front and whatever survives is the extra citation. In the 2026-08-26 build this leaves zero document-name text in the residual across all 132,562 non-null values. That anchoring is deliberate: unanchored matching is this repo's single most frequent bug (see reference/AUDIT_2026-07-16.md).

Missing stays missing. Where the source field is &lt;NA&gt; every output column is &lt;NA&gt;, never False — an unrecorded source is not evidence that the document is absent.

def provenance_selftest() -> list[str]:
    """Assert the anchoring rules. Needs no data; returns a list of failures."""
    fails: list[str] = []

    def want(label, got, exp):
        if got != exp:
            fails.append(f"{label}: got {got!r}, expected {exp!r}")

    both = f"{SPIS} {KARTA}"
    want("both documents", _split_source(both)[:3], (True, True, False))
    want("spis only", _split_source(SPIS)[:3], (True, False, False))
    want("karta only", _split_source(KARTA)[:3], (False, True, False))

    # An extra citation survives verbatim, and only the extra.
    s_, k_, e_, r_ = _split_source(f"{both} JAKL, Tomáš. Hrdinové z Londýna. Praha: VHÚ, 2016.")
    want("extra flags", (s_, k_, e_), (True, True, True))
    if not (r_ or "").startswith("JAKL"):
        fails.append(f"extra ref not stripped clean: {r_!r}")
    if SPIS in (r_ or "").lower() or KARTA in (r_ or "").lower():
        fails.append("document name leaked into the extra citation")

    # A literature-only record claims neither document.
    want("literature only", _split_source("Seznam sibiřské armády. Praha: MNO, 1919.")[:3],
         (False, False, True))

    # Missing stays missing — never False.
    for empty in (pd.NA, None, "", "   "):
        if not all(v is pd.NA for v in _split_source(empty)):
            fails.append(f"missing value {empty!r} did not stay <NA>")

    # The document name is matched only at the front, never mid-string.
    want("mid-string document name is not a match",
         _split_source(f"rodinný archiv {SPIS}")[:3], (False, False, True))

    return fails

"""
provenance.py — what source csol.cz cites for a legionnaire record
==================================================================
Parses the free-text ``Zdroj informací`` field into two separate families,
because the field concatenates two things that mean different things:

1. **Archival personnel document** — which of VÚA-VHA's two standard
   legionnaire records backs the entry: the *legionářský poslužný spis*
   (service file) and/or the *osobní karta legionáře* (personal card).
   → ``source_spis`` / ``source_karta``
2. **Additional cited source** — anything csol's editors added beyond those
   two: a published okres volume, a regimental memorial book, a family
   archive, a museum fond. This is a marker of *modern editorial attention*,
   not of a different wartime document.
   → ``source_extra`` / ``source_extra_ref``

Keeping them apart matters: a record citing *Legionáři okresu Pardubice*
(2006) was worked over by a local historian ninety years after the fact, which
is a statement about recording completeness, not about the archive.

Matching is **anchored at the start of the string and never a substring test**
— the two document names are stripped in sequence from the front and whatever
survives is the extra citation. In the 2026-08-26 build this leaves zero
document-name text in the residual across all 132,562 non-null values. That
anchoring is deliberate: unanchored matching is this repo's single most
frequent bug (see ``reference/AUDIT_2026-07-16.md``).

**Missing stays missing.** Where the source field is ``<NA>`` every output
column is ``<NA>``, never ``False`` — an unrecorded source is not evidence
that the document is absent.

Usage
-----
    from utils.parsing.provenance import add_provenance_columns, provenance_report
    df = add_provenance_columns(df)
    provenance_report(df)
"""

from __future__ import annotations

import re

import pandas as pd

__version__ = "1.0.0"

# The two standard VÚA-VHA legionnaire personnel documents, lowercased.
SPIS = "legionářský poslužný spis"
KARTA = "osobní karta legionáře"

_WS = re.compile(r"\s+")


def _split_source(value) -> tuple[object, object, object, object]:
    """Return (spis, karta, extra, extra_ref) for one raw value."""
    if value is None or value is pd.NA or (isinstance(value, float) and pd.isna(value)):
        return pd.NA, pd.NA, pd.NA, pd.NA

    text = _WS.sub(" ", str(value)).strip()
    if not text:
        return pd.NA, pd.NA, pd.NA, pd.NA

    rest = text
    spis = karta = False

    # Anchored, ordered strip: spis leads whenever present, karta follows it.
    low = rest.lower()
    if low.startswith(SPIS):
        spis = True
        rest = rest[len(SPIS):].strip()
        low = rest.lower()
    if low.startswith(KARTA):
        karta = True
        rest = rest[len(KARTA):].strip()

    extra = bool(rest)
    return spis, karta, extra, (rest if extra else pd.NA)


def add_provenance_columns(
    df: pd.DataFrame,
    source_col: str = "Zdroj informací",
    prefix: str = "source",
) -> pd.DataFrame:
    """Add ``{prefix}_spis``, ``_karta``, ``_extra``, ``_extra_ref``."""
    df = df.copy()
    if source_col not in df.columns:
        return df

    parsed = df[source_col].map(_split_source)
    df[f"{prefix}_spis"] = pd.array([p[0] for p in parsed], dtype="boolean")
    df[f"{prefix}_karta"] = pd.array([p[1] for p in parsed], dtype="boolean")
    df[f"{prefix}_extra"] = pd.array([p[2] for p in parsed], dtype="boolean")
    df[f"{prefix}_extra_ref"] = pd.array([p[3] for p in parsed], dtype="string")
    return df


def provenance_report(df: pd.DataFrame, prefix: str = "source") -> None:
    n = len(df)
    spis, karta = df[f"{prefix}_spis"], df[f"{prefix}_karta"]
    extra, ref = df[f"{prefix}_extra"], df[f"{prefix}_extra_ref"]
    recorded = spis.notna()
    print(f"provenance: {recorded.sum():,}/{n:,} records cite a source "
          f"({(~recorded).sum():,} <NA>)")
    print(f"  poslužný spis (service file) : {spis.fillna(False).sum():>7,} "
          f"({spis.fillna(False).mean() * 100:.1f}%)")
    print(f"  osobní karta  (personal card): {karta.fillna(False).sum():>7,} "
          f"({karta.fillna(False).mean() * 100:.1f}%)")
    print(f"  both                         : {(spis.fillna(False) & karta.fillna(False)).sum():>7,}")
    print(f"  extra citation beyond the two: {extra.fillna(False).sum():>7,} "
          f"({extra.fillna(False).mean() * 100:.1f}%), "
          f"{ref.dropna().nunique():,} distinct")
