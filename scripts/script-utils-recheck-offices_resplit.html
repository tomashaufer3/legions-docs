How it works

The German cutter cuts each column of a page into segments at the headings it finds in the image, and reads each heading off a crop. Where it misses one — an Ämter, Anstalten usw. heading half way down a column, on a page the court runs onto — the segment below it keeps the kind of the segment above. Every reader still transcribes the whole crop, heading included, and the volume's matcher recognises the heading in their text (('offices', 'Neubenatek', 51) on scan 0152); but lexikon_1910_vision.sections_from_segments only strips a found heading and never starts a section at one, since a section's identity comes from where the crop was cut. The offices entries below the missed heading are therefore filed as parts, and the decoder, which reads offices sections only, never sees them. All four readers lost the same lines, so no disagreement flagged it.

This module rebuilds each reader's segments from its saved notes (notes_&lt;label&gt;.csv, written by run_r5_german_notes.assemble), splits a segment where a line is a heading of the other kind, carries the new kind into the segments the cutter marked as continuations, and decodes the result with the route's own decoder, as assemble does. Nothing is read again and nothing in the route changes: it writes only under output/recheck/&lt;edition&gt;/resplit/ the rows the reader's own inventory lacks, which notebook 17 settles as the route settles claims (agreed by every reader: stands; disputed: Opus) and tests against the Czech volume where Böhmen has both.

def load_read(label: str) -> list[tuple[lxnt.Segment, list[str]]]:
    """``[(segment, lines)]`` in reading order, as ``assemble`` passed them to the decoder: the
    reader's cleaned lines per crop, with the crop's kind, court and whether it was carried."""
    carried = {(r["page"], r["region"], r["order"]): r.get("carried") == "1"
               for r in _rows("crop_audit", label)}
    slots: dict[tuple, dict] = {}
    for r in _rows("notes", label):
        if r["kind"] == "legend":
            continue
        slot = (r["page"], r["region"], r["order"])
        s = slots.setdefault(slot, {"kind": r["kind"], "district": r["court_de"], "lines": []})
        s["lines"].append((int(r["line_no"]), r["text"]))
    read = []
    for slot in sorted(slots, key=lambda k: (k[0], int(k[2]))):
        s = slots[slot]
        segment = lxnt.Segment(kind=s["kind"], district=s["district"], page=slot[0],
                               region=slot[1], box=(0, 0, 0, 0), order=int(slot[2]),
                               carried=carried.get(slot, False))
        read.append((segment, [t for _n, t in sorted(s["lines"])]))
    return read

def other_court(crosswalk: dict):
    """``(a, b) -> bool``: the two German court names are different 1910 courts — both resolve
    through the route's crosswalk, to different Czech courts. A name that does not resolve, or
    resolves to the same court, is a spelling of the crop's own court and changes nothing."""
    def differ(a: str, b: str) -> bool:
        ca, cb = lxdn.court_name(a, crosswalk)[0], lxdn.court_name(b, crosswalk)[0]
        return bool(ca and cb and lxnt.normalise(ca) != lxnt.normalise(cb))
    return differ

def resplit_sections(read: list, find, differ=lambda a, b: False) -> tuple[list[lxnt.Section], list[dict]]:
    """:func:`sections_from_segments` with the split it does not make: a line the matcher reads as
    a heading of the other kind, or of another court (``differ``), opens a section of its own, and a
    carried segment continues whatever section its court's previous segment ended in.
    Returns ``(sections, splits)``."""
    merged: dict[tuple, lxnt.Section] = {}
    order: list[tuple] = []
    last_kind: dict[str, str] = {}                 # court key -> kind in force at its last line
    splits = []

    def section(page, kind, district):
        key = (page, kind, lxnt.normalise(district))
        if key not in merged:
            merged[key] = lxnt.Section(kind=kind, district=district, page=page, lines=[])
            order.append(key)
        return merged[key]

    for segment, lines in read:
        kind, district = segment.kind, segment.district
        if segment.carried and last_kind.get(segment.key, kind) != kind:
            splits.append({"page": segment.page, "region": segment.region, "order": segment.order,
                           "court_de": district, "from": kind, "to": last_kind[segment.key],
                           "how": "carried"})
            kind = last_kind[segment.key]
        current = section(segment.page, kind, district)
        for line in lines:
            heading = find(line)
            # Split on a change of kind, or on a heading of another 1910 court. A court name read
            # off the reader's text that is only a spelling of the crop's court changes nothing:
            # the court stays the crop's, as in the route.
            real = bool(heading) and heading[0] in ("parts", "offices") and opens_heading(line, heading)
            new_court = real and differ(heading[1], district)
            if real and (heading[0] != kind or new_court):
                splits.append({"page": segment.page, "region": segment.region, "order": segment.order,
                               "court_de": district, "heading_court": heading[1],
                               "from": kind, "to": heading[0],
                               "how": "court" if new_court else "heading"})
                kind = heading[0]
                district = heading[1] if new_court else district
                current = section(segment.page, kind, district)
            tail = line[heading[2]:] if heading else line
            if tail.strip():
                current.lines.append(tail)
        last_kind[segment.key] = kind
        last_kind[lxnt.normalise(district)] = kind
    return [merged[k] for k in order], splits

def resplit_reader(edition: str, label: str) -> dict:
    """One reader: the rows the re-split adds to its inventory, and the rows it would lose (which
    must be none — the re-split only moves lines out of parts sections)."""
    profile = lxvol.load_profile(rf.GERMAN[edition])
    differ = other_court(lxdn.load_court_crosswalk())
    sections, splits = resplit_sections(load_read(label), profile.find_heading, differ)
    new = decoded_inventory(sections, edition)
    old = _rows("inventory", label)
    have = {_key(r) for r in old}
    added = [r for r in new if _key(r) not in have]
    lost = have - {_key(r) for r in new}
    return {"label": label, "splits": splits, "added": added, "lost": sorted(lost),
            "old_rows": len(old), "new_rows": len(new), "inventory": new}

def run(edition: str, pages: str = "full", out: Path = OUT) -> dict:
    """Every reader of one German edition; writes ``resplit/inventory_<label>.csv`` (the added
    rows), ``splits_<label>.csv`` and ``summary.json`` under ``output/recheck/<edition>/``."""
    if edition not in GERMAN:
        raise ValueError(f"{edition}: the re-split is for the German editions (the Czech volume "
                         f"is cut by its text layer, and lost no offices lines)")
    folder = out / edition / "resplit"
    folder.mkdir(parents=True, exist_ok=True)
    summary = {"module": __version__, "edition": edition, "pages": pages,
               "run": time.strftime("%Y-%m-%d %H:%M:%S"), "readers": {}}
    for label in readers(edition, pages):
        r = resplit_reader(edition, label)
        _write(folder / f"added_{label}.csv", r["added"])
        _write(folder / f"inventory_{label}.csv", r["inventory"])
        _write(folder / f"lost_{label}.csv", [dict(zip(CLAIM, k)) for k in r["lost"]])
        _write(folder / f"splits_{label}.csv", r["splits"])
        summary["readers"][label] = {
            "splits": dict(Counter(s["how"] for s in r["splits"])),
            "rows_added": len(r["added"]), "rows_lost": len(r["lost"]),
            "entries_added": len({_key(x)[:4] for x in r["added"]}),
            "pages_added": len({x["page"] for x in r["added"]})}
        print(f"  {label}: {summary['readers'][label]}")
    (folder / "summary.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return summary

def consensus(edition: str, pages: str = "full", out: Path = OUT) -> dict:
    """The added claims across readers, settled as the route settles them (``routed_claims``):
    a claim every reader makes stands; one only some make is disputed, for Opus. Only entries
    every reader's re-split reached are compared."""
    folder = out / edition / "resplit"
    labels = readers(edition, pages)
    claims, touched = {}, set()
    for label in labels:
        with open(folder / f"inventory_{label}.csv", encoding="utf-8", newline="") as fh:
            claims[label] = {_key(r) for r in csv.DictReader(fh) if r.get("code")}
        with open(folder / f"added_{label}.csv", encoding="utf-8", newline="") as fh:
            touched |= {_key(r)[:4] for r in csv.DictReader(fh)}
    # The entries the re-split changed for any reader, settled as the route settles claims: where
    # every reader's whole re-split inventory has the entry, a claim all make stands and one only
    # some make is disputed (``routed_claims``); elsewhere the reader of record's claim stands
    # alone (the route's "not asked: reader of record"). A reader may have left the heading line
    # out of its text, and then its notes cannot be split there.
    record = labels[0]
    entries = touched & set.intersection(*({c[:4] for c in cs} for cs in claims.values()))
    shared = {l: {c for c in cs if c[:4] in entries} for l, cs in claims.items()}
    union, agreed = set().union(*shared.values()), set.intersection(*shared.values())
    rows = [dict(zip(CLAIM, c), status="agreed" if c in agreed else "disputed",
                 support=sum(c in s for s in shared.values())) for c in sorted(union)]
    alone = {c for c in claims[record] if c[:4] in touched - entries}
    rows += [dict(zip(CLAIM, c), status="reader of record", support=1) for c in sorted(alone)]
    _write(folder / "consensus.csv", rows)
    return {"entries_touched": len(touched), "entries_every_reader": len(entries),
            "entries_reader_of_record": len({c[:4] for c in alone}),
            "agreed": len(agreed), "disputed": len(union - agreed), "reader_of_record": len(alone)}

def czech_test(pages: str = "full", out: Path = OUT, statuses=("agreed", "reader of record"),
               folder: str = "resplit") -> pd.DataFrame:
    """German Böhmen's re-split claims against the Czech volume on the municipalities they fill.

    The claims are rolled up to the municipality (parts included, as in
    ``check_offices_coverage.bohemia_editions``) and paired with the Czech municipality through
    the crosswalk's ``cz_muni_seq``; a municipality counts only if the route gave it no pictogram
    in German before. Per symbol: both, Czech only, German only, agreement = both / either.
    ``folder``: whose ``consensus.csv`` — ``resplit`` (§3) or ``reread`` (§4, ``offices_reread``)."""
    reg = ROOT / "data" / "final" / "register"
    places = pd.read_parquet(reg / "register_places.parquet")
    inst = pd.read_parquet(reg / "register_institutions.parquet")
    inv = pd.read_csv(rf.out_dir("de-bohemia", pages) / "final_inventory.csv", dtype=str,
                      keep_default_na=False)
    court_of = dict(inv[["district", "court_id"]].drop_duplicates().itertuples(index=False))
    rows = pd.read_csv(out / "de-bohemia" / folder / "consensus.csv", dtype=str, keep_default_na=False)
    rows = rows[rows.status.isin(statuses) & rows.code.str.startswith("[")]
    rows = rows.assign(court_key=rows.district.map(court_of), muni_seq=rows.muni_seq.astype(int))
    unmapped = int(rows.court_key.isna().sum())
    new = rows.dropna(subset=["court_key"]).groupby(["court_key", "muni_seq"]).code.agg(set)
    de = places[(places.edition == "de-bohemia") & (places.place_level == "municipality")
                & places.cz_muni_seq.notna()]
    pic = inst[inst.source == "pictogram"]
    had = set(zip(pic[pic.edition == "de-bohemia"].court_key, pic[pic.edition == "de-bohemia"].muni_seq))
    cz = pic[pic.edition == "cz"].groupby(["court_key", "muni_seq"]).mark.agg(set)
    pairs = [(new[(k, m)], cz.get((k, int(c)), set())) for k, m, c in
             zip(de.court_key, de.muni_seq, de.cz_muni_seq)
             if (k, m) in new.index and (k, m) not in had]
    out_rows = []
    for mark in TESTED:
        both = sum(mark in d and mark in c for d, c in pairs)
        c_only = sum(mark in c and mark not in d for d, c in pairs)
        d_only = sum(mark in d and mark not in c for d, c in pairs)
        out_rows.append({"mark": mark, "both": both, "cz_only": c_only, "de_only": d_only,
                         "agree": round(both / max(1, both + c_only + d_only), 3)})
    table = pd.DataFrame(out_rows)
    table.attrs.update(municipalities=len(pairs), unmapped_claims=unmapped)
    return table

"""The offices lines every German reader transcribed and the route filed as parts, decoded (D-192).

The German cutter cuts each column of a page into segments at the headings it finds in the image,
and reads each heading off a crop. Where it misses one — an *Ämter, Anstalten usw.* heading half
way down a column, on a page the court runs onto — the segment below it keeps the kind of the
segment above. Every reader still transcribes the whole crop, heading included, and the volume's
matcher recognises the heading in their text (``('offices', 'Neubenatek', 51)`` on scan 0152); but
:func:`lexikon_1910_vision.sections_from_segments` only strips a found heading and never starts a
section at one, since a section's identity comes from where the crop was cut. The offices entries
below the missed heading are therefore filed as parts, and the decoder, which reads offices
sections only, never sees them. All four readers lost the same lines, so no disagreement flagged it.

This module rebuilds each reader's segments from its saved notes (``notes_<label>.csv``, written by
``run_r5_german_notes.assemble``), splits a segment where a line is a heading of the other kind,
carries the new kind into the segments the cutter marked as continuations, and decodes the result
with the route's own decoder, as ``assemble`` does. Nothing is read again and nothing in the
route changes: it writes only under ``output/recheck/<edition>/resplit/`` the rows the reader's
own inventory lacks, which notebook 17 settles as the route settles claims (agreed by every reader:
stands; disputed: Opus) and tests against the Czech volume where Böhmen has both.

    python -m utils.recheck.offices_resplit --edition de-moravia
"""

def readers(edition: str, pages: str = "full") -> list[str]:
    """The run's readers, reader of record first (``run_frozen.labels``)."""
    lab = rf.labels(pages, edition)
    return [lab["gemini"], *lab["haiku"]]


def _rows(kind: str, label: str) -> list[dict]:
    path = bo.find_pass_file(kind, label)
    if path is None:
        raise FileNotFoundError(f"no {kind}_{label}.csv under {bo.BAKEOFF}")
    with open(path, encoding="utf-8", newline="") as fh:
        return list(csv.DictReader(fh))


def load_read(label: str) -> list[tuple[lxnt.Segment, list[str]]]:
    """``[(segment, lines)]`` in reading order, as ``assemble`` passed them to the decoder: the
    reader's cleaned lines per crop, with the crop's kind, court and whether it was carried."""
    carried = {(r["page"], r["region"], r["order"]): r.get("carried") == "1"
               for r in _rows("crop_audit", label)}
    slots: dict[tuple, dict] = {}
    for r in _rows("notes", label):
        if r["kind"] == "legend":
            continue
        slot = (r["page"], r["region"], r["order"])
        s = slots.setdefault(slot, {"kind": r["kind"], "district": r["court_de"], "lines": []})
        s["lines"].append((int(r["line_no"]), r["text"]))
    read = []
    for slot in sorted(slots, key=lambda k: (k[0], int(k[2]))):
        s = slots[slot]
        segment = lxnt.Segment(kind=s["kind"], district=s["district"], page=slot[0],
                               region=slot[1], box=(0, 0, 0, 0), order=int(slot[2]),
                               carried=carried.get(slot, False))
        read.append((segment, [t for _n, t in sorted(s["lines"])]))
    return read


#: How a heading line opens, unaccented and letters only: *Ortsbestandteile* / *Ämter, Anstalten*
#: (an OCR'd ``Amter``, or ``mter`` with the umlaut lost) / *Části osad* / *Úřady, ústavy*.
HEADING_OPENS = ("ortsbest", "amter", "mter", "casti", "urady")


def opens_heading(line: str, heading: tuple) -> bool:
    """The matcher's hit is a heading line, not a tail found inside an entry: the line opens with a
    heading's own words and names a court of three letters or more. The route only strips what the
    matcher finds, so a false hit there costs a few characters; here it would switch the section."""
    return (lxnt.normalise(line[:heading[2]]).startswith(HEADING_OPENS)
            and len(lxnt.normalise(heading[1])) >= 3)


def other_court(crosswalk: dict):
    """``(a, b) -> bool``: the two German court names are different 1910 courts — both resolve
    through the route's crosswalk, to different Czech courts. A name that does not resolve, or
    resolves to the same court, is a spelling of the crop's own court and changes nothing."""
    def differ(a: str, b: str) -> bool:
        ca, cb = lxdn.court_name(a, crosswalk)[0], lxdn.court_name(b, crosswalk)[0]
        return bool(ca and cb and lxnt.normalise(ca) != lxnt.normalise(cb))
    return differ


def resplit_sections(read: list, find, differ=lambda a, b: False) -> tuple[list[lxnt.Section], list[dict]]:
    """:func:`sections_from_segments` with the split it does not make: a line the matcher reads as
    a heading of the other kind, or of another court (``differ``), opens a section of its own, and a
    carried segment continues whatever section its court's previous segment ended in.
    Returns ``(sections, splits)``."""
    merged: dict[tuple, lxnt.Section] = {}
    order: list[tuple] = []
    last_kind: dict[str, str] = {}                 # court key -> kind in force at its last line
    splits = []

    def section(page, kind, district):
        key = (page, kind, lxnt.normalise(district))
        if key not in merged:
            merged[key] = lxnt.Section(kind=kind, district=district, page=page, lines=[])
            order.append(key)
        return merged[key]

    for segment, lines in read:
        kind, district = segment.kind, segment.district
        if segment.carried and last_kind.get(segment.key, kind) != kind:
            splits.append({"page": segment.page, "region": segment.region, "order": segment.order,
                           "court_de": district, "from": kind, "to": last_kind[segment.key],
                           "how": "carried"})
            kind = last_kind[segment.key]
        current = section(segment.page, kind, district)
        for line in lines:
            heading = find(line)
            # Split on a change of kind, or on a heading of another 1910 court. A court name read
            # off the reader's text that is only a spelling of the crop's court changes nothing:
            # the court stays the crop's, as in the route.
            real = bool(heading) and heading[0] in ("parts", "offices") and opens_heading(line, heading)
            new_court = real and differ(heading[1], district)
            if real and (heading[0] != kind or new_court):
                splits.append({"page": segment.page, "region": segment.region, "order": segment.order,
                               "court_de": district, "heading_court": heading[1],
                               "from": kind, "to": heading[0],
                               "how": "court" if new_court else "heading"})
                kind = heading[0]
                district = heading[1] if new_court else district
                current = section(segment.page, kind, district)
            tail = line[heading[2]:] if heading else line
            if tail.strip():
                current.lines.append(tail)
        last_kind[segment.key] = kind
        last_kind[lxnt.normalise(district)] = kind
    return [merged[k] for k in order], splits


def decoded_inventory(sections: list, edition: str) -> list[dict]:
    """The sections decoded and inventoried exactly as ``run_r5_german_notes.assemble`` does."""
    profile = lxvol.load_profile(rf.GERMAN[edition])
    abbreviations = lxin.load_abbreviations(profile.abbreviations, lang="de", volume=profile.slug)
    lxdn.relabel_sections(sections, lxdn.load_court_crosswalk())
    records, _entries = lxin.decode_sections(sections, abbreviations)
    return lxin.build_inventory(records, [])


def _key(row: dict) -> tuple:
    return tuple(str(row.get(k, "")) for k in CLAIM)


def resplit_reader(edition: str, label: str) -> dict:
    """One reader: the rows the re-split adds to its inventory, and the rows it would lose (which
    must be none — the re-split only moves lines out of parts sections)."""
    profile = lxvol.load_profile(rf.GERMAN[edition])
    differ = other_court(lxdn.load_court_crosswalk())
    sections, splits = resplit_sections(load_read(label), profile.find_heading, differ)
    new = decoded_inventory(sections, edition)
    old = _rows("inventory", label)
    have = {_key(r) for r in old}
    added = [r for r in new if _key(r) not in have]
    lost = have - {_key(r) for r in new}
    return {"label": label, "splits": splits, "added": added, "lost": sorted(lost),
            "old_rows": len(old), "new_rows": len(new), "inventory": new}


def run(edition: str, pages: str = "full", out: Path = OUT) -> dict:
    """Every reader of one German edition; writes ``resplit/inventory_<label>.csv`` (the added
    rows), ``splits_<label>.csv`` and ``summary.json`` under ``output/recheck/<edition>/``."""
    if edition not in GERMAN:
        raise ValueError(f"{edition}: the re-split is for the German editions (the Czech volume "
                         f"is cut by its text layer, and lost no offices lines)")
    folder = out / edition / "resplit"
    folder.mkdir(parents=True, exist_ok=True)
    summary = {"module": __version__, "edition": edition, "pages": pages,
               "run": time.strftime("%Y-%m-%d %H:%M:%S"), "readers": {}}
    for label in readers(edition, pages):
        r = resplit_reader(edition, label)
        _write(folder / f"added_{label}.csv", r["added"])
        _write(folder / f"inventory_{label}.csv", r["inventory"])
        _write(folder / f"lost_{label}.csv", [dict(zip(CLAIM, k)) for k in r["lost"]])
        _write(folder / f"splits_{label}.csv", r["splits"])
        summary["readers"][label] = {
            "splits": dict(Counter(s["how"] for s in r["splits"])),
            "rows_added": len(r["added"]), "rows_lost": len(r["lost"]),
            "entries_added": len({_key(x)[:4] for x in r["added"]}),
            "pages_added": len({x["page"] for x in r["added"]})}
        print(f"  {label}: {summary['readers'][label]}")
    (folder / "summary.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return summary


def _write(path: Path, rows: list[dict]) -> None:
    fields = sorted({k for r in rows for k in r}) if rows else ["page"]
    with open(path, "w", encoding="utf-8", newline="") as fh:
        w = csv.DictWriter(fh, fieldnames=fields, lineterminator="\n")
        w.writeheader()
        w.writerows(rows)


def consensus(edition: str, pages: str = "full", out: Path = OUT) -> dict:
    """The added claims across readers, settled as the route settles them (``routed_claims``):
    a claim every reader makes stands; one only some make is disputed, for Opus. Only entries
    every reader's re-split reached are compared."""
    folder = out / edition / "resplit"
    labels = readers(edition, pages)
    claims, touched = {}, set()
    for label in labels:
        with open(folder / f"inventory_{label}.csv", encoding="utf-8", newline="") as fh:
            claims[label] = {_key(r) for r in csv.DictReader(fh) if r.get("code")}
        with open(folder / f"added_{label}.csv", encoding="utf-8", newline="") as fh:
            touched |= {_key(r)[:4] for r in csv.DictReader(fh)}
    # The entries the re-split changed for any reader, settled as the route settles claims: where
    # every reader's whole re-split inventory has the entry, a claim all make stands and one only
    # some make is disputed (``routed_claims``); elsewhere the reader of record's claim stands
    # alone (the route's "not asked: reader of record"). A reader may have left the heading line
    # out of its text, and then its notes cannot be split there.
    record = labels[0]
    entries = touched & set.intersection(*({c[:4] for c in cs} for cs in claims.values()))
    shared = {l: {c for c in cs if c[:4] in entries} for l, cs in claims.items()}
    union, agreed = set().union(*shared.values()), set.intersection(*shared.values())
    rows = [dict(zip(CLAIM, c), status="agreed" if c in agreed else "disputed",
                 support=sum(c in s for s in shared.values())) for c in sorted(union)]
    alone = {c for c in claims[record] if c[:4] in touched - entries}
    rows += [dict(zip(CLAIM, c), status="reader of record", support=1) for c in sorted(alone)]
    _write(folder / "consensus.csv", rows)
    return {"entries_touched": len(touched), "entries_every_reader": len(entries),
            "entries_reader_of_record": len({c[:4] for c in alone}),
            "agreed": len(agreed), "disputed": len(union - agreed), "reader_of_record": len(alone)}


#: The symbols the Czech test reports on: common enough to give a rate on a few hundred places.
TESTED = ("[fire-brigade]", "[school]", "[midwife]", "[rc-parish]", "[post]", "[telegraph]",
          "[gendarme]", "[rail-station]", "[doctor]", "[rc-filial]", "[castle]", "[poorhouse]")


def czech_test(pages: str = "full", out: Path = OUT, statuses=("agreed", "reader of record"),
               folder: str = "resplit") -> pd.DataFrame:
    """German Böhmen's re-split claims against the Czech volume on the municipalities they fill.

    The claims are rolled up to the municipality (parts included, as in
    ``check_offices_coverage.bohemia_editions``) and paired with the Czech municipality through
    the crosswalk's ``cz_muni_seq``; a municipality counts only if the route gave it no pictogram
    in German before. Per symbol: both, Czech only, German only, agreement = both / either.
    ``folder``: whose ``consensus.csv`` — ``resplit`` (§3) or ``reread`` (§4, ``offices_reread``)."""
    reg = ROOT / "data" / "final" / "register"
    places = pd.read_parquet(reg / "register_places.parquet")
    inst = pd.read_parquet(reg / "register_institutions.parquet")
    inv = pd.read_csv(rf.out_dir("de-bohemia", pages) / "final_inventory.csv", dtype=str,
                      keep_default_na=False)
    court_of = dict(inv[["district", "court_id"]].drop_duplicates().itertuples(index=False))
    rows = pd.read_csv(out / "de-bohemia" / folder / "consensus.csv", dtype=str, keep_default_na=False)
    rows = rows[rows.status.isin(statuses) & rows.code.str.startswith("[")]
    rows = rows.assign(court_key=rows.district.map(court_of), muni_seq=rows.muni_seq.astype(int))
    unmapped = int(rows.court_key.isna().sum())
    new = rows.dropna(subset=["court_key"]).groupby(["court_key", "muni_seq"]).code.agg(set)
    de = places[(places.edition == "de-bohemia") & (places.place_level == "municipality")
                & places.cz_muni_seq.notna()]
    pic = inst[inst.source == "pictogram"]
    had = set(zip(pic[pic.edition == "de-bohemia"].court_key, pic[pic.edition == "de-bohemia"].muni_seq))
    cz = pic[pic.edition == "cz"].groupby(["court_key", "muni_seq"]).mark.agg(set)
    pairs = [(new[(k, m)], cz.get((k, int(c)), set())) for k, m, c in
             zip(de.court_key, de.muni_seq, de.cz_muni_seq)
             if (k, m) in new.index and (k, m) not in had]
    out_rows = []
    for mark in TESTED:
        both = sum(mark in d and mark in c for d, c in pairs)
        c_only = sum(mark in c and mark not in d for d, c in pairs)
        d_only = sum(mark in d and mark not in c for d, c in pairs)
        out_rows.append({"mark": mark, "both": both, "cz_only": c_only, "de_only": d_only,
                         "agree": round(both / max(1, both + c_only + d_only), 3)})
    table = pd.DataFrame(out_rows)
    table.attrs.update(municipalities=len(pairs), unmapped_claims=unmapped)
    return table


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--edition", default="all", choices=(*GERMAN, "all"))
    args = ap.parse_args(argv)
    for ed in GERMAN if args.edition == "all" else (args.edition,):
        print(ed)
        run(ed)
        print("  consensus:", consensus(ed))
    return 0
