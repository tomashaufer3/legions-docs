How it works

The list of parts prints a pictogram beside each place saying what it is — a town (město, Stadt), a market town (městys, Markt), a village (ves, Dorf), a hamlet, a mill, a manor farm, a gamekeeper's lodge … (Výklad značek / Zeichenerklärung, p. XI). The route settles each type as it settles an institution — all four readings alike, else Opus, else the author (D-131) — and writes final_types.csv; notebook 17 carries it through every step of the merge, the post-1910 renumbering included, to final_types_rechecked.csv. The register module never read it (it is frozen, D-119), so the types reached data/final only as the reader of record's raw text in register_places.components. This adds them, settled, in two forms:

  • register_place_types — one row per place and type, keyed as the institutions are (the place key
  • code), with the type's names in the volume's two languages and in English (reference/places/institution_abbreviations.csv), how it was settled and how many readings named it. Only the codes that file classes as settlement or industry — the mill, the brickworks and the factory are pictograms of what a place is in the list of parts (a mill is a settlement), though the same file files them under industry. The types file also carries the connector [besides] and a few institutions read into the type column ([telegraph], [gendarme], [jewish] …), which the institutions table holds where they belong; they are counted, not kept.
  • place_status on register_places and register_municipalities — the place's legal status from its own types: town where it carries [mesto], else market_town for [mestys], else village for [ves], else empty (a part, a hamlet, a single house …). 22 municipalities print both a market town's and a village's pictogram and 21 a town's and a village's; the higher status is the place's, as the volume prints the status of the municipality and the village of its seat.

A type whose dispute no call was sent for (not asked) is kept and labelled so, and gives no status.

What it is blind to. The types are read off the apparatus's list of parts (Části osad, Ortsbestandteile), which names a municipality only where it has parts: in the Czech volume 3,059 of 7,760 municipalities. So an empty place_status means not in the list of parts, not not a village; a town with no parts has no status here. The table's name cell prints the same pictogram for every municipality, but the German reader writes it as [?] and the Czech reader often misreads the glyph, so that is a reading of its own, not done here.

def table(run: reg.RunInput, recheck: Path, names: pd.DataFrame | None = None) -> tuple[pd.DataFrame, dict]:
    """One volume's settled types, keyed as its institutions are, and what was left out."""
    path = Path(recheck) / run.edition / FILE
    if not path.exists():
        return pd.DataFrame(columns=COLUMNS), {"edition": run.edition, "file": "missing"}
    raw = pd.read_csv(path, dtype=str, keep_default_na=False)
    names = names if names is not None else labels()
    units = reg._Units(run)
    raw["edition"] = run.edition
    raw["court_key"] = [units.key(p, d, c) for p, d, c in zip(raw["page"], raw["district"], raw["court_id"])]
    raw["muni_seq"] = pd.to_numeric(raw["muni_seq"], errors="coerce")
    raw["settlement_seq"] = pd.to_numeric(raw["settlement_seq"].replace("", "0"), errors="coerce")
    unkeyed = int(raw["muni_seq"].isna().sum() + raw["settlement_seq"].isna().sum())
    raw = raw.dropna(subset=["muni_seq", "settlement_seq"])
    raw = raw.astype({"muni_seq": int, "settlement_seq": int})
    raw["settled"] = raw["rule"].map(_settled)
    keep = raw["code"].isin(set(names["code"]))
    other = raw.loc[~keep, "code"].value_counts().to_dict()
    out = raw[keep].merge(names, on="code", how="left", validate="m:1")
    order = {name: k for k, (_p, name) in enumerate(SETTLED)}
    out = (out.assign(_rank=out["settled"].map(order).fillna(len(order)))
              .sort_values(KEY + ["_rank"], kind="stable"))
    dupes = int(out.duplicated(KEY).sum())                 # one type named twice for a place: the strongest kept
    out = out.drop_duplicates(KEY).drop(columns="_rank").reindex(columns=COLUMNS).reset_index(drop=True)
    return out, {"edition": run.edition, "rows": len(out), "not_a_place_type": other,
                 "same_type_twice": dupes, "unkeyed": unkeyed}

    """``place_status`` per place key, from the place's own settled types (not the *not asked* ones)."""

def add(tables: dict, runs: list, recheck: Path) -> dict:
    """``tables`` with ``register_place_types`` and ``place_status`` added; a report of what was
    left out and how many places got a status. Every type row must name a place of the register."""
    names = labels()
    parts, reports = [], []
    for run in runs:
        frame, report = table(run, recheck, names)
        parts.append(frame)
        reports.append(report)
    types = pd.concat(parts, ignore_index=True) if parts else pd.DataFrame(columns=COLUMNS)
    places = tables["register_places"]
    known = places[reg.PLACE_KEY].astype({"court_key": str}).drop_duplicates()
    joined = types.astype({"court_key": str}).merge(known, on=reg.PLACE_KEY, how="left", indicator=True,
                                                     validate="m:1")
    orphan = joined["_merge"] == "left_only"
    for rep in reports:
        rep["no_register_place"] = int((orphan & (joined["edition"] == rep["edition"])).sum())
    types = joined.loc[~orphan, COLUMNS].reset_index(drop=True)
    assert not types.duplicated(KEY).any(), "register_place_types: duplicate key"
    st = status(types)
    n0 = len(places)
    places = places.drop(columns="place_status", errors="ignore").merge(
        st, on=reg.PLACE_KEY, how="left", validate="1:1")
    assert len(places) == n0, "place_status: the merge changed the row count"
    places["place_status"] = places["place_status"].fillna("")
    muni = tables["register_municipalities"]
    if {"edition", "court_id", "muni_seq"} <= set(muni.columns):
        own = places.loc[places["settlement_seq"] == 0, ["edition", "court_key", "muni_seq", "place_status"]]
        m0 = len(muni)
        muni = (muni.drop(columns="place_status", errors="ignore")
                    .assign(_ck=muni["court_id"].astype("string").fillna(""))
                    .merge(own.rename(columns={"court_key": "_ck"}), on=["edition", "_ck", "muni_seq"],
                           how="left", validate="m:1").drop(columns="_ck"))
        assert len(muni) == m0, "place_status: the municipalities merge changed the row count"
        muni["place_status"] = muni["place_status"].fillna("")
    out = {**tables, "register_place_types": types, "register_places": places, "register_municipalities": muni}
    by = places.loc[places["settlement_seq"] == 0, "place_status"].value_counts().to_dict()
    out["_meta"] = {**tables.get("_meta", {}),             # a report, not a table: under _meta, which callers skip
                    "place_types": {"volumes": reports, "rows": len(types), "municipality_status": by}}
    return out

__version__ = "0.1.0"   # 0.1.0 the settled types as a table and the legal status as a column (D-216)

from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT), str(ROOT / "utils" / "lexikon_1910")]

FILE = "final_types_rechecked.csv"
KEY = reg.PLACE_KEY + ["code"]
#: The abbreviation categories whose pictograms say what a place is (the list of parts' type column).
CATEGORIES = ("settlement", "industry")
#: The legal statuses, highest first: the code that gives each.
STATUS = (("[mesto]", "town"), ("[mestys]", "market_town"), ("[ves]", "village"))
#: How the route labels a settled type (``run_frozen.final_types``), strongest first.
SETTLED = (("author", "author"), ("opus", "opus"), ("agreed", "agreed"), ("not asked", "not_asked"))
COLUMNS = KEY + ["type_en", "type_cs", "type_de", "settled", "support", "rule"]

def _settled(rule: str) -> str:
    rule = str(rule).strip().lower()
    return next((name for prefix, name in SETTLED if rule.startswith(prefix)), "other")

    left out and how many places got a status. Every type row must name a place of the register."""
    names = labels()
    parts, reports = [], []
    for run in runs:
        frame, report = table(run, recheck, names)
        parts.append(frame)
        reports.append(report)
    types = pd.concat(parts, ignore_index=True) if parts else pd.DataFrame(columns=COLUMNS)
    places = tables["register_places"]
    known = places[reg.PLACE_KEY].astype({"court_key": str}).drop_duplicates()
    joined = types.astype({"court_key": str}).merge(known, on=reg.PLACE_KEY, how="left", indicator=True,
                                                     validate="m:1")
    orphan = joined["_merge"] == "left_only"
    for rep in reports:
        rep["no_register_place"] = int((orphan & (joined["edition"] == rep["edition"])).sum())
    types = joined.loc[~orphan, COLUMNS].reset_index(drop=True)
    assert not types.duplicated(KEY).any(), "register_place_types: duplicate key"
    st = status(types)
    n0 = len(places)
    places = places.drop(columns="place_status", errors="ignore").merge(
        st, on=reg.PLACE_KEY, how="left", validate="1:1")
    assert len(places) == n0, "place_status: the merge changed the row count"
    places["place_status"] = places["place_status"].fillna("")
    muni = tables["register_municipalities"]
    if {"edition", "court_id", "muni_seq"} <= set(muni.columns):
        own = places.loc[places["settlement_seq"] == 0, ["edition", "court_key", "muni_seq", "place_status"]]
        m0 = len(muni)
        muni = (muni.drop(columns="place_status", errors="ignore")
                    .assign(_ck=muni["court_id"].astype("string").fillna(""))
                    .merge(own.rename(columns={"court_key": "_ck"}), on=["edition", "_ck", "muni_seq"],
                           how="left", validate="m:1").drop(columns="_ck"))
        assert len(muni) == m0, "place_status: the municipalities merge changed the row count"
        muni["place_status"] = muni["place_status"].fillna("")
    out = {**tables, "register_place_types": types, "register_places": places, "register_municipalities": muni}
    by = places.loc[places["settlement_seq"] == 0, "place_status"].value_counts().to_dict()
    out["_meta"] = {**tables.get("_meta", {}),             # a report, not a table: under _meta, which callers skip
                    "place_types": {"volumes": reports, "rows": len(types), "municipality_status": by}}
    return out


def write(tables: dict, out_dir: Path) -> None:
    """The type table and the two tables that gained ``place_status``, as the register writes its own."""
    for name in ("register_place_types", "register_places", "register_municipalities"):
        tables[name].to_parquet(Path(out_dir) / f"{name}.parquet", index=False)
        tables[name].to_csv(Path(out_dir) / f"{name}.csv", index=False, encoding="utf-8")
