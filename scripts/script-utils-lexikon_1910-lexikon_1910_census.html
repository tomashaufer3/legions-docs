How it works

The register's own identities — sexes to total, confessions to total, languages plus foreigners to total, settlements to their municipality — are all internal. They are strong, but they share one blind spot: they cannot see a municipality the parser never emitted, or a block it invented. An okres that is missing three villages satisfies every one of them, because each surviving row is individually consistent. Nothing inside the page can detect a row that is not on it.

This module closes that gap by summing the transcribed municipality rows across a whole okres and comparing the totals to census_okres_1910, which reports the same 1910 enumeration at okres level from the Rumpler-Seger workbooks. Where the two disagree, either the transcription lost or duplicated a row, or the confession grouping below is wrong — and the two are told apart by how they disagree: a lost row is a deficit in every column at once, a bad grouping is an exact offset in one.

This is not independent evidence about 1910. Both sides descend from the same census, so agreement validates our reading of the register, not the register itself. That is exactly what is wanted here, but it means the reconciliation must never be cited as corroboration of the census.

Two things about the comparison need stating rather than assuming:

  • Only municipality rows are summed. Settlements are subdivisions of their municipality and are already counted in it — the same double-count that invariant I4 exists to check. Summing both would inflate every total by roughly a factor of two and would do it consistently, which is the kind of error that looks like a result.
  • It needs a complete okres. A partial okres under-sums by construction, so a run that covers half a district would report a spurious deficit in every column. Coverage is therefore a precondition, checked against the anchoring, not a caveat in the output.

class Reconciliation:
    """What one okres reconciled to, column by column."""

    okres_id: int
    okres_cs: str
    complete: bool
    n_blocks: int
    n_expected: int
    #: Distinct known obce actually anchored. This, not ``n_blocks``, is what
    #: coverage means: a page can carry more blocks than the okres has obce (a
    #: settlement row wrongly promoted) while still being short of real ones, so
    #: counting blocks would call an incomplete okres complete.
    n_found: int = 0
    #: District summary rows excluded from the sums. Never zero on a complete
    #: okres — the register always opens one — so a zero here on a full district
    #: means the detector missed them and the totals are inflated.
    n_summary: int = 0
    findings: list[dict] = field(default_factory=list)
    unreadable: dict[str, int] = field(default_factory=dict)

    @property
    def missing(self) -> int:
        return max(self.n_expected - self.n_found, 0)

def register_totals(blocks) -> tuple[dict[str, int], dict[str, int], int]:
    """Sum the municipality rows of an okres, column by column.

    Returns ``(totals, unreadable, n_summary_skipped)``. Two kinds of row are
    excluded and both would otherwise double-count: settlements, which are already
    counted inside their municipality, and district summary rows, which are already
    counted across all of them.

    A column with any unreadable cell still reports a total, but the count of cells
    that could not be read travels beside it — a sum that silently skipped nine bad
    cells is not a total, and a reconciliation that treats it as one would report
    the transcription's own gaps as a census disagreement.
    """
    from utils.lexikon_1910 import lexikon_1910 as lx

    rows = [b.municipality for b in blocks]
    kept = [r for r in rows if not is_summary_row(r)]
    totals: dict[str, int] = {}
    unreadable: dict[str, int] = {}
    for column in lx.INT_COLS:
        running = 0
        bad = 0
        for row in kept:
            value, status = lx.parse_int_cell(row.get(column))
            if status == "bad":
                bad += 1
            elif status == "ok":
                running += value
        totals[column] = running
        unreadable[column] = bad
    return totals, unreadable, len(rows) - len(kept)

def census_targets(row) -> dict[str, int]:
    """The expected okres totals, in the register's own column names."""
    targets: dict[str, int] = {}
    for register_col, census_cols in {**DIRECT_MAP, **CONFESSION_MAP,
                                      **LANGUAGE_MAP}.items():
        present = [c for c in census_cols if c in row and _isfinite(row[c])]
        if not present:
            continue
        targets[register_col] = int(sum(float(row[c]) for c in present))
    return targets

def house_candidates(row) -> dict[str, int]:
    """The two readings of the census house count, both reported, neither graded."""
    def band_sum(suffix: str) -> int:
        cols = [f"pop_{b}_houses_{suffix}" for b in HOUSE_BANDS]
        return int(sum(float(row[c]) for c in cols
                       if c in row and _isfinite(row[c])))

    inhabited = band_sum("inhabited")
    return {"inhabited": inhabited,
            "inhabited+uninhabited": inhabited + band_sum("uninhabited")}

def reconcile(blocks, census_row, okres_id: int, okres_cs: str = "",
              n_expected: int = 0, n_found: int = 0, complete: bool | None = None,
              tolerance: int = 0) -> Reconciliation:
    """Compare an okres's transcribed municipality totals to the census.

    ``complete`` says whether every municipality the okres is known to contain was
    actually found. When it is False the columns are still compared and reported,
    but every one of them is marked ``incomplete`` rather than ``fail``: a partial
    okres under-sums by construction, and letting that score as a disagreement
    would bury the real ones under arithmetic that was never going to balance.
    """
    totals, unreadable, n_summary = register_totals(blocks)
    targets = census_targets(census_row)
    if complete is None:
        complete = bool(n_expected) and n_found >= n_expected

    findings: list[dict] = []
    for column, expected in sorted(targets.items()):
        got = totals.get(column, 0)
        delta = got - expected
        if not complete:
            verdict = "incomplete"
        elif unreadable.get(column):
            # An unreadable cell means the sum is missing a term of unknown size,
            # so the comparison cannot distinguish a census disagreement from the
            # transcription's own hole. Say so instead of guessing which.
            verdict = "unparseable"
        elif abs(delta) <= tolerance:
            verdict = "pass"
        else:
            verdict = "fail"
        findings.append({
            "okres_id": okres_id, "okres_cs": okres_cs, "column": column,
            "check": f"I6_okres:{column}", "verdict": verdict,
            "register": got, "census": expected, "delta": delta,
            "unreadable_cells": unreadable.get(column, 0),
            "detail": "" if verdict == "pass" else
                      f"register {got} vs census {expected} ({delta:+d})",
        })

    # Houses: reported against both candidate definitions, graded against neither.
    houses = house_candidates(census_row)
    for label, expected in houses.items():
        got = totals.get("houses", 0)
        findings.append({
            "okres_id": okres_id, "okres_cs": okres_cs, "column": "houses",
            "check": f"I6_okres:houses[{label}]", "verdict": "ungraded",
            "register": got, "census": expected, "delta": got - expected,
            "unreadable_cells": unreadable.get("houses", 0),
            "detail": f"register {got} vs census {expected} "
                      f"({got - expected:+d}) — definition unsettled",
        })

    return Reconciliation(okres_id=okres_id, okres_cs=okres_cs, complete=complete,
                          n_blocks=len(blocks), n_expected=n_expected,
                          n_found=n_found, n_summary=n_summary,
                          findings=findings, unreadable=unreadable)

def diagnose(reconciliation: Reconciliation) -> str:
    """Name the shape of a failure, where the shape is unambiguous.

    A lost or duplicated row moves *every* column in the same direction at once; a
    wrong confession grouping moves one column by an exact amount and leaves the
    rest untouched. Distinguishing them is the whole reason for comparing column by
    column rather than on population alone.
    """
    graded = [f for f in reconciliation.findings if f["verdict"] in ("pass", "fail")]
    if not graded:
        return "not graded"
    failed = [f for f in graded if f["verdict"] == "fail"]
    if not failed:
        return "all graded columns reconcile exactly"
    if len(failed) == len(graded) and len({f["delta"] < 0 for f in failed}) == 1:
        direction = "short of" if failed[0]["delta"] < 0 else "over"
        return (f"every column is {direction} the census — consistent with "
                f"{'missing' if failed[0]['delta'] < 0 else 'duplicated'} rows, "
                f"not with a column mapping error")
    if len(failed) == 1:
        f = failed[0]
        return (f"only {f['column']} disagrees ({f['delta']:+d}) — consistent with "
                f"a grouping error in that column, not with a lost row")
    names = ", ".join(f["column"] for f in failed)
    return f"{len(failed)} of {len(graded)} columns disagree: {names}"

def lexikon_1910_census_report(reconciliations) -> str:
    """Ledger over one or more okresy."""
    lines = [f"lexikon_1910_census v{__version__} — okres reconciliation", ""]
    if not reconciliations:
        return "\n".join(lines + ["  no okres was complete enough to reconcile"])

    lines.append(f"  {'okres':22s} {'blocks':>12s}  {'pass':>5s} {'fail':>5s} "
                 f"{'unrd':>5s} {'incm':>5s}")
    lines.append("  " + "-" * 62)
    graded_total = passed_total = 0
    for rec in reconciliations:
        counts: dict[str, int] = {}
        for f in rec.findings:
            if f["verdict"] == "ungraded":
                continue
            counts[f["verdict"]] = counts.get(f["verdict"], 0) + 1
        coverage = (f"{rec.n_found}/{rec.n_expected}" if rec.n_expected
                    else str(rec.n_blocks))
        lines.append(f"  {(rec.okres_cs or rec.okres_id)!s:22.22s} {coverage:>12s}  "
                     f"{counts.get('pass', 0):5d} {counts.get('fail', 0):5d} "
                     f"{counts.get('unparseable', 0):5d} "
                     f"{counts.get('incomplete', 0):5d}")
        graded_total += counts.get("pass", 0) + counts.get("fail", 0)
        passed_total += counts.get("pass", 0)
        note = diagnose(rec)
        if rec.complete:
            lines.append(f"      {note}")
        else:
            lines.append(f"      incomplete — {rec.missing} of {rec.n_expected} "
                         f"known obce not anchored on these pages "
                         f"({rec.n_blocks} blocks read); columns reported, "
                         f"none graded")
            # A near-complete okres is still worth reading: the shortfall in
            # pop_total is an estimate of how big the missing municipalities are,
            # which is what you search for to find them. A shortfall that is
            # implausible for that many villages says the gap is not coverage.
            if 0 < rec.missing <= 3:
                deficit = next((f["delta"] for f in rec.findings
                                if f["column"] == "pop_total"), None)
                if deficit is not None and deficit < 0:
                    average = (f", {-deficit // rec.missing} each on average"
                               if rec.missing > 1 else "")
                    noun = "obec" if rec.missing == 1 else "obce"
                    lines.append(f"      the {rec.missing} missing {noun} account "
                                 f"for {-deficit} inhabitants{average}")

    lines.append("  " + "-" * 62)
    rate = f"{passed_total / graded_total:.1%}" if graded_total else "--"
    lines.append(f"  {passed_total}/{graded_total} graded columns reconcile  {rate}")
    lines.append("")
    lines.append("  houses is reported against both census definitions and graded "
                 "against neither;")
    lines.append(f"  {', '.join(NOT_IN_CENSUS)} are not carried at okres level at all.")
    lines.append("  incomplete okresy under-sum by construction — coverage, not a "
                 "disagreement.")
    return "\n".join(lines)

# --------------------------------------------------------------------------- #
# Register column -> census column
# --------------------------------------------------------------------------- #
#
# The register prints four confession columns; the census reports seventeen. The
# grouping below is the standard Austrian one, and the two marginal calls are
# named here rather than buried, because each is a decision that shifts a column:
#
#   * **Greek and Armenian Catholics count as Catholic.** They are Catholic
#     *rites*, which is what the register's "katolické" names. In Bohemia they are
#     numerically negligible, so this choice is nearly free here — but it would not
#     be in Galicia or the Bukovina, and this module is written to be moved.
#   * **Old Catholics do NOT.** Altkatholiken are a separate denomination that
#     broke with Rome in 1870, not a rite, so they fall to "other". Bohemia is
#     where they are least negligible, which is exactly why the call is explicit.
#
# If either is wrong the reconciliation shows it in the cleanest possible form: a
# constant offset in one column, equal to that group's own count, with every other
# column exact.

CONFESSION_MAP: dict[str, tuple[str, ...]] = {
    "conf_catholic": ("rel_roman_catholic", "rel_greek_catholic",
                      "rel_armenian_catholic"),
    # Augsburger + Helvetischer Bekenntnis — together the census's "evangelisch",
    # confirmed against census_codebook.csv.
    "conf_evangelical": ("rel_lutheran", "rel_calvinist"),
    "conf_israelite": ("rel_jewish",),
    "conf_other": ("rel_old_catholic", "rel_greek_orthodox", "rel_armenian_orthodox",
                   "rel_moravian", "rel_anglican", "rel_mennonite", "rel_unitarian",
                   "rel_lipovan", "rel_muslim", "rel_other", "rel_none"),
}

#: The register's ``lang_czech`` is "česko-moravsko-slovácké", which the census
#: carries as the single category "böhmisch-mährisch-slowakisch" — a one-to-one
#: match, not a grouping. Everything other than German and Czech falls to
#: ``lang_other``.
LANGUAGE_MAP: dict[str, tuple[str, ...]] = {
    "lang_german": ("lang_german",),
    "lang_czech": ("lang_czech",),
    "lang_other": ("lang_polish", "lang_ruthenian", "lang_slovene",
                   "lang_serbocroat", "lang_italian_ladin", "lang_romanian",
                   "lang_magyar"),
}

#: Straight one-to-one columns.
DIRECT_MAP: dict[str, tuple[str, ...]] = {
    "pop_total": ("pop_total",),
    "foreigners": ("foreign_nationals_present",),
}

#: The census splits houses by locality-size band and by whether they were
#: inhabited; the register prints one number. Which of the two candidate totals it
#: is, is a definitional question this module does not pretend to have settled — so
#: both are computed and reported side by side, and ``houses`` is graded against
#: neither until the volume's preface says which. See ``report`` below.
HOUSE_BANDS = ("lt500", "500_2000", "2001_5000", "5001_10000",
               "10001_20000", "gt20000")

#: Columns the register prints that the okres census does not carry at all. Listed
#: so their absence is visible in the ledger rather than looking like a pass.
NOT_IN_CENSUS = ("area_ha", "pop_male", "pop_female")

# Totals

@dataclass
class Reconciliation:
    """What one okres reconciled to, column by column."""

    okres_id: int
    okres_cs: str
    complete: bool
    n_blocks: int
    n_expected: int
    #: Distinct known obce actually anchored. This, not ``n_blocks``, is what
    #: coverage means: a page can carry more blocks than the okres has obce (a
    #: settlement row wrongly promoted) while still being short of real ones, so
    #: counting blocks would call an incomplete okres complete.
    n_found: int = 0
    #: District summary rows excluded from the sums. Never zero on a complete
    #: okres — the register always opens one — so a zero here on a full district
    #: means the detector missed them and the totals are inflated.
    n_summary: int = 0
    findings: list[dict] = field(default_factory=list)
    unreadable: dict[str, int] = field(default_factory=dict)

    @property
    def missing(self) -> int:
        return max(self.n_expected - self.n_found, 0)


def register_totals(blocks) -> tuple[dict[str, int], dict[str, int], int]:
    """Sum the municipality rows of an okres, column by column.

    Returns ``(totals, unreadable, n_summary_skipped)``. Two kinds of row are
    excluded and both would otherwise double-count: settlements, which are already
    counted inside their municipality, and district summary rows, which are already
    counted across all of them.

    A column with any unreadable cell still reports a total, but the count of cells
    that could not be read travels beside it — a sum that silently skipped nine bad
    cells is not a total, and a reconciliation that treats it as one would report
    the transcription's own gaps as a census disagreement.
    """
    from utils.lexikon_1910 import lexikon_1910 as lx

    rows = [b.municipality for b in blocks]
    kept = [r for r in rows if not is_summary_row(r)]
    totals: dict[str, int] = {}
    unreadable: dict[str, int] = {}
    for column in lx.INT_COLS:
        running = 0
        bad = 0
        for row in kept:
            value, status = lx.parse_int_cell(row.get(column))
            if status == "bad":
                bad += 1
            elif status == "ok":
                running += value
        totals[column] = running
        unreadable[column] = bad
    return totals, unreadable, len(rows) - len(kept)


def census_targets(row) -> dict[str, int]:
    """The expected okres totals, in the register's own column names."""
    targets: dict[str, int] = {}
    for register_col, census_cols in {**DIRECT_MAP, **CONFESSION_MAP,
                                      **LANGUAGE_MAP}.items():
        present = [c for c in census_cols if c in row and _isfinite(row[c])]
        if not present:
            continue
        targets[register_col] = int(sum(float(row[c]) for c in present))
    return targets


def house_candidates(row) -> dict[str, int]:
    """The two readings of the census house count, both reported, neither graded."""
    def band_sum(suffix: str) -> int:
        cols = [f"pop_{b}_houses_{suffix}" for b in HOUSE_BANDS]
        return int(sum(float(row[c]) for c in cols
                       if c in row and _isfinite(row[c])))

    inhabited = band_sum("inhabited")
    return {"inhabited": inhabited,
            "inhabited+uninhabited": inhabited + band_sum("uninhabited")}


def _isfinite(value) -> bool:
    try:
        return value == value and value not in (None, "")   # NaN-safe
    except TypeError:                                       # pragma: no cover
        return False


# --------------------------------------------------------------------------- #
# The comparison
# --------------------------------------------------------------------------- #

def reconcile(blocks, census_row, okres_id: int, okres_cs: str = "",
              n_expected: int = 0, n_found: int = 0, complete: bool | None = None,
              tolerance: int = 0) -> Reconciliation:
    """Compare an okres's transcribed municipality totals to the census.

    ``complete`` says whether every municipality the okres is known to contain was
    actually found. When it is False the columns are still compared and reported,
    but every one of them is marked ``incomplete`` rather than ``fail``: a partial
    okres under-sums by construction, and letting that score as a disagreement
    would bury the real ones under arithmetic that was never going to balance.
    """
    totals, unreadable, n_summary = register_totals(blocks)
    targets = census_targets(census_row)
    if complete is None:
        complete = bool(n_expected) and n_found >= n_expected

    findings: list[dict] = []
    for column, expected in sorted(targets.items()):
        got = totals.get(column, 0)
        delta = got - expected
        if not complete:
            verdict = "incomplete"
        elif unreadable.get(column):
            # An unreadable cell means the sum is missing a term of unknown size,
            # so the comparison cannot distinguish a census disagreement from the
            # transcription's own hole. Say so instead of guessing which.
            verdict = "unparseable"
        elif abs(delta) <= tolerance:
            verdict = "pass"
        else:
            verdict = "fail"
        findings.append({
            "okres_id": okres_id, "okres_cs": okres_cs, "column": column,
            "check": f"I6_okres:{column}", "verdict": verdict,
            "register": got, "census": expected, "delta": delta,
            "unreadable_cells": unreadable.get(column, 0),
            "detail": "" if verdict == "pass" else
                      f"register {got} vs census {expected} ({delta:+d})",
        })

    # Houses: reported against both candidate definitions, graded against neither.
    houses = house_candidates(census_row)
    for label, expected in houses.items():
        got = totals.get("houses", 0)
        findings.append({
            "okres_id": okres_id, "okres_cs": okres_cs, "column": "houses",
            "check": f"I6_okres:houses[{label}]", "verdict": "ungraded",
            "register": got, "census": expected, "delta": got - expected,
            "unreadable_cells": unreadable.get("houses", 0),
            "detail": f"register {got} vs census {expected} "
                      f"({got - expected:+d}) — definition unsettled",
        })

    return Reconciliation(okres_id=okres_id, okres_cs=okres_cs, complete=complete,
                          n_blocks=len(blocks), n_expected=n_expected,
                          n_found=n_found, n_summary=n_summary,
                          findings=findings, unreadable=unreadable)


def diagnose(reconciliation: Reconciliation) -> str:
    """Name the shape of a failure, where the shape is unambiguous.

    A lost or duplicated row moves *every* column in the same direction at once; a
    wrong confession grouping moves one column by an exact amount and leaves the
    rest untouched. Distinguishing them is the whole reason for comparing column by
    column rather than on population alone.
    """
    graded = [f for f in reconciliation.findings if f["verdict"] in ("pass", "fail")]
    if not graded:
        return "not graded"
    failed = [f for f in graded if f["verdict"] == "fail"]
    if not failed:
        return "all graded columns reconcile exactly"
    if len(failed) == len(graded) and len({f["delta"] < 0 for f in failed}) == 1:
        direction = "short of" if failed[0]["delta"] < 0 else "over"
        return (f"every column is {direction} the census — consistent with "
                f"{'missing' if failed[0]['delta'] < 0 else 'duplicated'} rows, "
                f"not with a column mapping error")
    if len(failed) == 1:
        f = failed[0]
        return (f"only {f['column']} disagrees ({f['delta']:+d}) — consistent with "
                f"a grouping error in that column, not with a lost row")
    names = ", ".join(f["column"] for f in failed)
    return f"{len(failed)} of {len(graded)} columns disagree: {names}"


def lexikon_1910_census_report(reconciliations) -> str:
    """Ledger over one or more okresy."""
    lines = [f"lexikon_1910_census v{__version__} — okres reconciliation", ""]
    if not reconciliations:
        return "\n".join(lines + ["  no okres was complete enough to reconcile"])

    lines.append(f"  {'okres':22s} {'blocks':>12s}  {'pass':>5s} {'fail':>5s} "
                 f"{'unrd':>5s} {'incm':>5s}")
    lines.append("  " + "-" * 62)
    graded_total = passed_total = 0
    for rec in reconciliations:
        counts: dict[str, int] = {}
        for f in rec.findings:
            if f["verdict"] == "ungraded":
                continue
            counts[f["verdict"]] = counts.get(f["verdict"], 0) + 1
        coverage = (f"{rec.n_found}/{rec.n_expected}" if rec.n_expected
                    else str(rec.n_blocks))
        lines.append(f"  {(rec.okres_cs or rec.okres_id)!s:22.22s} {coverage:>12s}  "
                     f"{counts.get('pass', 0):5d} {counts.get('fail', 0):5d} "
                     f"{counts.get('unparseable', 0):5d} "
                     f"{counts.get('incomplete', 0):5d}")
        graded_total += counts.get("pass", 0) + counts.get("fail", 0)
        passed_total += counts.get("pass", 0)
        note = diagnose(rec)
        if rec.complete:
            lines.append(f"      {note}")
        else:
            lines.append(f"      incomplete — {rec.missing} of {rec.n_expected} "
                         f"known obce not anchored on these pages "
                         f"({rec.n_blocks} blocks read); columns reported, "
                         f"none graded")
            # A near-complete okres is still worth reading: the shortfall in
            # pop_total is an estimate of how big the missing municipalities are,
            # which is what you search for to find them. A shortfall that is
            # implausible for that many villages says the gap is not coverage.
            if 0 < rec.missing <= 3:
                deficit = next((f["delta"] for f in rec.findings
                                if f["column"] == "pop_total"), None)
                if deficit is not None and deficit < 0:
                    average = (f", {-deficit // rec.missing} each on average"
                               if rec.missing > 1 else "")
                    noun = "obec" if rec.missing == 1 else "obce"
                    lines.append(f"      the {rec.missing} missing {noun} account "
                                 f"for {-deficit} inhabitants{average}")

    lines.append("  " + "-" * 62)
    rate = f"{passed_total / graded_total:.1%}" if graded_total else "--"
    lines.append(f"  {passed_total}/{graded_total} graded columns reconcile  {rate}")
    lines.append("")
    lines.append("  houses is reported against both census definitions and graded "
                 "against neither;")
    lines.append(f"  {', '.join(NOT_IN_CENSUS)} are not carried at okres level at all.")
    lines.append("  incomplete okresy under-sum by construction — coverage, not a "
                 "disagreement.")
    return "\n".join(lines)
