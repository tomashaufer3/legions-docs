How it works

After the table's structure is mended (D-211), the municipalities the register still holds institutions for and no table row are rows no block was cut for: the run's table cutter found 6 blocks on Czech 0360 (Přísečnice and Vejprty), 5 on Mähren 0151 (Vsetín), and elsewhere it joined a row to the next one's block (Děčín 15 and 16 on Czech 0068) or left it out between two (Louny 38, Teplá 31, Pohořelice 12, Vranov 15). Reading the same blocks again would miss them again, so this reads the page: each page's table body is cut into horizontal bands, full table width, scaled to the vision ceiling as the run scales a block (so the type is read at the run's resolution), each band BAND_RATIO of the table's width high and overlapping the next by OVERLAP of it, so that no row falls only on a band's edge. The run's two table readers — Gemini 3.8 Flash (the reader of record) and Claude Haiku 5.5, with the run's settings and the volume's own table prompt and row schema — transcribe every printed line of each band. One instruction differs from the run's, because a band holds several blocks: INSTRUCTION.

The pages are the ones §9's table check lists a municipality with institutions and no table row on (output/recheck/table_check.csv). The replies are written to output/recheck/&lt;edition&gt;/table_reread/replies_&lt;reader&gt;.json; which of their rows enter the register, and under which court, is decided in §9 (table_rows.reread_rows), where the court's list of parts is at hand.

Nothing is sent without spend, and then only on the machine of record (D-68). Every band's reply is cached under its crop, instruction, model, prompt and settings, so a rerun spends nothing.

def pages_to_read(check: Path = OUT / "table_check.csv", final: Path = FINAL,
                  before: Path = OUT / "table_reread.json") -> dict[str, list[str]]:
    """``{edition: [page, …]}``: every page §9's check lists a municipality with institutions and
    no table row on, **and every page an earlier re-read read**. Once a page's rows are in, the check
    no longer lists it; without the earlier pages, the replies would be rewritten without them and
    their rows would leave the register on the next build. An earlier page is served from the cache."""
    faults = pd.read_csv(check, dtype=str)
    inst = pd.read_parquet(final / "register_institutions.parquet", columns=["edition", "court_key", "muni_seq"])
    held = {(e, str(k), str(m)) for e, k, m in zip(inst["edition"], inst["court_key"], inst["muni_seq"])}
    miss = faults[faults["issue"] == "missing_row"]
    out: dict[str, set] = {}
    for r in miss.itertuples():
        seq = str(r.muni_seq).split(".")[0]
        if (r.edition, str(r.court_key), seq) in held:
            for page in str(r.page).split(";"):
                out.setdefault(r.edition, set()).add(page.strip().zfill(4))
    if Path(before).exists():
        for edition, pages in json.loads(Path(before).read_text(encoding="utf-8")).get("pages", {}).items():
            out.setdefault(edition, set()).update(pages)
    return {e: sorted(p) for e, p in out.items()}

def cut(edition: str, pages: list[str]) -> list[dict]:
    """Every band of every page: ``{"band", "page", "top", "bottom", "parts"}`` with the run's crop."""
    jobs = []
    if edition == "cz":
        template = lx.read_grid_template(pilot.GRID_CSV, "cechy_1910_cz")
        for page in pages:
            alto = lx.read_alto(pilot.CZ_PAGES / f"{page}.alto.xml")
            image = lx.render_page(pilot.CZ_PDF, int(page), alto.width)
            # the table's top and foot from the image (the run's notes geometry), not the text layer's
            # rules: on 0360 those put the body's top below Přísečnice 11–17 and Vejprty 1–2
            geo = lximg.detect(image)
            if geo.kind == "table" and geo.header_bottom and geo.table_foot:
                top, bottom = geo.header_bottom, min(image.height, geo.table_foot + int(FOOT_MARGIN * image.width))
            else:
                top, bottom = lx.body_band(alto)
            cols = lx.fit_columns(alto, template)[0]
            for k, (t, b) in enumerate(_bands(top, bottom, int(cols[-1] - cols[0]))):
                tiles = lx.tile_block(image, alto, SimpleNamespace(top=t, bottom=b), template, tiles=1)
                parts, _run_instruction = pilot.block_parts(tiles)
                jobs.append({"band": f"{page}#B{k:02d}", "page": page, "top": t, "bottom": b, "parts": parts})
        return jobs
    profile = lxvol.load_profile(rf.GERMAN[edition])
    for page in pages:
        image = lxdn.render_native(profile.source_pdf, int(page))
        geo = lxdt.detect(image, profile)
        if geo.kind != "table" or geo.frame is None:
            continue
        top = geo.header_bottom or geo.table_top or 0
        bottom = min(image.height, (geo.table_foot or image.height) + int(FOOT_MARGIN * image.width))
        for k, (t, b) in enumerate(_bands(top, bottom, geo.frame[1] - geo.frame[0])):
            block = lxdt.TableBlock(page=page, index=k, kind="municipality",
                                    rows=[lxdt.PrintedRow(top=t, bottom=b, area=True, seq=True)])
            jobs.append({"band": f"{page}#B{k:02d}", "page": page, "top": t, "bottom": b,
                         "parts": [(lxdt.crop_block(image, geo, block), "")]})
    return jobs

def price(jobs: list[dict], name: str) -> float:
    """USD for the bands not in the cache: the image at the vision ceiling, the reply at
    :data:`EST_OUTPUT` (thinking included for Haiku)."""
    rate_in, rate_out = (rdt.ARMS[name].price if name in rdt.ARMS else (0.75, 3.75))
    image = sum(min(4784, im.width * im.height // 750) for job in jobs for im, _c in job["parts"])
    return (image + 2000 * len(jobs)) * rate_in / 1e6 + EST_OUTPUT * len(jobs) * rate_out / 1e6

def read(spend: bool = False, out: Path = OUT, pages: dict[str, list[str]] | None = None,
         allow_other_machine: bool = False) -> dict:
    """Read every band of the pages (``pages_to_read`` by default) with both readers; write each
    reader's replies. Returns the summary also written to ``output/recheck/table_reread.json``."""
    if spend and bo.HOST != MACHINE_OF_RECORD and not allow_other_machine:
        raise RuntimeError(f"the table re-read spends on {MACHINE_OF_RECORD} only (D-68): this is {bo.HOST}")
    if spend:
        from utils.lexikon_1910 import lexikon_1910_pages as lxp
        lxp.load_credentials()
    pages = pages if pages is not None else pages_to_read()
    summary = {"module": __version__, "host": bo.HOST, "run": time.strftime("%Y-%m-%d %H:%M:%S"),
               "pages": pages, "editions": {}}
    for edition, page_list in pages.items():
        jobs = cut(edition, page_list)
        folder = Path(out) / edition / SUB
        entry = {"bands": len(jobs), "readers": {}}
        for name, reader in readers(edition).items():
            cache = folder / "cache" / name
            cache.mkdir(parents=True, exist_ok=True)
            missing = [j for j in jobs if not (cache / f"{_key(j, reader)}.json").exists()]
            state = {"cached": len(jobs) - len(missing), "to_read": len(missing),
                     "usd_live": round(price(missing, name), 2)}
            if missing and not spend:
                state["state"] = "not read (spend is off)"
                entry["readers"][name] = state
                continue
            failed, errors = 0, []
            for job in missing:
                try:
                    rows, usage = _read_band(reader, job)
                except Exception as exc:                                # noqa: BLE001
                    failed += 1
                    errors.append(f"{job['band']}: {type(exc).__name__}: {str(exc)[:200]}")
                    continue
                (cache / f"{_key(job, reader)}.json").write_text(json.dumps(
                    {"lines": rows, "usage": usage, "band": job["band"], "model": getattr(reader, "model", ""),
                     "at": time.time()}), encoding="utf-8")
            replies = []
            for job in jobs:
                hit = cache / f"{_key(job, reader)}.json"
                if hit.exists():
                    replies.append({"band": job["band"], "page": job["page"], "top": job["top"],
                                    "bottom": job["bottom"],
                                    "rows": json.loads(hit.read_text(encoding="utf-8"))["lines"]})
            (folder / f"replies_{name}.json").write_text(json.dumps(replies, indent=1, ensure_ascii=False),
                                                          encoding="utf-8")
            state.update({"failed": failed, "errors": errors[:5], "read": len(replies),
                          "state": "read" if len(replies) == len(jobs) else "incomplete: run again"})
            entry["readers"][name] = state
        summary["editions"][edition] = entry
    # the register takes the reader of record's rows; the second reader's agreement is reported, so a
    # band it could not read holds nothing back (Mähren 0078 B01 ran Haiku out of tokens at 16,000)
    summary["complete"] = all(e["readers"].get(READERS[0], {}).get("state") == "read"
                              for e in summary["editions"].values())
    summary["second_complete"] = all(e["readers"].get(READERS[1], {}).get("state") == "read"
                                     for e in summary["editions"].values())
    Path(out).mkdir(parents=True, exist_ok=True)
    (Path(out) / "table_reread.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return summary

def report(summary: dict) -> str:
    lines = [f"table re-read ({summary['host']}, {summary['run']}):"]
    for edition, e in summary["editions"].items():
        pages = ", ".join(summary["pages"][edition])
        lines.append(f"  {edition}: {e['bands']} bands on pages {pages}")
        for name, r in e["readers"].items():
            lines.append(f"    {name}: {r}")
    return "\n".join(lines)

from utils.lexikon_1910 import lexikon_1910_volume as lxvol            # noqa: E402

MACHINE_OF_RECORD = "heavyhorse6"
OUT = ROOT / "output" / "recheck"
FINAL = ROOT / "data" / "final" / "register_rechecked"
SUB = "table_reread"
BAND_RATIO = 0.30          # a band's height, as a share of the table's width (≈ 8 rows of the Czech table)
FOOT_MARGIN = 0.04         # past the detected foot: on Mähren 0151 it sat above the last row
OVERLAP = 0.06             # how far a band overlaps the next, as a share of the table's width (≈ 1½ rows)
INSTRUCTION = ("This image is a horizontal band of the table holding several consecutive blocks, not one. "
               "Transcribe every printed line in it, in printed order, as the rules say; ignore a line "
               "cut off at the top or bottom edge.")
#: The run's two table readers (``run_table_arm.READERS``, ``run_de_table.ARMS``), reader of record first.
READERS = ("gemini38", "haiku55")
MAX_TOKENS = 32000         # a band's reply cap, after doubling from the run's 8,000
#: Output tokens a band, for the estimate only: about eight blocks of the run's ≈ 600.
EST_OUTPUT = 4800


def pages_to_read(check: Path = OUT / "table_check.csv", final: Path = FINAL,
                  before: Path = OUT / "table_reread.json") -> dict[str, list[str]]:
    """``{edition: [page, …]}``: every page §9's check lists a municipality with institutions and
    no table row on, **and every page an earlier re-read read**. Once a page's rows are in, the check
    no longer lists it; without the earlier pages, the replies would be rewritten without them and
    their rows would leave the register on the next build. An earlier page is served from the cache."""
    faults = pd.read_csv(check, dtype=str)
    inst = pd.read_parquet(final / "register_institutions.parquet", columns=["edition", "court_key", "muni_seq"])
    held = {(e, str(k), str(m)) for e, k, m in zip(inst["edition"], inst["court_key"], inst["muni_seq"])}
    miss = faults[faults["issue"] == "missing_row"]
    out: dict[str, set] = {}
    for r in miss.itertuples():
        seq = str(r.muni_seq).split(".")[0]
        if (r.edition, str(r.court_key), seq) in held:
            for page in str(r.page).split(";"):
                out.setdefault(r.edition, set()).add(page.strip().zfill(4))
    if Path(before).exists():
        for edition, pages in json.loads(Path(before).read_text(encoding="utf-8")).get("pages", {}).items():
            out.setdefault(edition, set()).update(pages)
    return {e: sorted(p) for e, p in out.items()}


def _bands(top: int, bottom: int, width: int) -> list[tuple[int, int]]:
    """``[(top, bottom), …]`` covering the body, each ``BAND_RATIO`` of the width high, overlapping."""
    height, step = int(BAND_RATIO * width), int((BAND_RATIO - OVERLAP) * width)
    out, y = [], top
    while True:
        out.append((y, min(bottom, y + height)))
        if y + height >= bottom:
            return out
        y += step


def cut(edition: str, pages: list[str]) -> list[dict]:
    """Every band of every page: ``{"band", "page", "top", "bottom", "parts"}`` with the run's crop."""
    jobs = []
    if edition == "cz":
        template = lx.read_grid_template(pilot.GRID_CSV, "cechy_1910_cz")
        for page in pages:
            alto = lx.read_alto(pilot.CZ_PAGES / f"{page}.alto.xml")
            image = lx.render_page(pilot.CZ_PDF, int(page), alto.width)
            # the table's top and foot from the image (the run's notes geometry), not the text layer's
            # rules: on 0360 those put the body's top below Přísečnice 11–17 and Vejprty 1–2
            geo = lximg.detect(image)
            if geo.kind == "table" and geo.header_bottom and geo.table_foot:
                top, bottom = geo.header_bottom, min(image.height, geo.table_foot + int(FOOT_MARGIN * image.width))
            else:
                top, bottom = lx.body_band(alto)
            cols = lx.fit_columns(alto, template)[0]
            for k, (t, b) in enumerate(_bands(top, bottom, int(cols[-1] - cols[0]))):
                tiles = lx.tile_block(image, alto, SimpleNamespace(top=t, bottom=b), template, tiles=1)
                parts, _run_instruction = pilot.block_parts(tiles)
                jobs.append({"band": f"{page}#B{k:02d}", "page": page, "top": t, "bottom": b, "parts": parts})
        return jobs
    profile = lxvol.load_profile(rf.GERMAN[edition])
    for page in pages:
        image = lxdn.render_native(profile.source_pdf, int(page))
        geo = lxdt.detect(image, profile)
        if geo.kind != "table" or geo.frame is None:
            continue
        top = geo.header_bottom or geo.table_top or 0
        bottom = min(image.height, (geo.table_foot or image.height) + int(FOOT_MARGIN * image.width))
        for k, (t, b) in enumerate(_bands(top, bottom, geo.frame[1] - geo.frame[0])):
            block = lxdt.TableBlock(page=page, index=k, kind="municipality",
                                    rows=[lxdt.PrintedRow(top=t, bottom=b, area=True, seq=True)])
            jobs.append({"band": f"{page}#B{k:02d}", "page": page, "top": t, "bottom": b,
                         "parts": [(lxdt.crop_block(image, geo, block), "")]})
    return jobs


def readers(edition: str) -> dict:
    """The run's two table readers for the volume, with its prompt and row schema."""
    if edition == "cz":
        url, key_env = pilot.READER_ENDPOINTS["gemini"]
        return {"gemini38": pilot.OpenAICompatibleReader("gemini-3.8-flash", url, key_env, temperature=0.0,
                                                         reasoning_effort="none"),
                "haiku55": pilot.ClaudeReader("claude-haiku-5-5", "high", thinking="adaptive", max_tokens=8000)}
    profile = lxvol.load_profile(rf.GERMAN[edition])
    return {name: rdt.make_reader(rdt.ARMS[name], profile) for name in READERS}


def _key(job: dict, reader) -> str:
    return rdt.cache_key(job["parts"][0][0], INSTRUCTION, reader)


def _read_band(reader, job: dict):
    """One band's reply; a reply cut off at the token cap is asked again with the cap doubled, up to
    :data:`MAX_TOKENS` (the cap is not in the cache key, so the answer is cached as any other)."""
    while True:
        try:
            return reader(job["parts"], INSTRUCTION)
        except pilot.Truncated:
            cap = getattr(reader, "max_tokens", 8000)
            if cap >= MAX_TOKENS:
                raise
            reader.max_tokens = min(cap * 2, MAX_TOKENS)


def price(jobs: list[dict], name: str) -> float:
    """USD for the bands not in the cache: the image at the vision ceiling, the reply at
    :data:`EST_OUTPUT` (thinking included for Haiku)."""
    rate_in, rate_out = (rdt.ARMS[name].price if name in rdt.ARMS else (0.75, 3.75))
    image = sum(min(4784, im.width * im.height // 750) for job in jobs for im, _c in job["parts"])
    return (image + 2000 * len(jobs)) * rate_in / 1e6 + EST_OUTPUT * len(jobs) * rate_out / 1e6


def read(spend: bool = False, out: Path = OUT, pages: dict[str, list[str]] | None = None,
         allow_other_machine: bool = False) -> dict:
    """Read every band of the pages (``pages_to_read`` by default) with both readers; write each
    reader's replies. Returns the summary also written to ``output/recheck/table_reread.json``."""
    if spend and bo.HOST != MACHINE_OF_RECORD and not allow_other_machine:
        raise RuntimeError(f"the table re-read spends on {MACHINE_OF_RECORD} only (D-68): this is {bo.HOST}")
    if spend:
        from utils.lexikon_1910 import lexikon_1910_pages as lxp
        lxp.load_credentials()
    pages = pages if pages is not None else pages_to_read()
    summary = {"module": __version__, "host": bo.HOST, "run": time.strftime("%Y-%m-%d %H:%M:%S"),
               "pages": pages, "editions": {}}
    for edition, page_list in pages.items():
        jobs = cut(edition, page_list)
        folder = Path(out) / edition / SUB
        entry = {"bands": len(jobs), "readers": {}}
        for name, reader in readers(edition).items():
            cache = folder / "cache" / name
            cache.mkdir(parents=True, exist_ok=True)
            missing = [j for j in jobs if not (cache / f"{_key(j, reader)}.json").exists()]
            state = {"cached": len(jobs) - len(missing), "to_read": len(missing),
                     "usd_live": round(price(missing, name), 2)}
            if missing and not spend:
                state["state"] = "not read (spend is off)"
                entry["readers"][name] = state
                continue
            failed, errors = 0, []
            for job in missing:
                try:
                    rows, usage = _read_band(reader, job)
                except Exception as exc:                                # noqa: BLE001
                    failed += 1
                    errors.append(f"{job['band']}: {type(exc).__name__}: {str(exc)[:200]}")
                    continue
                (cache / f"{_key(job, reader)}.json").write_text(json.dumps(
                    {"lines": rows, "usage": usage, "band": job["band"], "model": getattr(reader, "model", ""),
                     "at": time.time()}), encoding="utf-8")
            replies = []
            for job in jobs:
                hit = cache / f"{_key(job, reader)}.json"
                if hit.exists():
                    replies.append({"band": job["band"], "page": job["page"], "top": job["top"],
                                    "bottom": job["bottom"],
                                    "rows": json.loads(hit.read_text(encoding="utf-8"))["lines"]})
            (folder / f"replies_{name}.json").write_text(json.dumps(replies, indent=1, ensure_ascii=False),
                                                          encoding="utf-8")
            state.update({"failed": failed, "errors": errors[:5], "read": len(replies),
                          "state": "read" if len(replies) == len(jobs) else "incomplete: run again"})
            entry["readers"][name] = state
        summary["editions"][edition] = entry
    # the register takes the reader of record's rows; the second reader's agreement is reported, so a
    # band it could not read holds nothing back (Mähren 0078 B01 ran Haiku out of tokens at 16,000)
    summary["complete"] = all(e["readers"].get(READERS[0], {}).get("state") == "read"
                              for e in summary["editions"].values())
    summary["second_complete"] = all(e["readers"].get(READERS[1], {}).get("state") == "read"
                                     for e in summary["editions"].values())
    Path(out).mkdir(parents=True, exist_ok=True)
    (Path(out) / "table_reread.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return summary


def replies(edition: str, out: Path = OUT) -> dict[str, list[dict]]:
    """``{reader: [band, …]}`` as :func:`read` wrote them; empty where the volume was not re-read."""
    folder = Path(out) / edition / SUB
    return {name: json.loads((folder / f"replies_{name}.json").read_text(encoding="utf-8"))
            for name in READERS if (folder / f"replies_{name}.json").exists()}


def report(summary: dict) -> str:
    lines = [f"table re-read ({summary['host']}, {summary['run']}):"]
    for edition, e in summary["editions"].items():
        pages = ", ".join(summary["pages"][edition])
        lines.append(f"  {edition}: {e['bands']} bands on pages {pages}")
        for name, r in e["readers"].items():
            lines.append(f"    {name}: {r}")
    return "\n".join(lines)


def main(argv=None) -> int:
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--spend", action="store_true", help="send what is not cached (Heavyhorse6 only)")
    args = ap.parse_args(argv)
    print(report(read(spend=args.spend)))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
