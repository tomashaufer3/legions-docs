TL;DR

atlas_iv37_parties.csv is a hand transcription of a photograph of a printed table (Akademický atlas českých dějin, plate IV.37, p. 318). The page is curved in the photo, so the notes column drifts against the rows it belongs to — a note can sit visibly above its own row near the page edges and level with it near the centre. That is exactly the failure the project's validation register warns about for the 1910 register: a row wrong in two compensating places still looks right.

How it works

The table defends itself. Its notes reference other rows by code ("1908 sjednocení s Č 10 v Č 15"), so a mis-assigned note usually produces a reference that cannot be reconciled. These checks make that mechanical.

python utils/reference_tools/parties/check_atlas_parties.py

Exit code 0 if every invariant holds, 1 otherwise.

What it CANNOT see (stated, per the register's convention)

  • A note swapped between two adjacent rows that both carry a resolvable reference — the graph stays consistent, only the attribution is wrong.
  • A row dropped wholesale: the codes would still run 1..n if the drop is at the end of a block, which is why P1 checks contiguity from 1.
  • Anything in the year columns. They are read off the photo and nothing here cross-checks them; treat founded as indicative, not evidential.

def run_checks(df: pd.DataFrame) -> list[tuple[bool, str]]:
    out: list[tuple[bool, str]] = []
    codes = set(df["code"])

    # P1 — every block's codes run 1..n with no gap and no repeat.
    for prefix, bloc in BLOCS.items():
        nums = sorted(df.loc[df["prefix"] == prefix, "num"])
        ok = nums == list(range(1, len(nums) + 1))
        out.append((ok, f"{bloc}: codes run 1..{len(nums)} without gaps "
                        f"({'ok' if ok else nums})"))

    # P2 — bloc label agrees with the code prefix.
    bad = df[df["bloc"] != df["prefix"].map(BLOCS)]
    out.append((bad.empty, f"bloc label matches code prefix "
                           f"({len(bad)} mismatched)"))

    # P3 — every code named in `refs` exists in the table.
    dangling: list[str] = []
    for _, r in df.iterrows():
        for ref in filter(None, r["refs"].split(";")):
            if _norm(ref) not in codes:
                dangling.append(f"{r['code']}→{ref}")
    out.append((not dangling, f"every `refs` target exists "
                              f"({len(dangling)} dangling: {dangling[:4]})"))

    # P4 — every code mentioned in the note text is also declared in `refs`.
    #      Catches a note transcribed onto the wrong row *and* a ref column
    #      that was not kept in step with the note it summarises.
    missed: list[str] = []
    for _, r in df.iterrows():
        declared = {_norm(x) for x in r["refs"].split(";") if x}
        mentioned = {_norm(m) for m in re.findall(r"[ČNPDV] ?\d+", r["note_cs"])}
        if mentioned - declared:
            missed.append(f"{r['code']}: {sorted(mentioned - declared)}")
    out.append((not missed, f"every code named in a note is in `refs` "
                            f"({len(missed)} unlisted: {missed[:4]})"))

    # P5 — a merger is reciprocal: if A says "merged with B into C", then B
    #      must say "merged with A into C", and C must exist.
    asym: list[str] = []
    notes = dict(zip(df["code"], df["note_cs"]))
    for code, note in notes.items():
        m = _MERGE.search(note)
        if not m:
            continue
        partner, into = _norm(m.group(1)), _norm(m.group(2))
        back = _MERGE.search(notes.get(partner, ""))
        if not back or _norm(back.group(1)) != code or _norm(back.group(2)) != into:
            asym.append(f"{code}+{partner}→{into}")
    out.append((not asym, f"mergers are reciprocal ({len(asym)} one-sided: {asym})"))

    # P6 — a successor claim is reciprocal in the other direction:
    #      "nástupkyně X" / "pokračovatelka X" should face a note on X.
    lonely: list[str] = []
    for code, note in notes.items():
        m = re.search(r"(?:nástupkyně|pokračovatelka) ([ČNPDV] ?\d+)", note)
        if m:
            target = _norm(m.group(1))
            if code not in notes.get(target, "").replace(" ", ""):
                lonely.append(f"{code}←{target}")
    out.append((not lonely, f"successor claims face a note on the predecessor "
                            f"({len(lonely)} unmatched: {lonely})"))

    # P7 — a party must operate somewhere, or say why not.
    nowhere = df[(df["cechy"] == "") & (df["morava"] == "") & (df["slezsko"] == "")]
    out.append((len(nowhere) <= 1,
                f"at most one row with no area of operation "
                f"({len(nowhere)}: {list(nowhere['code'])})"))

    # P8 — the transcription is complete against the printed block sizes.
    EXPECTED = {"czech": 25, "german": 11, "polish": 8, "workers": 7, "landowner": 4}
    sizes = df["bloc"].value_counts().to_dict()
    ok = sizes == EXPECTED
    out.append((ok, f"block sizes match the plate {EXPECTED} (got {sizes})"))

    return out

"""
check_atlas_parties.py — invariants on the transcribed IV.37 party register
===========================================================================
``atlas_iv37_parties.csv`` is a **hand transcription of a photograph** of a
printed table (*Akademický atlas českých dějin*, plate IV.37, p. 318).  The
page is curved in the photo, so the notes column drifts against the rows it
belongs to — a note can sit visibly above its own row near the page edges and
level with it near the centre.  That is exactly the failure the project's
validation register warns about for the 1910 register: a row wrong in two
compensating places still *looks* right.

The table defends itself.  Its notes reference other rows by code ("1908
sjednocení s Č 10 v Č 15"), so a mis-assigned note usually produces a
reference that cannot be reconciled.  These checks make that mechanical.

    python utils/reference_tools/parties/check_atlas_parties.py

Exit code 0 if every invariant holds, 1 otherwise.

What it CANNOT see (stated, per the register's convention)
----------------------------------------------------------
* A note swapped between two adjacent rows that both carry a *resolvable*
  reference — the graph stays consistent, only the attribution is wrong.
* A row dropped wholesale: the codes would still run 1..n if the drop is at
  the end of a block, which is why P1 checks contiguity from 1.
* Anything in the year columns.  They are read off the photo and nothing here
  cross-checks them; treat `founded` as indicative, not evidential.
"""

from __future__ import annotations

import re
import sys
from pathlib import Path

import pandas as pd

__version__ = "1.0.0"

CSV = Path(__file__).resolve().parents[3] / "reference" / "parties" / "atlas_iv37_parties.csv"

BLOCS = {"Č": "czech", "N": "german", "P": "polish",
         "D": "workers", "V": "landowner"}

# Notes that assert a directed relation, and the reciprocal each implies.
#   pattern → (relation, reciprocal wording that the target should carry)
_MERGE = re.compile(r"sjednocení s ([ČNPDV] ?\d+) v ([ČNPDV] ?\d+)")


def _norm(code: str) -> str:
    return code.replace(" ", "")


def load() -> pd.DataFrame:
    df = pd.read_csv(CSV, dtype=str).fillna("")
    df["prefix"] = df["code"].str[0]
    df["num"] = df["code"].str[1:].astype(int)
    return df


def run_checks(df: pd.DataFrame) -> list[tuple[bool, str]]:
    out: list[tuple[bool, str]] = []
    codes = set(df["code"])

    # P1 — every block's codes run 1..n with no gap and no repeat.
    for prefix, bloc in BLOCS.items():

        out.append((ok, f"{bloc}: codes run 1..{len(nums)} without gaps "
                        f"({'ok' if ok else nums})"))

    # P2 — bloc label agrees with the code prefix.
    bad = df[df["bloc"] != df["prefix"].map(BLOCS)]
    out.append((bad.empty, f"bloc label matches code prefix "
                           f"({len(bad)} mismatched)"))

    # P3 — every code named in `refs` exists in the table.
    dangling: list[str] = []
    for _, r in df.iterrows():
        for ref in filter(None, r["refs"].split(";")):
            if _norm(ref) not in codes:
                dangling.append(f"{r['code']}→{ref}")
    out.append((not dangling, f"every `refs` target exists "
                              f"({len(dangling)} dangling: {dangling[:4]})"))

    # P4 — every code mentioned in the note text is also declared in `refs`.
    #      Catches a note transcribed onto the wrong row *and* a ref column
    #      that was not kept in step with the note it summarises.
    missed: list[str] = []
    for _, r in df.iterrows():
        declared = {_norm(x) for x in r["refs"].split(";") if x}
        mentioned = {_norm(m) for m in re.findall(r"[ČNPDV] ?\d+", r["note_cs"])}
        if mentioned - declared:
            missed.append(f"{r['code']}: {sorted(mentioned - declared)}")
    out.append((not missed, f"every code named in a note is in `refs` "
                            f"({len(missed)} unlisted: {missed[:4]})"))

    # P5 — a merger is reciprocal: if A says "merged with B into C", then B
    #      must say "merged with A into C", and C must exist.
    asym: list[str] = []
    notes = dict(zip(df["code"], df["note_cs"]))
    for code, note in notes.items():
        m = _MERGE.search(note)
        if not m:
            continue
        partner, into = _norm(m.group(1)), _norm(m.group(2))
        back = _MERGE.search(notes.get(partner, ""))
        if not back or _norm(back.group(1)) != code or _norm(back.group(2)) != into:
            asym.append(f"{code}+{partner}→{into}")
    out.append((not asym, f"mergers are reciprocal ({len(asym)} one-sided: {asym})"))

    # P6 — a successor claim is reciprocal in the other direction:
    #      "nástupkyně X" / "pokračovatelka X" should face a note on X.
    lonely: list[str] = []
    for code, note in notes.items():
        m = re.search(r"(?:nástupkyně|pokračovatelka) ([ČNPDV] ?\d+)", note)
        if m:
            target = _norm(m.group(1))
            if code not in notes.get(target, "").replace(" ", ""):
                lonely.append(f"{code}←{target}")
    out.append((not lonely, f"successor claims face a note on the predecessor "
                            f"({len(lonely)} unmatched: {lonely})"))

    # P7 — a party must operate somewhere, or say why not.
    nowhere = df[(df["cechy"] == "") & (df["morava"] == "") & (df["slezsko"] == "")]
    out.append((len(nowhere) <= 1,
                f"at most one row with no area of operation "
                f"({len(nowhere)}: {list(nowhere['code'])})"))

    # P8 — the transcription is complete against the printed block sizes.
    EXPECTED = {"czech": 25, "german": 11, "polish": 8, "workers": 7, "landowner": 4}
    sizes = df["bloc"].value_counts().to_dict()
    ok = sizes == EXPECTED
    out.append((ok, f"block sizes match the plate {EXPECTED} (got {sizes})"))

    return out


def main() -> int:
    df = load()
    print(f"IV.37 party register — {len(df)} parties, {CSV.name}\n")
    results = run_checks(df)
    for ok, msg in results:
        print(f"  [{'pass' if ok else 'FAIL'}] {msg}")
    n_ok = sum(ok for ok, _ in results)
    print(f"\n{n_ok}/{len(results)} invariants hold")
    return 0 if n_ok == len(results) else 1


if __name__ == "__main__":
    sys.exit(main())
