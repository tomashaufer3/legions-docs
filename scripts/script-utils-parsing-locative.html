How it works

Nineteenth-century member lists give a residence after a preposition, so the name is in the locative case, while every place table here (settlements_1910, its aliases) holds the nominative. Rather than guess a nominative back from each printed form, this module declines every 1910 name forward into the locatives Czech grammar allows and indexes them, so a printed Vodňanech finds Vodňany and nothing else is invented. The rules over-generate on purpose (an unused form costs nothing); what keeps them honest is that a key claimed by more than one settlement is decided only by evidence the entry carries — a court seat, a qualifier (u Příbrami, na Moravě) — and refused otherwise.

The match, for one residence:

matched one settlement (how says why: unique, seat, qualifier, land, obec). prague Prague and its historic quarters, which settlements_1910 does not carry as settlements: okres Praha, no settlement. ambiguous two or more settlements and nothing to choose between them (n_candidates). outside a place outside the Czech lands the lists name often (Vienna, Pest, Galicia …). unmatched nothing found.

def noun_locatives(w: str) -> set[str]:
    """Locatives a noun may take (singular and plural place names alike)."""
    out = {w}
    if len(w) > 4 and w[-2] == "e" and w[-1] not in "aeiouyáéěíóúůý":   # e-drop:
        out |= {w[:-2] + w[-1] + "i", w[:-2] + w[-1] + "u", w[:-2] + w[-1] + "e"}  # Litovel → Litovli
    if w.endswith("ice"):                                   # Pardubice → Pardubicích, Roudnice → Roudnici
        out |= {w[:-1] + "ích", w[:-1] + "i"}
    elif w.endswith(("y", "i")) and len(w) > 3:             # Vodňany → Vodňanech, Vary → Varech
        out |= {w[:-1] + "ech", w[:-1] + "ách", w[:-1] + "ích"}
    elif w.endswith(("e", "ě")):                            # Lázně → Lázních, Hradiště → Hradišti
        out |= {w[:-1] + "ích", w[:-1] + "i", w[:-1] + "ích"}
    elif w.endswith("í"):                                   # Meziříčí, Ústí: unchanged
        pass
    elif w.endswith("a"):                                   # Praha → Praze, Hora → Hoře
        stem = w[:-1]
        out |= {_palatalise(stem) + "e", stem + "ě", stem + "e", stem + "i"}
    elif w.endswith("o"):                                   # Brno → Brně, Rakousko → Rakousku
        stem = w[:-1]
        out |= {stem + "ě", stem + "u", _palatalise(stem) + "e", stem + "e"}
    else:                                                   # consonant: Tábor, Jičín, Písek …
        out |= {w + "ě", w + "e", w + "u", _palatalise(w) + "e", w + "i"}
        if w.endswith("ek"):                                # Písek → Písku, Hrádek → Hrádku
            out |= {w[:-2] + "ku", w[:-2] + "ce"}
        if w.endswith("ec"):                                # Hradec → Hradci
            out.add(w[:-2] + "ci")
        if w.endswith("eň"):                                # Plzeň → Plzni
            out.add(w[:-2] + "ni")
        if w.endswith("es"):                                # Ves → Vsi
            out.add(w[:-2] + "si")
        if "ů" in w:                                        # Dvůr → Dvoře, Brůmov → Brumově?
            v = w[::-1].replace("ů", "o", 1)[::-1]
            out |= {v + "ě", v + "e", _palatalise(v) + "e", v + "u"}
    return out

def adjective_locatives(w: str) -> set[str]:
    """Locatives an adjective (or a possessive) in a place name may take."""
    out: set[str] = set()
    if w.endswith("á"):                                     # Kutná → Kutné
        out.add(w[:-1] + "é")
    elif w.endswith("é"):                                   # Nové → Novém; Velké (pl.) → Velkých
        out |= {w[:-1] + "ém", w[:-1] + "ých"}
    elif w.endswith("ý"):                                   # Český → Českém
        out |= {w[:-1] + "ém"}
    elif w.endswith("í"):                                   # Horní → Horní / Horním / Horních
        out |= {w + "m", w[:-1] + "ích"}

        out.add(w[:-2] + "ově")

        out.add(w[:-1] + "ých")
    elif w.endswith("in"):                                  # Libušin → Libušině
        out.add(w + "ě")
    return out

def locative_keys(name: str, cap: int = 4000) -> set[str]:
    """Every normalised locative and genitive (and the nominative itself) of a place name."""
    words = name.lower().split()
    if not words:
        return set()
    cut = next((i for i, t in enumerate(words) if i > 0 and t in _NAME_PREPS), len(words))
    head, tail = words[:cut], words[cut:]
    if len(head) > 4:
        return {norm(name)}
    keys = set()
    for adj, noun in ((adjective_locatives, noun_locatives), (adjective_genitives, noun_genitives)):
        options = [{w} | adj(w) if w.endswith(("á", "é", "ý", "ův", "ova", "ovo"))
                   else {w} | noun(w) | adj(w) for w in head]
        for combo in itertools.islice(itertools.product(*options), cap):
            keys.add(norm(" ".join(list(combo) + tail)))
    return keys

def czech_spellings(german: str) -> set[str]:
    """A German place name as a Czech list would spell it, with the usual endings
    (*-berg* as *-berk*/*-perk*, *-burg* as *-burk*/*-purk*)."""
    s = german.lower()
    for a, b in _CZECHIZE:
        s = s.replace(a, b)
    out = {s}
    for a, bs in (("berg", ("berk", "perk")), ("burg", ("burk", "purk"))):
        if s.endswith(a):
            out |= {s[: -len(a)] + b for b in bs}
    return out

def build_index(settlements: pd.DataFrame, aliases: pd.DataFrame | None = None,
                districts: pd.DataFrame | None = None) -> dict[str, list[dict]]:
    """Normalised locative → the settlements it may name.

    Each settlement is indexed under its Czech name and, where it differs, its 1910 name;
    each alias (an older Czech name, ``settlement_aliases_1910``) under its own name, for
    the settlement it points to. A candidate records whether the settlement is the seat
    of its court district (the name of a ``gerichtsbezirk_cs`` in its own okres)."""
    seats: set[tuple[str, int]] = set()
    if districts is not None:
        for r in districts.itertuples(index=False):
            okres_id = int(str(r.court_id)[:6]) if pd.notna(r.court_id) else None
            for nm in (r.gerichtsbezirk_cs, r.okres_cs):
                if isinstance(nm, str) and okres_id:
                    nm = re.sub(r"\s*\(.*\)", "", nm)
                    seats.add((norm(nm), okres_id))
                    if short_name(nm):
                        seats.add((norm(short_name(nm)), okres_id))
    s = settlements.dropna(subset=["okres_id"])
    by_name: dict[tuple[str, int], str] = {}
    index: dict[str, list[dict]] = defaultdict(list)

    def add(name: str, rec: dict, via: str) -> None:
        uid = rec["settlement_id"] or f"okres:{rec['okres_id']}"
        for key in locative_keys(name):
            if not any((c["settlement_id"] or f"okres:{c['okres_id']}") == uid for c in index[key]):
                index[key].append({**rec, "via": via})

    for r in s.itertuples(index=False):
        okres_id = int(r.okres_id)
        short = short_name(_title(r.cs_name))
        rec = {"settlement_id": r.settlement_id, "okres_id": okres_id, "land": r.land,
               "status": r.status_1910, "okres_name": r.okres_1910 or "",
               "court": getattr(r, "court_cs", "") or "",
               "seat": (norm(r.cs_name), okres_id) in seats
               or (short is not None and (norm(short), okres_id) in seats),
               "lat": getattr(r, "lat", None), "lon": getattr(r, "lon", None)}
        by_name[(norm(r.cs_name), okres_id)] = r.settlement_id
        add(_title(r.cs_name), rec, "cs_name")
        if short:
            add(short, rec, "cs_name")                     # the town as the lists print it
        if isinstance(r.name_1910, str) and norm(r.name_1910) != norm(r.cs_name):
            add(r.name_1910, rec, "name_1910")
        de = getattr(r, "de_name", None)
        if isinstance(de, str) and norm(de) != norm(r.cs_name):
            add(de, rec, "de_name")
            for cz in czech_spellings(de):
                add(cz, rec, "de_name")
    if aliases is not None:
        recs = {x["settlement_id"]: {k: v for k, v in x.items() if k != "via"}
                for lst in index.values() for x in lst}
        targets: dict[tuple[str, str], list[str]] = defaultdict(list)
        for r in s.itertuples(index=False):
            targets[(norm(r.cs_name), norm(r.okres_1910 or ""))].append(r.settlement_id)
        for a in aliases.itertuples(index=False):
            hit = targets.get((norm(a.viz_target), norm(a.viz_okres if isinstance(a.viz_okres, str)
                                                         else "")), [])
            if len(hit) == 1 and hit[0] in recs:
                add(_title(a.cs_name), recs[hit[0]], "alias")
    # A court or okres seat with no settlement row of its own in its okres (28 courts:
    # towns the table lost or holds under a garbled name, e.g. "ÚSTÍ NAD ORL ICÍ") is
    # indexed at okres level: settlement unknown, okres certain.
    if districts is not None:
        # a statutory city's row sits under the city okres (Brno under "Brno (město)"),
        # so a seat whose name is an obec anywhere is not missing, only filed elsewhere
        obce = s[s["status_1910"] == "obec"]["cs_name"]
        have = {n for n, _ in by_name} | {norm(n) for n in obce} | {
            norm(short_name(_title(n))) for n in obce if short_name(_title(n))}
        for r in districts.itertuples(index=False):
            okres_id = int(str(r.court_id)[:6]) if pd.notna(r.court_id) else None
            for nm in {r.gerichtsbezirk_cs, r.okres_cs}:
                if not isinstance(nm, str) or not okres_id:
                    continue
                nm = re.sub(r"\s*\(.*\)", "", nm)
                if norm(nm) in have:
                    continue
                rec = {"settlement_id": None, "okres_id": okres_id, "land": r.land,
                       "status": "district", "okres_name": r.okres_cs or "",
                       "court": r.gerichtsbezirk_cs or "", "seat": True}
                add(nm, rec, "district")
                if short_name(nm):
                    add(short_name(nm), rec, "district")
    for old, new in HISTORIC_NAMES.items():
        pick, _ = decide([dict(c, via="cs_name") for c in index.get(norm(new), [])], "", None)
        if pick is not None:
            add(old, {k: v for k, v in pick.items() if k != "via"}, "historic")
    return dict(index)

def split_residence(text: str) -> tuple[str, str, str | None]:
    """``(place, qualifier, land)`` of a printed residence: the place before any
    ``u …``/``v …``/comma qualifier, the qualifier itself, and a land named in it."""
    raw = re.sub(r"\([^)]*\)", " ", str(text))
    raw = re.sub(r"[«»^|\"'‘’„“*]", "", raw)                  # stray OCR marks
    raw = re.sub(r"(?<=\s)\.(?=\w)", "", raw)                  # "Novém .Městě"
    key = re.sub(r"\b(nad|pod|na)\.", r"\1", norm(raw))
    key = re.sub(r"\bva (?=mor)", "na ", key)                  # "va Mor." = na Mor.
    land = next((_LAND_OF[k] for k, rx in _LAND_HINTS if rx.search(key)), None)
    m = re.search(r"\s(?:u|v|ve|na|pri|blize|blizko|pod)\s|,", key)
    place, qualifier = (key[:m.start()], key[m.end():]) if m else (key, "")
    # a name that itself carries "nad/pod/na" keeps it: Kostelci nad Orlicí, Ústí nad Labem
    m2 = re.match(r"^(.+?\s(?:nad|n|pod)\s\S+)", key)
    if m2 and len(m2.group(1)) > len(place):
        place, qualifier = m2.group(1), key[m2.end():]
    # an interior "n." is "nad" (Týnci n. Sázavou); a leading "N." is Nový / Německý
    place = re.sub(r"(?<=\S)\s+n\.?\s+(?=\S)", " nad ", place)
    return place.strip(" ."), qualifier.strip(" ."), land

def fuzzy_key(place: str, keys: list[str]) -> str | None:
    """The index key an OCR-misspelt place most likely is (*voduanech* → *vodnanech*):
    one key scoring at least ``FUZZY_CUTOFF`` and ``FUZZY_MARGIN`` above any other, among
    keys of nearly the same length; places under six letters are never guessed."""
    from rapidfuzz import fuzz, process
    if len(place) < 6:
        return None
    pool = [k for k in keys if abs(len(k) - len(place)) <= 2]
    hits = process.extract(place, pool, scorer=fuzz.ratio, limit=2, score_cutoff=FUZZY_CUTOFF - 10)
    if not hits or hits[0][1] < FUZZY_CUTOFF:
        return None
    if len(hits) > 1 and hits[0][1] - hits[1][1] < FUZZY_MARGIN:
        return None
    return hits[0][0]

"""
Czech place names printed in the locative (*v Brně*, *ve Vodňanech*, *v Hradci Králové*)
matched to the 1910 settlements of the Czech lands.

Nineteenth-century member lists give a residence after a preposition, so the name is in
the locative case, while every place table here (``settlements_1910``, its aliases) holds
the nominative. Rather than guess a nominative back from each printed form, this module
**declines every 1910 name forward** into the locatives Czech grammar allows and indexes
them, so a printed *Vodňanech* finds *Vodňany* and nothing else is invented. The rules
over-generate on purpose (an unused form costs nothing); what keeps them honest is that a
key claimed by more than one settlement is decided only by evidence the entry carries —
a court seat, a qualifier (*u Příbrami*, *na Moravě*) — and refused otherwise.

The match, for one residence:

``matched``
    one settlement (``how`` says why: ``unique``, ``seat``, ``qualifier``, ``land``,
    ``obec``).
``prague``
    Prague and its historic quarters, which ``settlements_1910`` does not carry as
    settlements: okres Praha, no settlement.
``ambiguous``
    two or more settlements and nothing to choose between them (``n_candidates``).
``outside``
    a place outside the Czech lands the lists name often (Vienna, Pest, Galicia …).
``unmatched``
    nothing found.
"""

def norm(text: str) -> str:
    """Lowercase, no accents, one space — the key both sides are compared on."""
    s = "".join(c for c in unicodedata.normalize("NFKD", str(text)) if not unicodedata.combining(c))
    return re.sub(r"\s+", " ", s.lower().replace("-", " ")).strip(" .,")

# Declension: nominative → candidate locatives
# ---------------------------------------------------------------------------
_PALATAL = (("ch", "š"), ("h", "z"), ("k", "c"), ("r", "ř"), ("g", "z"))
_SOFT_END = tuple("žščřňďťjcl") + ("m", "v", "ť", "d", "t", "s", "z", "b", "p", "n", "f")
#: Prepositions inside a name; the words after one are not declined (*nad Orlicí*).
_NAME_PREPS = {"nad", "pod", "na", "u", "v", "ve", "při", "pri", "n.", "p."}


def _palatalise(stem: str) -> str:
    for a, b in _PALATAL:
        if stem.endswith(a):
            return stem[: -len(a)] + b
    return stem


def noun_locatives(w: str) -> set[str]:
    """Locatives a noun may take (singular and plural place names alike)."""
    out = {w}
    if len(w) > 4 and w[-2] == "e" and w[-1] not in "aeiouyáéěíóúůý":   # e-drop:
        out |= {w[:-2] + w[-1] + "i", w[:-2] + w[-1] + "u", w[:-2] + w[-1] + "e"}  # Litovel → Litovli
    if w.endswith("ice"):                                   # Pardubice → Pardubicích, Roudnice → Roudnici
        out |= {w[:-1] + "ích", w[:-1] + "i"}
    elif w.endswith(("y", "i")) and len(w) > 3:             # Vodňany → Vodňanech, Vary → Varech
        out |= {w[:-1] + "ech", w[:-1] + "ách", w[:-1] + "ích"}
    elif w.endswith(("e", "ě")):                            # Lázně → Lázních, Hradiště → Hradišti
        out |= {w[:-1] + "ích", w[:-1] + "i", w[:-1] + "ích"}
    elif w.endswith("í"):                                   # Meziříčí, Ústí: unchanged
        pass
    elif w.endswith("a"):                                   # Praha → Praze, Hora → Hoře
        stem = w[:-1]
        out |= {_palatalise(stem) + "e", stem + "ě", stem + "e", stem + "i"}
    elif w.endswith("o"):                                   # Brno → Brně, Rakousko → Rakousku
        stem = w[:-1]
        out |= {stem + "ě", stem + "u", _palatalise(stem) + "e", stem + "e"}
    else:                                                   # consonant: Tábor, Jičín, Písek …
        out |= {w + "ě", w + "e", w + "u", _palatalise(w) + "e", w + "i"}
        if w.endswith("ek"):                                # Písek → Písku, Hrádek → Hrádku
            out |= {w[:-2] + "ku", w[:-2] + "ce"}
        if w.endswith("ec"):                                # Hradec → Hradci
            out.add(w[:-2] + "ci")
        if w.endswith("eň"):                                # Plzeň → Plzni
            out.add(w[:-2] + "ni")
        if w.endswith("es"):                                # Ves → Vsi
            out.add(w[:-2] + "si")
        if "ů" in w:                                        # Dvůr → Dvoře, Brůmov → Brumově?
            v = w[::-1].replace("ů", "o", 1)[::-1]
            out |= {v + "ě", v + "e", _palatalise(v) + "e", v + "u"}
    return out


def adjective_locatives(w: str) -> set[str]:

    out: set[str] = set()

        out.add(w[:-1] + "é")

        pass
    elif w.endswith("a"):                                   # Praha → Prahy, Lhota → Lhoty
        out |= {w[:-1] + "y", w[:-1] + "i"}
    elif w.endswith("o"):                                   # Brno → Brna
        out.add(w[:-1] + "a")
    else:                                                   # Nymburk → Nymburka, Písek → Písku
        out |= {w + "a", w + "u", w + "i", w + "e"}
        if len(w) > 4 and w[-2] == "e":
            out |= {w[:-2] + w[-1] + "u", w[:-2] + w[-1] + "a", w[:-2] + w[-1] + "e"}
        if w.endswith("eň"):                                # Plzeň → Plzně
            out.add(w[:-2] + "ně")
        if "ů" in w:
            v = w[::-1].replace("ů", "o", 1)[::-1]
            out |= {v + "a", v + "u"}
    return out


def adjective_genitives(w: str) -> set[str]:
    out: set[str] = set()
    if w.endswith("á"):
        out.add(w[:-1] + "é")
    elif w.endswith(("é", "ý")):
        out |= {w[:-1] + "ého", w[:-1] + "ých"}
    elif w.endswith("í"):
        out |= {w[:-1] + "ího", w[:-1] + "ích"}
    elif w.endswith("ův"):
        out.add(w[:-2] + "ova")
    elif w.endswith("ovy"):
        out.add(w[:-1] + "ých")
    return out


def locative_keys(name: str, cap: int = 4000) -> set[str]:
    """Every normalised locative and genitive (and the nominative itself) of a place name."""
    words = name.lower().split()
    if not words:
        return set()
    cut = next((i for i, t in enumerate(words) if i > 0 and t in _NAME_PREPS), len(words))
    head, tail = words[:cut], words[cut:]
    if len(head) > 4:
        return {norm(name)}
    keys = set()
    for adj, noun in ((adjective_locatives, noun_locatives), (adjective_genitives, noun_genitives)):
        options = [{w} | adj(w) if w.endswith(("á", "é", "ý", "ův", "ova", "ovo"))
                   else {w} | noun(w) | adj(w) for w in head]
        for combo in itertools.islice(itertools.product(*options), cap):
            keys.add(norm(" ".join(list(combo) + tail)))
    return keys


#: German spelling → the Czech spelling the lists print a German name in (Schwarzbach →
#: Švarcbach, Schildberg → Šildperk), applied in order.
_CZECHIZE = (("tsch", "č"), ("sch", "š"), ("tz", "c"), ("z", "c"), ("w", "v"), ("ei", "aj"),
             ("eu", "oj"), ("äu", "oj"), ("ck", "k"), ("ß", "s"), ("ss", "s"), ("ie", "í"),
             ("ä", "e"), ("ö", "e"), ("ü", "i"), ("ph", "f"), ("th", "t"), ("qu", "kv"),
             ("y", "i"))


def czech_spellings(german: str) -> set[str]:
    """A German place name as a Czech list would spell it, with the usual endings
    (*-berg* as *-berk*/*-perk*, *-burg* as *-burk*/*-purk*)."""
    s = german.lower()
    for a, b in _CZECHIZE:
        s = s.replace(a, b)
    out = {s}
    for a, bs in (("berg", ("berk", "perk")), ("burg", ("burk", "purk"))):
        if s.endswith(a):
            out |= {s[: -len(a)] + b for b in bs}
    return out


def short_name(name: str) -> str | None:
    """A town's name without the river or neighbour that qualifies it (*Roudnice nad
    Labem* → *Roudnice*, *Slavkov u Brna* → *Slavkov*), as the lists often print it."""
    m = re.match(r"^(.+?)\s(?:nad|pod|u|na|v)\s", name, flags=re.I)
    return m.group(1) if m else None

def _title(cs_name: str) -> str:
    return " ".join(w if w.isupper() and len(w) <= 4 and w.endswith(".") else w.capitalize()
                    for w in str(cs_name).lower().split())


def build_index(settlements: pd.DataFrame, aliases: pd.DataFrame | None = None,
                districts: pd.DataFrame | None = None) -> dict[str, list[dict]]:
    """Normalised locative → the settlements it may name.

    Each settlement is indexed under its Czech name and, where it differs, its 1910 name;
    each alias (an older Czech name, ``settlement_aliases_1910``) under its own name, for
    the settlement it points to. A candidate records whether the settlement is the seat
    of its court district (the name of a ``gerichtsbezirk_cs`` in its own okres)."""
    seats: set[tuple[str, int]] = set()
    if districts is not None:
        for r in districts.itertuples(index=False):
            okres_id = int(str(r.court_id)[:6]) if pd.notna(r.court_id) else None
            for nm in (r.gerichtsbezirk_cs, r.okres_cs):
                if isinstance(nm, str) and okres_id:
                    nm = re.sub(r"\s*\(.*\)", "", nm)
                    seats.add((norm(nm), okres_id))
                    if short_name(nm):
                        seats.add((norm(short_name(nm)), okres_id))
    s = settlements.dropna(subset=["okres_id"])
    by_name: dict[tuple[str, int], str] = {}
    index: dict[str, list[dict]] = defaultdict(list)

    def add(name: str, rec: dict, via: str) -> None:
        uid = rec["settlement_id"] or f"okres:{rec['okres_id']}"
        for key in locative_keys(name):
            if not any((c["settlement_id"] or f"okres:{c['okres_id']}") == uid for c in index[key]):
                index[key].append({**rec, "via": via})

    for r in s.itertuples(index=False):
        okres_id = int(r.okres_id)
        short = short_name(_title(r.cs_name))
        rec = {"settlement_id": r.settlement_id, "okres_id": okres_id, "land": r.land,
               "status": r.status_1910, "okres_name": r.okres_1910 or "",
               "court": getattr(r, "court_cs", "") or "",
               "seat": (norm(r.cs_name), okres_id) in seats
               or (short is not None and (norm(short), okres_id) in seats),
               "lat": getattr(r, "lat", None), "lon": getattr(r, "lon", None)}
        by_name[(norm(r.cs_name), okres_id)] = r.settlement_id
        add(_title(r.cs_name), rec, "cs_name")
        if short:
            add(short, rec, "cs_name")                     # the town as the lists print it
        if isinstance(r.name_1910, str) and norm(r.name_1910) != norm(r.cs_name):
            add(r.name_1910, rec, "name_1910")
        de = getattr(r, "de_name", None)
        if isinstance(de, str) and norm(de) != norm(r.cs_name):
            add(de, rec, "de_name")
            for cz in czech_spellings(de):
                add(cz, rec, "de_name")
    if aliases is not None:
        recs = {x["settlement_id"]: {k: v for k, v in x.items() if k != "via"}
                for lst in index.values() for x in lst}
        targets: dict[tuple[str, str], list[str]] = defaultdict(list)
        for r in s.itertuples(index=False):
            targets[(norm(r.cs_name), norm(r.okres_1910 or ""))].append(r.settlement_id)
        for a in aliases.itertuples(index=False):
            hit = targets.get((norm(a.viz_target), norm(a.viz_okres if isinstance(a.viz_okres, str)
                                                         else "")), [])
            if len(hit) == 1 and hit[0] in recs:
                add(_title(a.cs_name), recs[hit[0]], "alias")
    # A court or okres seat with no settlement row of its own in its okres (28 courts:
    # towns the table lost or holds under a garbled name, e.g. "ÚSTÍ NAD ORL ICÍ") is
    # indexed at okres level: settlement unknown, okres certain.
    if districts is not None:
        # a statutory city's row sits under the city okres (Brno under "Brno (město)"),
        # so a seat whose name is an obec anywhere is not missing, only filed elsewhere
        obce = s[s["status_1910"] == "obec"]["cs_name"]
        have = {n for n, _ in by_name} | {norm(n) for n in obce} | {
            norm(short_name(_title(n))) for n in obce if short_name(_title(n))}
        for r in districts.itertuples(index=False):
            okres_id = int(str(r.court_id)[:6]) if pd.notna(r.court_id) else None
            for nm in {r.gerichtsbezirk_cs, r.okres_cs}:
                if not isinstance(nm, str) or not okres_id:
                    continue
                nm = re.sub(r"\s*\(.*\)", "", nm)
                if norm(nm) in have:
                    continue
                rec = {"settlement_id": None, "okres_id": okres_id, "land": r.land,
                       "status": "district", "okres_name": r.okres_cs or "",
                       "court": r.gerichtsbezirk_cs or "", "seat": True}
                add(nm, rec, "district")
                if short_name(nm):
                    add(short_name(nm), rec, "district")
    for old, new in HISTORIC_NAMES.items():
        pick, _ = decide([dict(c, via="cs_name") for c in index.get(norm(new), [])], "", None)
        if pick is not None:
            add(old, {k: v for k, v in pick.items() if k != "via"}, "historic")
    return dict(index)

def split_residence(text: str) -> tuple[str, str, str | None]:
    """``(place, qualifier, land)`` of a printed residence: the place before any
    ``u …``/``v …``/comma qualifier, the qualifier itself, and a land named in it."""
    raw = re.sub(r"\([^)]*\)", " ", str(text))
    raw = re.sub(r"[«»^|\"'‘’„“*]", "", raw)                  # stray OCR marks
    raw = re.sub(r"(?<=\s)\.(?=\w)", "", raw)                  # "Novém .Městě"
    key = re.sub(r"\b(nad|pod|na)\.", r"\1", norm(raw))
    key = re.sub(r"\bva (?=mor)", "na ", key)                  # "va Mor." = na Mor.
    land = next((_LAND_OF[k] for k, rx in _LAND_HINTS if rx.search(key)), None)
    m = re.search(r"\s(?:u|v|ve|na|pri|blize|blizko|pod)\s|,", key)
    place, qualifier = (key[:m.start()], key[m.end():]) if m else (key, "")
    # a name that itself carries "nad/pod/na" keeps it: Kostelci nad Orlicí, Ústí nad Labem
    m2 = re.match(r"^(.+?\s(?:nad|n|pod)\s\S+)", key)
    if m2 and len(m2.group(1)) > len(place):
        place, qualifier = m2.group(1), key[m2.end():]
    # an interior "n." is "nad" (Týnci n. Sázavou); a leading "N." is Nový / Německý
    place = re.sub(r"(?<=\S)\s+n\.?\s+(?=\S)", " nad ", place)
    return place.strip(" ."), qualifier.strip(" ."), land


def _abbrev_regex(place: str) -> re.Pattern | None:
    """A printed name with abbreviated words (``jindr. hradci``, ``hradci kral``) as a
    pattern: a word printed with a full stop, or of three letters or fewer, may be the
    start of a longer one; the others must match whole."""
    toks = place.split()
    if len(toks) < 2 or not any(t.endswith(".") or len(t) <= 4 for t in toks):
        return None
    parts = [re.escape(t.rstrip(".")) + (r"\S*" if t.endswith(".") or len(t.rstrip(".")) <= 4 else "")
             for t in toks]
    return re.compile("^" + r"\s".join(parts) + "$")


def _variants(place: str) -> list[str]:
    """The printed key, then old spellings and an inverted two-word name
    (*Meziříčí Velké* for *Velké Meziříčí*)."""
    out = [place, re.sub(r"^ou", "u", place), re.sub(r"^holomou", "olomou", place)]
    toks = place.split()
    if len(toks) == 2:
        out.append(f"{toks[1]} {toks[0]}")
    # the OCR's commonest misreadings of this typeface: "cli" for "ch", "x" for "h",
    # a word-initial "Y" for "V" (Yorlíka), a word cut in two (Libranti cích)
    repaired = re.sub(r"cli\b", "ch", place).replace("x", "h")
    repaired = re.sub(r"(^|\s)y", r"\1v", repaired)
    out += [repaired, re.sub(r"(?<=\w) (?=\w{1,4}\b)", "", repaired)]
    # nineteenth-century spelling: "ou" for "ú/u" (Žehouň), Hořejní/Dolejší for
    # Horní/Dolní, Mezeříčí for Meziříčí
    old = re.sub(r"\bhorejs?n?(?=[ie])", "horn", repaired)        # horejni(m) → horni(m)
    old = re.sub(r"\bdolejs(?=[ie])", "doln", old)                # dolejsi(ch) → dolni(ch)
    old = old.replace("mezer", "mezir")
    out += [old, old.replace("ou", "u")]
    return list(dict.fromkeys(v for v in out if v))


#: Set by reading all 85 fuzzy matches the matcher made at 92 (2026-10-29): every one at
#: 96 or above was right, while between 92 and 96 several were wrong places (*Černém
#: Kostelci* → Červený Kostelec, *Zlomcích* → Lomec, *Charkově* → Charov for Kharkiv).
FUZZY_CUTOFF = 96      # rapidfuzz ratio a misspelt key must reach …
FUZZY_MARGIN = 4       # … and by how much it must beat the next-best key


def fuzzy_key(place: str, keys: list[str]) -> str | None:
    """The index key an OCR-misspelt place most likely is (*voduanech* → *vodnanech*):
    one key scoring at least ``FUZZY_CUTOFF`` and ``FUZZY_MARGIN`` above any other, among
    keys of nearly the same length; places under six letters are never guessed."""
    from rapidfuzz import fuzz, process
    if len(place) < 6:
        return None
    pool = [k for k in keys if abs(len(k) - len(place)) <= 2]
    hits = process.extract(place, pool, scorer=fuzz.ratio, limit=2, score_cutoff=FUZZY_CUTOFF - 10)
    if not hits or hits[0][1] < FUZZY_CUTOFF:
        return None
    if len(hits) > 1 and hits[0][1] - hits[1][1] < FUZZY_MARGIN:
        return None
    return hits[0][0]


NEAR_KM = 30            # a qualifier (*u Plzně*) corroborates a settlement this close to it
FUZZY_NEAR_CUTOFF = 85  # a misspelling this close is accepted only when so corroborated


def _km(a: tuple[float, float], b: tuple[float, float]) -> float:
    """Great-circle distance in km between two (lat, lon) points."""
    import math
    la1, lo1, la2, lo2 = map(math.radians, (*a, *b))
    h = (math.sin((la2 - la1) / 2) ** 2
         + math.cos(la1) * math.cos(la2) * math.sin((lo2 - lo1) / 2) ** 2)
    return 6371 * 2 * math.asin(math.sqrt(h))


def qualifier_point(qualifier: str, index: dict[str, list[dict]]) -> tuple[float, float] | None:
    """Where the place a qualifier names lies (*u Plzně* → Plzeň), if that place is
    itself found without doubt (one settlement, or its own name, or a court seat)."""
    head = re.split(r"\s+(?:u|v|ve|na)\s+|,", qualifier or "")[0].strip()
    if not head:
        return None
    pick, why = decide([dict(c) for c in index.get(head, [])], "", None)
    if pick is None or why not in ("unique", "own_name", "seat") or pd.isna(pick.get("lat")):
        return None
    return float(pick["lat"]), float(pick["lon"])


def near_candidates(cands: list[dict], near: tuple[float, float]) -> list[dict]:
    return [c for c in cands if pd.notna(c.get("lat")) and _km(near, (c["lat"], c["lon"])) <= NEAR_KM]


def fuzzy_near(place: str, keys: list[str], index: dict[str, list[dict]],
               near: tuple[float, float]) -> dict | None:
    """A misspelt place scoring at least ``FUZZY_NEAR_CUTOFF`` against an index key,
    accepted only if exactly one settlement among the close keys lies within
    ``NEAR_KM`` of the place its qualifier names (*Pia šech u Plzně* → Plasy)."""
    from rapidfuzz import fuzz, process
    if len(place) < 5:
        return None
    pool = [k for k in keys if abs(len(k) - len(place)) <= 3]
    hits = process.extract(place, pool, scorer=fuzz.ratio, limit=8, score_cutoff=FUZZY_NEAR_CUTOFF)
    found = {c["settlement_id"]: c for k, _, _ in hits
             for c in near_candidates(index[k], near) if c["settlement_id"]}
    return next(iter(found.values())) if len(found) == 1 else None


def match_residence(residence: str, index: dict[str, list[dict]],
                    by_last: dict[str, list[str]] | None = None,
                    fuzzy_keys: list[str] | None = None) -> dict:
    """The 1910 settlement a printed residence names, with how it was decided."""
    out = {"settlement_id": None, "okres_id": None, "place_status": "unmatched",
           "place_how": "", "n_candidates": 0, "place_key": ""}
    if not isinstance(residence, str) or not residence.strip():
        out["place_status"] = "none"
        return out
    place, qualifier, land = split_residence(residence)
    out["place_key"] = place
    if any(v in PRAGUE or re.match(r"^(?:praze|pr)\b", v) for v in _variants(place)):
        out.update(okres_id=PRAGUE_OKRES_ID, place_status="prague", place_how="prague")
        return out
    parts = [p for p in re.split(r"\s+(?:u|v|ve|na)\s+|,\s*", qualifier or "") if p]
    if OUTSIDE.match(place) or any(OUTSIDE.match(p) for p in parts):
        out.update(place_status="outside", place_how="outside")
        return out
    if place in LAND_ONLY:
        out.update(place_status="land_only", place_how="land_only")
        return out
    cands, how = [], "exact"
    for v in _variants(place):
        cands = index.get(v, [])
        if cands:
            how = "exact" if v == place else "variant"
            break
    if not cands and by_last is not None:
        for v in _variants(place):
            rx = _abbrev_regex(v)
            if rx:
                keys = [k for k in by_last.get(v.split()[-1].rstrip(".")[:3], []) if rx.match(k)]
                cands = list({c["settlement_id"]: c for k in keys for c in index[k]}.values())
                how = "abbrev"
                if cands:
                    break
    plain = [v for v in _variants(place) if "." not in v]       # the forms fuzzy may try
    if not cands and fuzzy_keys is not None:
        key = next((k for k in (fuzzy_key(v, fuzzy_keys) for v in plain) if k), None)
        if key:
            cands, how = index[key], "fuzzy"
    near = qualifier_point(qualifier, index) if qualifier else None
    if not cands and fuzzy_keys is not None and near is not None:
        hit = next((h for h in (fuzzy_near(v, fuzzy_keys, index, near) for v in plain) if h), None)
        if hit is not None:
            out.update(settlement_id=hit["settlement_id"], okres_id=hit["okres_id"],
                       place_status="matched", place_how="fuzzy:qualifier_near", n_candidates=1)
            return out
    out["n_candidates"] = len(cands)
    if not cands:
        return out
    pick, why = decide(cands, qualifier, land, near)
    if pick is None:
        out.update(place_status="ambiguous", place_how=why)
        return out
    out.update(settlement_id=pick["settlement_id"], okres_id=pick["okres_id"],
               place_status="matched", place_how=f"{how}:{why}")
    return out


def decide(cands: list[dict], qualifier: str, land: str | None,
           near: tuple[float, float] | None = None) -> tuple[dict | None, str]:
    """One settlement out of several, or ``None``: by uniqueness, then the land named,
    then a qualifier naming the okres or court (``u Příbrami``) or a place within
    ``NEAR_KM`` of it (``near``), then a name that is the settlement's own present one
    (not its 1910 form or an alias), then a court seat, then the one *obec* among
    *osady*. The last two are priors, not evidence: their ``how`` says so, and a strict
    reading drops them."""
    if len(cands) == 1:
        return cands[0], "unique"
    pool = cands
    if land:
        in_land = [c for c in pool if c["land"] == land]
        if len(in_land) == 1:
            return in_land[0], "land"
        pool = in_land or pool
    if qualifier:
        stems = {w[:4] for w in qualifier.split() if len(w) >= 4}
        named = [c for c in pool if stems & {w[:4] for w in norm(f"{c['okres_name']} {c['court']}").split()
                                             if len(w) >= 4}]
        if len(named) == 1:
            return named[0], "qualifier"
        pool = named or pool
    if near is not None:
        close = near_candidates(pool, near)
        if len(close) == 1:
            return close[0], "qualifier_near"
        pool = close or pool
    own = [c for c in pool if c.get("via") in ("cs_name", "district")]
    if len(own) == 1:
        return own[0], "own_name"
    pool = own or pool
    seats = [c for c in pool if c["seat"]]
    if len(seats) == 1:
        return seats[0], "seat"
    obce = [c for c in pool if c["status"] == "obec"]
    if len(obce) == 1 and len(pool) > 1:
        return obce[0], "obec"
    return None, f"{len(pool)} candidates"


def last_word_index(index: dict[str, list[dict]]) -> dict[str, list[str]]:
    """Index keys by their last word, for matching abbreviated names."""
    out: dict[str, list[str]] = defaultdict(list)
    for k in index:
        out[k.split()[-1][:3]].append(k)
    return dict(out)


def match_all(residences: pd.Series, index: dict[str, list[dict]]) -> pd.DataFrame:
    """:func:`match_residence` for a column of residences (cached per distinct string)."""
    by_last = last_word_index(index)
    keys = list(index)
    cache = {r: match_residence(r, index, by_last, keys) for r in residences.dropna().unique()}
    empty = match_residence("", index)
    return pd.DataFrame([cache.get(r, empty) if isinstance(r, str) else empty
                         for r in residences], index=residences.index)
