How it works

  • ? / - / empty strings used as missing-value placeholders;
  • surnames wrapped in parentheses or quotes — (Abel), ("Adamek-ova") — sometimes with a trailing ? flagging an uncertain reading ((Adet?));
  • place names carrying a trailing + verification marker ("Oloví, okres Falknov nad Ohří+").

The module is deliberately dataset-agnostic so it can be reused on the legionnaire records as well as the VÚA fallen-soldier dump.

What is kept on purpose

okres … district text and German exonym parentheticals inside place fields are preserved verbatim — they are the join key for the later 1910 Cisleithanian census merge. Only the trailing + marker and redundant whitespace are removed.

def normalize_missing(
    df: pd.DataFrame,
    cols: list[str] | None = None,
) -> pd.DataFrame:
    """Map placeholder strings (``?``, ``-``, empty, …) to ``pd.NA`` and strip.

    Parameters
    ----------
    df:
        Input DataFrame (not modified in place).
    cols:
        Columns to normalise.  Defaults to every text (object/string) column.

    Returns
    -------
    pd.DataFrame
        Copy of *df* with placeholders replaced by ``pd.NA`` and surviving
        string values whitespace-stripped.  Idempotent.
    """
    df = df.copy()
    if cols is None:
        cols = list(df.select_dtypes(include=["object", "string"]).columns)

    for c in cols:
        if c not in df.columns:
            continue
        df[c] = df[c].map(lambda v: pd.NA if _is_missing(v) else str(v).strip())

    return df

    df: pd.DataFrame,
    surname_col: str = "Příjmení",
    given_col: str = "Jméno",
) -> pd.DataFrame:
    """Append cleaned ``surname`` / ``given`` columns and a ``name_uncertain`` flag.

    The original raw columns are left untouched.  ``name_uncertain`` is True
    when either the surname or given name carried a trailing ``?``.
    """

    if surname_col in df.columns:

        df["surname"] = res.map(lambda t: t[0])

        df["given"] = res.map(lambda t: t[0])

def strip_place_markers(v) -> object:
    """Drop ``+`` verification markers (ALL occurrences) and collapse whitespace.

    Keeps ``okres …`` text and German exonym parentheticals intact (needed for
    the census merge); returns ``pd.NA`` for empty results.

    if pd.isna(v):

    s = re.sub(r"\s{2,}", " ", s).strip()
    return s if s else pd.NA

"""
clean.py — Generic raw-field cleaning
=====================================
First-pass normalisation of scraped archival records, applied **before** any
field-specific parsing (dates, causes, units, …).  The VÚA and csol.cz dumps
share the same surface noise:

* ``?`` / ``-`` / empty strings used as missing-value placeholders;
* surnames wrapped in parentheses or quotes — ``(Abel)``, ``("Adamek-ova")`` —
  sometimes with a trailing ``?`` flagging an uncertain reading (``(Adet?)``);
* place names carrying a trailing ``+`` verification marker
  (``"Oloví, okres Falknov nad Ohří+"``).

The module is deliberately dataset-agnostic so it can be reused on the
legionnaire records as well as the VÚA fallen-soldier dump.

What is **kept** on purpose
---------------------------
``okres …`` district text and German exonym parentheticals inside place
fields are preserved verbatim — they are the join key for the later 1910
Cisleithanian census merge.  Only the trailing ``+`` marker and redundant
whitespace are removed.

Public API
----------
    from utils.parsing.clean import clean_dataframe, clean_report
    df = clean_dataframe(df)          # normalise missing + names + places
    clean_report(df)

    # or the individual passes
    df = normalize_missing(df)
    df = clean_names(df)
    df = clean_places(df)
"""

from __future__ import annotations

import re

import pandas as pd

__version__ = "1.0.1"

# ── Missing-value detection ───────────────────────────────────────────────────
# A cell is treated as missing when, after stripping, it is empty or consists
# solely of punctuation / question marks (``?``, ``??``, ``(?)``, ``(?)+``,
# ``-``, ``–``, ``.``, ``/`` …).  Any alphanumeric content keeps the value.
_NONINFO_RE = re.compile(r"^[\s?\-–—.()+/,;:]*$")


def _is_missing(v) -> bool:
    """True if *v* is NaN/None or a non-informative placeholder string."""
    if pd.isna(v):
        return True
    s = str(v).strip()
    return s == "" or bool(_NONINFO_RE.match(s))


def normalize_missing(
    df: pd.DataFrame,
    cols: list[str] | None = None,
) -> pd.DataFrame:
    """Map placeholder strings (``?``, ``-``, empty, …) to ``pd.NA`` and strip.

    Parameters
    ----------
    df:
        Input DataFrame (not modified in place).
    cols:
        Columns to normalise.  Defaults to every text (object/string) column.

    Returns
    -------
    pd.DataFrame
        Copy of *df* with placeholders replaced by ``pd.NA`` and surviving
        string values whitespace-stripped.  Idempotent.
    """
    df = df.copy()
    if cols is None:
        cols = list(df.select_dtypes(include=["object", "string"]).columns)

    for c in cols:
        if c not in df.columns:
            continue
        df[c] = df[c].map(lambda v: pd.NA if _is_missing(v) else str(v).strip())

    return df


# ── Name cleaning ─────────────────────────────────────────────────────────────
_WRAP_QUOTES = "\"'„“”‚‘’"
_TRAILING_Q_RE = re.compile(r"\?+\s*$")


def _clean_name_value(v) -> tuple[object, bool]:
    """Return ``(clean_name, uncertain)`` for a single raw name cell.

    Repeatedly peels wrapping parentheses and quotes (``("Adamek-ova")`` →
    ``Adamek-ova``); a trailing ``?`` (``(Adet?)`` → ``Adet``) sets the
    uncertainty flag.
    """
    if pd.isna(v):
        return pd.NA, False

    s = str(v).strip()
    uncertain = bool(_TRAILING_Q_RE.search(s))

    prev = None
    while prev != s and s:
        prev = s
        if len(s) >= 2 and s[0] in "(" and s[-1] in ")":
            s = s[1:-1].strip()
        if len(s) >= 2 and s[0] in _WRAP_QUOTES and s[-1] in _WRAP_QUOTES:
            s = s[1:-1].strip()
        if _TRAILING_Q_RE.search(s):
            uncertain = True
            s = _TRAILING_Q_RE.sub("", s).strip()

    s = s.strip(_WRAP_QUOTES).strip()
    return (s if s else pd.NA), uncertain


def clean_names(
    df: pd.DataFrame,
    surname_col: str = "Příjmení",
    given_col: str = "Jméno",
) -> pd.DataFrame:

    The original raw columns are left untouched.  ``name_uncertain`` is True
    when either the surname or given name carried a trailing ``?``.

        res = df[surname_col].map(_clean_name_value)
        df["surname"] = res.map(lambda t: t[0])
        uncertain = uncertain | res.map(lambda t: t[1])

    if given_col in df.columns:
        res = df[given_col].map(_clean_name_value)
        df["given"] = res.map(lambda t: t[0])
        uncertain = uncertain | res.map(lambda t: t[1])

    df["name_uncertain"] = uncertain
    return df


# ── Place cleaning ────────────────────────────────────────────────────────────
_PLACE_COLS_DEFAULT = (
    "Místo narození",
    "Domovem",
    "Místo úmrtí",
    "Místo pohřbení",
)

        return pd.NA

def clean_places(
    df: pd.DataFrame,
    cols: tuple[str, ...] = _PLACE_COLS_DEFAULT,

def clean_dataframe(
    df: pd.DataFrame,
    *,
    names: bool = True,
    places: bool = True,
) -> pd.DataFrame:
    """Run the full cleaning pipeline: missing → names → places.

    Parameters
    ----------
    df:
        Raw input DataFrame (not modified in place).
    names:
        Apply :func:`clean_names` (adds ``surname``/``given``/``name_uncertain``).

    if places:
        df = clean_places(df)
    return df
