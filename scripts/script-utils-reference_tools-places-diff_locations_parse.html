How it works

Builds output/places/locations_v12_diff_* (until 2026-11-16 a scratch script, so those files had no builder). The old parser is the frozen copy _locations_v110 (git show HEAD:utils/parsing/locations.py at commit 853df72, version 1.1.0); the new one is utils.parsing.locations as it stands. Both read each distinct string once, after utils.parsing.clean.strip_place_markers (notebook 02's cleaning), and a string's row count is how often it occurs.

Columns compared: the v1.1 five (municipality, okres, german, land, cisleithanian) plus uncertain; n_candidates is reported separately, since 1.2 counts candidates rather than every ;. A changed string is labelled with the 1.2 reading that produced it (head:okres, candidate:statutory (head bare), ...); okres debris only marks a change that is nothing but 1.1's trailing - / ) on the okres.

The csol.cz columns are parsed too although notebook 02 never sends them through this parser (records_geo reads csol's two fields directly): they show what the parser would do to the shape.

Outputs (output/places/)::

locations_v12_diff_{vua_birth,vua_home,csol_birth_obec,csol_birth_okres,
                    csol_home_obec,csol_home_okres}.csv   one row per changed distinct string
locations_v12_diff_summary.txt                             the printed summary

Run from anywhere::

python utils/reference_tools/places/diff_locations_parse.py

Version 1.0.0 (2026-11-16): moved here from the scratchpad, unchanged in what it measures; also writes the summary file it used to print. Version 1.0.1 (2026-11-16): every writer LF (lineterminator / newline); the outputs of earlier runs were CRLF.

def diff_column(series: pd.Series, old) -> tuple[pd.DataFrame, dict]:
    """One row per distinct string whose parse changed, and the column's counts."""
    s = series.map(strip_place_markers).dropna()
    s = s[s.astype(str).str.strip() != ""]
    vc = s.value_counts()
    rows, ncand = [], collections.Counter()
    for txt, n in vc.items():
        o, w = old.parse_location(txt), new.parse_location(txt)
        if o["n_candidates"] != w["n_candidates"]:
            ncand[(o["n_candidates"], w["n_candidates"])] += n
        changed = [c for c in OLD7 if not same(o[c], w[c])]
        if o["uncertain"] != w["uncertain"]:
            changed.append("uncertain")
        if not changed:
            continue
        pat = label(txt)
        if changed == ["okres"] and pd.notna(o["okres"]) and same(new._clean_okres(o["okres"]), w["okres"]):
            pat = "okres debris only (" + pat + ")"
        rows.append({"raw": txt, "n": n, "pattern": pat, "changed": "+".join(changed),
                     **{f"old_{c}": o[c] for c in OLD7}, **{f"new_{c}": w[c] for c in OLD7},
                     "quarter": w["quarter"], "quarter_no": w["quarter_no"],
                     "statutory_city": w["statutory_city"], "n_candidates": w["n_candidates"],
                     "candidates_agree": w["candidates_agree"]})
    d = pd.DataFrame(rows)
    if len(d):
        d = d.sort_values("n", ascending=False)
    return d, {"non_empty": len(s), "distinct": len(vc), "ncand": ncand}

def report(name: str, d: pd.DataFrame, stats: dict) -> None:
    print(f"\n=== {name}: {stats['non_empty']:,} non-empty, {stats['distinct']:,} distinct")
    if d.empty:
        print("  no row changes in the v1.1 columns")
    else:
        print(f"  rows changed: {d.n.sum():,} ({len(d):,} distinct strings)")
        for c in OLD7:
            print(f"    {c:<14}{d[d.changed.str.contains(c)].n.sum():>8,}")
        print("  by pattern x changed columns:")
        print(d.groupby(["pattern", "changed"]).n.sum().sort_values(ascending=False).to_string())
    nc = stats["ncand"]
    print(f"  n_candidates changed: {sum(nc.values()):,} rows; top (old,new):", nc.most_common(8))

def build(out_dir: Path = OUT) -> dict[str, pd.DataFrame]:
    """Every column diffed; CSVs and the summary written to *out_dir*. Returns the diff frames."""
    out_dir.mkdir(parents=True, exist_ok=True)
    old = load_old()
    frames: dict[str, pd.DataFrame] = {}
    buf = io.StringIO()
    with redirect_stdout(buf):
        print(f"locations {old.__version__} (frozen copy) vs {new.__version__}  "
              f"[{Path(__file__).name} {__version__}]")
        sources = [(VUA_CSV, VUA_COLS), (CSOL_CSV, CSOL_COLS)]
        for path, cols in sources:
            raw = pd.read_csv(path, usecols=list(cols), dtype=str)
            for col, name in cols.items():
                d, stats = diff_column(raw[col], old)
                report(name, d, stats)
                frames[name] = d
                target = out_dir / f"locations_v12_diff_{name}.csv"
                if len(d):
                    d.to_csv(target, index=False, encoding="utf-8", lineterminator="\n")
                elif target.exists():
                    target.unlink()               # no change left: a stale file would claim one
    text = buf.getvalue()
    (out_dir / "locations_v12_diff_summary.txt").write_text(text, encoding="utf-8", newline="\n")
    print(text)
    return frames
