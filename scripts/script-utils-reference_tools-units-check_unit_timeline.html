unit_timeline.csv is parsed by build_unit_timeline.py from the formulas of the Fidler (2025) encyclopedia. A parser of prose fails quietly: a clause it does not recognise is dropped, or lands in the wrong column, and the table still looks full. These checks make the failures that can be seen mechanical.

How it works

python utils/reference_tools/units/check_unit_timeline.py

Exit code 0 if every hard invariant holds, 1 otherwise.

What it CANNOT see (stated, per the register's convention)

  • A month the encyclopedia itself has wrong. Nothing here is checked against a second source; the chronicles in library/diaries/ are the place to do that.
  • A clause dropped whole. T6 bounds it from one side only (every unit entry that has a location formula yields location rows), not sentence by sentence.
  • Place names. They are kept in the grammatical case of the source ("Borispolu") and are not matched to a gazetteer, so two spellings of one place are two places here.
  • The order of events inside one phase: the source gives the order, not the dates.

def checks(t: pd.DataFrame) -> list[tuple[str, bool, str, bool]]:
    """(label, ok, detail, hard) for every invariant."""
    out = []
    auth = pd.read_csv(AUTHORITY, dtype=str, encoding="utf-8")

    key = ["unit_cs", "subunit", "seq"]
    out.append(("T1 key (unit_cs, subunit, seq) unique and never missing",
                not t.duplicated(key).any() and (t["unit_cs"] != "").all()
                and (t["seq"] != "").all(),
                f"{t.duplicated(key).sum()} duplicated keys", True))

    stray = set(t["unit_cs"]) - set(auth["unit_cs"])
    out.append(("T2 every unit resolves to unit_authority.csv", not stray,
                f"{len(stray)} unknown: {sorted(stray)[:3]}", True))

    bad_kind = [(k, s) for k, s in zip(t["kind"], t["state"])
                if k not in KINDS or (KINDS[k] is not None and s not in KINDS[k])]
    out.append(("T3 kind and state in the declared vocabulary", not bad_kind,
                f"{len(bad_kind)} rows, e.g. {bad_kind[:3]}", True))

    bad_date = t[[not re.fullmatch(PRECISION.get(p, "x"), d)
                  for p, d in zip(t["date_from_precision"], t["date_from"])]]
    out.append(("T4 date_from matches its stated precision", bad_date.empty,
                f"{len(bad_date)} rows", True))
    dated = t[t["date_from"] != ""]
    span = dated[(dated["date_from"].str[:7] < FIRST) | (dated["date_from"].str[:7] > LAST)]
    out.append((f"T5 every date within {FIRST} … {LAST}", span.empty,
                f"{len(span)} rows, e.g. {span['date_from'].head(3).tolist()}", True))

    units = auth[auth["kind"] == "unit"]["unit_cs"]
    located = set(t[t["kind"] == "location"]["unit_cs"])
    missing = [u for u in units if u not in located]
    out.append(("T6 at least 95% of unit entries carry a location chain",
                len(missing) <= 0.05 * len(units),
                f"{len(missing)} of {len(units)} without: {missing[:3]}", True))

    back = 0
    ordered = t[(t["kind"] == "location") & (t["date_from"] != "")
                & (t["date_suspect"] != "1")]
    for _, g in ordered.groupby(key[:2]):
        months = g.sort_values("seq", key=lambda s: s.astype(int))["date_from"].str[:7]
        back += int((months.values[1:] < months.values[:-1]).sum())
    out.append(("T7 no location chain runs backwards once the flagged rows are set aside",
                back == 0, f"{back} reversals", True))
    suspect = (t["date_suspect"] == "1").sum()
    out.append(("T12 at most 30 rows flagged as misprinted dates in the source",
                suspect <= 30, f"{suspect} rows", False))

    cmd = t[(t["kind"] == "commander") & (t["date_from"] != "") & (t["date_to"] != "")]
    wrong = cmd[cmd["date_to"] < cmd["date_from"]]
    out.append(("T8 a tenure ends no earlier than it begins", wrong.empty,
                f"{len(wrong)} rows", True))

    strength = t[t["kind"] == "strength"]["value"]
    out.append(("T9 transport strength is a positive number",
                strength.str.fullmatch(r"[1-9]\d*").all(), "non-numeric strength", True))

    long_place = t[(t["kind"] == "location") & (t["value"].str.len() > 30)]
    out.append(("T10 no place longer than 30 characters (an unparsed clause)",
                len(long_place) <= 5, f"{len(long_place)} rows: "
                f"{long_place['value'].head(3).tolist()}", False))
    unknown = (t["date_from_precision"] == "unknown").sum()
    out.append(("T11 fewer than 1% of rows without a date", unknown <= 0.01 * len(t),
                f"{unknown} of {len(t)}", False))
    return out

def main() -> int:

        mark = "ok  " if ok else ("FAIL" if hard else "warn")

"""
check_unit_timeline.py — invariants on reference/units/unit_timeline.csv
========================================================================
``unit_timeline.csv`` is parsed by ``build_unit_timeline.py`` from the formulas of the
Fidler (2025) encyclopedia. A parser of prose fails quietly: a clause it does not
recognise is dropped, or lands in the wrong column, and the table still looks full.
These checks make the failures that can be seen mechanical.

    python utils/reference_tools/units/check_unit_timeline.py

Exit code 0 if every hard invariant holds, 1 otherwise.

  source; the chronicles in ``library/diaries/`` are the place to do that.

from __future__ import annotations

import re
import sys
from pathlib import Path

import pandas as pd

__version__ = "1.0.0"

HERE = Path(__file__).resolve().parents[3] / "reference" / "units"   # the data, not this script
CSV = HERE / "unit_timeline.csv"
AUTHORITY = HERE / "unit_authority.csv"

KINDS = {"location": {"place", "moving", "field", "moving_home"},
         "lifecycle": {"formed", "renamed", "merged", "disbanded", "sailed", "arrived_home"},
         "parent": {"", "professional", "general"},
         "commander": None, "composition": {""}, "strength": {""}}
PRECISION = {"day": r"\d{4}-\d{2}-\d{2}", "month": r"\d{4}-\d{2}",
             "formation": r"\d{4}-\d{2}-\d{2}", "unknown": r""}
FIRST, LAST = "1914-01", "1921-12"   # the Legions' lifetime, generously


def load() -> pd.DataFrame:
    return pd.read_csv(CSV, dtype=str, encoding="utf-8").fillna("")

def main() -> int:
    t = load()
    failed = 0
    for label, ok, detail, hard in checks(t):
        mark = "ok  " if ok else ("FAIL" if hard else "warn")
        print(f"[{mark}] {label}" + ("" if ok else f" — {detail}"))
        failed += (not ok) and hard
    print(f"\n{len(t):,} rows, {t['unit_cs'].nunique()} units; "
          f"{failed} hard failure(s)")
