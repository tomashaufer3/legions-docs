How it works

(Not to be confused with utils.parsing.theater, which is the war theatre — Russia / Italy / France.)

What "before 1914" means when a year is fuzzy

An ensemble's start is printed as a year, a decade (191x) or a century (19xx) and is carried with its precision. starts_before is the single rule:

year      1913            -> yes        1914            -> no
decade    190x (≤1909)    -> yes        191x (1910–19)  -> ambiguous     192x -> no
century   18xx (≤1899)    -> yes        19xx            -> ambiguous

"Before 1914" in every table below means yes — the ambiguous are counted and shown beside it, never folded in.

The join

A database place (present-day obec / místní část, with coordinates) is tied to a row of settlements_1910 (the ČSÚ lexikon crosswalk; coordinates for about half the obce and few hamlets) in three tiers, most trusted first:

name+coords     same normalised name, both geocoded, ≤ `max_km` apart
name+okres      same normalised name, the settlement not geocoded, and the place's
                present-day okres on the settlement's okres timeline (`okres_all`)
                — accepted only where that leaves exactly one candidate on each side
fuzzy+coords    names ≥ `min_similarity` alike, ≤ `fuzzy_km` apart

The match is one-to-one on both sides and the result has exactly one row per place — both asserted. settlements_1910 has no declared key (a name can be an obec and an osada in the same okres), so its row position is used as settlement_row.

Town size

There is no settlement-level 1910 population in the repo (the ČSÚ lexikon volume that carries it is a PDF whose digits do not extract), so size is proxied by three classes read off what the crosswalk does carry: the seat of a 1910 judicial district, any other commune, and a dependent hamlet or part. If the settlements frame is given a population_1910 column, coverage_tables bands on that instead.

def haversine_km(lat1, lon1, lat2, lon2):
    """Great-circle distance in km (vectorised; broadcasts)."""
    lat1, lon1, lat2, lon2 = (np.radians(np.asarray(v, dtype=float)) for v in (lat1, lon1, lat2, lon2))
    a = np.sin((lat2 - lat1) / 2) ** 2 + np.cos(lat1) * np.cos(lat2) * np.sin((lon2 - lon1) / 2) ** 2
    return 2 * EARTH_RADIUS_KM * np.arcsin(np.sqrt(a))

def settlement_frame(settlements: pd.DataFrame, districts: pd.DataFrame | None = None) -> pd.DataFrame:
    """``settlements_1910`` with the columns the join and the tabulation need.

    Adds ``settlement_row`` (row position — the frame's only usable key), ``name_key``,
    ``name_key_alt`` (:func:`city_part_name`), ``okres_tokens``, ``okres_now`` (the last
    okres on the timeline), ``in_1910`` (the
    settlement has a resolved 1910 okres and status), ``okres_1910_cs`` (the district's
    name from ``districts_1910``, free of the register's OCR breaks) and ``size_class``.
    """
    out = settlements.reset_index(drop=True).copy()
    out.insert(0, "settlement_row", np.arange(len(out)))
    out["name_key"] = out["cs_name"].map(normalise_name)
    out["name_key_alt"] = out["name_key"].map(city_part_name)
    out["okres_tokens"] = out["okres_all"].map(okres_tokens)
    out["okres_now"] = out["okres_tokens"].map(lambda t: t[-1] if t else None)
    out["in_1910"] = out["okres_id"].notna() & out["status_1910"].notna()

    seats: set[tuple[int, str]] = set()
    out["okres_1910_cs"] = out["okres_1910"]
    if districts is not None:
        codes = districts["okres_code"].str.split("-", expand=True).astype(int)
        okres_id = codes[0] * 1000 + codes[1]
        names = dict(zip(okres_id, districts["okres_cs"]))
        out["okres_1910_cs"] = out["okres_id"].map(names).fillna(out["okres_1910"])
        seats = set(zip(okres_id, districts["gerichtsbezirk_cs"].map(normalise_name)))

    is_seat = pd.Series([(o, n) in seats for o, n in zip(out["okres_id"], out["name_key"])],
                        index=out.index)
    commune = out["status_1910"].eq("obec")
    out["size_class"] = pd.Series("3 hamlet / part", index=out.index, dtype=object)
    out.loc[commune, "size_class"] = "2 other commune"
    out.loc[commune & is_seat, "size_class"] = "1 court-seat town"
    out.loc[~out["in_1910"], "size_class"] = None
    if "population_1910" in out.columns:
        out["size_class"] = pd.cut(out["population_1910"], POPULATION_BANDS,
                                   labels=POPULATION_LABELS, right=False).astype(object)
    return out

def join_places_to_settlements(
    places: pd.DataFrame,
    settlements: pd.DataFrame,
    *,
    max_km: float = 5.0,
    fuzzy_km: float = 2.0,
    min_similarity: float = 85.0,
) -> pd.DataFrame:
    """Tie each database place to at most one 1910 settlement, by distance with a name check.

    Args:
        places: the scraper's places table (``geo_id, name, type, lat, lon, okres``).
        settlements: :func:`settlement_frame` output.
        max_km: furthest an exact-name pair may be apart.
        fuzzy_km: furthest a near-name pair may be apart.
        min_similarity: rapidfuzz ratio (0–100) on the normalised, de-accented names.

    Returns:
        One row per place — ``geo_id, settlement_row, match_method, distance_km,
        name_similarity`` — with ``settlement_row`` missing where nothing matched and
        ``match_method`` saying why (``unmatched: no such name`` / ``… too far`` /
        ``… ambiguous``).  Asserted: one row per place, no settlement used twice.
    """
    P = places[["geo_id", "name", "type", "lat", "lon", "okres"]].copy()
    P["name_key"] = P["name"].map(normalise_name)
    S = settlements[["settlement_row", "name_key", "status_1910", "lat", "lon", "okres_tokens"]]
    # a city part is reachable under its own name as well as under "CITY IV-PART"
    alt = settlements.loc[settlements["name_key_alt"].notna(), S.columns.drop("name_key")].assign(
        name_key=settlements["name_key_alt"].dropna())
    by_name = pd.concat([S, alt[S.columns]], ignore_index=True)

    pairs = P.merge(by_name, on="name_key", suffixes=("", "_s"))
    pairs = pairs[pairs["name_key"] != ""]
    geocoded = pairs["lat_s"].notna()
    pairs["distance_km"] = np.where(
        geocoded, haversine_km(pairs["lat"], pairs["lon"], pairs["lat_s"].fillna(0),
                               pairs["lon_s"].fillna(0)), np.nan)
    pairs["okres_ok"] = [o in toks for o, toks in zip(pairs["okres"], pairs["okres_tokens"])]
    pairs["penalty"] = [_type_penalty(t, s) for t, s in zip(pairs["type"], pairs["status_1910"])]

    used_places: dict[int, tuple] = {}
    used_settlements: set[int] = set()

    def accept(geo_id, row, method, km, similarity) -> None:
        used_places[int(geo_id)] = (int(row), method, km, similarity)
        used_settlements.add(int(row))

    # tier 1 — exact name, both geocoded, close: nearest pair first
    near = pairs[geocoded & (pairs["distance_km"] <= max_km)].sort_values(["distance_km", "penalty"])
    for r in near.itertuples():
        if r.geo_id not in used_places and r.settlement_row not in used_settlements:
            accept(r.geo_id, r.settlement_row, "name+coords", round(float(r.distance_km), 2), 100.0)

    # tier 2 — exact name, settlement not geocoded, okres on its timeline, unique both ways
    loose = pairs[~geocoded & pairs["okres_ok"]
                  & ~pairs["geo_id"].isin(list(used_places))
                  & ~pairs["settlement_row"].isin(used_settlements)]
    loose = loose[loose["penalty"] == loose.groupby("geo_id")["penalty"].transform("min")]
    n_for_place = loose.groupby("geo_id")["settlement_row"].transform("nunique")
    n_for_settlement = loose.groupby("settlement_row")["geo_id"].transform("nunique")
    ambiguous = set(loose.loc[(n_for_place > 1) | (n_for_settlement > 1), "geo_id"])
    for r in loose[(n_for_place == 1) & (n_for_settlement == 1)].itertuples():
        accept(r.geo_id, r.settlement_row, "name+okres", np.nan, 100.0)

    # tier 3 — near-identical name, very close
    left = P[~P["geo_id"].isin(list(used_places)) & P["lat"].notna() & (P["name_key"] != "")]
    right = S[~S["settlement_row"].isin(used_settlements) & S["lat"].notna()]
    fuzzy_rows = []
    if len(left) and len(right):
        right_lat, right_lon = right["lat"].to_numpy(), right["lon"].to_numpy()
        right_keys = right["name_key"].map(strip_diacritics).to_numpy()
        right_ids = right["settlement_row"].to_numpy()
        for p in left.itertuples():
            km = haversine_km(p.lat, p.lon, right_lat, right_lon)
            key = strip_diacritics(p.name_key)
            for j in np.flatnonzero(km <= fuzzy_km):
                similarity = fuzz.ratio(key, right_keys[j])
                if similarity >= min_similarity:
                    fuzzy_rows.append((float(km[j]), -similarity, p.geo_id, right_ids[j]))
    for km, neg_similarity, geo_id, row in sorted(fuzzy_rows):
        if geo_id not in used_places and row not in used_settlements:
            accept(geo_id, row, "fuzzy+coords", round(km, 2), float(-neg_similarity))

    # why the rest did not match
    has_name = set(pairs["geo_id"])
    too_far = set(pairs.loc[geocoded & (pairs["distance_km"] > max_km), "geo_id"])
    rows = []
    for geo_id in P["geo_id"]:
        if int(geo_id) in used_places:
            rows.append((geo_id, *used_places[int(geo_id)]))
            continue
        reason = ("unmatched: ambiguous" if geo_id in ambiguous
                  else "unmatched: too far" if geo_id in too_far
                  else "unmatched: name elsewhere" if geo_id in has_name
                  else "unmatched: no such name")
        rows.append((geo_id, pd.NA, reason, np.nan, np.nan))
    out = pd.DataFrame(rows, columns=["geo_id", "settlement_row", "match_method",
                                      "distance_km", "name_similarity"])
    out["settlement_row"] = out["settlement_row"].astype("Int64")

    assert len(out) == len(places) and out["geo_id"].is_unique, "join: not one row per place"
    assert not out["settlement_row"].dropna().duplicated().any(), "join: a settlement matched twice"
    return out

def starts_before(year, precision, cutoff: int = PRE_WAR_CUTOFF) -> str:
    """Does a (possibly fuzzy) year fall before ``cutoff``?  ``yes / no / ambiguous / unknown``."""
    if precision not in _SPAN_OF or pd.isna(year):
        return "unknown"
    earliest, latest = int(year), int(year) + _SPAN_OF[precision]
    if latest < cutoff:
        return "yes"
    if earliest >= cutoff:
        return "no"
    return "ambiguous"

def place_dates(places: pd.DataFrame, ensembles: pd.DataFrame,
                cutoff: int = PRE_WAR_CUTOFF) -> pd.DataFrame:
    """Per place: how many ensembles, and whether theatre is recorded before ``cutoff``.

    ``ensemble_pre`` is True when at least one ensemble certainly starts before the
    cutoff; ``ensemble_ambiguous`` when none certainly does but one may (``191x``);
    ``first_record_pre`` reads the place page's own 'První zpráva'.  Key ``geo_id``.
    """
    E = ensembles.copy()
    E["pre"] = [starts_before(y, p, cutoff) for y, p in zip(E["year_from"], E["year_from_precision"])]
    per_place = E.groupby("geo_id").agg(
        n_ensembles=("ensemble_id", "size"),
        n_pre=("pre", lambda s: int((s == "yes").sum())),
        n_ambiguous=("pre", lambda s: int((s == "ambiguous").sum())),
        n_undated=("pre", lambda s: int((s == "unknown").sum())),
    )
    out = places[["geo_id"]].merge(per_place, on="geo_id", how="left", validate="1:1")
    counts = ["n_ensembles", "n_pre", "n_ambiguous", "n_undated"]
    out[counts] = out[counts].fillna(0).astype(int)
    out["ensemble_pre"] = out["n_pre"] > 0
    out["ensemble_ambiguous"] = (out["n_pre"] == 0) & (out["n_ambiguous"] > 0)
    first = [starts_before(y, p, cutoff) for y, p in
             zip(places["first_record_year"], places["first_record_precision"])]
    out["first_record_pre"] = [f == "yes" for f in first]
    out["first_record_known"] = [f != "unknown" for f in first]
    assert len(out) == len(places)
    return out

def association_counts(ensembles: pd.DataFrame, cutoff: int = PRE_WAR_CUTOFF) -> tuple[pd.DataFrame, pd.Series]:
    """Counts of pre-``cutoff`` ensembles by association, and the unclassified names.

    Returns:
        ``(counts, unclassified)`` — counts has one row per label plus *unclassified*,
        *more than one label* and *all pre-1914 ensembles*; ``unclassified`` is the
        frequency table of the names nothing matched, so the remainder can be read.
    """
    E = ensembles.drop_duplicates("ensemble_id").copy()
    E = E[[starts_before(y, p, cutoff) == "yes" for y, p in zip(E["year_from"], E["year_from_precision"])]]
    labels = E["name"].map(classify_association)
    rows = [(label, int(labels.map(lambda found, lab=label: lab in found).sum()))
            for label in ASSOCIATION_PATTERNS]
    rows += [("unclassified", int((labels.str.len() == 0).sum())),
             ("more than one label", int((labels.str.len() > 1).sum())),
             (f"all ensembles starting before {cutoff}", len(E))]
    counts = pd.DataFrame(rows, columns=["association", "ensembles"])
    counts["share"] = (counts["ensembles"] / max(len(E), 1)).round(3)
    return counts, E.loc[labels.str.len() == 0, "name"].value_counts()

    """From step A alone: which 1910 communes have *any* entry in the database.

    Presence only — the map XML has no dates.  The denominator is the 1910 communes
    (``status_1910 == "obec"``) of ``settlements_1910``; each is assigned to a present-day
    kraj through the latest okres on its timeline that the database knows.
    """
    S = settlement_frame_
    join = join_places_to_settlements(places, S)
    matched = join.dropna(subset=["settlement_row"])
    okres_kraj = (places.dropna(subset=["okres", "kraj"]).groupby("okres")["kraj"]
                  .agg(lambda s: s.mode().iloc[0]).to_dict())
    # the database files Prague under its numbered districts; the lexikon under "Praha"
    prague = next((k for k in okres_kraj.values() if k.startswith("Praha")), None)
    if prague:
        okres_kraj.setdefault("Praha", prague)

    def kraj_of(tokens: list[str]):
        for okres in reversed(tokens):
            if okres in okres_kraj:
                return okres_kraj[okres]
        return None

    communes = S[S["in_1910"] & S["status_1910"].eq("obec")].copy()
    communes["kraj"] = communes["okres_tokens"].map(kraj_of)
    communes["in_database"] = communes["settlement_row"].isin(matched["settlement_row"])

    def share(by: str) -> pd.DataFrame:
        table = communes.groupby(by, dropna=False).agg(
            communes_1910=("settlement_row", "size"),
            in_database=("in_database", "sum")).reset_index()
        table["share"] = (table["in_database"] / table["communes_1910"]).round(3)
        return table

    by_kraj = share("kraj").merge(
        places.groupby("kraj").agg(
            database_places=("geo_id", "size"),
            database_obce=("type", lambda s: int((s == "Obec").sum()))).reset_index(),
        on="kraj", how="outer", validate="1:1")
    return {
        "country_join": (join["match_method"].value_counts().rename("places").reset_index()),
        "country_communes_by_kraj": by_kraj.sort_values("share", ascending=False),
        "country_communes_by_land": share("land"),
        "country_places_by_type": places.groupby("type").agg(
            places=("geo_id", "size"),
            matched=("geo_id", lambda s: int(s.isin(matched["geo_id"]).sum()))).reset_index(),
    }

amateur_theatre.py — is the amateur-theatre database usable as a pre-1914 covariate?
====================================================================================
**Feasibility pilot.**  Nothing here feeds ``data/final``.  The module takes the two
tables written by :mod:`utils.scraping.scraper_amaterskedivadlo` (places, ensembles) and
asks the one question that decides whether the source is worth building on: *how evenly
is it documented* — across town sizes, across Bohemia and Moravia, across Czech- and
German-speaking districts — before any of it is read as a measure of associational life.

(Not to be confused with :mod:`utils.parsing.theater`, which is the *war* theatre —
Russia / Italy / France.)

What "before 1914" means when a year is fuzzy
---------------------------------------------
An ensemble's start is printed as a year, a decade (``191x``) or a century (``19xx``) and
is carried with its precision.  :func:`starts_before` is the single rule:

    year      1913            -> yes        1914            -> no
    decade    190x (≤1909)    -> yes        191x (1910–19)  -> ambiguous     192x -> no
    century   18xx (≤1899)    -> yes        19xx            -> ambiguous

"Before 1914" in every table below means **yes** — the ambiguous are counted and shown
beside it, never folded in.

The join
--------
A database place (present-day obec / místní část, with coordinates) is tied to a row of
``settlements_1910`` (the ČSÚ lexikon crosswalk; coordinates for about half the obce and
few hamlets) in three tiers, most trusted first:

    name+coords     same normalised name, both geocoded, ≤ ``max_km`` apart
    name+okres      same normalised name, the settlement not geocoded, and the place's
                    present-day okres on the settlement's okres timeline (``okres_all``)
                    — accepted only where that leaves exactly one candidate on each side
    fuzzy+coords    names ≥ ``min_similarity`` alike, ≤ ``fuzzy_km`` apart

The match is one-to-one on both sides and the result has exactly one row per place —
both asserted.  ``settlements_1910`` has no declared key (a name can be an *obec* and an
*osada* in the same okres), so its row position is used as ``settlement_row``.

Town size
---------
There is **no settlement-level 1910 population in the repo** (the ČSÚ lexikon volume that
carries it is a PDF whose digits do not extract), so size is proxied by three classes read
off what the crosswalk does carry: the seat of a 1910 judicial district, any other
commune, and a dependent hamlet or part.  If the settlements frame is given a
``population_1910`` column, :func:`coverage_tables` bands on that instead.

Usage
-----
    from utils.parsing import amateur_theatre as at
    tables = at.coverage_tables(places, ensembles, settlements, districts,
                                okresy=["Kutná Hora", "Trutnov", "Přerov"])
    at.coverage_report(tables)
"""

# 1910 okres by its Czech-speaking share (census Umgangssprache)
LANGUAGE_BANDS = [0, 0.5, 0.9, 1.01]
LANGUAGE_LABELS = ["German-majority (<50% Czech)", "mixed (50–90% Czech)", "Czech (90%+)"]
MIN_OKRES_SETTLEMENTS = 20           # smaller 1910 okresy are left out of the spread table


# ---------------------------------------------------------------------------
# Names and distance
# ---------------------------------------------------------------------------
_PAREN = re.compile(r"\s*\([^)]*\)")
_VIZ = re.compile(r"\s+viz\s+.*$", re.I)
_FORMERLY = re.compile(r",\s*do\s+\d{4}\b.*$", re.I)
_NON_WORD = re.compile(r"[^\w]+", re.UNICODE)
_CITY_PART = re.compile(r"^.+\s[ivxl]+\s(.+)$")


def normalise_name(name) -> str:
    """A place name reduced to what should be equal across the two sources.

    Case-folded; parentheticals, ``X viz Y`` cross-reference tails and the database's
    ``, do 1925 Bračice`` former-name notes dropped; punctuation and hyphens turned into
    single spaces.  **Diacritics are kept** — in Czech they separate different villages
    (*Bělá* / *Bílá*).
    """
    if name is None or (isinstance(name, float) and np.isnan(name)):
        return ""
    text = _FORMERLY.sub("", _VIZ.sub("", _PAREN.sub("", str(name))))
    return _NON_WORD.sub(" ", text).strip().casefold()


def city_part_name(name_key: str) -> str | None:
    """The part's own name where the lexikon files it under its city.

    The register heads a village since absorbed by a town as ``HRANICE IV-DRAHOTUŠE`` or
    ``PŘEROV VII-ČEKYNĚ``; the database calls it *Drahotuše*, *Čekyně*.  Returns the part
    after the roman numeral, or None for an ordinary name.
    """
    m = _CITY_PART.match(name_key or "")
    return m.group(1) if m else None


def strip_diacritics(text: str) -> str:
    return "".join(c for c in unicodedata.normalize("NFD", text) if not unicodedata.combining(c))


def haversine_km(lat1, lon1, lat2, lon2):
    """Great-circle distance in km (vectorised; broadcasts)."""
    lat1, lon1, lat2, lon2 = (np.radians(np.asarray(v, dtype=float)) for v in (lat1, lon1, lat2, lon2))
    a = np.sin((lat2 - lat1) / 2) ** 2 + np.cos(lat1) * np.cos(lat2) * np.sin((lon2 - lon1) / 2) ** 2
    return 2 * EARTH_RADIUS_KM * np.arcsin(np.sqrt(a))

def okres_tokens(okres_all) -> list[str]:
    """The okres timeline of a settlement as a list (stray brackets and spaces removed)."""
    if not isinstance(okres_all, str):
        return []
    return [t for t in (re.sub(r"\s+", " ", tok.strip(" ()")) for tok in okres_all.split("|")) if t]


def settlement_frame(settlements: pd.DataFrame, districts: pd.DataFrame | None = None) -> pd.DataFrame:
    """``settlements_1910`` with the columns the join and the tabulation need.

    Adds ``settlement_row`` (row position — the frame's only usable key), ``name_key``,
    ``name_key_alt`` (:func:`city_part_name`), ``okres_tokens``, ``okres_now`` (the last
    okres on the timeline), ``in_1910`` (the
    settlement has a resolved 1910 okres and status), ``okres_1910_cs`` (the district's
    name from ``districts_1910``, free of the register's OCR breaks) and ``size_class``.
    """
    out = settlements.reset_index(drop=True).copy()
    out.insert(0, "settlement_row", np.arange(len(out)))
    out["name_key"] = out["cs_name"].map(normalise_name)
    out["name_key_alt"] = out["name_key"].map(city_part_name)
    out["okres_tokens"] = out["okres_all"].map(okres_tokens)
    out["okres_now"] = out["okres_tokens"].map(lambda t: t[-1] if t else None)
    out["in_1910"] = out["okres_id"].notna() & out["status_1910"].notna()

    seats: set[tuple[int, str]] = set()
    out["okres_1910_cs"] = out["okres_1910"]
    if districts is not None:
        codes = districts["okres_code"].str.split("-", expand=True).astype(int)
        okres_id = codes[0] * 1000 + codes[1]
        names = dict(zip(okres_id, districts["okres_cs"]))
        out["okres_1910_cs"] = out["okres_id"].map(names).fillna(out["okres_1910"])
        seats = set(zip(okres_id, districts["gerichtsbezirk_cs"].map(normalise_name)))

    is_seat = pd.Series([(o, n) in seats for o, n in zip(out["okres_id"], out["name_key"])],
                        index=out.index)
    commune = out["status_1910"].eq("obec")
    out["size_class"] = pd.Series("3 hamlet / part", index=out.index, dtype=object)
    out.loc[commune, "size_class"] = "2 other commune"
    out.loc[commune & is_seat, "size_class"] = "1 court-seat town"
    out.loc[~out["in_1910"], "size_class"] = None
    if "population_1910" in out.columns:
        out["size_class"] = pd.cut(out["population_1910"], POPULATION_BANDS,
                                   labels=POPULATION_LABELS, right=False).astype(object)
    return out

_TYPE_PREFERENCE = {"Obec": {"obec": 0}, "Místní část": {"osada": 0, "cast": 0},
                    "Část": {"osada": 0, "cast": 0}}


def _type_penalty(place_type, status) -> int:
    """0 where the place's type and the settlement's 1910 status agree, else 1."""
    return _TYPE_PREFERENCE.get(place_type, {}).get(status, 1)


def join_places_to_settlements(
    places: pd.DataFrame,
    settlements: pd.DataFrame,
    *,
    max_km: float = 5.0,
    fuzzy_km: float = 2.0,
    min_similarity: float = 85.0,
) -> pd.DataFrame:
    """Tie each database place to at most one 1910 settlement, by distance with a name check.

    Args:
        places: the scraper's places table (``geo_id, name, type, lat, lon, okres``).
        settlements: :func:`settlement_frame` output.
        max_km: furthest an exact-name pair may be apart.
        fuzzy_km: furthest a near-name pair may be apart.
        min_similarity: rapidfuzz ratio (0–100) on the normalised, de-accented names.

    Returns:
        One row per place — ``geo_id, settlement_row, match_method, distance_km,
        name_similarity`` — with ``settlement_row`` missing where nothing matched and
        ``match_method`` saying why (``unmatched: no such name`` / ``… too far`` /
        ``… ambiguous``).  Asserted: one row per place, no settlement used twice.
    """
    P = places[["geo_id", "name", "type", "lat", "lon", "okres"]].copy()
    P["name_key"] = P["name"].map(normalise_name)
    S = settlements[["settlement_row", "name_key", "status_1910", "lat", "lon", "okres_tokens"]]
    # a city part is reachable under its own name as well as under "CITY IV-PART"
    alt = settlements.loc[settlements["name_key_alt"].notna(), S.columns.drop("name_key")].assign(
        name_key=settlements["name_key_alt"].dropna())
    by_name = pd.concat([S, alt[S.columns]], ignore_index=True)

    pairs = P.merge(by_name, on="name_key", suffixes=("", "_s"))
    pairs = pairs[pairs["name_key"] != ""]
    geocoded = pairs["lat_s"].notna()
    pairs["distance_km"] = np.where(
        geocoded, haversine_km(pairs["lat"], pairs["lon"], pairs["lat_s"].fillna(0),
                               pairs["lon_s"].fillna(0)), np.nan)
    pairs["okres_ok"] = [o in toks for o, toks in zip(pairs["okres"], pairs["okres_tokens"])]
    pairs["penalty"] = [_type_penalty(t, s) for t, s in zip(pairs["type"], pairs["status_1910"])]

    used_places: dict[int, tuple] = {}
    used_settlements: set[int] = set()

    def accept(geo_id, row, method, km, similarity) -> None:
        used_places[int(geo_id)] = (int(row), method, km, similarity)
        used_settlements.add(int(row))

    # tier 1 — exact name, both geocoded, close: nearest pair first
    near = pairs[geocoded & (pairs["distance_km"] <= max_km)].sort_values(["distance_km", "penalty"])
    for r in near.itertuples():
        if r.geo_id not in used_places and r.settlement_row not in used_settlements:
            accept(r.geo_id, r.settlement_row, "name+coords", round(float(r.distance_km), 2), 100.0)

    # tier 2 — exact name, settlement not geocoded, okres on its timeline, unique both ways
    loose = pairs[~geocoded & pairs["okres_ok"]
                  & ~pairs["geo_id"].isin(list(used_places))
                  & ~pairs["settlement_row"].isin(used_settlements)]
    loose = loose[loose["penalty"] == loose.groupby("geo_id")["penalty"].transform("min")]
    n_for_place = loose.groupby("geo_id")["settlement_row"].transform("nunique")
    n_for_settlement = loose.groupby("settlement_row")["geo_id"].transform("nunique")
    ambiguous = set(loose.loc[(n_for_place > 1) | (n_for_settlement > 1), "geo_id"])
    for r in loose[(n_for_place == 1) & (n_for_settlement == 1)].itertuples():
        accept(r.geo_id, r.settlement_row, "name+okres", np.nan, 100.0)

    # tier 3 — near-identical name, very close
    left = P[~P["geo_id"].isin(list(used_places)) & P["lat"].notna() & (P["name_key"] != "")]
    right = S[~S["settlement_row"].isin(used_settlements) & S["lat"].notna()]
    fuzzy_rows = []
    if len(left) and len(right):
        right_lat, right_lon = right["lat"].to_numpy(), right["lon"].to_numpy()
        right_keys = right["name_key"].map(strip_diacritics).to_numpy()
        right_ids = right["settlement_row"].to_numpy()
        for p in left.itertuples():
            km = haversine_km(p.lat, p.lon, right_lat, right_lon)
            key = strip_diacritics(p.name_key)
            for j in np.flatnonzero(km <= fuzzy_km):
                similarity = fuzz.ratio(key, right_keys[j])
                if similarity >= min_similarity:
                    fuzzy_rows.append((float(km[j]), -similarity, p.geo_id, right_ids[j]))
    for km, neg_similarity, geo_id, row in sorted(fuzzy_rows):
        if geo_id not in used_places and row not in used_settlements:
            accept(geo_id, row, "fuzzy+coords", round(km, 2), float(-neg_similarity))

    # why the rest did not match
    has_name = set(pairs["geo_id"])
    too_far = set(pairs.loc[geocoded & (pairs["distance_km"] > max_km), "geo_id"])
    rows = []
    for geo_id in P["geo_id"]:
        if int(geo_id) in used_places:
            rows.append((geo_id, *used_places[int(geo_id)]))
            continue
        reason = ("unmatched: ambiguous" if geo_id in ambiguous
                  else "unmatched: too far" if geo_id in too_far
                  else "unmatched: name elsewhere" if geo_id in has_name
                  else "unmatched: no such name")
        rows.append((geo_id, pd.NA, reason, np.nan, np.nan))
    out = pd.DataFrame(rows, columns=["geo_id", "settlement_row", "match_method",
                                      "distance_km", "name_similarity"])
    out["settlement_row"] = out["settlement_row"].astype("Int64")

    assert len(out) == len(places) and out["geo_id"].is_unique, "join: not one row per place"
    assert not out["settlement_row"].dropna().duplicated().any(), "join: a settlement matched twice"
    return out

_SPAN_OF = {"year": 0, "decade": 9, "century": 99}


def starts_before(year, precision, cutoff: int = PRE_WAR_CUTOFF) -> str:
    """Does a (possibly fuzzy) year fall before ``cutoff``?  ``yes / no / ambiguous / unknown``."""
    if precision not in _SPAN_OF or pd.isna(year):
        return "unknown"
    earliest, latest = int(year), int(year) + _SPAN_OF[precision]
    if latest < cutoff:
        return "yes"
    if earliest >= cutoff:
        return "no"
    return "ambiguous"


_PERIOD_EDGES = [0, 1800, 1850, 1870, 1880, 1890, 1900, 1910, PRE_WAR_CUTOFF, 1920, 1950, 9999]
_PERIOD_LABELS = ["<1800", "1800–49", "1850–69", "1870s", "1880s", "1890s", "1900s",
                  "1910–13", "1914–19", "1920–49", "1950+"]
PERIOD_ORDER = _PERIOD_LABELS + ["191x (decade only)", "century only", "not recorded"]


def first_record_period(year, precision) -> str:
    """The period a place's first report falls in, for the distribution table.

    An exact year goes to its bin.  A decade does too, except ``191x`` — the one decade
    the 1914 line cuts — which is shown as its own row rather than assigned to a side.
    """
    if pd.isna(year) or precision not in _SPAN_OF:
        return "not recorded"
    if precision == "century":
        return "century only"
    if precision == "decade" and int(year) == 1910:
        return "191x (decade only)"
    return _PERIOD_LABELS[int(np.searchsorted(_PERIOD_EDGES, int(year), side="right")) - 1]


def place_dates(places: pd.DataFrame, ensembles: pd.DataFrame,
                cutoff: int = PRE_WAR_CUTOFF) -> pd.DataFrame:
    """Per place: how many ensembles, and whether theatre is recorded before ``cutoff``.

    ``ensemble_pre`` is True when at least one ensemble certainly starts before the
    cutoff; ``ensemble_ambiguous`` when none certainly does but one may (``191x``);
    ``first_record_pre`` reads the place page's own 'První zpráva'.  Key ``geo_id``.
    """
    E = ensembles.copy()
    E["pre"] = [starts_before(y, p, cutoff) for y, p in zip(E["year_from"], E["year_from_precision"])]
    per_place = E.groupby("geo_id").agg(
        n_ensembles=("ensemble_id", "size"),
        n_pre=("pre", lambda s: int((s == "yes").sum())),
        n_ambiguous=("pre", lambda s: int((s == "ambiguous").sum())),
        n_undated=("pre", lambda s: int((s == "unknown").sum())),
    )
    out = places[["geo_id"]].merge(per_place, on="geo_id", how="left", validate="1:1")
    counts = ["n_ensembles", "n_pre", "n_ambiguous", "n_undated"]
    out[counts] = out[counts].fillna(0).astype(int)
    out["ensemble_pre"] = out["n_pre"] > 0
    out["ensemble_ambiguous"] = (out["n_pre"] == 0) & (out["n_ambiguous"] > 0)
    first = [starts_before(y, p, cutoff) for y, p in
             zip(places["first_record_year"], places["first_record_precision"])]
    out["first_record_pre"] = [f == "yes" for f in first]
    out["first_record_known"] = [f != "unknown" for f in first]
    assert len(out) == len(places)
    return out


# ---------------------------------------------------------------------------
# What kind of association an ensemble's name points to
# ---------------------------------------------------------------------------
# Word-anchored on both sides, with the inflected forms listed: `sokol` must not fire
# inside the town of Sokolov, nor `orel` inside a longer word.  An *Osvětová beseda* is the
# post-1945 state cultural club, not the nineteenth-century reading society, and is left out.
ASSOCIATION_PATTERNS: dict[str, re.Pattern] = {
    "Sokol": re.compile(
        r"\bsokol(?:a|u|e|em|ů|ové|sk[áéýou]\w*|stv\w*|ík\w*)?\b", re.I),
    "Hasiči": re.compile(
        r"\bhasič\w*|\bSDH\b", re.I),
    "Beseda": re.compile(
        r"(?<!osvětová\s)(?<!osvětové\s)(?<!osvětovou\s)\bbesed[aáyěou]\b", re.I),
    "Orel": re.compile(
        r"\borel\b|\borl[aue]\b|\borlem\b|\borelsk\w+", re.I),
    "DTJ": re.compile(
        r"\bDTJ\b|\bD\.\s?T\.\s?J\b|\bdělnick\w+\s+tělocvičn\w+\s+jednot\w+", re.I),
}


def classify_association(name) -> list[str]:
    """Labels of :data:`ASSOCIATION_PATTERNS` an ensemble name matches (possibly none)."""
    if not isinstance(name, str):
        return []
    return [label for label, pattern in ASSOCIATION_PATTERNS.items() if pattern.search(name)]


def association_counts(ensembles: pd.DataFrame, cutoff: int = PRE_WAR_CUTOFF) -> tuple[pd.DataFrame, pd.Series]:
    """Counts of pre-``cutoff`` ensembles by association, and the unclassified names.

    Returns:
        ``(counts, unclassified)`` — counts has one row per label plus *unclassified*,
        *more than one label* and *all pre-1914 ensembles*; ``unclassified`` is the
        frequency table of the names nothing matched, so the remainder can be read.
    """
    E = ensembles.drop_duplicates("ensemble_id").copy()
    E = E[[starts_before(y, p, cutoff) == "yes" for y, p in zip(E["year_from"], E["year_from_precision"])]]
    labels = E["name"].map(classify_association)
    rows = [(label, int(labels.map(lambda found, lab=label: lab in found).sum()))
            for label in ASSOCIATION_PATTERNS]
    rows += [("unclassified", int((labels.str.len() == 0).sum())),
             ("more than one label", int((labels.str.len() > 1).sum())),
             (f"all ensembles starting before {cutoff}", len(E))]
    counts = pd.DataFrame(rows, columns=["association", "ensembles"])
    counts["share"] = (counts["ensembles"] / max(len(E), 1)).round(3)
    return counts, E.loc[labels.str.len() == 0, "name"].value_counts()

def _share_table(frame: pd.DataFrame, by: list[str]) -> pd.DataFrame:
    table = frame.groupby(by, dropna=False).agg(
        settlements=("settlement_row", "size"),
        in_database=("geo_id", lambda s: int(s.notna().sum())),
        ensemble_pre1914=("ensemble_pre", "sum"),
        ambiguous_191x=("ensemble_ambiguous", "sum"),
        first_record_pre1914=("first_record_pre", "sum"),
    ).reset_index()
    for col in ("in_database", "ensemble_pre1914", "first_record_pre1914"):
        table[f"share_{col}"] = (table[col] / table["settlements"]).round(3)
    return table


def _documentation_table(paged: pd.DataFrame, by: str) -> pd.DataFrame:
    """How much the database records per place, grouped by ``by``."""
    return paged.groupby(by, dropna=False).agg(
        places=("geo_id", "size"),
        with_first_record=("first_record_known", "mean"),
        first_record_pre1914=("first_record_pre", "mean"),
        with_any_ensemble=("n_ensembles", lambda s: (s > 0).mean()),
        ensembles_per_place=("n_ensembles", "mean"),
        with_ensemble_pre1914=("ensemble_pre", "mean"),
        with_archive_citation=("archive_chars", lambda s: (s > 0).mean()),
        with_bibliography=("bibliography_chars", lambda s: (s > 0).mean()),
        median_chronicle_chars=("text_chars", "median"),
    ).round(3).reset_index()


def coverage_tables(
    places: pd.DataFrame,
    ensembles: pd.DataFrame,
    settlements: pd.DataFrame,
    districts: pd.DataFrame | None = None,
    *,
    okresy: list[str] | None,
    census: pd.DataFrame | None = None,
    cutoff: int = PRE_WAR_CUTOFF,
) -> dict[str, pd.DataFrame]:
    """Every table of the pilot, for the scraped sample and (from step A) the country.

    Args:
        places: scraper ``places.csv`` (all places; page fields filled for the sample).
        ensembles: scraper ``ensembles.csv``.
        settlements: ``data/final/geo/settlements_1910``.
        districts: ``data/final/geo/districts_1910`` (district names + court seats).
        okresy: the present-day okresy that were scraped; ``None`` once all of them are,
            in which case the denominator is every 1910 settlement rather than those whose
            okres timeline ends in a sample okres.
        census: ``census_okres_1910`` — if given, each 1910 okres gets its Czech-speaking
            share, the first thing coverage of a *Czech* theatre database depends on.

    Returns:
        A dict of DataFrames; :func:`coverage_report` prints them.
    """
    S = settlement_frame(settlements, districts)
    whole_country = okresy is None
    okresy = sorted(places["okres"].dropna().unique()) if whole_country else list(okresy)
    sample = places.copy() if whole_country else places[places["okres"].isin(okresy)].copy()
    missing_pages = int((~sample["page_cached"]).sum())
    tables: dict[str, pd.DataFrame] = {}

    # ── the join, on the sample ─────────────────────────────────────────────
    join = join_places_to_settlements(sample, S)
    n_before = len(sample)
    sample = sample.merge(join, on="geo_id", how="left", validate="1:1")
    assert len(sample) == n_before, "place ↔ join merge changed the row count"
    dates = place_dates(sample, ensembles[ensembles["geo_id"].isin(sample["geo_id"])], cutoff)
    sample = sample.merge(dates, on="geo_id", how="left", validate="1:1")
    assert len(sample) == n_before
    sample["matched"] = sample["settlement_row"].notna()
    # a place's crown land is that of the 1910 settlement it ties to
    sample = sample.merge(S[["settlement_row", "land"]].astype({"settlement_row": "Int64"}),
                          on="settlement_row", how="left", validate="m:1")
    assert len(sample) == n_before

    tables["join_by_method"] = (sample.groupby(["type", "match_method"]).size()
                                .unstack("type", fill_value=0).assign(all=lambda t: t.sum(axis=1)))
    tables["join_rate"] = sample.groupby("okres").agg(
        places=("geo_id", "size"), matched=("matched", "sum"),
        match_rate=("matched", "mean")).round(3).reset_index()
    tables["unmatched_places"] = sample.loc[~sample["matched"],
                                            ["geo_id", "name", "type", "okres", "match_method",
                                             "n_ensembles", "first_record_year"]]

    # ── the 1910 settlements of the sample area ─────────────────────────────
    in_scope = S["in_1910"] if whole_country else S["in_1910"] & S["okres_now"].isin(okresy)
    scope = S[in_scope]
    n_scope = len(scope)
    tied = sample.loc[sample["matched"], ["settlement_row", "geo_id", "ensemble_pre",
                                          "ensemble_ambiguous", "first_record_pre",
                                          "n_ensembles"]].astype({"settlement_row": "int64"})
    outside_scope = int((~tied["settlement_row"].isin(scope["settlement_row"])).sum())
    cover = scope.merge(tied, on="settlement_row", how="left", validate="1:1")
    assert len(cover) == n_scope, "settlement ↔ place merge changed the row count"
    for col in ("ensemble_pre", "ensemble_ambiguous", "first_record_pre"):
        cover[col] = cover[col].eq(True)          # no database place → not recorded
    by_okres = _share_table(cover, ["okres_now", "land", "okres_id", "okres_1910_cs"])
    if census is not None:
        by_okres = by_okres.merge(census[["okres_id", "lang_czech_share"]], on="okres_id",
                                  how="left", validate="m:1")
        by_okres["lang_czech_share"] = by_okres["lang_czech_share"].round(3)
    tables["settlements_by_okres_1910"] = by_okres
    tables["settlements_by_land"] = _share_table(cover, ["land"])
    tables["settlements_by_size"] = _share_table(cover, ["size_class"])
    tables["settlements_by_land_and_size"] = _share_table(cover, ["land", "size_class"])
    if census is not None:
        # a database of *Czech* theatre: coverage has to be read within language bands
        banded = cover.merge(census[["okres_id", "lang_czech_share"]], on="okres_id",
                             how="left", validate="m:1")
        assert len(banded) == n_scope, "settlement ↔ census merge changed the row count"
        banded["okres_language"] = pd.cut(
            banded["lang_czech_share"], LANGUAGE_BANDS, right=False,
            labels=LANGUAGE_LABELS).astype(object).fillna("no census row")
        czech = banded[banded["okres_language"] == LANGUAGE_LABELS[-1]]
        tables["settlements_by_land_and_language"] = _share_table(banded, ["land", "okres_language"])
        tables["settlements_czech_okresy_by_land_and_size"] = _share_table(czech, ["land", "size_class"])
        spread = _share_table(czech, ["land", "okres_id", "okres_1910_cs"])
        spread = spread[spread["settlements"] >= MIN_OKRES_SETTLEMENTS]
        tables["czech_okresy_spread_by_land"] = spread.groupby("land")["share_ensemble_pre1914"].describe(
            percentiles=[0.1, 0.25, 0.5, 0.75, 0.9]).round(3).reset_index()
    tables["scope_note"] = pd.DataFrame({
        "what": ["1910 settlements in the denominator (whole country: all; a sample: those "
                 "whose latest okres is a sample okres)",
                 "sample okres on the timeline but not its latest okres (left out)",
                 "sample places tied to a settlement outside that denominator",
                 "sample places whose page is not cached"],
        "n": [n_scope,
              int((S["in_1910"] & ~in_scope
                   & S["okres_tokens"].map(lambda t: bool(set(t) & set(okresy)))).sum()),
              outside_scope, missing_pages]})

    # ── how the database documents a place, by okres ────────────────────────
    paged = sample[sample["page_cached"]]
    tables["documentation_by_okres"] = _documentation_table(paged, "okres")
    tables["documentation_by_land"] = _documentation_table(paged, "land")

    period = pd.Series([first_record_period(y, p) for y, p in
                        zip(paged["first_record_year"], paged["first_record_precision"])],
                       index=paged.index).rename("period")
    for name, by in (("first_record_distribution", "okres"),
                     ("first_record_distribution_by_land", "land")):
        dist = pd.crosstab(period, paged[by].fillna("(no 1910 settlement)")).reindex(
            PERIOD_ORDER, fill_value=0)
        dist["all"] = dist.sum(axis=1)
        tables[name] = dist
    # heaping: a year that many places share is a source being read, not theatre beginning
    exact = paged[paged["first_record_precision"].eq("year")]
    tables["first_record_heaping"] = (
        exact.groupby("okres")["first_record_year"]
        .apply(lambda s: ", ".join(f"{int(y)}×{n}" for y, n in s.value_counts().head(6).items()))
        .rename("commonest first-report years (year×places)").reset_index())
    tables["first_record_precision"] = (paged.groupby(["first_record_precision",
                                                       "first_record_qualified"]).size()
                                        .rename("places").reset_index())

    counts, unclassified = association_counts(
        ensembles[ensembles["geo_id"].isin(sample["geo_id"])], cutoff)
    tables["associations_pre1914"] = counts
    tables["unclassified_names"] = unclassified.head(25).rename("ensembles").reset_index()
    tables["ensemble_year_precision"] = (ensembles.groupby(
        ["year_from_precision", "year_to_precision", "open_ended"]).size()
        .rename("ensembles").reset_index())

    # ── the whole country, from step A alone ────────────────────────────────
    tables.update(country_tables(places, S))
    return tables


def country_tables(places: pd.DataFrame, settlement_frame_: pd.DataFrame) -> dict[str, pd.DataFrame]:
    """From step A alone: which 1910 communes have *any* entry in the database.

    Presence only — the map XML has no dates.  The denominator is the 1910 communes
    (``status_1910 == "obec"``) of ``settlements_1910``; each is assigned to a present-day
    kraj through the latest okres on its timeline that the database knows.
    """
    S = settlement_frame_
    join = join_places_to_settlements(places, S)
    matched = join.dropna(subset=["settlement_row"])
    okres_kraj = (places.dropna(subset=["okres", "kraj"]).groupby("okres")["kraj"]
                  .agg(lambda s: s.mode().iloc[0]).to_dict())
    # the database files Prague under its numbered districts; the lexikon under "Praha"
    prague = next((k for k in okres_kraj.values() if k.startswith("Praha")), None)
    if prague:
        okres_kraj.setdefault("Praha", prague)

    def kraj_of(tokens: list[str]):
        for okres in reversed(tokens):
            if okres in okres_kraj:
                return okres_kraj[okres]
        return None

    communes = S[S["in_1910"] & S["status_1910"].eq("obec")].copy()
    communes["kraj"] = communes["okres_tokens"].map(kraj_of)
    communes["in_database"] = communes["settlement_row"].isin(matched["settlement_row"])

    def share(by: str) -> pd.DataFrame:
        table = communes.groupby(by, dropna=False).agg(
            communes_1910=("settlement_row", "size"),
            in_database=("in_database", "sum")).reset_index()
        table["share"] = (table["in_database"] / table["communes_1910"]).round(3)
        return table

    by_kraj = share("kraj").merge(
        places.groupby("kraj").agg(
            database_places=("geo_id", "size"),
            database_obce=("type", lambda s: int((s == "Obec").sum()))).reset_index(),
        on="kraj", how="outer", validate="1:1")
    return {
        "country_join": (join["match_method"].value_counts().rename("places").reset_index()),
        "country_communes_by_kraj": by_kraj.sort_values("share", ascending=False),
        "country_communes_by_land": share("land"),
        "country_places_by_type": places.groupby("type").agg(
            places=("geo_id", "size"),
            matched=("geo_id", lambda s: int(s.isin(matched["geo_id"]).sum()))).reset_index(),
    }
