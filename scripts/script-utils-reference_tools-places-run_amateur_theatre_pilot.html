How it works

Re-runnable, offline, and writes nothing but the parsed tables the scraper keeps beside its cache (data/raw/theatre/places.csv, ensembles.csv, and ensemble_pages.csv once step C is cached — re-parsed even with --reuse-tables, since it takes seconds) and the chronicle read line by line (data/interim/theatre/chronicle_lines.csv; nothing goes to data/final) and its clauses and the measure built from them (beside it, all in data/interim/theatre/). It

  1. re-parses the scraper's cache into those tables (no network),
  2. ties the database's places to settlements_1910 and prints the coverage tables of utils.parsing.amateur_theatre — for the scraped okresy, and from the map XML alone for the whole country,
  3. reads every place's chronicle into dated events (amateur_theatre.chronicle_lines), cuts them into dated clauses (amateur_theatre.chronicle_clauses, written to data/interim/theatre/chronicle_clauses.csv), codes them by the author's rulings (D-141), does the same for each ensemble page's own chronicle (ensemble_chronicle_clauses.csv, D-145) and builds the measure beside them, the pages' active decades and ends included: theatre_evidence.csv, the place × year panel place_year_panel.parquet and its 1910 settlement / court / okres sums ({settlement,court,okres}_year_panel.parquet), and prints what it found,
  4. reports whether the two pages of a place agree (which decides whether the full run needs one request per place or two) and what the full run would cost.

The sample is whatever okresy have been scraped; the default is the three of the pilot — Kutná Hora and Trutnov in Bohemia, Přerov in Moravia.

To fetch the sample first (about 400 requests, a quarter of an hour)::

python utils/scraping/scraper_amaterskedivadlo.py --step map
python utils/scraping/scraper_amaterskedivadlo.py --step places         --okres "Kutná Hora" Trutnov Přerov --pages geo

Usage::

python utils/reference_tools/places/run_amateur_theatre_pilot.py
python utils/reference_tools/places/run_amateur_theatre_pilot.py --okres "Kutná Hora" Kolín
python utils/reference_tools/places/run_amateur_theatre_pilot.py --all        # after the full scrape

def run(okresy: list[str] | None, out_dir: Path = ad.DEFAULT_OUT_DIR, *,
        reuse_tables: bool = False) -> dict[str, pd.DataFrame]:
    """Build and print every pilot table; returns them.

    ``okresy=None`` tabulates every okres whose places are all cached — the whole country
    once the full scrape is in.  ``reuse_tables`` reads ``places.csv`` / ``ensembles.csv``
    as they stand instead of re-parsing the cache (minutes, for the full scrape).
    """
    if reuse_tables:
        places = pd.read_csv(out_dir / "places.csv")
        ensembles = pd.read_csv(out_dir / "ensembles.csv")
    else:
        places, ensembles = ad.build_tables(out_dir)
    label = ", ".join(okresy) if okresy else ""
    if okresy is None and not places["page_cached"].all():
        complete = places.groupby("okres")["page_cached"].all()
        okresy = sorted(complete[complete].index)
        label = f"the {len(okresy)} fully scraped okresy"
        if not okresy:
            raise SystemExit("no okres is fully scraped yet — run step B first")
    elif okresy is None:
        label = f"the whole country ({places['okres'].nunique()} okresy, {len(places):,} places)"
    settlements = pd.read_parquet(FINAL / "geo" / "settlements_1910.parquet")
    districts = pd.read_parquet(FINAL / "geo" / "districts_1910.parquet")
    census = pd.read_parquet(FINAL / "census" / "census_okres_1910.parquet",
                             columns=["okres_id", "lang_czech_share"])

    tables = at.coverage_tables(places, ensembles, settlements, districts,
                                okresy=okresy, census=census)
    print(f"amateur-theatre pilot — scraper {ad.__version__}, tabulation {at.__version__}")
    print(f"source: {ad.CITATION}")
    print(f"sample: {label}")
    at.coverage_report(tables)

    lines = at.chronicle_lines(places)
    at.chronicle_report(places, lines, ensembles)
    INTERIM.mkdir(parents=True, exist_ok=True)
    # utf-8-sig: the database's own prose, so Excel must open it as UTF-8
    lines.to_csv(INTERIM / "chronicle_lines.csv", index=False, encoding="utf-8-sig")
    print(f"wrote {INTERIM / 'chronicle_lines.csv'} ({len(lines):,} lines)")
    tables["chronicle_lines"] = lines

    clauses = at.code_clauses(at.chronicle_clauses(lines))
    at.clause_report(clauses)
    print("\n── pre-1914 non-puppet clauses by event cue\n"
          + at.cue_table(clauses).to_string(index=False))
    clauses.to_csv(INTERIM / "chronicle_clauses.csv", index=False, encoding="utf-8-sig")
    print(f"wrote {INTERIM / 'chronicle_clauses.csv'} ({len(clauses):,} clauses)")
    tables["chronicle_clauses"] = clauses

    # the ensembles' own pages (step C, D-145): re-parsed from the cache every run — a few
    # seconds — so their decades always follow the scraper's current reader
    pages = ensemble_clauses = None
    if ad._cached_ids(out_dir, ad.ENSEMBLE_KIND):
        pages = ad.build_ensemble_pages(out_dir)
        summary = at.ensemble_page_summary(pages, ensembles)
        print("\n── ensemble pages (step C), one count per ensemble\n"
              + "\n".join(f"{k:>24}: {v:,}" for k, v in summary.items()))
        ensemble_clauses = at.code_clauses(at.ensemble_chronicle_clauses(pages, ensembles))
        ensemble_clauses.to_csv(INTERIM / "ensemble_chronicle_clauses.csv", index=False, encoding="utf-8-sig")
        print(f"wrote {INTERIM / 'ensemble_chronicle_clauses.csv'} ({len(ensemble_clauses):,} clauses, "
              f"{int(ensemble_clauses['needs_model'].sum()):,} left to the model pass)")
        tables.update(ensemble_pages=pages, ensemble_chronicle_clauses=ensemble_clauses)

    # the measure (D-141, D-145): evidence → place × year → 1910 settlement / court / okres × year
    evidence = at.evidence_table(places, ensembles, clauses, pages, ensemble_clauses)
    panel = at.place_year_panel(evidence, places["geo_id"])
    at.panel_report(panel, evidence)
    frame = at.settlement_frame(settlements, districts)
    areas = at.area_year_tables(panel, at.join_places_to_settlements(places, frame), frame)
    evidence.to_csv(INTERIM / "theatre_evidence.csv", index=False, encoding="utf-8-sig")
    panel.to_parquet(INTERIM / "place_year_panel.parquet", index=False)
    for level, table in areas.items():
        table.to_parquet(INTERIM / f"{level}_year_panel.parquet", index=False)
    print(f"wrote theatre_evidence.csv ({len(evidence):,} rows), place_year_panel.parquet"
          f" ({len(panel):,}), " + ", ".join(f"{k}_year_panel.parquet ({len(v):,})" for k, v in areas.items()))
    tables.update(evidence=evidence, place_year_panel=panel,
                  **{f"{k}_year_panel": v for k, v in areas.items()})

    agreement = ad.compare_sources(out_dir)
    print("\n── list page vs place page (places with both cached)")
    print(f"identical on {agreement['places_identical']} of {agreement['places_compared']}"
          f"; differing geo ids: {agreement['differing'][:20]}")

    print("\n── the full step B, from this cache's own timings")
    for pages in (("geo",), ("geo", "list")):
        estimate = ad.estimate_full_run(out_dir, pages=pages)
        print(f"pages={pages}: {estimate}")
    return tables
