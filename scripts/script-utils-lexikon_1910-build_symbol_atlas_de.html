How it works

The German twin of build_symbol_atlas. The Spezialortsrepertorium von Böhmen explains its pictograms on the plate Zeichenerklärung, p. XI — scan 0012 of the UBW scan — three columns wide, one symbol to a line with its German meaning beside it, exactly as the Czech Výklad značek does on its p. XI.

The German reader is shown its own volume's plate, not the Czech one. The two plates are not the same list: the German one prints 80 symbols, and the three the Czech plate has beyond them — [cooperative], [port] and [rc-parish-exp] — are not on it (the German volume writes cooperatives as letters only, and a parish expositure as the word Exp.). The prompt tells the reader that the key "is the whole of what this volume prints", so a Czech key would be a false statement to a German reader, three times over.

How it works is the Czech builder's, reused rather than copied: the plate is self-labelling, so each printed gloss names its token through the de pictogram rows of institution_abbreviations.csv (whose expansions were read off this same plate), and the layout — symbol band, gutter, gloss — gives the crop. The one difference is the text layer: this PDF has no ALTO, so the words come from the PDF's own positional OCR layer (pymupdf get_text("words")), scaled onto a render at the Czech plate's width so the Czech builder's pixel constants keep their meaning.

What comes out, in reference/places/symbols/de/:

&lt;token&gt;.png one crop per pictogram manifest.csv token, printed gloss, match score, source box atlas.png the crops with their tokens, for a person atlas_prompt.png the dense sheet a reader is sent (--atlas arms)

def load_targets() -> list[tuple]:
    """``[(token, printed gloss, english, category)]`` for every German pictogram.

    The gloss is the expansion **plus the qualifier printed with it**, which the
    table keeps in ``note`` after the plate reference (``Rotte`` + ``Häuser in
    loserer Verteilung …``). The plate prints both on one entry, and the OCR
    damages short words worst — it reads *Rotte* as ``Hotte`` — so the long
    qualifier is what lets the entry be matched at all.
    """
    rows = lxin.load_abbreviations(bsa.ABBREVIATIONS, lang="de")
    out = []
    for a in rows:
        if a.kind != "pictogram" or a.lang != "de" or not a.expansion:
            continue
        qualifier = a.note.split(";", 1)[1].strip() if ";" in a.note else ""
        printed = f"{a.expansion} ({qualifier})" if qualifier and "connector" not in qualifier             else a.expansion
        out.append((a.code, printed, a.english, a.category))
    return out

def plate_words(pdf_path: Path, scan: int, width: int) -> list[tuple]:
    """The plate's OCR words as ALTO-shaped ``(top, left, width, height, text)``.

    That is the tuple :func:`build_symbol_atlas.cluster_lines` and
    :func:`~build_symbol_atlas.columns_of` index into, so the Czech layout code
    reads the German plate unchanged.
    """
    import fitz

    with fitz.open(str(pdf_path)) as doc:
        page = doc[scan - 1]
        zoom = width / page.rect.width
        out = []
        for x0, y0, x1, y1, text, *_rest in page.get_text("words"):
            if text.strip():
                out.append((int(y0 * zoom), int(x0 * zoom),
                            max(1, int((x1 - x0) * zoom)),
                            max(1, int((y1 - y0) * zoom)), text))
    return out

def dark_mask(image, share: float = PAPER_SHARE):

def plate_ink(dark, box) -> float:
    """Share of ``box`` that is ink in the plate's own mask."""
    left, top, right, bottom = box
    cell = dark[top:bottom, left:right]
    return float(cell.mean()) if cell.size else 0.0

def ink_columns(dark, top: int, bottom: int) -> list[tuple[int, int, int]]:
    """``(column_left, gloss_left, column_right)`` for each printed column.

    Ink runs between blank vertical bands of rows ``top..bottom``; a run under
    :data:`SYMBOL_MAX` followed directly by one over :data:`GLOSS_MIN` opens a
    column, and a column ends where the next one starts.
    """
    profile = dark[top:bottom].sum(axis=0)
    runs, start = [], None
    for x, value in enumerate(list(profile) + [0] * BLANK_MIN):
        if value > 2 and start is None:
            start = x
        elif value <= 2 and start is not None:
            end = x
            # A blank band shorter than BLANK_MIN does not end the run.
            if x + BLANK_MIN <= len(profile) and profile[x:x + BLANK_MIN].max() > 2:
                continue
            runs.append((start, end))
            start = None
    columns = []
    for (a, b), (c, d) in zip(runs, runs[1:]):
        if b - a < SYMBOL_MAX and d - c >= GLOSS_MIN:
            columns.append([a, c, None])
    for here, after in zip(columns, columns[1:]):
        here[2] = after[0] - 1
    if columns:
        columns[-1][2] = runs[-1][1]
    return [tuple(c) for c in columns]

def plate_entries(words: list[tuple], bounds, dark, top_row: int = 0) -> list[dict]:
    """``[{gloss, top, bottom, left, gloss_left}]``, one per printed meaning.

    The wrap rule is the Czech builder's — a line continues the meaning above
    when that meaning is unclosed and the line is adjacent (the long note in
    :func:`build_symbol_atlas.build`) — except that "nothing in the symbol band"
    is read off the ink rather than off the OCR's symbol words, which on this
    plate are glued to the gloss or missing.
    """
    entries: list[dict] = []
    for left, gloss_left, right in bounds:
        # Below the title only: the running head ``XI`` and the title itself
        # sit inside a column's x range.
        column = [w for w in words if gloss_left - 6 <= w[1] + w[2] // 2 <= right
                  and w[0] >= top_row]
        mine: list[dict] = []
        for gloss, _line, top, bottom, _x0, _x1 in bsa.cluster_lines(column):
            if not gloss.strip():
                continue
            band = dark[top:bottom + 1, left:max(left + 1, gloss_left - 4)]
            has_symbol = band.size and band.mean() >= BAND_INK
            if (not has_symbol and mine and top - mine[-1]["bottom"] <= bsa.MAX_WRAP_GAP
                    and not bsa._ENTRY_END_RE.search(mine[-1]["gloss"])):
                mine[-1]["gloss"] += " " + gloss
                mine[-1]["bottom"] = bottom
                continue
            mine.append({"gloss": gloss, "top": top, "bottom": bottom,
                         "left": left, "gloss_left": gloss_left})
        entries += mine
    return entries

def body_rows(words: list[tuple], height: int) -> tuple[int, int]:
    """The rows the legend occupies: under its title, down to its last word."""
    from utils.lexikon_1910.lexikon_1910_notes import normalise

    title = [w for w in words if "zeichenerk" in normalise(w[4])]

def build(scan: int, show: bool = False) -> int:
    from PIL import Image, ImageDraw, ImageFont

    profile = lxvol.load_profile(PROFILE)
    image = lx.render_page(profile.source_pdf, scan, RENDER_WIDTH)
    words = plate_words(profile.source_pdf, scan, RENDER_WIDTH)
    targets = load_targets()
    dark = dark_mask(image)
    top, bottom = body_rows(words, image.height)
    bounds = ink_columns(dark_mask(image, COLUMN_SHARE), top, bottom)
    print(f"  plate: scan {scan:04d} of {profile.slug}, {image.width}x{image.height}, "
          f"{len(words)} words; {len(targets)} curated German pictograms; "
          f"{len(bounds)} printed columns "
          + ", ".join(f"{a}-{g} | {g}-{b}" for a, g, b in bounds))

    OUT_DIR.mkdir(parents=True, exist_ok=True)
    manifest, unmatched, used = [], [], {}
    for entry in plate_entries(words, bounds, dark, top):
        token, expansion, score = bsa.match_gloss(entry["gloss"], targets)
        box = (max(0, entry["left"] - 10), max(0, entry["top"] - 6),
               min(image.width, entry["gloss_left"] - 8),
               min(image.height, entry["bottom"] + 6))
        record = {"token": token, "gloss_printed": entry["gloss"][:70],
                  "gloss_curated": expansion, "score": round(score, 1),
                  "scan": f"{scan:04d}", "left": box[0], "top": box[1],
                  "right": box[2], "bottom": box[3], "file": "", "ink": ""}
        if (score < bsa._MIN_SCORE or box[2] - box[0] < bsa._MIN_SYMBOL_PX
                or box[3] - box[1] < bsa._MIN_SYMBOL_PX):
            record["verdict"] = ("no curated token matches this printed meaning"
                                 if score < bsa._MIN_SCORE
                                 else "no symbol region on this line")
            unmatched.append(record)
            continue
        crop = image.crop(box)
        record["ink"] = round(plate_ink(dark, box), 4)
        if record["ink"] < bsa._MIN_INK_SHARE:
            record["verdict"] = "the crop holds no ink"
            unmatched.append(record)
            continue
        used[token] = used.get(token, 0) + 1
        name = token.strip("[]") + ("" if used[token] == 1 else f"_{used[token]}")
        crop.save(OUT_DIR / f"{name}.png")
        record["file"] = f"{name}.png"
        record["verdict"] = "ok" if used[token] == 1 else "duplicate token"
        manifest.append(record)

    manifest.sort(key=lambda r: r["token"])
    with (OUT_DIR / "manifest.csv").open("w", newline="", encoding="utf-8") as fh:
        writer = csv.DictWriter(fh, fieldnames=list((manifest or unmatched)[0]))
        writer.writeheader()
        writer.writerows(manifest + unmatched)
    bsa._montage([r for r in manifest if r["verdict"] == "ok"], Image, ImageDraw,
                 ImageFont, out_dir=OUT_DIR, title=TITLE)
    return report(manifest, unmatched, targets, show)

def report(manifest, unmatched, targets, show: bool) -> int:
    """What the plate gave, and which tokens it did not give a symbol for."""
    named = {r["token"] for r in manifest if r["verdict"] == "ok"}
    missing = [t for t, *_ in targets if t not in named]
    dup = [r for r in manifest if r["verdict"] == "duplicate token"]
    print(f"\n  {len(named)} of {len(targets)} German pictograms cropped and "
          f"named -> {OUT_DIR}")
    for label, rows in (("tokens matched by more than one plate line", dup),
                        ("plate lines no token matches", [r for r in unmatched
                         if r["verdict"].startswith("no curated")]),
                        ("crops rejected as blank paper", [r for r in unmatched
                         if r["verdict"].startswith("the crop")])):
        if rows:
            print(f"  ! {len(rows)} {label}:")
            for r in rows[:12]:
                print(f"      {r['token'] or '-':22s} {r['gloss_printed'][:50]!r}")
    if missing:
        print(f"  ! curated tokens WITHOUT a symbol from the plate: "
              f"{', '.join(missing)}")
    if show:
        for r in manifest:
            print(f"    {r['token']:22s} {r['score']:5.1f}  {r['gloss_printed'][:44]!r}")
    return 0 if not missing else 1

    The gloss is the expansion **plus the qualifier printed with it**, which the
    table keeps in ``note`` after the plate reference (``Rotte`` + ``Häuser in
    loserer Verteilung …``). The plate prints both on one entry, and the OCR
    damages short words worst — it reads *Rotte* as ``Hotte`` — so the long
    qualifier is what lets the entry be matched at all.

            continue
        qualifier = a.note.split(";", 1)[1].strip() if ";" in a.note else ""
        printed = f"{a.expansion} ({qualifier})" if qualifier and "connector" not in qualifier             else a.expansion
        out.append((a.code, printed, a.english, a.category))
    return out


def plate_words(pdf_path: Path, scan: int, width: int) -> list[tuple]:
    """The plate's OCR words as ALTO-shaped ``(top, left, width, height, text)``.

    That is the tuple :func:`build_symbol_atlas.cluster_lines` and
    :func:`~build_symbol_atlas.columns_of` index into, so the Czech layout code
    reads the German plate unchanged.
    """
    import fitz

    with fitz.open(str(pdf_path)) as doc:
        page = doc[scan - 1]
        zoom = width / page.rect.width
        out = []
        for x0, y0, x1, y1, text, *_rest in page.get_text("words"):
            if text.strip():
                out.append((int(y0 * zoom), int(x0 * zoom),
                            max(1, int((x1 - x0) * zoom)),
                            max(1, int((y1 - y0) * zoom)), text))
    return out


#: Columns are found in the plate's ink, not in its words. Measured on the
#: render (2026-10-19): the German plate's gutters are ~30 px where the Czech
#: plate's are 62-81, under the Czech builder's 50 px column gap, and its OCR
#: glues a symbol to its gloss in one word (``(SA)ev.``), so the word-based
#: split of :func:`build_symbol_atlas.columns_of` found two columns of three.
#: The ink is unambiguous: a narrow band of symbols, a blank gap, a wide gloss.
BLANK_MIN = 8               # a vertical blank band at least this wide separates
SYMBOL_MAX = 120            # a symbol band is narrower than this ...
GLOSS_MIN = 300             # ... and is followed by a gloss at least this wide

#: The share of a line's symbol band that must be ink for the line to carry a
#: symbol. A wrapped gloss has none; a printed symbol is several per cent.
BAND_INK = 0.01

#: What counts as ink: darker than this share of the paper's brightness. **Not
#: the Czech builder's 0.72.** This scan prints faint grey (~147 on paper ~178,
#: the profile's note, i.e. 0.83 of paper), so at 0.72 the thin outline symbols
#: — ○ *Dorf*, ⊙ *Markt*, † *Filialkirche* — held 0.2-1.4 % ink and were
#: rejected as blank paper, twelve of them. At 0.80 the faintest real symbol on
#: the plate holds 1.7 % and a blank band of the same size 0.0 % (measured
#: 2026-10-19); the Czech minimum share, ``_MIN_INK_SHARE``, is kept.
PAPER_SHARE = 0.80

#: Where the columns are is read on the strict cut, the Czech one. At
#: :data:`PAPER_SHARE` the faint grey between a symbol and its gloss bridges the
#: ~17 px gap in the middle column and the symbol band merges into the gloss;
#: at 0.72 the three columns come out exactly as measured on the render.
COLUMN_SHARE = 0.72


def dark_mask(image, share: float = PAPER_SHARE):
    """Pixels darker than ``share`` of the plate's paper (its 90th percentile)."""
    import numpy as np

    grey = np.asarray(image.convert("L")).astype(int)
    return grey < np.percentile(grey, 90) * share


def plate_ink(dark, box) -> float:
    """Share of ``box`` that is ink in the plate's own mask."""
    left, top, right, bottom = box
    cell = dark[top:bottom, left:right]
    return float(cell.mean()) if cell.size else 0.0


def ink_columns(dark, top: int, bottom: int) -> list[tuple[int, int, int]]:
    """``(column_left, gloss_left, column_right)`` for each printed column.

    Ink runs between blank vertical bands of rows ``top..bottom``; a run under
    :data:`SYMBOL_MAX` followed directly by one over :data:`GLOSS_MIN` opens a
    column, and a column ends where the next one starts.
    """
    profile = dark[top:bottom].sum(axis=0)
    runs, start = [], None
    for x, value in enumerate(list(profile) + [0] * BLANK_MIN):
        if value > 2 and start is None:
            start = x
        elif value <= 2 and start is not None:
            end = x
            # A blank band shorter than BLANK_MIN does not end the run.
            if x + BLANK_MIN <= len(profile) and profile[x:x + BLANK_MIN].max() > 2:
                continue
            runs.append((start, end))
            start = None
    columns = []
    for (a, b), (c, d) in zip(runs, runs[1:]):
        if b - a < SYMBOL_MAX and d - c >= GLOSS_MIN:
            columns.append([a, c, None])
    for here, after in zip(columns, columns[1:]):
        here[2] = after[0] - 1
    if columns:
        columns[-1][2] = runs[-1][1]
    return [tuple(c) for c in columns]


def plate_entries(words: list[tuple], bounds, dark, top_row: int = 0) -> list[dict]:
    """``[{gloss, top, bottom, left, gloss_left}]``, one per printed meaning.

    The wrap rule is the Czech builder's — a line continues the meaning above
    when that meaning is unclosed and the line is adjacent (the long note in
    :func:`build_symbol_atlas.build`) — except that "nothing in the symbol band"
    is read off the ink rather than off the OCR's symbol words, which on this
    plate are glued to the gloss or missing.
    """
    entries: list[dict] = []
    for left, gloss_left, right in bounds:
        # Below the title only: the running head ``XI`` and the title itself
        # sit inside a column's x range.
        column = [w for w in words if gloss_left - 6 <= w[1] + w[2] // 2 <= right
                  and w[0] >= top_row]
        mine: list[dict] = []
        for gloss, _line, top, bottom, _x0, _x1 in bsa.cluster_lines(column):
            if not gloss.strip():
                continue
            band = dark[top:bottom + 1, left:max(left + 1, gloss_left - 4)]
            has_symbol = band.size and band.mean() >= BAND_INK
            if (not has_symbol and mine and top - mine[-1]["bottom"] <= bsa.MAX_WRAP_GAP
                    and not bsa._ENTRY_END_RE.search(mine[-1]["gloss"])):
                mine[-1]["gloss"] += " " + gloss
                mine[-1]["bottom"] = bottom
                continue
            mine.append({"gloss": gloss, "top": top, "bottom": bottom,
                         "left": left, "gloss_left": gloss_left})
        entries += mine
    return entries


def body_rows(words: list[tuple], height: int) -> tuple[int, int]:
    """The rows the legend occupies: under its title, down to its last word."""
    from utils.lexikon_1910.lexikon_1910_notes import normalise

    title = [w for w in words if "zeichenerk" in normalise(w[4])]
    top = (max(w[0] + w[3] for w in title) + 20) if title else height // 5
    below = [w[0] + w[3] for w in words if w[0] > top]
    return top, (max(below) if below else height)


def build(scan: int, show: bool = False) -> int:
    from PIL import Image, ImageDraw, ImageFont

    profile = lxvol.load_profile(PROFILE)
    image = lx.render_page(profile.source_pdf, scan, RENDER_WIDTH)
    words = plate_words(profile.source_pdf, scan, RENDER_WIDTH)
    targets = load_targets()
    dark = dark_mask(image)
    top, bottom = body_rows(words, image.height)
    bounds = ink_columns(dark_mask(image, COLUMN_SHARE), top, bottom)
    print(f"  plate: scan {scan:04d} of {profile.slug}, {image.width}x{image.height}, "
          f"{len(words)} words; {len(targets)} curated German pictograms; "
          f"{len(bounds)} printed columns "
          + ", ".join(f"{a}-{g} | {g}-{b}" for a, g, b in bounds))

    OUT_DIR.mkdir(parents=True, exist_ok=True)
    manifest, unmatched, used = [], [], {}
    for entry in plate_entries(words, bounds, dark, top):
        token, expansion, score = bsa.match_gloss(entry["gloss"], targets)
        box = (max(0, entry["left"] - 10), max(0, entry["top"] - 6),
               min(image.width, entry["gloss_left"] - 8),
               min(image.height, entry["bottom"] + 6))
        record = {"token": token, "gloss_printed": entry["gloss"][:70],
                  "gloss_curated": expansion, "score": round(score, 1),
                  "scan": f"{scan:04d}", "left": box[0], "top": box[1],
                  "right": box[2], "bottom": box[3], "file": "", "ink": ""}
        if (score < bsa._MIN_SCORE or box[2] - box[0] < bsa._MIN_SYMBOL_PX
                or box[3] - box[1] < bsa._MIN_SYMBOL_PX):
            record["verdict"] = ("no curated token matches this printed meaning"
                                 if score < bsa._MIN_SCORE
                                 else "no symbol region on this line")
            unmatched.append(record)
            continue
        crop = image.crop(box)
        record["ink"] = round(plate_ink(dark, box), 4)
        if record["ink"] < bsa._MIN_INK_SHARE:
            record["verdict"] = "the crop holds no ink"
            unmatched.append(record)
            continue
        used[token] = used.get(token, 0) + 1
        name = token.strip("[]") + ("" if used[token] == 1 else f"_{used[token]}")
        crop.save(OUT_DIR / f"{name}.png")
        record["file"] = f"{name}.png"
        record["verdict"] = "ok" if used[token] == 1 else "duplicate token"
        manifest.append(record)

    manifest.sort(key=lambda r: r["token"])
    with (OUT_DIR / "manifest.csv").open("w", newline="", encoding="utf-8") as fh:
        writer = csv.DictWriter(fh, fieldnames=list((manifest or unmatched)[0]))
        writer.writeheader()
        writer.writerows(manifest + unmatched)
    bsa._montage([r for r in manifest if r["verdict"] == "ok"], Image, ImageDraw,
                 ImageFont, out_dir=OUT_DIR, title=TITLE)
    return report(manifest, unmatched, targets, show)


def report(manifest, unmatched, targets, show: bool) -> int:
    """What the plate gave, and which tokens it did not give a symbol for."""
    named = {r["token"] for r in manifest if r["verdict"] == "ok"}
    missing = [t for t, *_ in targets if t not in named]
    dup = [r for r in manifest if r["verdict"] == "duplicate token"]
    print(f"\n  {len(named)} of {len(targets)} German pictograms cropped and "
          f"named -> {OUT_DIR}")
    for label, rows in (("tokens matched by more than one plate line", dup),
                        ("plate lines no token matches", [r for r in unmatched
                         if r["verdict"].startswith("no curated")]),
                        ("crops rejected as blank paper", [r for r in unmatched
                         if r["verdict"].startswith("the crop")])):
        if rows:
            print(f"  ! {len(rows)} {label}:")
            for r in rows[:12]:
                print(f"      {r['token'] or '-':22s} {r['gloss_printed'][:50]!r}")
    if missing:
        print(f"  ! curated tokens WITHOUT a symbol from the plate: "
              f"{', '.join(missing)}")
    if show:
        for r in manifest:
            print(f"    {r['token']:22s} {r['score']:5.1f}  {r['gloss_printed'][:44]!r}")
    return 0 if not missing else 1
