How it works

Prague and Liberec (Reichenberg) were Städte mit eigenem Statut and the register describes them differently: their institutions are printed as running text over the full width of scans 0015–0018, not as the two-column apparatus of keys and pictograms. lexikon_1910_notes.extract_notes finds no divider there (correctly — there is none), so no section crop and no reader ever saw this text.

This writes it out, one file per scan, as the text layer has it: lines rebuilt from the ALTO words in reading order, paragraphs split at the wider vertical gaps the register leaves between them. On a page that opens with a table (0015 Prague, 0018 Liberec) only the print below the table's column rules is taken; the table itself belongs to the table channel. The OCR is good on this type — a slip every few lines (depositili for depositní) — so the files are a reading to check, not a transcription. No model is called.

def table_foot(page) -> int:
    """The lowest foot of the table's inner column rules, or 0 with no table.

    Inner: the frame's own verticals are the outermost rules on the sheet and
    run past the table, so the rules within 100 px of the leftmost and rightmost
    are left out. What a page prints under the table — the table's footnotes and
    then the prose — is kept.
    """
    tall = [r for r in page.verticals if r["HEIGHT"] >= 150]
    if not tall:
        return 0
    xs = [r["HPOS"] for r in tall]
    inner = [r for r in tall if min(xs) + 100 < r["HPOS"] < max(xs) - 100
             and r["VPOS"] < 0.5 * page.height]
    return max((r["VPOS"] + r["HEIGHT"] for r in inner), default=0)

def vision_read(scan: int, top: int, spend: bool, allow_other_machine: bool) -> str | None:
    """The page's prose read by the vision model, cached in ``prose_pages/raw``."""
    import hashlib
    import io
    import json
    import os
    import platform
    import re

    raw_dir = OUT_DIR / "raw"
    raw_dir.mkdir(parents=True, exist_ok=True)
    page = lx.read_alto(CZ_PAGES / f"{scan:04d}.alto.xml")
    image = lx.render_page(CZ_PDF, scan, page.width)
    foot = min(page.height, max(s[0] + s[3] for s in page.strings) + 20)
    middle = (top + foot) // 2
    texts = []
    for half, (y0, y1) in enumerate(((max(0, top - 10), middle + 40),
                                     (middle - 40, foot))):
        crop = image.crop((0, y0, image.width, y1))
        buffer = io.BytesIO()
        crop.save(buffer, format="PNG")
        png = buffer.getvalue()
        key = hashlib.sha256(png + VISION_PROMPT.encode() + VISION_MODEL.encode()).hexdigest()[:16]
        cached = raw_dir / f"{scan:04d}_{half}_{key}.json"
        if cached.exists():
            texts.append(json.loads(cached.read_text(encoding="utf-8"))["text"])
            continue
        host = re.sub(r"[^a-z0-9]+", "-", (os.environ.get("BAKEOFF_HOST") or platform.node()).lower()).strip("-")
        if not spend:
            print(f"    {scan:04d} half {half}: not cached; --spend to read it")
            return None
        if host != MACHINE_OF_RECORD and not allow_other_machine:
            raise SystemExit(f"this is {host}, not {MACHINE_OF_RECORD} (D-68); nothing was sent")
        import base64

        from openai import OpenAI

        sys.path.insert(0, str(ROOT))
        from utils.lexikon_1910 import lexikon_1910_pages as lxp
        lxp.load_credentials()
        client = OpenAI(base_url=GEMINI_URL, api_key=os.environ["GEMINI_API_KEY"])
        reply = client.chat.completions.create(
            model=VISION_MODEL, temperature=0.0, reasoning_effort="none",
            messages=[{"role": "user", "content": [
                {"type": "text", "text": VISION_PROMPT},
                {"type": "image_url", "image_url": {
                    "url": "data:image/png;base64," + base64.b64encode(png).decode()}}]}])
        text = reply.choices[0].message.content or ""
        usage = reply.usage
        out_tokens = usage.total_tokens - usage.prompt_tokens   # thinking is billed as output
        cost = (usage.prompt_tokens * PRICE_IN + out_tokens * PRICE_OUT) / 1e6
        cached.write_text(json.dumps({"scan": scan, "half": half, "model": VISION_MODEL,
                                      "box": [0, y0, image.width, y1], "text": text,
                                      "prompt_tokens": usage.prompt_tokens,
                                      "output_tokens": out_tokens, "cost_usd": round(cost, 5)},
                                     ensure_ascii=False, indent=1), encoding="utf-8")
        print(f"    {scan:04d} half {half}: read, ${cost:.4f}")
        texts.append(text)
    return "\n\n".join(t.strip() for t in texts)

import sys

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

LIBRARY = ROOT / "library" / "lexicons"
CZ_PAGES = LIBRARY / "NDK_Podrobny-seznam-mist-Cechy-1910_cz_pages"
OUT_DIR = ROOT / "output" / "lexikon_1910_pilot" / "prose_pages"

#: The statutory towns' pages in the Czech volume: Prague 0015–0017, Liberec 0018.
PROSE_SCANS = (15, 16, 17, 18)

#: A paragraph break is a line gap this many times the page's median line gap.
PARAGRAPH_GAP = 1.6

def _join_breaks(text: str) -> str:
    import re
    return re.sub(r"(\w)- (?=[a-zà-žß])", r"\1", text)


#: The vision read (``--vision``). Gemini Flash, thinking off, as the apparatus
#: reader; each page is sent as two halves of its prose so the 8-point type
#: survives the image budget. Four pages, eight calls: cents.

across lines with a hyphen. Do not correct, expand or translate anything. If a
character cannot be read, write [?]. If a pictogram is printed, write [pictogram].
Return only the text."""
