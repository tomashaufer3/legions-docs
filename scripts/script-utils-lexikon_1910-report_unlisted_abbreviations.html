How it works

Decodes the offices text each reading wrote (notes_&lt;label&gt;.csv) with the current code table and lists, most often read first, every item no code accounts for, spelled out through the volume's own abbreviation parts (abbreviation_components.csv). It is the census the code table is grown from: an item several readings print and no code reaches is an institution the inventory loses. No API, no model; reads the bake-off pass files, writes nothing::

python utils/lexikon_1910/report_unlisted_abbreviations.py cz gold
python utils/lexikon_1910/report_unlisted_abbreviations.py de-bohemia gold --top 80

The sections are re-cut from the notes as written, without the text layer's identities, so a district label can differ from the reading's own sidecar (mechanice for nechanice): fine for a census of the text, not a substitute for the reading's decoded records.

"""The offices items no code in ``institution_abbreviations.csv`` reaches, on one run's readings (D-156).

Decodes the offices text each reading wrote (``notes_<label>.csv``) with the current code table
and lists, most often read first, every item no code accounts for, spelled out through the
volume's own abbreviation parts (``abbreviation_components.csv``). It is the census the code
table is grown from: an item several readings print and no code reaches is an institution the
inventory loses. No API, no model; reads the bake-off pass files, writes nothing::

    python utils/lexikon_1910/report_unlisted_abbreviations.py cz gold

The sections are re-cut from the notes as written, without the text layer's identities, so a
district label can differ from the reading's own sidecar (``mechanice`` for ``nechanice``):
fine for a census of the text, not a substitute for the reading's decoded records.

from __future__ import annotations

import argparse
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
for _p in (ROOT, Path(__file__).resolve().parent):
    if str(_p) not in sys.path:
        sys.path.insert(0, str(_p))

import pandas as pd  # noqa: E402

import bakeoff as bo  # noqa: E402
import run_frozen as rf  # noqa: E402
from utils.lexikon_1910 import lexikon_1910_institutions as lxin  # noqa: E402
from utils.lexikon_1910 import lexikon_1910_notes as lxnt  # noqa: E402
from utils.lexikon_1910 import lexikon_1910_vision as lxv  # noqa: E402

__version__ = "1.0.0"

ABBREVIATIONS = ROOT / "reference" / "places" / "institution_abbreviations.csv"
COMPONENTS = ROOT / "reference" / "places" / "abbreviation_components.csv"
#: A German heading line, which the German notes keep as the section's first line.
_DE_HEADING = re.compile(r"^\s*(Ämter|Amter)\b")


def reading_entries(label: str) -> list[lxin.Entry]:
    """The offices entries of one reading, cut from its notes file."""
    path = bo.find_pass_file("notes", label)
    if path is None:
        return []
    notes = pd.read_csv(path, dtype=str, keep_default_na=False)
    if "kind" in notes:                       # a German pass writes its sections as cut
        out = []
        offices = notes[notes["kind"] == "offices"]
        for (page, _order, court), g in offices.groupby(["page", "order", "court_de"], sort=False):
            lines = [t for t in g.sort_values("line_no", key=lambda s: s.astype(int))["text"]
                     if not _DE_HEADING.match(t)]
            out += lxin.parse_entries(lxnt.Section("offices", court, page, lines))
        return out
    pages = [lxv.notes_from_regions(p, {r: g.loc[g["region"] == r, "text"].tolist()
                                        for r in ("left", "right", "footer")})
             for p, g in notes.groupby("page", sort=True)]
    return [e for s in lxnt.collect_sections(pages) if s.kind == "offices"
            for e in lxin.parse_entries(s)]


def unlisted(edition: str, pages: str) -> tuple[list[dict], int]:
    """``(rows, readings)``: every unlisted item of every reading of the run."""
    lang = rf._lang(edition)
    table = lxin.load_abbreviations(ABBREVIATIONS, lang=lang, volume=rf.GERMAN.get(edition, ""))
    parts = lxin.load_components(COMPONENTS, lang=lang)
    lab = rf.labels(pages, edition)
    readers = [lab["gemini"], *lab["haiku"]]
    rows = []
    for label in readers:
        for entry in reading_entries(label):
            rows += [dict(r, reader=label) for r in lxin.unlisted_items(entry, table, parts)]
    return rows, len(readers)


def main(argv: list[str] | None = None) -> None:
    ap = argparse.ArgumentParser(description=__doc__.split("\n")[0])
    ap.add_argument("edition")
    ap.add_argument("pages")
    ap.add_argument("--top", type=int, default=60)
    args = ap.parse_args(argv)
    rows, readers = unlisted(args.edition, args.pages)
    print(lxin.unlisted_report(rows, readers=readers, top=args.top))


if __name__ == "__main__":
    main()
