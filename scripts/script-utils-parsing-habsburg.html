TL;DR

Field-specific harmonisation for the VÚA padlí v 1. sv. válce dump (data/raw/habsburg/habsburg_fallen.csv, ~186k fallen Habsburg soldiers). Run after utils.parsing.clean has normalised placeholders and names.

How it works

VÚA-specific fields are parsed here; dates are delegated to utils.parsing.dates and place fields to utils.parsing.locations.

Fields handled

Příčina úmrtí → cause (primary) + cause_&lt;CODE&gt; dummies + cause_uncertain Bojiště → theatre + in_captivity Příslušnost k útvaru rakousko-uherské armády → ah_unit_primary, ah_unit_type (arm), ah_unit_org (Common army / Landwehr / Honvéd / Landsturm), ah_regiment_no, ah_n_units Poznámka → note_legion (+ note_legion_ref), note_register_ref, note_declared_dead (+ note_dead_ref), note_rank, note_disease Místo narození / Domovem → birth_* / home_* via utils.parsing.locations.add_location_columns

Design mirrors utils.parsing.exit_codes: canonical code dicts, a primary code plus boolean dummies, and a *_report diagnostic. All keyword matching is case-insensitive (the raw field mixes Zemřel/zemřel) and stem-based so gendered variants fold together (Zemřela→DIS, Padla→KIA, Nezvěstná→MIS).

Usage

from utils.parsing.habsburg import add_habsburg_columns, habsburg_report
df = add_habsburg_columns(df)     # dates + cause + theatre + Habsburg units
habsburg_report(df)

def add_cause_columns(
    df: pd.DataFrame,
    source_col: str = "Příčina úmrtí",
) -> pd.DataFrame:
    """Append harmonised cause-of-death columns.

    Adds ``cause`` (primary code), ``cause_label`` (English), ``cause_uncertain``
    (raw contained ``?``), ``cause_codes`` (frozenset of all codes present), and
    one boolean ``cause_<CODE>`` per code in :data:`CAUSES_CS`.

    **A blank cause is ``<NA>``, not ``False``** (v1.2.0, the D-52 rule). ``cause_KIA``
    = False asserts "this man was not killed in action"; on the 8.6k records whose
    ``Příčina úmrtí`` is empty the source says nothing of the kind. A cause that *is*
    recorded but matches no stem keeps ``False`` on every dummy.
    """
    df = df.copy()
    scanned = df[source_col].map(_scan_causes)
    recorded = df[source_col].notna()

    df["cause_codes"] = scanned.map(lambda t: t[0])
    df["cause"] = scanned.map(lambda t: t[1])
    df["cause_uncertain"] = scanned.map(lambda t: t[2]).where(recorded, pd.NA).astype("boolean")
    df["cause_label"] = df["cause"].map(lambda c: CAUSES_EN.get(c) if pd.notna(c) else pd.NA)

    for code in CAUSES_CS:
        hit = df["cause_codes"].map(lambda s, c=code: c in s)
        df[f"cause_{code}"] = hit.where(recorded, pd.NA).astype("boolean")

    return df

def cause_report(df: pd.DataFrame) -> None:
    """Print a frequency table of parsed causes (primary + any-mention)."""
    if "cause_codes" not in df.columns:
        raise ValueError("Run add_cause_columns(df) first.")

    n = len(df)
    print(f"Cause-of-death report  (N = {n:,})\n")
    print(f"  {'Code':<5}  {'English':<26}  {'primary':>9}  {'any':>9}")
    print("  " + "-" * 56)
    for code in CAUSES_CS:
        n_primary = int((df["cause"] == code).sum())
        n_any = int(df[f"cause_{code}"].sum())
        print(f"  {code:<5}  {CAUSES_EN[code]:<26}  {n_primary:>9,}  {n_any:>9,}")

    n_none = int(df["cause"].isna().sum())
    n_unc = int(df["cause_uncertain"].fillna(False).sum())
    print(f"\n  {'—':<5}  {'No cause parsed':<26}  {n_none:>9,}")
    print(f"  {'?':<5}  {'Marked uncertain':<26}  {n_unc:>9,}")

def extract_theatre(raw) -> str:
    """Map a raw *Bojiště* cell to a theatre label (``"Unknown"`` if none)."""
    if pd.isna(raw):
        return "Unknown"
    s = str(raw).lower()
    best: tuple[int, str] | None = None
    for stem, theatre in _THEATRE_KEYWORDS:
        idx = s.find(stem)
        if idx != -1 and (best is None or idx < best[0]):
            best = (idx, theatre)
    return best[1] if best else "Unknown"

def parse_ah_units(raw) -> list[dict]:
    """Split a raw Habsburg unit cell on ``;`` and classify each unit.

    Returns a list of ``{"unit", "type", "org", "number"}`` dicts, one per
    unit (empty list for missing input).
    """
    if pd.isna(raw):
        return []
    parts = [p.strip() for p in str(raw).split(";") if p.strip()]
    out: list[dict] = []
    for p in parts:
        arm, org, number = _classify_unit(p)
        out.append({"unit": p, "type": arm, "org": org, "number": number})
    return out

def add_ah_unit_columns(
    df: pd.DataFrame,
    source_col: str = "Příslušnost k útvaru rakousko-uherské armády",
) -> pd.DataFrame:
    """Append Habsburg unit columns derived from the *primary* (first) unit.

    Adds ``ah_unit_primary`` (raw first unit), ``ah_unit_type`` (arm),
    ``ah_unit_org`` (Common army / Landwehr / Honvéd / Landsturm),
    ``ah_regiment_no`` (int / NA), and ``ah_n_units`` (count of ``;``-joined units).
    """
    df = df.copy()
    parsed = df[source_col].map(parse_ah_units)

    df["ah_n_units"] = parsed.map(len)
    first = parsed.map(lambda lst: lst[0] if lst else None)
    df["ah_unit_primary"] = first.map(lambda d: d["unit"] if d else pd.NA)
    df["ah_unit_type"] = first.map(lambda d: d["type"] if d else pd.NA)
    df["ah_unit_org"] = first.map(lambda d: d["org"] if d else pd.NA)
    df["ah_regiment_no"] = first.map(lambda d: d["number"] if d else pd.NA)

    return df

def habsburg_selftest() -> list[str]:
    """Return the note-parsing behaviours that regressed; empty means all hold."""
    fails: list[str] = []
    for raw, field, expected in _SELFTEST_NOTES:
        got = _scan_note(raw)[field]
        if got != expected:
            fails.append(f"{raw!r}: {field} = {got!r}, expected {expected!r}")
    return fails

def add_note_columns(
    df: pd.DataFrame,
    source_col: str = "Poznámka",
) -> pd.DataFrame:
    """Append structured columns extracted from the free-text note field.

    Adds ``note_legion`` (bool) + ``note_legion_ref`` (matched phrase),
    ``note_register_ref`` (the interwar Czechoslovak unit whose register lists the man —
    provenance), ``note_declared_dead`` (bool) + ``note_dead_ref`` (file no.),
    ``note_rank`` (canonical Czech rank term), and ``note_disease`` (English cause detail).
    The booleans mean "the note says so": they are ``False`` where there is no note.
    """
    df = df.copy()
    if source_col not in df.columns:
        return df

    scanned = df[source_col].map(_scan_note)
    df["note_legion"]        = scanned.map(lambda d: d["legion"])
    df["note_legion_ref"]    = scanned.map(lambda d: d["legion_ref"])
    df["note_register_ref"]  = scanned.map(lambda d: d["register_ref"])
    df["note_declared_dead"] = scanned.map(lambda d: d["declared_dead"])
    df["note_dead_ref"]      = scanned.map(lambda d: d["dead_ref"])
    df["note_rank"]          = scanned.map(lambda d: d["rank"])
    df["note_disease"]       = scanned.map(lambda d: d["disease"])
    return df

def note_report(df: pd.DataFrame) -> None:
    """Print coverage of the four note-derived signals."""
    if "note_legion" not in df.columns:
        raise ValueError("Run add_note_columns(df) first.")

    n = len(df)
    n_note = int(df["Poznámka"].notna().sum()) if "Poznámka" in df.columns else n
    print(f"Note report  (N = {n:,};  non-null Poznámka = {n_note:,})\n")

        ("note_legion", "Legion cross-link"),
        ("note_declared_dead", "Declared dead"),

    n_dis = int(df["note_disease"].notna().sum())
    print(f"  {'Rank recovered':<22}: {n_rank:>8,}  ({n_rank/n:.1%})")
    print(f"  {'Disease detail':<22}: {n_dis:>8,}  ({n_dis/n:.1%})")

    if n_rank:
        print("\n  Top note ranks:")
        print(df["note_rank"].value_counts().head(10).to_string())

"""
habsburg.py — VÚA fallen-soldier field parsing
==============================================
Field-specific harmonisation for the VÚA *padlí v 1. sv. válce* dump
(``data/raw/habsburg/habsburg_fallen.csv``, ~186k fallen Habsburg soldiers).
Run **after** :mod:`utils.parsing.clean` has normalised placeholders and names.

VÚA-specific fields are parsed here; dates are delegated to
:mod:`utils.parsing.dates` and place fields to :mod:`utils.parsing.locations`.

Fields handled
--------------
``Příčina úmrtí``  → ``cause`` (primary) + ``cause_<CODE>`` dummies + ``cause_uncertain``
``Bojiště``        → ``theatre`` + ``in_captivity``
``Příslušnost k útvaru rakousko-uherské armády``
                   → ``ah_unit_primary``, ``ah_unit_type`` (arm), ``ah_unit_org``
                     (Common army / Landwehr / Honvéd / Landsturm),
                     ``ah_regiment_no``, ``ah_n_units``
``Poznámka``       → ``note_legion`` (+ ``note_legion_ref``), ``note_register_ref``,
                     ``note_declared_dead`` (+ ``note_dead_ref``), ``note_rank``, ``note_disease``
``Místo narození`` / ``Domovem`` → ``birth_*`` / ``home_*`` via
                     :func:`utils.parsing.locations.add_location_columns`

Design mirrors :mod:`utils.parsing.exit_codes`: canonical code dicts, a primary
code plus boolean dummies, and a ``*_report`` diagnostic.  All keyword matching
is **case-insensitive** (the raw field mixes ``Zemřel``/``zemřel``) and stem-based
so gendered variants fold together (``Zemřela``→DIS, ``Padla``→KIA,
``Nezvěstná``→MIS).

Usage
-----
    from utils.parsing.habsburg import add_habsburg_columns, habsburg_report
    df = add_habsburg_columns(df)     # dates + cause + theatre + Habsburg units
    habsburg_report(df)
"""

from __future__ import annotations

import re

import pandas as pd

from utils.parsing.dates import add_date_columns, date_report
from utils.parsing.locations import add_location_columns, location_report

__version__ = "1.2.0"   # Legion note = explicit Legion vocabulary; register units split off (audit 2026-09-22)


# ═══════════════════════════════════════════════════════════════════════════
# 1. Cause of death  (Příčina úmrtí)
# ═══════════════════════════════════════════════════════════════════════════

#: Canonical cause codes → Czech / English labels
CAUSES_CS: dict[str, str] = {
    "KIA": "Padl v boji",
    "DIS": "Zemřel (nemoc / zranění)",
    "MIS": "Nezvěstný",
    "EXE": "Popraven / zastřelen",
    "DRO": "Utonul",
    "SUI": "Sebevražda",
    "CAP": "Zajat",
}

CAUSES_EN: dict[str, str] = {
    "KIA": "Killed in action",
    "DIS": "Died (illness / wounds)",
    "MIS": "Missing",
    "EXE": "Executed / shot",
    "DRO": "Drowned",
    "SUI": "Suicide",
    "CAP": "Captured",
}

# Lower-case stem → code.  Stems chosen so gendered/declined forms fold in
# (``padl`` ⊂ ``padla``; ``nezvěst`` ⊂ ``nezvěstný``/``nezvěstná``;
# ``zemřel`` ⊂ ``zemřela``).  Order is the tie-break when two stems start at
# the same position; actual primary is the earliest-occurring stem.
_CAUSE_KEYWORDS: list[tuple[str, str]] = [
    ("padl", "KIA"),
    ("nezvěst", "MIS"),
    ("zemřel", "DIS"),
    ("zamřel", "DIS"),   # observed typo of zemřel
    ("zahynul", "DIS"),
    ("popraven", "EXE"),
    ("zastřelen", "EXE"),
    ("oběšen", "EXE"),
    ("utonul", "DRO"),
    ("sebevra", "SUI"),
    ("zajat", "CAP"),
]


def _scan_causes(raw) -> tuple[frozenset[str], object, bool]:
    """Return ``(all_codes, primary_code, uncertain)`` for one raw cause cell."""
    if pd.isna(raw):
        return frozenset(), pd.NA, False

    s = str(raw).lower()
    uncertain = "?" in s

    hits: list[tuple[int, str]] = []
    found: set[str] = set()
    for stem, code in _CAUSE_KEYWORDS:
        idx = s.find(stem)
        if idx != -1:
            found.add(code)
            hits.append((idx, code))

    if not hits:
        return frozenset(), pd.NA, uncertain

    hits.sort(key=lambda t: t[0])
    return frozenset(found), hits[0][1], uncertain


def parse_cause(raw) -> object:
    """Return the primary canonical cause code for a raw *Příčina úmrtí* cell."""
    return _scan_causes(raw)[1]


def add_cause_columns(
    df: pd.DataFrame,
    source_col: str = "Příčina úmrtí",
) -> pd.DataFrame:
    """Append harmonised cause-of-death columns.

    Adds ``cause`` (primary code), ``cause_label`` (English), ``cause_uncertain``
    (raw contained ``?``), ``cause_codes`` (frozenset of all codes present), and
    one boolean ``cause_<CODE>`` per code in :data:`CAUSES_CS`.

    **A blank cause is ``<NA>``, not ``False``** (v1.2.0, the D-52 rule). ``cause_KIA``
    = False asserts "this man was not killed in action"; on the 8.6k records whose
    ``Příčina úmrtí`` is empty the source says nothing of the kind. A cause that *is*
    recorded but matches no stem keeps ``False`` on every dummy.
    """
    df = df.copy()
    scanned = df[source_col].map(_scan_causes)
    recorded = df[source_col].notna()

    df["cause_codes"] = scanned.map(lambda t: t[0])
    df["cause"] = scanned.map(lambda t: t[1])
    df["cause_uncertain"] = scanned.map(lambda t: t[2]).where(recorded, pd.NA).astype("boolean")
    df["cause_label"] = df["cause"].map(lambda c: CAUSES_EN.get(c) if pd.notna(c) else pd.NA)

    for code in CAUSES_CS:
        hit = df["cause_codes"].map(lambda s, c=code: c in s)
        df[f"cause_{code}"] = hit.where(recorded, pd.NA).astype("boolean")

    return df


def cause_report(df: pd.DataFrame) -> None:
    """Print a frequency table of parsed causes (primary + any-mention)."""
    if "cause_codes" not in df.columns:
        raise ValueError("Run add_cause_columns(df) first.")

    n = len(df)
    print(f"Cause-of-death report  (N = {n:,})\n")
    print(f"  {'Code':<5}  {'English':<26}  {'primary':>9}  {'any':>9}")
    print("  " + "-" * 56)
    for code in CAUSES_CS:
        n_primary = int((df["cause"] == code).sum())
        n_any = int(df[f"cause_{code}"].sum())
        print(f"  {code:<5}  {CAUSES_EN[code]:<26}  {n_primary:>9,}  {n_any:>9,}")

    n_none = int(df["cause"].isna().sum())
    n_unc = int(df["cause_uncertain"].fillna(False).sum())
    print(f"\n  {'—':<5}  {'No cause parsed':<26}  {n_none:>9,}")
    print(f"  {'?':<5}  {'Marked uncertain':<26}  {n_unc:>9,}")


# ═══════════════════════════════════════════════════════════════════════════
# 2. Theatre  (Bojiště)
# ═══════════════════════════════════════════════════════════════════════════

# Lower-case stem → theatre label (English, sharing the legionnaire label
# space where they overlap).  More specific fronts are listed first so they
# win on tie; the primary theatre is the earliest-occurring stem otherwise.
_THEATRE_KEYWORDS: list[tuple[str, str]] = [
    ("makedon", "Macedonia"),
    ("rumun", "Romania"),
    ("francouz", "France"),
    ("ital", "Italy"),
    ("srb", "Serbia"),
    ("rus", "Russia"),
    ("zápol", "Rear"),
]

THEATRES = ["Russia", "Italy", "Serbia", "Macedonia", "Romania", "France", "Rear", "Unknown"]


def extract_theatre(raw) -> str:
    """Map a raw *Bojiště* cell to a theatre label (``"Unknown"`` if none)."""
    if pd.isna(raw):
        return "Unknown"
    s = str(raw).lower()
    best: tuple[int, str] | None = None
    for stem, theatre in _THEATRE_KEYWORDS:
        idx = s.find(stem)
        if idx != -1 and (best is None or idx < best[0]):
            best = (idx, theatre)
    return best[1] if best else "Unknown"


def _is_captivity(raw) -> object:
    """True if the raw *Bojiště* cell mentions captivity (``v zajetí``); ``<NA>`` if blank.

    A blank battlefield cell says nothing about captivity, so it is not evidence of
    *not* dying in captivity (v1.2.0, the D-52 rule).
    """
    if pd.isna(raw):
        return pd.NA
    return "zajet" in str(raw).lower() or "zajat" in str(raw).lower()


def add_theatre_column(
    df: pd.DataFrame,
    source_col: str = "Bojiště",
) -> pd.DataFrame:
    """Append ``theatre`` (label) and ``in_captivity`` (boolean, ``<NA>`` if blank) columns."""
    df = df.copy()
    df["theatre"] = df[source_col].map(extract_theatre)
    df["in_captivity"] = df[source_col].map(_is_captivity).astype("boolean")
    return df


def theatre_report(df: pd.DataFrame) -> None:
    """Print the theatre composition and captivity share."""
    if "theatre" not in df.columns:
        raise ValueError("Run add_theatre_column(df) first.")

    n = len(df)
    print(f"Theatre report  (N = {n:,})\n")
    counts = df["theatre"].value_counts()
    print(f"  {'Theatre':<12}  {'n':>9}  {'%':>6}")
    print("  " + "-" * 32)
    for theatre in THEATRES:
        c = int(counts.get(theatre, 0))
        print(f"  {theatre:<12}  {c:>9,}  {c/n:>5.1%}")

    n_cap = int(df["in_captivity"].fillna(False).sum())
    print(f"\n  In captivity (v zajetí): {n_cap:,}  ({n_cap/n:.1%})")


# ═══════════════════════════════════════════════════════════════════════════
# 3. Austro-Hungarian unit  (Příslušnost k útvaru rakousko-uherské armády)
# ═══════════════════════════════════════════════════════════════════════════
#
# Two orthogonal dimensions are extracted:
#   * arm/branch (``ah_unit_type``) — Infantry / Artillery / Cavalry / …
#   * organisation (``ah_unit_org``) — k.u.k. Common army, k.k. Landwehr
#     (zeměbrana), k.u. Honvéd, or Landsturm (domobrana).
# The full-word Czech vocabulary differs entirely from the abbreviation-based
# legionnaire field, so this is independent of :mod:`utils.parsing.units`.

# Arm keywords, specific arms before the dominant "pěší" so e.g.
# "zeměbranecký pěší pluk" → Infantry but "horský dělostřelecký pluk" → Artillery.
_ARM_KEYWORDS: list[tuple[str, str]] = [
    ("dělostřel", "Artillery"),
    ("houfnic", "Artillery"),
    ("baterie", "Artillery"),
    ("střeleck", "Rifle"),
    ("myslivec", "Jäger"),
    ("myslivc", "Jäger"),
    ("dragoun", "Cavalry"),
    ("husar", "Cavalry"),
    ("hulán", "Cavalry"),
    ("ulán", "Cavalry"),
    ("jezdeck", "Cavalry"),
    ("sanit", "Medical"),
    ("zdravot", "Medical"),
    ("zákop", "Engineer"),
    ("telegraf", "Engineer"),
    ("železnič", "Engineer"),
    ("technick", "Engineer"),
    ("automobil", "Engineer"),
    ("pevnost", "Fortress"),
    ("námoř", "Navy"),
    ("letec", "Aviation"),
    ("letad", "Aviation"),
    ("pracovní", "Labour"),
    ("muniční", "Labour"),
    ("vozataj", "Train"),
    ("pěší", "Infantry"),
    ("pěch", "Infantry"),
]

# Generic unit nouns used as a fallback "Other unit" classification.
_GENERIC_UNIT_NOUNS = (
    "pluk", "prapor", "oddíl", "rota", "setnina", "baterie",
    "divize", "velitelství", "stanice",
)

# Organisation keywords (recruitment base / army branch).
_ORG_KEYWORDS: list[tuple[str, str]] = [
    ("honvéd", "Honvéd"),
    ("zeměbran", "Landwehr"),
    ("domobran", "Landsturm"),
]

_NUM_RE = re.compile(r"\d+")


def _classify_unit(text: str) -> tuple[object, object, object]:
    """Return ``(arm, org, number)`` for a single unit phrase."""
    s = text.lower()

    arm: object = pd.NA
    for stem, label in _ARM_KEYWORDS:
        if stem in s:
            arm = label
            break
    if arm is pd.NA and any(n in s for n in _GENERIC_UNIT_NOUNS):
        arm = "Other unit"

    # Organisation is detected independently of arm, so a bare "Domobrana" or
    # "domobranecké okresní velitelství" still resolves to Landsturm.
    org: object = pd.NA
    for stem, label in _ORG_KEYWORDS:
        if stem in s:
            org = label
            break
    if org is pd.NA and arm is not pd.NA:
        org = "Common army"

    m = _NUM_RE.search(s)
    number: object = int(m.group(0)) if m else pd.NA

    return arm, org, number


def parse_ah_units(raw) -> list[dict]:
    """Split a raw Habsburg unit cell on ``;`` and classify each unit.

    Returns a list of ``{"unit", "type", "org", "number"}`` dicts, one per
    unit (empty list for missing input).
    """
    if pd.isna(raw):
        return []
    parts = [p.strip() for p in str(raw).split(";") if p.strip()]
    out: list[dict] = []
    for p in parts:
        arm, org, number = _classify_unit(p)
        out.append({"unit": p, "type": arm, "org": org, "number": number})
    return out


def add_ah_unit_columns(
    df: pd.DataFrame,
    source_col: str = "Příslušnost k útvaru rakousko-uherské armády",
) -> pd.DataFrame:
    """Append Habsburg unit columns derived from the *primary* (first) unit.

    Adds ``ah_unit_primary`` (raw first unit), ``ah_unit_type`` (arm),
    ``ah_unit_org`` (Common army / Landwehr / Honvéd / Landsturm),
    ``ah_regiment_no`` (int / NA), and ``ah_n_units`` (count of ``;``-joined units).
    """
    df = df.copy()
    parsed = df[source_col].map(parse_ah_units)

    df["ah_n_units"] = parsed.map(len)
    first = parsed.map(lambda lst: lst[0] if lst else None)
    df["ah_unit_primary"] = first.map(lambda d: d["unit"] if d else pd.NA)
    df["ah_unit_type"] = first.map(lambda d: d["type"] if d else pd.NA)
    df["ah_unit_org"] = first.map(lambda d: d["org"] if d else pd.NA)
    df["ah_regiment_no"] = first.map(lambda d: d["number"] if d else pd.NA)

    return df


def ah_unit_report(df: pd.DataFrame, top_n: int = 15) -> None:
    """Print arm / organisation composition and the most common units."""
    if "ah_unit_type" not in df.columns:
        raise ValueError("Run add_ah_unit_columns(df) first.")

    n = len(df)
    print(f"Habsburg unit report  (N = {n:,})\n")

    print("  By arm (ah_unit_type):")
    print(df["ah_unit_type"].value_counts(dropna=False).to_string())
    print("\n  By organisation (ah_unit_org):")
    print(df["ah_unit_org"].value_counts(dropna=False).to_string())

    n_multi = int((df["ah_n_units"] >= 2).sum())
    print(f"\n  Records with ≥2 units: {n_multi:,}  ({n_multi/n:.1%})")

    print(f"\n  Top {top_n} primary units:")
    print(df["ah_unit_primary"].value_counts().head(top_n).to_string())

# 4. Free-text note enrichment  (Poznámka)
# ═══════════════════════════════════════════════════════════════════════════
#
# The note field is mostly archival provenance, but a minority of records carry
# four analytically useful signals, surfaced here as structured columns.

# Legion cross-link: an **explicit** mention of Czechoslovak-Legion service ties this
# fallen Habsburg soldier to the legionnaire population (double-desertion design). The
# capture group keeps the matched phrase as a reference.
#
# Until v1.2.0 any "čs. … pluk/prapor N" counted, and that phrase is overwhelmingly
# **archival provenance**, not service: the interwar Czechoslovak army's regiments kept
# the death registers of their recruiting region, so a note reads "náhradní prapor čs.
# pěšího pluku 33" or "Opsáno na podkladě: úmrtní knihy, … čs. pěšího pluku 33" for a man
# who fell in Serbia in September 1914. 10,509 of the 11,042 old flags were that, and the
# flagged men died at a median of 1915 — before any Czechoslovak unit existed. Tested
# directly: those records match a legionnaire on name+year+okres at 2.9 %, *below* the
# 4.8 % of records with no note at all (audit 2026-09-22 §A2). "Seznam vozatajského pluku
# 8" is the same kind of list for the k.u.k. train regiment, and "Sibiř" marks POWs.
#
# What stays is vocabulary that can only mean the Legions: legionář/legie, KLEG and
# "Kancelář legií" (the Legion records office, previously missed as KLEG), "čs.
# dobrovolník", rota Nazdar, Česká družina, and the twelve Legion rifle regiments of
# Russia ("čs. střelecký pluk 1–12"). A higher number, or a page reference ("čs.
# střelecký pluk 8, str. 7" on a man who died on 28 July 1914), is a register list again
# and is refused. Word-start anchored.
_WS = r"(?<![^\W\d_])"
_LEGION_RE = re.compile(
    _WS + r"(legion\w*"
    r"|kancelář\s+legií"
    r"|legi[eií](?![^\W\d_])"
    r"|kleg(?![^\W\d_])"
    r"|čs\.?\s*dobrovol\w*"
    r"|rot\w*\s+nazdar"
    r"|česk\w*\s+družin\w*"
    r"|čs\.?\s*střel\w*\.?\s*(?:pluk\w*|pl\.)"
    r"(?:\s*(?:č\.?\s*)?(?:1[0-2]|[1-9]|\?)(?!\d)(?!\s*,?\s*str\.)"   # a number 1-12 or "?"
    r"|(?!\s*(?:č\.?\s*)?\d)))",                                       # or no number at all
    re.IGNORECASE,
)

# The interwar Czechoslovak unit whose register lists the man — provenance, kept as its
# own column so the information the old flag carried is not thrown away with the error.
_REGISTER_RE = re.compile(
    _WS + r"((?:náhradn\w*\s+(?:prapor\w*|rot\w*)\s+)?"
    r"čs\.?\s*(?:pěš\w*|horsk\w*|hraničářsk\w*|jezdeck\w*|dělostř\w*)\s*"
    r"(?:pluk\w*|prapor\w*)\s*(?:č\.?\s*)?\d+"
    r"|(?:\d+\.\s*)?vozatajsk\w*\s+pluk\w*(?:\s*\d+)?)",
    re.IGNORECASE,
)

# Legal death declaration ("prohlášen za mrtvého") + its file reference.
_DEAD_RE = re.compile(r"prohláš\w*\s+za\s+mrtv", re.IGNORECASE)
_REF_RE = re.compile(r"(\d{3,}/\d{2})")

# Rank terms in priority order (the first term present wins). Each must start a word
# (v1.2.0): as a bare substring "střelec" was found inside "dělostřelec" (428 notes
# returned the wrong term) and "major" inside "generálmajor" — the shape `ranks` 1.4.0
# removed. Matched term is returned as-is.
RANK_TERMS: list[str] = [
    "nadporučík", "podporučík", "poručík",
    "nadrotmistr", "štábní rotmistr", "rotmistr",
    "podplukovník", "plukovník",
    "štábní kapitán", "kapitán", "setník", "generálmajor", "major", "generál",
    "praporčík", "šikovatel", "četař", "desátník", "svobodník",
    "poddůstojník", "důstojník",
    "dělostřelec", "střelec", "jezdec", "vojín",
]
_RANK_TERM_RE = {t: re.compile(_WS + re.escape(t)) for t in RANK_TERMS}

# Specific illness/cause detail → English canonical label.
_DISEASE_TERMS: list[tuple[str, str]] = [
    ("tyfus", "typhus"), ("skvrnit", "typhus"),
    ("úplavic", "dysentery"), ("enteri", "dysentery"),
    ("cholera", "cholera"),
    ("tuberk", "tuberculosis"), ("tbc", "tuberculosis"), ("souchot", "tuberculosis"),
    ("malár", "malaria"),
    ("zápal plic", "pneumonia"), ("pneumon", "pneumonia"),
    ("chřipk", "influenza"), ("španěl", "influenza"),
    ("úraz", "accident"), ("utonul", "drowning"),
]


def _scan_note(raw) -> dict:
    """Extract the four note signals from one raw *Poznámka* cell."""
    if pd.isna(raw):
        return {"legion": False, "legion_ref": pd.NA, "register_ref": pd.NA,
                "declared_dead": False, "dead_ref": pd.NA, "rank": pd.NA, "disease": pd.NA}

    s = str(raw)
    low = s.lower()

    m = _LEGION_RE.search(s)
    legion = bool(m)
    legion_ref = m.group(0).strip() if m else pd.NA
    mr_ = _REGISTER_RE.search(s)
    register_ref = mr_.group(0).strip() if mr_ else pd.NA

    declared_dead = bool(_DEAD_RE.search(low))
    dead_ref = pd.NA
    if declared_dead:
        mr = _REF_RE.search(s)
        dead_ref = mr.group(1) if mr else pd.NA

    rank: object = pd.NA
    for term in RANK_TERMS:
        if _RANK_TERM_RE[term].search(low):
            rank = term
            break

    disease: object = pd.NA
    for stem, label in _DISEASE_TERMS:
        if stem in low:
            disease = label
            break

    return {"legion": legion, "legion_ref": legion_ref, "register_ref": register_ref,
            "declared_dead": declared_dead, "dead_ref": dead_ref,
            "rank": rank, "disease": disease}


# One case per behaviour that shipped wrong once; run by the QA gate.
_SELFTEST_NOTES: tuple[tuple[str, str, object], ...] = (
    # archival provenance is not Legion service (audit 2026-09-22 §A2)

    # word-anchored rank terms
    ("dělostřelec", "rank", "dělostřelec"),
    ("generálmajor", "rank", "generálmajor"),

def habsburg_selftest() -> list[str]:
    """Return the note-parsing behaviours that regressed; empty means all hold."""
    fails: list[str] = []
    for raw, field, expected in _SELFTEST_NOTES:
        got = _scan_note(raw)[field]
        if got != expected:
            fails.append(f"{raw!r}: {field} = {got!r}, expected {expected!r}")
    return fails


def add_note_columns(
    df: pd.DataFrame,
    source_col: str = "Poznámka",
) -> pd.DataFrame:
    """Append structured columns extracted from the free-text note field.

    Adds ``note_legion`` (bool) + ``note_legion_ref`` (matched phrase),
    ``note_register_ref`` (the interwar Czechoslovak unit whose register lists the man —
    provenance), ``note_declared_dead`` (bool) + ``note_dead_ref`` (file no.),
    ``note_rank`` (canonical Czech rank term), and ``note_disease`` (English cause detail).
    The booleans mean "the note says so": they are ``False`` where there is no note.
    """
    df = df.copy()
    if source_col not in df.columns:
        return df

    scanned = df[source_col].map(_scan_note)
    df["note_legion"]        = scanned.map(lambda d: d["legion"])
    df["note_legion_ref"]    = scanned.map(lambda d: d["legion_ref"])
    df["note_register_ref"]  = scanned.map(lambda d: d["register_ref"])
    df["note_declared_dead"] = scanned.map(lambda d: d["declared_dead"])
    df["note_dead_ref"]      = scanned.map(lambda d: d["dead_ref"])
    df["note_rank"]          = scanned.map(lambda d: d["rank"])
    df["note_disease"]       = scanned.map(lambda d: d["disease"])
    return df


def note_report(df: pd.DataFrame) -> None:
    """Print coverage of the four note-derived signals."""
    if "note_legion" not in df.columns:
        raise ValueError("Run add_note_columns(df) first.")

    n = len(df)
    n_note = int(df["Poznámka"].notna().sum()) if "Poznámka" in df.columns else n
    print(f"Note report  (N = {n:,};  non-null Poznámka = {n_note:,})\n")

    for col, lab in [
        ("note_legion", "Legion cross-link"),
        ("note_declared_dead", "Declared dead"),
    ]:
        c = int(df[col].sum())
        print(f"  {lab:<22}: {c:>8,}  ({c/n:.1%})")
    if "note_register_ref" in df.columns:
        c = int(df["note_register_ref"].notna().sum())
        print(f"  {'ČSA register (provenance)':<22}: {c:>8,}  ({c/n:.1%})")

    n_rank = int(df["note_rank"].notna().sum())
    n_dis = int(df["note_disease"].notna().sum())
    print(f"  {'Rank recovered':<22}: {n_rank:>8,}  ({n_rank/n:.1%})")
    print(f"  {'Disease detail':<22}: {n_dis:>8,}  ({n_dis/n:.1%})")

    if n_rank:
        print("\n  Top note ranks:")
        print(df["note_rank"].value_counts().head(10).to_string())

def add_habsburg_columns(df: pd.DataFrame) -> pd.DataFrame:
    """Run the full VÚA parsing pipeline.

    Order: dates → cause → theatre → Habsburg units → notes → birth/home locations.
    Returns a copy of *df* with all derived columns appended.  Expects a
    DataFrame already passed through :func:`utils.parsing.clean.clean_dataframe`.
    """
    df = add_date_columns(df)
    df = add_cause_columns(df)
    df = add_theatre_column(df)
    df = add_ah_unit_columns(df)
    df = add_note_columns(df)
    df = add_location_columns(df, "Místo narození", prefix="birth")
    df = add_location_columns(df, "Domovem", prefix="home")
    return df


def habsburg_report(df: pd.DataFrame) -> None:
    """Print all VÚA parsing diagnostics in sequence."""
    sep = "\n" + "=" * 64 + "\n"
    date_report(df)
    print(sep); cause_report(df)
    print(sep); theatre_report(df)
    print(sep); ah_unit_report(df)
    print(sep); note_report(df)
    print(sep); location_report(df, "birth")
    print(sep); location_report(df, "home")
