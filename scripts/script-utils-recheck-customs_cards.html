How it works

Customs (Zollamt / celní úřad) and the revenue office (Finanzwach-Kontrollbezirksleitung / správa kontrolního okresu finanční stráže) are one glyph in both keys, a circle on a hook: a dot inside for customs, empty for the revenue office (hatched is the revenue guard; the same glyphs in brackets are the cadastre office and technical revenue control). The two editions agree on 0.78 (customs) and 0.86 (revenue office) of the municipalities both read, against 0.96-0.99 for the other common institutions, and the disagreements run both ways, so the Czech volume cannot judge them (unlike the castle and poorhouse, D-190). The author rules them by eye.

One card per one-to-one paired municipality (the pairing of check_offices_coverage.bohemia_editions) where either symbol is held by one edition and not the other, both editions holding some pictogram there. The card shows each edition's entry, cut from the full run's section images around the boxes the readers placed for that municipality, with no outline: nothing on it says which edition holds what. Pages held for certification, reserve or calibration are left out (D-133, D-173).

Reads data/final/register_rechecked/register_places.parquet, output/recheck/{cz,de-bohemia}/ final_inventory_rechecked.csv and output/frozen/{cz,de-bohemia}_full/sections/; writes output/recheck/customs_rulings/ (key.csv, cards.json, sections/). Run from the repo root: python -m utils.recheck.customs_cards.

def pairs(register: Path = REGISTER) -> pd.DataFrame:
    """German Böhmen municipalities paired one to one with a Czech one (as the editions check pairs them)."""
    places = pd.read_parquet(register)
    places["court_key"] = places.court_key.astype(str)
    munis = places[places.place_level == "municipality"]
    de = munis[(munis.edition == "de-bohemia") & munis.cz_muni_seq.notna()]
    out = pd.DataFrame({"court_key": de.court_key, "de": de.muni_seq.astype(int),
                        "cz": de.cz_muni_seq.astype(int), "de_name": de.printed_name})
    one = ~out.duplicated(["court_key", "cz"], keep=False) & ~out.duplicated(["court_key", "de"], keep=False)
    out = out[one]
    cz = munis[munis.edition == "cz"][["court_key", "muni_seq", "printed_name"]].drop_duplicates(["court_key", "muni_seq"])
    cz = cz.rename(columns={"muni_seq": "cz", "printed_name": "cz_name"}).astype({"cz": int})
    merged = out.merge(cz, on=["court_key", "cz"], how="left", validate="1:1")
    assert len(merged) == len(out)
    return merged

def disagreements(register: Path = REGISTER, recheck: Path = RECHECK, include: frozenset = frozenset(),
                  codes: tuple = PAIR) -> list[dict]:
    """``[{court_key, de, cz, names, de_rows, cz_rows, holds}]`` where the pair differs between the editions."""
    inv = {side: inventory(ed, recheck) for side, ed in EDITIONS.items()}
    groups = {side: dict(tuple(d.groupby(["court_key", "muni"]))) for side, d in inv.items()}
    out = []
    for r in pairs(register).itertuples():
        rows = {"de": groups["de"].get((r.court_key, r.de)), "cz": groups["cz"].get((r.court_key, r.cz))}
        if rows["de"] is None or rows["cz"] is None:
            continue                                       # one edition holds nothing there: not judged
        holds = {side: sorted(set(rows[side].code) & set(codes)) for side in rows}
        if holds["de"] == holds["cz"] and f"cr-{r.court_key}-{r.de}-{r.cz}" not in include:
            continue      # a card named in ``include`` is drawn though the tables agree (a ruling to make again)
        out.append({"court_key": r.court_key, "de": r.de, "cz": r.cz, "de_name": r.de_name,
                    "cz_name": r.cz_name, "rows": rows, "holds": holds})
    return out

def boxes(rows: pd.DataFrame) -> dict[str, list[tuple]]:
    """``{section image: [boxes]}`` the readers placed for every claim of the municipality."""
    out = defaultdict(list)
    for marks in rows.marks:
        for mark in str(marks).split(" | "):
            image, sep, box = mark.partition("#")
            if sep and image.endswith(".png"):
                try:
                    out[image].append(tuple(int(float(v)) for v in box.split(",")))
                except ValueError:
                    continue
    return dict(out)

def band(edition: str, name: str, placed: list[tuple]):
    """The rows of section image ``name`` around ``placed``, with a neutral bar in the left margin
    over the rows the route filed under this number."""
    from PIL import Image, ImageDraw

    scan = Image.open(FROZEN / f"{edition}_full" / "sections" / name).convert("RGB")
    line = sorted(b[3] - b[1] for b in placed)[len(placed) // 2]
    first, last = min(b[1] for b in placed), max(b[3] for b in placed)
    top, foot = max(0, first - LINES_AROUND * line), min(scan.height, last + LINES_AROUND * line)
    cut = scan.crop((0, top, scan.width, foot))
    ImageDraw.Draw(cut).rectangle((0, first - top, 10, last - top), fill=(70, 110, 170))
    return _scaled(cut, BAND_WIDTH)

def sections_of(edition: str, rows: pd.DataFrame) -> list[str]:
    """The offices section images of every (page, district) the municipality's claims sit on."""
    folder = FROZEN / f"{edition}_full" / "sections"
    names = set()
    for page, district in set(zip(rows.page, rows.district)):
        found = {p.name for p in folder.glob(f"{page}_{district}_offices_*.png")}
        # a missed offices heading (§4's regions, Leitmeritz 0247): the lines are in the parts images
        names |= found or {p.name for p in folder.glob(f"{page}_{district}_parts_*.png")}
    return sorted(names)

def build(out: Path = OUT, register: Path = REGISTER, recheck: Path = RECHECK,
          ruled: frozenset = frozenset(), include: frozenset = frozenset(), codes: tuple = PAIR,
          prefix: str = "cr") -> list[dict]:
    """The cards, leaving out those on held pages and the ``ruled`` card ids (a page already ruled);
    ``include``: cards drawn though the two tables agree (0.2.1)."""
    held = {side_: bs.held_out(ed) for side_, ed in EDITIONS.items()}
    cards, key, left_out = [], [], []
    (out / "sections").mkdir(parents=True, exist_ok=True)
    wanted = defaultdict(set)
    for d in disagreements(register, recheck, include, codes):
        pages = {s: set(d["rows"][s].page) for s in EDITIONS}
        if any(pages[s] & held[s] for s in EDITIONS):
            left_out.append(d)
            continue
        card_id = f"{prefix}-{d['court_key']}-{d['de']}-{d['cz']}"
        if card_id in ruled:
            continue
        sides = {s: side(EDITIONS[s], d["rows"][s], d[s], d[f"{s}_name"]) for s in EDITIONS}
        for s in EDITIONS:
            wanted[EDITIONS[s]] |= set(sides[s]["sections"])
        cards.append({"id": card_id, **sides})
        key.append({"card": card_id, "court_key": d["court_key"], "de_muni": d["de"], "cz_muni": d["cz"],
                    "de_name": d["de_name"], "cz_name": d["cz_name"],
                    "de_pages": ";".join(sides["de"]["pages"]), "cz_pages": ";".join(sides["cz"]["pages"]),
                    "de_holds": " ".join(d["holds"]["de"]), "cz_holds": " ".join(d["holds"]["cz"])})
    from PIL import Image
    for edition, names in wanted.items():
        for name in names:
            target = out / "sections" / f"{edition}_{Path(name).stem}.webp"
            if not target.exists():
                image = Image.open(FROZEN / f"{edition}_full" / "sections" / name).convert("RGB")
                target.write_bytes(_webp(_scaled(image, SECTION_WIDTH), 68))
    order = sorted(range(len(cards)), key=lambda i: (cards[i]["de"]["pages"][0], int(cards[i]["de"]["muni"])))
    cards = [cards[i] for i in order]
    pd.DataFrame(key).to_csv(out / "key.csv", index=False)
    (out / "cards.json").write_text(json.dumps(cards), encoding="utf-8")
    print(f"customs_cards {__version__}: {len(cards)} cards, {len(left_out)} left out on held pages, "
          f"{sum(len(v) for v in wanted.values())} section images")
    return cards

def single_candidates(edition: str, recheck: Path = RECHECK) -> list[dict]:
    """``[{court, district, muni, rows, hits, holds, read}]``: each municipality whose table holds
    customs or the revenue office, or where some reader read either (a claim the route dropped)."""
    inv = inventory(edition, recheck)
    inv["court"] = inv.court_key.where(inv.court_key != "", "unresolved:" + inv.district)
    court_of = {(p, d): c for p, d, c in zip(inv.page, inv.district, inv.court)}
    hits = _ledger_hits(edition)
    hits["court"] = [court_of.get((p, d), "unresolved:" + d) for p, d in zip(hits.page, hits.district)]
    keys = set(map(tuple, inv[inv.code.isin(FAMILY)][["court", "district", "muni"]].astype(str).values))
    keys |= set(zip(hits.court, hits.district, hits.muni.astype(int).astype(str)))
    out = []
    for court, district, muni in sorted(keys):
        rows = inv[(inv.court == court) & (inv.district == district) & (inv.muni.astype(str) == muni)]
        mine = hits[(hits.court == court) & (hits.district == district) & (hits.muni.astype(str) == muni)]
        out.append({"court": court, "district": district, "muni": int(muni), "rows": rows, "hits": mine,
                    "holds": sorted(set(rows.code) & set(FAMILY)), "read": sorted(set(mine["read"]))})
    return out

def family_candidates(edition: str, recheck: Path = RECHECK, family: tuple = FAMILY_ALL) -> list[dict]:
    """``single_candidates``' shape for the municipalities where the readers disagreed on a mark of the
    circle-on-hook family: some read one of its six glyphs there, another read something else (0.4.0)."""
    folder = FROZEN / f"{edition}_full" / "ledger"
    ledger = pd.read_csv(folder / f"ledger_{edition}_full.csv", dtype=str, keep_default_na=False)
    ledger = ledger[ledger["step"] == "read"]
    reads = ledger.groupby("mark")["read"].agg(set)
    split = [m for m, rs in reads.items() if rs & set(family) and len(rs) > 1]
    marks = pd.read_csv(folder / f"marks_{edition}_full.csv", dtype=str, keep_default_na=False)
    hits = marks[marks["mark"].isin(split)][["mark", "page", "district", "muni"]].copy()
    hits["read"] = [" / ".join(sorted(reads[m])) for m in hits["mark"]]
    inv = inventory(edition, recheck)
    inv["court"] = inv.court_key.where(inv.court_key != "", "unresolved:" + inv.district)
    court_of = {(p, d): c for p, d, c in zip(inv.page, inv.district, inv.court)}
    hits["court"] = [court_of.get((p, d), "unresolved:" + d) for p, d in zip(hits.page, hits.district)]
    out = []
    for (court, district, muni), mine in hits.groupby(["court", "district", "muni"]):
        rows = inv[(inv.court == court) & (inv.district == district) & (inv.muni.astype(str) == str(int(muni)))]
        out.append({"court": court, "district": district, "muni": int(muni), "rows": rows, "hits": mine,
                    "holds": sorted(set(rows.code) & set(family)), "read": sorted(set(mine["read"]))})
    return out

def build_single(edition: str, out: Path, recheck: Path = RECHECK, candidates: list | None = None,
                 prefix: str = "cs") -> list[dict]:
    """The one-volume cards of ``edition`` (``key.csv``, ``cards.json``, ``sections/`` under ``out``);
    ``candidates`` default to :func:`single_candidates`."""
    from PIL import Image
    held = bs.held_out(edition)
    cards, key, wanted, left_out = [], [], set(), 0
    (out / "sections").mkdir(parents=True, exist_ok=True)
    for c in (candidates if candidates is not None else single_candidates(edition, recheck)):
        both = pd.concat([c["rows"][["page", "district", "marks"]],
                          c["hits"].rename(columns={"mark": "marks"})[["page", "district", "marks"]]])
        if set(both.page) & held:
            left_out += 1
            continue
        placed = boxes(both)
        if any("_offices_" in img for img in placed):       # a place type read in the parts list: not the entry
            placed = {img: bx for img, bx in placed.items() if "_offices_" in img}
        bands = [base64.b64encode(_webp(band(edition, img, bx))).decode() for img, bx in sorted(placed.items())]
        sections = sorted(set(placed) | set(sections_of(edition, both)))
        wanted |= set(sections)
        card_id = f"{prefix}-{SHORT[edition]}-{c['court'].replace(':', '_')}-{c['muni']}"
        cards.append({"id": card_id, "edition": edition, "land": LAND[edition],
                      "de": {"muni": str(c["muni"]), "printed": str(c["muni"] % sk.OFFSET), "name": "",
                             "pages": sorted(set(both.page)), "district": c["district"], "bands": bands,
                             "sections": sections}})
        key.append({"card": card_id, "edition": edition, "court_key": c["court"], "district": c["district"],
                    "muni": c["muni"], "pages": ";".join(sorted(set(both.page))), "holds": " ".join(c["holds"]),
                    "read_by_some_reader": " ".join(c["read"])})
    for name in wanted:
        target = out / "sections" / f"{edition}_{Path(name).stem}.webp"
        if not target.exists():
            image = Image.open(FROZEN / f"{edition}_full" / "sections" / name).convert("RGB")
            target.write_bytes(_webp(_scaled(image, SECTION_WIDTH), 68))
    print(f"customs_cards {__version__} ({edition}): {len(cards)} cards, {left_out} left out on held pages")
    return cards, key

"""The customs / revenue-office cards: German and Czech Böhmen where the two editions disagree.

Customs (*Zollamt / celní úřad*) and the revenue office (*Finanzwach-Kontrollbezirksleitung /
správa kontrolního okresu finanční stráže*) are one glyph in both keys, a circle on a hook: a dot
inside for customs, empty for the revenue office (hatched is the revenue guard; the same glyphs in
brackets are the cadastre office and technical revenue control). The two editions agree on 0.78
(customs) and 0.86 (revenue office) of the municipalities both read, against 0.96-0.99 for the
other common institutions, and the disagreements run both ways, so the Czech volume cannot judge
them (unlike the castle and poorhouse, D-190). The author rules them by eye.

One card per one-to-one paired municipality (the pairing of ``check_offices_coverage.bohemia_editions``)
where either symbol is held by one edition and not the other, both editions holding some pictogram
there. The card shows each edition's entry, cut from the full run's section images around the boxes
the readers placed for that municipality, with no outline: nothing on it says which edition holds
what. Pages held for certification, reserve or calibration are left out (D-133, D-173).

Reads ``data/final/register_rechecked/register_places.parquet``, ``output/recheck/{cz,de-bohemia}/
final_inventory_rechecked.csv`` and ``output/frozen/{cz,de-bohemia}_full/sections/``; writes
``output/recheck/customs_rulings/`` (``key.csv``, ``cards.json``, ``sections/``). Run from the repo root:
``python -m utils.recheck.customs_cards``.
"""
from __future__ import annotations

import base64
import io
import json
import sys
from collections import defaultdict
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from utils.recheck import blind_sample as bs                         # noqa: E402
from utils.recheck import section_keys as sk                         # noqa: E402

__version__ = "0.4.0"   # 0.4.0 the circle-on-hook family: codes for the Böhmen pairs, family_candidates for one volume; 0.3.0 one-volume cards for Mähren and Schlesien (build_single); 0.2.1 include: cards to rule again though the tables agree; 0.2.0 the rechecked register's pairs and keys (D-208), the printed number on each side, cards already ruled left out; 0.1.0 the cards for the customs / revenue-office disagreements between the Böhmen editions

PAIR = ("[customs]", "[revenue-admin]")
EDITIONS = {"de": "de-bohemia", "cz": "cz"}
REGISTER = ROOT / "data" / "final" / "register_rechecked" / "register_places.parquet"   # one number, one place (D-208)
RECHECK = ROOT / "output" / "recheck"
FROZEN = ROOT / "output" / "frozen"
OUT = RECHECK / "customs_rulings"
BAND_WIDTH = 1500        # px: a band is scaled to this width for the page
SECTION_WIDTH = 1300     # px: the whole section image, opened on request
LINES_AROUND = 2         # an entry's band reaches this many line heights above and below its boxes


def pairs(register: Path = REGISTER) -> pd.DataFrame:
    """German Böhmen municipalities paired one to one with a Czech one (as the editions check pairs them)."""
    places = pd.read_parquet(register)
    places["court_key"] = places.court_key.astype(str)
    munis = places[places.place_level == "municipality"]
    de = munis[(munis.edition == "de-bohemia") & munis.cz_muni_seq.notna()]
    out = pd.DataFrame({"court_key": de.court_key, "de": de.muni_seq.astype(int),
                        "cz": de.cz_muni_seq.astype(int), "de_name": de.printed_name})
    one = ~out.duplicated(["court_key", "cz"], keep=False) & ~out.duplicated(["court_key", "de"], keep=False)
    out = out[one]
    cz = munis[munis.edition == "cz"][["court_key", "muni_seq", "printed_name"]].drop_duplicates(["court_key", "muni_seq"])
    cz = cz.rename(columns={"muni_seq": "cz", "printed_name": "cz_name"}).astype({"cz": int})
    merged = out.merge(cz, on=["court_key", "cz"], how="left", validate="1:1")
    assert len(merged) == len(out)
    return merged


def inventory(edition: str, recheck: Path = RECHECK) -> pd.DataFrame:
    d = pd.read_csv(Path(recheck) / edition / "final_inventory_rechecked.csv", dtype=str, keep_default_na=False)
    d["court_key"], d["muni"] = d.court_id.astype(str), d.muni_seq.astype(int)
    return d


def disagreements(register: Path = REGISTER, recheck: Path = RECHECK, include: frozenset = frozenset(),
                  codes: tuple = PAIR) -> list[dict]:
    """``[{court_key, de, cz, names, de_rows, cz_rows, holds}]`` where the pair differs between the editions."""
    inv = {side: inventory(ed, recheck) for side, ed in EDITIONS.items()}
    groups = {side: dict(tuple(d.groupby(["court_key", "muni"]))) for side, d in inv.items()}
    out = []
    for r in pairs(register).itertuples():
        rows = {"de": groups["de"].get((r.court_key, r.de)), "cz": groups["cz"].get((r.court_key, r.cz))}
        if rows["de"] is None or rows["cz"] is None:
            continue                                       # one edition holds nothing there: not judged
        holds = {side: sorted(set(rows[side].code) & set(codes)) for side in rows}
        if holds["de"] == holds["cz"] and f"cr-{r.court_key}-{r.de}-{r.cz}" not in include:
            continue      # a card named in ``include`` is drawn though the tables agree (a ruling to make again)
        out.append({"court_key": r.court_key, "de": r.de, "cz": r.cz, "de_name": r.de_name,
                    "cz_name": r.cz_name, "rows": rows, "holds": holds})
    return out


def boxes(rows: pd.DataFrame) -> dict[str, list[tuple]]:
    """``{section image: [boxes]}`` the readers placed for every claim of the municipality."""
    out = defaultdict(list)
    for marks in rows.marks:
        for mark in str(marks).split(" | "):
            image, sep, box = mark.partition("#")
            if sep and image.endswith(".png"):
                try:
                    out[image].append(tuple(int(float(v)) for v in box.split(",")))
                except ValueError:
                    continue
    return dict(out)


def _webp(image, quality: int = 72) -> bytes:
    buf = io.BytesIO()
    image.save(buf, "WEBP", quality=quality)
    return buf.getvalue()


def _scaled(image, width: int):
    if image.width <= width:
        return image
    return image.resize((width, round(image.height * width / image.width)))


def band(edition: str, name: str, placed: list[tuple]):
    """The rows of section image ``name`` around ``placed``, with a neutral bar in the left margin
    over the rows the route filed under this number."""
    from PIL import Image, ImageDraw

    scan = Image.open(FROZEN / f"{edition}_full" / "sections" / name).convert("RGB")
    line = sorted(b[3] - b[1] for b in placed)[len(placed) // 2]
    first, last = min(b[1] for b in placed), max(b[3] for b in placed)
    top, foot = max(0, first - LINES_AROUND * line), min(scan.height, last + LINES_AROUND * line)
    cut = scan.crop((0, top, scan.width, foot))
    ImageDraw.Draw(cut).rectangle((0, first - top, 10, last - top), fill=(70, 110, 170))
    return _scaled(cut, BAND_WIDTH)


def sections_of(edition: str, rows: pd.DataFrame) -> list[str]:
    """The offices section images of every (page, district) the municipality's claims sit on."""
    folder = FROZEN / f"{edition}_full" / "sections"
    names = set()
    for page, district in set(zip(rows.page, rows.district)):
        found = {p.name for p in folder.glob(f"{page}_{district}_offices_*.png")}
        # a missed offices heading (§4's regions, Leitmeritz 0247): the lines are in the parts images
        names |= found or {p.name for p in folder.glob(f"{page}_{district}_parts_*.png")}
    return sorted(names)


def side(edition: str, rows: pd.DataFrame, muni: int, name: str) -> dict:
    placed = boxes(rows)
    bands = [base64.b64encode(_webp(band(edition, img, bs_))).decode() for img, bs_ in sorted(placed.items())]
    return {"muni": str(muni), "printed": str(muni % sk.OFFSET), "name": name or "", "pages": sorted(set(rows.page)),
            "district": sorted(set(rows.district))[0], "bands": bands,
            "sections": sorted(set(placed) | set(sections_of(edition, rows)))}


def build(out: Path = OUT, register: Path = REGISTER, recheck: Path = RECHECK,
          ruled: frozenset = frozenset(), include: frozenset = frozenset(), codes: tuple = PAIR,
          prefix: str = "cr") -> list[dict]:
    """The cards, leaving out those on held pages and the ``ruled`` card ids (a page already ruled);
    ``include``: cards drawn though the two tables agree (0.2.1)."""
    held = {side_: bs.held_out(ed) for side_, ed in EDITIONS.items()}
    cards, key, left_out = [], [], []
    (out / "sections").mkdir(parents=True, exist_ok=True)
    wanted = defaultdict(set)
    for d in disagreements(register, recheck, include, codes):
        pages = {s: set(d["rows"][s].page) for s in EDITIONS}
        if any(pages[s] & held[s] for s in EDITIONS):
            left_out.append(d)
            continue
        card_id = f"{prefix}-{d['court_key']}-{d['de']}-{d['cz']}"
        if card_id in ruled:
            continue
        sides = {s: side(EDITIONS[s], d["rows"][s], d[s], d[f"{s}_name"]) for s in EDITIONS}
        for s in EDITIONS:
            wanted[EDITIONS[s]] |= set(sides[s]["sections"])
        cards.append({"id": card_id, **sides})
        key.append({"card": card_id, "court_key": d["court_key"], "de_muni": d["de"], "cz_muni": d["cz"],
                    "de_name": d["de_name"], "cz_name": d["cz_name"],
                    "de_pages": ";".join(sides["de"]["pages"]), "cz_pages": ";".join(sides["cz"]["pages"]),
                    "de_holds": " ".join(d["holds"]["de"]), "cz_holds": " ".join(d["holds"]["cz"])})
    from PIL import Image
    for edition, names in wanted.items():
        for name in names:
            target = out / "sections" / f"{edition}_{Path(name).stem}.webp"
            if not target.exists():
                image = Image.open(FROZEN / f"{edition}_full" / "sections" / name).convert("RGB")
                target.write_bytes(_webp(_scaled(image, SECTION_WIDTH), 68))
    order = sorted(range(len(cards)), key=lambda i: (cards[i]["de"]["pages"][0], int(cards[i]["de"]["muni"])))
    cards = [cards[i] for i in order]
    pd.DataFrame(key).to_csv(out / "key.csv", index=False)
    (out / "cards.json").write_text(json.dumps(cards), encoding="utf-8")
    print(f"customs_cards {__version__}: {len(cards)} cards, {len(left_out)} left out on held pages, "
          f"{sum(len(v) for v in wanted.values())} section images")
    return cards


if __name__ == "__main__":
    build()


# --------------------------------------------------------------------------------------------- #
# One volume: Mähren and Schlesien, which have no second edition to disagree with (0.3.0)
# --------------------------------------------------------------------------------------------- #

FAMILY = ("[customs]", "[customs-exp]", "[revenue-admin]")
#: the whole circle-on-hook family: dotted, empty, hatched, each with or without brackets (0.4.0)
FAMILY_ALL = FAMILY + ("[cadastre-office]", "[revenue-guard]", "[revenue-technical]")
BRACKETED = ("[cadastre-office]", "[revenue-guard]", "[revenue-technical]")
LAND = {"de-moravia": "Mähren", "de-silesia": "Schlesien"}
SHORT = {"de-moravia": "mo", "de-silesia": "si"}


def _ledger_hits(edition: str) -> pd.DataFrame:
    """Every mark some reader read as customs or the revenue office, with its place
    (``page, district, muni``), from the run's mark ledger."""
    folder = FROZEN / f"{edition}_full" / "ledger"
    ledger = pd.read_csv(folder / f"ledger_{edition}_full.csv", dtype=str, keep_default_na=False)
    marks = pd.read_csv(folder / f"marks_{edition}_full.csv", dtype=str, keep_default_na=False)
    hit = ledger[ledger["read"].isin(FAMILY)][["mark", "read"]].drop_duplicates()
    return hit.merge(marks[["mark", "page", "district", "muni"]], on="mark", how="inner")


def single_candidates(edition: str, recheck: Path = RECHECK) -> list[dict]:
    """``[{court, district, muni, rows, hits, holds, read}]``: each municipality whose table holds
    customs or the revenue office, or where some reader read either (a claim the route dropped)."""
    inv = inventory(edition, recheck)
    inv["court"] = inv.court_key.where(inv.court_key != "", "unresolved:" + inv.district)
    court_of = {(p, d): c for p, d, c in zip(inv.page, inv.district, inv.court)}
    hits = _ledger_hits(edition)
    hits["court"] = [court_of.get((p, d), "unresolved:" + d) for p, d in zip(hits.page, hits.district)]
    keys = set(map(tuple, inv[inv.code.isin(FAMILY)][["court", "district", "muni"]].astype(str).values))
    keys |= set(zip(hits.court, hits.district, hits.muni.astype(int).astype(str)))
    out = []
    for court, district, muni in sorted(keys):
        rows = inv[(inv.court == court) & (inv.district == district) & (inv.muni.astype(str) == muni)]
        mine = hits[(hits.court == court) & (hits.district == district) & (hits.muni.astype(str) == muni)]
        out.append({"court": court, "district": district, "muni": int(muni), "rows": rows, "hits": mine,
                    "holds": sorted(set(rows.code) & set(FAMILY)), "read": sorted(set(mine["read"]))})
    return out


def family_candidates(edition: str, recheck: Path = RECHECK, family: tuple = FAMILY_ALL) -> list[dict]:
    """``single_candidates``' shape for the municipalities where the readers disagreed on a mark of the
    circle-on-hook family: some read one of its six glyphs there, another read something else (0.4.0)."""
    folder = FROZEN / f"{edition}_full" / "ledger"
    ledger = pd.read_csv(folder / f"ledger_{edition}_full.csv", dtype=str, keep_default_na=False)
    ledger = ledger[ledger["step"] == "read"]
    reads = ledger.groupby("mark")["read"].agg(set)
    split = [m for m, rs in reads.items() if rs & set(family) and len(rs) > 1]
    marks = pd.read_csv(folder / f"marks_{edition}_full.csv", dtype=str, keep_default_na=False)
    hits = marks[marks["mark"].isin(split)][["mark", "page", "district", "muni"]].copy()
    hits["read"] = [" / ".join(sorted(reads[m])) for m in hits["mark"]]
    inv = inventory(edition, recheck)
    inv["court"] = inv.court_key.where(inv.court_key != "", "unresolved:" + inv.district)
    court_of = {(p, d): c for p, d, c in zip(inv.page, inv.district, inv.court)}
    hits["court"] = [court_of.get((p, d), "unresolved:" + d) for p, d in zip(hits.page, hits.district)]
    out = []
    for (court, district, muni), mine in hits.groupby(["court", "district", "muni"]):
        rows = inv[(inv.court == court) & (inv.district == district) & (inv.muni.astype(str) == str(int(muni)))]
        out.append({"court": court, "district": district, "muni": int(muni), "rows": rows, "hits": mine,
                    "holds": sorted(set(rows.code) & set(family)), "read": sorted(set(mine["read"]))})
    return out


def build_single(edition: str, out: Path, recheck: Path = RECHECK, candidates: list | None = None,
                 prefix: str = "cs") -> list[dict]:
    """The one-volume cards of ``edition`` (``key.csv``, ``cards.json``, ``sections/`` under ``out``);
    ``candidates`` default to :func:`single_candidates`."""
    from PIL import Image
    held = bs.held_out(edition)
    cards, key, wanted, left_out = [], [], set(), 0
    (out / "sections").mkdir(parents=True, exist_ok=True)
    for c in (candidates if candidates is not None else single_candidates(edition, recheck)):
        both = pd.concat([c["rows"][["page", "district", "marks"]],
                          c["hits"].rename(columns={"mark": "marks"})[["page", "district", "marks"]]])
        if set(both.page) & held:
            left_out += 1
            continue
        placed = boxes(both)
        if any("_offices_" in img for img in placed):       # a place type read in the parts list: not the entry
            placed = {img: bx for img, bx in placed.items() if "_offices_" in img}
        bands = [base64.b64encode(_webp(band(edition, img, bx))).decode() for img, bx in sorted(placed.items())]
        sections = sorted(set(placed) | set(sections_of(edition, both)))
        wanted |= set(sections)
        card_id = f"{prefix}-{SHORT[edition]}-{c['court'].replace(':', '_')}-{c['muni']}"
        cards.append({"id": card_id, "edition": edition, "land": LAND[edition],
                      "de": {"muni": str(c["muni"]), "printed": str(c["muni"] % sk.OFFSET), "name": "",
                             "pages": sorted(set(both.page)), "district": c["district"], "bands": bands,
                             "sections": sections}})
        key.append({"card": card_id, "edition": edition, "court_key": c["court"], "district": c["district"],
                    "muni": c["muni"], "pages": ";".join(sorted(set(both.page))), "holds": " ".join(c["holds"]),
                    "read_by_some_reader": " ".join(c["read"])})
    for name in wanted:
        target = out / "sections" / f"{edition}_{Path(name).stem}.webp"
        if not target.exists():
            image = Image.open(FROZEN / f"{edition}_full" / "sections" / name).convert("RGB")
            target.write_bytes(_webp(_scaled(image, SECTION_WIDTH), 68))
    print(f"customs_cards {__version__} ({edition}): {len(cards)} cards, {left_out} left out on held pages")
    return cards, key
