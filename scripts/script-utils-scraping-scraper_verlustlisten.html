How it works

Feasibility pilot first. The index of the k.u.k. casualty lists, transcribed by volunteers of the Verein für Computergenealogie (CompGen), Familia Austria and partners (the VHÚ Praha among them), is served by CompGen's Data Entry System (DES) as project ou1wk. Statistics page, 2026-10-15: 2,694,074 entries on 36,850 list pages, every page "confirmed complete" since 2022-01-16.

How the site can be read — and why that sets the cost

  • Each entry has one server-rendered page, /search/show/{record_id}, with no login. robots.txt disallows only /search/suggest/ (checked 2026-10-15).
  • The project search returns no result list without a login, and the global search returns name + UUID only. Neither is used: this module never logs in.
  • record_id is one integer space shared by every DES project (address books, civil registers, the German casualty lists …). ou1wk entries sit roughly between IDs 13 M and 21 M, interleaved with other projects; an unused ID answers 302 to the home page. So a complete read is a walk over every ID in that band, and its cost is set by how densely ou1wk fills it — which is what the pilot measures.

The pilot (run_pilot) fetches a seeded, stratified random sample of IDs across the band plus a few short consecutive windows, and pilot_report turns that into (a) the ou1wk share by ID band and an estimate of requests and days for a full walk, (b) run lengths inside the windows (whether skipping foreign blocks could ever pay), (c) per-field fill rates and status values on a random sample of entries.

Fields of one entry

Page Number ({Ausgabe}_{page}), Ausgabe, Datum (date of the list), Last name, First name, Offizier/Mannschaft, Charge, Truppenkörper, Unterabteilung, Land, Bezirk, Heimatort, Geburtsjahr, Status, Ereignisdatum, Ort/Region/Land des Ereignisses — plus the entry's UUID (the only identifier CompGen asks to be cited) and the link to the page scan. Values are stored verbatim; a field the page does not know is kept in extra_fields (JSON), never dropped.

Politeness — and identity

  • Sequential, one connection, never parallel; delay below MIN_DELAY is refused; the pause stretches to twice the server's last response time; a rest of batch_pause s every batch_size requests.
  • Backs off on 429/5xx (honouring Retry-After), stops after a run of consecutive failures (ServerUnhealthyError), and stops at once on a bot challenge or a login redirect (BlockedError) — neither is worked around.
  • Every probed ID goes to ledger.csv and is never requested twice; re-runs resume.
  • No identity is sent. The User-Agent names this script and its version only; the session refuses all cookies (the server hands out a JSESSIONID on every response), so requests carry no session either. There is no credential, login or e-mail anywhere in this module, and none is to be added.

def parse_record(html: str, record_id: int) -> dict | None:
    """Parse one ``ou1wk`` entry page into a flat row of :data:`RECORD_COLUMNS`.

    The entry is the two-column table beside the scan: labelled rows (``td.name`` /
    ``td.value``) and a few unlabelled ones (``Page Number``, ``Last name``,
    ``First name``) whose first cell is the label.  Both are read the same way — first
    cell label, second cell value.  A label not in :data:`FIELD_MAP` goes to
    ``extra_fields`` as JSON, so a layout change loses nothing silently.

    Returns:
        The row, or ``None`` when the page is not an ``ou1wk`` entry.
    """
    if project_of(html) != PROJECT_SLUG:
        return None
    soup = BeautifulSoup(html, "html.parser")
    table = soup.find("table", class_="table-striped")
    if table is None:
        return None

    row: dict = {col: "" for col in RECORD_COLUMNS}
    extra: dict[str, str] = {}
    for tr in table.find_all("tr"):
        cells = tr.find_all("td")
        if len(cells) < 2:
            continue
        label, value = _clean(cells[0].get_text()), _clean(cells[1].get_text())
        if label in FIELD_MAP:
            row[FIELD_MAP[label]] = value
        elif label:
            extra[label] = value

    uuid_el = soup.find(id="entry-uuid")
    row["uuid"] = _clean(uuid_el.get_text()) if uuid_el else ""
    scan = soup.find("a", href=re.compile(r"files\.genealogy\.net/verlustlisten/"))
    row["scan_url"] = scan["href"] if scan else ""
    row["extra_fields"] = json.dumps(extra, ensure_ascii=False) if extra else ""
    row["record_id"] = int(record_id)
    row["scraped_at"] = datetime.now(timezone.utc).isoformat(timespec="seconds")
    row["scraper_version"] = __version__
    return row

class DESClient:
    """One polite, cookie-less session against des.genealogy.net."""

    def __init__(
        self,
        *,
        delay: float = DEFAULT_DELAY,
        timeout: float = 60.0,
        retries: int = 4,
        max_requests: int | None = None,
        max_consecutive_failures: int = 5,
        batch_size: int = DEFAULT_BATCH_SIZE,
        batch_pause: float = DEFAULT_BATCH_PAUSE,
    ) -> None:
        if delay < MIN_DELAY:
            raise ValueError(f"delay={delay} s is below the {MIN_DELAY} s floor — "
                             "this site is not to be requested faster than that")
        self.delay = float(delay)
        self.timeout = timeout
        self.retries = retries
        self.max_requests = max_requests
        self.max_consecutive_failures = max_consecutive_failures
        self.batch_size = batch_size
        self.batch_pause = batch_pause
        self.session = make_session()
        self.n_requests = 0
        self._pause = self.delay
        self._last_done: float | None = None
        self._consecutive_failures = 0

    def _wait(self) -> None:
        """Sleep until ``_pause`` seconds have passed since the last response."""
        if self._last_done is not None:
            remaining = self._pause - (time.monotonic() - self._last_done)
            if remaining > 0:
                time.sleep(remaining)
        if self.batch_size and self.n_requests and self.n_requests % self.batch_size == 0:
            log.info("  %d requests this run — resting %.0f s", self.n_requests, self.batch_pause)
            time.sleep(self.batch_pause)

    def fetch(self, record_id: int) -> tuple[str, str | None, int | str, float, str | None]:
        """Fetch one ID.  Redirects are not followed: a 302 to ``/`` means no entry.

        Returns:
            ``(outcome, project, http_status, elapsed_s, html)`` where outcome is
            ``ou1wk`` | ``other`` | ``missing`` | ``unparsed`` | ``error``.

        Raises:
            BlockedError, ServerUnhealthyError, RequestBudgetExceeded.
        """
        url = RECORD_URL.format(record_id=int(record_id))
        status: int | str = "none"
        elapsed = 0.0
        for attempt in range(1, self.retries + 1):
            if self.max_requests is not None and self.n_requests >= self.max_requests:
                raise RequestBudgetExceeded(f"stopped at max_requests={self.max_requests}")
            self._wait()
            started = time.monotonic()
            try:
                response = self.session.get(url, timeout=self.timeout, allow_redirects=False)
                status = response.status_code
            except requests.RequestException as exc:
                response, status = None, exc.__class__.__name__
            elapsed = time.monotonic() - started
            self.n_requests += 1
            self._last_done = time.monotonic()
            # give a slow server room: wait at least twice as long as it took to answer
            self._pause = max(self.delay, min(MAX_ADAPTIVE_DELAY, 2 * elapsed))

            if response is not None and (_is_challenge(response) or _is_login_redirect(response)):
                raise BlockedError(
                    f"{url} answered with a challenge or a login redirect (HTTP {status}). "
                    "Stopping — do not work around it; ask the operators instead.")
            if response is not None and status == 200:
                self._consecutive_failures = 0
                project = project_of(response.text)
                if project is None:
                    return "unparsed", None, status, elapsed, response.text
                outcome = PROJECT_SLUG if project == PROJECT_SLUG else "other"
                return outcome, project, status, elapsed, response.text
            if response is not None and status in (301, 302, 404):
                self._consecutive_failures = 0
                return "missing", None, status, elapsed, None

            wait = min(MAX_BACKOFF, self.delay * 4 ** attempt)
            if response is not None:
                retry_after = _float_or_none(response.headers.get("Retry-After"))
                if retry_after:
                    wait = min(MAX_BACKOFF, max(wait, retry_after))
                if status not in RETRY_STATUS:
                    log.warning("  HTTP %s on %s — not retried", status, url)
                    break
            log.warning("  %s on %s — attempt %d/%d, waiting %.0f s",
                        status, url, attempt, self.retries, wait)
            if attempt < self.retries:
                time.sleep(wait)

        self._consecutive_failures += 1
        if self._consecutive_failures >= self.max_consecutive_failures:
            raise ServerUnhealthyError(
                f"{self._consecutive_failures} IDs in a row failed — the server looks unwell, "
                "so the run stops here. Everything so far is in the ledger; re-run to resume.")
        return "error", None, status, elapsed, None

def walk_ids(
    ids: Iterable[int],
    out_dir: str | Path = DEFAULT_OUT_DIR,
    *,
    keep_html: bool = False,
    progress: bool = True,
    client: DESClient | None = None,
    **client_kwargs,
) -> Counter:
    """Probe every ID in *ids* once, in the order given; resumable.

    IDs already in ``ledger.csv`` are skipped.  Each probe appends one ledger row; an
    ``ou1wk`` entry also appends one ``records.csv`` row.  With ``keep_html`` the raw page
    of every ``ou1wk`` entry is cached so the parser can be re-run without the network.

    Returns:
        Counter of outcomes for this run (``skipped`` = already in the ledger).
    """
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    write_source_note(out_dir)
    ledger_path, records_path = out_dir / "ledger.csv", out_dir / "records.csv"
    done = read_ledger(out_dir)
    ids = [int(i) for i in ids]
    todo = [i for i in ids if i not in done]
    counts = Counter(skipped=len(ids) - len(todo))
    client = client or DESClient(**client_kwargs)
    log.info("%d IDs, %d already in the ledger, %d to fetch (delay %.1f s)",
             len(ids), counts["skipped"], len(todo), client.delay)

    with redirect_logging(), page_bar(len(todo), "ou1wk", enabled=progress, unit="id") as bar:
        for record_id in todo:
            outcome, project, status, elapsed, html = client.fetch(record_id)
            if outcome == PROJECT_SLUG:
                row = parse_record(html, record_id)
                if row is None:
                    outcome = "unparsed"
                else:
                    _append_csv(records_path, RECORD_COLUMNS, row)
                    if keep_html:
                        path = out_dir / "cache" / f"{record_id}.html.gz"
                        path.parent.mkdir(parents=True, exist_ok=True)
                        path.write_bytes(gzip.compress(html.encode("utf-8")))
            if outcome == "unparsed":
                log.warning("  %s: HTTP 200 but no project / entry table", record_id)
            _append_csv(ledger_path, LEDGER_COLUMNS, {
                "record_id": record_id, "outcome": outcome, "project": project or "",
                "http_status": status, "elapsed_s": f"{elapsed:.2f}",
                "fetched_at": datetime.now(timezone.utc).isoformat(timespec="seconds")})
            counts[outcome] += 1
            bar.update(1)
            bar.set_postfix(ou1wk=counts[PROJECT_SLUG], other=counts["other"],
                            missing=counts["missing"])
    log.info("done: %s", dict(counts))
    return counts

def check_canary(client: DESClient | None = None) -> dict:
    """Fetch :data:`CANARY_ID` and check it parses to the known values.

    Raises:
        CanaryError: the entry is missing or any known value differs.
    """
    client = client or DESClient()
    outcome, project, status, _, html = client.fetch(CANARY_ID)
    row = parse_record(html, CANARY_ID) if outcome == PROJECT_SLUG else None
    if row is None:
        raise CanaryError(f"canary {CANARY_ID}: outcome={outcome} project={project} HTTP {status}")
    wrong = {k: (row[k], v) for k, v in CANARY.items() if row[k] != v}
    if wrong or not row["uuid"]:
        raise CanaryError(f"canary {CANARY_ID} parsed wrong (got, expected): {wrong}")
    return row

def pilot_ids(
    n_random: int = 1_200,
    n_windows: int = 4,
    window_len: int = 200,
    low: int = ID_LOW,
    high: int = ID_HIGH,
    n_bands: int = PILOT_BANDS,
    seed: int = 1914,
) -> tuple[list[int], list[int]]:
    """The pilot's ID sample: ``(random_ids, window_ids)``, deterministic for a seed.

    ``random_ids`` are ``n_random // n_bands`` uniform draws inside each of ``n_bands``
    equal bands of ``[low, high)`` — a stratified sample, so every band gets a density
    estimate.  ``window_ids`` are ``n_windows`` runs of ``window_len`` consecutive IDs
    starting at random points, for the run-length view.
    """
    rng = random.Random(seed)
    width = (high - low) / n_bands
    per_band = max(1, n_random // n_bands)
    random_ids = sorted({int(low + b * width + rng.random() * width)
                         for b in range(n_bands) for _ in range(per_band)})
    window_ids: list[int] = []
    for _ in range(n_windows):
        start = rng.randrange(low, high - window_len)
        window_ids.extend(range(start, start + window_len))
    return random_ids, window_ids

def pilot_report(out_dir: str | Path = PILOT_OUT_DIR, *, delay: float = DEFAULT_DELAY,
                 print_output: bool = True) -> dict:
    """Read the pilot's ledger and records — no network — and estimate the full walk.

    Returns a dict of DataFrames / numbers: ``bands`` (ou1wk share per ID band, from the
    random sample only), ``walk`` (IDs, implied entries and days for a full walk over the
    bands that contain ou1wk), ``runs`` (run lengths inside the windows), ``fill`` (share
    of non-empty values per field), ``status`` (Status values).
    """
    import pandas as pd

    out_dir = Path(out_dir)
    design = json.loads((out_dir / "pilot_design.json").read_text(encoding="utf-8"))
    ledger = pd.read_csv(out_dir / "ledger.csv")
    ledger = ledger[ledger["outcome"] != "error"].drop_duplicates("record_id", keep="last")
    random_ids = set(design["random_ids"])
    sample = ledger[ledger["record_id"].isin(random_ids)].copy()
    width = (design["high"] - design["low"]) / design["n_bands"]
    sample["band"] = ((sample["record_id"] - design["low"]) // width).astype(int)
    bands = (sample.assign(is_ou1wk=sample["outcome"].eq(PROJECT_SLUG),
                           is_missing=sample["outcome"].eq("missing"))
             .groupby("band").agg(n=("record_id", "size"), ou1wk_share=("is_ou1wk", "mean"),
                                  missing_share=("is_missing", "mean")))
    bands["id_from"] = (design["low"] + bands.index * width).astype(int)
    bands["id_to"] = (bands["id_from"] + width).astype(int)
    bands["implied_entries"] = (bands["ou1wk_share"] * width).round().astype(int)

    # The client's own pacing: the pause is max(delay, 2 x response time), counted from
    # the end of the last response, plus a batch rest spread over its batch.
    response_s = float(ledger["elapsed_s"].median())
    sec_per_id = (max(delay, 2 * response_s) + response_s
                  + DEFAULT_BATCH_PAUSE / DEFAULT_BATCH_SIZE)
    occupied = bands[bands["ou1wk_share"] > 0]
    n_walk = int(len(occupied) * width)
    walk = {"ids_to_walk": n_walk,
            "implied_entries": int(bands["implied_entries"].sum()),
            "expected_entries": EXPECTED_ENTRIES,
            "seconds_per_id": round(sec_per_id, 2),
            "days_at_this_pace": round(n_walk * sec_per_id / 86_400, 1)}

    by_id = dict(zip(ledger["record_id"], ledger["outcome"]))
    run_rows = []
    for start in design["window_starts"]:
        seq = [by_id.get(i) for i in range(start, start + design["window_len"]) if i in by_id]
        for outcome, length in _runs(seq):
            run_rows.append({"window": start, "outcome": outcome, "length": length})
    runs = pd.DataFrame(run_rows)
    run_stats = (runs.groupby("outcome")["length"].describe()[["count", "mean", "50%", "max"]]
                 if len(runs) else runs)

    records_path = out_dir / "records.csv"
    records = (pd.read_csv(records_path, dtype=str, keep_default_na=False)
               if records_path.exists() else pd.DataFrame(columns=RECORD_COLUMNS))
    records = records[records["record_id"].astype(int).isin(random_ids)]
    fill = (records[FIELD_COLUMNS].ne("").mean().round(3)
            .rename("share_filled").to_frame())
    status = records["status"].value_counts().head(20)

    report = {"bands": bands, "walk": walk, "runs": run_stats, "fill": fill,
              "status": status, "n_random_entries": len(records)}
    if print_output:
        print(f"Pilot: {len(ledger):,} IDs probed, {len(sample):,} in the random sample, "
              f"{len(records):,} ou1wk entries in it\n")
        print("ou1wk share by ID band (random sample):")
        print(bands[["id_from", "id_to", "n", "ou1wk_share", "missing_share"]].to_string())
        print(f"\nFull walk over the occupied bands: {walk['ids_to_walk']:,} IDs -> "
              f"~{walk['implied_entries']:,} entries implied (site says {EXPECTED_ENTRIES:,}); "
              f"{walk['seconds_per_id']} s/ID -> ~{walk['days_at_this_pace']} days\n")
        print("Run lengths inside the consecutive windows:")
        print(run_stats.to_string() if len(runs) else "  (no window data)")
        print("\nShare of entries with the field filled (random sample):")
        print(fill.to_string())
        print("\nStatus values:")
        print(status.to_string())
    return report

#!/usr/bin/env python3
"""
Verlustlisten Österreich-Ungarns 1914–1919 (DES index) scraper — importable module + CLI.

**Feasibility pilot first.**  The index of the k.u.k. casualty lists, transcribed by
volunteers of the Verein für Computergenealogie (CompGen), Familia Austria and partners
(the VHÚ Praha among them), is served by CompGen's Data Entry System (DES) as project
``ou1wk``.  Statistics page, 2026-10-15: **2,694,074 entries on 36,850 list pages**, every
page "confirmed complete" since 2022-01-16.

How the site can be read — and why that sets the cost
-----------------------------------------------------
* Each entry has one server-rendered page, ``/search/show/{record_id}``, with no login.
  ``robots.txt`` disallows only ``/search/suggest/`` (checked 2026-10-15).
* The **project search returns no result list without a login**, and the global search
  returns name + UUID only.  Neither is used: this module never logs in.
* ``record_id`` is **one integer space shared by every DES project** (address books,
  civil registers, the German casualty lists …).  ``ou1wk`` entries sit roughly between
  IDs 13 M and 21 M, interleaved with other projects; an unused ID answers ``302`` to
  the home page.  So a complete read is a walk over every ID in that band, and its cost
  is set by how densely ``ou1wk`` fills it — which is what the pilot measures.

The pilot (:func:`run_pilot`) fetches a seeded, stratified random sample of IDs across
the band plus a few short consecutive windows, and :func:`pilot_report` turns that into
(a) the ``ou1wk`` share by ID band and an estimate of requests and days for a full walk,
(b) run lengths inside the windows (whether skipping foreign blocks could ever pay),
(c) per-field fill rates and status values on a random sample of entries.

Fields of one entry
-------------------
``Page Number`` (``{Ausgabe}_{page}``), ``Ausgabe``, ``Datum`` (date of the list),
``Last name``, ``First name``, ``Offizier/Mannschaft``, ``Charge``, ``Truppenkörper``,
``Unterabteilung``, ``Land``, ``Bezirk``, ``Heimatort``, ``Geburtsjahr``, ``Status``,

Politeness — and identity
-------------------------
* Sequential, one connection, never parallel; ``delay`` below :data:`MIN_DELAY` is
  refused; the pause stretches to twice the server's last response time; a rest of
  ``batch_pause`` s every ``batch_size`` requests.
* Backs off on 429/5xx (honouring ``Retry-After``), stops after a run of consecutive
  failures (:class:`ServerUnhealthyError`), and stops at once on a bot challenge or a
  login redirect (:class:`BlockedError`) — neither is worked around.

* **No identity is sent.**  The ``User-Agent`` names this script and its version only;
  the session **refuses all cookies** (the server hands out a ``JSESSIONID`` on every
  response), so requests carry no session either.  There is no credential, login or
  e-mail anywhere in this module, and none is to be added.

# Slugs may carry dots (``Kreis-St.Goarshausen-AB-1926``) — the pilot's 8 "unparsed" pages.
_PROJECT_LINK = re.compile(r'href="/([^"/]+)/list/overview"')
_WS = re.compile(r"\s+")


def _clean(text: str | None) -> str:
    return _WS.sub(" ", text or "").strip()


def project_of(html: str) -> str | None:
    """The DES project an entry page belongs to (``ou1wk`` …), read off its menu link."""
    match = _PROJECT_LINK.search(html)
    return match.group(1) if match else None


def parse_record(html: str, record_id: int) -> dict | None:
    """Parse one ``ou1wk`` entry page into a flat row of :data:`RECORD_COLUMNS`.

    The entry is the two-column table beside the scan: labelled rows (``td.name`` /
    ``td.value``) and a few unlabelled ones (``Page Number``, ``Last name``,
    ``First name``) whose first cell is the label.  Both are read the same way — first
    cell label, second cell value.  A label not in :data:`FIELD_MAP` goes to
    ``extra_fields`` as JSON, so a layout change loses nothing silently.

    Returns:
        The row, or ``None`` when the page is not an ``ou1wk`` entry.
    """
    if project_of(html) != PROJECT_SLUG:
        return None
    soup = BeautifulSoup(html, "html.parser")
    table = soup.find("table", class_="table-striped")
    if table is None:
        return None

    row: dict = {col: "" for col in RECORD_COLUMNS}
    extra: dict[str, str] = {}
    for tr in table.find_all("tr"):
        cells = tr.find_all("td")
        if len(cells) < 2:
            continue
        label, value = _clean(cells[0].get_text()), _clean(cells[1].get_text())
        if label in FIELD_MAP:
            row[FIELD_MAP[label]] = value
        elif label:
            extra[label] = value

    uuid_el = soup.find(id="entry-uuid")
    row["uuid"] = _clean(uuid_el.get_text()) if uuid_el else ""
    scan = soup.find("a", href=re.compile(r"files\.genealogy\.net/verlustlisten/"))
    row["scan_url"] = scan["href"] if scan else ""
    row["extra_fields"] = json.dumps(extra, ensure_ascii=False) if extra else ""
    row["record_id"] = int(record_id)
    row["scraped_at"] = datetime.now(timezone.utc).isoformat(timespec="seconds")
    row["scraper_version"] = __version__
    return row


def _is_challenge(response: requests.Response) -> bool:
    """A bot challenge rather than the site's own answer."""
    if response.headers.get("cf-mitigated", "").lower() == "challenge":
        return True
    if response.status_code in (403, 503):
        head = response.text[:2000]
        return "Just a moment" in head or "cf-chl" in head or "captcha" in head.lower()
    return False


def _is_login_redirect(response: requests.Response) -> bool:
    return response.is_redirect and "/login" in response.headers.get("Location", "")


def _float_or_none(value) -> float | None:
    try:
        return float(str(value).strip())
    except (TypeError, ValueError):
        return None

def make_session() -> requests.Session:
    """A session that names the script and **refuses every cookie** (see Politeness)."""
    session = requests.Session()
    session.headers.update({"User-Agent": USER_AGENT, "Accept-Language": "de,en;q=0.8"})
    session.cookies.set_policy(DefaultCookiePolicy(allowed_domains=[]))
    return session


class DESClient:
    """One polite, cookie-less session against des.genealogy.net."""

    def __init__(
        self,
        *,
        delay: float = DEFAULT_DELAY,
        timeout: float = 60.0,
        retries: int = 4,
        max_requests: int | None = None,
        max_consecutive_failures: int = 5,
        batch_size: int = DEFAULT_BATCH_SIZE,
        batch_pause: float = DEFAULT_BATCH_PAUSE,
    ) -> None:
        if delay < MIN_DELAY:
            raise ValueError(f"delay={delay} s is below the {MIN_DELAY} s floor — "
                             "this site is not to be requested faster than that")
        self.delay = float(delay)
        self.timeout = timeout
        self.retries = retries
        self.max_requests = max_requests
        self.max_consecutive_failures = max_consecutive_failures
        self.batch_size = batch_size
        self.batch_pause = batch_pause
        self.session = make_session()
        self.n_requests = 0
        self._pause = self.delay
        self._last_done: float | None = None
        self._consecutive_failures = 0

    def _wait(self) -> None:
        """Sleep until ``_pause`` seconds have passed since the last response."""
        if self._last_done is not None:
            remaining = self._pause - (time.monotonic() - self._last_done)
            if remaining > 0:
                time.sleep(remaining)
        if self.batch_size and self.n_requests and self.n_requests % self.batch_size == 0:
            log.info("  %d requests this run — resting %.0f s", self.n_requests, self.batch_pause)
            time.sleep(self.batch_pause)

    def fetch(self, record_id: int) -> tuple[str, str | None, int | str, float, str | None]:
        """Fetch one ID.  Redirects are not followed: a 302 to ``/`` means no entry.

        Returns:
            ``(outcome, project, http_status, elapsed_s, html)`` where outcome is
            ``ou1wk`` | ``other`` | ``missing`` | ``unparsed`` | ``error``.

        Raises:
            BlockedError, ServerUnhealthyError, RequestBudgetExceeded.
        """
        url = RECORD_URL.format(record_id=int(record_id))
        status: int | str = "none"
        elapsed = 0.0
        for attempt in range(1, self.retries + 1):
            if self.max_requests is not None and self.n_requests >= self.max_requests:
                raise RequestBudgetExceeded(f"stopped at max_requests={self.max_requests}")
            self._wait()
            started = time.monotonic()
            try:
                response = self.session.get(url, timeout=self.timeout, allow_redirects=False)
                status = response.status_code
            except requests.RequestException as exc:
                response, status = None, exc.__class__.__name__
            elapsed = time.monotonic() - started
            self.n_requests += 1
            self._last_done = time.monotonic()
            # give a slow server room: wait at least twice as long as it took to answer
            self._pause = max(self.delay, min(MAX_ADAPTIVE_DELAY, 2 * elapsed))

            if response is not None and (_is_challenge(response) or _is_login_redirect(response)):
                raise BlockedError(
                    f"{url} answered with a challenge or a login redirect (HTTP {status}). "
                    "Stopping — do not work around it; ask the operators instead.")
            if response is not None and status == 200:
                self._consecutive_failures = 0
                project = project_of(response.text)
                if project is None:
                    return "unparsed", None, status, elapsed, response.text
                outcome = PROJECT_SLUG if project == PROJECT_SLUG else "other"
                return outcome, project, status, elapsed, response.text
            if response is not None and status in (301, 302, 404):
                self._consecutive_failures = 0
                return "missing", None, status, elapsed, None

            wait = min(MAX_BACKOFF, self.delay * 4 ** attempt)
            if response is not None:
                retry_after = _float_or_none(response.headers.get("Retry-After"))
                if retry_after:
                    wait = min(MAX_BACKOFF, max(wait, retry_after))
                if status not in RETRY_STATUS:
                    log.warning("  HTTP %s on %s — not retried", status, url)
                    break
            log.warning("  %s on %s — attempt %d/%d, waiting %.0f s",
                        status, url, attempt, self.retries, wait)
            if attempt < self.retries:
                time.sleep(wait)

        self._consecutive_failures += 1
        if self._consecutive_failures >= self.max_consecutive_failures:
            raise ServerUnhealthyError(
                f"{self._consecutive_failures} IDs in a row failed — the server looks unwell, "
                "so the run stops here. Everything so far is in the ledger; re-run to resume.")
        return "error", None, status, elapsed, None

def _append_csv(path: Path, columns: list[str], row: dict) -> None:
    new = not path.exists() or path.stat().st_size == 0
    with open(path, "a", newline="", encoding="utf-8") as fh:
        writer = csv.DictWriter(fh, fieldnames=columns, extrasaction="ignore")
        if new:
            writer.writeheader()
        writer.writerow(row)


def read_ledger(out_dir: str | Path) -> dict[int, str]:
    """``{record_id: outcome}`` for every ID already probed (errors excluded, so they retry)."""
    path = Path(out_dir) / "ledger.csv"
    if not path.exists():
        return {}
    done: dict[int, str] = {}
    with open(path, newline="", encoding="utf-8") as fh:
        for row in csv.DictReader(fh):
            if row["outcome"] != "error":
                done[int(row["record_id"])] = row["outcome"]
    return done

# ---------------------------------------------------------------------------
# The walk
# ---------------------------------------------------------------------------
def walk_ids(
    ids: Iterable[int],
    out_dir: str | Path = DEFAULT_OUT_DIR,
    *,
    keep_html: bool = False,
    progress: bool = True,
    client: DESClient | None = None,
    **client_kwargs,
) -> Counter:
    """Probe every ID in *ids* once, in the order given; resumable.

    IDs already in ``ledger.csv`` are skipped.  Each probe appends one ledger row; an
    ``ou1wk`` entry also appends one ``records.csv`` row.  With ``keep_html`` the raw page
    of every ``ou1wk`` entry is cached so the parser can be re-run without the network.

    Returns:
        Counter of outcomes for this run (``skipped`` = already in the ledger).
    """
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    write_source_note(out_dir)
    ledger_path, records_path = out_dir / "ledger.csv", out_dir / "records.csv"
    done = read_ledger(out_dir)
    ids = [int(i) for i in ids]
    todo = [i for i in ids if i not in done]
    counts = Counter(skipped=len(ids) - len(todo))
    client = client or DESClient(**client_kwargs)
    log.info("%d IDs, %d already in the ledger, %d to fetch (delay %.1f s)",
             len(ids), counts["skipped"], len(todo), client.delay)

    with redirect_logging(), page_bar(len(todo), "ou1wk", enabled=progress, unit="id") as bar:
        for record_id in todo:
            outcome, project, status, elapsed, html = client.fetch(record_id)
            if outcome == PROJECT_SLUG:
                row = parse_record(html, record_id)
                if row is None:
                    outcome = "unparsed"
                else:
                    _append_csv(records_path, RECORD_COLUMNS, row)
                    if keep_html:
                        path = out_dir / "cache" / f"{record_id}.html.gz"
                        path.parent.mkdir(parents=True, exist_ok=True)
                        path.write_bytes(gzip.compress(html.encode("utf-8")))
            if outcome == "unparsed":
                log.warning("  %s: HTTP 200 but no project / entry table", record_id)
            _append_csv(ledger_path, LEDGER_COLUMNS, {
                "record_id": record_id, "outcome": outcome, "project": project or "",
                "http_status": status, "elapsed_s": f"{elapsed:.2f}",
                "fetched_at": datetime.now(timezone.utc).isoformat(timespec="seconds")})
            counts[outcome] += 1
            bar.update(1)
            bar.set_postfix(ou1wk=counts[PROJECT_SLUG], other=counts["other"],
                            missing=counts["missing"])
    log.info("done: %s", dict(counts))
    return counts


def check_canary(client: DESClient | None = None) -> dict:
    """Fetch :data:`CANARY_ID` and check it parses to the known values.

    Raises:
        CanaryError: the entry is missing or any known value differs.
    """
    client = client or DESClient()
    outcome, project, status, _, html = client.fetch(CANARY_ID)
    row = parse_record(html, CANARY_ID) if outcome == PROJECT_SLUG else None
    if row is None:
        raise CanaryError(f"canary {CANARY_ID}: outcome={outcome} project={project} HTTP {status}")
    wrong = {k: (row[k], v) for k, v in CANARY.items() if row[k] != v}
    if wrong or not row["uuid"]:
        raise CanaryError(f"canary {CANARY_ID} parsed wrong (got, expected): {wrong}")
    return row


def scrape_range(
    start_id: int,
    end_id: int,
    out_dir: str | Path = DEFAULT_OUT_DIR,
    *,
    progress: bool = True,
    **client_kwargs,
) -> Counter:
    """The full walk over ``start_id..end_id`` (inclusive), resumable.  Canary first.

    At the default 1.5 s a million IDs is ~19 days — run :func:`run_pilot` first, and
    prefer CompGen's export if it comes.  ``client_kwargs`` go to :class:`DESClient`.
    """
    client = DESClient(**client_kwargs)
    check_canary(client)
    return walk_ids(range(int(start_id), int(end_id) + 1), out_dir,
                    progress=progress, client=client)

def pilot_ids(
    n_random: int = 1_200,
    n_windows: int = 4,
    window_len: int = 200,
    low: int = ID_LOW,
    high: int = ID_HIGH,
    n_bands: int = PILOT_BANDS,
    seed: int = 1914,
) -> tuple[list[int], list[int]]:
    """The pilot's ID sample: ``(random_ids, window_ids)``, deterministic for a seed.

    ``random_ids`` are ``n_random // n_bands`` uniform draws inside each of ``n_bands``
    equal bands of ``[low, high)`` — a stratified sample, so every band gets a density
    estimate.  ``window_ids`` are ``n_windows`` runs of ``window_len`` consecutive IDs
    starting at random points, for the run-length view.
    """
    rng = random.Random(seed)
    width = (high - low) / n_bands
    per_band = max(1, n_random // n_bands)
    random_ids = sorted({int(low + b * width + rng.random() * width)
                         for b in range(n_bands) for _ in range(per_band)})
    window_ids: list[int] = []
    for _ in range(n_windows):
        start = rng.randrange(low, high - window_len)
        window_ids.extend(range(start, start + window_len))
    return random_ids, window_ids


def run_pilot(
    out_dir: str | Path = PILOT_OUT_DIR,
    *,
    n_random: int = 1_200,
    n_windows: int = 4,
    window_len: int = 200,
    seed: int = 1914,
    **client_kwargs,
) -> dict:
    """Canary, then the pilot sample (~2,000 requests, ~1 h at 1.5 s), then the report.

    Raw pages are cached (``keep_html=True``) so the parser can be re-run offline.
    Resumable: an interrupted pilot continues where it stopped.
    """
    out_dir = Path(out_dir)
    random_ids, window_ids = pilot_ids(n_random, n_windows, window_len,
                                       n_bands=PILOT_BANDS, seed=seed)
    out_dir.mkdir(parents=True, exist_ok=True)
    # The report reads the sample back from here, so it never depends on regenerating it.
    (out_dir / "pilot_design.json").write_text(json.dumps({
        "seed": seed, "low": ID_LOW, "high": ID_HIGH, "n_bands": PILOT_BANDS,
        "window_len": window_len, "window_starts": window_ids[::window_len],
        "random_ids": random_ids}), encoding="utf-8")
    client = DESClient(**client_kwargs)
    check_canary(client)
    walk_ids(random_ids + window_ids, out_dir, keep_html=True, client=client)
    return pilot_report(out_dir)


def _runs(outcomes: list[str]) -> list[tuple[str, int]]:
    """Run-length encode a sequence of outcomes: ``[(outcome, length), …]``."""
    runs: list[tuple[str, int]] = []
    for outcome in outcomes:
        if runs and runs[-1][0] == outcome:
            runs[-1] = (outcome, runs[-1][1] + 1)
        else:
            runs.append((outcome, 1))
    return runs


def pilot_report(out_dir: str | Path = PILOT_OUT_DIR, *, delay: float = DEFAULT_DELAY,
                 print_output: bool = True) -> dict:
    """Read the pilot's ledger and records — no network — and estimate the full walk.

    Returns a dict of DataFrames / numbers: ``bands`` (ou1wk share per ID band, from the
    random sample only), ``walk`` (IDs, implied entries and days for a full walk over the
    bands that contain ou1wk), ``runs`` (run lengths inside the windows), ``fill`` (share
    of non-empty values per field), ``status`` (Status values).
    """
    import pandas as pd

    out_dir = Path(out_dir)
    design = json.loads((out_dir / "pilot_design.json").read_text(encoding="utf-8"))
    ledger = pd.read_csv(out_dir / "ledger.csv")
    ledger = ledger[ledger["outcome"] != "error"].drop_duplicates("record_id", keep="last")
    random_ids = set(design["random_ids"])
    sample = ledger[ledger["record_id"].isin(random_ids)].copy()
    width = (design["high"] - design["low"]) / design["n_bands"]
    sample["band"] = ((sample["record_id"] - design["low"]) // width).astype(int)
    bands = (sample.assign(is_ou1wk=sample["outcome"].eq(PROJECT_SLUG),
                           is_missing=sample["outcome"].eq("missing"))
             .groupby("band").agg(n=("record_id", "size"), ou1wk_share=("is_ou1wk", "mean"),
                                  missing_share=("is_missing", "mean")))
    bands["id_from"] = (design["low"] + bands.index * width).astype(int)
    bands["id_to"] = (bands["id_from"] + width).astype(int)
    bands["implied_entries"] = (bands["ou1wk_share"] * width).round().astype(int)

    # The client's own pacing: the pause is max(delay, 2 x response time), counted from
    # the end of the last response, plus a batch rest spread over its batch.
    response_s = float(ledger["elapsed_s"].median())
    sec_per_id = (max(delay, 2 * response_s) + response_s
                  + DEFAULT_BATCH_PAUSE / DEFAULT_BATCH_SIZE)
    occupied = bands[bands["ou1wk_share"] > 0]
    n_walk = int(len(occupied) * width)
    walk = {"ids_to_walk": n_walk,
            "implied_entries": int(bands["implied_entries"].sum()),
            "expected_entries": EXPECTED_ENTRIES,
            "seconds_per_id": round(sec_per_id, 2),
            "days_at_this_pace": round(n_walk * sec_per_id / 86_400, 1)}

    by_id = dict(zip(ledger["record_id"], ledger["outcome"]))
    run_rows = []
    for start in design["window_starts"]:
        seq = [by_id.get(i) for i in range(start, start + design["window_len"]) if i in by_id]
        for outcome, length in _runs(seq):
            run_rows.append({"window": start, "outcome": outcome, "length": length})
    runs = pd.DataFrame(run_rows)
    run_stats = (runs.groupby("outcome")["length"].describe()[["count", "mean", "50%", "max"]]
                 if len(runs) else runs)

    records_path = out_dir / "records.csv"
    records = (pd.read_csv(records_path, dtype=str, keep_default_na=False)
               if records_path.exists() else pd.DataFrame(columns=RECORD_COLUMNS))
    records = records[records["record_id"].astype(int).isin(random_ids)]
    fill = (records[FIELD_COLUMNS].ne("").mean().round(3)
            .rename("share_filled").to_frame())
    status = records["status"].value_counts().head(20)

    report = {"bands": bands, "walk": walk, "runs": run_stats, "fill": fill,
              "status": status, "n_random_entries": len(records)}
    if print_output:
        print(f"Pilot: {len(ledger):,} IDs probed, {len(sample):,} in the random sample, "
              f"{len(records):,} ou1wk entries in it\n")
        print("ou1wk share by ID band (random sample):")
        print(bands[["id_from", "id_to", "n", "ou1wk_share", "missing_share"]].to_string())
        print(f"\nFull walk over the occupied bands: {walk['ids_to_walk']:,} IDs -> "
              f"~{walk['implied_entries']:,} entries implied (site says {EXPECTED_ENTRIES:,}); "
              f"{walk['seconds_per_id']} s/ID -> ~{walk['days_at_this_pace']} days\n")
        print("Run lengths inside the consecutive windows:")
        print(run_stats.to_string() if len(runs) else "  (no window data)")
        print("\nShare of entries with the field filled (random sample):")
        print(fill.to_string())
        print("\nStatus values:")
        print(status.to_string())
    return report

def test_id(record_id: int = CANARY_ID) -> dict | None:
    """Fetch one ID and pretty-print it.  Returns the parsed row, or ``None``."""
    outcome, project, status, elapsed, html = DESClient().fetch(record_id)
    print(f"{RECORD_URL.format(record_id=record_id)}  ->  {outcome} "
          f"(project={project}, HTTP {status}, {elapsed:.2f} s)")
    row = parse_record(html, record_id) if outcome == PROJECT_SLUG else None
    if row:
        width = max(len(k) for k in row)
        for key, value in row.items():
            print(f"  {key:<{width}}  {value}")
    return row

def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(
        description="Read the DES index of the Austro-Hungarian casualty lists (ou1wk).",
        formatter_class=argparse.ArgumentDefaultsHelpFormatter)
    mode = p.add_mutually_exclusive_group(required=True)
    mode.add_argument("--test-id", type=int, metavar="ID", help="Fetch and print one entry.")
    mode.add_argument("--pilot", action="store_true", help="Run the pilot sample, then report.")
    mode.add_argument("--report", action="store_true", help="Pilot report only (no network).")
    mode.add_argument("--start", type=int, help="First ID of a full-walk slice.")
    p.add_argument("--end", type=int, help="Last ID (inclusive) of the slice.")
    p.add_argument("--out-dir", default=None,
                   help="Output directory (default: data/raw/verlustlisten[/pilot]).")
    p.add_argument("--delay", type=float, default=DEFAULT_DELAY,
                   help=f"Seconds between requests (floor {MIN_DELAY}).")
    p.add_argument("--max-requests", type=int, default=None, help="Stop after this many.")
    p.add_argument("--log-level", default="INFO",
                   choices=["DEBUG", "INFO", "WARNING", "ERROR"])
    return p


def main(argv: list[str] | None = None) -> None:
    args = build_parser().parse_args(argv)
    logging.basicConfig(level=getattr(logging, args.log_level),
                        format="%(asctime)s %(levelname)s %(message)s", datefmt="%H:%M:%S")
    client_kwargs = {"delay": args.delay, "max_requests": args.max_requests}
    if args.test_id:
        test_id(args.test_id)
    elif args.pilot:
        run_pilot(args.out_dir or PILOT_OUT_DIR, **client_kwargs)
    elif args.report:
        pilot_report(args.out_dir or PILOT_OUT_DIR, delay=args.delay)
    else:
        if args.end is None:
            raise SystemExit("--start needs --end")
        scrape_range(args.start, args.end, args.out_dir or DEFAULT_OUT_DIR, **client_kwargs)


if __name__ == "__main__":
    main()
