How it works

Source

One SPARQL query to the Wikidata Query Service (CC0). It returns every item that

  • carries a zanikleobce.cz ID (P3003) — the volunteer register Zaniklé obce a objekty po roce 1945 (https://www.zanikleobce.cz), whose identifiers Wikidata imported item by item; or
  • is an instance (P31) of abandoned village (Q350895), abandoned hamlet (Q27654220), deserted farmyard (Q28807616) or former entity (Q15893266) located in Czechia (P17 Q213).

Per item: QID, Czech and German labels, coordinates (P625), classes, the zanikleobce.cz ID, the administrative unit it is filed under (P131), inception (P571) and dissolution (P576). zanikleobce.cz itself is not fetched: its robots.txt asks crawlers off its query pages and it states no licence; Wikidata's copy of its identifiers is CC0, and every point keeps the ID so it can be cited back to the page it came from (https://www.zanikleobce.cz/index.php?obec=&lt;id&gt;).

The raw result is cached under data/raw/geo/wikidata_vanished/ beside a SOURCE.txt holding the query text, the endpoint, the retrieval time (UTC), the row count and the sha256 of the file — so the build can be re-run without re-querying, and the exact query is on record.

Output: vanished_places [qid] — labels, point, classes, zanikleobce_id, the 1910 court and okres the point falls in, and the modern obec whose territory holds the site today.

def fetch_vanished(cache_path, refresh: bool = False) -> pd.DataFrame:
    """The raw query result (CSV), from the cache unless *refresh* or the cache is missing."""
    import requests

    cache_path = Path(cache_path)
    if cache_path.exists() and not refresh:
        return pd.read_csv(cache_path, dtype=str)
    resp = requests.get(ENDPOINT, params={"query": QUERY},
                        headers={"Accept": "text/csv", "User-Agent": USER_AGENT}, timeout=300)
    resp.raise_for_status()
    cache_path.parent.mkdir(parents=True, exist_ok=True)
    cache_path.write_bytes(resp.content)
    raw = pd.read_csv(cache_path, dtype=str)
    (cache_path.parent / "SOURCE.txt").write_text(
        "Wikidata Query Service — vanished / abandoned settlements of Czechia\n"
        f"endpoint:   {ENDPOINT}\n"
        f"retrieved:  {datetime.now(timezone.utc).isoformat(timespec='seconds')}\n"
        f"file:       {cache_path.name}\n"
        f"rows:       {len(raw)}\n"
        f"bytes:      {cache_path.stat().st_size}\n"
        f"sha256:     {_sha256(cache_path)}\n"
        "licence:    CC0 (Wikidata). zanikleobce.cz IDs (P3003) cite https://www.zanikleobce.cz\n"
        f"built by:   utils/parsing/vanished.py {__version__} (fetch_vanished)\n"
        "approved:   by the author in chat, 2026-10-29\n\n"
        "query:\n" + QUERY, encoding="utf-8")
    return raw

def build_vanished(raw: pd.DataFrame, locator, obce: pd.DataFrame) -> pd.DataFrame:
    """Tidy the query result; locate each point in the 1910 courts and in today's obce."""
    from utils.parsing import ruian as ru

    out = pd.DataFrame({"qid": raw["item"].str.rsplit("/", n=1).str[-1]})
    out["label_cs"] = raw["label_cs"].str.strip()
    out["label_de"] = raw["label_de"].str.strip()
    ll = raw["coord"].map(_point)
    out["lon"] = [a for a, _b in ll]
    out["lat"] = [b for _a, b in ll]
    out["n_coord"] = pd.to_numeric(raw["n_coord"]).astype("Int64")
    out["zanikleobce_id"] = raw["zanikleobce_id"]
    out["classes"] = raw["classes"]
    out["settlement_like"] = out["classes"].fillna("").str.split("|").map(
        lambda cs: any(c in SETTLEMENT_CLASSES for c in cs))
    out["admin_qid"] = raw["admin"].str.rsplit("/", n=1).str[-1]
    out["admin_cs"] = raw["admin_cs"]
    out["inception"] = raw["inception"].str[:10]
    out["dissolved"] = raw["dissolved"].str[:10]
    court, okres = ru.locate_courts(out["lon"].to_numpy(float), out["lat"].to_numpy(float), locator)
    out["court_id"] = pd.array(court, dtype="Int64")
    out["okres_id_1910"] = pd.array(okres, dtype="Int64")
    x, y = ru.wgs84_to_sjtsk(out["lon"].to_numpy(float), out["lat"].to_numpy(float))
    out["obec_kod"] = pd.array(ru.locate_obce(x, y, obce), dtype="Int64")
    assert out["qid"].is_unique and out["qid"].notna().all()
    return out

"""
vanished.py — vanished and abandoned settlements of the Czech lands, from Wikidata
=================================================================================
RÚIAN and Geonames list what exists today. A 1910 settlement that has since disappeared — a
German hamlet left empty after 1945, a village cleared for a reservoir or a military area, a
farmstead that fell in — is on neither, so the crosswalk (:mod:`utils.parsing.ruian`) cannot
place it however good the name match. This module supplies the places that are gone.

Source
------
One SPARQL query to the Wikidata Query Service (CC0). It returns every item that

* carries a **zanikleobce.cz ID** (P3003) — the volunteer register *Zaniklé obce a objekty po
  roce 1945* (https://www.zanikleobce.cz), whose identifiers Wikidata imported item by item; or
* is an instance (P31) of *abandoned village* (Q350895), *abandoned hamlet* (Q27654220),
  *deserted farmyard* (Q28807616) or *former entity* (Q15893266) located in Czechia (P17 Q213).

Per item: QID, Czech and German labels, coordinates (P625), classes, the zanikleobce.cz ID,
the administrative unit it is filed under (P131), inception (P571) and dissolution (P576).
zanikleobce.cz itself is **not** fetched: its robots.txt asks crawlers off its query pages and
it states no licence; Wikidata's copy of its identifiers is CC0, and every point keeps the ID so
it can be cited back to the page it came from (``https://www.zanikleobce.cz/index.php?obec=<id>``).

The raw result is cached under ``data/raw/geo/wikidata_vanished/`` beside a ``SOURCE.txt``
holding the query text, the endpoint, the retrieval time (UTC), the row count and the sha256 of
the file — so the build can be re-run without re-querying, and the exact query is on record.

Output: ``vanished_places`` [``qid``] — labels, point, classes, ``zanikleobce_id``, the 1910 court
and okres the point falls in, and the modern obec whose territory holds the site today.
"""

from __future__ import annotations

import hashlib
from datetime import datetime, timezone
from pathlib import Path

import numpy as np
import pandas as pd

__version__ = "1.0.0"   # first build (2026-10-29)

ENDPOINT = "https://query.wikidata.org/sparql"
# Polite and anonymous: the script names itself, never the author (project scraping rule).
USER_AGENT = "Legions-thesis utils/parsing/vanished.py/1.0 (research; one query, cached)"
# Classes that make an item a place a person could have lived in — the only items the
# crosswalk will match. Churches, chapels, synagogues, monuments, streets and trees are kept in
# the table (``settlement_like`` False) but never matched to a settlement name. Labels as
# Wikidata gives them (wbgetentities, 2026-10-29).
SETTLEMENT_CLASSES = {
    "Q350895": "abandoned village", "Q27654220": "abandoned hamlet",
    "Q28807616": "deserted farmyard", "Q1138414": "isolated settlement", "Q5084": "hamlet",
    "Q532": "village", "Q486972": "human settlement", "Q5153359": "municipality of the Czech Republic",
    "Q61089180": "municipal part of the Czech Republic", "Q329245": "basic unit of settlement",
    "Q20871353": "cadastral area in the Czech Republic", "Q56822897": "mill building",
    "Q185187": "watermill",
}
CLASSES = {
    "Q350895": "abandoned village",
    "Q27654220": "abandoned hamlet",
    "Q28807616": "deserted farmyard",
    "Q15893266": "former entity",
}

QUERY = """
SELECT ?item
       (SAMPLE(?csL) AS ?label_cs) (SAMPLE(?deL) AS ?label_de)
       (SAMPLE(?c) AS ?coord) (COUNT(DISTINCT ?c) AS ?n_coord)
       (SAMPLE(?z) AS ?zanikleobce_id)
       (GROUP_CONCAT(DISTINCT STRAFTER(STR(?cls), "entity/"); separator="|") AS ?classes)
       (SAMPLE(?adm) AS ?admin) (SAMPLE(?admL) AS ?admin_cs)
       (SAMPLE(?inc) AS ?inception) (SAMPLE(?dis) AS ?dissolved)
WHERE {
  { ?item wdt:P3003 ?z0 }
  UNION
  { VALUES ?k { %CLASSES% } ?item wdt:P31 ?k ; wdt:P17 wd:Q213 }
  OPTIONAL { ?item wdt:P3003 ?z }
  OPTIONAL { ?item rdfs:label ?csL FILTER(LANG(?csL) = "cs") }
  OPTIONAL { ?item rdfs:label ?deL FILTER(LANG(?deL) = "de") }
  OPTIONAL { ?item wdt:P625 ?c }
  OPTIONAL { ?item wdt:P31 ?cls }
  OPTIONAL { ?item wdt:P131 ?adm . OPTIONAL { ?adm rdfs:label ?admL FILTER(LANG(?admL) = "cs") } }
  OPTIONAL { ?item wdt:P571 ?inc }
  OPTIONAL { ?item wdt:P576 ?dis }
}
GROUP BY ?item
""".replace("%CLASSES%", " ".join(f"wd:{q}" for q in CLASSES))


def _sha256(path: Path) -> str:
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def _point(wkt) -> tuple[float, float]:
    """``'Point(14.1 50.2)'`` → ``(lon, lat)``."""
    if not isinstance(wkt, str) or not wkt.startswith("Point("):
        return (np.nan, np.nan)
    lon, lat = wkt[6:-1].split()[:2]
    return (float(lon), float(lat))
