How it works

The two societies printed their members as lists, and utils/scraping/run_matice_downloads.py keeps every page's OCR with its ALTO word coordinates in library/associations/ (DATA_SOURCES.md #28, #29). This module reads those pages back into one row per printed entry, keeping the entry's own text beside every field read from it.

Three layouts occur:

mc Matice česká, one founder per line: "Adamec Frant., obchodník v Studnici u Č. Skalice, r. 83, č. 5359." An entry ends with its membership number, so lines are joined only while that number is still missing. A Roman number marks a corporate founder (a reading club, a school, a town); a leading dash repeats words of the entry above. The deceased carry the number and the year of death instead. lines Matice moravská 1909, one member per line with a hanging indent: a line set in from the left margin continues the entry above (read from the ALTO HPOS). para Matice moravská 1876–1891, members run on in a paragraph, separated by a comma or a full stop. An entry is cut where a residence (v Brně) or an amount paid (100 zl.) has been read and the text after the separator opens a new name. A paragraph of Place: name, occupation. name, occupation. groups is read by place, the place heading becoming each entry's residence. payments Matice česká 1885–1914, the monthly Zpráva o Matici České in the Časopis Musea: everyone who paid into the society that month, run on and separated by dashes, each entry ending in its amount (21 K., 10 zl. 50 kr.), the month closed by its total (Úhrnem), against which every month is checked (parse_payments). A founder paid 105 K in instalments, so a payer's payments are gathered to a person (payment_persons), and the persons who began paying in a window — the joiners — are Bohemia's measure (D-184).

The residences are matched to 1910 settlements by utils.parsing.locative (the printed form is mostly locative, v Brně); this module then classes each member's occupation as posted (clergy, teachers, officials — placed by an employer) or local, and sums members to the 1910 census okres (okres_measure, D-161). It does not decide which entries of two issues are one person beyond the Matice česká membership number, which the society kept for life.

def read_pdf_lines(pdf: Path, first: int, last: int) -> pd.DataFrame:
    """The text lines of PDF pages ``first``–``last`` (1-based) from the file's own text
    layer, in the shape of :func:`read_lines`. The Internet Archive marks a hyphenated
    line end with ``¬``; there is no geometry, so ``hpos``/``vpos`` are 0 and the line
    number."""
    from pypdf import PdfReader
    reader = PdfReader(str(pdf))
    rows = []
    for page in range(first, last + 1):
        text = reader.pages[page - 1].extract_text() or ""
        for i, raw in enumerate(text.splitlines(), 1):
            line = normalise(raw)
            hyphen = line.endswith("¬")
            rows.append({"page": page, "line": i, "hpos": 0, "vpos": i,
                         "text": line.rstrip("¬").rstrip(), "hyphen": hyphen})
    return pd.DataFrame(rows, columns=["page", "line", "hpos", "vpos", "text", "hyphen"])

def collapse_spacing(name: str, *, strict: bool = False) -> str:
    """Undo the letter-spacing of names set in *Sperrsatz* (``D r ej šuk`` → ``Drejšuk``,
    ``A b e 1 o v á`` → ``Abelová``) and a surname the OCR split (``Dur dík`` → ``Durdík``).

    ``strict`` (for whole lines of text rather than a name) joins a lower-case fragment
    only to a one- or two-letter capitalised one, so *Jan farář* stays two words."""
    short_cap = rf"[{_UC}][{_LC}]{{0,1}}" if strict else rf"[{_UC}][{_LC}]{{0,2}}"
    toks = name.split()
    out: list[str] = []
    merged_last = False
    i = 0
    while i < len(toks):
        # a run of three or more one- or two-letter tokens is a spaced-out word
        j = i
        while j < len(toks) and re.fullmatch(rf"[{_UC}{_LC}1]{{1,2}}", toks[j]):
            j += 1
        if j - i >= 3 and any(len(t) == 1 for t in toks[i:j]):
            word = "".join(toks[i:j])
            word = word[0] + word[1:].replace("1", "l")
            out.append(word)
            merged_last = True
            i = j
            continue
        t = toks[i]
        prev = out[-1] if out else ""
        if (prev and re.fullmatch(rf"[{_LC}]+\.?,?", t) and t.rstrip(".,") not in _PARTICLES
                and not prev.endswith((".", ",")) and prev not in _SHORT_WORDS
                and (merged_last or re.fullmatch(short_cap, prev))):
            out[-1] = prev + t
            merged_last = True
        else:
            out.append(t)
            merged_last = False
        i += 1
    return " ".join(out)

def parse_person(text: str) -> dict:
    """Honorific, name, occupation, residence and amount of one entry's text."""
    out = {"honorific": "", "name": "", "occupation": "", "residence": "",
           "residence_prep": "", "amount_zl": pd.NA}
    text = _CONTINUED.sub("", text).strip(" ,;")
    m = _AMOUNT.search(text)
    if m:
        out["amount_zl"] = int(m.group("a") or m.group("b"))
        text = text[:m.start()].strip(" ,")
    m = _HONORIFIC.match(text)
    if m:
        out["honorific"] = re.sub(r"P\.\s*T\.\s*(?:pánové|pp\.)\s*:?", "", m.group(0)).strip()
        text = text[m.end():]
    head, _, rest = text.partition(",")
    if not rest:                                  # no comma: the name runs to the place
        head, prep, place = split_residence(text)
        out.update(name=head, residence_prep=prep, residence=place)
    else:
        before, prep, place = split_residence(rest)
        if prep == "" and _PLACE.search(head):    # "Durasová Jaroslava v Roudnici, r. 49"
            head, prep, place = split_residence(head)
            before = rest
        out.update(name=head.strip(), occupation=before.strip(" ,"),
                   residence_prep=prep, residence=place)
    out["name"] = collapse_spacing(out["name"].strip(" ,"))
    return out

def roman_to_int(raw: str) -> int | None:
    """A Roman numeral as the OCR printed it, or ``None`` if it cannot be read as one.

    Repairs the common misreadings (``l``/``1``/``í`` for I, ``H`` for II, ``Ш`` for III,

    s = raw.replace(" ", "").replace("Ш", "III").replace("H", "II").translate(_ROMAN_FIX)
    candidates = [s]
    if s.endswith("L") and len(s) > 1 and s[-2] == "I":
        candidates.append(s[:-1] + "I")
    for c in candidates:
        if re.fullmatch(r"[IVXLC]+", c):
            n = _roman_value(c)
            if n and int_to_roman(n) == c:
                return n
    return None

Matice česká and Matice moravská member registers — OCR to one row per printed entry.

keeps every page's OCR with its ALTO word coordinates in ``library/associations/``
(``DATA_SOURCES.md`` #28, #29). This module reads those pages back into one row per
printed entry, keeping the entry's own text beside every field read from it.

    joined only while that number is still missing. A Roman number marks a corporate
    founder (a reading club, a school, a town); a leading dash repeats words of the
    entry above. The deceased carry the number and the year of death instead.

    or a full stop. An entry is cut where a residence (*v Brně*) or an amount paid
    (*100 zl.*) has been read and the text after the separator opens a new name. A
    paragraph of ``Place: name, occupation. name, occupation.`` groups is read by place,
    the place heading becoming each entry's residence.
``payments``
    Matice česká 1885–1914, the monthly *Zpráva o Matici České* in the *Časopis Musea*:
    everyone who paid into the society that month, run on and separated by dashes, each
    entry ending in its amount (*21 K.*, *10 zl. 50 kr.*), the month closed by its total
    (*Úhrnem*), against which every month is checked (:func:`parse_payments`). A
    founder paid 105 K in instalments, so a payer's payments are gathered to a person
    (:func:`payment_persons`), and the persons who began paying in a window — the
    joiners — are Bohemia's measure (D-184).

printed form is mostly locative, *v Brně*); this module then classes each member's
occupation as *posted* (clergy, teachers, officials — placed by an employer) or *local*,
and sums members to the 1910 census okres (:func:`okres_measure`, D-161). It does not
decide which entries of two issues are one person beyond the Matice česká membership

_LINE = re.compile(r"<TextLine ([^>]*)>(.*?)</TextLine>", re.S)
_STRING = re.compile(r"<String ([^>]*?)/?>")
_ATTR = re.compile(r'([A-Z_]+)="([^"]*)"')

#: Cyrillic letters the OCR put into Latin words, mapped back.
_HOMOGLYPHS = str.maketrans("АВЕКМНОРСТХаеорсхк", "ABEKMHOPCTXaeopcxk")


def normalise(text: str) -> str:
    """NFC, Latin for Cyrillic look-alikes, one space between words."""
    text = unicodedata.normalize("NFC", text).translate(_HOMOGLYPHS)
    return re.sub(r"\s+", " ", text.replace("﻿", "")).strip()


def alto_lines(path: Path) -> list[dict]:
    """The text lines of one ALTO page: left edge, top, text, and whether the line ends
    on the first half of a hyphenated word."""
    out = []
    for m in _LINE.finditer(path.read_text(encoding="utf-8")):
        attrs = dict(_ATTR.findall(m.group(1)))
        words, hyphen = [], False
        for s in _STRING.finditer(m.group(2)):
            sa = dict(_ATTR.findall(s.group(1)))
            words.append(html.unescape(sa.get("CONTENT", "")))
            hyphen = sa.get("SUBS_TYPE") == "HypPart1"
        out.append({"hpos": int(attrs.get("HPOS", 0)), "vpos": int(attrs.get("VPOS", 0)),
                    "text": normalise(" ".join(words)), "hyphen": hyphen})
    return out


def read_pdf_lines(pdf: Path, first: int, last: int) -> pd.DataFrame:
    """The text lines of PDF pages ``first``–``last`` (1-based) from the file's own text
    layer, in the shape of :func:`read_lines`. The Internet Archive marks a hyphenated
    line end with ``¬``; there is no geometry, so ``hpos``/``vpos`` are 0 and the line
    number."""
    from pypdf import PdfReader
    reader = PdfReader(str(pdf))
    rows = []
    for page in range(first, last + 1):
        text = reader.pages[page - 1].extract_text() or ""
        for i, raw in enumerate(text.splitlines(), 1):
            line = normalise(raw)
            hyphen = line.endswith("¬")
            rows.append({"page": page, "line": i, "hpos": 0, "vpos": i,
                         "text": line.rstrip("¬").rstrip(), "hyphen": hyphen})
    return pd.DataFrame(rows, columns=["page", "line", "hpos", "vpos", "text", "hyphen"])


def read_lines(pages_dir: Path) -> pd.DataFrame:
    """Every ALTO line of an item's page cache, keyed by page position and line number."""
    rows = []
    for f in sorted(Path(pages_dir).glob("*.alto.xml")):
        page = int(f.name[:4])
        for i, ln in enumerate(alto_lines(f), 1):
            rows.append({"page": page, "line": i, **ln})
    return pd.DataFrame(rows, columns=["page", "line", "hpos", "vpos", "text", "hyphen"])

def assign_sections(lines: pd.DataFrame) -> pd.DataFrame:
    """Tag each line with the list it belongs to (``kind``; ``None`` outside a list) and
    whether it lies under a *Seznam nových členů* heading; headings and page furniture
    get ``kind=None``."""
    kind, new = None, False
    kinds, news = [], []
    for text in lines["text"]:
        short = len(text) < 110
        hit = next((k for k, rx in HEADINGS if short and rx.search(text)), None)
        if short and _NEW_MEMBERS.search(text):
            kind, new = None, True
            kinds.append(None)
        elif hit:
            kind = hit
            kinds.append(None)
        elif _STOP.search(text):
            kind, new = None, False
            kinds.append(None)
        elif (_PAGE_NO.match(text) or _HEADING_TAIL.search(text) or _FOOTNOTE.match(text)
              or (text.rstrip().endswith("složili:") and not _PAID_YEAR.search(text))):
            kinds.append(None)
        else:
            kinds.append(kind)
        news.append(new)
        if kind and _CONTINUED.search(text):
            kind = None
    out = lines.copy()
    out["kind"], out["new_members"] = kinds, news
    return out

def collapse_spacing(name: str, *, strict: bool = False) -> str:
    """Undo the letter-spacing of names set in *Sperrsatz* (``D r ej šuk`` → ``Drejšuk``,
    ``A b e 1 o v á`` → ``Abelová``) and a surname the OCR split (``Dur dík`` → ``Durdík``).

    ``strict`` (for whole lines of text rather than a name) joins a lower-case fragment
    only to a one- or two-letter capitalised one, so *Jan farář* stays two words."""
    short_cap = rf"[{_UC}][{_LC}]{{0,1}}" if strict else rf"[{_UC}][{_LC}]{{0,2}}"
    toks = name.split()
    out: list[str] = []
    merged_last = False
    i = 0
    while i < len(toks):
        # a run of three or more one- or two-letter tokens is a spaced-out word
        j = i
        while j < len(toks) and re.fullmatch(rf"[{_UC}{_LC}1]{{1,2}}", toks[j]):
            j += 1
        if j - i >= 3 and any(len(t) == 1 for t in toks[i:j]):
            word = "".join(toks[i:j])
            word = word[0] + word[1:].replace("1", "l")
            out.append(word)
            merged_last = True
            i = j
            continue
        t = toks[i]
        prev = out[-1] if out else ""
        if (prev and re.fullmatch(rf"[{_LC}]+\.?,?", t) and t.rstrip(".,") not in _PARTICLES
                and not prev.endswith((".", ",")) and prev not in _SHORT_WORDS
                and (merged_last or re.fullmatch(short_cap, prev))):
            out[-1] = prev + t
            merged_last = True
        else:
            out.append(t)
            merged_last = False
        i += 1
    return " ".join(out)


def split_residence(text: str) -> tuple[str, str, str]:
    """``(before, preposition, residence)``: the first ``v|ve|na|z|ze|ku|při`` + capital
    word starts the residence; ``u`` only where no other preposition does."""
    hits = [m for m in _PLACE.finditer(text) if m.group(1) != "u"] or list(_PLACE.finditer(text))
    if not hits:
        return text.strip(" ,"), "", ""
    m = hits[0]
    return text[:m.start()].strip(" ,"), m.group(1), text[m.end():].strip(" ,.")


def is_corporate(name: str) -> bool:
    return name.startswith(CORPORATE_HEADS)


def parse_person(text: str) -> dict:
    """Honorific, name, occupation, residence and amount of one entry's text."""
    out = {"honorific": "", "name": "", "occupation": "", "residence": "",
           "residence_prep": "", "amount_zl": pd.NA}
    text = _CONTINUED.sub("", text).strip(" ,;")
    m = _AMOUNT.search(text)
    if m:
        out["amount_zl"] = int(m.group("a") or m.group("b"))
        text = text[:m.start()].strip(" ,")
    m = _HONORIFIC.match(text)
    if m:
        out["honorific"] = re.sub(r"P\.\s*T\.\s*(?:pánové|pp\.)\s*:?", "", m.group(0)).strip()
        text = text[m.end():]
    head, _, rest = text.partition(",")
    if not rest:                                  # no comma: the name runs to the place
        head, prep, place = split_residence(text)
        out.update(name=head, residence_prep=prep, residence=place)
    else:
        before, prep, place = split_residence(rest)
        if prep == "" and _PLACE.search(head):    # "Durasová Jaroslava v Roudnici, r. 49"
            head, prep, place = split_residence(head)
            before = rest
        out.update(name=head.strip(), occupation=before.strip(" ,"),
                   residence_prep=prep, residence=place)
    out["name"] = collapse_spacing(out["name"].strip(" ,"))
    return out


def surname_of(name: str, order: str) -> str:
    """The surname, as far as the printed order allows: the first word for lists printed
    surname first, the last capitalised word for lists printed given name first."""
    words = [w for w in re.split(r"\s+", name) if w and w.strip(".,") not in _PARTICLES]
    caps = [w for w in words if re.match(rf"[{_UC}]", w) and not _GIVEN_ABBREV.match(w)]
    if order != "surname_first" and len(caps) > 1:
        caps = [w for w in caps if w not in GIVEN_NAMES] or caps
    if not caps:
        return ""
    if order == "given_first":
        return caps[-1].strip(".,")
    if order == "mixed" and len(words) >= 2 and (_GIVEN_ABBREV.match(words[0])
                                                 or words[0] in GIVEN_NAMES):
        return caps[-1].strip(".,")
    return caps[0].strip(".,")

    """A Roman numeral as the OCR printed it, or ``None`` if it cannot be read as one.

    Repairs the common misreadings (``l``/``1``/``í`` for I, ``H`` for II, ``Ш`` for III,
    a final ``L`` for I) and accepts the result only if it is a well-formed numeral."""
    s = raw.replace(" ", "").replace("Ш", "III").replace("H", "II").translate(_ROMAN_FIX)
    candidates = [s]
    if s.endswith("L") and len(s) > 1 and s[-2] == "I":
        candidates.append(s[:-1] + "I")
    for c in candidates:
        if re.fullmatch(r"[IVXLC]+", c):
            n = _roman_value(c)
            if n and int_to_roman(n) == c:
                return n
    return None


def _roman_value(s: str) -> int:
    vals = {"I": 1, "V": 5, "X": 10, "L": 50, "C": 100}
    total = 0
    for a, b in zip(s, s[1:] + " "):
        v = vals[a]
        total += -v if b != " " and vals[b] > v else v
    return total


def int_to_roman(n: int) -> str:
    out = ""
    for v, r in ((100, "C"), (90, "XC"), (50, "L"), (40, "XL"), (10, "X"), (9, "IX"),
                 (5, "V"), (4, "IV"), (1, "I")):
        while n >= v:
            out, n = out + r, n - v
    return out


def mc_complete(text: str, kind: str) -> bool:
    """Whether an accumulated founder line already ends with its membership number."""
    rx = _MC_DECEASED if kind == "founder_deceased" else _MC_LIVING
    return bool(rx.search(text))


_ENTRY_START = re.compile(rf"^(?:{_HONORIFIC.pattern[1:]})?[{_UC}][\w'’\- ]{{1,40}},")


def assemble_mc(block: pd.DataFrame, kind: str) -> list[dict]:
    """Join a Matice česká block's lines into founder entries: a line continues the entry
    above only while that entry still lacks its membership number."""
    units: list[dict] = []
    acc: dict | None = None
    for row in block.itertuples(index=False):
        text = row.text
        if acc and not mc_complete(acc["text"], kind) and not (
                _ENTRY_START.match(text) and mc_complete(text, kind)):
            joiner = "" if acc["hyphen"] or acc["text"].endswith("-") else " "
            acc["text"] = acc["text"].rstrip("-") + joiner + text
            acc["hyphen"] = row.hyphen
            continue
        if acc:
            units.append(acc)
        acc = {"page": row.page, "line": row.line, "text": text, "hyphen": row.hyphen}
    if acc:
        units.append(acc)
    return [u for u in units if len(re.findall(rf"[{_UC}{_LC}]", u["text"])) >= 8]


def parse_mc(text: str, kind: str) -> dict:
    """Fields of one Matice česká founder entry."""
    out = {"entry_year": pd.NA, "member_no": pd.NA, "member_no_raw": "", "death_year": pd.NA,
           "death_date": "", "flags": []}
    if kind == "founder_deceased":
        m = _MC_DECEASED.search(text)
        if m:
            out["member_no"] = int(re.sub("[OÓ]", "0", m.group("no")).replace("Í", "1"))
            out["member_no_raw"] = m.group("no")
            if m.group("year"):
                out["death_year"] = 1800 + int(m.group("year"))
                out["death_date"] = (m.group("date") or "").strip(" ,.")
            text = text[:m.start()]
        else:
            out["flags"].append("no_member_no")
    else:
        m = _MC_LIVING.search(text)
        if m:
            year = int(re.sub("[OÓ]", "0", m.group("year")).replace("Í", "1"))
            out["entry_year"] = 1800 + year if 31 <= year <= 99 else pd.NA
            if pd.isna(out["entry_year"]):
                out["flags"].append("entry_year_unread")
            raw = m.group("no").strip()
            out["member_no_raw"] = raw
            if re.fullmatch(r"[0-9OÓÍ]{1,5}", raw):
                out["member_no"] = int(re.sub("[OÓ]", "0", raw).replace("Í", "1"))
            else:
                n = roman_to_int(raw)
                out["member_no"] = n if n is not None else pd.NA
                out["roman"] = True
                if n is None:
                    out["flags"].append("roman_unread")
            text = text[:m.start()]
        else:
            out["flags"].append("no_member_no")
    out["head"] = text.strip(" ,.")
    return out


_DITTO = re.compile(r"^((?:[—–\-]+\s*)+)")
_DITTO_QUOTE = re.compile(r"^((?:[„\"“,]\s*)+)(?=\S)")


def resolve_ditto(head: str, previous: str, previous_ditto: bool = False) -> tuple[str, bool]:
    """Expand the marks that repeat the corporate name above.

    Matice česká uses dashes (``— v Černilově`` after *Beseda čtenářská v Čáslavi* is
    *Beseda čtenářská v Černilove*): a single dash before a place after a name printed
    in full stands for that whole name; otherwise each dash repeats one word of the name
    above (``— Jičínské`` then ``— v Jindřichově Hradci`` after *Gymnasium v Hradci
    Králové* are *Gymnasium Jičínské* and *Gymnasium*). A corporate line opening in lower
    case lost its dash in the OCR and counts one. Matice moravská 1909 uses „ marks (or a
    comma, as the OCR reads them), one per word repeated."""
    q = _DITTO_QUOTE.match(head)
    if q:
        k, rest = len(re.findall(r"[„\"“,]", q.group(1))), head[q.end():].strip()
        return (" ".join(previous.split()[:k] + [rest]).strip(), True) if previous else (head, True)
    m = _DITTO.match(head)
    if m:
        dashes, rest = len(re.findall(r"[—–]+|-{1,2}", m.group(1))), head[m.end():].strip()
    elif re.match(rf"[{_LC}]", head) and not re.match(r"(?:z|ze|von|de)\s", head):
        dashes, rest = 1, head
    else:
        return head, False
    if not previous:
        return head, True
    if (_PLACE.match(rest) or not rest) and dashes == 1 and not previous_ditto:
        return f"{previous} {rest}".strip(), True
    words = previous.split()
    return " ".join(words[:max(dashes, 1)] + [rest]).strip(), True

_ENTRY_OPEN = re.compile(
    rf"^(?:{_HONORIFIC.pattern[1:]})?(?:"
    rf"(?:[{_UC}][\w'’\-]*\.?\s+(?:[{_LC}]{{1,4}}\s+)?){{0,3}}[{_UC}][\w'’\-]*\.?\s*,"  # Surname Given,
    rf"|(?:[{_UC}][{_LC}]{{0,5}}\.\s*){{1,2}}[{_UC}][\w'’\-]+"        # Frant. Jarušek
    rf"|(?:{'|'.join(re.escape(h) for h in CORPORATE_HEADS)})"
    rf"|[{_UC}][\w'’\-]+\s+(?:hrabě|rytíř|kníže|svob|baron|šlechtic)"
    rf"|[{_UC}][\w'’\-]+\s+[{_UC}][\w'’\-]+\s+(?:v|ve|na|z|ze)\s+[{_UC}]"
    # Surname + given name or initial, then an occupation or a place, no comma between
    # (*Musil K. ve Ždaře*, *Tekly Edv. správce hospodářství*: the 1878 list prints so)
    rf"|[{_UC}][\w'’\-]+\s+[{_UC}][\w'’\-]{{0,8}}\.?\s+(?:[{_LC}]{{3,}}|(?:v|ve|na|z|ze)\s+[{_UC}]))")
#: Where the OCR lost the full stop between two entries: before a "Surname Given," that
#: follows a residence (*… v Hranicích Štolba Josef, …*).
_LOST_STOP = rf"\s+(?=[{_UC}][\w'’\-]+\s+[{_UC}][\w'’\-]{{0,8}}\.?,)"
#: Abbreviations a place name may contain before its full stop (``v Uh. Brodě``).
_PLACE_ABBREV = {"Uh", "Uher", "Mor", "Král", "Nov", "Val", "Č", "Něm", "Vel", "Kr", "Mnich",
                 "Mal", "Hor", "Dol", "Stř", "Vys", "Bystř", "Ml", "St", "Sv", "n", "p", "nad",
                 "pod", "Kut", "Jindř", "Hrad", "Dom", "Vid", "Mal", "Malé", "Brod", "sv"}
_PLACE_HEADER = re.compile(
    rf"(?:(?<=^)|(?<=[.;—]\s)|(?<=—))"
    rf"((?:[{_UC}][\w\-]*\.?\s?){{1,4}}?|(?:[{_UC}{_LC}]\s){{2,}}[{_LC}])\s*:\s+")


def join_lines(block: pd.DataFrame) -> tuple[str, list[int], list[tuple[int, int]]]:
    """A block's lines as one string (hyphenated words rejoined), with the offset each
    line starts at and its (page, line), for tracing an entry back to the page."""
    text, starts, where = "", [], []
    hyphen = False
    for row in block.itertuples(index=False):
        piece = collapse_spacing(_CONTINUED.sub("", row.text).strip(), strict=True)
        if not piece:
            continue
        if text:
            if hyphen or (text.endswith("-") and not _HYPHEN_AFTER_AMOUNT.search(text[-16:])):
                text = text.rstrip("-")            # a word broken at the line, not "21 K.-"
            else:
                text += " "
        starts.append(len(text))
        where.append((row.page, row.line))
        text += piece
        hyphen = row.hyphen
    return text, starts, where


def split_run_on(text: str, *, need_place: bool = True) -> list[tuple[int, str]]:
    """Cut a run-on paragraph into entries; returns ``(offset, entry text)`` pairs.

    A comma, full stop or semicolon ends an entry when (1) the text since the last cut
    holds a residence or an amount paid — or, inside a place group (``need_place=False``),
    a comma — (2) the word before a full stop is not an abbreviation a place name carries
    (``Uh.``, ``Mor.``) or an initial, and (3) the text after it opens a new name."""
    cuts = [0]
    # a separator, or the space before a priest's or doctor's title where the OCR lost it
    for m in re.finditer(rf"[,.;]\s+|\s+(?=(?:P|Dr|JUDr|MUDr)\.\s*[{_UC}])|{_LOST_STOP}", text):
        left = text[cuts[-1]:m.start()]
        right = text[m.end():]
        if not _ENTRY_OPEN.match(right):
            continue
        last = re.findall(r"[\w'’]+$", left)
        last = last[0] if last else ""
        sep = text[m.start()] if text[m.start()] in ",.;" else " "
        if sep == "." and (last in _PLACE_ABBREV or re.fullmatch(rf"[{_UC}]", last)):
            continue
        if sep == " " and not _PLACE.search(left):
            continue
        has_amount = bool(re.search(r"zl\.?\s*\d+$|\d+\s*zl\.?$", left))
        has_place = bool(_PLACE.search(left))
        if has_amount or (has_place and need_place) or (not need_place and ("," in left
                                                                            or is_corporate(left))):
            cuts.append(m.end())
    pieces = []
    for a, b in zip(cuts, cuts[1:] + [len(text)]):
        piece = text[a:b].strip(" ,.;—-")
        if piece:
            pieces.append((a, piece))
    return pieces


def split_by_place(text: str) -> list[tuple[int, str, str]] | None:
    """``Place: entries. Place: entries.`` → ``(offset, place, entry text)``; ``None`` if
    the paragraph is not grouped by place (fewer than two place headings)."""
    heads = list(_PLACE_HEADER.finditer(text))
    if len(heads) < 2:
        return None
    out = []
    for h, nxt in zip(heads, heads[1:] + [None]):
        body_start, body_end = h.end(), (nxt.start() if nxt else len(text))
        place = collapse_spacing(h.group(1).strip())
        for off, piece in split_run_on(text[body_start:body_end], need_place=False):
            out.append((body_start + off, place, piece))
    return out


def column_margins(hpos: pd.Series, gap: int = 150) -> list[float]:
    """The left margins of a page's text columns: line starts cluster by column (a
    hanging indent stays inside its column's cluster), and each cluster's margin is its
    10th percentile."""
    xs = sorted(hpos)
    groups = [[xs[0]]]
    for x in xs[1:]:
        if x - groups[-1][-1] > gap:
            groups.append([x])
        else:
            groups[-1].append(x)
    return [float(pd.Series(g).quantile(0.1)) for g in groups if len(g) >= 3] or [float(xs[0])]


def assemble_lines(block: pd.DataFrame) -> list[dict]:
    """Join an indented list's lines into entries, column by column: a line set in from
    its column's left margin by more than 15 px continues the entry above."""
    units: list[dict] = []
    for page, lines in block.groupby("page", sort=True):
        margins = column_margins(lines["hpos"])
        lines = lines.assign(
            col=[max([i for i, m in enumerate(margins) if m <= h + 20], default=0)
                 for h in lines["hpos"]]).sort_values(["col", "vpos"], kind="stable")
        last_col = None
        for row in lines.itertuples(index=False):
            text = collapse_spacing(row.text, strict=True)
            margin = margins[row.col]
            cont = units and units[-1]["page"] == page and last_col == row.col and (
                row.hpos > margin + 15 or units[-1]["hyphen"] or units[-1]["text"].endswith("-")
            ) and not _DITTO_QUOTE.match(text)
            last_col = row.col
            if cont:
                prev = units[-1]
                joiner = "" if prev["hyphen"] or prev["text"].endswith("-") else " "
                prev["text"] = prev["text"].rstrip("-") + joiner + text
                prev["hyphen"] = row.hyphen
            else:
                units.append({"page": page, "line": row.line, "text": text,
                              "hyphen": row.hyphen})
    return units

ENTRY_COLUMNS = [
    "entry_id", "source", "society", "list_date", "list_kind", "new_members", "page", "line",
    "text", "honorific", "name", "surname", "is_corporate", "ditto", "occupation",
    "residence_prep", "residence", "place_group", "amount_zl", "paid_for_year", "entry_year",
    "member_no", "member_no_raw", "death_year", "death_date", "flags",
]


def paid_year_segments(text: str) -> list[tuple[int, str, int | None]]:
    """Cut a paragraph at its *Za rok NNNN* markers: ``(offset, segment, year)``, the year
    ``None`` before the first marker."""
    out, pos, year = [], 0, None
    for m in _PAID_YEAR.finditer(text):
        if m.start() > pos:
            out.append((pos, text[pos:m.start()], year))
        pos, year = m.end(), int(m.group(1))
    out.append((pos, text[pos:], year))
    return [(o, s, y) for o, s, y in out if s.strip(" ,.;")]


def _blocks(lines: pd.DataFrame):
    """Consecutive runs of lines in the same list (kind, new_members)."""
    keep = lines[lines["kind"].notna()]
    if keep.empty:
        return
    run = (keep[["kind", "new_members"]] != keep[["kind", "new_members"]].shift()).any(axis=1)
    gap = keep.index.to_series().diff().fillna(1) > 50          # far apart: a new block
    for _, block in keep.groupby((run | gap).cumsum(), sort=True):
        yield block.iloc[0]["kind"], bool(block.iloc[0]["new_members"]), block


def parse_source(key: str, path: Path) -> pd.DataFrame:
    """All entries of one downloaded item, one row each (columns ``ENTRY_COLUMNS``).
    ``path`` is the item's ``_pages/`` cache, or the PDF for a source read from a PDF."""
    cfg = SOURCES[key]
    raw = read_pdf_lines(path, *cfg["pdf_pages"]) if "pdf" in cfg else read_lines(path)
    lines = assign_sections(raw)
    rows: list[dict] = []
    for kind, new, block in _blocks(lines):
        if cfg["layout"] == "mc":
            rows += _mc_rows(block, kind)
        elif cfg["layout"] == "lines":
            previous = ""
            for u in assemble_lines(block):
                head, ditto = resolve_ditto(u["text"], previous) if _DITTO_QUOTE.match(
                    u["text"]) else (u["text"], False)
                fields = parse_person(head.rstrip(" ."))
                if ditto:
                    fields["name"], fields["is_corporate"] = fields["name"], True
                previous = fields["name"]
                rows.append({**u, "text": u["text"], "list_kind": kind, "new_members": new,
                             "ditto": ditto, **fields})
        else:
            text, starts, where = join_lines(block)
            for base, segment, paid in paid_year_segments(text):
                grouped = split_by_place(segment)
                pieces = (grouped if grouped is not None
                          else [(o, "", p) for o, p in split_run_on(segment)])
                for off, place, piece in pieces:
                    page, line = where[max(bisect_right(starts, base + off) - 1, 0)]
                    fields = parse_person(piece)
                    if place and not fields["residence"]:
                        fields["residence"] = place
                    rows.append({"page": page, "line": line, "text": piece, "list_kind": kind,
                                 "new_members": new, "place_group": place,
                                 "paid_for_year": paid, **fields})
    df = pd.DataFrame(rows)
    if df.empty:
        return pd.DataFrame(columns=ENTRY_COLUMNS)
    df["source"], df["society"], df["list_date"] = key, cfg["society"], cfg["list_date"]
    df = _finish(df, key)
    df["surname"] = [surname_of(n, cfg["name_order"]) if not c else ""
                     for n, c in zip(df["name"], df["is_corporate"])]
    return df


def _mc_rows(block: pd.DataFrame, kind: str) -> list[dict]:
    rows, previous_corporate, previous_ditto = [], "", False
    for u in assemble_mc(block, kind):
        f = parse_mc(u["text"], kind)
        roman = f.pop("roman", False)
        corporate = roman or kind == "founder_corporate"
        head, ditto = (resolve_ditto(f.pop("head"), previous_corporate, previous_ditto)
                       if corporate else (f.pop("head"), False))
        if corporate:
            name, prep, place = split_residence(head)
            fields = {"honorific": "", "name": name, "occupation": "", "residence_prep": prep,
                      "residence": place, "amount_zl": pd.NA}
            previous_corporate, previous_ditto = name, ditto
        else:
            fields = parse_person(head)
        list_kind = ("founder_corporate" if corporate and kind == "founder"
                     else kind)
        rows.append({**u, **fields, **f, "list_kind": list_kind, "new_members": False,
                     "is_corporate": corporate, "ditto": ditto})
    return rows


def _finish(df: pd.DataFrame, key: str) -> pd.DataFrame:
    """Defaults, flags, the key, and the column order."""
    for col, default in (("is_corporate", None), ("ditto", False), ("place_group", ""),
                         ("member_no_raw", ""), ("death_date", ""), ("flags", None),
                         ("surname", "")):
        if col not in df:
            df[col] = default
    for col in ("entry_year", "member_no", "death_year", "amount_zl", "paid_for_year"):
        if col not in df:
            df[col] = pd.NA
        df[col] = pd.array(df[col].astype("object").where(df[col].notna(), None), dtype="Int64")
    corp = [is_corporate(n) for n in df["name"]]
    df["is_corporate"] = [c if isinstance(c, bool) else k for c, k in zip(df["is_corporate"], corp)]
    flags = []
    for r in df.itertuples(index=False):
        f = list(r.flags) if isinstance(r.flags, list) else []
        if not r.name:
            f.append("no_name")
        if not r.residence:
            f.append("no_residence")
        if len(r.text) > 160:
            f.append("long_entry")
        flags.append(";".join(dict.fromkeys(f)))
    df["flags"] = flags
    df = df.sort_values(["page", "line"], kind="stable").reset_index(drop=True)
    df["entry_id"] = [f"{key}:{i + 1:05d}" for i in range(len(df))]
    return df[ENTRY_COLUMNS]


def parse_all(library_dir: Path, keys: list[str] | None = None) -> pd.DataFrame:
    """Every source whose page cache is on disk, stacked (key ``entry_id``)."""
    frames = []
    for key in keys or list(SOURCES):
        cfg = SOURCES[key]
        path = Path(library_dir) / (cfg["pdf"] if "pdf" in cfg else f"{cfg['stem']}_pages")
        if path.exists():
            frames.append(parse_source(key, path))
    return pd.concat(frames, ignore_index=True) if frames else pd.DataFrame(columns=ENTRY_COLUMNS)

def totals_report(entries: pd.DataFrame) -> pd.DataFrame:
    """Entries found against the totals each society printed (``PRINTED_TOTALS``)."""
    found = entries.groupby(["source", "list_kind"]).size()
    rows = []
    for (src, kind), printed in PRINTED_TOTALS.items():
        n = int(found.get((src, kind), 0))
        rows.append({"source": src, "list_kind": kind, "printed": printed, "found": n,
                     "diff": n - printed, "pct": round(100 * n / printed, 1)})
    return pd.DataFrame(rows)


def mc_number_order(entries: pd.DataFrame, window: int = 31, tolerance: int = 5) -> pd.Series:
    """Matice česká numbers were issued in order of entry, so a person's entry year should
    match those of the founders numbered next to them. ``True`` where it differs from the
    median of the ``window`` nearest numbers by more than ``tolerance`` years — an OCR
    misreading of the year or the number, or a printing slip."""
    out = pd.Series(False, index=entries.index)
    mc = entries[(entries["society"] == "MC") & entries["entry_year"].notna()
                 & entries["member_no"].notna() & ~entries["is_corporate"].astype(bool)]
    for _, grp in mc.groupby("source"):
        g = grp.sort_values("member_no")
        med = g["entry_year"].astype(float).rolling(window, center=True, min_periods=5).median()
        out.loc[g.index] = (g["entry_year"].astype(float) - med).abs() > tolerance
    return out


def entry_year_from_number(entries: pd.DataFrame) -> pd.Series:
    """For a deceased Matice česká founder (no entry year printed), the entry year of the
    living founder with the nearest lower number in either issue; ``<NA>`` elsewhere."""
    ref = entries[(entries["society"] == "MC") & entries["entry_year"].notna()
                  & entries["member_no"].notna() & ~entries["is_corporate"].astype(bool)
                  & ~mc_number_order(entries)]
    ref = (ref[["member_no", "entry_year"]].astype("int64").drop_duplicates("member_no")
           .sort_values("member_no"))
    dead = entries[(entries["list_kind"] == "founder_deceased") & entries["member_no"].notna()]
    if ref.empty or dead.empty:
        return pd.Series(pd.NA, index=entries.index, dtype="Int64")
    probe = dead[["member_no"]].astype("int64").reset_index().sort_values("member_no")
    hit = pd.merge_asof(probe, ref, on="member_no", direction="backward")
    out = pd.Series(pd.NA, index=entries.index, dtype="Int64")
    out.loc[hit["index"].to_numpy()] = hit["entry_year"].astype("Int64").to_numpy()
    return out

def occupation_class(occupation: str, honorific: str = "", corporate: bool = False) -> str:
    """The class of a member's occupation (``OCCUPATION_CLASSES``): ``corporate`` for a
    society, school or town, ``clergy`` for a priest's *P.* with no occupation printed,
    ``unknown`` where nothing matches."""
    if corporate:
        return "corporate"
    text = f"{occupation or ''} ".lower()
    for cls, rx in OCCUPATION_CLASSES:
        if re.search(rx, text):
            return cls
    if re.match(r"P\.", honorific or ""):
        return "clergy"
    return "unknown"


def member_group(cls: str) -> str:
    """``posted`` (clergy, teachers, officials), ``local`` (the rest of the occupied),
    or the class itself (``student``, ``corporate``, ``unknown``)."""
    return "posted" if cls in POSTED else "local" if cls in LOCAL else cls

#: Which entry stands for a Matice česká founder listed more than once: the most recent
#: list he appears in (his latest residence), the living list before the deceased one.
_MC_PRIORITY = {("mc1890", "founder"): 0, ("mc1890", "founder_deceased"): 1,
                ("mc1873", "founder"): 2, ("mc1873", "founder_deceased"): 3,
                ("mc1890", "founder_corporate"): 0, ("mc1873", "founder_corporate"): 2}


def mc_founders(entries: pd.DataFrame, corporate: bool = False) -> pd.DataFrame:
    """Every Matice česká founder once: entries sharing a membership number and a surname
    (:func:`founder_key`) are one founder, kept at the most recent list he appears in
    (``_MC_PRIORITY``); an entry whose number was not read stands alone."""
    mc = entries[(entries["society"] == "MC")
                 & (entries["is_corporate"].astype(bool) == corporate)].copy()
    mc["_prio"] = [_MC_PRIORITY.get((s, k), 9) for s, k in zip(mc["source"], mc["list_kind"])]
    mc["_who"] = founder_key(mc)
    numbered = mc[mc["member_no"].notna()].sort_values("_prio").drop_duplicates("_who")
    return pd.concat([numbered, mc[mc["member_no"].isna()]]).drop(columns=["_prio", "_who"])


def founder_key(entries: pd.DataFrame) -> pd.Series:
    """Who a numbered founder is: the membership number and the first three letters of the
    surname, folded. The number alone is not enough — the OCR misreads a digit, and in the
    1890 living list 58 of the 59 numbers printed twice belong to two different surnames
    (*Hofer Jan … č. 262*, *Hradecký Jan … c. 262*); three letters keep an OCR variant of
    one surname together (*S tefan*, *Štefán*). Corporate founders, who print no surname,
    are told apart by their name."""
    who = [(_fold(s) or _fold(n))[:3] for s, n in zip(entries["surname"].fillna(""),
                                                     entries["name"].fillna(""))]
    return entries["member_no"].astype("string") + "|" + pd.Series(who, index=entries.index)


def eb_rate(count: pd.Series, exposure: pd.Series) -> pd.Series:
    """Rates shrunk toward the overall rate in proportion to their noise — the global
    empirical-Bayes estimator of Marshall (1991, *Applied Statistics* 40:283–294):

    .. math:: \\hat r_i = m + C_i (r_i - m), \\quad C_i = \\frac{A}{A + m / n_i},
              \\quad A = \\max\\Big(0,\\; \\frac{\\sum n_i (r_i - m)^2}{\\sum n_i}
              - \\frac{m}{\\bar n}\\Big)

    with :math:`r_i = y_i / n_i` and :math:`m = \\sum y_i / \\sum n_i`."""
    y, n = count.astype(float), exposure.astype(float)
    ok = n > 0
    r = (y / n).where(ok)
    m = y[ok].sum() / n[ok].sum()
    a = max(0.0, (n[ok] * (r[ok] - m) ** 2).sum() / n[ok].sum() - m / n[ok].mean())
    c = a / (a + m / n)
    return (m + c * (r - m)).where(ok)


def measure_groups(e: pd.DataFrame, payers: pd.DataFrame | None = None) -> dict[str, pd.DataFrame]:
    """The member counts the measure is built from: column name → the rows counted. With
    *payers* (:func:`payment_persons`), also the Matice česká joiners — first payment in
    ``JOINER_WINDOW`` (``mc_joiners``, Bohemia's measure since D-184) — the local ones
    among them, those whose payments reached the fee, and the joiners of 1891–1913."""
    persons = mc_founders(e)
    not_corp = ~e["is_corporate"].astype(bool)
    joined = {}
    if payers is not None:
        joined = {
            MC_OWN: joiners(payers),
            f"{MC_OWN}_local": joiners(payers)[lambda d: d["member_group"] == "local"],
            f"{MC_OWN}_paid": joiners(payers, founders_only=True),
            "mc_joiners_1891_1913": joiners(payers, (1891, 1913)),
        }
    return {
        **joined,
        "mc_founders": persons,
        "mc_founders_posted": persons[persons["member_group"] == "posted"],
        "mc_founders_local": persons[persons["member_group"] == "local"],
        "mc_living_1873": e[(e["source"] == "mc1873") & (e["list_kind"] == "founder")],
        "mc_living_1890": e[(e["source"] == "mc1890") & (e["list_kind"] == "founder")],
        "mc_corporate": mc_founders(e, corporate=True),
        "mm_members_1890": e[(e["source"] == "mm1891") & not_corp],
        "mm_members_1909": e[(e["source"] == "mm1909") & not_corp],
        "mm_members_1909_local": e[(e["source"] == "mm1909") & (e["member_group"] == "local")],
        "mm_members_1876_78": e[e["source"].isin(["mm1876", "mm1877", "mm1878"]) & not_corp
                                ].drop_duplicates(["surname", "residence"]),
    }


def okres_measure(entries: pd.DataFrame, census: pd.DataFrame,
                  crosswalk: pd.DataFrame, payers: pd.DataFrame | None = None) -> pd.DataFrame:
    """Members per 1910 census okres (key ``census_okres_id``, all of them, zeros kept):
    each count of :func:`measure_groups`, its rate per 1,000 inhabitants and per 1,000
    Czech speakers (1910 census), and the main rates shrunk by :func:`eb_rate`. Only
    entries placed in an okres count; refused and unmatched residences do not.
    ``matice_own_z`` and its robustness forms (:func:`add_own_society`) need *payers*
    (Bohemia's measure, D-184)."""
    xw = crosswalk[["okres_id", "census_okres_id"]]
    e = entries[entries["okres_id"].notna()].merge(xw, on="okres_id", how="left", validate="m:1")
    if payers is not None:
        payers = payers[payers["okres_id"].notna()].merge(xw, on="okres_id", how="left",
                                                          validate="m:1")
    base = census[["census_okres_id", "okres_cs", "land", "pop_total", "lang_czech",
                   "lang_czech_share"]].copy()
    groups = measure_groups(e, payers)
    for col, rows in groups.items():
        counts = rows.groupby("census_okres_id").size().rename(col)
        base = base.merge(counts, left_on="census_okres_id", right_index=True, how="left",
                          validate="1:1")
        base[col] = base[col].fillna(0).astype(int)
        base[f"{col}_per_1000"] = 1000 * base[col] / base["pop_total"]
        base[f"{col}_per_1000_czech"] = (1000 * base[col] / base["lang_czech"]).where(
            base["lang_czech"] > 0)
    for col in [c for c in SHRUNK if c in groups]:
        base[f"{col}_per_1000_eb"] = 1000 * eb_rate(base[col], base["pop_total"])
    base = add_own_society(base, PRAGUE_CENSUS_OKRES_ID)
    return base


#: The Matice moravská list that stands for Moravia and Silesia: the roster of
#: 1 March 1909, the one nearest the 1910 census and the war (D-182; the 1890 list until then).
MM_OWN = "mm_members_1909"
#: What stands for Bohemia: the Matice česká joiners, payers whose first payment for a
#: founder's share falls in ``JOINER_WINDOW``, read off the monthly payment lists (D-184;
#: the founders of 1831–1890 until then).
MC_OWN = "mc_joiners"
#: The counts that are also given as shrunk rates.
SHRUNK = (MC_OWN, f"{MC_OWN}_local", f"{MC_OWN}_paid", "mc_founders", "mc_founders_local",
          "mm_members_1890", "mm_members_1909")

#: Each land's own society: Matice česká recruited in Bohemia, Matice moravská in
#: Moravia and Silesia (placed members 3,017 vs 305; Matice moravská 475 vs 107 in 1909,
#: 478 vs 79 in 1890 — D-161, counts of `run_matice_pilot.py` on 2026-11-02).
OWN_SOCIETY = {"Böhmen": f"{MC_OWN}_per_1000_eb", "Mähren": f"{MM_OWN}_per_1000_eb",
               "Schlesien": f"{MM_OWN}_per_1000_eb"}


#: The combined measure is standardised on a log scale (D-185), as Squicciarini and
#: Voigtländer (2015, §3.1) use ln(1 + subscriptions per 1,000) "to reduce the influence of
#: extreme values": here ln(1 + members per ``LOG_PER`` inhabitants). The unit is chosen so
#: the +1 is small beside the typical rate — the median okres has about 3 Bohemian joiners
#: and 8 Moravian members per 100,000 — where per 1,000, as theirs, rates of 0.03 would sit
#: on the linear stretch of the log and leave Prague at z ≈ 7.4 (3.1 per 100,000).
LOG_PER = 100_000
#: Census okres Praha, the city itself; its suburbs (Karlín, Královské Vinohrady, Smíchov,
#: Žižkov) are okresy of their own. The seat of the Matice česká, where students and
#: officials from all of Bohemia paid while they lived there — every regression on the
#: measure carries a dummy for it, as theirs does for Paris (D-185).
PRAGUE_CENSUS_OKRES_ID = 301072
#: Each land's own society's member count, beside ``OWN_SOCIETY``'s rate.
OWN_COUNT = {"Böhmen": MC_OWN, "Mähren": MM_OWN, "Schlesien": MM_OWN}
_TERRITORY = {"Böhmen": "MC", "Mähren": "MM", "Schlesien": "MM"}


def _own(measure: pd.DataFrame, columns: dict[str, str]) -> pd.Series:
    return pd.Series([measure.at[i, columns[land]] if land in columns else None
                      for i, land in measure["land"].items()], index=measure.index, dtype=float)


def own_society_z(measure: pd.DataFrame, log: bool = True) -> pd.Series:
    """One measure across the Czech lands: the shrunk member rate of the unit's own land's
    society (``OWN_SOCIETY``), as ln(1 + members per ``LOG_PER``) (D-185; ``log=False`` the
    rate itself, as before), standardised within that society's territory (mean 0, s.d. 1)
    — Bohemia for Matice česká, Moravia and Silesia together for Matice moravská, since
    Silesia alone (7 placed members) is too thin to standardise. Provisional (D-161)."""
    rate = _own(measure, OWN_SOCIETY)
    if log:
        rate = (rate * LOG_PER / 1000).map(math.log1p)
    territory = measure["land"].map(_TERRITORY)
    return rate.groupby(territory).transform(lambda s: (s - s.mean()) / s.std(ddof=0))


#: Below this many Czech speakers a rate per Czech speaker is noise (Aš has 5; one posted
#: official among Podbořany's 1,350 would put it near the top), so the unit has no per-Czech form.
MIN_CZECH_SPEAKERS = 1000


def own_society_czech(measure: pd.DataFrame) -> tuple[pd.Series, pd.Series]:
    """The second denominator (D-186): the own society's members per 1,000 Czech speakers
    (1910 Umgangssprache), shrunk (:func:`eb_rate`) over the units with at least
    ``MIN_CZECH_SPEAKERS``, and its log z within territory as :func:`own_society_z` builds
    it per inhabitant. A unit with fewer Czech speakers is ``NaN`` in both."""
    czech = measure[measure["lang_czech"] >= MIN_CZECH_SPEAKERS].copy()
    for col in (MC_OWN, MM_OWN):
        czech[f"{col}_per_1000_eb"] = 1000 * eb_rate(czech[col], czech["lang_czech"])
    return (_own(czech, OWN_SOCIETY).reindex(measure.index),
            own_society_z(czech).reindex(measure.index))


def add_own_society(measure: pd.DataFrame, prague_id: int) -> pd.DataFrame:
    """The combined measure and its robustness forms: ``matice_own_z`` (log, the baseline,
    D-185), ``matice_own_z_level`` (the rate, standardised as it stands), ``matice_own_any``
    (the own society has at least one member placed in the unit), the same per Czech
    speaker (``matice_own_per_1000_czech``, ``matice_own_z_czech``; D-186) and ``prague``
    (the unit is Prague, the dummy every regression on the measure carries). ``prague_id``
    is the unit's key value for Prague."""
    key = measure.columns[0]
    measure["matice_own_z"] = own_society_z(measure)
    measure["matice_own_z_level"] = own_society_z(measure, log=False)
    measure["matice_own_per_1000_czech"], measure["matice_own_z_czech"] = own_society_czech(measure)
    count = _own(measure, OWN_COUNT)
    measure["matice_own_any"] = (count > 0).where(count.notna())
    measure["prague"] = measure[key] == prague_id
    return measure

# The same measure one level down: the court district (soudní okres)
# ---------------------------------------------------------------------------
#: Prague's members are placed in okres Praha, not a settlement; its one court.
PRAGUE_COURT_ID = 301072155


def census_court_pool(districts: pd.DataFrame, census_ids: set[int]) -> dict[int, int]:
    """Court → the census court it is counted in. The court census has no row for nine
    courts of a statutory city's rural okres (Liberec, Opava, Olomouc …): it counts each
    with the city court of the same seat, so they are pooled the same way here."""
    d = districts.assign(court_id=districts["court_id"].astype(int))
    pool = {}
    for r in d.itertuples(index=False):
        if r.court_id in census_ids:
            pool[r.court_id] = r.court_id
            continue
        twin = d[(d["gerichtsbezirk_cs"] == r.gerichtsbezirk_cs) & d["court_id"].isin(census_ids)]
        if len(twin) == 1:
            pool[r.court_id] = int(twin["court_id"].iloc[0])
    return pool


def _with_court(frame: pd.DataFrame, settlement_courts: pd.Series,
                pool: dict[int, int]) -> pd.DataFrame:
    """*frame*'s rows placed in a census court (``census_court_id``), Prague's at its court."""
    court = frame["settlement_id"].map(settlement_courts)
    court = court.where(frame["place_status"] != "prague", PRAGUE_COURT_ID)
    out = frame.assign(census_court_id=court.map(lambda c: pool.get(int(c)) if pd.notna(c) else None))
    return out[out["census_court_id"].notna()]


def court_measure(entries: pd.DataFrame, settlement_courts: pd.Series,
                  court_census: pd.DataFrame, districts: pd.DataFrame,
                  payers: pd.DataFrame | None = None) -> pd.DataFrame:
    """Members per 1910 court district (key ``census_court_id``, zeros kept), placed by
    the court their settlement's point falls in (``settlement_courts``: settlement_id →
    court_id); Prague's go to its court, and a member placed only at okres level is left
    out. Counts, rates per 1,000 inhabitants (court census), shrunk rates, and
    ``matice_own_z`` as for the okres (:func:`own_society_z`)."""
    census_ids = set(court_census["unit_id"].astype(int))
    pool = census_court_pool(districts, census_ids)
    e = _with_court(entries, settlement_courts, pool)
    if payers is not None:
        payers = _with_court(payers, settlement_courts, pool)
    d = districts.assign(court_id=districts["court_id"].astype(int))
    names = d[d["court_id"].isin(census_ids)].set_index("court_id")[["gerichtsbezirk_cs", "okres_cs", "land"]]
    base = (court_census[["unit_id", "rel_zusammen", "umgspr_bohmisch_mahrisch_slowakisch"]]
            .rename(columns={"unit_id": "census_court_id", "rel_zusammen": "pop_total",
                             "umgspr_bohmisch_mahrisch_slowakisch": "lang_czech"})
            .astype({"census_court_id": int}).merge(names, left_on="census_court_id",
                                                    right_index=True, how="left", validate="1:1"))
    base["lang_czech_share"] = base["lang_czech"] / base["pop_total"]
    groups = measure_groups(e, payers)
    for col in [c for c in SHRUNK if c in groups]:
        counts = groups[col].groupby("census_court_id").size().rename(col)
        base = base.merge(counts, left_on="census_court_id", right_index=True, how="left",
                          validate="1:1")
        base[col] = base[col].fillna(0).astype(int)
        base[f"{col}_per_1000"] = 1000 * base[col] / base["pop_total"]
        base[f"{col}_per_1000_eb"] = 1000 * eb_rate(base[col], base["pop_total"])
    base = add_own_society(base, PRAGUE_COURT_ID)
    return base

#: The gulden form, with the OCR's readings of ``zl`` (*zL*, *zJ*, *«1*, *.zl*) and of a
#: kreuzer figure it damaged (*5■)*, *f.O*, *h)*), which :func:`payment_amount` repairs.
_AMOUNT_ZL = re.compile(r"(?P<zl>\d+)\s*\.?\s*(?:[z«][lLJI1|!i]|z\s+i|al)(?!\w)\.?"
                        r"(?:\s*v?\s*,?\s*(?P<kr>\S{1,3}?)\s*\.?\s*v?kr?\b\.{0,2}"
                        r"|\s+(?P<kr50>50)\.?)?$")
#: A founder admitted on the Jungmann foundation, who paid nothing himself.
_FOUNDATION = re.compile(r"\(\s*zakladatel\w*\s+z\s+nadac\w*[^)]*\)?", re.I)
_TRAILING = re.compile(r"[\s.,;:'\"■—–\-]+$")
#: Letters the OCR reads for digits inside an amount (*2Í K*, *1O zl.*), mapped back in a
#: token that holds a real digit and stands right before the currency.
_OCR_DIGITS = str.maketrans({"Í": "1", "í": "1", "I": "1", "l": "1", "|": "1", "L": "1",
                             "i": "1", "O": "0", "o": "0", "C": "0", "Q": "0"})
_AMOUNT_TOKEN = re.compile(r"(?<![\w])([\dÍíIl|LiOoCQ][\dÍíIl|LiOoCQ.]*)"
                           r"(?=\s*(?:[,'\"·*’]\s*[\dOo]{2}\s*)?[*'’f]?\s*(?:K\b|kor|z[lLJI1|!i]?\b|al\b))")
#: A digit the OCR split off its number before the currency (*2 i K*, *2 L K*).
_SPLIT_DIGIT = re.compile(r"(?<=\d)\s(?=[iIlL]\s*(?:K\b|z))")
_SEPARATOR = re.compile(r"\s*[■'’]*\s*(?:[—–]+|-{2,})\s*")
#: A plain hyphen the OCR read for the dash after an amount (*21 K. - Novák …*, *21 K.-*).
_HYPHEN_AFTER_AMOUNT = re.compile(r"(\d\s*(?:K|kr|zl|kor\w*)\.?)\s*-(?=\s|$)")
#: A stray bar between the figure and the currency (*21|K.*), not a digit.
_STRAY_BAR = re.compile(r"(?<=\d\d)[|!](?=\s?K\b)")
_DASH = re.compile(r"[—–]|-{2,}")
#: This many lines in a row with neither an amount nor a dash end a month that printed no
#: total: what follows is the journal's next section (acquisitions, accounts, an article).
PROSE_RUN = 3
#: Titles printed before a payer's name (*P.*, *Pan*, *Slečna*, *MUC.*, *J. U. Dr.* …).
#: *P.* is a priest's *Pater* in the lists printed surname first, but *Pan* in those of
#: the 1880s, printed given name first (*P. Rudolf Mendl, účetní v Praze*).
_PAY_TITLE = re.compile(r"^(?:(?:P\.|P[#„;:,'’\"]|P(?=\s+[A-ZÁČĎÉĚÍŇÓŘŠŤÚŮÝŽ])|Pan|Paní|Slečna|"
                        r"Ph\.\s*Dr\.|PhDr\.|ThDr\.|Ph\.\s*C\.|PhC\.|"
                        r"M\.\s*U\.\s*C\.|MUC\.|J\.\s*U\.\s*C\.|JUC\.|J\.\s*U\.\s*Dr\.|JUDr\.|"
                        r"MUDr\.|Dr\.|Ing\.|[Ss][lI]\.|[Ss]lč\.|si\.|[Pp]í\.|Msgr\.)(?!\w)\s*)+")
#: An institution named anywhere in a payer's name (*Knížecí arcibiskupská knihovna*,
#: *Purkmistrovský úřad*, *Výbor okresní*) — corporate unless the name holds a given name
#: or a person's occupation, as a person whose comma the OCR lost does
#: (*Novák Josef ředitel gymnasia*).
_INSTITUTION = re.compile(r"(?i)(?:knihovn|škol|gymn|re[áa]lk|spol(?:ek|ku|ečnost)|besed|\bobec\b|"
                          r"obecní|úřad|ústav|\bsbor|seminář|muse|kapitul|konvent|\bfara\b|farní|"
                          r"seniorát|jednot|zastupitel|výbor|záložn|spořitel|klub|družstv|"
                          r"universit|kolej|gremi|čtenář|kasin|ředitelstv|\bspráva\b)")
_PERSON_WORK = re.compile(r"(?i)\b(?:ředitel|učitel\w*|prof\w*|katechet\w*|farář|kaplan|úředník|"
                          r"inženýr|stud\w*|žák\w*|syn|dcera)\b")
#: First words of a corporate payer beyond ``CORPORATE_HEADS`` (schools, libraries,
#: teachers' bodies, which pay the corporate fee in instalments of their own).
_PAY_CORPORATE = re.compile(r"^(?:Re[áa]lk|[ÚU]stav|Místní|Král\.\s|Česk[áéý]\s+[a-zá-ž]|"
                            r"Okres|Měst|Gymn|Náprstk|Veřejn|Vyšší|Nižší|Knihovn|Sbor)")


def month_heading(text: str) -> tuple[int, int | None] | None:
    """``(month, year)`` of a month heading (*Prosinec.*, *Říjen r. 1898.*, *Příspěvky
    v měsíci červnu 1913.*), the year ``None`` where none is printed; ``None`` for any
    other line. A combined heading (*Červenec. Srpen.*) gives its first month."""
    t = text.strip()
    if not t or len(t) > 90 or _ANY_AMOUNT.search(t):
        return None
    hits = sorted((m.start(), month) for month, rx in _MONTH_RX for m in rx.finditer(t))
    if not hits or not _HEADING_LEAD.match(t[:hits[0][0]]):
        return None
    rest = t
    for _, rx in _MONTH_RX:
        rest = rx.sub(" ", rest)
    unknown = [w for w in re.findall(r"[^\W\d_]+", rest) if not _HEADING_WORDS.fullmatch(w)]
    if len(unknown) > 1 or any(len(w) > 8 for w in unknown):
        return None                              # one short word is OCR noise (*do leonce října*)
    if re.match(r"^\W*\d", t) and not re.search(r"\b[Oo]d\b", t):
        return None                              # a date opening prose: "13. října 1877 …"
    year = _YEAR.search(t)
    return hits[0][1], int(year.group(1)) if year else None


def payment_amount(text: str) -> tuple[str, float | None, str]:
    """``(rest, amount in korun, currency)`` of an entry ending in an amount; the amount
    ``None`` where none is read. A kreuzer figure the OCR damaged is read as 50 after
    10 zl. (the instalment, 10 zl. 50 kr.) and as 0 otherwise, the currency then
    ``zl?`` so the guess stays visible."""
    t = _SPLIT_DIGIT.sub("", _STRAY_BAR.sub("", _TRAILING.sub("", text)))
    t = _AMOUNT_TOKEN.sub(lambda m: m.group(1).translate(_OCR_DIGITS)
                          if re.search(r"\d", m.group(1)) else m.group(1), t)
    m = _AMOUNT_K.search(t)
    if m:
        k = (int(m.group("k").replace(".", ""))
             + int(m.group("h") or m.group("h2") or 0) / 100)
        return t[:m.start()].strip(" ,"), k, "K"
    m = _AMOUNT_K_FIRST.search(t)
    if m:
        return t[:m.start()].strip(" ,"), float(m.group("k")), "K"
    m = _AMOUNT_ZL.search(t)
    if m:
        zl, currency = int(m.group("zl")), "zl"
        kr = (m.group("kr") or m.group("kr50") or "").translate(_OCR_DIGITS)
        if kr and not kr.isdigit():
            kr, currency = ("50" if zl == 10 else "0"), "zl?"
        return (t[:m.start()].strip(" ,"), (zl + int(kr or 0) / 100) * KRONE_PER_GULDEN,
                currency)
    m = _AMOUNT_KR.search(t)
    if m:
        return t[:m.start()].strip(" ,"), int(m.group("kr")) / 100 * KRONE_PER_GULDEN, "zl"
    return t.strip(" ,"), None, ""


def parse_payer(text: str) -> dict:
    """Title, name, occupation, residence and corporate flag of one payer's text."""
    m = _PAY_TITLE.match(text)
    title = m.group(0).strip() if m else ""
    fields = parse_person(text[m.end():] if m else text)
    first = fields["name"].split()[0] if fields["name"].split() else ""
    if re.match(r"P\W?(?:\s|$)", title) and (first in GIVEN_NAMES or _GIVEN_ABBREV.match(first)):
        title = "Pan" + title[title.find(" "):] if " " in title else "Pan"   # P. here is Pan
    fields["honorific"] = " ".join(x for x in (title, fields["honorific"]) if x)
    fields.pop("amount_zl", None)
    name = fields["name"]
    words = [w.strip(".,") for w in name.split()]
    person = (any(w in GIVEN_NAMES or _GIVEN_ABBREV.match(w + ".") for w in words[1:])
              or bool(_PERSON_WORK.search(name)))
    fields["is_corporate"] = bool(is_corporate(name) or _PAY_CORPORATE.match(name)
                                  or (_INSTITUTION.search(name) and not person))
    return fields


def read_text_lines(pages_dir: Path) -> pd.DataFrame:
    """The OCR text lines of a page cache, in the shape of :func:`read_lines`, for a page
    whose ALTO is not on disk (a line ending in ``-`` or ``¬`` is hyphenated)."""
    rows = []
    for f in sorted(Path(pages_dir).glob("*.txt")):
        page = int(f.name[:4])
        for i, raw in enumerate(f.read_text(encoding="utf-8").splitlines(), 1):
            line = normalise(raw)
            hyphen = (line.endswith(("¬", "-")) and not line.endswith((" -", "—"))
                      and not _HYPHEN_AFTER_AMOUNT.search(line))   # "21 K.-" ends an entry
            rows.append({"page": page, "line": i, "hpos": 0, "vpos": i,
                         "text": line.rstrip("¬").rstrip(), "hyphen": hyphen})
    return pd.DataFrame(rows, columns=["page", "line", "hpos", "vpos", "text", "hyphen"])


def slice_lines(pages_dir: Path) -> pd.DataFrame:
    """A slice's lines from its ALTO pages, a page without ALTO from its OCR text."""
    alto = read_lines(pages_dir)
    text = read_text_lines(pages_dir)
    return (pd.concat([alto, text[~text["page"].isin(alto["page"])]])
            .sort_values(["page", "line"], kind="stable").reset_index(drop=True))


def month_blocks(lines: pd.DataFrame, state: dict) -> list[dict]:
    """Cut a slice's lines into month blocks: ``{month, year, year_how, heading, chain,
    lines}``.

    A block opens at a month heading and closes once its *Úhrnem* total has been read;
    lines between a total and the next heading (accounts, notices, the next section)
    belong to none. ``state`` carries the open block and the last month and year across
    slices, so a list continued in the next slice, and a heading without a year, are
    read in sequence: the year is the heading's own, else the last one, plus one when the
    month runs back (December → January), else the volume's year (``state['volume']``),
    the year before for a month from June on. A month continued into the next slice is a
    second block with the same ``chain``."""
    blocks: list[dict] = []
    open_block = state.get("open")
    if open_block is not None:
        open_block = dict(open_block, lines=[])
    awaiting_total = False
    for row in lines.itertuples(index=False):
        text = row.text.strip()
        if not text or _PAYMENT_FURNITURE.match(text):
            continue
        head = month_heading(text)
        if head:
            month, year = head
            how = "printed"
            if year is not None and not state["volume"] - 2 <= year <= state["volume"]:
                year = None                       # an OCR misreading (1880 in the 1886 volume)
            if year is None and state.get("month"):
                year = state["year"] + (month < state["month"])
                how = "carried"
            elif year is None:
                year = state["volume"] - (month >= 6)
                how = "volume"
            state.update(month=month, year=year, chains=state.get("chains", 0) + 1)
            open_block = {"month": month, "year": year, "year_how": how, "heading": text,
                          "chain": state["chains"], "lines": []}
            blocks.append(open_block)
            awaiting_total = False
            continue
        if open_block is None:
            continue
        if not any(b is open_block for b in blocks):     # continued from the previous slice
            blocks.append(open_block)
        open_block["lines"].append(row)
        dry = open_block["lines"][-PROSE_RUN:]
        if len(dry) == PROSE_RUN and not any(_ANY_AMOUNT.search(r.text) or _DASH.search(r.text)
                                             for r in dry):
            del open_block["lines"][-PROSE_RUN:]   # the list ended without a total: prose follows
            open_block, awaiting_total = None, False
            continue
        previous = open_block["lines"][-2] if len(open_block["lines"]) > 1 else None
        probe = (previous.text.rstrip("-") + text if previous is not None and previous.hyphen
                 else text)                      # a total hyphenated across lines: Úh-|rnem
        total = _TOTAL.search(probe)
        if total or awaiting_total:
            if _ANY_AMOUNT.search(probe[total.start():] if total else probe):
                open_block, awaiting_total = None, False
            else:
                awaiting_total = True
    state["open"] = open_block
    return blocks


def block_payments(block: dict) -> tuple[list[dict], float | None]:
    """The payments of one month block and its printed total (korun)."""
    frame = pd.DataFrame(block["lines"], columns=["page", "line", "hpos", "vpos", "text",
                                                  "hyphen"])
    text, starts, where = join_lines(frame)
    text = _HYPHEN_AFTER_AMOUNT.sub(r"\1 —", text)   # each hit moves later offsets by ≤ 1
    rows, total = [], None
    pos = 0
    for piece in _SEPARATOR.split(text):
        off = text.find(piece, pos)
        pos = off + len(piece) if off >= 0 else pos
        piece = piece.strip()
        if not piece:
            continue
        word = _TOTAL.search(piece)
        if word:
            _, total, _ = payment_amount(re.sub(r"vybrán\w*", "", piece[word.end():]))
            break                                # nothing after the total is the month's
        foundation = bool(_FOUNDATION.search(piece))
        rest, amount, currency = payment_amount(_FOUNDATION.sub("", piece))
        page, line = where[max(bisect_right(starts, max(off, 0)) - 1, 0)]
        payer = parse_payer(rest)
        restored = (amount is not None and amount > MAX_PERSON_PAYMENT_K
                    and not payer["is_corporate"])
        if restored:
            amount = amount / 100
        flags = [f for f, bad in (("foundation", foundation), ("decimal_restored", restored),
                                  ("no_amount", amount is None and not foundation),
                                  ("no_name", not payer["name"]),
                                  ("no_residence", not payer["residence"]),
                                  ("long_entry", len(piece) > 200)) if bad]
        rows.append({"page": page, "line": line, "text": piece, "amount_k": amount,
                     "currency": currency, **payer, "flags": ";".join(flags)})
    return rows, total


PAYMENT_COLUMNS = [
    "payment_id", "slice", "volume", "issue", "block_id", "period_year", "period_month",
    "year_how", "page", "line", "text", "honorific", "name", "surname", "is_corporate",
    "occupation", "residence_prep", "residence", "amount_k", "currency", "flags",
]


def parse_payments(library_dir: Path, manifest: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame]:
    """Every payment in the slices of *manifest* (read in its order) and every month block
    against its printed total: ``(payments, months)``. ``payments`` is keyed
    ``payment_id``, ``months`` ``block_id``; ``months['diff_k']`` is the parsed sum less
    the printed *Úhrnem* (missing where no total was read)."""
    state: dict = {}
    pay_rows, month_rows = [], []
    for s in manifest.itertuples(index=False):
        pages_dir = Path(library_dir) / f"NM_Casopis-Musea_{s.key}_Matice-prispevky_pages"
        if not pages_dir.exists():
            continue
        state["volume"] = int(s.year)
        for block in month_blocks(slice_lines(pages_dir), state):
            block_id = f"{s.key}:{block['year']}-{block['month']:02d}:{len(month_rows) + 1:04d}"
            rows, total = block_payments(block)
            for r in rows:
                pay_rows.append({**r, "slice": s.key, "volume": int(s.year), "issue": s.issue,
                                 "block_id": block_id, "period_year": block["year"],
                                 "period_month": block["month"], "year_how": block["year_how"]})
            amounts = [r["amount_k"] for r in rows if r["amount_k"] is not None]
            month_rows.append({"block_id": block_id, "chain": block["chain"], "slice": s.key,
                               "volume": int(s.year), "period_year": block["year"],
                               "period_month": block["month"], "year_how": block["year_how"],
                               "heading": block["heading"], "entries": len(rows),
                               "no_amount": len(rows) - len(amounts),
                               "sum_k": round(sum(amounts), 2), "printed_k": total})
    months = month_check(pd.DataFrame(month_rows))
    pay = pd.DataFrame(pay_rows)
    if pay.empty:
        return pd.DataFrame(columns=PAYMENT_COLUMNS), months
    # printed given name first until 1892, surname first from 1893: each month block's
    # order is read off the majority of its names, then applied to all of them
    order = pay.groupby("block_id")["name"].transform(block_name_order)
    pay["surname"] = [surname_of(n, o) if not c else ""
                      for n, o, c in zip(pay["name"], order, pay["is_corporate"])]
    pay["payment_id"] = [f"mcp:{i + 1:05d}" for i in range(len(pay))]
    return pay[PAYMENT_COLUMNS], months


def block_name_order(names: pd.Series) -> str:
    """``given_first`` where most names of a month block open with a given name (*Rudolf
    Mendl*, *Frant. Kozák*), else ``surname_first`` (*Benetka Čestmír*) — so a given name
    the list of :data:`GIVEN_NAMES` lacks (*Bořivoj Nejedlý*) is not taken for a surname."""
    firsts = [n.split()[0] for n in names.dropna() if isinstance(n, str) and n.split()]
    given = sum(f in GIVEN_NAMES or bool(_GIVEN_ABBREV.match(f)) for f in firsts)
    return "given_first" if firsts and given > len(firsts) / 2 else "surname_first"


def month_check(blocks: pd.DataFrame) -> pd.DataFrame:
    """One row per month as printed (a chain of blocks, a month continued into the next
    slice being one), its parsed sum against its *Úhrnem*: ``diff_k`` is the sum less the
    printed total, missing where no total was read; ``agrees`` within half a korun."""
    if blocks.empty:
        return blocks
    first = blocks.groupby("chain", sort=False).first()
    out = first[["block_id", "slice", "volume", "period_year", "period_month", "year_how",
                 "heading"]].copy()
    agg = blocks.groupby("chain", sort=False).agg(
        blocks=("block_id", "size"), entries=("entries", "sum"), no_amount=("no_amount", "sum"),
        sum_k=("sum_k", "sum"), printed_k=("printed_k", "last"))
    out = out.join(agg).reset_index()
    out["diff_k"] = (out["sum_k"] - out["printed_k"]).round(2)
    out["agrees"] = out["diff_k"].abs() <= 0.5
    return out


# ---------------------------------------------------------------------------
# From payments to founders who joined
# ---------------------------------------------------------------------------
#: The window a joiner's first payment must fall in to count in the headline measure
#: (D-184; ruled 1895–1909 on 2026-11-03, D-188): it ends at the Matice moravská roster of
#: 1 March 1909 and the 1910 census, and starts in 1895 because a decade is too few people —
#: its two halves rank okresy at Spearman 0.64 with each other. The lists run from 1886, so a
#: payer first seen from 1895 on is very unlikely to have begun his instalments before them.
JOINER_WINDOW = (1895, 1909)


def _fold(text: str) -> str:
    text = "" if not isinstance(text, str) else text
    text = unicodedata.normalize("NFKD", text).encode("ascii", "ignore").decode()
    return re.sub(r"[^a-z]", "", text.lower())


def payer_key(surname: str, name: str, place: str) -> str:
    """Who a payment is from: the folded surname, the first letter of the given name, and
    the place (the 1910 okres, else the first letters of the residence, folded). The initial is
    all the lists print consistently (*Kubeš Jaroši.*, *Kubeš Jarosl.*, *Kubeš J.*); it is
    the first word that is not the surname, so either name order gives the same key."""
    return f"{_fold(surname)}|{_initial(surname, name)}|{place}"


def _initial(surname: str, name: str) -> str:
    """The folded first letter of the first word of *name* that is not the surname."""
    name = name if isinstance(name, str) else ""
    rest = [w for w in name.split() if w.strip(".,") != surname and w.strip(".,") not in _PARTICLES]
    return _fold(rest[0])[:1] if rest else ""


_PLACE_COLS = ["settlement_id", "okres_id", "place_status"]


def borrow_places(payments: pd.DataFrame) -> pd.DataFrame:
    """Place a payment whose residence was refused as ambiguous, or not found, by the same
    payer's other payments: those with the same surname, initial and printed place name
    (``place_key``: *Benešově* and *Benešově u Prahy* share it) that were placed. Borrowed
    only where they agree on one place; ``place_how`` becomes ``borrowed:payer``. Evidence
    the lists print, never a prior."""
    p = payments.copy()
    key = pd.Series([f"{_fold(s)}|{_initial(s, n)}|{k}" for s, n, k in
                     zip(p["surname"], p["name"], p["place_key"].fillna(""))], index=p.index)
    placed = p["place_status"].isin(["matched", "prague"]) & p["surname"].astype(bool)
    donors = (p[placed].assign(_key=key[placed])
              .drop_duplicates(["_key"] + _PLACE_COLS))
    one = donors.groupby("_key").filter(lambda g: len(g) == 1).set_index("_key")[_PLACE_COLS]
    take = (p["place_status"].isin(["ambiguous", "unmatched"]) & p["surname"].astype(bool)
            & key.isin(one.index) & p["place_key"].fillna("").astype(bool))
    for col in _PLACE_COLS:
        p.loc[take, col] = key[take].map(one[col]).to_numpy()
    p.loc[take, "place_how"] = "borrowed:payer"
    return p


def payment_persons(payments: pd.DataFrame) -> pd.DataFrame:
    """One row per payer who is a person (key ``payer_id``): first and last year paid,
    payments, the sum in korun, whether it reached the founder's fee (``FOUNDER_FEE_K``,
    half a korun's tolerance) and the year it did, and the place and occupation of the
    first payment. Needs the place columns of ``locative.match_all`` on *payments*."""
    p = payments[~payments["is_corporate"].astype(bool) & payments["amount_k"].notna()
                 & payments["surname"].astype(bool)].copy()
    # the okres, not the settlement: an inflected name read twice (Vraňan / Vraňanech,
    # Slaném / Slaným) or a move within the district must not split a payer
    place = ("o" + p["okres_id"].astype("string")).fillna(
        p["place_status"].where(p["place_status"] == "prague")).fillna(
        p["residence"].map(lambda r: _fold(r)[:5]))
    p["payer_id"] = [payer_key(s, n, str(pl)) for s, n, pl in zip(p["surname"], p["name"], place)]
    p = p.sort_values(["period_year", "period_month", "payment_id"], kind="stable")
    p["cum_k"] = p.groupby("payer_id")["amount_k"].cumsum()
    full = p[p["cum_k"] >= FOUNDER_FEE_K - 0.5].groupby("payer_id")["period_year"].min()
    # whole rows, never column-wise first(): a payer's place, and his occupation with its
    # class, each come from one payment — the place from his first, the occupation from the
    # first that prints one
    first = p.drop_duplicates("payer_id").set_index("payer_id")
    out = first[["name", "surname", "honorific", "residence", "settlement_id", "okres_id",
                 "place_status"]].copy()
    work = ["occupation"] + (["member_group"] if "member_group" in p else [])
    printed = p[p["occupation"].fillna("").astype(str).str.strip().astype(bool)]
    job = printed.drop_duplicates("payer_id").set_index("payer_id")[work]
    out[work] = job.reindex(out.index)
    out["occupation"] = out["occupation"].fillna(first["occupation"])
    if "member_group" in work:
        out["member_group"] = out["member_group"].fillna(first["member_group"])
    out = out[["name", "surname", "honorific", "occupation", "residence", "settlement_id",
               "okres_id", "place_status"] + work[1:]]
    out["first_year"] = first["period_year"]
    out["last_year"] = p.groupby("payer_id")["period_year"].max()
    out["payments"] = p.groupby("payer_id").size()
    out["paid_k"] = p.groupby("payer_id")["amount_k"].sum().round(2)
    out["founder"] = out.index.isin(full.index)
    out["founder_year"] = full.reindex(out.index).astype("Int64")
    return out.reset_index()


def joiners(persons: pd.DataFrame, window: tuple[int, int] = JOINER_WINDOW,
            founders_only: bool = False) -> pd.DataFrame:
    """The payers whose first payment falls in *window*: a joiner is someone who began
    paying for a founder's share then (D-184; dating by first payment ruled 2026-11-03, the
    window provisional). ``founders_only`` keeps only
    those whose payments reached the fee within the series — a thin count (instalments ran
    for years, often in 5 K sums, and a payer the lists name differently splits in two)."""
    lo, hi = window
    keep = persons["first_year"].between(lo, hi)
    if founders_only:
        keep &= persons["founder"]
    return persons[keep]
