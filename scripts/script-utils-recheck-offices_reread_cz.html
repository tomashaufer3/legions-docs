How it works

The section cutter labelled three Czech offices regions parts (D-192): Dvůr Králové nad Labem on page 0083 (two columns, below the Úřady, ústavy heading the cutter missed) and Vodňany on 0317. The route's readers named the symbols in those lines, but the lines sat in a Části osad section, so the decoder never read them as offices (and Gemini, the reader of record, skipped the left column's foot): the Czech register shows no institution for Dvůr Králové's 15 offices entries or Vodňany's 2, the volume's largest blind block. offices_reread re-reads such regions for the German volumes with the German readers; this module does it for the Czech volume with the Czech readers, as the frozen run built them (run_frozen.read_plan):

  • the crop — the Czech page as the pilot renders it (lexikon_1910.render_page at the ALTO page's width, strips cut on the page's ink as under --geometry image), the region's box taken from the run's sections/index.csv (the section image is the region enlarged by enlarge) from the located heading down, cut into strips 500 px tall and upscaled (--notes-max-height 500 --notes-upscale), with the pilot's own captions;
  • the readers — built as run_lexikon_1910_pilot.build_notes_reader builds them from the run's flags (its parser is inline in main, so the arguments are restated here, with the pilot's defaults: 4,000 output tokens for Gemini, 8,000 for Haiku): Gemini 3.8 Flash once (temperature 0, reasoning off, the key image) and Haiku 5.5 three times (adaptive thinking, effort high, the key image and the run's contrast sheet), each with the Czech offices prompt (lexikon_1910_vision.build_system(section="offices")), each label with a crop cache of its own under output/recheck/cz/reread/;
  • the decode — the Czech decoder (lexikon_1910_institutions.decode_sections) on the lines split at any heading they carry (offices_resplit.resplit_sections, a full-stop key mended);
  • the claims — settled as §4 settles the German ones (offices_reread.settle): every reader agrees → added; disputed → Opus (§6); an entry only some readers reached → the reader of record's.

Nothing is sent without spend, and then only on the machine of record (D-68).

def cut(pages: str = "full") -> tuple[list[tuple], dict]:
    """``(jobs, segments)``: every Czech region of the coverage check cut as the pilot cuts a
    section — jobs ``(slot, crop, caption, "offices")`` in printed order, segments by slot."""
    index = {r["image"]: r for r in _read(rf.out_dir(EDITION, pages) / "sections" / "index.csv")}
    by_page = defaultdict(list)
    for r in orr.regions(EDITION):
        row = index[r["section"] + ".png"]
        by_page[row["page"]].append((r, row))
    jobs, segments = [], {}
    for page in sorted(by_page):
        alto = lx.read_alto(pilot.CZ_PAGES / f"{page}.alto.xml")
        image = lx.render_page(pilot.CZ_PDF, int(page), alto.width)
        geo = lximg.detect(image)               # --geometry image: strips cut on the page's ink
        ink = lxnt.page_ink(image, geo) if geo.apparatus() else None
        for r, row in sorted(by_page[page], key=lambda pair: int(pair[1]["order"])):
            slot = (row["page"], row["region"], int(row["order"]))
            segment = lxnt.Segment(kind="offices", district=row["district"], page=row["page"],
                                   region=row["region"], box=orr.page_box(row, r["y_from"], r["y_to"]),
                                   order=int(row["order"]))
            segments[slot] = segment
            for crop, _s, i, n in lxnt.crop_segments(image, [segment], alto, upscale=True,
                                                     max_height=MAX_HEIGHT, ink=ink):
                jobs.append((slot, crop, orr._caption(i, n), "offices"))
    return jobs, segments

def readers(pages: str = "full") -> dict:
    """``{label: spec}`` — the frozen run's four Czech readers with the Czech offices prompt."""
    from PIL import Image

    abbr = abbreviations()
    sheet = rf.out_dir(EDITION, pages) / "freeze" / "contrast_sheet.png"
    contrast = Image.open(sheet).convert("RGB")
    digest = hashlib.sha256(sheet.read_bytes()).hexdigest()[:16]
    out = {}
    with orr._placeholder_keys(("ANTHROPIC_API_KEY", pilot.READER_ENDPOINTS["gemini"][1])):
        for label in rs.readers(EDITION, pages):

                system = lxv.build_system(abbr, style="annotate", section="offices", atlas=True)
                url, key_env = pilot.READER_ENDPOINTS["gemini"]
                reader = pilot.OpenAICompatibleReader(
                    GEMINI[0], url, key_env, system=system, schema=lxv.NOTES_SCHEMA, payload_key="lines",
                    schema_name="notes", max_tokens=GEMINI[1], temperature=0.0, reasoning_effort="none")
                reader.prompt_style = "annotate"
                reader.shared_parts = 1
                out[label] = {"arm": gn.ARMS["gemini38"], "reader": reader, "system": system,
                              "contrast": None, "contrast_digest": ""}
            else:
                system = lxv.build_system(abbr, style="annotate", section="offices", atlas=True,
                                          think=True, contrast=True)
                reader = pilot.ClaudeReader(HAIKU[0], "high", thinking="adaptive", max_tokens=HAIKU[1],
                                            system=system, schema=lxv.NOTES_SCHEMA, payload_key="lines")
                reader.prompt_style = "annotate"
                reader.shared_parts = 2
                out[label] = {"arm": gn.ARMS["haiku55nh"], "reader": reader, "system": system,
                              "contrast": contrast, "contrast_digest": digest}
    return out

def decode(label: str, segments: dict, by_slot: dict, folder: Path) -> list[dict]:
    """One Czech reader's lines into its inventory, as :func:`offices_reread.decode` does for a
    German one: split at any heading the lines carry, a full-stop key mended, decoded by the Czech
    decoder. Writes ``notes_<label>.csv``, ``splits_<label>.csv`` and ``inventory_<label>.csv``."""
    read_ = [(segments[slot], by_slot.get(slot, [])) for slot in sorted(segments, key=lambda s: (s[0], s[2]))]
    sections, splits = rs.resplit_sections(read_, lxnt.find_heading, mend=True)
    records, _entries = lxin.decode_sections(sections, abbreviations())
    inventory = lxin.build_inventory(records, [])
    rs._write(folder / f"notes_{label}.csv",
              [{"page": s.page, "region": s.region, "order": s.order, "court_de": s.district,
                "line_no": i, "text": line} for s, lines in read_ for i, line in enumerate(lines)])
    rs._write(folder / f"splits_{label}.csv", splits)
    rs._write(folder / f"inventory_{label}.csv", inventory)
    return inventory

def read(pages: str = "full", spend: bool = False, out: Path = OUT, allow_other_machine: bool = False) -> dict:
    """Cut, price, and read the Czech regions with every Czech reader; decode what is read.
    Without ``spend`` nothing is sent. Returns the summary also written to ``summary.json``."""
    if spend and bo.HOST != orr.MACHINE_OF_RECORD and not allow_other_machine:
        raise RuntimeError(f"the re-read spends on {orr.MACHINE_OF_RECORD} only (D-68): this is {bo.HOST}")
    from PIL import Image

    atlas = Image.open(ATLAS).convert("RGB")
    jobs, segments = cut(pages)
    folder = Path(out) / EDITION / SUB
    folder.mkdir(parents=True, exist_ok=True)
    if spend:
        from utils.lexikon_1910 import lexikon_1910_pages as lxp
        lxp.load_credentials()
    summary = {"module": __version__, "edition": EDITION, "pages": pages, "host": bo.HOST,
               "run": time.strftime("%Y-%m-%d %H:%M:%S"), "regions": len(segments),
               "crops": len(jobs), "readers": {}}
    for label, spec in readers(pages).items():
        cache = pilot.key_image_cache(folder / "crop_cache" / label, pilot.key_image_digest(atlas))
        missing = [j for j in jobs if not orr._cached(j, spec, cache)]
        usd = orr.price(missing, spec, atlas, len(spec["system"]))
        entry = {"cached": len(jobs) - len(missing), "to_read": len(missing), "usd_live": round(usd, 2)}
        if missing and not spend:
            entry["state"] = "not read (spend is off)"
            summary["readers"][label] = entry
            continue
        results, parked = gn.read_crops(jobs, {"offices": spec["reader"]}, atlas, cache,
                                        contrast=spec["contrast"], contrast_digest=spec["contrast_digest"])
        by_slot, counts = gn.clean_lines(results, lxv.vocabulary(abbreviations()))
        entry.update({k: counts.get(k, 0) for k in ("failed", "empty", "rejected", "input", "output")})
        entry["state"] = ("incomplete: run again to finish (the cache keeps what was read)"
                          if parked.get("at") is not None or counts.get("failed") else "read")
        entry["rows"] = len(decode(label, segments, by_slot, folder))
        summary["readers"][label] = entry
        print(f"  {label}: {entry}")
    (folder / "summary.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return summary

            else:
                system = lxv.build_system(abbr, style="annotate", section="offices", atlas=True,
                                          think=True, contrast=True)
                reader = pilot.ClaudeReader(HAIKU[0], "high", thinking="adaptive", max_tokens=HAIKU[1],
                                            system=system, schema=lxv.NOTES_SCHEMA, payload_key="lines")
                reader.prompt_style = "annotate"
                reader.shared_parts = 2
                out[label] = {"arm": gn.ARMS["haiku55nh"], "reader": reader, "system": system,
                              "contrast": contrast, "contrast_digest": digest}
    return out


def decode(label: str, segments: dict, by_slot: dict, folder: Path) -> list[dict]:
    """One Czech reader's lines into its inventory, as :func:`offices_reread.decode` does for a
    German one: split at any heading the lines carry, a full-stop key mended, decoded by the Czech
    decoder. Writes ``notes_<label>.csv``, ``splits_<label>.csv`` and ``inventory_<label>.csv``."""
    read_ = [(segments[slot], by_slot.get(slot, [])) for slot in sorted(segments, key=lambda s: (s[0], s[2]))]
    sections, splits = rs.resplit_sections(read_, lxnt.find_heading, mend=True)
    records, _entries = lxin.decode_sections(sections, abbreviations())
    inventory = lxin.build_inventory(records, [])
    rs._write(folder / f"notes_{label}.csv",
              [{"page": s.page, "region": s.region, "order": s.order, "court_de": s.district,
                "line_no": i, "text": line} for s, lines in read_ for i, line in enumerate(lines)])
    rs._write(folder / f"splits_{label}.csv", splits)
    rs._write(folder / f"inventory_{label}.csv", inventory)
    return inventory


def read(pages: str = "full", spend: bool = False, out: Path = OUT, allow_other_machine: bool = False) -> dict:
    """Cut, price, and read the Czech regions with every Czech reader; decode what is read.
    Without ``spend`` nothing is sent. Returns the summary also written to ``summary.json``."""
    if spend and bo.HOST != orr.MACHINE_OF_RECORD and not allow_other_machine:
        raise RuntimeError(f"the re-read spends on {orr.MACHINE_OF_RECORD} only (D-68): this is {bo.HOST}")
    from PIL import Image

    atlas = Image.open(ATLAS).convert("RGB")
    jobs, segments = cut(pages)
    folder = Path(out) / EDITION / SUB
    folder.mkdir(parents=True, exist_ok=True)
    if spend:
        from utils.lexikon_1910 import lexikon_1910_pages as lxp
        lxp.load_credentials()
    summary = {"module": __version__, "edition": EDITION, "pages": pages, "host": bo.HOST,
               "run": time.strftime("%Y-%m-%d %H:%M:%S"), "regions": len(segments),
               "crops": len(jobs), "readers": {}}
    for label, spec in readers(pages).items():
        cache = pilot.key_image_cache(folder / "crop_cache" / label, pilot.key_image_digest(atlas))
        missing = [j for j in jobs if not orr._cached(j, spec, cache)]
        usd = orr.price(missing, spec, atlas, len(spec["system"]))
        entry = {"cached": len(jobs) - len(missing), "to_read": len(missing), "usd_live": round(usd, 2)}
        if missing and not spend:
            entry["state"] = "not read (spend is off)"
            summary["readers"][label] = entry
            continue
        results, parked = gn.read_crops(jobs, {"offices": spec["reader"]}, atlas, cache,
                                        contrast=spec["contrast"], contrast_digest=spec["contrast_digest"])
        by_slot, counts = gn.clean_lines(results, lxv.vocabulary(abbreviations()))
        entry.update({k: counts.get(k, 0) for k in ("failed", "empty", "rejected", "input", "output")})
        entry["state"] = ("incomplete: run again to finish (the cache keeps what was read)"
                          if parked.get("at") is not None or counts.get("failed") else "read")
        entry["rows"] = len(decode(label, segments, by_slot, folder))
        summary["readers"][label] = entry
        print(f"  {label}: {entry}")
    (folder / "summary.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return summary


def main(argv=None) -> int:
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    summary = read()
    print(orr.report(summary))
    if all(r["state"] == "read" for r in summary["readers"].values()):
        print("  consensus:", orr.consensus(EDITION, "full", OUT))
    return 0
