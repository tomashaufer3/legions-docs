How it works

Until 2026-08-29 both of these lived in notebooks/05_ocr_model_bakeoff.ipynb and the notebook was the driver. That is what produced 05_ocr_model_bakeoff (Heavyhorse6's conflicted copy 2026-08-14).ipynb: a Dropbox-synced JSON document cannot be the thing two machines both run, because the run mutates it (outputs, execution counts) on both sides of the sync.

So the run moved here and the notebook became a READER of results. The split is the point:

  • this module and run_bakeoff.py hold the matrix, the flags and the loop. They are plain text, they diff, and a merge conflict in them is legible.
  • the notebook keeps sections 6a-6f, which read finished CSVs and draw the comparison. Two machines can have that open at once without harm.

Pass artefacts are written per host (bakeoff/&lt;host&gt;/), so two machines running the same matrix cannot write the same path. Only the summary CSVs need to travel. The human transcription is NOT per host - there is one gold set and both machines score against it.

def select_models(wanted, models: list | None = None) -> list:
    """Narrow a model list by label or by ollama id.

    ``"all"`` (or ``None``) keeps everything. Anything else is a list of short
    labels (``"qwen38"``) or full ids (``"glm-5.3-flash:cloud"``), matched
    case-insensitively.

    **An unknown name raises rather than being skipped.** A typo that silently
    selects nothing would start a night that runs no passes and reports success,
    and a typo that silently selects a subset is worse — the comparison would be
    missing a model and nothing would say which.
    """
    pool = list(models if models is not None else MODELS)
    if wanted in (None, "all", "ALL"):
        return pool
    if isinstance(wanted, str):
        wanted = [wanted]
    asked = [str(w).strip().lower() for w in wanted if str(w).strip()]
    picked, unknown = [], []
    for name in asked:
        hit = [(m, l) for m, l in pool
               if l.lower() == name or m.lower() == name]
        if hit:
            picked += [h for h in hit if h not in picked]
        else:
            unknown.append(name)
    if unknown:
        known = ", ".join(sorted(f"{l} ({m})" for m, l in pool))
        raise SystemExit(f"unknown model(s): {', '.join(unknown)}\n"
                         f"known here: {known}")
    return picked

def runnable(verbose: bool = True) -> list:
    """The models this daemon can actually run, as ``(model, label)``.

    Two different problems that must not be reported as one. A model that is not
    pulled is an `ollama pull` away; a model that is pulled but reports no
    vision cannot do this job at all and no amount of pulling will change that.
    """
    pulled = pulled_models()
    if not pulled:
        if verbose:
            print(f"MISSING ollama at {OLLAMA}. Start the service, or run: "
                  f"ollama serve")
        return []
    out = []
    for model, label in MODELS:
        if model not in pulled:
            if verbose:
                print(f"  {model:22s} not pulled on this daemon")
            continue
        try:
            capabilities = ollama_show(model).get("capabilities", []) or []
        except Exception as exc:                               # noqa: BLE001
            if verbose:
                print(f"  {model:22s} show failed: {exc}")
            continue
        if "vision" not in capabilities:
            if verbose:
                print(f"  {model:22s} reports no vision - cannot read a crop")
            continue
        out.append((model, label))
    return out

def n_pages(spec: str) -> int:
    """How many pages a --pages string names: ``118,270`` -> 2, ``118-137`` -> 20."""
    total = 0
    for part in str(spec).split(","):
        if "-" in part:
            first, last = part.split("-", 1)
            total += int(last) - int(first) + 1
        elif part.strip():
            total += 1
    return max(total, 1)

def plan(kind: str, models: list, encoder: dict, only=None) -> list:
    """The worklist: one row per pass, in the order it will run.

    Variant-major: every model at `base`, then every model at `again`, and so on
    down `VARIANTS`. A night that stops early then leaves a complete comparison
    across models at the variants that matter most, rather than the first two
    models finished and the last two never started.
    """
    if kind not in KINDS:
        raise SystemExit(f"unknown kind {kind!r}; known: {', '.join(KINDS)}")
    pages = {"accuracy": accuracy_pages(), "gold": gold_pages(),
             "table": gold_pages()}.get(kind, PAGES_THROUGHPUT)
    names = {"accuracy": [v["name"] for v in VARIANTS],
             "gold": GOLD_VARIANTS,
             "table": TABLE_VARIANTS}.get(kind, THROUGHPUT_VARIANTS)
    wanted = [VARIANT[n] for n in names if n in VARIANT]
    if only:
        # A NARROWING THAT SELECTS NOTHING SAYS SO. Silently, this returns an
        # empty worklist, `run_matrix` prints "nothing outstanding", and a night
        # that was meant to read thirteen pages with six models ends in under a
        # second looking like a completed run. `select_models` has raised on an
        # unknown model since it was written, for exactly this reason; the
        # variant filter beside it did not.
        #
        # Two different mistakes, reported differently. A name that is not a
        # variant at all is a typo and raises. A real variant that this kind does
        # not have - `noup` under `gold`, which is base and atlas only - is a
        # misunderstanding of the matrix, and saying which variants the kind does
        # have is more use than a traceback.
        unknown = [n for n in only if n not in VARIANT]
        if unknown:
            raise SystemExit(f"unknown variant(s): {', '.join(unknown)}\n"
                             f"known: {', '.join(VARIANT)}")
        missing = [n for n in only if n not in names]
        if missing:
            print(f"  .. {kind} has no {', '.join(missing)} variant - it runs "
                  f"{', '.join(names)}")
        wanted = [v for v in wanted if v["name"] in only]
        if not wanted:
            print(f"  !! nothing in --only ({', '.join(only)}) is a {kind} "
                  f"variant, so this plan is EMPTY and nothing will run.")
    # Cloud models last, whatever order they sit in `MODELS`. A parked cloud pass
    # costs nothing once the card has finished its own work; parked first it would
    # leave the GPU idle behind a quota it cannot influence.
    models = sorted(models, key=lambda m: m[0] in CLOUD_MODELS)
    jobs = []
    for variant in wanted:
        for model, label0 in models:
            if model in NOT_NAMERS or not applies(model, variant, encoder):
                continue
            label = label_for(label0, variant["name"], kind)
            upscale, height = resolution_for(model, variant, encoder)
            jobs.append({
                "kind": kind, "model": model, "variant": variant["name"],
                "label": label, "pages": pages,
                "atlas": variant["atlas"], "upscale": upscale,
                "height": height, "num_ctx": NUM_CTX,
                "cloud": model in CLOUD_MODELS,
                # A pass that PARKED is not done. Its inventory exists and is
                # short, so the plain "inventory is on disk" test would skip it
                # for ever and the model would never be retried on a fresh quota.
                "done": ((PASS_DIR / f"inventory_{label}.csv").exists()
                         and not (PASS_DIR / f"ratelimit_{label}.json").exists()),
                "parked": (PASS_DIR / f"ratelimit_{label}.json").exists(),
            })
    return jobs

def estimate(jobs: list, timings: dict, force: bool = False) -> float:
    """Hours the outstanding jobs will take, from what this machine has measured.

    A model with no history gets the field's median, and a first-ever run gets
    40 s a page, which is the only number here that is a guess.

    history: dict = {}
    for stamp in timings.values():
        for run in stamp.values():
            if run.get("seconds") and run.get("model") and run.get("pages"):
                history.setdefault(run["model"], []).append(
                    run["seconds"] / n_pages(run["pages"]))
    per_model = {m: sum(v) / len(v) for m, v in history.items()}
    default = (sorted(per_model.values())[len(per_model) // 2]
               if per_model else 40.0)
    return sum(per_model.get(job["model"], default) * n_pages(job["pages"])
               for job in jobs if force or not job["done"]) / 3600

class Tee(io.TextIOBase):
    """Print to the console and to a log file at once.

    The runs are long. Without this the choice is a live view that vanishes when
    the terminal closes, or a log nobody watches.
    """

    def __init__(self, *streams):
        self.streams = streams

    def write(self, text):
        for stream in self.streams:
            try:
                stream.write(text)
            except UnicodeEncodeError:
                # ONE STREAM'S ALPHABET MUST NOT END A PASS. The parts ledger
                # prints the marks the OCR engine returned - `Ш`, `©`, `«“■` -
                # and a cp1250 console cannot encode the first of those. Left to
                # raise, it kills the pilot *after* the crops have been read and
                # paid for, `run_pilot` records the pass as a failure, and three
                # of those abort the rest of the matrix. The log file is UTF-8
                # and keeps the real characters either way.
                stream.write(text.encode("ascii", "replace").decode("ascii"))
        return len(text)

    def flush(self):
        for stream in self.streams:
            try:
                stream.flush()
            except Exception:                                  # noqa: BLE001
                pass

        print("  every model already classified - "
              + ", ".join(f"{m}={k}" for m, k in encoder.items()))
        return encoder
    native, big = encoder_crop(upscale=False), encoder_crop(upscale=True)
    print(f"crop native {native.width}x{native.height}, "
          f"upscaled {big.width}x{big.height} "
          f"({big.width * big.height / (native.width * native.height):.1f}x "
          f"the pixels)\n")
    print(f"  {'model':22s} {'native':>8s} {'upscaled':>9s} {'ratio':>7s}  "
          f"encoder")
    for model, _label in todo:
        try:
            small = prompt_tokens(model, native)
            large = prompt_tokens(model, big)
        except Exception as exc:                               # noqa: BLE001
            print(f"  {model:22s} probe failed: {exc}")
            continue
        finally:
            unload(model)
        ratio = (large / small) if small else 0.0
        # 1.25x is well clear of both outcomes measured on this crop: a dynamic
        # encoder came back at 1.66x and a fixed-grid one at 0.99x. It is a
        # threshold between two clusters, not a tuned parameter.
        kind = "dynamic" if ratio >= 1.25 else "fixed"
        encoder[model] = kind
        print(f"  {model:22s} {small:8d} {large:9d} {ratio:7.2f}  {kind}")
    ENCODERS.write_text(json.dumps(encoder, indent=2), encoding="utf-8")
    print(f"\n  written to {ENCODERS}")
    return encoder

def lexikon_1910_rate_limit_report(events: list) -> str:
    """What the limit was, and whether the model can read the volume behind it.

    Three questions, and the third is the one that decides whether a cloud model
    is a candidate at all:

    * **When did it stop, and how much had it read?** Measured exactly — crops
      completed, seconds elapsed, and the rate between them.
    * **How big is the limit?** Only partly answerable from one pass. The crops
      completed before the refusal is a **lower bound** on the allowance, not the
      allowance: the window may have been part-spent when the run started, and
      nothing here can see that. It is reported as a bound and never as the limit.
    * **Could it do the full sweep?** That needs the reset window, which is
      knowable only if the server sent `Retry-After`. Where it did, the arithmetic
      is honest and printed. Where it did not, **no number is invented** — the
      report says what would have to be measured instead.
    """
    lines = [f"bakeoff v{__version__} — rate limits", ""]
    if not events:
        return "\n".join(lines + ["  no pass has been rate limited on this host"])
    total = volume_crops()
    for event in events:
        done = event.get("crops_done_before_limit") or 0
        seconds = event.get("seconds_before_limit") or 0
        rate = event.get("crops_per_second_before_limit")
        retry = event.get("retry_after_seconds")
        lines.append(f"  {event.get('label', '?')}   {event.get('model', '?')}"
                     f"   {event.get('limit_at', '?')}")
        lines.append(f"    read {done} of {event.get('crops_total', '?')} crops "
                     f"in {seconds:.0f}s before the refusal"
                     + (f" ({rate:.3f} crops/s)" if rate else ""))
        lines.append(f"    status {event.get('status')}   "
                     f"{str(event.get('error', ''))[:90]}")
        lines.append(f"    allowance: AT LEAST {done} requests in "
                     f"{seconds / 60:.1f} min — a lower bound, because the "
                     f"window may")
        lines.append(f"               already have been part spent when this "
                     f"pass started")
        lines.append(f"    full Čechy sweep at sections={SECTIONS}: "
                     f"{total:.0f} crops")
        if retry:
            batches = max(0, -(-total // max(done, 1)) - 1)
            waiting = batches * retry / 3600
            reading = (total / rate / 3600) if rate else float("nan")
            lines.append(f"    the server asked for {retry:.0f}s before "
                         f"retrying, so at {done} crops a window:")
            lines.append(f"      {batches:.0f} waits x {retry / 60:.1f} min = "
                         f"{waiting:.1f} h idle"
                         + (f", plus {reading:.1f} h reading = "
                            f"{waiting + reading:.1f} h total"
                            if rate else ""))
        else:
            lines.append(f"    the server sent NO Retry-After, so the reset "
                         f"window is unknown and the")
            lines.append(f"    sweep cost cannot be computed. To measure it: "
                         f"re-run this one pass and")
            lines.append(f"    time the gap until a request succeeds again — "
                         f"that gap is the window.")
        lines.append("")
    lines.append("  A parked pass is marked `incomplete` in timings.json and is "
                 "excluded from")
    lines.append("  every accuracy table. It measures the quota, not the model.")
    return "\n".join(lines)

"""The bake-off, as a module: the configuration and the run loop, in one place.

Until 2026-08-29 both of these lived in ``notebooks/05_ocr_model_bakeoff.ipynb``
and the notebook was the driver. That is what produced
``05_ocr_model_bakeoff (Heavyhorse6's conflicted copy 2026-08-14).ipynb``: a
Dropbox-synced JSON document cannot be the thing two machines both run, because
the run mutates it (outputs, execution counts) on both sides of the sync.

So the run moved here and the notebook became a READER of results. The split is
the point:

  * this module and ``run_bakeoff.py`` hold the matrix, the flags and the loop.
    They are plain text, they diff, and a merge conflict in them is legible.
  * the notebook keeps sections 6a-6f, which read finished CSVs and draw the
    comparison. Two machines can have that open at once without harm.

Pass artefacts are written **per host** (``bakeoff/<host>/``), so two machines
running the same matrix cannot write the same path. Only the summary CSVs need
to travel. The human transcription is NOT per host - there is one gold set and
both machines score against it.

TABLE_CHANNELS = "B,A"

# fails every remaining pass in about a second each, and without this the

ABORT_AFTER = 3

def gold_pages() -> str:
    """The pages the gold sweep reads, from the drawn sample itself.

    Read off `gold_sample_pages.csv` rather than listed here, so that a page added
    to the sample joins the sweep instead of the two quietly diverging — which is
    the same rule :func:`accuracy_pages` follows against the transcription.
    """
    if not GOLD_SAMPLE.exists():
        return ""
    with GOLD_SAMPLE.open(encoding="utf-8", newline="") as fh:
        scans = sorted({int(r["scan"]) for r in csv.DictReader(fh)
                        if r["role"] in GOLD_ROLES})
    return ",".join(str(s) for s in scans)


def accuracy_pages() -> str:
    """The graded pages, READ OFF the human transcription rather than hardcoded,
    so that when the gold set grows the graded pages follow it instead of
    silently staying behind."""
    if not HUMAN.exists():
        return "118,270,450,451"
    with HUMAN.open(encoding="utf-8", newline="") as fh:
        pages = sorted({r["page"] for r in csv.DictReader(fh)})
    return ",".join(str(int(p)) for p in pages) or "118,270,450,451"

def ollama_show(model: str) -> dict:
    response = requests.post(f"{OLLAMA}/api/show", json={"model": model},
                             timeout=60)
    response.raise_for_status()
    return response.json()


def unload(model: str) -> None:
    """Drop the weights from VRAM so the next model has the card to itself."""
    try:
        requests.post(f"{OLLAMA}/api/generate",
                      json={"model": model, "keep_alive": 0}, timeout=60)
    except Exception:                                          # noqa: BLE001
        pass


def daemon_up() -> bool:
    """Is Ollama still answering? Asked only after a pass fails."""
    try:
        requests.get(f"{OLLAMA}/api/tags", timeout=10).raise_for_status()
        return True
    except Exception:                                          # noqa: BLE001
        return False


def pulled_models() -> set:
    try:
        tags = requests.get(f"{OLLAMA}/api/tags", timeout=10).json()
        return {m["name"] for m in tags.get("models", [])}
    except Exception:                                          # noqa: BLE001
        return set()


def select_models(wanted, models: list | None = None) -> list:
    """Narrow a model list by label or by ollama id.

    ``"all"`` (or ``None``) keeps everything. Anything else is a list of short
    labels (``"qwen38"``) or full ids (``"glm-5.3-flash:cloud"``), matched
    case-insensitively.

    **An unknown name raises rather than being skipped.** A typo that silently
    selects nothing would start a night that runs no passes and reports success,
    and a typo that silently selects a subset is worse — the comparison would be
    missing a model and nothing would say which.
    """
    pool = list(models if models is not None else MODELS)
    if wanted in (None, "all", "ALL"):
        return pool
    if isinstance(wanted, str):
        wanted = [wanted]
    asked = [str(w).strip().lower() for w in wanted if str(w).strip()]
    picked, unknown = [], []
    for name in asked:
        hit = [(m, l) for m, l in pool
               if l.lower() == name or m.lower() == name]
        if hit:
            picked += [h for h in hit if h not in picked]
        else:
            unknown.append(name)
    if unknown:
        known = ", ".join(sorted(f"{l} ({m})" for m, l in pool))
        raise SystemExit(f"unknown model(s): {', '.join(unknown)}\n"
                         f"known here: {known}")
    return picked


def runnable(verbose: bool = True) -> list:
    """The models this daemon can actually run, as ``(model, label)``.

    Two different problems that must not be reported as one. A model that is not
    pulled is an `ollama pull` away; a model that is pulled but reports no
    vision cannot do this job at all and no amount of pulling will change that.
    """
    pulled = pulled_models()
    if not pulled:
        if verbose:
            print(f"MISSING ollama at {OLLAMA}. Start the service, or run: "
                  f"ollama serve")
        return []
    out = []
    for model, label in MODELS:
        if model not in pulled:
            if verbose:
                print(f"  {model:22s} not pulled on this daemon")
            continue
        try:
            capabilities = ollama_show(model).get("capabilities", []) or []
        except Exception as exc:                               # noqa: BLE001
            if verbose:
                print(f"  {model:22s} show failed: {exc}")
            continue
        if "vision" not in capabilities:
            if verbose:
                print(f"  {model:22s} reports no vision - cannot read a crop")
            continue
        out.append((model, label))
    return out

#: Where the encoder classes lived before the per-host split.
LEGACY_ENCODERS = OUT_DIR / "bakeoff_encoders.json"


def load_encoders() -> dict:
    """model id -> "dynamic" | "fixed". ONLY the encoder class is persisted: a
    stored crop height outlives the config that produced it, and on 2026-08-14 a
    stored 250 overrode a config that said 500 while the ledger printed 500.

    The fallback to the pre-split file is safe for exactly the reason a stored
    height is not: the class is a fact about the MODEL, not about the run. The
    run tag moved to r3 because `SECTIONS` did, and no setting under this tag can
    change how a model's encoder answers more pixels. Without the fallback this
    returns {} on a fresh host, `resolution_for` then calls every model dynamic,
    and gemma and nemotron - measured `fixed` - are sent `--notes-upscale`: hours
    spent on a no-op, and a `base` that means two different things across one
    matrix.
    """
    for path in (ENCODERS, LEGACY_ENCODERS):
        if path.exists():
            return json.loads(path.read_text(encoding="utf-8"))
    return {}


#: What each kind of pass appends to its label. The page set is part of a pass's
#: identity, not a footnote to it: the same model at the same variant over four
#: pages and over thirteen are two different measurements, and one label for both
#: would make the skip-if-done rule accept whichever ran first.
KIND_SUFFIX = {"accuracy": "", "throughput": "20", "gold": "g", "table": "t"}

#: The kinds of pass there are. `plan` checks against it rather than falling
#: through to the throughput defaults: a mistyped kind would otherwise run the
#: wrong pages under the wrong variants and report nothing unusual.
KINDS = tuple(KIND_SUFFIX)


def label_for(label: str, variant: str, kind: str = "accuracy") -> str:
    """The label one pass writes under: model, run tag, variant, page set.

    Every artefact of a pass - checks_, notes_, inventory_, symbol_join_,
    ratelimit_ - is named from this, so the whole matrix lands in one directory
    without collisions and the notebook can find any pass from its coordinates.

        qwen36_r3base     4 pages          qwen36_r3base20    19 pages
        qwen36_r3atlas    4 pages, atlas   qwen36_r3baseg     13 gold pages
    """
    return f"{label}_{RUN_TAG}{variant}{KIND_SUFFIX.get(kind, '')}"


def resolution_for(model: str, variant: dict, encoder: dict):
    """Upscale and crop height: the measured encoder class, this run's heights."""
    kind = encoder.get(model, "dynamic")
    upscale = ((kind == "dynamic") if variant["upscale"] == "measured"
               else bool(variant["upscale"]))
    height = MAX_HEIGHT if kind == "dynamic" else MAX_HEIGHT_FIXED
    if variant["height"] == "half":
        height = max(120, height // 2)
    return upscale, height


def flags_for(model: str, label: str, pages: str, variant: dict,
              encoder: dict, kind: str = "accuracy") -> list:
    """Every flag this pass sends. The single source of truth for the run AND
    for the ledger, so the table can never describe a setting that was not
    used."""
    if kind == "table":
        # A DIFFERENT CHANNEL AND A DIFFERENT READER. Channel A reads the
        # seventeen printed columns off a block crop; the notes flags below do
        # not reach it and would be silently ignored. `--reader ollama` is a
        # preset for this same daemon, and `--reader-temperature 0` is not
        # optional: an OpenAI-compatible endpoint is sent no temperature unless
        # one is given, and Ollama then samples at 0.8.
        return [
            "--out", str(PASS_DIR),
            "--pages", pages,
            "--channels", TABLE_CHANNELS,
            "--reader", "ollama",
            "--reader-model", model,
            "--reader-temperature", "0",
            "--notes-label", label,
            "--workers", str(WORKERS),
        ]
    upscale, height = resolution_for(model, variant, encoder)
    argv = [
        "--out", str(PASS_DIR),
        "--pages", pages,
        "--channels", "B,G",
        "--notes-reader", "ollama",
        "--notes-model", model,
        "--notes-label", label,
        "--notes-sections", SECTIONS,
        "--notes-num-ctx", str(NUM_CTX),
        "--notes-temperature", "0",
        "--workers", str(WORKERS),
        "--notes-max-height", str(height),
    ]
    if upscale:
        argv += ["--notes-upscale"]
    if variant["atlas"]:
        argv += ["--notes-atlas"]
    if variant.get("think"):
        argv += ["--notes-think",
                 "--notes-max-tokens", str(THINK_MAX_TOKENS)]
    return argv


def applies(model: str, variant: dict, encoder: dict) -> bool:
    """Is this variant meaningful for this model's encoder?"""
    if variant["encoders"] == "any":
        return True
    return encoder.get(model, "dynamic") == variant["encoders"]


VARIANT = {v["name"]: v for v in VARIANTS}
ATLAS_WANTED = any(v["atlas"] for v in VARIANTS)

def n_pages(spec: str) -> int:
    """How many pages a --pages string names: ``118,270`` -> 2, ``118-137`` -> 20."""
    total = 0
    for part in str(spec).split(","):
        if "-" in part:
            first, last = part.split("-", 1)
            total += int(last) - int(first) + 1
        elif part.strip():
            total += 1
    return max(total, 1)


def plan(kind: str, models: list, encoder: dict, only=None) -> list:
    """The worklist: one row per pass, in the order it will run.

    Variant-major: every model at `base`, then every model at `again`, and so on
    down `VARIANTS`. A night that stops early then leaves a complete comparison
    across models at the variants that matter most, rather than the first two
    models finished and the last two never started.
    """
    if kind not in KINDS:
        raise SystemExit(f"unknown kind {kind!r}; known: {', '.join(KINDS)}")
    pages = {"accuracy": accuracy_pages(), "gold": gold_pages(),
             "table": gold_pages()}.get(kind, PAGES_THROUGHPUT)
    names = {"accuracy": [v["name"] for v in VARIANTS],
             "gold": GOLD_VARIANTS,
             "table": TABLE_VARIANTS}.get(kind, THROUGHPUT_VARIANTS)
    wanted = [VARIANT[n] for n in names if n in VARIANT]
    if only:
        # A NARROWING THAT SELECTS NOTHING SAYS SO. Silently, this returns an
        # empty worklist, `run_matrix` prints "nothing outstanding", and a night
        # that was meant to read thirteen pages with six models ends in under a
        # second looking like a completed run. `select_models` has raised on an
        # unknown model since it was written, for exactly this reason; the
        # variant filter beside it did not.
        #
        # Two different mistakes, reported differently. A name that is not a
        # variant at all is a typo and raises. A real variant that this kind does
        # not have - `noup` under `gold`, which is base and atlas only - is a
        # misunderstanding of the matrix, and saying which variants the kind does
        # have is more use than a traceback.
        unknown = [n for n in only if n not in VARIANT]
        if unknown:
            raise SystemExit(f"unknown variant(s): {', '.join(unknown)}\n"
                             f"known: {', '.join(VARIANT)}")
        missing = [n for n in only if n not in names]
        if missing:
            print(f"  .. {kind} has no {', '.join(missing)} variant - it runs "
                  f"{', '.join(names)}")
        wanted = [v for v in wanted if v["name"] in only]
        if not wanted:
            print(f"  !! nothing in --only ({', '.join(only)}) is a {kind} "
                  f"variant, so this plan is EMPTY and nothing will run.")
    # Cloud models last, whatever order they sit in `MODELS`. A parked cloud pass
    # costs nothing once the card has finished its own work; parked first it would
    # leave the GPU idle behind a quota it cannot influence.
    models = sorted(models, key=lambda m: m[0] in CLOUD_MODELS)
    jobs = []
    for variant in wanted:
        for model, label0 in models:
            if model in NOT_NAMERS or not applies(model, variant, encoder):
                continue
            label = label_for(label0, variant["name"], kind)
            upscale, height = resolution_for(model, variant, encoder)
            jobs.append({
                "kind": kind, "model": model, "variant": variant["name"],
                "label": label, "pages": pages,
                "atlas": variant["atlas"], "upscale": upscale,
                "height": height, "num_ctx": NUM_CTX,
                "cloud": model in CLOUD_MODELS,
                # A pass that PARKED is not done. Its inventory exists and is
                # short, so the plain "inventory is on disk" test would skip it
                # for ever and the model would never be retried on a fresh quota.
                "done": ((PASS_DIR / f"inventory_{label}.csv").exists()
                         and not (PASS_DIR / f"ratelimit_{label}.json").exists()),
                "parked": (PASS_DIR / f"ratelimit_{label}.json").exists(),
            })
    return jobs


#: Where the timings lived before the per-host split.
LEGACY_TIMINGS = OUT_DIR / "bakeoff_timings.json"


def load_timings() -> dict:
    """This host's timings, falling back to the pre-split file.

    The fallback is not cosmetic: :func:`estimate` is built entirely out of
    seconds-per-page history, so a fresh per-host file would make every first
    run on a machine report the 40 s guess and call it a measurement.
    """
    for path in (TIMINGS, LEGACY_TIMINGS):
        if path.exists():
            return json.loads(path.read_text(encoding="utf-8"))
    return {}


def estimate(jobs: list, timings: dict, force: bool = False) -> float:
    """Hours the outstanding jobs will take, from what this machine has measured.

    A model with no history gets the field's median, and a first-ever run gets
    40 s a page, which is the only number here that is a guess.
    """
    history: dict = {}
    for stamp in timings.values():
        for run in stamp.values():
            if run.get("seconds") and run.get("model") and run.get("pages"):
                history.setdefault(run["model"], []).append(
                    run["seconds"] / n_pages(run["pages"]))
    per_model = {m: sum(v) / len(v) for m, v in history.items()}
    default = (sorted(per_model.values())[len(per_model) // 2]
               if per_model else 40.0)
    return sum(per_model.get(job["model"], default) * n_pages(job["pages"])
               for job in jobs if force or not job["done"]) / 3600


def show_plan(jobs: list, timings: dict, force: bool = False) -> None:
    todo = [j for j in jobs if force or not j["done"]]
    print(f"  {len(jobs)} passes, {len(jobs) - len(todo)} already on disk, "
          f"{len(todo)} to run   (~{estimate(jobs, timings, force):.1f} h)")
    print()
    print(f"  {'label':24s} {'model':22s} {'atlas':>5s} {'up':>3s} "
          f"{'height':>6s}  status")
    for job in jobs:
        print(f"  {job['label']:24s} {job['model']:22s} "
              f"{str(job['atlas']):>5s} {str(job['upscale'])[0]:>3s} "
              f"{job['height']:6d}  "
              + ("done" if job["done"] and not force else "to run"))

def file_stamp(path: Path):
    """This file's modification time in ns, or ``None`` if it is not there.

    Compared BEFORE and AFTER a pass to answer "did this run write it", which
    ``.exists()`` cannot: every artefact is named from the label, so the previous
    attempt at the same label left a file with exactly that name. Relative, not
    absolute - two stamps off the same filesystem, never a wall clock compared
    with a file's, so no clock skew can make a finished pass look empty.
    """
    try:
        return path.stat().st_mtime_ns
    except OSError:
        return None


class Tee(io.TextIOBase):
    """Print to the console and to a log file at once.

    The runs are long. Without this the choice is a live view that vanishes when
    the terminal closes, or a log nobody watches.
    """

    def __init__(self, *streams):
        self.streams = streams

    def write(self, text):
        for stream in self.streams:
            try:
                stream.write(text)
            except UnicodeEncodeError:
                # ONE STREAM'S ALPHABET MUST NOT END A PASS. The parts ledger
                # prints the marks the OCR engine returned - `Ш`, `©`, `«“■` -
                # and a cp1250 console cannot encode the first of those. Left to
                # raise, it kills the pilot *after* the crops have been read and
                # paid for, `run_pilot` records the pass as a failure, and three
                # of those abort the rest of the matrix. The log file is UTF-8
                # and keeps the real characters either way.
                stream.write(text.encode("ascii", "replace").decode("ascii"))
        return len(text)

    def flush(self):
        for stream in self.streams:
            try:
                stream.flush()
            except Exception:                                  # noqa: BLE001
                pass


def run_pilot(argv: list, log_name: str):
    """Drive the pilot in-process, tee its output to a log, return (rc, seconds)."""
    ensure_dirs()
    live = sys.stdout
    log_path = LOG_DIR / f"{log_name}.log"
    started = time.time()
    sys.argv = ["pilot", *argv]
    with log_path.open("w", encoding="utf-8") as fh:
        with redirect_stdout(Tee(live, fh)):
            print("$ python utils/lexikon_1910/run_lexikon_1910_pilot.py "
                  + " ".join(argv) + "\n")
            try:
                rc = pilot.main()
            except SystemExit as exc:
                rc = int(exc.code or 0)
            except Exception as exc:                           # noqa: BLE001
                rc = 1
                print(f"\n!! {type(exc).__name__}: {exc}")
    return rc, time.time() - started


def run_matrix(kind: str, models: list, encoder: dict,
               force: bool = False, only=None) -> list:
    """Run every outstanding pass of one kind, resumably, and record what each
    one actually did.

    Two things make this safe to leave alone overnight. A pass that raises is
    caught by :func:`run_pilot` and costs one run, not the night. And a run of
    failures is checked against the daemon: if Ollama has died, every remaining
    pass would fail in about a second each and the morning's result would be an
    empty matrix, so the loop stops and says so instead.
    """
    ensure_dirs()
    timings = load_timings()
    jobs = plan(kind, models, encoder, only)
    todo = [j for j in jobs if force or not j["done"]]
    show_plan(jobs, timings, force)
    # TWO WAYS TO HAVE NOTHING TO DO, AND ONLY ONE OF THEM IS FINE. An empty
    # WORKLIST means the matrix selected nothing at all - no model reached
    # `plan`, or `only` narrowed to a variant this kind does not have - and the
    # single old message reported that as a completed night. `select_models`
    # has raised on an unknown model name since it was written; this is that
    # same rule extended to the other two ways the list can come out empty.
    if not jobs:
        raise SystemExit(
            f"\n!! plan({kind!r}) is EMPTY - {len(models)} model(s) in, "
            f"only={only!r}. Nothing would run, and without this the matrix "
            f"would report a finished night. Check the model list and `only`.")
    # An empty `todo` against a full `jobs` is the resume rule working, so it
    # names the directory it read that from: the answer is per host, and a
    # BAKEOFF_HOST pointing somewhere unexpected looks exactly like this.
    if not todo:
        print(f"\n  nothing outstanding for {kind}: all {len(jobs)} passes "
              f"are already on disk in {PASS_DIR}\n  Pass force=True to "
              f"re-run them.")
        return jobs
    started_all, failures, done = time.time(), 0, []
    park: dict = {}
    for n, job in enumerate(todo, 1):
        # PARK THE MODEL, NOT THE PASS. A spent quota belongs to the account and
        # the window, so the next variant of the same model would hit it too — and
        # would spend a fresh set of requests to find out. Every OTHER model keeps
        # running, which is the whole point: the night still produces a complete
        # local comparison, and what the cloud model managed is reported beside it
        # rather than mixed into it.
        if job["model"] in park:
            print(f"\n[{n}/{len(todo)}]  SKIPPED  {job['label']} — "
                  f"{job['model']} was parked at {park[job['model']]}")
            continue
        elapsed_h = (time.time() - started_all) / 3600
        print("\n" + "=" * 78)
        print(f"[{n}/{len(todo)}]  {kind.upper()}  {job['model']}  ->  "
              f"{job['label']}")
        if kind == "table":
            print(f"        pages {job['pages']}, channels {TABLE_CHANNELS}, "
                  f"temperature 0 — the 17 printed columns, per municipality "
                  f"block")
        else:
            print(f"        pages {job['pages']}, atlas={job['atlas']}, "
                  f"upscale={job['upscale']}, max_height={job['height']}, "
                  f"num_ctx={job['num_ctx']}, sections={SECTIONS}")
        print(f"        {VARIANT[job['variant']]['why']}")
        print(f"        {elapsed_h:.1f} h in, "
              f"{time.strftime('%H:%M', time.localtime())}")
        print("=" * 78)
        argv = flags_for(job["model"], job["label"], job["pages"],
                         VARIANT[job["variant"]], encoder, kind)
        # A PASS IS JUDGED ON THE FILES *IT* WROTE, AND ONLY THOSE. Every
        # artefact is named from the label, so a re-run of the same label lands
        # on top of the previous attempt's files - and until now the two were
        # read as one. Both halves of that were wrong in the same direction, and
        # the marker was the half that ended nights:
        #
        #   * a stale `ratelimit_<label>.json` from a spent quota is what
        #     `plan()` uses to keep the pass outstanding, so the retry is
        #     correctly scheduled - and then the marker is still on disk when the
        #     retry finishes, so a pass that read all 81 crops on a fresh quota
        #     was recorded INCOMPLETE, the model was parked, every later pass of
        #     it was SKIPPED, and the whole matrix ended in minutes. The cloud
        #     model could never come back.
        #   * a stale `inventory_<label>.csv` makes `wrote` true for a pass that
        #     wrote nothing, which under `--force` turns a failed re-run into a
        #     success and stops ABORT_AFTER ever counting it.
        #
        # The marker is MOVED rather than ignored: `plan`, `passes_on_disk` and
        # `rate_limit_events` all key on its name, so leaving it beside a pass
        # that has since succeeded would keep that pass out of every table. The
        # superseded copy stays on disk under a name none of those globs match,
        # because what a limit cost is a measurement and not a scratch file.
        limit_file = PASS_DIR / f"ratelimit_{job['label']}.json"
        if limit_file.exists():
            kept = (PASS_DIR / f"superseded_ratelimit_{job['label']}"
                              f"_{time.strftime('%Y%m%d-%H%M%S')}.json")
            limit_file.replace(kept)
            print(f"        a previous attempt was rate limited; that marker is "
                  f"now {kept.name}\n"
                  f"        and this pass is judged on its own "
                  f"result")
        inventory_file = PASS_DIR / f"inventory_{job['label']}.csv"
        inventory_was = file_stamp(inventory_file)
        rc, seconds = run_pilot(argv, f"{kind}_{job['label']}")
        # Unload only when the next pass needs a different model. The card holds
        # one 30B vision model, so the swap is obligatory - but paying for it
        # between two passes of the SAME model is a reload bought for nothing.
        if n >= len(todo) or todo[n]["model"] != job["model"]:
            unload(job["model"])
        # The settings are recorded WITH the timing, so the notebook reports the
        # flags this pass sent rather than whatever the config says when the
        # table is drawn. That is the 2026-08-14 defect, fixed at the source.
        timings.setdefault(job["label"], {})[kind] = {
            "model": job["model"], "variant": job["variant"],
            "pages": job["pages"], "seconds": seconds, "rc": rc,
            "atlas": job["atlas"], "upscale": job["upscale"],
            "max_height": job["height"], "num_ctx": job["num_ctx"],
            "sections": SECTIONS, "run_tag": RUN_TAG, "host": HOST,
            "finished": time.strftime("%Y-%m-%d %H:%M:%S"),
        }
        # Changed, not merely present: see the note above the run.
        wrote = file_stamp(inventory_file) not in (None, inventory_was)
        limited = None
        if limit_file.exists():
            limited = json.loads(limit_file.read_text(encoding="utf-8"))
            park[job["model"]] = limited["limit_at"]
        # `incomplete` is what stops this pass being read as a result. The
        # notebook drops it from every accuracy table rather than scoring a model
        # on the crops it was allowed to send before the quota ran out.
        timings[job["label"]][kind]["incomplete"] = bool(limited)
        if limited:
            timings[job["label"]][kind]["rate_limit"] = limited
        TIMINGS.write_text(json.dumps(timings, indent=2), encoding="utf-8")
        print(f"\n  [{job['label']}: rc={rc}, {seconds / 60:.1f} min, "
              f"inventory {'written' if wrote else 'MISSING'}"
              + (f", INCOMPLETE — rate limited after "
                 f"{limited['crops_done_before_limit']}/"
                 f"{limited['crops_total']} crops]" if limited else "]"))
        done.append(job)
        # A parked pass is neither a success nor a fault of the rig, so it must
        # not count toward ABORT_AFTER — three parked cloud passes would
        # otherwise stop a matrix whose local models are running perfectly.
        if limited:
            failures = 0
        else:
            failures = 0 if (rc == 0 and wrote) else failures + 1
        if failures >= ABORT_AFTER:
            alive = daemon_up()
            print(f"\n!! {failures} passes in a row produced nothing. Ollama is "
                  f"{'up' if alive else 'DOWN'}.")
            print("   Stopping here rather than burning the rest of the matrix "
                  "on the same fault.")
            print("   Fix it, then re-run: finished passes are skipped, so it "
                  "resumes.")
            break
    print(f"\n{'=' * 78}\n{kind}: {len(done)} passes in "
          f"{(time.time() - started_all) / 3600:.1f} h")
    return jobs

def load_pass(path) -> list:
    with Path(path).open(encoding="utf-8", newline="") as fh:
        return list(csv.DictReader(fh))


def passes_on_disk(legacy: bool = False, hosts: str = "",
                   include_incomplete: bool = False) -> dict:
    """Every inventory of this run tag, keyed by label.

    ``hosts="all"`` pools every machine's directory, which is how two nights on
    two cards become one table. A label that appears on both is kept once, from
    whichever host sorts first, and the collision is REPORTED rather than
    silently resolved - two machines that ran the same cell should agree, and if
    they do not that is the finding, not a coin toss.

    **A rate-limited pass is dropped by default, and that is the guard the whole
    parking design exists for.** Its inventory is on disk and is short — the
    model read some crops, then the quota refused the rest — so nothing about the
    file itself says it was cut off. Scored, it reports a cloud model as a poor
    reader of the register when what was actually measured was an account limit.
    The marker written beside it says so, and this drops it. ``include_incomplete``
    is for looking at what it *did* manage, never for a comparison table.
    """
    out: dict = {}
    seen: dict = {}
    directories = []
    if hosts == "all":
        directories = sorted(d for d in BAKEOFF.glob("*") if d.is_dir())
    elif PASS_DIR.exists():
        directories = [PASS_DIR]
    if legacy:
        directories.append(LEGACY_DIR)
    for directory in directories:
        for path in sorted(directory.glob("inventory_*.csv")):
            label = path.stem[len("inventory_"):]
            if label == "human":
                continue
            mine = any(label.startswith(f"{l}_{RUN_TAG}")
                       for _m, l in ALL_MODELS)
            if not (mine or legacy):
                continue
            if not include_incomplete and (
                    directory / f"ratelimit_{label}.json").exists():
                print(f"  .. {label} was rate limited and is INCOMPLETE - "
                      f"excluded (run_bakeoff.py --limits)")
                continue
            if label in out:
                print(f"  !! {label} exists on {seen[label]} and "
                      f"{directory.name} - keeping {seen[label]}")
                continue
            out[label] = load_pass(path)
            seen[label] = directory.name
    return out

def encoder_crop(page_number: int = 270, upscale: bool = False):
    """The tallest offices section of one page - the crop a real pass sends."""
    from utils.lexikon_1910 import lexikon_1910 as lx
    from utils.lexikon_1910 import lexikon_1910_notes as lxnt
    page = lx.read_alto(pilot.CZ_PAGES / f"{page_number:04d}.alto.xml")
    notes = lxnt.extract_notes(page)
    if notes is None:
        raise RuntimeError(f"page {page_number} has no footnote divider")
    image = lx.render_page(pilot.CZ_PDF, page_number, page.width)
    wanted = [s for s in lxnt.section_segments(notes, page)
              if s.kind == "offices"]
    target = max(wanted, key=lambda s: s.height)
    crops = lxnt.crop_segments(image, [target], page, upscale=upscale,
                               max_height=MAX_HEIGHT)
    return crops[0][0]


def prompt_tokens(model: str, crop) -> int:
    """What the daemon says this image cost to encode."""
    body = {"model": model, "stream": False, "think": False,
            "keep_alive": "10m",
            "options": {"temperature": 0, "num_predict": 1,
                        "num_ctx": NUM_CTX},
            "messages": [{"role": "user", "content": "Transcribe.",
                          "images": [pilot.encode_png(crop)]}]}
    response = requests.post(f"{OLLAMA}/api/chat", json=body, timeout=900)
    response.raise_for_status()
    return response.json().get("prompt_eval_count", 0) or 0


def measure_encoders(models: list, force: bool = False) -> dict:
    """Classify each model's vision encoder, and persist ONLY the class.

    Same crop, native and upscaled, num_predict=1 - only the PROMPT tokens are
    compared, so this costs two forward passes per model and no generation. A
    dynamic encoder spends more patches on more pixels; a fixed-grid one throws
    them away, and the ratio says which.
    """
    ensure_dirs()
    encoder = load_encoders()
    todo = [(m, l) for m, l in models if force or m not in encoder]
    if not todo:
        print("  every model already classified - "
              + ", ".join(f"{m}={k}" for m, k in encoder.items()))
        return encoder
    native, big = encoder_crop(upscale=False), encoder_crop(upscale=True)
    print(f"crop native {native.width}x{native.height}, "
          f"upscaled {big.width}x{big.height} "
          f"({big.width * big.height / (native.width * native.height):.1f}x "
          f"the pixels)\n")
    print(f"  {'model':22s} {'native':>8s} {'upscaled':>9s} {'ratio':>7s}  "
          f"encoder")
    for model, _label in todo:
        try:
            small = prompt_tokens(model, native)
            large = prompt_tokens(model, big)
        except Exception as exc:                               # noqa: BLE001
            print(f"  {model:22s} probe failed: {exc}")
            continue
        finally:
            unload(model)
        ratio = (large / small) if small else 0.0
        # 1.25x is well clear of both outcomes measured on this crop: a dynamic
        # encoder came back at 1.66x and a fixed-grid one at 0.99x. It is a
        # threshold between two clusters, not a tuned parameter.
        kind = "dynamic" if ratio >= 1.25 else "fixed"
        encoder[model] = kind
        print(f"  {model:22s} {small:8d} {large:9d} {ratio:7.2f}  {kind}")
    ENCODERS.write_text(json.dumps(encoder, indent=2), encoding="utf-8")
    print(f"\n  written to {ENCODERS}")
    return encoder


def probe(models: list, encoder: dict, page: int = 270,
          force: bool = False) -> None:
    """One crop each: does this model see the image at all, and how fast.

    Read by eye, and its log is kept - so a model already probed under this run
    tag has nothing left to show, and re-probing it is a model load spent on a
    log that already exists.
    """
    ensure_dirs()
    for model, label in models:
        log = LOG_DIR / f"probe_{label}_{RUN_TAG}.log"
        if log.exists() and not force:
            print(f"  probe {model:22s} already on disk - {log.name}")
            continue
        upscale, height = resolution_for(model, VARIANT["base"], encoder)
        print("\n" + "=" * 78)
        print(f"PROBE  {model}   (upscale={upscale}, max_height={height})")
        print("=" * 78)
        argv = ["--out", str(PASS_DIR), "--probe", "--probe-page", str(page),
                "--notes-reader", "ollama", "--notes-model", model,
                "--notes-sections", SECTIONS,
                "--notes-num-ctx", str(NUM_CTX),
                "--notes-temperature", "0",
                "--notes-max-height", str(height)]
        if upscale:
            argv += ["--notes-upscale"]
        _rc, elapsed = run_pilot(argv, f"probe_{label}_{RUN_TAG}")
        unload(model)
        print(f"\n  [probe wall clock {elapsed:.0f}s including model load]")

def rate_limit_events(hosts: str = "") -> list:
    """Every parked pass on disk, newest first."""
    directories = (sorted(d for d in BAKEOFF.glob("*") if d.is_dir())
                   if hosts == "all" else [PASS_DIR])
    out = []
    for directory in directories:
        for path in sorted(directory.glob("ratelimit_*.json")):
            try:
                record = json.loads(path.read_text(encoding="utf-8"))
            except (OSError, ValueError):
                continue
            record["host"] = directory.name
            out.append(record)
    return sorted(out, key=lambda r: r.get("limit_at", ""), reverse=True)


def volume_crops(sections: str = "") -> float:
    """Crops in a full Čechy sweep at this section setting."""
    per_page = pilot.VOLUME_SECTIONS_PER_PAGE.get(sections or SECTIONS, 3.0)
    return pilot.N_TABLE_PAGES * per_page


def lexikon_1910_rate_limit_report(events: list) -> str:
    """What the limit was, and whether the model can read the volume behind it.

    Three questions, and the third is the one that decides whether a cloud model
    is a candidate at all:

    * **When did it stop, and how much had it read?** Measured exactly — crops
      completed, seconds elapsed, and the rate between them.
    * **How big is the limit?** Only partly answerable from one pass. The crops
      completed before the refusal is a **lower bound** on the allowance, not the
      allowance: the window may have been part-spent when the run started, and
      nothing here can see that. It is reported as a bound and never as the limit.
    * **Could it do the full sweep?** That needs the reset window, which is
      knowable only if the server sent `Retry-After`. Where it did, the arithmetic
      is honest and printed. Where it did not, **no number is invented** — the
      report says what would have to be measured instead.
    """
    lines = [f"bakeoff v{__version__} — rate limits", ""]
    if not events:
        return "\n".join(lines + ["  no pass has been rate limited on this host"])
    total = volume_crops()
    for event in events:
        done = event.get("crops_done_before_limit") or 0
        seconds = event.get("seconds_before_limit") or 0
        rate = event.get("crops_per_second_before_limit")
        retry = event.get("retry_after_seconds")
        lines.append(f"  {event.get('label', '?')}   {event.get('model', '?')}"
                     f"   {event.get('limit_at', '?')}")
        lines.append(f"    read {done} of {event.get('crops_total', '?')} crops "
                     f"in {seconds:.0f}s before the refusal"
                     + (f" ({rate:.3f} crops/s)" if rate else ""))
        lines.append(f"    status {event.get('status')}   "
                     f"{str(event.get('error', ''))[:90]}")
        lines.append(f"    allowance: AT LEAST {done} requests in "
                     f"{seconds / 60:.1f} min — a lower bound, because the "
                     f"window may")
        lines.append(f"               already have been part spent when this "
                     f"pass started")
        lines.append(f"    full Čechy sweep at sections={SECTIONS}: "
                     f"{total:.0f} crops")
        if retry:
            batches = max(0, -(-total // max(done, 1)) - 1)
            waiting = batches * retry / 3600
            reading = (total / rate / 3600) if rate else float("nan")
            lines.append(f"    the server asked for {retry:.0f}s before "
                         f"retrying, so at {done} crops a window:")
            lines.append(f"      {batches:.0f} waits x {retry / 60:.1f} min = "
                         f"{waiting:.1f} h idle"
                         + (f", plus {reading:.1f} h reading = "
                            f"{waiting + reading:.1f} h total"
                            if rate else ""))
        else:
            lines.append(f"    the server sent NO Retry-After, so the reset "
                         f"window is unknown and the")
            lines.append(f"    sweep cost cannot be computed. To measure it: "
                         f"re-run this one pass and")
            lines.append(f"    time the gap until a request succeeds again — "
                         f"that gap is the window.")
        lines.append("")
    lines.append("  A parked pass is marked `incomplete` in timings.json and is "
                 "excluded from")
    lines.append("  every accuracy table. It measures the quota, not the model.")
    return "\n".join(lines)
