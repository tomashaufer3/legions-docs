How it works

The reading pass (reference/background/_evidence/) left 73 ledgers in ~470 part files and 13 topic digests. This script joins them: every ledger entry is parsed with its fields (date, theatre, units, page, confidence, note), every digest line is matched back to its entry, so each claim knows which digest group(s) hold it. The result is one self-contained HTML page (data gzipped inline, Libertinus embedded) that shows the topics as a map, each digest's groups as a tree, the contradiction notes and gaps, and a search over all claims.

Orientation only: nothing here is a pipeline input, and nothing is added to the ledgers. Entries tagged only POLITICS or OTHER reach no digest (an open decision of the reading pass); the page shows them grouped by theatre so they are not invisible.

usage: python utils/reference_tools/background/build_reading_map.py writes: output/reading_map/reading_map.html (gitignored; rebuild after any digest rebuild)

def parse_ledgers() -> tuple[list[dict], dict]:
    entries, headers = [], {}
    for path in sorted((EVIDENCE / "parts").glob("*.part*.md"), key=_part_sort_key):
        key = path.name.split(".part")[0]
        text = path.read_text(encoding="utf-8")
        head = headers.setdefault(key, {"key": key, "work": work_of(key), "parts": 0})
        head["parts"] += 1
        for field in ("source", "pages", "covers"):
            m = re.search(rf"^- {field}:\s*(.+)$", text, re.M)
            if m and field not in head:
                head[field] = m.group(1).strip()
        for block in re.split(r"^### ", text, flags=re.M)[1:]:
            e = _parse_block(block)
            if e:
                e["key"] = key
                entries.append(e)
    for i, e in enumerate(entries):
        e["id"] = i
        e["year"] = year_of(e.get("date", ""))
        e["th"] = theatre_main(e.get("theatre", ""))
        e["topics"] = sorted({TAG_TO_TOPIC[t] for t in e["tags"] if t in TAG_TO_TOPIC},
                             key=TOPICS.index)
        headers[e["key"]]["n"] = headers[e["key"]].get("n", 0) + 1
    return entries, headers

def md_inline(text: str) -> str:
    out = html.escape(text, quote=False)
    spans: list[str] = []  # code spans and refs are set aside so * inside them stays literal

    def keep(fragment: str) -> str:
        spans.append(fragment)
        return f"\x00{len(spans) - 1}\x00"

    out = re.sub(r"`([a-z0-9_]+) · ([^`]+)`",
                 lambda m: keep(f'<button class="ref" data-k="{m.group(1)}" '
                                f'data-p="{first_num(m.group(2))}">{m.group(1)} · {m.group(2)}</button>'),
                 out)
    out = re.sub(r"`([^`]+)`", lambda m: keep(f"<code>{m.group(1)}</code>"), out)
    out = re.sub(r"\*\*(.+?)\*\*", r"<strong>\1</strong>", out)
    out = re.sub(r"(?<![\w*])\*(?!\s)(.+?)(?<!\s)\*(?![\w*])", r"<em>\1</em>", out)
    return re.sub(r"\x00(\d+)\x00", lambda m: spans[int(m.group(1))], out)

def md_block(text: str) -> str:
    """Paragraphs, bullet lists and pipe tables: all the digests' prose sections use."""
    out, para, items, rows = [], [], [], []

    def flush():
        if para:
            out.append("<p>" + md_inline(" ".join(para)) + "</p>")
            para.clear()
        if items:
            out.append("<ul>" + "".join(f"<li>{md_inline(i)}</li>" for i in items) + "</ul>")
            items.clear()
        if rows:
            cells = [[c.strip() for c in r.strip().strip("|").split(" | ")] for r in rows
                     if not re.match(r"^\|[\s|:-]+\|$", r.strip())]
            head, body = cells[0], cells[1:]
            out.append('<div class="tblwrap"><table><thead><tr>'
                       + "".join(f"<th>{md_inline(c)}</th>" for c in head) + "</tr></thead><tbody>"
                       + "".join("<tr>" + "".join(f"<td>{md_inline(c)}</td>" for c in r) + "</tr>"
                                 for r in body) + "</tbody></table></div>")
            rows.clear()

    for line in text.split("\n"):
        s = line.strip()
        if not s:
            flush()
        elif s.startswith("|"):
            if para or items:
                flush()
            rows.append(s)
        elif s.startswith("- "):
            if para or rows:
                flush()
            items.append(s[2:])
        elif s.startswith("> "):
            flush()
            out.append('<blockquote>' + md_inline(s[2:]) + "</blockquote>")
        elif re.match(r"^#{2,4} ", s):
            flush()
            out.append("<h4>" + md_inline(s.lstrip("# ")) + "</h4>")
        else:
            if items:
                items[-1] += " " + s
            else:
                if rows:
                    flush()
                para.append(s)
    flush()
    return "".join(out)

def parse_digest(tag: str) -> dict:
    text = (EVIDENCE / "digests" / f"{tag}.md").read_text(encoding="utf-8")
    title = text.split("\n", 1)[0].lstrip("# ").strip()
    sections = re.split(r"^## ", text, flags=re.M)
    bullets = _header_bullets(sections[0])
    digest = {"tag": tag, "title": title, "covers": md_inline(covers_text(bullets.get("Covers", ""))),
              "reading": md_inline(bullets.get("Reading note", "")),
              "built": md_inline(bullets.get("Built", "")), "groups": [], "extra": [],
              "gaps": ""}
    for sec in sections[1:]:
        heading, _, body = sec.partition("\n")
        heading = heading.strip()
        if heading.lower().startswith("gaps"):
            digest["gaps"] = md_block(body)
        elif re.match(r"^\d+\.", heading):
            digest["groups"].append(_parse_group(heading, body))
        elif heading.lower() != "table":
            digest["extra"].append({"title": md_inline(heading), "html": md_block(body)})
    return digest

def match_lines(digest: dict, by_key: dict, tag: str) -> Counter:
    """Point every digest line at its ledger entry: same key, claim prefix, page breaks ties."""
    stats = Counter()
    for gi, group in enumerate(digest["groups"]):
        for sub in group["subs"]:
            ids = []
            for line in sub["lines"]:
                target = norm(line["claim"])[:70]
                cands = [e for e in by_key.get(line["key"], [])
                         if tag in e["topics"] and norm(e["claim"]).startswith(target)]
                if len(cands) > 1:
                    cands = [e for e in cands if first_num(e.get("page", "")) == line["page"]] or cands
                if cands:
                    ids.append(cands[0]["id"])
                    stats["matched"] += 1
                else:
                    stats["unmatched"] += 1
            sub["ids"] = ids
            del sub["lines"]
    return stats

def theatre_groups(entries: list[dict], topic: str, note: str) -> list[dict]:
    """Groups by theatre for the topics with no grouped digest (MOVE+EVENT, POLITICS, OTHER)."""
    by_th = defaultdict(list)
    for e in entries:
        if topic in e["topics"]:
            by_th[e["th"]].append(e)
    groups = []
    for i, th in enumerate(t for t in THEATRES if by_th[t]):
        rows = sorted(by_th[th], key=lambda e: (e["year"] or 9999, e.get("date", "")))
        groups.append({"num": str(i + 1), "title": th, "lead": note, "notes": [],
                       "subs": [{"title": "", "ids": [e["id"] for e in rows]}]})
    return groups

def build_topics(entries: list[dict]) -> tuple[list[dict], dict]:
    by_key = defaultdict(list)
    for e in entries:
        by_key[e["key"]].append(e)
    topics, stats = [], {}
    for tag in TOPICS:
        if tag in DIGESTED:
            d = parse_digest(tag)
            if tag == "MOVE_EVENT":
                d["groups"] = theatre_groups(
                    entries, tag, "The digest is one dated table; here split by the ledger's theatre, in date order.")
                stats[tag] = {"matched": sum(tag in e["topics"] for e in entries), "unmatched": 0}
            else:
                stats[tag] = dict(match_lines(d, by_key, tag))
        else:
            d = {"tag": tag, "title": f"{tag} (no digest)", "covers": "", "reading": "",
                 "built": "", "extra": [], "gaps": "",
                 "groups": theatre_groups(entries, tag, "No digest exists for this tag; grouped by the ledger's theatre.")}
            stats[tag] = {"matched": sum(tag in e["topics"] for e in entries), "unmatched": 0}
        placed = {i for g in d["groups"] for s in g["subs"] for i in s["ids"]}
        missing = [e["id"] for e in entries if tag in e["topics"] and e["id"] not in placed]
        if missing:  # tagged but no digest line matched: keep them visible
            d["groups"].append({"num": "–", "title": "Not matched to a digest line", "lead":
                                "Entries with this tag whose digest line the build could not find.",
                                "notes": [], "subs": [{"title": "", "ids": missing}]})
        d["label"] = TOPIC_LABEL[tag]
        d["priority"] = tag in PRIORITY
        d["digested"] = tag in DIGESTED
        d["n"] = sum(tag in e["topics"] for e in entries)
        topics.append(d)
    return topics, stats

def cooccurrence(entries: list[dict]) -> list[list[int]]:
    idx = {t: i for i, t in enumerate(TOPICS)}
    mat = [[0] * len(TOPICS) for _ in TOPICS]
    for e in entries:
        for a in e["topics"]:
            for b in e["topics"]:
                mat[idx[a]][idx[b]] += 1
    return mat

def payload(entries, headers, topics) -> dict:
    keys = sorted(headers, key=lambda k: (list(WORKS).index(next(p for p in WORKS if k.startswith(p))), k))
    kidx = {k: i for i, k in enumerate(keys)}
    rows = []
    for e in entries:
        rows.append([kidx[e["key"]], " ".join(e["tags"]), e["claim"], e.get("date", ""),
                     e["year"] or 0, e["th"], e.get("theatre", ""), (e.get("units", "") or "")[:140],
                     e.get("page", ""), conf_level(e.get("confidence", "")),
                     e.get("confidence", "") if conf_level(e.get("confidence", "")) != "high" else "",
                     e.get("note", "")])
    return {"built": date.today().isoformat(), "version": __version__,
            "topics": topics, "topicOrder": TOPICS, "theatres": THEATRES,
            "ledgers": [headers[k] for k in keys], "works": list(WORKS.values()),
            "cooc": cooccurrence(entries), "E": rows}

def fontfaces() -> str:
    faces = [("Libertinus Sans", "normal", 400, "LibertinusSans-Regular"),
             ("Libertinus Sans", "italic", 400, "LibertinusSans-Italic"),
             ("Libertinus Sans", "normal", 700, "LibertinusSans-Bold"),
             ("Libertinus Serif", "normal", 700, "LibertinusSerif-Bold")]
    css = []
    for family, style, weight, name in faces:
        b64 = base64.b64encode((FONTS / f"{name}.woff").read_bytes()).decode()
        css.append(f'@font-face{{font-family:"{family}";font-style:{style};font-weight:{weight};'
                   f'font-display:swap;src:url(data:font/woff;base64,{b64}) format("woff")}}')
    return "\n".join(css)

def main() -> None:
    entries, headers = parse_ledgers()
    topics, stats = build_topics(entries)
    data = json.dumps(payload(entries, headers, topics), ensure_ascii=False, separators=(",", ":"))
    blob = base64.b64encode(gzip.compress(data.encode("utf-8"), 9)).decode()
    page = (TEMPLATE.read_text(encoding="utf-8")
            .replace("/*{{FONTFACES}}*/", fontfaces())
            .replace("{{DATA}}", blob)
            .replace("{{N_ENTRIES}}", f"{len(entries):,}"))
    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text(page, encoding="utf-8")

    print(f"ledgers {len(headers)}, entries {len(entries):,}")
    for tag in TOPICS:
        s = stats[tag]
        print(f"  {tag:<11} {s.get('matched', 0):>5} matched  {s.get('unmatched', 0):>4} unmatched")
    undig = sum(1 for e in entries if not set(e["topics"]) & set(DIGESTED))
    bad = Counter(t for e in entries for t in e["tags"] if t not in TAG_TO_TOPIC)
    print(f"entries in no digest: {undig:,};  tags outside the brief: {dict(bad)}")
    print(f"json {len(data) / 1e6:.1f} MB -> gzip+b64 {len(blob) / 1e6:.1f} MB; page {OUT.stat().st_size / 1e6:.1f} MB")
    print(f"wrote {OUT.relative_to(ROOT)}")

"""Build the reading-pass map: one browsable page over the ledgers and the 13 digests.

The reading pass (reference/background/_evidence/) left 73 ledgers in ~470 part files and
13 topic digests. This script joins them: every ledger entry is parsed with its fields
(date, theatre, units, page, confidence, note), every digest line is matched back to its
entry, so each claim knows which digest group(s) hold it. The result is one self-contained
HTML page (data gzipped inline, Libertinus embedded) that shows the topics as a map, each
digest's groups as a tree, the contradiction notes and gaps, and a search over all claims.

Orientation only: nothing here is a pipeline input, and nothing is added to the ledgers.
Entries tagged only POLITICS or OTHER reach no digest (an open decision of the reading
pass); the page shows them grouped by theatre so they are not invisible.

usage:  python utils/reference_tools/background/build_reading_map.py
writes: output/reading_map/reading_map.html   (gitignored; rebuild after any digest rebuild)
"""
from __future__ import annotations

__version__ = "1.0"

import base64
import gzip
import html
import json
import re
import sys
from collections import Counter, defaultdict
from datetime import date
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
EVIDENCE = ROOT / "reference" / "background" / "_evidence"
FONTS = ROOT / "utils" / "docs" / "fonts"
TEMPLATE = Path(__file__).with_name("reading_map_template.html")
OUT = ROOT / "output" / "reading_map" / "reading_map.html"

# The reader brief's tag set (READER_BRIEF.md). MOVE and EVENT share one digest.
DIGESTED = ["ASSIGN", "DESERT", "RANK", "DATA", "MOVE_EVENT", "RECRUIT", "FORM", "CMD",
            "NATION", "EXIT", "MATERIAL", "HABS", "THREAT"]
UNDIGESTED = ["POLITICS", "OTHER"]
TOPICS = DIGESTED + UNDIGESTED
TAG_TO_TOPIC = {t: t for t in TOPICS} | {"MOVE": "MOVE_EVENT", "EVENT": "MOVE_EVENT"}
# READER_BRIEF.md: "ASSIGN, DESERT, RANK, DATA and MOVE matter most."
PRIORITY = {"ASSIGN", "DESERT", "RANK", "DATA", "MOVE_EVENT"}
TOPIC_LABEL = {
    "ASSIGN": "How a recruit reached a unit", "DESERT": "Desertion, refusal, discipline",
    "RANK": "Rank on entry, promotion, pay by rank", "DATA": "How personnel records were kept",
    "MOVE_EVENT": "Where units were; dated events", "RECRUIT": "Who enlisted, and how",
    "FORM": "Formation, reorganisation, strength", "CMD": "Commanders and committees",
    "NATION": "Relations with other nationalities", "EXIT": "Deaths, discharge, repatriation",
    "MATERIAL": "Pay, food, supply, promises", "HABS": "The Habsburg side",
    "THREAT": "Threats to inference (residue tag)", "POLITICS": "Factions, councils, press",
    "OTHER": "Important, fits no tag",
}
THEATRES = ["Russia", "Italy", "France", "Habsburg", "Serbia", "home", "general"]
WORKS = {  # slice-key prefix -> short name of the work (full title is in each ledger header)
    "zs1": "Za svobodu I", "zs2": "Za svobodu II", "zs3": "Za svobodu III",
    "zs4": "Za svobodu IV", "vboj": "V boj", "nazdar": "Rota Nazdar",
    "oeulk": "ÖULK (official history)", "fidler": "Fidler 2025",
}

_TAGS_RE = re.compile(r"^((?:\[[A-Z_]+\])+)\s*(.*)$", re.S)
_FIELD_RE = re.compile(r"^- (date|theatre|units|page|locator|confidence|note):\s?(.*)$")
_LINE_RE = re.compile(r"^- (.*) — \*(.*)\* · `([a-z0-9_]+) · ([^`]+)`\s*$")
_NUM_RE = re.compile(r"\d+")


def work_of(key: str) -> str:
    for prefix, name in WORKS.items():
        if key.startswith(prefix):
            return name
    return key


def norm(text: str) -> str:
    text = text.replace("[…]", " ").replace("[...]", " ")
    text = re.sub(r"[\"'“”„‘’`*]", "", text.lower())
    return re.sub(r"\s+", " ", text).strip(" .;,:")


def first_num(page: str) -> str:
    m = _NUM_RE.search(page or "")
    return str(int(m.group())) if m else ""


def theatre_main(raw: str) -> str:
    for word in re.findall(r"[A-Za-z]+", raw or ""):
        if word in THEATRES:
            return word
    return "general"


def year_of(date_field: str) -> int | None:
    m = re.search(r"\b(18[6-9]\d|19[0-4]\d)\b", date_field or "")
    return int(m.group()) if m else None


# ---------------------------------------------------------------- ledgers

def _part_sort_key(path: Path) -> tuple[str, int]:
    key, _, rest = path.name.partition(".part")
    return key, int(re.match(r"\d+", rest).group())


def _parse_block(block: str) -> dict | None:
    lines = block.split("\n")
    m = _TAGS_RE.match(lines[0].strip())
    if not m:
        return None
    tags = re.findall(r"\[([A-Z_]+)\]", m.group(1))
    entry = {"tags": tags, "claim": m.group(2).strip()}
    last = None
    for line in lines[1:]:
        if re.match(r"^#{1,2} |^<!--", line):  # Silences / remarks / next part's comment
            break
        f = _FIELD_RE.match(line)
        if f:
            last = f.group(1)
            entry[last] = f.group(2).strip()
        elif last and line.strip() and not line.startswith("- "):
            entry[last] += " " + line.strip()
    return entry


def parse_ledgers() -> tuple[list[dict], dict]:
    entries, headers = [], {}
    for path in sorted((EVIDENCE / "parts").glob("*.part*.md"), key=_part_sort_key):
        key = path.name.split(".part")[0]
        text = path.read_text(encoding="utf-8")
        head = headers.setdefault(key, {"key": key, "work": work_of(key), "parts": 0})
        head["parts"] += 1
        for field in ("source", "pages", "covers"):
            m = re.search(rf"^- {field}:\s*(.+)$", text, re.M)
            if m and field not in head:
                head[field] = m.group(1).strip()
        for block in re.split(r"^### ", text, flags=re.M)[1:]:
            e = _parse_block(block)
            if e:
                e["key"] = key
                entries.append(e)
    for i, e in enumerate(entries):
        e["id"] = i
        e["year"] = year_of(e.get("date", ""))
        e["th"] = theatre_main(e.get("theatre", ""))
        e["topics"] = sorted({TAG_TO_TOPIC[t] for t in e["tags"] if t in TAG_TO_TOPIC},
                             key=TOPICS.index)
        headers[e["key"]]["n"] = headers[e["key"]].get("n", 0) + 1
    return entries, headers


# ---------------------------------------------------------------- markdown (digest prose)

def md_inline(text: str) -> str:
    out = html.escape(text, quote=False)
    spans: list[str] = []  # code spans and refs are set aside so * inside them stays literal

    def keep(fragment: str) -> str:
        spans.append(fragment)
        return f"\x00{len(spans) - 1}\x00"

    out = re.sub(r"`([a-z0-9_]+) · ([^`]+)`",
                 lambda m: keep(f'<button class="ref" data-k="{m.group(1)}" '
                                f'data-p="{first_num(m.group(2))}">{m.group(1)} · {m.group(2)}</button>'),
                 out)
    out = re.sub(r"`([^`]+)`", lambda m: keep(f"<code>{m.group(1)}</code>"), out)
    out = re.sub(r"\*\*(.+?)\*\*", r"<strong>\1</strong>", out)
    out = re.sub(r"(?<![\w*])\*(?!\s)(.+?)(?<!\s)\*(?![\w*])", r"<em>\1</em>", out)
    return re.sub(r"\x00(\d+)\x00", lambda m: spans[int(m.group(1))], out)


def covers_text(raw: str) -> str:
    """Drop the digest's selection boilerplate ('every ledger entry tagged ... (heading match ...):')."""
    if raw.lower().startswith("every ledger entry") and "): " in raw:
        raw = raw.split("): ", 1)[1]
    return raw[:1].upper() + raw[1:]


def md_block(text: str) -> str:
    """Paragraphs, bullet lists and pipe tables: all the digests' prose sections use."""
    out, para, items, rows = [], [], [], []

    def flush():
        if para:
            out.append("<p>" + md_inline(" ".join(para)) + "</p>")
            para.clear()
        if items:
            out.append("<ul>" + "".join(f"<li>{md_inline(i)}</li>" for i in items) + "</ul>")
            items.clear()
        if rows:
            cells = [[c.strip() for c in r.strip().strip("|").split(" | ")] for r in rows
                     if not re.match(r"^\|[\s|:-]+\|$", r.strip())]
            head, body = cells[0], cells[1:]
            out.append('<div class="tblwrap"><table><thead><tr>'
                       + "".join(f"<th>{md_inline(c)}</th>" for c in head) + "</tr></thead><tbody>"
                       + "".join("<tr>" + "".join(f"<td>{md_inline(c)}</td>" for c in r) + "</tr>"
                                 for r in body) + "</tbody></table></div>")
            rows.clear()

    for line in text.split("\n"):
        s = line.strip()
        if not s:
            flush()
        elif s.startswith("|"):
            if para or items:
                flush()
            rows.append(s)
        elif s.startswith("- "):
            if para or rows:
                flush()
            items.append(s[2:])
        elif s.startswith("> "):
            flush()
            out.append('<blockquote>' + md_inline(s[2:]) + "</blockquote>")
        elif re.match(r"^#{2,4} ", s):
            flush()
            out.append("<h4>" + md_inline(s.lstrip("# ")) + "</h4>")
        else:
            if items:
                items[-1] += " " + s
            else:
                if rows:
                    flush()
                para.append(s)
    flush()
    return "".join(out)


# ---------------------------------------------------------------- digests

def _header_bullets(head: str) -> dict:
    bullets = {}
    for m in re.finditer(r"^- \*\*(.+?):\*\*\s*(.*?)(?=^- \*\*|\Z)", head, re.M | re.S):
        bullets[m.group(1).strip()] = m.group(2).strip()
    return bullets


def parse_digest(tag: str) -> dict:
    text = (EVIDENCE / "digests" / f"{tag}.md").read_text(encoding="utf-8")
    title = text.split("\n", 1)[0].lstrip("# ").strip()
    sections = re.split(r"^## ", text, flags=re.M)
    bullets = _header_bullets(sections[0])
    digest = {"tag": tag, "title": title, "covers": md_inline(covers_text(bullets.get("Covers", ""))),
              "reading": md_inline(bullets.get("Reading note", "")),
              "built": md_inline(bullets.get("Built", "")), "groups": [], "extra": [],
              "gaps": ""}
    for sec in sections[1:]:
        heading, _, body = sec.partition("\n")
        heading = heading.strip()
        if heading.lower().startswith("gaps"):
            digest["gaps"] = md_block(body)
        elif re.match(r"^\d+\.", heading):
            digest["groups"].append(_parse_group(heading, body))
        elif heading.lower() != "table":
            digest["extra"].append({"title": md_inline(heading), "html": md_block(body)})
    return digest


def _parse_group(heading: str, body: str) -> dict:
    num, _, title = heading.partition(". ")
    group = {"num": num.strip(), "title": title.strip(), "lead": "", "notes": [], "subs": []}
    sub = {"title": "", "lines": []}
    group["subs"].append(sub)
    for line in body.split("\n"):
        s = line.strip()
        if not group["lead"] and s.startswith("*") and s.endswith("*") and not s.startswith("**"):
            group["lead"] = md_inline(s.strip("*"))
        elif s.startswith("### "):
            sub = {"title": s[4:].strip(), "lines": []}
            group["subs"].append(sub)
        elif s.startswith("> "):
            group["notes"].append(md_inline(re.sub(r"^\*\*Note\.\*\*\s*", "", s[2:])))
        else:
            m = _LINE_RE.match(s)
            if m:  # FORM 16-18 prefix the claim with its theatre: "[Russia] ..."
                claim = re.sub(r"^\[[A-Za-z /;→()]+\]\s*", "", m.group(1))
                sub["lines"].append({"claim": claim, "key": m.group(3),
                                     "page": first_num(m.group(4))})
    group["subs"] = [s for s in group["subs"] if s["lines"] or s["title"]]
    return group


def match_lines(digest: dict, by_key: dict, tag: str) -> Counter:
    """Point every digest line at its ledger entry: same key, claim prefix, page breaks ties."""
    stats = Counter()
    for gi, group in enumerate(digest["groups"]):
        for sub in group["subs"]:
            ids = []
            for line in sub["lines"]:
                target = norm(line["claim"])[:70]
                cands = [e for e in by_key.get(line["key"], [])
                         if tag in e["topics"] and norm(e["claim"]).startswith(target)]
                if len(cands) > 1:
                    cands = [e for e in cands if first_num(e.get("page", "")) == line["page"]] or cands
                if cands:
                    ids.append(cands[0]["id"])
                    stats["matched"] += 1
                else:
                    stats["unmatched"] += 1
            sub["ids"] = ids
            del sub["lines"]
    return stats


def theatre_groups(entries: list[dict], topic: str, note: str) -> list[dict]:
    """Groups by theatre for the topics with no grouped digest (MOVE+EVENT, POLITICS, OTHER)."""
    by_th = defaultdict(list)
    for e in entries:
        if topic in e["topics"]:
            by_th[e["th"]].append(e)
    groups = []
    for i, th in enumerate(t for t in THEATRES if by_th[t]):
        rows = sorted(by_th[th], key=lambda e: (e["year"] or 9999, e.get("date", "")))
        groups.append({"num": str(i + 1), "title": th, "lead": note, "notes": [],
                       "subs": [{"title": "", "ids": [e["id"] for e in rows]}]})
    return groups


# ---------------------------------------------------------------- assemble

def build_topics(entries: list[dict]) -> tuple[list[dict], dict]:
    by_key = defaultdict(list)
    for e in entries:
        by_key[e["key"]].append(e)
    topics, stats = [], {}
    for tag in TOPICS:
        if tag in DIGESTED:
            d = parse_digest(tag)
            if tag == "MOVE_EVENT":
                d["groups"] = theatre_groups(
                    entries, tag, "The digest is one dated table; here split by the ledger's theatre, in date order.")
                stats[tag] = {"matched": sum(tag in e["topics"] for e in entries), "unmatched": 0}
            else:
                stats[tag] = dict(match_lines(d, by_key, tag))
        else:
            d = {"tag": tag, "title": f"{tag} (no digest)", "covers": "", "reading": "",
                 "built": "", "extra": [], "gaps": "",
                 "groups": theatre_groups(entries, tag, "No digest exists for this tag; grouped by the ledger's theatre.")}
            stats[tag] = {"matched": sum(tag in e["topics"] for e in entries), "unmatched": 0}
        placed = {i for g in d["groups"] for s in g["subs"] for i in s["ids"]}
        missing = [e["id"] for e in entries if tag in e["topics"] and e["id"] not in placed]
        if missing:  # tagged but no digest line matched: keep them visible
            d["groups"].append({"num": "–", "title": "Not matched to a digest line", "lead":
                                "Entries with this tag whose digest line the build could not find.",
                                "notes": [], "subs": [{"title": "", "ids": missing}]})
        d["label"] = TOPIC_LABEL[tag]
        d["priority"] = tag in PRIORITY
        d["digested"] = tag in DIGESTED
        d["n"] = sum(tag in e["topics"] for e in entries)
        topics.append(d)
    return topics, stats


def cooccurrence(entries: list[dict]) -> list[list[int]]:
    idx = {t: i for i, t in enumerate(TOPICS)}
    mat = [[0] * len(TOPICS) for _ in TOPICS]
    for e in entries:
        for a in e["topics"]:
            for b in e["topics"]:
                mat[idx[a]][idx[b]] += 1
    return mat


def conf_level(raw: str) -> str:
    word = (raw or "").split(" ", 1)[0].strip(" ,;—-").lower()
    return word if word in {"high", "medium", "low"} else ""


def payload(entries, headers, topics) -> dict:
    keys = sorted(headers, key=lambda k: (list(WORKS).index(next(p for p in WORKS if k.startswith(p))), k))
    kidx = {k: i for i, k in enumerate(keys)}
    rows = []
    for e in entries:
        rows.append([kidx[e["key"]], " ".join(e["tags"]), e["claim"], e.get("date", ""),
                     e["year"] or 0, e["th"], e.get("theatre", ""), (e.get("units", "") or "")[:140],
                     e.get("page", ""), conf_level(e.get("confidence", "")),
                     e.get("confidence", "") if conf_level(e.get("confidence", "")) != "high" else "",
                     e.get("note", "")])
    return {"built": date.today().isoformat(), "version": __version__,
            "topics": topics, "topicOrder": TOPICS, "theatres": THEATRES,
            "ledgers": [headers[k] for k in keys], "works": list(WORKS.values()),
            "cooc": cooccurrence(entries), "E": rows}


def fontfaces() -> str:
    faces = [("Libertinus Sans", "normal", 400, "LibertinusSans-Regular"),
             ("Libertinus Sans", "italic", 400, "LibertinusSans-Italic"),
             ("Libertinus Sans", "normal", 700, "LibertinusSans-Bold"),
             ("Libertinus Serif", "normal", 700, "LibertinusSerif-Bold")]
    css = []
    for family, style, weight, name in faces:
        b64 = base64.b64encode((FONTS / f"{name}.woff").read_bytes()).decode()
        css.append(f'@font-face{{font-family:"{family}";font-style:{style};font-weight:{weight};'
                   f'font-display:swap;src:url(data:font/woff;base64,{b64}) format("woff")}}')
    return "\n".join(css)


def main() -> None:
    entries, headers = parse_ledgers()
    topics, stats = build_topics(entries)
    data = json.dumps(payload(entries, headers, topics), ensure_ascii=False, separators=(",", ":"))
    blob = base64.b64encode(gzip.compress(data.encode("utf-8"), 9)).decode()
    page = (TEMPLATE.read_text(encoding="utf-8")
            .replace("/*{{FONTFACES}}*/", fontfaces())
            .replace("{{DATA}}", blob)
            .replace("{{N_ENTRIES}}", f"{len(entries):,}"))
    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text(page, encoding="utf-8")

    print(f"ledgers {len(headers)}, entries {len(entries):,}")
    for tag in TOPICS:
        s = stats[tag]
        print(f"  {tag:<11} {s.get('matched', 0):>5} matched  {s.get('unmatched', 0):>4} unmatched")
    undig = sum(1 for e in entries if not set(e["topics"]) & set(DIGESTED))
    bad = Counter(t for e in entries for t in e["tags"] if t not in TAG_TO_TOPIC)
    print(f"entries in no digest: {undig:,};  tags outside the brief: {dict(bad)}")
    print(f"json {len(data) / 1e6:.1f} MB -> gzip+b64 {len(blob) / 1e6:.1f} MB; page {OUT.stat().st_size / 1e6:.1f} MB")
    print(f"wrote {OUT.relative_to(ROOT)}")


if __name__ == "__main__":
    sys.exit(main())
