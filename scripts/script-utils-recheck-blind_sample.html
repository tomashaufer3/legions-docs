How it works

German Böhmen's rechecked claims were judged against the Czech volume (§8a). Mähren and Schlesien have no second volume, and they are where the rechecked register differs from the frozen one, so their changes are judged by eye. The sample, per stratum:

s4_agreed     §4's adds every reader made                               60 drawn
s4_disputed   §4's adds Opus ruled printed                              all
s4_record     §4's adds only the reader of record made                  60 drawn
s3_moved      §3's parts moved to another court                         all (one question a municipality)
cp_unchanged  castle and poorhouse claims no step changed (control)     60 drawn
no_pictogram  municipalities of 500+ people without a pictogram (control)  50 drawn

Left out: certification, reserve and calibration pages (rf.sealed_pages | rf.in_sample_pages); every entry the author has already ruled on; and §3's dropped place types — a village or manor-farm sign is not an institution, so taking it out of the institutions table is right whatever the print shows (they matter again when place types reach data/final).

Blind: a card shows the entry and asks, of each symbol it names, is it printed for this entry? The key — which rows were drawn, from which stratum, and what the rechecked register holds — is written apart (key.csv), never into the page. Every card on a §4 entry also asks one common symbol the rechecked register does not hold there, and a castle/poorhouse card asks both, so a row's being asked says nothing about its answer.

Score: per stratum, the share of drawn rows the author bears out — printed where the register holds the symbol, not printed where it does not — with a Wilson 95 % lower bound. A step is a net gain if the bound clears 0.5 and stands if it clears 0.8. Printed for another part bears a claim out at the municipality but not at the part; both levels are reported. Cannot tell is left out.

python -m utils.recheck.blind_sample draw     # writes output/recheck/blind_sample/key.csv
python -m utils.recheck.blind_sample score    # reads the staged rulings, writes score.json/.txt

def holdings(recheck: Path, edition: str) -> tuple[dict, dict, dict]:
    """``({part: codes}, {(page, district, muni): codes}, {court_id: district keys})`` of the rechecked tables."""
    by_part, by_muni, keys = defaultdict(set), defaultdict(set), defaultdict(set)
    for name in ("final_inventory_rechecked.csv", "final_abbreviations_rechecked.csv"):
        t = _table(recheck, edition, name)
        for r in t.itertuples():
            by_part[(r.page, r.district, r.muni_seq, r.settlement_seq)].add(r.code)
            by_muni[(r.page, r.district, r.muni_seq)].add(r.code)
            if r.court_id:
                keys[r.court_id].add(r.district)
    return by_part, by_muni, keys

def ruled_claims(edition: str, changes: pd.DataFrame) -> set[tuple]:
    """``(page, district, muni, code)`` of every claim the author has ruled on: a change of the author
    step; a hand-check mark, both the route's symbol there and the ruled one (a ruling that kept the
    route makes no change); the castle and poorhouse of an Unboxed or Last Castle card."""
    out = {(r.page, r.district, r.muni_seq, r.code) for r in changes.itertuples() if r.step == "author"}
    marks = _table(ROOT / "output" / "frozen", f"{edition}_full", f"ledger/marks_{edition}_full.csv")

        if r["edition"] == edition and r["mark"] in where:

    for r in mg.unboxed_rulings(edition) + mg.last_rulings(edition):
        out |= {r["part"][:3] + (code,) for code in PAIR}
    return out

def pools(recheck: Path = mg.OUT) -> dict[str, list[dict]]:
    """Every claim each stratum can draw, sorted, across the two volumes."""
    out = defaultdict(list)
    places = pd.read_parquet(ROOT / "data" / "final" / "register" / "register_places.parquet")
    for ed in EDITIONS:
        held = held_out(ed)
        changes = _table(recheck, ed, "changes.csv")
        ruled = ruled_claims(ed, changes)
        inv = _table(recheck, ed, "final_inventory_rechecked.csv")
        court_of = {(r.page, r.district, r.muni_seq): r.court_id for r in inv.itertuples()}
        by_part, _, keys = holdings(recheck, ed)
        free = lambda page, district, muni, code="": (page not in held                              # noqa: E731
                                                      and (page, district, muni, code) not in ruled)
        for r in changes.itertuples():
            if r.step == "§4" and r.action == "add" and free(r.page, r.district, r.muni_seq, r.code):
                stratum = next((s for p, s in S4_REASONS.items() if r.reason.startswith(p)), None)
                if stratum:
                    out[stratum].append(_claim(ed, r.page, r.district, court_of.get((r.page, r.district, r.muni_seq), ""),
                                               r.muni_seq, r.settlement_seq, r.code, r.table, stratum))
        moved = changes[(changes.step == "§3") & changes.reason.str.startswith("moved from")]
        for (page, district, muni), g in moved.groupby(["page", "district", "muni_seq"]):
            old = changes[(changes.step == "§3") & (changes.page == page) & (changes.muni_seq == muni)
                          & changes.reason.str.startswith("moved:")].district
            if free(page, district, muni) and len(old):
                out["s3_moved"].append(_claim(ed, page, old.iloc[0], court_of.get((page, district, muni), ""), muni,
                                              "0", "court", "court", "s3_moved",
                                              old_district=old.iloc[0], new_district=district))
        touched = set(zip(changes.page, changes.district, changes.muni_seq, changes.settlement_seq, changes.code))
        for r in inv[inv.code.isin(PAIR)].itertuples():
            if free(r.page, r.district, r.muni_seq, r.code) and (r.page, r.district, r.muni_seq, r.settlement_seq, r.code) not in touched:
                out["cp_unchanged"].append(_claim(ed, r.page, r.district, r.court_id, r.muni_seq, r.settlement_seq,
                                                  r.code, "inventory", "cp_unchanged"))
        with_pictogram = {(r.court_id, r.muni_seq) for r in inv.itertuples() if r.code.startswith("[")}
        towns = places[(places.edition == ed) & (places.place_level == "municipality")
                       & (places.pop_total.fillna(0) >= MIN_POP) & places.muni_seq.notna()]
        for r in towns.itertuples():
            court, muni, page = str(r.court_id), str(int(r.muni_seq)), re.split(r"[;,\s]+", str(r.pages).strip())[0]
            district = next(iter(sorted(keys.get(court, ()))), "")
            if (court, muni) in with_pictogram or not district or not free(page, district, muni):
                continue
            out["no_pictogram"].append(_claim(ed, page, district, court, muni, "0", ANY, "any", "no_pictogram"))
    return {s: sorted(v, key=lambda c: tuple(c[k] for k in ("edition", "page", "district", "muni", "seq", "code")))
            for s, v in out.items()}

def draw(recheck: Path = mg.OUT, seed: int = SEED, sample: dict = SAMPLE) -> pd.DataFrame:
    """The key: one row per question asked, the drawn claims with their stratum, the decoys beside them."""
    rng = random.Random(seed)
    every = pools(recheck)
    drawn = []
    for stratum, n in sample.items():
        pool = every.get(stratum, [])
        drawn += pool if n is None or len(pool) <= n else rng.sample(pool, n)
    holding = {ed: holdings(recheck, ed) for ed in EDITIONS}
    cards = defaultdict(dict)
    for c in drawn:
        card = "-".join((mg.SHORT[c["edition"]], c["page"], c["district"], c["muni"],
                         "court" if c["table"] == "court" else c["seq"]))
        cards[card][c["code"]] = c
    rows = []
    for card, asked in sorted(cards.items()):
        first = next(iter(asked.values()))
        by_part, by_muni, _ = holding[first["edition"]]
        part = (first["page"], first["district"], first["muni"], first["seq"])
        muni_codes = by_muni.get(part[:3], set())
        extra = {}
        if any(c["stratum"] == "cp_unchanged" for c in asked.values()):
            extra = {code: "context" for code in PAIR if code not in asked}
        elif any(c["stratum"].startswith("s4") for c in asked.values()):
            choices = [d for d in DECOYS if d not in by_part.get(part, set()) and d not in asked]
            extra = {rng.choice(choices): "decoy"} if choices else {}
        for code, why in extra.items():
            asked[code] = _claim(first["edition"], *part[:2], first["court_id"], *part[2:], code, "inventory", why)
        for code, c in asked.items():
            held = code in by_part.get(part, set()) if c["table"] != "court" else True
            if c["stratum"] == "context":
                c["stratum"] = "cp_unchanged_other" if held else "decoy"
            rows.append({**{k: "" for k in KEY_FIELDS}, **c, "card": card, "held": held,
                         "held_muni": (code in muni_codes) if c["table"] not in ("court", "any") else held})
    return pd.DataFrame(rows, columns=list(KEY_FIELDS))

def wilson_lower(k: int, n: int, z: float = 1.96) -> float:
    """The Wilson score interval's lower bound for k of n."""
    if n == 0:
        return float("nan")
    p = k / n
    lower = (p + z * z / (2 * n) - z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n))) / (1 + z * z / n)
    return max(0.0, round(lower, 12))

def borne_out(row: dict, answer: str, level: str = "part") -> bool | None:
    """Whether the author's answer bears the rechecked register out on one row; ``None``: no answer.
    Answers: printed, other_part (printed for another part of the municipality), absent, unsure;
    a court question: the district key of the court the author names, or unsure."""
    if answer in ("", "unsure", None):
        return None
    if row["table"] == "court":
        return answer == row["new_district"]
    held = str(row["held_muni" if level == "municipality" else "held"]) == "True"
    printed = answer == "printed" or (answer == "other_part" and level == "municipality")
    return printed == held

def as_rulings(edition: str, key: Path | None = None, folder: Path = RULINGS) -> list[dict]:
    """The author's answers on the page as rulings on one symbol of one part, for the merge's author
    step (``merge.apply_chat``): *printed* and *not printed* only — *another part*, *can't tell*, the
    court question and the any-pictogram question set no symbol. 0.2.0."""
    path = Path(key) if key else OUT / "key.csv"
    if not path.exists():
        return []
    answers = rulings(folder)
    out = []
    for r in pd.read_csv(path, dtype=str, keep_default_na=False).itertuples():
        answer = answers.get((r.card, r.code), "")
        if r.edition != edition or r.table in ("court", "any") or answer not in ("printed", "absent"):
            continue
        out.append({"part": (r.page, r.district, r.muni, r.seq), "code": r.code,
                    "verdict": "printed" if answer == "printed" else "not printed",
                    "ruled": "blind sample, 2026-11-09", "note": ""})
    return out

def score(key: pd.DataFrame, answers: dict) -> pd.DataFrame:
    """Per stratum (and the decoys): rows drawn, ruled, borne out at the part and the municipality."""
    out = []
    for stratum in (*STRATA, "decoy"):
        rows = key[key.stratum == stratum].to_dict("records")
        line = {"stratum": stratum, "drawn": len(rows)}
        for level in ("part", "municipality"):
            judged = [borne_out(r, answers.get((r["card"], r["code"]), ""), level) for r in rows]
            judged = [j for j in judged if j is not None]
            k, n = sum(judged), len(judged)
            lower = wilson_lower(k, n)
            line |= {f"ruled_{level}": n, f"borne_{level}": k, f"share_{level}": round(k / n, 3) if n else None,
                     f"lower_{level}": None if math.isnan(lower) else round(lower, 3), f"verdict_{level}": verdict(lower)}
        out.append(line)
    return pd.DataFrame(out)

def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n")[0])
    ap.add_argument("what", choices=("draw", "score"))
    ap.add_argument("--recheck", default=str(mg.OUT), help="the folder notebook 17 §7 wrote (default output/recheck)")
    args = ap.parse_args(argv)
    OUT.mkdir(parents=True, exist_ok=True)
    if args.what == "draw":
        key = draw(Path(args.recheck))
        key.to_csv(OUT / "key.csv", index=False)
        print(f"blind_sample {__version__}: {key.card.nunique()} cards, {len(key)} questions")
        print(key.groupby("stratum").size().to_string())
        return 0
    key = pd.read_csv(OUT / "key.csv", dtype=str, keep_default_na=False)
    table = score(key, rulings())
    (OUT / "score.json").write_text(json.dumps(table.to_dict("records"), indent=1), encoding="utf-8")
    (OUT / "score.txt").write_text(table.to_string(index=False), encoding="utf-8")
    print(table.to_string(index=False))
    return 0

"""A blind sample of what the correction changed in Mähren and Schlesien, ruled by the author (D-199).

German Böhmen's rechecked claims were judged against the Czech volume (§8a). Mähren and Schlesien
have no second volume, and they are where the rechecked register differs from the frozen one, so
their changes are judged by eye. The sample, per stratum:

    s4_agreed     §4's adds every reader made                               60 drawn
    s4_disputed   §4's adds Opus ruled printed                              all
    s4_record     §4's adds only the reader of record made                  60 drawn
    s3_moved      §3's parts moved to another court                         all (one question a municipality)
    cp_unchanged  castle and poorhouse claims no step changed (control)     60 drawn
    no_pictogram  municipalities of 500+ people without a pictogram (control)  50 drawn

Left out: certification, reserve and calibration pages (``rf.sealed_pages | rf.in_sample_pages``);
every entry the author has already ruled on; and §3's dropped place types — a village or
manor-farm sign is not an institution, so taking it out of the institutions table is right whatever
the print shows (they matter again when place types reach ``data/final``).

Blind: a card shows the entry and asks, of each symbol it names, *is it printed for this entry?*
The key — which rows were drawn, from which stratum, and what the rechecked register holds — is
written apart (``key.csv``), never into the page. Every card on a §4 entry also asks one common
symbol the rechecked register does not hold there, and a castle/poorhouse card asks both, so a
row's being asked says nothing about its answer.

Score: per stratum, the share of drawn rows the author bears out — *printed* where the register
holds the symbol, *not printed* where it does not — with a Wilson 95 % lower bound. A step is a net
gain if the bound clears 0.5 and stands if it clears 0.8. *Printed for another part* bears a claim
out at the municipality but not at the part; both levels are reported. *Cannot tell* is left out.

    python -m utils.recheck.blind_sample draw     # writes output/recheck/blind_sample/key.csv
    python -m utils.recheck.blind_sample score    # reads the staged rulings, writes score.json/.txt
"""

import run_frozen as rf                                                # noqa: E402
from utils.recheck import merge as mg                                 # noqa: E402
from utils.recheck import opus_pass as op                             # noqa: E402

EDITIONS = ("de-moravia", "de-silesia")
SAMPLE = {"s4_agreed": 60, "s4_disputed": None, "s4_record": 60, "s3_moved": None,
          "cp_unchanged": 60, "no_pictogram": 50}                    # None: every claim in the pool
STRATA = tuple(SAMPLE)
S4_REASONS = {"offices read again: agreed": "s4_agreed",              # merge.py's §4 reasons, by prefix
              "offices read again: disputed": "s4_disputed",
              "offices read again: reader of record": "s4_record"}
SEED = 1910
MIN_POP = 500
PAIR = ("[castle]", "[poorhouse]")
DECOYS = ("[school]", "[fire-brigade]", "[midwife]", "[cooperative]", "[rc-parish]", "[post]",
          "[doctor]", "[telegraph]", "[gendarme]", "[rail-station]", "[pharmacy]", "[kindergarten]")
ANY = "any pictogram"
NET_GAIN, STANDS = 0.5, 0.8                                           # bars on the Wilson lower bound
OUT = mg.OUT / "blind_sample"
RULINGS = ROOT / "output" / "review_readback" / "blind_ms"          # one JSON per ruled card, as the page saved it
KEY_FIELDS = ("card", "edition", "page", "district", "court_id", "muni", "seq", "code", "table", "stratum",
              "held", "held_muni", "old_district", "new_district")


def _table(folder: Path, edition: str, name: str) -> pd.DataFrame:
    path = Path(folder) / edition / name
    return pd.read_csv(path, dtype=str, keep_default_na=False) if path.exists() else pd.DataFrame()


def held_out(edition: str) -> set[str]:
    """Certification, reserve and calibration pages: never in a sample the author looks at (D-133, D-173)."""
    return rf.sealed_pages(edition) | rf.in_sample_pages(edition)


def holdings(recheck: Path, edition: str) -> tuple[dict, dict, dict]:
    """``({part: codes}, {(page, district, muni): codes}, {court_id: district keys})`` of the rechecked tables."""
    by_part, by_muni, keys = defaultdict(set), defaultdict(set), defaultdict(set)
    for name in ("final_inventory_rechecked.csv", "final_abbreviations_rechecked.csv"):
        t = _table(recheck, edition, name)
        for r in t.itertuples():
            by_part[(r.page, r.district, r.muni_seq, r.settlement_seq)].add(r.code)
            by_muni[(r.page, r.district, r.muni_seq)].add(r.code)
            if r.court_id:
                keys[r.court_id].add(r.district)
    return by_part, by_muni, keys


def ruled_claims(edition: str, changes: pd.DataFrame) -> set[tuple]:
    """``(page, district, muni, code)`` of every claim the author has ruled on: a change of the author
    step; a hand-check mark, both the route's symbol there and the ruled one (a ruling that kept the
    route makes no change); the castle and poorhouse of an Unboxed or Last Castle card."""
    out = {(r.page, r.district, r.muni_seq, r.code) for r in changes.itertuples() if r.step == "author"}
    marks = _table(ROOT / "output" / "frozen", f"{edition}_full", f"ledger/marks_{edition}_full.csv")
    where = {m.mark: ((m.page, m.district, str(m.muni)), m.final) for m in marks.itertuples()} if len(marks) else {}
    for r in op.hand_rulings():
        if r["edition"] == edition and r["mark"] in where:
            entry, final = where[r["mark"]]
            out |= {entry + (final,), entry + (r["author"],)}
    for r in mg.unboxed_rulings(edition) + mg.last_rulings(edition):
        out |= {r["part"][:3] + (code,) for code in PAIR}
    return out


def _claim(edition, page, district, court_id, muni, seq, code, table, stratum, **kw) -> dict:
    return {"edition": edition, "page": page, "district": district, "court_id": court_id, "muni": str(muni),
            "seq": str(seq), "code": code, "table": table, "stratum": stratum, **kw}


def pools(recheck: Path = mg.OUT) -> dict[str, list[dict]]:
    """Every claim each stratum can draw, sorted, across the two volumes."""
    out = defaultdict(list)
    places = pd.read_parquet(ROOT / "data" / "final" / "register" / "register_places.parquet")
    for ed in EDITIONS:
        held = held_out(ed)
        changes = _table(recheck, ed, "changes.csv")
        ruled = ruled_claims(ed, changes)
        inv = _table(recheck, ed, "final_inventory_rechecked.csv")
        court_of = {(r.page, r.district, r.muni_seq): r.court_id for r in inv.itertuples()}
        by_part, _, keys = holdings(recheck, ed)
        free = lambda page, district, muni, code="": (page not in held                              # noqa: E731
                                                      and (page, district, muni, code) not in ruled)
        for r in changes.itertuples():
            if r.step == "§4" and r.action == "add" and free(r.page, r.district, r.muni_seq, r.code):
                stratum = next((s for p, s in S4_REASONS.items() if r.reason.startswith(p)), None)
                if stratum:
                    out[stratum].append(_claim(ed, r.page, r.district, court_of.get((r.page, r.district, r.muni_seq), ""),
                                               r.muni_seq, r.settlement_seq, r.code, r.table, stratum))
        moved = changes[(changes.step == "§3") & changes.reason.str.startswith("moved from")]
        for (page, district, muni), g in moved.groupby(["page", "district", "muni_seq"]):
            old = changes[(changes.step == "§3") & (changes.page == page) & (changes.muni_seq == muni)
                          & changes.reason.str.startswith("moved:")].district
            if free(page, district, muni) and len(old):
                out["s3_moved"].append(_claim(ed, page, old.iloc[0], court_of.get((page, district, muni), ""), muni,
                                              "0", "court", "court", "s3_moved",
                                              old_district=old.iloc[0], new_district=district))
        touched = set(zip(changes.page, changes.district, changes.muni_seq, changes.settlement_seq, changes.code))
        for r in inv[inv.code.isin(PAIR)].itertuples():
            if free(r.page, r.district, r.muni_seq, r.code) and (r.page, r.district, r.muni_seq, r.settlement_seq, r.code) not in touched:
                out["cp_unchanged"].append(_claim(ed, r.page, r.district, r.court_id, r.muni_seq, r.settlement_seq,
                                                  r.code, "inventory", "cp_unchanged"))
        with_pictogram = {(r.court_id, r.muni_seq) for r in inv.itertuples() if r.code.startswith("[")}
        towns = places[(places.edition == ed) & (places.place_level == "municipality")
                       & (places.pop_total.fillna(0) >= MIN_POP) & places.muni_seq.notna()]
        for r in towns.itertuples():
            court, muni, page = str(r.court_id), str(int(r.muni_seq)), re.split(r"[;,\s]+", str(r.pages).strip())[0]
            district = next(iter(sorted(keys.get(court, ()))), "")
            if (court, muni) in with_pictogram or not district or not free(page, district, muni):
                continue
            out["no_pictogram"].append(_claim(ed, page, district, court, muni, "0", ANY, "any", "no_pictogram"))
    return {s: sorted(v, key=lambda c: tuple(c[k] for k in ("edition", "page", "district", "muni", "seq", "code")))
            for s, v in out.items()}


def draw(recheck: Path = mg.OUT, seed: int = SEED, sample: dict = SAMPLE) -> pd.DataFrame:
    """The key: one row per question asked, the drawn claims with their stratum, the decoys beside them."""
    rng = random.Random(seed)
    every = pools(recheck)
    drawn = []
    for stratum, n in sample.items():
        pool = every.get(stratum, [])
        drawn += pool if n is None or len(pool) <= n else rng.sample(pool, n)
    holding = {ed: holdings(recheck, ed) for ed in EDITIONS}
    cards = defaultdict(dict)
    for c in drawn:
        card = "-".join((mg.SHORT[c["edition"]], c["page"], c["district"], c["muni"],
                         "court" if c["table"] == "court" else c["seq"]))
        cards[card][c["code"]] = c
    rows = []
    for card, asked in sorted(cards.items()):
        first = next(iter(asked.values()))
        by_part, by_muni, _ = holding[first["edition"]]
        part = (first["page"], first["district"], first["muni"], first["seq"])
        muni_codes = by_muni.get(part[:3], set())
        extra = {}
        if any(c["stratum"] == "cp_unchanged" for c in asked.values()):
            extra = {code: "context" for code in PAIR if code not in asked}
        elif any(c["stratum"].startswith("s4") for c in asked.values()):
            choices = [d for d in DECOYS if d not in by_part.get(part, set()) and d not in asked]
            extra = {rng.choice(choices): "decoy"} if choices else {}
        for code, why in extra.items():
            asked[code] = _claim(first["edition"], *part[:2], first["court_id"], *part[2:], code, "inventory", why)
        for code, c in asked.items():
            held = code in by_part.get(part, set()) if c["table"] != "court" else True
            if c["stratum"] == "context":
                c["stratum"] = "cp_unchanged_other" if held else "decoy"
            rows.append({**{k: "" for k in KEY_FIELDS}, **c, "card": card, "held": held,
                         "held_muni": (code in muni_codes) if c["table"] not in ("court", "any") else held})
    return pd.DataFrame(rows, columns=list(KEY_FIELDS))


def wilson_lower(k: int, n: int, z: float = 1.96) -> float:
    """The Wilson score interval's lower bound for k of n."""
    if n == 0:
        return float("nan")
    p = k / n
    lower = (p + z * z / (2 * n) - z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n))) / (1 + z * z / n)
    return max(0.0, round(lower, 12))                                # 0 of n is 0, not -1e-17


def borne_out(row: dict, answer: str, level: str = "part") -> bool | None:
    """Whether the author's answer bears the rechecked register out on one row; ``None``: no answer.
    Answers: printed, other_part (printed for another part of the municipality), absent, unsure;
    a court question: the district key of the court the author names, or unsure."""
    if answer in ("", "unsure", None):
        return None
    if row["table"] == "court":
        return answer == row["new_district"]
    held = str(row["held_muni" if level == "municipality" else "held"]) == "True"
    printed = answer == "printed" or (answer == "other_part" and level == "municipality")
    return printed == held


def rulings(folder: Path = RULINGS) -> dict[tuple, str]:
    """``{(card, code): answer}`` from the page's staged JSON, one file a card; ``rows`` is the page's
    list of ``{c: code, v: answer}``."""
    out = {}
    for path in sorted(Path(folder).glob("*.json")):
        d = json.loads(path.read_text("utf-8"))
        for row in d.get("rows") or []:
            out[(d["id"], row["c"])] = row["v"]
    return out


def as_rulings(edition: str, key: Path | None = None, folder: Path = RULINGS) -> list[dict]:
    """The author's answers on the page as rulings on one symbol of one part, for the merge's author
    step (``merge.apply_chat``): *printed* and *not printed* only — *another part*, *can't tell*, the
    court question and the any-pictogram question set no symbol. 0.2.0."""
    path = Path(key) if key else OUT / "key.csv"
    if not path.exists():
        return []
    answers = rulings(folder)
    out = []
    for r in pd.read_csv(path, dtype=str, keep_default_na=False).itertuples():
        answer = answers.get((r.card, r.code), "")
        if r.edition != edition or r.table in ("court", "any") or answer not in ("printed", "absent"):
            continue
        out.append({"part": (r.page, r.district, r.muni, r.seq), "code": r.code,
                    "verdict": "printed" if answer == "printed" else "not printed",
                    "ruled": "blind sample, 2026-11-09", "note": ""})
    return out


def verdict(lower: float) -> str:
    if math.isnan(lower):
        return "no rows"
    return "stands" if lower > STANDS else "net gain" if lower > NET_GAIN else "not shown"


def score(key: pd.DataFrame, answers: dict) -> pd.DataFrame:
    """Per stratum (and the decoys): rows drawn, ruled, borne out at the part and the municipality."""
    out = []
    for stratum in (*STRATA, "decoy"):
        rows = key[key.stratum == stratum].to_dict("records")
        line = {"stratum": stratum, "drawn": len(rows)}
        for level in ("part", "municipality"):
            judged = [borne_out(r, answers.get((r["card"], r["code"]), ""), level) for r in rows]
            judged = [j for j in judged if j is not None]
            k, n = sum(judged), len(judged)
            lower = wilson_lower(k, n)
            line |= {f"ruled_{level}": n, f"borne_{level}": k, f"share_{level}": round(k / n, 3) if n else None,
                     f"lower_{level}": None if math.isnan(lower) else round(lower, 3), f"verdict_{level}": verdict(lower)}
        out.append(line)
    return pd.DataFrame(out)


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.split("\n")[0])
    ap.add_argument("what", choices=("draw", "score"))
    ap.add_argument("--recheck", default=str(mg.OUT), help="the folder notebook 17 §7 wrote (default output/recheck)")
    args = ap.parse_args(argv)
    OUT.mkdir(parents=True, exist_ok=True)
    if args.what == "draw":
        key = draw(Path(args.recheck))
        key.to_csv(OUT / "key.csv", index=False)
        print(f"blind_sample {__version__}: {key.card.nunique()} cards, {len(key)} questions")
        print(key.groupby("stratum").size().to_string())
        return 0
    key = pd.read_csv(OUT / "key.csv", dtype=str, keep_default_na=False)
    table = score(key, rulings())
    (OUT / "score.json").write_text(json.dumps(table.to_dict("records"), indent=1), encoding="utf-8")
    (OUT / "score.txt").write_text(table.to_string(index=False), encoding="utf-8")
    print(table.to_string(index=False))
    return 0
