How it works

The field encodes the reason(s) a soldier left the Legions. In the raw data it takes the form "DD.MM.YYYY ( CODE )" where CODE is one or more single- or multi-letter tokens from the codebook below, optionally separated by commas, spaces, or slashes. Some records substitute full Czech words for the letter codes. A minority of records omit the date and contain only the code, or contain only the date with no code.

Codebook (source: csol.cz / VÚA-VHA Praha)

B Padl v boji Killed in action ČSA Sloužil v čs. armádě po r. 1918 Served in post-1918 Czechoslovak army D Dezertoval Deserted DM Demobilizován Demobilised N Nezvěstný Missing NS Neschopen služby Unfit for service P Popraven Executed S Zemřel Died V Vrátil se do vlasti Returned home X Viz poznámku See note Z Zemřel na následky zranění Died of wounds

Source type codes (archival material identifiers, not exit codes): Is Legionářský poslužný spis k Osobní karta legionáře o Osvědčení o vykonané legionářské službě

def parse_exit_codes(raw: str | float | None) -> frozenset[str]:
    """Parse a raw *Konec v legiích* string into a frozenset of canonical codes.

    Parameters
    ----------
    raw:
        A single cell value from the *Konec v legiích* column.

    Returns
    -------
    frozenset[str]
        Zero or more codes from :data:`CODES_CS`.  Empty if the input is NaN,
        blank, or contains only unrecognised tokens.

    Examples
    --------
    >>> parse_exit_codes("30.12.1920 ( Demobilizován )")
    frozenset({'DM'})
    >>> parse_exit_codes("1.1.1918 ( DM,D X )")
    frozenset({'D', 'DM', 'X'})
    >>> parse_exit_codes(float("nan"))
    frozenset()
    """
    # Guard EVERY missing scalar — including pd.NA, which clean.normalize_missing
    # substitutes for placeholders. The old float-only guard let pd.NA through as
    # the string "<NA>", whose letter N was scanned into a fabricated "Missing"
    # exit code on every empty cell (26,386 rows in the 2026-07 build).
    if raw is None or (not isinstance(raw, str) and pd.isna(raw)):
        return frozenset()

    token = str(raw).strip()

    # Extract parenthetical content when present; otherwise work with the
    # full string (some records omit the date wrapper entirely).
    m = re.search(r"\(([^)]+)\)", token)
    token = m.group(1).strip() if m else token

    # Substitute full Czech words before any further processing
    for word, code in _WORD_TO_CODE.items():
        token = token.replace(word, code)

    # Normalise to uppercase; collapse separators and whitespace
    token = token.upper()
    token = re.sub(r"[,./\-]+", " ", token)
    token = re.sub(r"\s+", " ", token).strip()

    # Strip leading date fragments that survived the substitutions
    token = _DATE_RE.sub("", token).strip()

    # Token-wise extraction. Each whitespace token is first checked against the
    # drop list (archival markers — struck whole, so "MD" can never yield D),
    # then must parse COMPLETELY as a sequence of codes (greedy, ČSA (3) before
    # DM/NS (2) before single letters). A token with unparseable residue is
    # skipped entirely — partial reads fabricated codes from unknown strings
    # ("SUPER" → S, "PROP" → P), so no partial extraction is allowed.
    found: set[str] = set()
    for tok in token.split():
        tok = _TOKEN_SUBS.get(tok, tok).replace("?", "")   # "D?", "DM?V" = uncertain — keep codes
        if not tok or tok in _DROP_TOKENS or not tok.isalpha():
            continue
        codes: set[str] = set()
        i = 0
        while i < len(tok):
            for length in (3, 2, 1):
                candidate = tok[i : i + length]
                if candidate in CODES_CS:
                    codes.add(candidate)
                    i += len(candidate)
                    break
            else:
                codes = set()        # unparseable residue → reject the whole token
                break
        found |= codes

    return frozenset(found)

def add_exit_columns(df: pd.DataFrame, source_col: str = "Konec v legiích") -> pd.DataFrame:
    """Add harmonised exit-code columns to a legionnaire DataFrame.

    Adds the following columns:

    ``exit_codes``
        frozenset of canonical code strings for each record.

    ``exit_<CODE>`` (one per code in :data:`CODES_CS`)
        Boolean indicator, True if the code is present in *exit_codes*.

    ``deserted``
        Shorthand for ``exit_D``; the primary outcome variable.

    ``exit_date``
        Date parsed from the text **before** the parenthesis, through the same machinery
        as every other date (:func:`utils.parsing.dates.parse_date`): range
        1914-01-01 – 1939-12-31, the century-slip fix both ways, and month-year / year-only
        fallbacks (``pd.NaT`` if absent, unparseable or out of range).

    ``exit_date_precision``
        ``day`` / ``month`` / ``year`` — how exact ``exit_date`` is; ``<NA>`` without a date.
        Until v1.2.0 the exit date was the one date in the pipeline with **no range check**
        (values ran from 1818 to 2192), and a year-only exit (``1920 ( Vrátil se do vlasti
        )``, 3,930 records) got no date at all. Only the text before the parenthesis is
        read, so the ``1918`` inside the label *Sloužil v čs. armádě po r. 1918* can never
        become an exit year (audit 2026-09-22 §A8).

    Parameters
    ----------
    df:
        Input DataFrame containing *source_col*.
    source_col:
        Column name of the raw exit field.  Defaults to ``"Konec v legiích"``.

    Returns
    -------
    pd.DataFrame
        The original DataFrame with new columns appended (copy).
    """
    df = df.copy()

    df["exit_codes"] = df[source_col].apply(parse_exit_codes)

    for code in CODES_CS:
        df[f"exit_{code}"] = df["exit_codes"].apply(lambda s, c=code: c in s)

    df["deserted"] = df["exit_D"]

    parsed = df[source_col].apply(parse_exit_date)
    df["exit_date"] = pd.to_datetime(parsed.map(lambda t: t[0]))
    df["exit_date_precision"] = parsed.map(lambda t: t[1]).astype("string")

    return df

def parse_exit_date(raw) -> tuple[pd.Timestamp, str | None]:
    """``(exit date, precision)`` from the text before the parenthesis of one exit cell."""
    if raw is None or (not isinstance(raw, str) and pd.isna(raw)):
        return pd.NaT, None
    head = str(raw).split("(", 1)[0].strip()
    head = re.sub(r"(?<=\d)\s+(?=\d)", ".", head)          # "1 1 1918" -> "1.1.1918"
    lo, hi = EXIT_DATE_RANGE
    return parse_date(head, lo, hi, partial=True)

def exit_code_report(df: pd.DataFrame) -> None:
    """Print a frequency table of parsed exit codes to stdout.

    Requires that :func:`add_exit_columns` has already been called on *df*.
    """
    from collections import Counter

    if "exit_codes" not in df.columns:
        raise ValueError("Run add_exit_columns(df) first.")

    counts = Counter(code for codes in df["exit_codes"] for code in codes)
    n = len(df)

    print(f"Exit code report  (N = {n:,})\n")
    print(f"  {'Code':<6}  {'Czech label':<45}  {'English label':<40}  {'n':>7}  {'%':>6}")
    print("  " + "-" * 114)
    for code in CODES_CS:
        cs = CODES_CS[code]
        en = CODES_EN[code]
        c = counts.get(code, 0)
        print(f"  {code:<6}  {cs:<45}  {en:<40}  {c:>7,}  {c/n*100:>5.2f}%")

    n_empty = sum(1 for s in df["exit_codes"] if len(s) == 0)
    print(f"\n  {'—':<6}  {'No code parsed':<45}  {'':<40}  {n_empty:>7,}  {n_empty/n*100:>5.2f}%")

"""
Exit code harmonisation for the 'Konec v legiích' field.

The field encodes the reason(s) a soldier left the Legions.  In the raw data it
takes the form  "DD.MM.YYYY ( CODE )"  where CODE is one or more single- or
multi-letter tokens from the codebook below, optionally separated by commas,
spaces, or slashes.  Some records substitute full Czech words for the letter
codes.  A minority of records omit the date and contain only the code, or
contain only the date with no code.

Codebook (source: csol.cz / VÚA-VHA Praha)
-------------------------------------------
B    Padl v boji                               Killed in action
ČSA  Sloužil v čs. armádě po r. 1918          Served in post-1918 Czechoslovak army
D    Dezertoval                                Deserted
DM   Demobilizován                             Demobilised
N    Nezvěstný                                 Missing
NS   Neschopen služby                          Unfit for service
P    Popraven                                  Executed
S    Zemřel                                    Died
V    Vrátil se do vlasti                       Returned home
X    Viz poznámku                              See note
Z    Zemřel na následky zranění               Died of wounds

Source type codes (archival material identifiers, not exit codes):
Is   Legionářský poslužný spis
k    Osobní karta legionáře
o    Osvědčení o vykonané legionářské službě

Usage
-----
    >>> from utils.parsing.exit_codes import parse_exit_codes, add_exit_columns
    >>> codes = parse_exit_codes("30.12.1920 ( Demobilizován )")
    >>> codes
    {'DM'}
    >>> codes = parse_exit_codes("1.1.1918 ( DM,D X )")
    >>> codes
    {'DM', 'D', 'X'}

    >>> import pandas as pd
    >>> df = pd.read_csv("data/raw/legionnaires/legionnaires.csv", low_memory=False)
    >>> df = add_exit_columns(df)

Author: Tomáš Haufer
"""

from __future__ import annotations

import re
from collections.abc import Iterable

import pandas as pd

from utils.parsing.dates import parse_date

__version__ = "1.2.0"   # exit date: range-checked, pre-parenthesis, partial dates (audit 2026-09-22)

# Codebook

#: Canonical codes → Czech label
CODES_CS: dict[str, str] = {
    "B":   "Padl v boji",
    "ČSA": "Sloužil v čs. armádě po r. 1918",
    "D":   "Dezertoval",
    "DM":  "Demobilizován",
    "N":   "Nezvěstný",
    "NS":  "Neschopen služby",
    "P":   "Popraven",
    "S":   "Zemřel",
    "V":   "Vrátil se do vlasti",
    "X":   "Viz poznámku",
    "Z":   "Zemřel na následky zranění",
}

#: Canonical codes → English label
CODES_EN: dict[str, str] = {
    "B":   "Killed in action",
    "ČSA": "Served in post-1918 Czechoslovak army",
    "D":   "Deserted",
    "DM":  "Demobilised",
    "N":   "Missing",
    "NS":  "Unfit for service",
    "P":   "Executed",
    "S":   "Died",
    "V":   "Returned home",
    "X":   "See note",
    "Z":   "Died of wounds",
}

# Full Czech words that appear verbatim in the raw field instead of letter codes.
# Replaced before tokenisation.
_WORD_TO_CODE: dict[str, str] = {
    "Demobilizován":                    "DM",
    "Dezertoval":                       "D",
    "Neschopen služby":                 "NS",
    "Nezvěstný":                        "N",
    "Padl v boji":                      "B",
    "Popraven":                         "P",
    "Sloužil v čs. armádě po r. 1918":  "ČSA",
    "Viz poznámku":                     "X",
    "Vrátil se do vlasti":              "V",
    "Zemřel na následky zranění":       "Z",
    "Zemřel":                           "S",   # must come after the longer variant
}

# Whole tokens observed in the raw data that are archival identifiers, annotations,
# or unresolvable code-like strings — struck from the token stream BEFORE code
# extraction, so their letters can never fabricate codes ("MD" must not yield D).
#
#   IS / K / O   source type codes (Is = poslužný spis, k = osobní karta, o = osvědčení)
#   C            isolated C — likely OCR artefact
#   LF / LI / LR unidentified archival markers
#   M            unidentified
#   MD / DMD     ambiguous (DM transposed/doubled, or DM+D?) — 2 records, refused rather
#                than guessed: legion_id 113629 ("14.8.1920 ( MD )", Russia) and
#                18666 ("19.7.1919 ( DMD )", France); both plausibly DM but not certain
#   SUPER / PROP annotation fragments
_DROP_TOKENS: frozenset[str] = frozenset({
    "IS", "K", "O",
    "C",
    "LF", "LI", "LR",
    "M", "MD", "DMD",
    "SUPER", "PROP",
})

# Whole-token substitutions applied before extraction (variant → canonical spelling).
_TOKEN_SUBS: dict[str, str] = {
    "X2": "X",               # variant note marker → see-note
    "DMM": "DM",             # DM with a glued M archival marker (1 row)
    "DMCSA": "DMČSA",        # DM+ČSA glued, diacritics dropped (1 row) → parses as DM, ČSA
}

# Pre-compiled pattern that strips leading date fragments  "DD.MM.YYYY" or
# partial dates so they do not pollute the token stream.
_DATE_RE = re.compile(r"^\s*[\d\s.]*")

# Core parsing logic
# ---------------------------------------------------------------------------

def parse_exit_codes(raw: str | float | None) -> frozenset[str]:
    """Parse a raw *Konec v legiích* string into a frozenset of canonical codes.

    Parameters
    ----------
    raw:
        A single cell value from the *Konec v legiích* column.

    Returns
    -------
    frozenset[str]
        Zero or more codes from :data:`CODES_CS`.  Empty if the input is NaN,
        blank, or contains only unrecognised tokens.

    Examples
    --------
    >>> parse_exit_codes("30.12.1920 ( Demobilizován )")
    frozenset({'DM'})
    >>> parse_exit_codes("1.1.1918 ( DM,D X )")
    frozenset({'D', 'DM', 'X'})
    >>> parse_exit_codes(float("nan"))
    frozenset()
    """
    # Guard EVERY missing scalar — including pd.NA, which clean.normalize_missing
    # substitutes for placeholders. The old float-only guard let pd.NA through as
    # the string "<NA>", whose letter N was scanned into a fabricated "Missing"
    # exit code on every empty cell (26,386 rows in the 2026-07 build).
    if raw is None or (not isinstance(raw, str) and pd.isna(raw)):
        return frozenset()

    token = str(raw).strip()

    # Extract parenthetical content when present; otherwise work with the
    # full string (some records omit the date wrapper entirely).
    m = re.search(r"\(([^)]+)\)", token)
    token = m.group(1).strip() if m else token

    # Substitute full Czech words before any further processing
    for word, code in _WORD_TO_CODE.items():
        token = token.replace(word, code)

    # Normalise to uppercase; collapse separators and whitespace
    token = token.upper()
    token = re.sub(r"[,./\-]+", " ", token)
    token = re.sub(r"\s+", " ", token).strip()

    # Strip leading date fragments that survived the substitutions
    token = _DATE_RE.sub("", token).strip()

    # Token-wise extraction. Each whitespace token is first checked against the
    # drop list (archival markers — struck whole, so "MD" can never yield D),
    # then must parse COMPLETELY as a sequence of codes (greedy, ČSA (3) before
    # DM/NS (2) before single letters). A token with unparseable residue is
    # skipped entirely — partial reads fabricated codes from unknown strings
    # ("SUPER" → S, "PROP" → P), so no partial extraction is allowed.
    found: set[str] = set()
    for tok in token.split():
        tok = _TOKEN_SUBS.get(tok, tok).replace("?", "")   # "D?", "DM?V" = uncertain — keep codes
        if not tok or tok in _DROP_TOKENS or not tok.isalpha():
            continue
        codes: set[str] = set()
        i = 0
        while i < len(tok):
            for length in (3, 2, 1):
                candidate = tok[i : i + length]
                if candidate in CODES_CS:
                    codes.add(candidate)
                    i += len(candidate)
                    break
            else:
                codes = set()        # unparseable residue → reject the whole token
                break
        found |= codes

    return frozenset(found)


def label_exit_codes(
    codes: Iterable[str],
    lang: str = "cs",
) -> dict[str, str]:
    """Map a collection of parsed codes to their human-readable labels.

    Parameters
    ----------
    codes:
        An iterable of canonical code strings (e.g. from :func:`parse_exit_codes`).
    lang:
        ``"cs"`` for Czech labels (default), ``"en"`` for English.

    Returns
    -------
    dict[str, str]
        Mapping of code → label for each recognised code.
    """
    lookup = CODES_CS if lang == "cs" else CODES_EN
    return {c: lookup[c] for c in codes if c in lookup}

def add_exit_columns(df: pd.DataFrame, source_col: str = "Konec v legiích") -> pd.DataFrame:
    """Add harmonised exit-code columns to a legionnaire DataFrame.

    Adds the following columns:

    ``exit_codes``
        frozenset of canonical code strings for each record.

    ``exit_<CODE>`` (one per code in :data:`CODES_CS`)
        Boolean indicator, True if the code is present in *exit_codes*.

    ``deserted``
        Shorthand for ``exit_D``; the primary outcome variable.

    ``exit_date``
        Date parsed from the text **before** the parenthesis, through the same machinery
        as every other date (:func:`utils.parsing.dates.parse_date`): range
        1914-01-01 – 1939-12-31, the century-slip fix both ways, and month-year / year-only
        fallbacks (``pd.NaT`` if absent, unparseable or out of range).

    ``exit_date_precision``
        ``day`` / ``month`` / ``year`` — how exact ``exit_date`` is; ``<NA>`` without a date.
        Until v1.2.0 the exit date was the one date in the pipeline with **no range check**
        (values ran from 1818 to 2192), and a year-only exit (``1920 ( Vrátil se do vlasti
        )``, 3,930 records) got no date at all. Only the text before the parenthesis is
        read, so the ``1918`` inside the label *Sloužil v čs. armádě po r. 1918* can never
        become an exit year (audit 2026-09-22 §A8).

    Parameters
    ----------
    df:
        Input DataFrame containing *source_col*.
    source_col:
        Column name of the raw exit field.  Defaults to ``"Konec v legiích"``.

    Returns
    -------
    pd.DataFrame
        The original DataFrame with new columns appended (copy).
    """
    df = df.copy()

    df["exit_codes"] = df[source_col].apply(parse_exit_codes)

    for code in CODES_CS:
        df[f"exit_{code}"] = df["exit_codes"].apply(lambda s, c=code: c in s)

    df["deserted"] = df["exit_D"]

    parsed = df[source_col].apply(parse_exit_date)
    df["exit_date"] = pd.to_datetime(parsed.map(lambda t: t[0]))
    df["exit_date_precision"] = parsed.map(lambda t: t[1]).astype("string")

    return df


EXIT_DATE_RANGE = (pd.Timestamp("1914-01-01"), pd.Timestamp("1939-12-31"))


def parse_exit_date(raw) -> tuple[pd.Timestamp, str | None]:
    """``(exit date, precision)`` from the text before the parenthesis of one exit cell."""
    if raw is None or (not isinstance(raw, str) and pd.isna(raw)):
        return pd.NaT, None
    head = str(raw).split("(", 1)[0].strip()
    head = re.sub(r"(?<=\d)\s+(?=\d)", ".", head)          # "1 1 1918" -> "1.1.1918"
    lo, hi = EXIT_DATE_RANGE
    return parse_date(head, lo, hi, partial=True)

def exit_code_report(df: pd.DataFrame) -> None:
    """Print a frequency table of parsed exit codes to stdout.

    Requires that :func:`add_exit_columns` has already been called on *df*.
    """
    from collections import Counter

    if "exit_codes" not in df.columns:
        raise ValueError("Run add_exit_columns(df) first.")

    counts = Counter(code for codes in df["exit_codes"] for code in codes)
    n = len(df)

    print(f"Exit code report  (N = {n:,})\n")
    print(f"  {'Code':<6}  {'Czech label':<45}  {'English label':<40}  {'n':>7}  {'%':>6}")
    print("  " + "-" * 114)
    for code in CODES_CS:
        cs = CODES_CS[code]
        en = CODES_EN[code]
        c = counts.get(code, 0)
        print(f"  {code:<6}  {cs:<45}  {en:<40}  {c:>7,}  {c/n*100:>5.2f}%")

    n_empty = sum(1 for s in df["exit_codes"] if len(s) == 0)
    print(f"\n  {'—':<6}  {'No code parsed':<45}  {'':<40}  {n_empty:>7,}  {n_empty/n*100:>5.2f}%")
