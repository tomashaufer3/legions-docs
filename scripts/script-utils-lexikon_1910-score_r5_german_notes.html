How it works

Two questions, both answered from files on disk; nothing is sent.

  1. How well does each reader read the German edition? Every German pass (bakeoff/&lt;host&gt;/de/inventory_*_r5*gde*.csv) against the German gold sheets, reference/places/transcription_gold/de/{calibration,purposive} — read by read_transcription_md.read_sheet, the reader the Czech gold goes through, and scored by lexikon_1910_institutions.score_against_reference, the Czech arms' scorer. Settlement and municipality level, per pass, per reader (mean and range over repeats, and a majority of the repeats). The postal legend, the column header and the heading reads are reported beside it.
  2. Are the two editions two measurements? Each Czech r5 pass against each German pass, paired by court and municipality name, never by page (D-67), through lexikon_1910_editions — the conditional-independence test D-58 asks for.

The German sheets are being transcribed now and most are drafts: a blank key is not an entry, so a draft scores only on what it holds, and the report prints each sheet's status. Development gold; it certifies nothing, and the certification sheets are never opened.

def read_gold(paths: list[Path], lang: str, volume: str = "") -> list[dict]:
    """Every sheet through the Czech gold's own reader, with the edition's tokens,
    its claim rows as written (:func:`as_written`).

    ``volume`` (a profile slug) gives a German volume its own plate's tokens (1.1.0,
    D-146). Read with Böhmen's, a Mähren sheet's ``[cooperative]`` or ``[rc-lokalie]``
    is not on the atlas: the reader drops it with a problem line, and the route's
    correct claim would score as false."""
    allowed = lxv.vocabulary(abbreviations(lang, volume)) | {OFF_ATLAS} | TOPOGRAPHIC
    sheets = [rt.read_sheet(path, allowed) for path in paths]
    for sheet in sheets:
        sheet["offices"] = as_written(sheet["offices"])
        sheet["legend"] = as_written(sheet["legend"])
    return sheets

def ledger_names(path: Path) -> dict:
    """``{(court slug, muni_seq): printed name}`` from a sheet's ``==`` lines.

    The sheet's own locator: the municipality names were pre-filled from the
    print of *that* edition, so they are the names the CZ–DE pairing joins on.
    """
    names: dict = {}
    for section, district, kind, lines in rt._blocks(path.read_text(encoding="utf-8")):
        if kind != "ledger" or section not in ("2a", "2b"):
            continue
        for line in lines:
            match = _MUNI_RE.match(line.strip())
            if match:
                names.setdefault((district, int(match.group(1))), match.group(2).strip())
    return names

def arm_summary(gold: list[dict], passes: dict, scored: dict) -> list[str]:
    """Mean and range over repeats per reader, and a majority of its repeats."""
    import run_e1_gemini_version as e1

    lines = ["", "  per reader — mean [min, max] over repeats, and the majority vote"]
    for reader in sorted({family_of(n) for n in passes}):
        names = sorted(n for n in passes if family_of(n) == reader)
        rows = [scored[n] for n in names if n in scored]
        if not rows:
            continue
        k = len(rows)
        cell = lambda key: (f"{sum(r[key] for r in rows) / k:.3f} "
                            f"[{min(r[key] for r in rows):.3f}, "
                            f"{max(r[key] for r in rows):.3f}]")
        vote = k // 2 + 1
        maj = lxin.score_against_reference(
            {"human": gold, "maj": e1.majority([passes[n] for n in names], vote)},
            reference="human")
        tail = (f"   ≥{vote}-of-{k}: F1 sett {maj[0]['f1_settlement']:.3f}"
                if maj else "")
        lines.append(f"  {reader:10s} n={k}  F1 sett {cell('f1_settlement')}   "
                     f"F1 muni {cell('f1')}   exact {cell('entry_exact')}{tail}")
    return lines

def legend_agreement(gold: list[dict], rows: list[dict]) -> tuple[int, int, int]:
    """``(gold keys, keys the pass has, of those the place agrees)``."""
    want = {(r["page"], r["district"], str(r["key"])): lxnt.normalise(r["place"])
            for r in gold}
    got = {(r["page"], r["district"], str(r["key"])): lxnt.normalise(r["place"])
           for r in rows}
    both = want.keys() & got.keys()
    return len(want), len(both), sum(1 for k in both if want[k] == got[k])

def side_lines(passes: dict, gold_legend: list[dict]) -> list[str]:
    """The legend, the header and the heading reads, one line a pass."""
    lines = ["", "  the rest of what a pass reads — legend keys right / gold keys; "
                 "header complete on pages; headings labelled"]
    for name in sorted(passes):
        n_gold, n_both, same = legend_agreement(gold_legend, side_files(name, "legend"))
        header = side_files(name, "header")
        full = sum(1 for r in header if r["columns_found"] == r["columns"])
        heads = side_files(name, "headings")
        labelled = sum(1 for r in heads if r["kind"])
        lines.append(f"  {name:26s} legend {same}/{n_gold} (read {n_both})   header "
                     f"{full}/{len(header)}   headings {labelled}/{len(heads)}")
    return lines

def score_german(sheets: list[dict], passes: dict) -> tuple[list[str], list[dict]]:
    gold = [row for sheet in sheets for row in sheet["offices"]]
    gold_legend = [row for sheet in sheets for row in sheet["legend"]]
    lines = ["  the German gold sheets — scan, status, filled offices rows, problems"]
    for sheet in sheets:
        lines.append(f"    {sheet['scan']}  {sheet['meta'].get('status', '?'):9s} "
                     f"{len(sheet['offices']):4d} claims rows  "
                     f"{len(sheet['problems'])} problems")
    lines.append(f"  gold: {lxin.describe_reference(gold)} (development gold; "
                 f"certifies nothing)")
    if not passes:
        return lines + ["", f"  no German pass of run {bo.RUN_TAG} on disk yet "
                            f"(run_r5_german_notes.py)"], []
    scored_rows = lxin.score_against_reference({"human": gold, **passes}, "human")
    scored = {r["pass"]: r for r in scored_rows}
    lines += ["", f"  {'pass':26s} {'entries':>7s} {'missed':>6s} {'P sett':>6s} "
                  f"{'R sett':>6s} {'F1 sett':>7s} {'F1 muni':>7s} {'exact':>6s} "
                  f"{'misfiled':>8s}"]
    for r in scored_rows:
        lines.append(f"  {r['pass']:26s} {r['entries_shared']:7d} {r['entries_missed']:6d} "
                     f"{r['precision_settlement']:6.3f} {r['recall_settlement']:6.3f} "
                     f"{r['f1_settlement']:7.3f} {r['f1']:7.3f} {r['entry_exact']:6.3f} "
                     f"{r['false_misfiled']:8d}")
    lines += arm_summary(gold, passes, scored)
    lines += side_lines(passes, gold_legend)
    return lines, [{"reader": family_of(r["pass"]), **r} for r in scored_rows]

def compare(de_sheets: list[dict], de_paths: list[Path], de_passes: dict,
            draws: int) -> tuple[list[str], list[dict]]:
    cz_paths = sheet_paths(CZ_GOLD_DIR, matched_cz_scans())
    cz_sheets = read_gold(cz_paths, "cs")
    cz_names, de_names = {}, {}
    for path in cz_paths:
        cz_names.update(ledger_names(path))
    for path in de_paths:
        de_names.update(ledger_names(path))
    cz_passes = {name: rows for name, rows in bo.passes_on_disk(hosts="all").items()
                 if CZ_PASS_RE.match(name)}
    result = lxed.compare_editions(
        [r for s in cz_sheets for r in s["offices"]],
        [r for s in de_sheets for r in s["offices"]],
        cz_names, de_names, cz_passes, de_passes, family=family_of, draws=draws,
        shared=lxv.vocabulary(abbreviations("cs")) & lxv.vocabulary(abbreviations("de")))
    head = [f"  Czech gold scans {', '.join(sorted(s['scan'] for s in cz_sheets))}; "
            f"Czech r5 passes on disk: {len(cz_passes)}; only the symbols both "
            f"plates print are compared"]
    return head + [lxed.lexikon_1910_editions_report(result)], result["rows"]

def main(argv=None) -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):
            pass
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--no-compare", action="store_true",
                    help="skip the CZ–DE independence test")
    ap.add_argument("--draws", type=int, default=lxed.DRAWS,
                    help="permutation draws for the independence test")
    args = ap.parse_args(argv)

    de_paths = sheet_paths(bo.DE_GOLD_DIR)
    de_sheets = read_gold(de_paths, "de")
    de_passes = bo.de_passes_on_disk(hosts="all")
    lines = [f"score_r5_german_notes v{__version__} — {time.strftime('%Y-%m-%d %H:%M')} "
             f"— run {bo.RUN_TAG}", ""]
    german, rows = score_german(de_sheets, de_passes)
    lines += german
    pair_rows: list[dict] = []
    if not args.no_compare:
        compared, pair_rows = compare(de_sheets, de_paths, de_passes, args.draws)
        lines += ["", ""] + compared
    report = "\n".join(lines)
    print(report)
    if rows or pair_rows:
        write_csv(OUT_CSV, rows)
        write_csv(OUT_PAIRS, pair_rows)
        OUT_REPORT.parent.mkdir(parents=True, exist_ok=True)
        OUT_REPORT.write_text(report + "\n", encoding="utf-8")
    return 0

"""Score the German notes arm of ``r5`` against the German gold, and set it beside the Czech (D-92).

Two questions, both answered from files on disk; nothing is sent.

1. **How well does each reader read the German edition?** Every German pass
   (``bakeoff/<host>/de/inventory_*_r5*gde*.csv``) against the German gold
   sheets, ``reference/places/transcription_gold/de/{calibration,purposive}`` —
   read by ``read_transcription_md.read_sheet``, the reader the Czech gold goes
   through, and scored by ``lexikon_1910_institutions.score_against_reference``,
   the Czech arms' scorer. Settlement and municipality level, per pass, per
   reader (mean and range over repeats, and a majority of the repeats). The
   postal legend, the column header and the heading reads are reported beside
   it.
2. **Are the two editions two measurements?** Each Czech r5 pass against each
   German pass, paired by court and municipality name, never by page (D-67),
   through ``lexikon_1910_editions`` — the conditional-independence test D-58
   asks for.

The German sheets are being transcribed now and most are drafts: a blank key is
not an entry, so a draft scores only on what it holds, and the report prints
each sheet's status. **Development gold; it certifies nothing**, and the
certification sheets are never opened.

    python utils/lexikon_1910/score_r5_german_notes.py

#: The Czech r5 passes the German ones are set against: the same two readers, as
#: ``run_e1_gemini_version.py`` (arm E) and ``run_e1b_second_reader.py`` label them.
CZ_PASS_RE = re.compile(rf"^(gemini38_{bo.RUN_TAG}atlasntg|haiku55h_{bo.RUN_TAG}atlasg)\d+$")

OUT_CSV = bo.BAKEOFF / f"german_notes_{bo.RUN_TAG}.csv"
OUT_PAIRS = bo.BAKEOFF / f"german_notes_{bo.RUN_TAG}_cz_de.csv"
OUT_REPORT = bo.BAKEOFF / f"german_notes_{bo.RUN_TAG}.txt"

_MUNI_RE = re.compile(r"^==\s*(\d{1,3})\s+(.*)$")


def family_of(label: str) -> str:
    """``haiku55h_r5atlasgde2`` -> ``haiku55h``."""
    return label.split(f"_{bo.RUN_TAG}")[0]


def abbreviations(lang: str, volume: str = ""):
    return lxin.load_abbreviations(rt.ABBREVIATIONS, lang=lang, volume=volume)


def sheet_paths(gold_dir: Path, scans=None) -> list[Path]:
    """The gold sheets of the calibration and purposive halves; never certification."""
    paths = []
    for role in bo.GOLD_ROLES:
        for path in sorted((gold_dir / role).glob("[0-9][0-9][0-9][0-9].md")):
            if scans is None or path.stem in scans:
                paths.append(path)
    return paths


def as_written(rows: list[dict]) -> list[dict]:
    """Rows as a CSV gives them back: every value text, ``None`` blank.

    **The scorer keys an entry on ``(page, district, muni_seq)`` as given.** A
    pass is read back from its CSV, so its ``muni_seq`` is ``"3"``; a sheet
    read in memory gives ``3``. Compared as they come, no entry is shared and
    every pass scores recall 0 without an error — so the gold is put through the
    same round trip the Czech gold takes via ``inventory_human.csv``.
    """
    return [{k: "" if v is None else str(v) for k, v in row.items()} for row in rows]


def read_gold(paths: list[Path], lang: str, volume: str = "") -> list[dict]:
    """Every sheet through the Czech gold's own reader, with the edition's tokens,
    its claim rows as written (:func:`as_written`).

    ``volume`` (a profile slug) gives a German volume its own plate's tokens (1.1.0,
    D-146). Read with Böhmen's, a Mähren sheet's ``[cooperative]`` or ``[rc-lokalie]``
    is not on the atlas: the reader drops it with a problem line, and the route's
    correct claim would score as false."""
    allowed = lxv.vocabulary(abbreviations(lang, volume)) | {OFF_ATLAS} | TOPOGRAPHIC
    sheets = [rt.read_sheet(path, allowed) for path in paths]
    for sheet in sheets:
        sheet["offices"] = as_written(sheet["offices"])
        sheet["legend"] = as_written(sheet["legend"])
    return sheets


def ledger_names(path: Path) -> dict:
    """``{(court slug, muni_seq): printed name}`` from a sheet's ``==`` lines.

    The sheet's own locator: the municipality names were pre-filled from the
    print of *that* edition, so they are the names the CZ–DE pairing joins on.
    """
    names: dict = {}
    for section, district, kind, lines in rt._blocks(path.read_text(encoding="utf-8")):
        if kind != "ledger" or section not in ("2a", "2b"):
            continue
        for line in lines:
            match = _MUNI_RE.match(line.strip())
            if match:
                names.setdefault((district, int(match.group(1))), match.group(2).strip())
    return names

# 1. The German edition against its own gold

def arm_summary(gold: list[dict], passes: dict, scored: dict) -> list[str]:
    """Mean and range over repeats per reader, and a majority of its repeats."""
    import run_e1_gemini_version as e1

    lines = ["", "  per reader — mean [min, max] over repeats, and the majority vote"]
    for reader in sorted({family_of(n) for n in passes}):
        names = sorted(n for n in passes if family_of(n) == reader)
        rows = [scored[n] for n in names if n in scored]
        if not rows:
            continue
        k = len(rows)
        cell = lambda key: (f"{sum(r[key] for r in rows) / k:.3f} "
                            f"[{min(r[key] for r in rows):.3f}, "
                            f"{max(r[key] for r in rows):.3f}]")
        vote = k // 2 + 1
        maj = lxin.score_against_reference(
            {"human": gold, "maj": e1.majority([passes[n] for n in names], vote)},
            reference="human")
        tail = (f"   ≥{vote}-of-{k}: F1 sett {maj[0]['f1_settlement']:.3f}"
                if maj else "")
        lines.append(f"  {reader:10s} n={k}  F1 sett {cell('f1_settlement')}   "
                     f"F1 muni {cell('f1')}   exact {cell('entry_exact')}{tail}")
    return lines


def side_files(name: str, stem: str) -> list[dict]:
    """A pass's side file (``legend_``, ``header_``, ``headings_``) from any host."""
    for directory in sorted(d / "de" for d in bo.BAKEOFF.glob("*") if (d / "de").is_dir()):
        path = directory / f"{stem}_{name}.csv"
        if path.exists():
            with path.open(encoding="utf-8", newline="") as fh:
                return list(csv.DictReader(fh))
    return []


def legend_agreement(gold: list[dict], rows: list[dict]) -> tuple[int, int, int]:
    """``(gold keys, keys the pass has, of those the place agrees)``."""
    want = {(r["page"], r["district"], str(r["key"])): lxnt.normalise(r["place"])
            for r in gold}
    got = {(r["page"], r["district"], str(r["key"])): lxnt.normalise(r["place"])
           for r in rows}
    both = want.keys() & got.keys()
    return len(want), len(both), sum(1 for k in both if want[k] == got[k])


def side_lines(passes: dict, gold_legend: list[dict]) -> list[str]:
    """The legend, the header and the heading reads, one line a pass."""
    lines = ["", "  the rest of what a pass reads — legend keys right / gold keys; "
                 "header complete on pages; headings labelled"]
    for name in sorted(passes):
        n_gold, n_both, same = legend_agreement(gold_legend, side_files(name, "legend"))
        header = side_files(name, "header")
        full = sum(1 for r in header if r["columns_found"] == r["columns"])
        heads = side_files(name, "headings")
        labelled = sum(1 for r in heads if r["kind"])
        lines.append(f"  {name:26s} legend {same}/{n_gold} (read {n_both})   header "
                     f"{full}/{len(header)}   headings {labelled}/{len(heads)}")
    return lines


def score_german(sheets: list[dict], passes: dict) -> tuple[list[str], list[dict]]:
    gold = [row for sheet in sheets for row in sheet["offices"]]
    gold_legend = [row for sheet in sheets for row in sheet["legend"]]
    lines = ["  the German gold sheets — scan, status, filled offices rows, problems"]
    for sheet in sheets:
        lines.append(f"    {sheet['scan']}  {sheet['meta'].get('status', '?'):9s} "
                     f"{len(sheet['offices']):4d} claims rows  "
                     f"{len(sheet['problems'])} problems")
    lines.append(f"  gold: {lxin.describe_reference(gold)} (development gold; "
                 f"certifies nothing)")
    if not passes:
        return lines + ["", f"  no German pass of run {bo.RUN_TAG} on disk yet "
                            f"(run_r5_german_notes.py)"], []
    scored_rows = lxin.score_against_reference({"human": gold, **passes}, "human")
    scored = {r["pass"]: r for r in scored_rows}
    lines += ["", f"  {'pass':26s} {'entries':>7s} {'missed':>6s} {'P sett':>6s} "
                  f"{'R sett':>6s} {'F1 sett':>7s} {'F1 muni':>7s} {'exact':>6s} "
                  f"{'misfiled':>8s}"]
    for r in scored_rows:
        lines.append(f"  {r['pass']:26s} {r['entries_shared']:7d} {r['entries_missed']:6d} "
                     f"{r['precision_settlement']:6.3f} {r['recall_settlement']:6.3f} "
                     f"{r['f1_settlement']:7.3f} {r['f1']:7.3f} {r['entry_exact']:6.3f} "
                     f"{r['false_misfiled']:8d}")
    lines += arm_summary(gold, passes, scored)
    lines += side_lines(passes, gold_legend)
    return lines, [{"reader": family_of(r["pass"]), **r} for r in scored_rows]

def matched_cz_scans() -> set[str]:
    """The Czech scans the German gold pages are matched to (gold_sample_pages_de.csv)."""
    with bo.DE_GOLD_SAMPLE.open(encoding="utf-8", newline="") as fh:
        return {f"{int(s):04d}" for row in csv.DictReader(fh)
                if row.get("role") in bo.GOLD_ROLES
                for s in (row.get("cz_scans") or "").split() if s.strip()}


def compare(de_sheets: list[dict], de_paths: list[Path], de_passes: dict,
            draws: int) -> tuple[list[str], list[dict]]:
    cz_paths = sheet_paths(CZ_GOLD_DIR, matched_cz_scans())
    cz_sheets = read_gold(cz_paths, "cs")
    cz_names, de_names = {}, {}
    for path in cz_paths:
        cz_names.update(ledger_names(path))
    for path in de_paths:
        de_names.update(ledger_names(path))
    cz_passes = {name: rows for name, rows in bo.passes_on_disk(hosts="all").items()
                 if CZ_PASS_RE.match(name)}
    result = lxed.compare_editions(
        [r for s in cz_sheets for r in s["offices"]],
        [r for s in de_sheets for r in s["offices"]],
        cz_names, de_names, cz_passes, de_passes, family=family_of, draws=draws,
        shared=lxv.vocabulary(abbreviations("cs")) & lxv.vocabulary(abbreviations("de")))
    head = [f"  Czech gold scans {', '.join(sorted(s['scan'] for s in cz_sheets))}; "
            f"Czech r5 passes on disk: {len(cz_passes)}; only the symbols both "
            f"plates print are compared"]
    return head + [lxed.lexikon_1910_editions_report(result)], result["rows"]


def write_csv(path: Path, rows: list[dict]) -> None:
    if not rows:
        return
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8", newline="") as fh:
        writer = csv.DictWriter(fh, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)


def main(argv=None) -> int:
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")
        except (AttributeError, ValueError):
            pass
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--no-compare", action="store_true",
                    help="skip the CZ–DE independence test")
    ap.add_argument("--draws", type=int, default=lxed.DRAWS,
                    help="permutation draws for the independence test")
    args = ap.parse_args(argv)

    de_paths = sheet_paths(bo.DE_GOLD_DIR)
    de_sheets = read_gold(de_paths, "de")
    de_passes = bo.de_passes_on_disk(hosts="all")
    lines = [f"score_r5_german_notes v{__version__} — {time.strftime('%Y-%m-%d %H:%M')} "
             f"— run {bo.RUN_TAG}", ""]
    german, rows = score_german(de_sheets, de_passes)
    lines += german
    pair_rows: list[dict] = []
    if not args.no_compare:
        compared, pair_rows = compare(de_sheets, de_paths, de_passes, args.draws)
        lines += ["", ""] + compared
    report = "\n".join(lines)
    print(report)
    if rows or pair_rows:
        write_csv(OUT_CSV, rows)
        write_csv(OUT_PAIRS, pair_rows)
        OUT_REPORT.parent.mkdir(parents=True, exist_ok=True)
        OUT_REPORT.write_text(report + "\n", encoding="utf-8")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
