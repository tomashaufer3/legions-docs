How it works

Everything lexikon_1910 and lexikon_1910_notes know about where things sit on a page comes from the NDK ALTO: the engine's rules (GraphicalElement) and its word boxes. Only the Czech volume has one. German Bohemia, Moravia and Silesia are scans with at most a plain or position-less text layer, and a crown-land volume not yet acquired may have nothing at all. This module reads the same geometry off the pixels, so that a volume can be cut into crops without any OCR:

  • the frame and the table's column rules, from long vertical runs of ink;
  • the header row between the table's top rule and the double rule under it, and the column schema it implies: how many columns, and which of them a spanning label groups together (a horizontal rule inside the header);
  • the table foot, where the interior column rules stop;
  • the apparatus divider — the short central rule below the table — and the two columns it opens;
  • the footer band below the divider's foot, down to the last printed line;
  • section headings inside each apparatus column. A heading is a line set centred in its column with white on both sides and a blank line above, whereas an entry opens at the column's left edge and a wrapped entry line at the hanging indent. Found by projection: no character is read, so a heading the OCR loses (scans 0090 0141 0190 0241 0304 0384 0456 of the Czech volume) is still a heading here;
  • blank rows, where a strip cut can fall without passing through ink.

Opt-in only. Nothing in the existing pipeline calls this module, and no default anywhere changes because it exists. It is measured against the ALTO geometry on the Czech volume (utils/lexikon_1910/_archive/measure_image_geometry.py) and its crops are checked for lost print by ink accounting (utils/lexikon_1910/_archive/measure_ink_accounting.py). Making it the default is a separate decision, gated on a re-read (see the hand-over).

All pixel thresholds are written at the scale of the Czech render — the ALTO frame, 2,928 px wide — and multiplied by PageGeometry.scale, which is the ratio of this page's frame (or, before the frame is found, its width) to that reference. A 300-dpi German scan therefore uses the same rules as the 399-dpi Czech one.

Only numpy is required.

    column_lines: dict[str, list[Line]] = field(default_factory=dict)

    rules: list[Rule] = field(default_factory=list)
    notes: list[str] = field(default_factory=list)

    def n_columns(self) -> int:
        return max(0, len(self.columns) - 1)

    @property
    def band_top(self) -> int | None:
        return self.table_foot if self.divider is not None else None

    def apparatus(self) -> dict[str, tuple[int, int, int, int]]:
        """The two apparatus columns, or ``{}`` on a page with no divider."""
        if self.divider is None or self.frame is None or self.table_foot is None:
            return {}

    def table_box(self) -> tuple[int, int, int, int] | None:
        """Header and body: frame top to the table foot, frame left to right."""
        if self.frame is None or self.table_top is None or self.table_foot is None:
            return None
        return (self.frame[0], self.table_top, self.frame[1], self.table_foot)

def ink_threshold(grey, share: float = INK_SHARE) -> int:
    """The grey level below which a pixel is ink on this page (see ``INK_SHARE``)."""
    import numpy as np

    h, w = grey.shape
    centre = grey[h // 10: 9 * h // 10, w // 10: 9 * w // 10]
    paper = float(np.median(centre))
    ink = float(np.percentile(centre, 0.5))
    # A blank page has no ink to measure: keep the cut well below the paper.
    return int(min(ink + share * (paper - ink), paper - 25))

def sheet_box(grey, threshold: int) -> tuple[int, int, int, int]:
    """The scanned page without the scanner's dark border.

    The Silesian scans carry a black margin and sometimes a strip of the facing
    page; a column or row of the image that is dark over most of its length is
    border, never print, and is trimmed from each edge inward.
    """
    import numpy as np

    dark = grey < threshold
    h, w = grey.shape

    def trim(share_along) -> tuple[int, int]:
        n = len(share_along)
        lo, hi = 0, n
        while lo < n // 4 and share_along[lo] > 0.5:
            lo += 1
        while hi > 3 * n // 4 and share_along[hi - 1] > 0.5:
            hi -= 1
        return lo, hi

    left, right = trim(dark.mean(axis=0))
    top, bottom = trim(dark.mean(axis=1))
    return int(left), int(top), int(right), int(bottom)

    h, w = shape

def ink_bands(rows, row_min: int = ROW_MIN_DARK, hole: int = BAND_HOLE):
    """``(first, last, dark)`` for each run of rows carrying ink (cf. msf)."""
    bands: list[list[int]] = []
    for index, dark in enumerate(rows):
        if dark < row_min:
            continue
        if bands and index - bands[-1][1] <= hole + 1:
            bands[-1][1] = index
            bands[-1][2] += int(dark)
        else:
            bands.append([index, index, int(dark)])
    return [tuple(b) for b in bands]

def lines_in(ink, box: tuple[int, int, int, int], scale: float,
             min_rows: int | None = None, min_dark: int | None = None) -> list[Line]:
    """The printed lines inside ``box`` of a rules-free ink mask."""
    import numpy as np

    left, top, right, bottom = (int(v) for v in box)
    if bottom - top < 2 or right - left < 2:
        return []
    window = ink[top:bottom, left:right]
    rows = window.sum(axis=1)
    min_rows = int(LINE_MIN_ROWS * scale) if min_rows is None else min_rows
    min_dark = int(LINE_MIN_DARK * scale * scale) if min_dark is None else min_dark
    out: list[Line] = []
    for first, last, dark in ink_bands(rows, hole=max(2, int(BAND_HOLE * scale))):
        if last - first + 1 < min_rows or dark < min_dark:
            continue
        cols = np.nonzero(window[first:last + 1].any(axis=0))[0]
        if dark < LINE_MIN_DENSITY * (last - first + 1) * (cols[-1] - cols[0] + 1):
            continue
        weight = window[first:last + 1].sum(axis=0).cumsum()
        core = np.searchsorted(weight, [0.01 * weight[-1], 0.99 * weight[-1]])
        # Runs of ink along the band's darkest row: type breaks into glyphs
        # there, a blot or the book's edge does not. (Along the whole band's
        # columns instead, a paragraph merged into one band reads as one run.)
        row = window[first + int(window[first:last + 1].sum(axis=1).argmax())]
        xs = np.nonzero(row)[0]
        pieces = 1 + int((np.diff(xs) > 2).sum()) if len(xs) else 0
        out.append(Line(top + first, top + last, left + int(cols[0]),
                        left + int(cols[-1]), int(dark),
                        left + int(core[0]), left + int(core[1]), pieces))
    return out

def find_headings(lines: list[Line], box: tuple[int, int, int, int],
                  region: str, scale: float = 1.0, full_width: bool = False) -> list[Heading]:
    """The lines of one apparatus column that are set as section headings.

    ``box`` is the column between its rules (inset off them); see
    ``HEADING_MARGIN`` for the rule and how it was fitted, and ``HEADING_FULL_MARGIN``
    for the heading that fills its line (``full_width``, German editions).
    """
    left, top, right, _bottom = box
    width = right - left
    if width <= 0:
        return []
    found: list[Heading] = []
    one_line = HEADING_FULL_HEIGHT * scale
    for i, line in enumerate(lines):
        lw = (line.core_left - left) / width
        rw = (right - line.core_right) / width
        centred = (lw >= HEADING_MARGIN and rw >= HEADING_MARGIN
                   and abs(lw - rw) <= HEADING_BALANCE)
        filled = (full_width and min(lw, rw) >= HEADING_FULL_MARGIN
                  and abs(lw - rw) <= HEADING_FULL_BALANCE and line.height <= one_line)
        if not (centred or filled):
            continue
        white = line.top - (lines[i - 1].bottom if i else top)
        previous = found[-1] if found else None
        # A district name that wraps is printed as a second centred line
        # directly under the first; it is one heading.
        if previous and line.top - previous.bottom <= 1.2 * line.height:
            previous.bottom, previous.lines = line.bottom, previous.lines + 1
            previous.left = min(previous.left, line.left)
            previous.right = max(previous.right, line.right)
            continue
        if white < HEADING_WHITE_ABOVE * scale and min(lw, rw) < HEADING_WIDE:
            continue
        found.append(Heading(region, line.top, line.bottom, line.left, line.right))
    return found

def blank_cuts(ink, box: tuple[int, int, int, int], slices: int,
               min_gap: int = 4) -> list[int]:
    """Where to cut ``box`` into ``slices`` strips through rows with no ink at all.

    The image-only counterpart of ``lexikon_1910_notes.line_cuts``: there the
    whitespace is the gaps between word boxes, here it is rows of the box that
    carry not one ink pixel. A cut is the middle of the blank run nearest each
    even division, each run spent once; a box with no blank run returns fewer
    cuts rather than a cut through print.
    """
    import numpy as np

    left, top, right, bottom = (int(v) for v in box)
    if slices <= 1 or bottom - top < 40:
        return []
    blank = ~ink[top:bottom, left:right].any(axis=1)
    runs: list[tuple[int, int]] = []
    start = None
    for i, b in enumerate(np.append(blank, False)):
        if b and start is None:
            start = i
        elif not b and start is not None:
            if i - start >= min_gap:
                runs.append((top + start, top + i - 1))
            start = None
    step = (bottom - top) / slices
    cuts: list[int] = []
    for i in range(1, slices):
        if not runs:
            break
        target = top + step * i
        run = min(runs, key=lambda r: abs((r[0] + r[1]) / 2 - target))
        runs.remove(run)
        cut = (run[0] + run[1]) // 2
        if top < cut < bottom:
            cuts.append(cut)
    return sorted(set(cuts))

def detect(image, expected_columns: int | None = None,
           full_width_headings: bool = False) -> PageGeometry:
    """Read the geometry of one rendered page. See the module docstring.
    ``full_width_headings``: also take a heading that fills its line (German, D-132)."""
    grey = to_grey(image)
    height, width = grey.shape
    threshold = ink_threshold(grey)
    mask = grey < threshold
    sheet = sheet_box(grey, threshold)
    _clip(mask, sheet)
    geo = PageGeometry(width=width, height=height, scale=width / REFERENCE_WIDTH,
                       threshold=threshold, sheet=sheet)

    # Rules are looked for at a lenient cut: the UBW scans print them faint
    # (column grey ~147 against paper ~178 on de 0200, where type is ~100), so at
    # the type's cut they break into pieces shorter than any rule. A glyph stays
    # short however faint its halo. A line of bold type can join into a long
    # horizontal run at this cut (si 0040), which is why a horizontal rule is only
    # used where it is frame-wide or ends on two column rules.
    faint = grey < ink_threshold(grey, RULE_SHARE)
    _clip(faint, sheet)
    pieces = vertical_rules(faint, geo.scale)
    wide = horizontal_rules(faint, geo.scale, int(0.5 * width))
    frame = _frame(pieces, wide, geo)
    if frame is not None:
        geo.frame = frame
        geo.scale = (frame[1] - frame[0]) / REFERENCE_FRAME
    s = geo.scale

    structural += wide

crown-land volume not yet acquired may have nothing at all. This module reads
the same geometry off the pixels, so that a volume can be cut into crops without
any OCR:

* the **frame** and the table's **column rules**, from long vertical runs of ink;
* the **header row** between the table's top rule and the double rule under it,
  and the **column schema** it implies: how many columns, and which of them a
  spanning label groups together (a horizontal rule inside the header);
* the **table foot**, where the interior column rules stop;
* the **apparatus divider** — the short central rule below the table — and the
  two columns it opens;
* the **footer band** below the divider's foot, down to the last printed line;
* **section headings** inside each apparatus column. A heading is a line set
  centred in its column with white on both sides and a blank line above, whereas
  an entry opens at the column's left edge and a wrapped entry line at the
  hanging indent. Found by projection: no character is read, so a heading the
  OCR loses (scans 0090 0141 0190 0241 0304 0384 0456 of the Czech volume) is
  still a heading here;
* **blank rows**, where a strip cut can fall without passing through ink.

**Opt-in only.** Nothing in the existing pipeline calls this module, and no
default anywhere changes because it exists. It is measured against the ALTO
geometry on the Czech volume (``utils/lexikon_1910/_archive/measure_image_geometry.py``)
and its crops are checked for lost print by ink accounting
(``utils/lexikon_1910/_archive/measure_ink_accounting.py``). Making it the default is a
separate decision, gated on a re-read (see the hand-over).

All pixel thresholds are written at the scale of the Czech render — the ALTO
frame, 2,928 px wide — and multiplied by :attr:`PageGeometry.scale`, which is the
ratio of this page's frame (or, before the frame is found, its width) to that
reference. A 300-dpi German scan therefore uses the same rules as the 399-dpi
Czech one.

Only numpy is required.
"""

from __future__ import annotations

from dataclasses import dataclass, field

__version__ = "0.2.0"

# --------------------------------------------------------------------------- #
# Constants (Czech-render pixels; multiplied by the page scale)
# --------------------------------------------------------------------------- #

#: Width of the Czech ALTO render, the frame the constants below are read in.
REFERENCE_WIDTH = 2928

#: The table frame's width on that render (left frame rule to right frame rule):
#: 2,651-2,658 px on the Czech development scans 0090 0118 0141 0190 0270 0456.
#: Used to turn a detected frame into a scale, which is steadier than the page
#: width: the German scans are cropped anywhere from 511 to 541 pt wide, the
#: frame never varies.
REFERENCE_FRAME = 2655

#: A pixel is ink when it is darker than ``ink + INK_SHARE * (paper - ink)``,
#: where ``paper`` is the page's median grey and ``ink`` its 0.5th percentile.
#: Relative, not absolute: the UBW German scans print a stroke at grey 100-130
#: where the NDK Czech scan prints it at 75-90, and an absolute cut that keeps
#: the Czech type (95) loses most of the German. At 0.5 neither shows verso
#: show-through (checked by eye on cz 0270, de 0200, mo 0100).
INK_SHARE = 0.50

#: The cut used to find rules, on the same scale (see :func:`detect`).
RULE_SHARE = 0.75

#: A vertical rule is a run of ink at least this long, after bridging gaps of up
#: to ``RULE_BRIDGE`` rows and allowing ``RULE_DRIFT`` px of sideways drift. The
#: tallest glyph stroke in the register (a bold capital in the running head) is
#: under 70 px; the shortest rule that matters, a header sub-division, ~90.
RULE_MIN = 120
RULE_BRIDGE = 6
RULE_DRIFT = 2

#: Rule pieces at the same x are one rule when the gap between them is at most
#: this. Kept short on purpose: the apparatus divider often stands right under
#: one of the table's column rules with only the heading band between them, and
#: joining the two is exactly the ALTO failure behind
#: ``lexikon_1910_notes.fallback_divider``.
RULE_JOIN = 25

#: Rules closer than this are one printed boundary (the register prints double
#: rules between column groups).
COLUMN_MERGE = 30

#: Ink rows: a row of a region carries ink with at least ``ROW_MIN_DARK`` dark
#: pixels; ink rows join into one band across holes of up to ``BAND_HOLE`` rows.
#: Same numbers as ``measure_segment_foot.py``, which read them off scans 0076
#: and 0316.
ROW_MIN_DARK = 2
BAND_HOLE = 8

#: A band is a printed line when it is at least ``LINE_MIN_ROWS`` tall and holds
#: ``LINE_MIN_DARK`` dark pixels (at reference scale; area scales with scale²).
LINE_MIN_ROWS = 12
LINE_MIN_DARK = 70

#: And its ink is not spread thin: dark pixels per pixel of the band's box. The
#: shortest printed line (a key and one pictogram, ~114 px of ink in ~25 x 150)
#: is ~0.03; a faint smear of show-through 300 rows tall (de 0030, under the
#: footer) is ~0.003.
LINE_MIN_DENSITY = 0.01

#: A heading line is set centred in its column: white on both sides of at least
#: ``HEADING_MARGIN`` of the column box (rule to rule), the two within
#: ``HEADING_BALANCE`` of each other, and ``HEADING_WHITE_ABOVE`` px of white
#: above it (a heading follows a blank line or opens the column) unless both
#: margins reach ``HEADING_WIDE``. An entry opens at the column's left edge and a
#: wrapped entry line at the hanging indent, so neither is balanced; a part line
#: (``2) Rajsko …``) can be, but sits directly under the line before it.
#:
#: **Fitted on the odd-numbered Czech table scans** (2026-10-14; the even ones,
#: and with them the sealed certification scans, held out) against the headings
#: the ALTO text layer finds: 417 of 435 found, and the 11 lines taken that the
#: ALTO has no heading at are, by eye, 11 headings the text layer dropped (0079
#: 0105 0123 0141 ×2 0147 0221 0241 0257 0317 0345). Measured against the box,
#: not the text block, because a text block's edges are not measurable on a
#: column of a few short lines (cz 0021).
HEADING_MARGIN = 0.08
HEADING_BALANCE = 0.05
HEADING_WHITE_ABOVE = 25
HEADING_WIDE = 0.15

#: **A heading set across the whole column** (German editions only, 0.2.0, D-132). A long
#: court name fills the line — *Ämter, Anstalten usw. im Gerichtsbezirk Reichenau an der
#: Kněžna* on de 0380 stands at 0.053 / 0.052 of the column, under ``HEADING_MARGIN`` —
#: and the section it opened was lost (28 of the German gold's 61 missed claims, D-128).
#: Taken when its two margins are at least ``HEADING_FULL_MARGIN`` and within
#: ``HEADING_FULL_BALANCE`` of each other, with the white above every heading needs. An
#: entry line opens at the key, 0.033–0.049 of the column on the German gold, so none
#: qualifies. **Fitted on the seven German gold pages** (the four sealed German pages not
#: opened): every centred single line with white above was a heading, and 0380's is the
#: only one the old rule missed. Off for the Czech volume, whose route is certified on
#: the old rule (D-127). It must also be **one printed line** — at most
#: ``HEADING_FULL_HEIGHT`` px at the reference width — or a justified block of entries
#: merged into one band qualifies too (de 0478: a three-line block, margins 0.052 / 0.043,
#: joined the heading above it and swallowed the one below). On the German gold a single
#: line is 21–31 px at scale 0.756 and a merged block 54 px or more; a column's median
#: will not do, since a column can hold more blocks than single lines.
HEADING_FULL_MARGIN = 0.04
HEADING_FULL_BALANCE = 0.02
HEADING_FULL_HEIGHT = 45

#: Where in the frame the apparatus divider stands, as a share of frame width.
DIVIDER_ZONE = (0.38, 0.62)
DIVIDER_MIN = 200

@dataclass
class Rule:
    """A printed rule: ``x0..x1`` × ``y0..y1`` in page pixels (inclusive)."""

    x0: int
    y0: int
    x1: int
    y1: int

    @property
    def x(self) -> int:
        return (self.x0 + self.x1) // 2

    @property
    def y(self) -> int:
        return (self.y0 + self.y1) // 2

    @property
    def height(self) -> int:
        return self.y1 - self.y0 + 1

    @property
    def width(self) -> int:
        return self.x1 - self.x0 + 1


@dataclass
class Line:
    """One printed line inside a region: its rows and its ink's left/right."""

    top: int
    bottom: int
    left: int
    right: int
    dark: int
    #: Where the bulk of the line's ink starts and ends (1st and 99th
    #: percentile of its dark pixels' x): one speck beside a heading (cz 0270,
    #: 140 px right of *Části osad … Most:*) must not unbalance its margins.
    core_left: int = 0
    core_right: int = 0
    #: Separate runs of ink along the line's darkest row (glyphs split by
    #: blank px). Print has many; a blot or a pencil mark has one or two (cz 0026
    #: carries a blot under its apparatus that is as dark as a line of type).
    pieces: int = 0

    @property
    def height(self) -> int:
        return self.bottom - self.top + 1


@dataclass
class Heading:
    """A line the projection takes for a section heading."""

    region: str            # "left" | "right"
    top: int
    bottom: int
    left: int
    right: int
    lines: int = 1         # a long district name can wrap onto a second line


@dataclass
class Segment:
    """One section's run within one apparatus column — image-only.

    The image says *where* a section begins; it cannot say which kind it is or
    which district it names. Those stay empty here and are the reader's to
    report (design: stage 4 of the hand-over).
    """

    region: str
    box: tuple[int, int, int, int]
    order: int
    carried: bool          # no heading inside: continues the previous section


@dataclass
class PageGeometry:
    """Everything this module finds on one page. Boxes are ``(l, t, r, b)``."""

    width: int
    height: int
    scale: float
    threshold: int
    kind: str = "blank"                     # table | prose | blank
    sheet: tuple[int, int, int, int] = (0, 0, 0, 0)
    frame: tuple[int, int] | None = None    # left and right frame rule x
    columns: list[int] = field(default_factory=list)
    table_top: int | None = None
    header_bottom: int | None = None
    header_spans: list[tuple[int, int, int]] = field(default_factory=list)
    table_foot: int | None = None
    divider: Rule | None = None
    footer: tuple[int, int, int, int] | None = None
    below: tuple[int, int, int, int] | None = None   # print under a table, no divider
    prose: tuple[int, int, int, int] | None = None
    head: tuple[int, int, int, int] | None = None
    #: Where each apparatus column ends: the divider's foot, or lower when the
    #: column's last line starts above the foot and ends below it (cz 0494).
    column_foot: dict[str, int] = field(default_factory=dict)
    column_lines: dict[str, list[Line]] = field(default_factory=dict)
    headings: list[Heading] = field(default_factory=list)
    rules: list[Rule] = field(default_factory=list)
    notes: list[str] = field(default_factory=list)

    @property
    def n_columns(self) -> int:
        return max(0, len(self.columns) - 1)

    @property
    def band_top(self) -> int | None:
        return self.table_foot if self.divider is not None else None

    def apparatus(self) -> dict[str, tuple[int, int, int, int]]:
        """The two apparatus columns, or ``{}`` on a page with no divider."""
        if self.divider is None or self.frame is None or self.table_foot is None:
            return {}
        left, right = self.frame
        split, foot = self.divider.x, self.divider.y1
        return {"left": (left, self.table_foot, split, self.column_foot.get("left", foot)),
                "right": (split, self.table_foot, right, self.column_foot.get("right", foot))}

    def table_box(self) -> tuple[int, int, int, int] | None:
        """Header and body: frame top to the table foot, frame left to right."""
        if self.frame is None or self.table_top is None or self.table_foot is None:
            return None
        return (self.frame[0], self.table_top, self.frame[1], self.table_foot)

def to_grey(image):
    """A numpy uint8 array from a Pillow image or an array."""
    import numpy as np

    if hasattr(image, "convert"):
        return np.asarray(image.convert("L"))
    return np.asarray(image, dtype=np.uint8)


def ink_threshold(grey, share: float = INK_SHARE) -> int:
    """The grey level below which a pixel is ink on this page (see ``INK_SHARE``)."""
    import numpy as np

    h, w = grey.shape
    centre = grey[h // 10: 9 * h // 10, w // 10: 9 * w // 10]
    paper = float(np.median(centre))
    ink = float(np.percentile(centre, 0.5))
    # A blank page has no ink to measure: keep the cut well below the paper.
    return int(min(ink + share * (paper - ink), paper - 25))


def sheet_box(grey, threshold: int) -> tuple[int, int, int, int]:
    """The scanned page without the scanner's dark border.

    The Silesian scans carry a black margin and sometimes a strip of the facing
    page; a column or row of the image that is dark over most of its length is
    border, never print, and is trimmed from each edge inward.
    """
    import numpy as np

    dark = grey < threshold
    h, w = grey.shape

    def trim(share_along) -> tuple[int, int]:
        n = len(share_along)
        lo, hi = 0, n
        while lo < n // 4 and share_along[lo] > 0.5:
            lo += 1
        while hi > 3 * n // 4 and share_along[hi - 1] > 0.5:
            hi -= 1
        return lo, hi

    left, right = trim(dark.mean(axis=0))
    top, bottom = trim(dark.mean(axis=1))
    return int(left), int(top), int(right), int(bottom)

def _runs_along_rows(mask, min_len: int, bridge: int, drift: int):
    """``(x, y0, y1)`` for every vertical run of ink at least ``min_len`` long."""
    import numpy as np

    m = mask.copy()
    for d in range(1, drift + 1):           # sideways drift of a skewed rule
        m[:, d:] |= mask[:, :-d]
        m[:, :-d] |= mask[:, d:]
    if bridge > 0:                           # close small vertical gaps
        c = np.zeros((m.shape[0] + 1, m.shape[1]), dtype=np.int32)
        np.cumsum(m, axis=0, out=c[1:])
        h = m.shape[0]
        idx = np.arange(h)
        above = c[idx + 1] - c[np.maximum(idx + 1 - bridge, 0)]
        below = c[np.minimum(idx + bridge, h)] - c[idx]
        m = m | ((above > 0) & (below > 0))
    padded = np.zeros((m.shape[0] + 2, m.shape[1]), dtype=np.int8)
    padded[1:-1] = m
    diff = np.diff(padded, axis=0)
    xs_s, ys_s = np.nonzero(diff.T == 1)
    xs_e, ys_e = np.nonzero(diff.T == -1)
    length = ys_e - ys_s
    keep = length >= min_len
    return list(zip(xs_s[keep].tolist(), ys_s[keep].tolist(),
                    (ys_e[keep] - 1).tolist()))


def _group(runs, join: int, merge_x: int) -> list[Rule]:
    """Runs → rules: chain runs at neighbouring x into one stroke, then join that
    stroke's pieces along its length across gaps of up to ``join``.

    A frame rule is printed thick-and-thin (12 + 3 px on the Czech scan) and a
    group boundary as a double rule; both are one boundary, so neighbouring x
    positions chain into one stroke whatever its width.
    """
    clusters: list[list[tuple[int, int, int]]] = []
    last_x = None
    for run in sorted(runs):
        if clusters and run[0] - last_x <= merge_x:
            clusters[-1].append(run)
        else:
            clusters.append([run])
        last_x = run[0]
    rules: list[Rule] = []
    for cluster in clusters:
        current = None
        for x, y0, y1 in sorted(cluster, key=lambda r: r[1]):
            if current is not None and y0 <= current.y1 + join:
                current.y1 = max(current.y1, y1)
                current.x0, current.x1 = min(current.x0, x), max(current.x1, x)
            else:
                current = Rule(x, y0, x, y1)
                rules.append(current)
    return rules


def vertical_rules(mask, scale: float) -> list[Rule]:
    """Every vertical rule on the page."""
    runs = _runs_along_rows(mask, max(40, int(RULE_MIN * scale)),
                            int(RULE_BRIDGE * scale), RULE_DRIFT)
    rules = _group(runs, int(RULE_JOIN * scale), RULE_DRIFT * 2 + 2)
    # A rule is long and thin; a long *and* wide block of ink is not one.
    return [r for r in rules if r.height >= 4 * r.width]


def horizontal_rules(mask, scale: float, min_len: int) -> list[Rule]:
    """Every horizontal rule at least ``min_len`` long."""
    runs = _runs_along_rows(mask.T, min_len, int(RULE_BRIDGE * scale), RULE_DRIFT)
    rules = _group(runs, int(RULE_JOIN * scale), RULE_DRIFT * 2 + 2)
    return [Rule(r.y0, r.x0, r.y1, r.x1) for r in rules if r.height >= 4 * r.width]


def rule_mask(shape, rules: list[Rule], pad: int):
    """A boolean mask covering ``rules`` grown by ``pad`` px."""
    import numpy as np

    out = np.zeros(shape, dtype=bool)
    h, w = shape
    for r in rules:
        out[max(0, r.y0 - pad):min(h, r.y1 + pad + 1),
            max(0, r.x0 - pad):min(w, r.x1 + pad + 1)] = True
    return out

def ink_bands(rows, row_min: int = ROW_MIN_DARK, hole: int = BAND_HOLE):
    """``(first, last, dark)`` for each run of rows carrying ink (cf. msf)."""
    bands: list[list[int]] = []
    for index, dark in enumerate(rows):
        if dark < row_min:
            continue
        if bands and index - bands[-1][1] <= hole + 1:
            bands[-1][1] = index
            bands[-1][2] += int(dark)
        else:
            bands.append([index, index, int(dark)])
    return [tuple(b) for b in bands]


def lines_in(ink, box: tuple[int, int, int, int], scale: float,
             min_rows: int | None = None, min_dark: int | None = None) -> list[Line]:
    """The printed lines inside ``box`` of a rules-free ink mask."""
    import numpy as np

    left, top, right, bottom = (int(v) for v in box)
    if bottom - top < 2 or right - left < 2:
        return []
    window = ink[top:bottom, left:right]
    rows = window.sum(axis=1)
    min_rows = int(LINE_MIN_ROWS * scale) if min_rows is None else min_rows
    min_dark = int(LINE_MIN_DARK * scale * scale) if min_dark is None else min_dark
    out: list[Line] = []
    for first, last, dark in ink_bands(rows, hole=max(2, int(BAND_HOLE * scale))):
        if last - first + 1 < min_rows or dark < min_dark:
            continue
        cols = np.nonzero(window[first:last + 1].any(axis=0))[0]
        if dark < LINE_MIN_DENSITY * (last - first + 1) * (cols[-1] - cols[0] + 1):
            continue
        weight = window[first:last + 1].sum(axis=0).cumsum()
        core = np.searchsorted(weight, [0.01 * weight[-1], 0.99 * weight[-1]])
        # Runs of ink along the band's darkest row: type breaks into glyphs
        # there, a blot or the book's edge does not. (Along the whole band's
        # columns instead, a paragraph merged into one band reads as one run.)
        row = window[first + int(window[first:last + 1].sum(axis=1).argmax())]
        xs = np.nonzero(row)[0]
        pieces = 1 + int((np.diff(xs) > 2).sum()) if len(xs) else 0
        out.append(Line(top + first, top + last, left + int(cols[0]),
                        left + int(cols[-1]), int(dark),
                        left + int(core[0]), left + int(core[1]), pieces))
    return out


def find_headings(lines: list[Line], box: tuple[int, int, int, int],
                  region: str, scale: float = 1.0, full_width: bool = False) -> list[Heading]:
    """The lines of one apparatus column that are set as section headings.

    ``box`` is the column between its rules (inset off them); see
    ``HEADING_MARGIN`` for the rule and how it was fitted, and ``HEADING_FULL_MARGIN``
    for the heading that fills its line (``full_width``, German editions).
    """
    left, top, right, _bottom = box
    width = right - left
    if width <= 0:
        return []
    found: list[Heading] = []
    one_line = HEADING_FULL_HEIGHT * scale
    for i, line in enumerate(lines):
        lw = (line.core_left - left) / width
        rw = (right - line.core_right) / width
        centred = (lw >= HEADING_MARGIN and rw >= HEADING_MARGIN
                   and abs(lw - rw) <= HEADING_BALANCE)
        filled = (full_width and min(lw, rw) >= HEADING_FULL_MARGIN
                  and abs(lw - rw) <= HEADING_FULL_BALANCE and line.height <= one_line)
        if not (centred or filled):
            continue
        white = line.top - (lines[i - 1].bottom if i else top)
        previous = found[-1] if found else None
        # A district name that wraps is printed as a second centred line
        # directly under the first; it is one heading.
        if previous and line.top - previous.bottom <= 1.2 * line.height:
            previous.bottom, previous.lines = line.bottom, previous.lines + 1
            previous.left = min(previous.left, line.left)
            previous.right = max(previous.right, line.right)
            continue
        if white < HEADING_WHITE_ABOVE * scale and min(lw, rw) < HEADING_WIDE:
            continue
        found.append(Heading(region, line.top, line.bottom, line.left, line.right))
    return found


def blank_cuts(ink, box: tuple[int, int, int, int], slices: int,
               min_gap: int = 4) -> list[int]:
    """Where to cut ``box`` into ``slices`` strips through rows with no ink at all.

    The image-only counterpart of ``lexikon_1910_notes.line_cuts``: there the
    whitespace is the gaps between word boxes, here it is rows of the box that
    carry not one ink pixel. A cut is the middle of the blank run nearest each
    even division, each run spent once; a box with no blank run returns fewer
    cuts rather than a cut through print.
    """
    import numpy as np

    left, top, right, bottom = (int(v) for v in box)
    if slices <= 1 or bottom - top < 40:
        return []
    blank = ~ink[top:bottom, left:right].any(axis=1)
    runs: list[tuple[int, int]] = []
    start = None
    for i, b in enumerate(np.append(blank, False)):
        if b and start is None:
            start = i
        elif not b and start is not None:
            if i - start >= min_gap:
                runs.append((top + start, top + i - 1))
            start = None
    step = (bottom - top) / slices
    cuts: list[int] = []
    for i in range(1, slices):
        if not runs:
            break
        target = top + step * i
        run = min(runs, key=lambda r: abs((r[0] + r[1]) / 2 - target))
        runs.remove(run)
        cut = (run[0] + run[1]) // 2
        if top < cut < bottom:
            cuts.append(cut)
    return sorted(set(cuts))

def _clip(mask, sheet) -> None:
    mask[:, :sheet[0]] = False
    mask[:, sheet[2]:] = False
    mask[:sheet[1], :] = False
    mask[sheet[3]:, :] = False


def _frame(pieces: list[Rule], wide: list[Rule], geo: PageGeometry):
    """``(left, right)`` x of the table frame, or None.

    Anchored on the frame-wide rule under the header: it is the widest rule in
    the upper part of the sheet and its two ends meet the frame's two sides. A
    tall rule at each end confirms it; that is what a page edge, a strip of the
    facing page or the scanner bed (all of which the Silesian scans show) never
    has. Without such a rule the frame is the pair of tall rules enclosing the
    most others.
    """
    s = geo.scale
    tall = [r for r in pieces if r.height >= 0.15 * geo.height]
    reach = 0.05 * geo.width

    def side(x: int, y: int, outward: int):
        # The outermost tall rule at this end of the frame-wide rule. That rule
        # is often broken short of the frame (si 0020 stops at the first column
        # rule), so a side up to ``reach`` beyond its end is taken too — unless
        # it runs off the top of the image, which the frame (below the running
        # head) never does and the book's edge or the facing page does (si 0020).
        # Close to the end, running off the top is allowed: there the frame's
        # side has merged with the edge's shadow (mo 0030).
        near = [r for r in tall if r.y0 - 60 * s <= y <= r.y1 + 60 * s
                and -40 * s <= (r.x - x) * outward <= reach
                and (r.y0 > geo.sheet[1] + 3 or (r.x - x) * outward <= 40 * s)]
        return max(near, key=lambda r: r.x * outward) if near else None

    best = None
    for h in sorted(wide, key=lambda h: -(h.x1 - h.x0)):
        if h.y > 0.45 * geo.height:
            continue
        a, b = side(h.x0, h.y, -1), side(h.x1, h.y, +1)
        if a is not None and b is not None and b.x - a.x >= 0.5 * geo.width:
            best = (a.x, b.x)
            break
    if best is not None:
        return best
    pair, n = None, -1
    for a in tall:
        for b in tall:
            span = b.x - a.x
            if not 0.55 * geo.width <= span <= 0.98 * geo.width:
                continue
            inside = sum(1 for r in tall if a.x + 20 * s < r.x < b.x - 20 * s)
            if inside > n or (inside == n and pair and span < pair[1] - pair[0]):
                pair, n = (a.x, b.x), inside
    return pair if n >= 3 else None


def _merge_positions(xs: list[int], within: int) -> list[int]:
    out: list[int] = []
    for x in sorted(xs):
        if out and x - out[-1] < within:
            out[-1] = (out[-1] + x) // 2
        else:
            out.append(x)
    return out


def _clusters(pieces: list[Rule], within: int) -> list[list[Rule]]:
    """Rule pieces grouped by x: one group per printed boundary."""
    groups: list[list[Rule]] = []
    for r in sorted(pieces, key=lambda r: r.x):
        if groups and r.x - groups[-1][-1].x < within:
            groups[-1].append(r)
        else:
            groups.append([r])
    return groups


def _coverage(group: list[Rule], top: int, bottom: int) -> float:
    """Share of ``top..bottom`` that the group's pieces cover."""
    if bottom <= top:
        return 0.0
    spans = sorted((max(top, r.y0), min(bottom, r.y1)) for r in group
                   if r.y1 > top and r.y0 < bottom)
    covered, end = 0, top
    for a, b in spans:
        a = max(a, end)
        if b > a:
            covered += b - a
            end = b
    return covered / (bottom - top)


def detect(image, expected_columns: int | None = None,
           full_width_headings: bool = False) -> PageGeometry:
    """Read the geometry of one rendered page. See the module docstring.
    ``full_width_headings``: also take a heading that fills its line (German, D-132)."""
    grey = to_grey(image)
    height, width = grey.shape
    threshold = ink_threshold(grey)
    mask = grey < threshold
    sheet = sheet_box(grey, threshold)
    _clip(mask, sheet)
    geo = PageGeometry(width=width, height=height, scale=width / REFERENCE_WIDTH,
                       threshold=threshold, sheet=sheet)

    # Rules are looked for at a lenient cut: the UBW scans print them faint
    # (column grey ~147 against paper ~178 on de 0200, where type is ~100), so at
    # the type's cut they break into pieces shorter than any rule. A glyph stays
    # short however faint its halo. A line of bold type can join into a long
    # horizontal run at this cut (si 0040), which is why a horizontal rule is only
    # used where it is frame-wide or ends on two column rules.
    faint = grey < ink_threshold(grey, RULE_SHARE)
    _clip(faint, sheet)
    pieces = vertical_rules(faint, geo.scale)
    wide = horizontal_rules(faint, geo.scale, int(0.5 * width))
    frame = _frame(pieces, wide, geo)
    if frame is not None:
        geo.frame = frame
        geo.scale = (frame[1] - frame[0]) / REFERENCE_FRAME
    s = geo.scale

    structural: list[Rule] = [r for r in pieces if r.height >= 0.2 * height]
    structural += wide
    if geo.frame is not None:
        horizontals = horizontal_rules(faint, s, int(120 * s))
        structural += _table(geo, pieces, horizontals, mask, expected_columns)
    # Rules come out of the ink only where they are the page's structure: a
    # stray vertical run inside a block of type (a stack of ``)`` at the lenient
    # cut) stays ink, so ink accounting cannot hide print behind it.
    if geo.frame is not None and geo.table_foot is not None:
        _divider(geo, pieces)
        if geo.divider is not None:
            structural.append(geo.divider)
    geo.rules = structural
    ink = mask & ~rule_mask(mask.shape, structural, max(2, int(3 * s)))

    if geo.frame is None or geo.table_top is None or geo.table_foot is None:
        geo.frame = None
        geo.columns, geo.table_top, geo.header_bottom, geo.table_foot = [], None, None, None
        _prose(geo, ink)
        return geo

    geo.kind = "table"
    if expected_columns is not None and geo.n_columns != expected_columns:
        geo.notes.append(f"columns {geo.n_columns} != expected {expected_columns}")
    _apparatus(geo, ink, full_width_headings)
    _head(geo, ink)
    return geo


def _table(geo: PageGeometry, pieces: list[Rule], horizontals: list[Rule],
           mask, expected: int | None = None) -> list[Rule]:
    """Header, columns and foot; returns the rules they are made of."""
    left, right = geo.frame
    s = geo.scale
    span = right - left
    used: list[Rule] = []
    sides = [r for r in pieces if (abs(r.x - left) <= 40 * s or abs(r.x - right) <= 40 * s)
             and r.height >= 0.1 * geo.height]
    used += sides
    inner_all = [r for r in pieces if left + 20 * s < r.x < right - 20 * s]
    wide = sorted((h for h in horizontals
                   if h.x0 <= left + 0.1 * span and h.x1 >= right - 0.1 * span
                   and h.y < 0.5 * geo.height), key=lambda h: h.y)
    groups: list[list[Rule]] = []
    for h in wide:                      # rules within ~15 px are one double rule
        if groups and h.y0 - groups[-1][-1].y1 <= 15 * s:
            groups[-1].append(h)
        else:
            groups.append([h])

    def crossing(group: list[Rule]) -> list[Rule]:
        y0, y1 = min(h.y0 for h in group), max(h.y1 for h in group)
        return [r for r in inner_all if r.y0 < y0 - 20 * s and r.y1 > y1 + 20 * s]

    # The header's foot is the frame-wide rule the column rules run *through*:
    # they start above it, in the header, and go on into the body. The table's
    # top rule is crossed by none of them (they start there), and a page edge or
    # the facing page's frame by none either.
    scored = [(len(crossing(g)), g) for g in groups]
    if not scored or max(n for n, _g in scored) < 3:
        geo.notes.append("no header rule")
        return used
    _n, header = max(scored, key=lambda t: (t[0], -min(h.y0 for h in t[1])))
    geo.header_bottom = max(h.y1 for h in header)
    top = min(r.y0 for r in crossing(header))
    # The top rule is printed thin and comes out broken on many scans (cz 0270:
    # two pieces of 235 px in 2,670), so it is consulted, never required.
    above = [min(h.y0 for h in g) for g in groups
             if g is not header and top - 40 * s <= min(h.y0 for h in g) < geo.header_bottom - 60 * s]
    geo.table_top = min([top] + above)
    used += [h for g in groups if g is header or min(h.y0 for h in g) in above for h in g]

    found = _columns(geo, pieces, mask)
    if found is None:
        return used
    # A broken frame side can leave the last column outside the frame. Widened
    # only while columns are missing against the volume's schema (or, with none
    # given, never across a rule that runs off the top of the scan).
    if expected is None or geo.n_columns < expected:
        before = geo.frame
        _widen_frame(geo, pieces, mask, expected is not None)
        if geo.frame != before:
            found = _columns(geo, pieces, mask) or found
    used += found
    geo.header_spans, spans = _header_spans(geo, horizontals)
    used += spans
    return used


def _columns(geo: PageGeometry, pieces: list[Rule], mask) -> list[Rule] | None:
    """Column rules and the table's foot inside the current frame; the rules
    used, or None when the frame holds no column rule.

    Each printed boundary is a group of pieces at one x; a faint rule comes out
    broken into several. The table's foot is where most of the groups end, and a
    group is a column rule when it covers most of the body.
    """
    left, right = geo.frame
    s = geo.scale
    inner = [r for r in pieces if left + 20 * s < r.x < right - 20 * s
             and r.y1 > geo.header_bottom + 100 * s]
    groups_x = _clusters(inner, int(COLUMN_MERGE * s))
    reach = geo.header_bottom + 0.6 * (geo.sheet[3] - geo.header_bottom)
    ends = []
    for g in groups_x:
        upper = [r for r in g if r.y0 <= reach]
        if upper and sum(r.height for r in upper) >= 0.2 * (reach - geo.header_bottom):
            ends.append(max(r.y1 for r in upper))
    if not ends:
        geo.notes.append("no column rules")
        return None
    window = 20 * s
    geo.table_foot = max(ends, key=lambda e: (sum(abs(e - o) <= window for o in ends), e))
    columns, used = [], []
    for g in groups_x:
        # A rule printed faint and broken (de 0217's last columns) covers less
        # of the body, but it still runs through the header's foot: a stack of
        # glyphs in the body never reaches up into the header.
        through = any(r.y0 < geo.header_bottom - 15 * s and r.y1 > geo.header_bottom + 15 * s
                      for r in g)
        cover = _coverage(g, geo.header_bottom, geo.table_foot)
        if cover >= 0.4 or (through and cover >= 0.15):
            columns.append(int(sum(r.x for r in g) / len(g)))
            used += [r for r in g if r.y0 < geo.table_foot]
    geo.columns = [left] + _merge_positions(columns, int(COLUMN_MERGE * s)) + [right]
    _drop_empty_edges(geo, mask)
    return used


def _widen_frame(geo: PageGeometry, pieces: list[Rule], mask,
                 schema_known: bool = False) -> None:
    """Move a frame side outward past a column it left out.

    Where the frame's side is printed broken, the frame-wide rule can end on a
    column rule instead and the last column falls outside the table (mo 0101 and
    de 0035, the station column). A rule further out that starts at the table's
    top, runs down into the body and has print between it and the side found
    is the side; the step repeats while there is one. The strip beside a true
    frame holds no print (si 0040) or the scanner's black bed, and neither
    moves it.
    """
    s = geo.scale
    left, right = geo.frame
    hb = geo.header_bottom
    body = (hb + int(10 * s), hb + int(0.4 * (geo.sheet[3] - hb)))
    reach = 0.25 * geo.width
    # A side may run off the top of the scan where it has merged with the edge
    # of the page (de 0035) — admitted only when the schema says a column is
    # missing, since the facing page's edge does the same (si 0020).
    starts = [r for r in pieces if r.y0 <= hb + 30 * s and r.y1 >= body[1]
              and (schema_known or r.y0 > geo.sheet[1] + 3)]

    def has_print(a: int, b: int) -> bool:
        # Not the two rules themselves: the strip is cut 12 px in from each.
        a, b = a + int(12 * s), b - int(12 * s)
        return b > a and _has_print(mask, (a, body[0], b, body[1]), s)

    for outward in (+1, -1):
        while True:
            side = right if outward > 0 else left
            beyond = sorted((r for r in starts if 40 * s < (r.x - side) * outward <= reach),
                            key=lambda r: abs(r.x - side))
            nxt = next((r for r in beyond[:1]
                        if has_print(*sorted((side, r.x)))), None)
            if nxt is None:
                break
            geo.notes.append(f"frame {'right' if outward > 0 else 'left'} side moved "
                             f"out to {nxt.x} past a column with print")
            if outward > 0:
                right = nxt.x
            else:
                left = nxt.x
    geo.frame = (left, right)


#: A column holds print: digits, names, or at least the nil dot on every row
#: (~1 % of a column's body). The strip between the frame and the shadow of the
#: book's edge, which can stand like a rule beside it (de 0217), holds none.
EMPTY_COLUMN = 0.003


def _has_print(mask, box, s: float) -> bool:
    """Whether a strip of the table holds print: some ink but not mostly ink
    (the scanner's black bed is mostly ink), as either an ink share of a column
    (``EMPTY_COLUMN``; a dense column's rows merge into one band, de 0035) or
    three printed lines (a sparse one, de 0172: 0.26 % ink)."""
    a, top, b, bottom = box
    share = mask[top:bottom, a:b].mean() if b > a and bottom > top else 0.0
    if share > 0.35:
        return False
    return share >= EMPTY_COLUMN or len(lines_in(mask, box, s)) >= 3


def _drop_empty_edges(geo: PageGeometry, mask) -> None:
    """Move the frame inward past an outer "column" with no print in it."""
    top, foot = geo.header_bottom + 10, geo.table_foot - 10
    if foot <= top:
        return

    def empty(a: int, b: int) -> bool:
        # No print in the body *and* none in the header cell. The station column
        # can be nearly empty in the body (two short lines on de 0172), but every
        # column carries a printed label in the header; the strip beside the
        # frame carries nothing.
        a, b = a + int(12 * geo.scale), b - int(12 * geo.scale)
        if b <= a:
            return False
        head = mask[geo.table_top + int(15 * geo.scale):geo.header_bottom - int(15 * geo.scale), a:b]
        return (not _has_print(mask, (a, top, b, foot), geo.scale)
                and (head.size == 0 or head.mean() < EMPTY_COLUMN))

    while len(geo.columns) > 3 and empty(geo.columns[0], geo.columns[1]):
        geo.columns.pop(0)
        geo.notes.append("outer left boundary held no print; frame moved in")
    while len(geo.columns) > 3 and empty(geo.columns[-2], geo.columns[-1]):
        geo.columns.pop()
        geo.notes.append("outer right boundary held no print; frame moved in")
    geo.frame = (geo.columns[0], geo.columns[-1])


def _header_spans(geo: PageGeometry, horizontals) -> tuple[list[tuple[int, int, int]], list[Rule]]:
    """``(first_col, last_col, y)`` for each rule strictly inside the header row.

    A label printed over several columns ("Přítomné obyvatelstvo", "Anwesende
    Bevölkerung") is closed by a rule under it running across exactly those
    columns, so the rule's extent *is* the grouping, read without reading the
    label. A rule counts only when both its ends meet a column rule: a run of
    type cannot do that.
    """
    s = geo.scale
    top = geo.table_top + 15 * s
    bottom = geo.header_bottom - 15 * s
    spans, rules = [], []
    for r in horizontals:
        if not (top <= r.y <= bottom):
            continue
        a = min(range(len(geo.columns)), key=lambda i: abs(geo.columns[i] - r.x0))
        b = min(range(len(geo.columns)), key=lambda i: abs(geo.columns[i] - r.x1))
        # Two columns at least: a label over one column needs no rule. And not
        # all of them: that is the frame's own (double) rule, read inside the
        # header when the top rule is set double (de 0236, mo 0103).
        full = a == 0 and b == len(geo.columns) - 1
        if (b > a + 1 and not full and abs(geo.columns[a] - r.x0) <= 15 * s
                and abs(geo.columns[b] - r.x1) <= 15 * s):
            spans.append((a, b - 1, r.y))
            rules.append(r)
    return sorted(set(spans), key=lambda t: (t[2], t[0])), rules


def _divider(geo: PageGeometry, pieces: list[Rule]) -> None:
    """The apparatus divider: the longest rule in the middle of the frame below
    the table's foot."""
    left, right = geo.frame
    s = geo.scale
    span = right - left
    foot = geo.table_foot
    zone = (left + DIVIDER_ZONE[0] * span, left + DIVIDER_ZONE[1] * span)
    # A divider standing right under a column rule may come out joined to it;
    # only its part below the table's foot is the divider.
    candidates = [Rule(r.x0, max(r.y0, foot), r.x1, r.y1) for r in pieces
                  if zone[0] <= r.x <= zone[1] and r.y1 - max(r.y0, foot) >= DIVIDER_MIN * s]
    if candidates:
        geo.divider = max(candidates, key=lambda r: r.height)


def _apparatus(geo: PageGeometry, ink, full_width: bool = False) -> None:
    left, right = geo.frame
    s = geo.scale
    foot = geo.table_foot
    bottom_edge = geo.sheet[3]
    if geo.divider is not None:
        foot = geo.divider.y1
        for region, box in geo.apparatus().items():
            inset = (box[0] + int(6 * s), box[1], box[2] - int(6 * s),
                     min(bottom_edge, foot + int(80 * s)))
            lines = [l for l in lines_in(ink, inset, s) if l.top < foot]
            geo.column_lines[region] = lines
            geo.column_foot[region] = max([foot] + [l.bottom + 1 for l in lines])
            geo.headings += find_headings(lines, inset, region, s, full_width)
        top = max(geo.column_foot.values())
        geo.footer = _block_below(geo, ink, top)
    else:
        geo.below = _block_below(geo, ink, foot)


def _block_below(geo: PageGeometry, ink, top: int):
    """The print under the apparatus (the footer) or under a table with none
    (a statutory city's prose): from ``top`` to its last line of print, as wide
    as the frame or as its lines, whichever is wider — running prose is set a
    little wider than the table above it (de 0021)."""
    left, right = geo.frame
    s = geo.scale
    margin = int(0.06 * (right - left))
    search = (max(geo.sheet[0], left - margin), top,
              min(geo.sheet[2], right + margin), geo.sheet[3])
    lines = _print_lines(ink, search, s, geo.sheet)
    if not lines:
        return None
    return (min([left] + [l.left for l in lines]), top,
            max([right] + [l.right + 1 for l in lines]), lines[-1].bottom + 1)


def is_edge(line: Line, sheet) -> bool:
    """The book's edge or the scanner's shadow, not print: a band touching the
    top or bottom of the sheet (de 0100). The register's print never runs off
    the sheet."""
    return line.top <= sheet[1] + 3 or line.bottom >= sheet[3] - 4


def _print_lines(ink, box, s, sheet) -> list[Line]:
    """Lines of type, not of the scan's edges: the lines of ``box`` with at least
    the ink of a short line of print and not touching the top or bottom of the
    image, where the book's edge and the scanner's shadow sit (de 0100)."""
    out = []
    for line in lines_in(ink, box, s):
        if is_edge(line, sheet):
            continue
        # The bulk of the ink must run along the line: a blot with a few specks
        # beside it (cz 0026) has three pieces but a core 48 px wide.
        if (line.dark >= 4 * LINE_MIN_DARK * s * s and line.core_right - line.core_left >= 60 * s
                and line.pieces >= 3):
            out.append(line)
    return out


def _head(geo: PageGeometry, ink) -> None:
    """Running head and page number: everything printed above the table."""
    left, right = geo.frame
    margin = int(0.06 * (right - left))
    box = (max(geo.sheet[0], left - margin), geo.sheet[1],
           min(geo.sheet[2], right + margin), geo.table_top)
    lines = lines_in(ink, box, geo.scale)
    if lines:
        # Opened 100 px above the first line measured: the page number is set
        # in a light face above the running head, and on some scans (cz 0396,
        # mo 0116, mo 0133, de 0068) it does not register as a line of its own.
        top = max(geo.sheet[1], lines[0].top - int(100 * geo.scale))
        geo.head = (box[0], top, box[2], geo.table_top)


def _prose(geo: PageGeometry, ink) -> None:
    """A page with no table: the text block, if there is one."""
    lines = lines_in(ink, geo.sheet, geo.scale)
    if not lines:
        geo.kind = "blank"
        return
    geo.kind = "prose"
    geo.prose = (min(l.left for l in lines), lines[0].top,
                 max(l.right for l in lines) + 1, lines[-1].bottom + 1)

def segments(geo: PageGeometry) -> list[Segment]:
    """Each apparatus column cut at its headings, tiling it top to foot.

    The image-only counterpart of ``lexikon_1910_notes.section_segments`` with
    ``tile=True, extend_to_foot=True``: the first segment opens at the band's top
    (and is *carried* unless a heading opens it), each later one at its heading's
    top, and the last runs to the divider's foot.
    """
    out: list[Segment] = []
    order = 0
    for region, (left, top, right, bottom) in geo.apparatus().items():
        tops = sorted(h.top for h in geo.headings if h.region == region)
        lines = geo.column_lines.get(region, [])
        edges = [top] + [t for t in tops if t > top] + [bottom]
        opened = top
        for n, (a, b) in enumerate(zip(edges, edges[1:])):
            if b - a < 4:
                continue
            if n == 0 and not any(a <= l.top < b for l in lines):
                continue            # nothing printed above the first heading
            # The column's first crop opens at the column's top whatever it
            # holds, so the crops tile the column (as ``tile=True`` does).
            out.append(Segment(region, (left, opened if not any(
                s.region == region for s in out) else a, right, b), order,
                carried=(n == 0 and (not tops or tops[0] > top + 2))))
            order += 1
    return out


def crops(geo: PageGeometry) -> dict[str, list[tuple[int, int, int, int]]]:
    """Every box a reader would be sent, by kind: head, table, apparatus, footer,
    below, prose."""
    out: dict[str, list[tuple[int, int, int, int]]] = {}
    if geo.head:
        out["head"] = [geo.head]
    if geo.table_box():
        out["table"] = [geo.table_box()]
    seg = [s.box for s in segments(geo)]
    if seg:
        out["apparatus"] = seg
    if geo.footer:
        out["footer"] = [geo.footer]
    if geo.below:
        out["below"] = [geo.below]
    if geo.prose:
        out["prose"] = [geo.prose]
    return out

def uncovered(image_or_geo, boxes: list[tuple[int, int, int, int]], pad: int = 0,
              geo: PageGeometry | None = None, grey=None) -> list[dict]:
    """Every printed line whose ink lies outside all of ``boxes``.

    Rules are excluded (they are the frame, not print). The search is limited to
    the page's own print area: the frame ± 6 % of its width horizontally (the
    facing page's edge and a scanner border are not this page's print), the
    sheet vertically; on a page with no frame, the sheet. ``pad`` grows every box
    the way the crop functions pad a crop.

    Returns one dict per uncovered line: rows, columns, dark pixels.
    """
    import numpy as np

    if grey is None:
        grey = to_grey(image_or_geo)
    if geo is None:
        geo = detect(grey)
    h, w = grey.shape
    mask = grey < geo.threshold
    mask &= ~rule_mask(mask.shape, geo.rules, max(2, int(3 * geo.scale)))
    l, t, r, b = geo.sheet
    if geo.frame is not None:
        margin = int(0.06 * (geo.frame[1] - geo.frame[0]))
        l, r = max(l, geo.frame[0] - margin), min(r, geo.frame[1] + margin)
    area = np.zeros_like(mask)
    area[t:b, l:r] = True
    covered = np.zeros_like(mask)
    for (x0, y0, x1, y1) in boxes:
        covered[max(0, int(y0) - pad):min(h, int(y1) + pad),
                max(0, int(x0) - pad):min(w, int(x1) + pad)] = True
    left_over = mask & area & ~covered
    out = []
    for line in lines_in(left_over, (0, 0, w, h), geo.scale):
        # Touching the top or bottom of the image: the book's edge or the
        # scanner's shadow (de 0100), kept but marked so a count can leave it out.
        edge = is_edge(line, (l, t, r, b))
        out.append({"top": line.top, "bottom": line.bottom, "left": line.left,
                    "right": line.right, "dark_px": line.dark, "pieces": line.pieces,
                    "zone": "edge" if edge else zone_of(geo, line)})
    return out


def zone_of(geo: PageGeometry, line: Line) -> str:
    """Which part of the page a line falls in: head / table / apparatus / footer /
    below / margin / page."""
    y = (line.top + line.bottom) / 2
    x = (line.left + line.right) / 2
    if geo.frame is None:
        return "page"
    if not geo.frame[0] <= x <= geo.frame[1]:
        return "margin"
    if geo.table_top is not None and y < geo.table_top:
        return "head"
    if geo.table_foot is not None and y < geo.table_foot:
        return "table"
    if geo.divider is not None and y <= geo.divider.y1:
        return "apparatus"
    return "footer" if geo.divider is not None else "below"


def lexikon_1910_image_report(geos: list[PageGeometry]) -> str:
