TL;DR

The Cisleithanian census (utils.parsing.census) carries German variable names (umgspr_bohmisch_mahrisch_slowakisch) and the Transleithanian one (utils.parsing.census_hu) Magyar ones (etn_tot_anyanyelv). This module renames every covariate to an English snake_case label and adds analysis-ready shares/rates.

How it works

Two-stage translation

  1. Curated clean names for the analytically-central families — colloquial language (lang_*), religion (rel_*), literacy/disability (lit_* / disab_*), and the settlement-size population breakdown (pop_*). See CURATED_CIS / CURATED_HU.
  2. Compositional token translation for the long tail (births, deaths, cause of death, livestock, occupation, justice, savings) via DE_TOKENS / HU_TOKENS — the family prefix is kept, content tokens translated, grammatical stopwords dropped, and any residual name collisions are disambiguated with a numeric suffix so the mapping stays 1-to-1.

Derived shares (fractions of the relevant base) are added by add_shares_cis / add_shares_hu: language shares, key religion shares, literacy rate, illegitimacy rate, infant-mortality proxy — the ratios that go on the RHS.

Usage

from utils.parsing import census_labels as cl
census = cl.to_english_census(census, frame="cis")   # rename + shares (okres panel)
census = cl.to_english_census(census, frame="hu")     # rename + shares (járás panel)

def english_rename_map(df: pd.DataFrame, frame: str) -> dict[str, str]:
    """Return the source→English column rename map for ``frame`` in {"cis","hu"}."""
    if frame == "cis":
        return _english_map(df.columns, CURATED_CIS, DE_TOKENS, _DE_STOP, _ID_CIS)
    if frame == "hu":
        return _english_map(df.columns, CURATED_HU, HU_TOKENS, _HU_STOP, _ID_HU)
    raise ValueError("frame must be 'cis' or 'hu'")

def add_shares_cis(df: pd.DataFrame) -> pd.DataFrame:
    """Add analysis-ready shares/rates to the (English-renamed) Cisleithanian panel."""
    out = df.copy()
    langs = ["german", "czech", "polish", "ruthenian", "slovene", "serbocroat",
             "italian_ladin", "romanian", "magyar"]
    if "lang_total" in out:
        for lg in langs:
            if f"lang_{lg}" in out:
                out[f"lang_{lg}_share"] = _safe_div(out[f"lang_{lg}"], out["lang_total"])
    if "rel_total" in out:
        for r in ["roman_catholic", "greek_catholic", "greek_orthodox", "lutheran",
                  "calvinist", "jewish"]:
            if f"rel_{r}" in out:
                out[f"rel_{r}_share"] = _safe_div(out[f"rel_{r}"], out["rel_total"])
    if {"lit_read_write", "lit_age_over10"} <= set(out.columns):
        out["literacy_rate"] = _safe_div(out["lit_read_write"], out["lit_age_over10"])
    # settlement-derived total population (sum of inhabitants across size classes);
    # min_count=1 keeps an all-missing okres NA instead of silently becoming 0
    inh = [c for c in out.columns if c.startswith("pop_") and c.endswith("_inhabitants")]
    if inh:
        out["pop_total"] = out[inh].apply(pd.to_numeric, errors="coerce").sum(axis=1, min_count=1)
    # the Zusammen (z) columns rename WITHOUT a _total suffix (the single-letter z is
    # dropped by _compose): births_livebirths_illegitimate / births_livebirths_total
    if {"births_livebirths_illegitimate", "births_livebirths_total"} <= set(out.columns):
        out["illegitimacy_rate"] = _safe_div(out["births_livebirths_illegitimate"],
                                             out["births_livebirths_total"])
    return out

def add_shares_hu(df: pd.DataFrame) -> pd.DataFrame:
    """Add key religion shares to the (English-renamed) Transleithanian panel.

    Language shares (``slovak_share`` etc.) are already produced by :mod:`census_hu`.
    """
    out = df.copy()
    base = out.get("pop_present")
    if base is not None:
        for r in ["roman_catholic", "greek_catholic", "calvinist", "lutheran", "orthodox", "jewish"]:
            if f"rel_{r}" in out:
                out[f"rel_{r}_share"] = _safe_div(out[f"rel_{r}"], base)
    return out

def to_english_census(df: pd.DataFrame, frame: str) -> pd.DataFrame:
    """Rename all covariates to English and append derived shares, for ``frame`` in {"cis","hu"}."""
    out = df.rename(columns=english_rename_map(df, frame))
    return add_shares_cis(out) if frame == "cis" else add_shares_hu(out)

def _compose(col: str, prefix: str, tokens: dict[str, str], stop: set[str]) -> str:
    """Translate a source column compositionally: keep translated tokens, drop stopwords."""
    parts = re.split(r"[_ ]+", col)
    out: list[str] = []
    for p in parts:
        pl = p.lower().strip(",.")
        if not pl or pl in stop:
            continue
        if re.fullmatch(r"v\d+|[a-d]\d+|spalten?|iii?|[a-z]", pl) and pl not in ("a", "b", "c", "d", "m", "w"):
            continue                                  # drop column-ref / agr-index / roman / stray-letter noise
        t = tokens.get(pl, pl)                       # unknown token → kept transliterated
        if t:
            out.append(t)
    name = "_".join(out) or prefix
    return re.sub(r"_+", "_", name).strip("_")


def _english_map(cols, curated: dict[str, str], tokens: dict[str, str], stop: set[str],
                 ids: set[str]) -> dict[str, str]:
    """Build a unique source→English rename map, resolving collisions with a numeric suffix."""
    seen: dict[str, int] = {}
    out: dict[str, str] = {}
    for c in cols:
        if c in ids:
            out[c] = c
            continue
        eng = curated.get(c) or _compose(c, c.split("_")[0], tokens, stop)
        if eng in seen:
            seen[eng] += 1
            eng = f"{eng}_{seen[eng]}"
        else:
            seen[eng] = 1
        out[c] = eng
    return out

def _safe_div(a, b):
    return pd.to_numeric(a, errors="coerce") / pd.to_numeric(b, errors="coerce").replace(0, pd.NA)


def add_shares_cis(df: pd.DataFrame) -> pd.DataFrame:
    """Add analysis-ready shares/rates to the (English-renamed) Cisleithanian panel."""
    out = df.copy()
    langs = ["german", "czech", "polish", "ruthenian", "slovene", "serbocroat",
             "italian_ladin", "romanian", "magyar"]
    if "lang_total" in out:
        for lg in langs:
            if f"lang_{lg}" in out:
                out[f"lang_{lg}_share"] = _safe_div(out[f"lang_{lg}"], out["lang_total"])
    if "rel_total" in out:
        for r in ["roman_catholic", "greek_catholic", "greek_orthodox", "lutheran",
                  "calvinist", "jewish"]:
            if f"rel_{r}" in out:
                out[f"rel_{r}_share"] = _safe_div(out[f"rel_{r}"], out["rel_total"])
    if {"lit_read_write", "lit_age_over10"} <= set(out.columns):
        out["literacy_rate"] = _safe_div(out["lit_read_write"], out["lit_age_over10"])
    # settlement-derived total population (sum of inhabitants across size classes);
    # min_count=1 keeps an all-missing okres NA instead of silently becoming 0
    inh = [c for c in out.columns if c.startswith("pop_") and c.endswith("_inhabitants")]
    if inh:
        out["pop_total"] = out[inh].apply(pd.to_numeric, errors="coerce").sum(axis=1, min_count=1)
    # the Zusammen (z) columns rename WITHOUT a _total suffix (the single-letter z is
    # dropped by _compose): births_livebirths_illegitimate / births_livebirths_total
    if {"births_livebirths_illegitimate", "births_livebirths_total"} <= set(out.columns):
        out["illegitimacy_rate"] = _safe_div(out["births_livebirths_illegitimate"],
                                             out["births_livebirths_total"])
    return out


def add_shares_hu(df: pd.DataFrame) -> pd.DataFrame:
    """Add key religion shares to the (English-renamed) Transleithanian panel.

    Language shares (``slovak_share`` etc.) are already produced by :mod:`census_hu`.
    """
    out = df.copy()
    base = out.get("pop_present")
    if base is not None:
        for r in ["roman_catholic", "greek_catholic", "calvinist", "lutheran", "orthodox", "jewish"]:
            if f"rel_{r}" in out:
                out[f"rel_{r}_share"] = _safe_div(out[f"rel_{r}"], base)
    return out


def to_english_census(df: pd.DataFrame, frame: str) -> pd.DataFrame:
    """Rename all covariates to English and append derived shares, for ``frame`` in {"cis","hu"}."""
    out = df.rename(columns=english_rename_map(df, frame))
    return add_shares_cis(out) if frame == "cis" else add_shares_hu(out)
