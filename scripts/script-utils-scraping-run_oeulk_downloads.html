How it works

The work (Kriegsarchiv, Wien 1930–38) sits on the DSMO Kramerius as one monograph holding sixteen monographunit children: seven text volumes, the seven Beilagen volumes of maps and tables that go with them, Ergänzungsheft 9 (Ehnl 1934 — also held separately as Ehnl-1934_…) and the Registerband. The runner reads that list off the root rather than carrying sixteen pids, names each volume from the library's own partNumber::

1. Band             ->  Kriegsarchiv_Oesterreich-Ungarns-letzter-Krieg_Bd1
1. Band - Beilagen  ->  …_Bd1-Beilagen
Ergänzungsheft 9    ->  …_Ergaenzungsheft-9
Registerband        ->  …_Registerband

and hands them one at a time to scraper_kramerius.download_document. One host, so the volumes run serially, at the module's own pacing (20 pages per batch, 10 s rest between batches).

Safe to interrupt and re-run. A volume is done when its PDF holds as many pages as the library lists for it; a done volume is skipped, a half-fetched one resumes from its &lt;name&gt;_pages/ cache, and a PDF that came out short (a scan that would not download) is fetched again for the missing pages only.

Scans are taken at native size (native_images=True): the IMG_FULL stream is scaled to 1500 px wide on this host, which leaves a map sheet 1500 of its 2370 px. --force re-fetches a volume whose PDF is already complete — the way to replace a 1500 px copy.

The viewer URL is on digitalnistudovna.mo.gov.cz — the module's default is the older …army.cz name — so the host is passed explicitly. Its API is Kramerius 5. TLS verification is off for the reason given in run_lexicon_downloads.py: the local certifi bundle cannot build a chain for this host although curl and the Windows trust store accept it.

Usage::

python utils/scraping/run_oeulk_downloads.py
python utils/scraping/run_oeulk_downloads.py --only Bd1-Beilagen Bd6
python utils/scraping/run_oeulk_downloads.py --list

def list_volumes(client: scraper_kramerius.KrameriusClient) -> list[dict]:
    """The work's volumes in library order, as ``{"key", "pid", "part"}``."""
    volumes = []
    for child in client.children(ROOT_PID):
        part = (child.get("details") or {}).get("partNumber") or child["pid"]
        volumes.append({"key": volume_key(part), "pid": child["pid"], "part": part})
    keys = [v["key"] for v in volumes]
    if len(set(keys)) != len(keys):  # two volumes would overwrite one file
        raise RuntimeError(f"volume keys are not unique: {keys}")
    return volumes

def run_volume(client: scraper_kramerius.KrameriusClient, volume: dict,
               out_dir: Path, *, progress: bool, force: bool = False) -> dict:
    """Download one volume unless its PDF is already complete; return a summary row."""
    stem = STEM_PREFIX + volume["key"]
    pdf = out_dir / f"{stem}.pdf"
    listed = sum(1 for _ in client.iter_pages(volume["pid"]))
    have = pages_on_disk(pdf)
    row = {"key": volume["key"], "listed": listed, "pdf": pdf}
    if have == listed and not force:
        log.info("%-18s done already (%d pp) — skipped", volume["key"], listed)
        return row | {"status": "skipped", "scans": have}
    if have is not None and have != listed:
        log.info("%-18s PDF holds %d of %d pp — fetching the rest", volume["key"], have, listed)

    log.info("\n=== %s  (%s, %d pp) ===", volume["key"], volume["part"], listed)
    res = scraper_kramerius.download_document(
        volume["pid"], out_dir=out_dir, host=HOST, name=stem, native_images=True,
        verify_tls=False, progress=progress, progress_desc=volume["key"],
    )
    # SHORT = a scan or an OCR page is missing; LOWRES = every page is there but
    # some came from the scaled IMG_FULL (a dropped connection, usually) — the
    # page count cannot see that, so only --force replaces them.
    if res["images"] != listed or res["ocr_pages"] != listed:
        status = "SHORT"
    elif res["native_fallbacks"]:
        status = "LOWRES"
    else:
        status = "ok"
    return row | {"status": status, "scans": res["images"], "ocr": res["ocr_pages"]}

def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(
        description=__doc__.split("\n\n")[0],
        formatter_class=argparse.ArgumentDefaultsHelpFormatter,
    )
    p.add_argument("--out", default=str(DEFAULT_OUT), help="output directory")
    p.add_argument("--only", nargs="+", default=None, metavar="KEY",
                   help="download just these volumes, in the order given (keys as in --list)")
    p.add_argument("--list", action="store_true", help="print the volume keys and exit")
    p.add_argument("--force", action="store_true",
                   help="re-fetch volumes whose PDF is already complete")
    p.add_argument("--no-progress", action="store_true",
                   help="no progress bar; log every 25 pages instead")
    p.add_argument("-v", "--verbose", action="store_true", help="debug logging")
    args = p.parse_args(argv)

    logging.basicConfig(level=logging.DEBUG if args.verbose else logging.INFO,
                        format="%(message)s")

    client = scraper_kramerius.KrameriusClient(HOST, verify_tls=False)
    volumes = list_volumes(client)
    if args.list:
        for v in volumes:
            print(f"{v['key']:<18} {v['pid']}  {v['part']}")
        return 0
    if args.only:
        by_key = {v["key"]: v for v in volumes}
        unknown = [k for k in args.only if k not in by_key]
        if unknown:
            p.error(f"unknown volume key(s) {unknown}; see --list")
        volumes = [by_key[k] for k in args.only]

    out_dir = Path(args.out)
    rows: list[dict] = []
    for volume in volumes:
        try:
            rows.append(run_volume(client, volume, out_dir,
                                   progress=not args.no_progress, force=args.force))
        except Exception as exc:                           # keep going; report at the end
            log.error("FAILED (%s): %s", volume["key"], exc)
            rows.append({"key": volume["key"], "error": exc})

    summarise(rows)
    return 1 if any(r.get("status") in ("SHORT", "LOWRES") or "error" in r for r in rows) else 0
