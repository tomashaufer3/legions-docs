How it works

Scrapes individual record pages from two databases hosted at vuapraha.cz:

legionari     Databáze legionářů          /legionari/{id}/
padli         Databáze padlých v 1. sv. válce  /padli-1-svetova/{id}/

Both databases share the same HTML structure: a sequence of div.inventory__item elements, each containing a label and a value. The parser extracts all label–value pairs generically, so new fields added by VÚA are captured automatically without code changes.

Notes on the data source

vuapraha.cz is the canonical VÚA-VHA Praha interface. The legionář database (~133k records) was assembled 1997–2004; the Habsburg fallen database has 186,194 records assembled 2012–2020. ID spaces are independent sequential integers with gaps; the ceiling must be probed empirically (see autodetect_max_id).

The site does not require authentication and individual record pages are server-rendered (no JS execution needed). The public search UI at /databaze-vha/?dtb=fallen-first is only a front door: it POSTs JSON to /wp-json/mo/v1/archive-database/html with a body of {page, name, surname, place, date, database} (database="fallen-first") and returns {"html": <10-row results table>} whose rows link to the /padli-1-svetova/{id} detail pages enumerated here. Direct ID enumeration of the detail pages is simpler and yields the complete field set, so that endpoint is documented only as an optional manifest / cross-check source.

TLS note: the server presents a Czech government-PKI certificate chain that modern OpenSSL rejects ("Basic Constraints of CA cert not marked critical"). Sessions therefore disable certificate verification by default; pass verify_tls=True (CLI --verify-tls) to re-enable it.

Detail-page field set (fallen DB): Příjmení, Jméno, Hodnost (often absent), Datum narození, Místo narození, Domovem, Příslušnost k útvaru rakousko-uherské armády, Datum úmrtí, Příčina úmrtí, Místo úmrtí, Bojiště, Místo pohřbení, Poznámka. The field set varies per record, so the CSV writer expands its header on the fly. "?" is the source's own null marker, stored verbatim.

def make_session(
    user_agent: str = USER_AGENT,
    verify_tls: bool = False,
) -> requests.Session:
    """
    Build a requests session for vuapraha.cz.

    TLS verification is disabled by default: the server's government-PKI
    certificate chain is rejected by modern OpenSSL ("Basic Constraints of CA
    cert not marked critical"), so verification has to be relaxed to connect
    at all.  Pass ``verify_tls=True`` to re-enable it (and silence the warning
    suppression below).
    """
    s = requests.Session()
    s.headers.update({"User-Agent": user_agent, "Accept-Language": "cs,en;q=0.8"})
    s.verify = verify_tls
    if not verify_tls:
        # Avoid flooding stderr with one InsecureRequestWarning per request.
        from urllib3.exceptions import InsecureRequestWarning
        import urllib3
        urllib3.disable_warnings(InsecureRequestWarning)
    return s

def fetch_with_retries(
    session: requests.Session,
    url: str,
    retries: int = 3,
    backoff: float = 2.0,
    timeout: int = 15,
) -> requests.Response | None:
    """GET *url*, retrying transient failures with exponential backoff.

    A 186k-record scrape will hit intermittent timeouts and 5xx no matter how polite it
    is, so a single failure must not abort the run. **404 is returned, not retried** — on
    this site it is a meaningful answer (the ID has no record) and the caller records it
    as MISSING rather than as an error.

    Args:
        session: pooled session from the caller.
        url: absolute record URL.
        retries: attempts before giving up.
        backoff: base seconds; the wait grows as ``backoff ** attempt``.
        timeout: per-request timeout in seconds.

    Returns:
        The response (including a 404), or ``None`` if every attempt failed.
    """
    for attempt in range(retries):
        try:
            r = session.get(url, timeout=timeout, allow_redirects=True)
            if r.status_code == 200:
                return r
            if r.status_code == 404:
                return r          # caller handles MISSING
            logging.warning("HTTP %s for %s (attempt %d)", r.status_code, url, attempt + 1)
        except requests.RequestException as exc:
            logging.warning("Request error for %s: %s (attempt %d)", url, exc, attempt + 1)
        if attempt < retries - 1:
            time.sleep(backoff * (attempt + 1))
    return None

def parse_record(html: str, record_id: int, db: DB) -> dict | None:
    """
    Parse a single vuapraha.cz record page into a flat dict.

    Returns None if the page does not contain a record (missing ID).
    All label–value pairs are extracted generically; the label text
    (stripped, with trailing colon removed) becomes the dict key.
    """
    soup = BeautifulSoup(html, "html.parser")
    inventory = soup.find("div", class_="inventory")
    if inventory is None:
        return None

    record: dict[str, str] = {}
    for item in inventory.find_all("div", class_="inventory__item"):
        label_el = item.find("div", class_="inventory__label")
        value_el = item.find("div", class_="inventory__value")
        if label_el is None or value_el is None:
            continue
        label = label_el.get_text(strip=True).rstrip(":")
        value = value_el.get_text(" ", strip=True)
        # Use first occurrence only (labels are unique per page)
        if label not in record:
            record[label] = value

    if not record:
        return None

    # Metadata
    record["vua_id"]          = record_id
    record["database"]        = db.value
    record["scraped_at"]      = datetime.now(timezone.utc).isoformat()
    record["scraper_version"] = SCRAPER_VERSION

    return record

class CSVWriter:
    """
    Append-mode CSV with on-the-fly header expansion (resume-aware).

    Records from the generic label/value parser have a *heterogeneous* field
    set — e.g. ``Hodnost`` (rank) is present on some records and absent on
    others.  A fixed header set from the first record would silently drop any
    field that first appears later, so when a record introduces new keys the
    whole file is rewritten with the widened header.  New keys appear only a
    handful of times early in a run; every other write is a plain O(1) append.
    """

    def __init__(self, path: Path) -> None:
        self.path = path
        self._fieldnames: list[str] = []
        self._written_ids: set[int] = set()
        if path.exists() and path.stat().st_size > 0:
            self._load_existing()

    def _load_existing(self) -> None:
        with open(self.path, newline="", encoding="utf-8") as f:
            reader = csv.DictReader(f)
            self._fieldnames = list(reader.fieldnames or [])
            for row in reader:
                try:
                    self._written_ids.add(int(row["vua_id"]))
                except (KeyError, ValueError, TypeError):
                    pass

    def already_scraped(self, record_id: int) -> bool:
        return record_id in self._written_ids

    def write(self, record: dict) -> None:
        new_keys = [k for k in record.keys() if k not in self._fieldnames]
        if new_keys or not self._fieldnames:
            # Widen the header: read back existing rows, rewrite with the new
            # column set, then append this record.
            existing_rows: list[dict] = []
            if self.path.exists() and self._fieldnames:
                with open(self.path, newline="", encoding="utf-8") as f:
                    existing_rows = list(csv.DictReader(f))
            self._fieldnames.extend(new_keys)
            with open(self.path, "w", newline="", encoding="utf-8") as f:
                w = csv.DictWriter(f, fieldnames=self._fieldnames, extrasaction="ignore")
                w.writeheader()
                for row in existing_rows:
                    w.writerow(row)
                w.writerow(record)
        else:
            with open(self.path, "a", newline="", encoding="utf-8") as f:
                w = csv.DictWriter(f, fieldnames=self._fieldnames, extrasaction="ignore")
                w.writerow(record)

        try:
            self._written_ids.add(int(record["vua_id"]))
        except (KeyError, ValueError, TypeError):
            pass

    def close(self) -> None:
        # No persistent file handle is held; present for API symmetry.
        pass

    def __enter__(self):
        return self

    def __exit__(self, *args):
        self.close()

def autodetect_max_id(
    session: requests.Session,
    db: DB,
    low: int = 1,
    high: int = 500_000,
) -> int:
    """
    Binary search for the highest ID that returns a valid record.
    Useful for determining the enumeration ceiling before a full scrape.
    """
    logging.info("Probing ID ceiling for %s (low=%d, high=%d)", db.value, low, high)
    result = low
    while low <= high:
        mid = (low + high) // 2
        url = _url(db, mid)
        r = fetch_with_retries(session, url)
        if r is not None and not _is_missing(r):
            result = mid
            low = mid + 1
        else:
            high = mid - 1
        time.sleep(0.5)
    logging.info("Estimated max ID for %s: %d", db.value, result)
    return result

def iter_scrape(
    start_id: int,
    end_id: int,
    db: DB,
    session: requests.Session | None = None,
    delay: float = 1.0,
    resume_ids: set[int] | None = None,
    missing_ids: set[int] | None = None,
) -> Iterator[tuple[int, dict | None, str]]:
    """
    Yield ``(record_id, record_dict_or_None, status)`` for each ID in range.

    status values: "ok" | "missing" | "error" | "skipped"
    """
    if session is None:
        session = make_session()
    resume_ids  = resume_ids  or set()
    missing_ids = missing_ids or set()

    for record_id in range(start_id, end_id + 1):
        if record_id in resume_ids or record_id in missing_ids:
            yield record_id, None, "skipped"
            continue

        url = _url(db, record_id)
        r = fetch_with_retries(session, url)

        if r is None:
            yield record_id, None, "error"
            time.sleep(delay)
            continue

        if _is_missing(r):
            yield record_id, None, "missing"
            time.sleep(delay * 0.5)
            continue

        record = parse_record(r.text, record_id, db)
        if record is None:
            yield record_id, None, "error"
        else:
            yield record_id, record, "ok"

        time.sleep(delay)

def scrape_range(
    start_id: int,
    end_id: int,
    db: DB,
    output_dir: str | Path = "data/raw/habsburg",
    delay: float = 1.0,
    resume: bool = True,
    progress: str = "tqdm",   # "tqdm" | "log" | "none"
    log_interval: int = 500,
    verify_tls: bool = False,
) -> tuple[list[dict], dict]:
    """
    Scrape a range of IDs from *db* and write results to CSV.

    Parameters
    ----------
    start_id, end_id : int
        Inclusive ID range.
    db : DB
        Target database (DB.LEGIONARI or DB.PADLI).
    output_dir : path-like
        Directory for output files.  Created if absent.
    delay : float
        Seconds between requests.
    resume : bool
        Skip IDs already present in the output CSV.
    progress : str
        "tqdm" for a progress bar, "log" for periodic log lines, "none" for silence.
    log_interval : int
        How often to log progress when progress="log".
    verify_tls : bool
        Enable TLS cert verification (off by default — see ``make_session``).

    Returns
    -------
    records : list[dict]
        All successfully scraped records (also written to CSV).
    summary : dict
        Counts of ok / missing / error / skipped.
    """
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    stem        = _DB_CONFIG[db]["default_stem"]
    csv_path    = output_dir / f"{stem}.csv"
    failed_path = output_dir / f"{stem}_failed.txt"

    session = make_session(verify_tls=verify_tls)

    resume_ids  = set()
    missing_ids = set()
    if resume:
        writer_probe = CSVWriter(csv_path)
        resume_ids   = writer_probe._written_ids
        writer_probe.close()
        missing_ids  = _existing_failed_ids(failed_path)

    summary = {"ok": 0, "missing": 0, "error": 0, "skipped": 0}
    records: list[dict] = []

    total = end_id - start_id + 1
    iterator = iter_scrape(
        start_id, end_id, db,
        session=session, delay=delay,
        resume_ids=resume_ids, missing_ids=missing_ids,
    )

    if progress == "tqdm":
        try:
            from tqdm import tqdm
            iterator = tqdm(iterator, total=total, unit="id")
        except ImportError:
            progress = "log"

    with CSVWriter(csv_path) as writer:
        for i, (record_id, record, status) in enumerate(iterator, 1):
            summary[status] += 1
            if status == "ok" and record:
                writer.write(record)
                records.append(record)
            elif status == "missing":
                _log_failure(failed_path, record_id, "MISSING")
            elif status == "error":
                _log_failure(failed_path, record_id, "ERROR")

            if progress == "log" and i % log_interval == 0:
                pct = i / total * 100
                logging.info(
                    "[%s] %d/%d (%.1f%%) — ok:%d missing:%d error:%d skipped:%d",
                    db.value, i, total, pct,
                    summary["ok"], summary["missing"],
                    summary["error"], summary["skipped"],
                )

    logging.info(
        "Done. ok=%d missing=%d error=%d skipped=%d → %s",
        summary["ok"], summary["missing"], summary["error"],
        summary["skipped"], csv_path,
    )
    return records, summary

def test_id(record_id: int, db: DB = DB.PADLI, verify_tls: bool = False) -> dict | None:
    """
    Fetch and pretty-print a single record.  Returns the parsed dict.

    >>> test_id(1,   db=DB.PADLI)        # fallen Habsburg soldier
    >>> test_id(668, db=DB.PADLI)

    session = make_session(verify_tls=verify_tls)
    url = _url(db, record_id)
    print(f"Fetching: {url}")

        return None

    for k, v in record.items():
        print(f"  {k:<{width}}  {v}")
    return record

def validate_output(output_dir: str | Path, db: DB = DB.PADLI, print_output: bool = True) -> dict:
    """
    Audit a completed scrape: record count, null rates, ID coverage.
    """
    output_dir = Path(output_dir)
    stem        = _DB_CONFIG[db]["default_stem"]
    csv_path    = output_dir / f"{stem}.csv"
    failed_path = output_dir / f"{stem}_failed.txt"

    if not csv_path.exists():
        raise FileNotFoundError(csv_path)

    import pandas as pd
    df = pd.read_csv(csv_path, low_memory=False)

    n           = len(df)
    id_min      = df["vua_id"].min()
    id_max      = df["vua_id"].max()
    null_rates  = (df.isnull().sum() / n * 100).round(1).sort_values(ascending=False)

    missing_count = 0
    error_count   = 0
    if failed_path.exists():
        with open(failed_path) as f:
            for line in f:
                tag = line.strip().split("\t")[-1]
                if tag == "MISSING":
                    missing_count += 1
                elif tag == "ERROR":
                    error_count += 1

    result = {
        "records":       n,
        "id_min":        id_min,
        "id_max":        id_max,
        "missing_ids":   missing_count,
        "error_ids":     error_count,
        "null_rates":    null_rates.to_dict(),
    }

    if print_output:
        print(f"Database   : {_DB_CONFIG[db]['label_cs']}")
        print(f"CSV        : {csv_path}")
        print(f"Records    : {n:,}")
        print(f"ID range   : {id_min} – {id_max}")
        print(f"Missing IDs: {missing_count:,}")
        print(f"Error IDs  : {error_count:,}")
        print("\nNull rates (top 15):")
        for col, pct in list(null_rates.items())[:15]:
            if pct > 0:
                print(f"  {col:<55}  {pct:.1f}%")

#!/usr/bin/env python3
"""
vuapraha.cz scraper — importable module + CLI.

Scrapes individual record pages from two databases hosted at vuapraha.cz:

    legionari     Databáze legionářů          /legionari/{id}/
    padli         Databáze padlých v 1. sv. válce  /padli-1-svetova/{id}/

Both databases share the same HTML structure: a sequence of
``div.inventory__item`` elements, each containing a label and a value.
The parser extracts all label–value pairs generically, so new fields added
by VÚA are captured automatically without code changes.

Notes on the data source
------------------------
vuapraha.cz is the canonical VÚA-VHA Praha interface.  The legionář database
(~133k records) was assembled 1997–2004; the Habsburg fallen database has
186,194 records assembled 2012–2020.  ID spaces are independent sequential
integers with gaps; the ceiling must be probed empirically (see
``autodetect_max_id``).

The site does not require authentication and individual record pages are
server-rendered (no JS execution needed).  The public search UI at
``/databaze-vha/?dtb=fallen-first`` is only a front door: it POSTs JSON to
``/wp-json/mo/v1/archive-database/html`` with a body of
``{page, name, surname, place, date, database}`` (``database="fallen-first"``)
and returns ``{"html": <10-row results table>}`` whose rows link to the
``/padli-1-svetova/{id}`` detail pages enumerated here.  Direct ID enumeration
of the detail pages is simpler and yields the complete field set, so that
endpoint is documented only as an optional manifest / cross-check source.

TLS note: the server presents a Czech government-PKI certificate chain that
modern OpenSSL rejects ("Basic Constraints of CA cert not marked critical").
Sessions therefore disable certificate verification by default; pass
``verify_tls=True`` (CLI ``--verify-tls``) to re-enable it.

Detail-page field set (fallen DB): Příjmení, Jméno, Hodnost (often absent),
Datum narození, Místo narození, Domovem, Příslušnost k útvaru rakousko-uherské
armády, Datum úmrtí, Příčina úmrtí, Místo úmrtí, Bojiště, Místo pohřbení,
Poznámka.  The field set varies per record, so the CSV writer expands its
header on the fly.  "?" is the source's own null marker, stored verbatim.

Jupyter usage
-------------
    >>> from scraper_vuapraha import scrape_range, test_id, validate_output, DB
    >>> test_id(1,   db=DB.PADLI)        # fallen Habsburg soldier
    >>> test_id(668, db=DB.PADLI)
    >>> scrape_range(1, 300, db=DB.PADLI, output_dir="data/raw/habsburg", delay=0.7)

CLI usage: see ``python scraper_vuapraha.py --help``
"""

class DB(str, Enum):
    """Supported vuapraha.cz databases."""
    LEGIONARI = "legionari"
    PADLI     = "padli"

_DB_CONFIG: dict[DB, dict] = {
    DB.LEGIONARI: {
        "url_prefix":   "https://www.vuapraha.cz/legionari/",
        "label_cs":     "Databáze legionářů",
        "default_stem": "legionari_vua",
    },
    DB.PADLI: {
        "url_prefix":   "https://www.vuapraha.cz/padli-1-svetova/",
        "label_cs":     "Databáze padlých v 1. sv. válce",
        "default_stem": "habsburg_fallen",
    },
}

def _url(db: DB, record_id: int) -> str:
    # No trailing slash — matches the real detail-page links and avoids a
    # redirect hop on every request.
    return f"{_DB_CONFIG[db]['url_prefix']}{record_id}"

# HTTP session

USER_AGENT = "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"


def make_session(
    user_agent: str = USER_AGENT,
    verify_tls: bool = False,
) -> requests.Session:
    """
    Build a requests session for vuapraha.cz.

    TLS verification is disabled by default: the server's government-PKI
    certificate chain is rejected by modern OpenSSL ("Basic Constraints of CA
    cert not marked critical"), so verification has to be relaxed to connect
    at all.  Pass ``verify_tls=True`` to re-enable it (and silence the warning
    suppression below).
    """
    s = requests.Session()
    s.headers.update({"User-Agent": user_agent, "Accept-Language": "cs,en;q=0.8"})
    s.verify = verify_tls
    if not verify_tls:
        # Avoid flooding stderr with one InsecureRequestWarning per request.
        from urllib3.exceptions import InsecureRequestWarning
        import urllib3
        urllib3.disable_warnings(InsecureRequestWarning)
    return s


def fetch_with_retries(
    session: requests.Session,
    url: str,
    retries: int = 3,
    backoff: float = 2.0,
    timeout: int = 15,
) -> requests.Response | None:
    """GET *url*, retrying transient failures with exponential backoff.

    A 186k-record scrape will hit intermittent timeouts and 5xx no matter how polite it
    is, so a single failure must not abort the run. **404 is returned, not retried** — on
    this site it is a meaningful answer (the ID has no record) and the caller records it
    as MISSING rather than as an error.

    Args:
        session: pooled session from the caller.
        url: absolute record URL.
        retries: attempts before giving up.
        backoff: base seconds; the wait grows as ``backoff ** attempt``.
        timeout: per-request timeout in seconds.

    Returns:
        The response (including a 404), or ``None`` if every attempt failed.
    """
    for attempt in range(retries):
        try:
            r = session.get(url, timeout=timeout, allow_redirects=True)
            if r.status_code == 200:
                return r
            if r.status_code == 404:
                return r          # caller handles MISSING
            logging.warning("HTTP %s for %s (attempt %d)", r.status_code, url, attempt + 1)
        except requests.RequestException as exc:
            logging.warning("Request error for %s: %s (attempt %d)", url, exc, attempt + 1)
        if attempt < retries - 1:
            time.sleep(backoff * (attempt + 1))
    return None

def _is_missing(response: requests.Response) -> bool:
    """Return True if the response indicates a non-existent record."""
    if response.status_code == 404:
        return True
    # vuapraha redirects missing IDs to the homepage or returns a page
    # without the inventory div.
    soup = BeautifulSoup(response.text, "html.parser")
    return soup.find("div", class_="inventory") is None


def parse_record(html: str, record_id: int, db: DB) -> dict | None:
    """
    Parse a single vuapraha.cz record page into a flat dict.

    Returns None if the page does not contain a record (missing ID).
    All label–value pairs are extracted generically; the label text
    (stripped, with trailing colon removed) becomes the dict key.
    """
    soup = BeautifulSoup(html, "html.parser")
    inventory = soup.find("div", class_="inventory")
    if inventory is None:
        return None

    record: dict[str, str] = {}
    for item in inventory.find_all("div", class_="inventory__item"):
        label_el = item.find("div", class_="inventory__label")
        value_el = item.find("div", class_="inventory__value")
        if label_el is None or value_el is None:
            continue
        label = label_el.get_text(strip=True).rstrip(":")
        value = value_el.get_text(" ", strip=True)
        # Use first occurrence only (labels are unique per page)
        if label not in record:
            record[label] = value

    if not record:
        return None

    # Metadata
    record["vua_id"]          = record_id
    record["database"]        = db.value
    record["scraped_at"]      = datetime.now(timezone.utc).isoformat()
    record["scraper_version"] = SCRAPER_VERSION

    return record

# CSV writer (streaming, resume-aware)

class CSVWriter:
    """
    Append-mode CSV with on-the-fly header expansion (resume-aware).

    Records from the generic label/value parser have a *heterogeneous* field
    set — e.g. ``Hodnost`` (rank) is present on some records and absent on
    others.  A fixed header set from the first record would silently drop any
    field that first appears later, so when a record introduces new keys the
    whole file is rewritten with the widened header.  New keys appear only a
    handful of times early in a run; every other write is a plain O(1) append.
    """

    def __init__(self, path: Path) -> None:
        self.path = path
        self._fieldnames: list[str] = []
        self._written_ids: set[int] = set()
        if path.exists() and path.stat().st_size > 0:
            self._load_existing()

    def _load_existing(self) -> None:
        with open(self.path, newline="", encoding="utf-8") as f:
            reader = csv.DictReader(f)
            self._fieldnames = list(reader.fieldnames or [])
            for row in reader:
                try:
                    self._written_ids.add(int(row["vua_id"]))
                except (KeyError, ValueError, TypeError):
                    pass

    def already_scraped(self, record_id: int) -> bool:
        return record_id in self._written_ids

    def write(self, record: dict) -> None:
        new_keys = [k for k in record.keys() if k not in self._fieldnames]
        if new_keys or not self._fieldnames:
            # Widen the header: read back existing rows, rewrite with the new
            # column set, then append this record.
            existing_rows: list[dict] = []
            if self.path.exists() and self._fieldnames:
                with open(self.path, newline="", encoding="utf-8") as f:
                    existing_rows = list(csv.DictReader(f))
            self._fieldnames.extend(new_keys)
            with open(self.path, "w", newline="", encoding="utf-8") as f:
                w = csv.DictWriter(f, fieldnames=self._fieldnames, extrasaction="ignore")
                w.writeheader()
                for row in existing_rows:
                    w.writerow(row)
                w.writerow(record)
        else:
            with open(self.path, "a", newline="", encoding="utf-8") as f:
                w = csv.DictWriter(f, fieldnames=self._fieldnames, extrasaction="ignore")
                w.writerow(record)

        try:
            self._written_ids.add(int(record["vua_id"]))
        except (KeyError, ValueError, TypeError):
            pass

    def close(self) -> None:
        # No persistent file handle is held; present for API symmetry.
        pass

    def __enter__(self):
        return self

    def __exit__(self, *args):
        self.close()


def _log_failure(failed_path: Path, record_id: int, tag: str) -> None:
    with open(failed_path, "a", encoding="utf-8") as f:
        f.write(f"{record_id}\t{tag}\n")


def _existing_failed_ids(failed_path: Path) -> set[int]:
    if not failed_path.exists():
        return set()
    ids: set[int] = set()
    with open(failed_path, encoding="utf-8") as f:
        for line in f:
            parts = line.strip().split("\t")
            try:
                ids.add(int(parts[0]))
            except (ValueError, IndexError):
                pass
    return ids

# ---------------------------------------------------------------------------
# ID ceiling detection
# ---------------------------------------------------------------------------

def autodetect_max_id(
    session: requests.Session,
    db: DB,
    low: int = 1,
    high: int = 500_000,
) -> int:
    """
    Binary search for the highest ID that returns a valid record.
    Useful for determining the enumeration ceiling before a full scrape.
    """
    logging.info("Probing ID ceiling for %s (low=%d, high=%d)", db.value, low, high)
    result = low
    while low <= high:
        mid = (low + high) // 2
        url = _url(db, mid)
        r = fetch_with_retries(session, url)
        if r is not None and not _is_missing(r):
            result = mid
            low = mid + 1
        else:
            high = mid - 1
        time.sleep(0.5)
    logging.info("Estimated max ID for %s: %d", db.value, result)
    return result

def iter_scrape(
    start_id: int,
    end_id: int,
    db: DB,
    session: requests.Session | None = None,
    delay: float = 1.0,
    resume_ids: set[int] | None = None,
    missing_ids: set[int] | None = None,
) -> Iterator[tuple[int, dict | None, str]]:
    """
    Yield ``(record_id, record_dict_or_None, status)`` for each ID in range.

    status values: "ok" | "missing" | "error" | "skipped"
    """
    if session is None:
        session = make_session()
    resume_ids  = resume_ids  or set()
    missing_ids = missing_ids or set()

    for record_id in range(start_id, end_id + 1):
        if record_id in resume_ids or record_id in missing_ids:
            yield record_id, None, "skipped"
            continue

        url = _url(db, record_id)
        r = fetch_with_retries(session, url)

        if r is None:
            yield record_id, None, "error"
            time.sleep(delay)
            continue

        if _is_missing(r):
            yield record_id, None, "missing"
            time.sleep(delay * 0.5)
            continue

        record = parse_record(r.text, record_id, db)
        if record is None:
            yield record_id, None, "error"
        else:
            yield record_id, record, "ok"

        time.sleep(delay)

def scrape_range(
    start_id: int,
    end_id: int,
    db: DB,
    output_dir: str | Path = "data/raw/habsburg",
    delay: float = 1.0,
    resume: bool = True,
    progress: str = "tqdm",   # "tqdm" | "log" | "none"
    log_interval: int = 500,
    verify_tls: bool = False,
) -> tuple[list[dict], dict]:
    """
    Scrape a range of IDs from *db* and write results to CSV.

    Parameters
    ----------
    start_id, end_id : int
        Inclusive ID range.
    db : DB
        Target database (DB.LEGIONARI or DB.PADLI).
    output_dir : path-like
        Directory for output files.  Created if absent.
    delay : float
        Seconds between requests.
    resume : bool
        Skip IDs already present in the output CSV.
    progress : str
        "tqdm" for a progress bar, "log" for periodic log lines, "none" for silence.
    log_interval : int
        How often to log progress when progress="log".
    verify_tls : bool
        Enable TLS cert verification (off by default — see ``make_session``).

    Returns
    -------
    records : list[dict]
        All successfully scraped records (also written to CSV).
    summary : dict
        Counts of ok / missing / error / skipped.
    """
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    stem        = _DB_CONFIG[db]["default_stem"]
    csv_path    = output_dir / f"{stem}.csv"
    failed_path = output_dir / f"{stem}_failed.txt"

    session = make_session(verify_tls=verify_tls)

    resume_ids  = set()
    missing_ids = set()
    if resume:
        writer_probe = CSVWriter(csv_path)
        resume_ids   = writer_probe._written_ids
        writer_probe.close()
        missing_ids  = _existing_failed_ids(failed_path)

    summary = {"ok": 0, "missing": 0, "error": 0, "skipped": 0}
    records: list[dict] = []

    total = end_id - start_id + 1
    iterator = iter_scrape(
        start_id, end_id, db,
        session=session, delay=delay,
        resume_ids=resume_ids, missing_ids=missing_ids,
    )

    if progress == "tqdm":
        try:
            from tqdm import tqdm
            iterator = tqdm(iterator, total=total, unit="id")
        except ImportError:
            progress = "log"

    with CSVWriter(csv_path) as writer:
        for i, (record_id, record, status) in enumerate(iterator, 1):
            summary[status] += 1
            if status == "ok" and record:
                writer.write(record)
                records.append(record)
            elif status == "missing":
                _log_failure(failed_path, record_id, "MISSING")
            elif status == "error":
                _log_failure(failed_path, record_id, "ERROR")

            if progress == "log" and i % log_interval == 0:
                pct = i / total * 100
                logging.info(
                    "[%s] %d/%d (%.1f%%) — ok:%d missing:%d error:%d skipped:%d",
                    db.value, i, total, pct,
                    summary["ok"], summary["missing"],
                    summary["error"], summary["skipped"],
                )

    logging.info(
        "Done. ok=%d missing=%d error=%d skipped=%d → %s",
        summary["ok"], summary["missing"], summary["error"],
        summary["skipped"], csv_path,
    )
    return records, summary

# Single-record test

def test_id(record_id: int, db: DB = DB.PADLI, verify_tls: bool = False) -> dict | None:
    """
    Fetch and pretty-print a single record.  Returns the parsed dict.

    >>> test_id(1,   db=DB.PADLI)        # fallen Habsburg soldier
    >>> test_id(668, db=DB.PADLI)
    """
    session = make_session(verify_tls=verify_tls)
    url = _url(db, record_id)
    print(f"Fetching: {url}")
    r = fetch_with_retries(session, url)
    if r is None:
        print("ERROR: no response after retries")
        return None
    if _is_missing(r):
        print(f"MISSING: no record at ID {record_id}")
        return None
    record = parse_record(r.text, record_id, db)
    if record is None:
        print("ERROR: page fetched but parser returned None")
        return None
    width = max(len(k) for k in record)
    for k, v in record.items():
        print(f"  {k:<{width}}  {v}")
    return record

def validate_output(output_dir: str | Path, db: DB = DB.PADLI, print_output: bool = True) -> dict:
    """
    Audit a completed scrape: record count, null rates, ID coverage.
    """
    output_dir = Path(output_dir)
    stem        = _DB_CONFIG[db]["default_stem"]
    csv_path    = output_dir / f"{stem}.csv"
    failed_path = output_dir / f"{stem}_failed.txt"

    if not csv_path.exists():
        raise FileNotFoundError(csv_path)

    import pandas as pd
    df = pd.read_csv(csv_path, low_memory=False)

    n           = len(df)
    id_min      = df["vua_id"].min()
    id_max      = df["vua_id"].max()
    null_rates  = (df.isnull().sum() / n * 100).round(1).sort_values(ascending=False)

    missing_count = 0
    error_count   = 0
    if failed_path.exists():
        with open(failed_path) as f:
            for line in f:
                tag = line.strip().split("\t")[-1]
                if tag == "MISSING":
                    missing_count += 1
                elif tag == "ERROR":
                    error_count += 1

    result = {
        "records":       n,
        "id_min":        id_min,
        "id_max":        id_max,
        "missing_ids":   missing_count,
        "error_ids":     error_count,
        "null_rates":    null_rates.to_dict(),
    }

    if print_output:
        print(f"Database   : {_DB_CONFIG[db]['label_cs']}")
        print(f"CSV        : {csv_path}")
        print(f"Records    : {n:,}")
        print(f"ID range   : {id_min} – {id_max}")
        print(f"Missing IDs: {missing_count:,}")
        print(f"Error IDs  : {error_count:,}")
        print("\nNull rates (top 15):")
        for col, pct in list(null_rates.items())[:15]:
            if pct > 0:
                print(f"  {col:<55}  {pct:.1f}%")

    return result

def build_parser() -> argparse.ArgumentParser:
    """Define the command-line interface for both vuapraha.cz databases.

    The key option is ``--db``: this scraper serves the *legionari* and *padli* (fallen
    WWI) databases, which share an HTML structure but have independent ID spaces —
    hence ``--autodetect-max`` to probe the ceiling rather than assume it. Defaults to
    ``padli``, the fallen database used as this thesis's comparison population.

    Returns:
        The configured parser; :func:`main` parses with it.
    """
    p = argparse.ArgumentParser(
        description="Scrape vuapraha.cz legionnaire or Habsburg fallen database.",
        formatter_class=argparse.ArgumentDefaultsHelpFormatter,
    )
    p.add_argument(
        "--db",
        choices=[db.value for db in DB],
        default=DB.PADLI.value,
        help="Target database (default: padli — the fallen WWI DB).",
    )
    p.add_argument("--start-id",  type=int, default=1)
    p.add_argument("--end-id",    type=int, default=200_000,
                   help="Inclusive upper ID bound (probe with --autodetect-max).")
    p.add_argument("--output-dir", default="data/raw/habsburg")
    p.add_argument("--delay",      type=float, default=1.0,
                   help="Seconds between requests.")
    p.add_argument("--no-resume",  action="store_true",
                   help="Re-scrape IDs already in the output CSV.")
    p.add_argument("--verify-tls", action="store_true",
                   help="Enable TLS cert verification (off by default; the "
                        "server's gov-PKI chain is rejected by modern OpenSSL).")
    p.add_argument("--progress",   choices=["tqdm", "log", "none"], default="tqdm")
    p.add_argument("--log-interval", type=int, default=500)
    p.add_argument("--test-id",    type=int, metavar="ID",
                   help="Fetch and print a single record, then exit.")
    p.add_argument("--autodetect-max", action="store_true",
                   help="Binary-search for the highest valid ID, then exit.")
    p.add_argument("--validate",   action="store_true",
                   help="Print validation report for an existing output file, then exit.")
    p.add_argument("--log-level",  default="INFO",
                   choices=["DEBUG", "INFO", "WARNING", "ERROR"])
    return p


def main(argv: list[str] | None = None) -> None:
    """CLI entry point — configure logging, pick the database, dispatch the mode.

    ``--test-id`` short-circuits to a single-record dump before any bulk work; otherwise
    the arguments are forwarded to the range scrape. Because a full run takes hours, the
    log level is set up first so progress is visible from the very first request.

    Args:
        argv: argument list; ``None`` reads ``sys.argv`` (the normal CLI path).
    """
    args = build_parser().parse_args(argv)
    logging.basicConfig(
        level=getattr(logging, args.log_level),
        format="%(asctime)s %(levelname)s %(message)s",
        datefmt="%H:%M:%S",
    )

    db = DB(args.db)

    if args.test_id:
        test_id(args.test_id, db=db, verify_tls=args.verify_tls)
        return

    if args.autodetect_max:
        session = make_session(verify_tls=args.verify_tls)
        ceiling = autodetect_max_id(session, db)
        print(f"Estimated max ID for {db.value}: {ceiling}")
        return

    if args.validate:
        validate_output(args.output_dir, db=db)
        return

    scrape_range(
        start_id   = args.start_id,
        end_id     = args.end_id,
        db         = db,
        output_dir = args.output_dir,
        delay      = args.delay,
        resume     = not args.no_resume,
        progress   = args.progress,
        log_interval = args.log_interval,
        verify_tls = args.verify_tls,
    )


if __name__ == "__main__":
    main()
