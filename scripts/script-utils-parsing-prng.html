How it works

Source: PRNG_MIEJSCOWOSCI_XLSX.zip from GUGiK, https://opendata.geoportal.gov.pl/prng/ — the localities (miejscowości) half of the register, refreshed daily; the geoportal states the data are "bezpłatnie i do dowolnego wykorzystania" (free, for any use). Coordinates are given in WGS 84 as degrees-minutes-seconds text (50°43'05" 16°39'17", latitude first), read by parse_dms; the PL-1992 pair beside them is not needed.

Only the area of the 1910 Silesian courts is kept — the caller passes the MPIDR court locator, and a locality is kept when its point falls inside a court of SILESIA_OKRES_IDS.

Output: prng_places [prng_id] — name, kind, parent locality, gmina/powiat, lat/lon, the German name where PRNG records one, and the 1910 court_id / okres_id it falls in.

def read_prng(xlsx_path) -> pd.DataFrame:
    """Read the localities workbook (``openpyxl``, read-only) into a frame, Silesian voivodeships
    only. *xlsx_path* may be the downloaded zip itself; the workbook is read from inside it."""
    import io
    import zipfile

    import openpyxl

    if str(xlsx_path).endswith(".zip"):
        with zipfile.ZipFile(xlsx_path) as zf:
            member = next(n for n in zf.namelist() if n.endswith("_XLSX.xlsx"))
            xlsx_path = io.BytesIO(zf.read(member))
    wb = openpyxl.load_workbook(xlsx_path, read_only=True)
    try:
        rows = wb[wb.sheetnames[0]].iter_rows(values_only=True)
        header = list(next(rows))
        idx = {out: header.index(src) for src, out in _COLS.items()}
        iw = idx["wojewodztwo"]
        recs = [{out: r[i] for out, i in idx.items()} for r in rows if r[iw] in _VOIVODESHIPS]
    finally:
        wb.close()
    df = pd.DataFrame(recs)
    ll = df["dms"].map(parse_dms)
    df["lat"] = [a for a, _b in ll]
    df["lon"] = [b for _a, b in ll]
    df["prng_id"] = df["prng_id"].astype("int64")
    for c in ("name_historical", "name_other", "name_other_lang"):
        df[c] = df[c].replace("", None)
    return df.drop(columns=["dms"])

def silesia_okres_ids(districts: pd.DataFrame) -> set[int]:
    """``okres_id`` (``region * 1000 + district``) of :data:`SILESIA_OKRES_CS` in the backbone."""
    rows = districts[districts["okres_cs"].isin(SILESIA_OKRES_CS)].drop_duplicates("okres_code")
    if set(rows["okres_cs"]) != set(SILESIA_OKRES_CS):
        raise ValueError(f"backbone lacks {set(SILESIA_OKRES_CS) - set(rows['okres_cs'])}")
    return {int(r) * 1000 + int(d) for r, d in rows["okres_code"].str.split("-")}

"""
prng.py — Polish localities from the PRNG (Państwowy Rejestr Nazw Geograficznych), Těšín Silesia
================================================================================================
1910 Austrian Silesia did not end at today's Czech border: the eastern half of the Teschen
duchy — okres Bílsko and most of okres Těšín, with parts of Frýštát — went to Poland in 1920.
The ČSÚ lexicon and RÚIAN stop at the border, so for those 1910 parts the only official modern
gazetteer is the Polish one.

Source: ``PRNG_MIEJSCOWOSCI_XLSX.zip`` from GUGiK, https://opendata.geoportal.gov.pl/prng/ —
the localities (*miejscowości*) half of the register, refreshed daily; the geoportal states the
data are "bezpłatnie i do dowolnego wykorzystania" (free, for any use). Coordinates are given in
WGS 84 as degrees-minutes-seconds text (``50°43'05" 16°39'17"``, latitude first), read by
:func:`parse_dms`; the PL-1992 pair beside them is not needed.

Only the area of the 1910 Silesian courts is kept — the caller passes the MPIDR court locator,
and a locality is kept when its point falls inside a court of :data:`SILESIA_OKRES_IDS`.

Output: ``prng_places`` [``prng_id``] — name, kind, parent locality, gmina/powiat, lat/lon,
the German name where PRNG records one, and the 1910 ``court_id`` / ``okres_id`` it falls in.
"""

from __future__ import annotations

import re

import numpy as np
import pandas as pd

__version__ = "1.0.0"   # first build (2026-10-29)

# Columns read from the XLSX (Polish headers) → output names.
_COLS = {
    "identyfikator PRNG": "prng_id",
    "nazwa główna": "name",
    "rodzaj obiektu": "kind",
    "nazwa miejscowości nadrzędnej": "parent_name",
    "nazwa historyczna": "name_historical",
    "nazwa dodatkowa": "name_other",
    "język nazwy dodatkowej": "name_other_lang",
    "współrzędne geograficzne": "dms",
    "województwo": "wojewodztwo",
    "powiat": "powiat",
    "gmina": "gmina",
}
# The two voivodeships that hold any of 1910 Austrian Silesia; a cheap first filter.
_VOIVODESHIPS = {"śląskie", "małopolskie"}
_DMS = re.compile(r"(\d+)°(\d+)'(\d+(?:[.,]\d+)?)\"")


def parse_dms(s) -> tuple[float, float]:
    """``'50°43\\'05" 16°39\\'17"'`` → ``(lat, lon)`` in decimal degrees; NaNs if unreadable."""
    if not isinstance(s, str):
        return (np.nan, np.nan)
    m = _DMS.findall(s)
    if len(m) < 2:
        return (np.nan, np.nan)
    (a, b, c), (d, e, f) = m[:2]
    lat = int(a) + int(b) / 60 + float(c.replace(",", ".")) / 3600
    lon = int(d) + int(e) / 60 + float(f.replace(",", ".")) / 3600
    return (lat, lon)

# The 1910 Silesian okresy that lie wholly or partly in today's Poland: Bílsko 303-1,
# Bílsko (město) 303-2, Fryštát 303-3, Těšín 303-9. Resolved by name against the backbone in
# :func:`silesia_okres_ids` — kept as names, not codes, so a renumbered backbone fails loudly.
SILESIA_OKRES_CS = ("Bílsko", "Bílsko (město)", "Fryštát", "Těšín")


def silesia_okres_ids(districts: pd.DataFrame) -> set[int]:

    rows = districts[districts["okres_cs"].isin(SILESIA_OKRES_CS)].drop_duplicates("okres_code")
    if set(rows["okres_cs"]) != set(SILESIA_OKRES_CS):
        raise ValueError(f"backbone lacks {set(SILESIA_OKRES_CS) - set(rows['okres_cs'])}")
    return {int(r) * 1000 + int(d) for r, d in rows["okres_code"].str.split("-")}
