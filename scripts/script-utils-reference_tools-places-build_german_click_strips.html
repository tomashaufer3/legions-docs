How it works

The page (german_click_page.html) shows one strip at a time; the author clicks every pictogram on it, optionally naming it, and the clicks are stored in section-image coordinates. Strips come from the section images the route itself cuts (output/frozen/de-bohemia_gold/sections), so a click needs no mapping before the locator or DINOv3 can use it. Offices sections of the German gold calibration and purposive pages only; never certification, never the reserve pages.

Two uses, in this order (the order of strips.json): 1. the oracle set — the first strips of the first page, every glyph named: does DINOv3 judge boxes that are certainly right? If it does not, no locator fix will pass the E3 gate. 2. the line labels — the rest, clicks only (naming optional): positives, and by omission the negatives, for a supervised glyph / not-glyph classifier.

def main(argv=None) -> int:
    import numpy as np
    from PIL import Image

    from build_token_review import text_lines

    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--pages", default="0142,0276,0480")
    parser.add_argument("--lines", type=int, default=3, help="printed lines per strip")
    parser.add_argument("--out", type=Path, default=OUT)
    args = parser.parse_args(argv)
    pages = [p for p in args.pages.split(",") if p]
    if any((GOLD_DE / role / f"{p}.md").exists() for p in pages for role in SEALED):
        raise SystemExit("refused: a sealed or reserve page")

    with (SECTIONS / "index.csv").open(encoding="utf-8", newline="") as fh:
        index = [r for r in csv.DictReader(fh) if r["page"] in pages and r["kind"] == "offices"]
    (args.out / "strips").mkdir(parents=True, exist_ok=True)
    (args.out / "glyph").mkdir(parents=True, exist_ok=True)

    strips, palette = [], {}
    for page in pages:
        palette[page] = [t for t, _n in page_tokens(page).most_common()]
        for row in sorted((r for r in index if r["page"] == page), key=lambda r: int(r["order"])):
            gray = Image.open(SECTIONS / row["image"]).convert("L")
            lines = text_lines(gray)
            for k in range(0, len(lines), args.lines):
                group = lines[k:k + args.lines]
                top, bottom = max(0, group[0][0] - PAD), min(gray.height, group[-1][1] + PAD)
                band = np.asarray(gray.crop((0, top, gray.width, bottom))) < 128
                sums = band.sum(axis=0)       # a column inked over most of the height is the printed frame
                cols = np.flatnonzero((sums >= 2) & (sums < 0.8 * band.shape[0]))
                right = min(gray.width, int(cols.max()) + MARGIN) if len(cols) else gray.width
                strip_id = f"{Path(row['image']).stem}_{top:04d}"
                gray.convert("RGB").crop((0, top, right, bottom)).save(
                    args.out / "strips" / f"{strip_id}.jpg", quality=86)
                strips.append({"id": strip_id, "page": page, "image": row["image"], "x0": 0, "y0": top,
                               "w": right, "h": bottom - top, "lines": len(group)})
    for n, strip in enumerate(strips):
        strip["oracle"] = n < ORACLE_STRIPS

    used = sorted({t for tokens in palette.values() for t in tokens})
    glyphs = {}
    for token in used:
        src = GLYPHS / f"{token.strip('[]')}.png"
        if src.exists():
            shutil.copy(src, args.out / "glyph" / src.name)
            glyphs[token] = f"glyph/{src.name}"
    data = {"built": "2026-10-31", "oracle_strips": ORACLE_STRIPS, "strips": strips,
            "palette": palette, "glyphs": glyphs}
    (args.out / "strips.json").write_text(json.dumps(data, ensure_ascii=False), encoding="utf-8")
    shutil.copy(TEMPLATE, args.out / "index.html") if TEMPLATE.exists() else None
    size = sum(p.stat().st_size for p in args.out.rglob("*") if p.is_file()) / 1e6
    print(f"{len(strips)} strips ({sum(s['lines'] for s in strips)} printed lines) from pages {pages}; "
          f"{len(used)} tokens, {len(glyphs)} glyph images; {size:.1f} MB in {args.out}")
    return 0
