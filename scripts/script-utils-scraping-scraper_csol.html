How it works

Notes on the data source: The csol.cz database is a community-enriched fork of VÚA-VHA Praha's Databáze legionářů (1997-2004, originally 128,141 names). csol.cz hosts sequential per-record pages at /legionar/{id}/ with photos and family annotations. The canonical archival version lives at vuapraha.cz, but that is search-only and not enumerable. Prefer Flosman's bulk export for methodological defensibility; scrape csol.cz only as a fallback or for the photo/annotation overlay.

Jupyter usage: >>> from scrape_csol import scrape_range, test_id, validate_output >>> records, summary = scrape_range( ... start_id=1, end_id=500, output_dir="./output", ... fmt="csv", delay=1.0, resume=True, progress="tqdm" ... ) >>> test_id(16930) # returns dict, prints fields >>> validate_output("./output") # prints a report

CLI usage: see python scrape_csol.py --help.

def parse_record(html: str, record_id: int) -> dict | None:
    """
    Extract labelled fields from a csol.cz record page.

    The page structure is:
        <p class="legionar-XXX"><span>Label:</span>Value</p>

    where Label ends with ':' and Value is the text content of the <p>
    following the <span>. The CSS class on <p> (e.g. legionar-jmeno) is a
    stable per-field identifier, but the human-readable Czech label inside
    the <span> is what we store as the column name — same wording as the
    archival source, and easier for the user to read.

    Returns a dict {field_name: value} or None if the page doesn't look like
    a real record (no <p class="legionar-*"> found — used for soft-404s).
    Repeated labels are suffixed _2, _3, ... (e.g. "Rakousko-uherský útvar
    v době zajetí" appears twice on most pages).

    The 'Doporučené odkazy' paragraph uses <strong> instead of <span>;
    we also capture it. Tracking links (legionar-databaziSpravuje) are
    skipped — they have no <span> label and contain site chrome.
    """
    soup = BeautifulSoup(html, "html.parser")

    # Scope to the record's content container if present, otherwise fall back
    # to the whole document. The real page uses <div class="legionar-content">.
    container = (
        soup.find("div", class_="legionar-content")
        or soup.find("div", class_="post-obsah")
        or soup.find("main")
        or soup
    )

    fields: dict[str, str] = {}
    label_counts: Counter[str] = Counter()

    # The <p> elements carrying fields all have a class beginning "legionar-".
    # NOTE: BeautifulSoup's class_= callable is invoked per-class-name (a str),
    # not per-list, so use a regex which it applies correctly.
    paragraphs = container.find_all("p", class_=re.compile(r"^legionar-"))
    for p in paragraphs:
        # Find the label tag: <span> for normal fields, <strong> for the
        # "Doporučené odkazy" block. Skip <p>s with no recognizable label.
        label_tag = p.find(["span", "strong"])
        if label_tag is None:
            continue

        label_text = label_tag.get_text(" ", strip=True)
        # Some labels include a stray </br> inside (e.g. "Zdroj informací:<br/>")
        # — get_text already drops the tag, but trailing whitespace remains.
        if ":" not in label_text:
            continue
        label = label_text.rstrip(": ").strip()
        if not label:
            continue

        # Value = full <p> text with label prefix removed. Using full text
        # rather than next_siblings handles cases where the value is wrapped
        # in nested tags (e.g. <a>) or split across line breaks.
        full = p.get_text(" ", strip=False)
        # Remove the leading label-with-colon (only once, only at start)
        idx = full.find(":")
        if idx == -1:
            continue
        value = full[idx + 1 :]
        value = re.sub(r"\s+", " ", value).strip()
        if not value:
            continue

        label_counts[label] += 1
        stored = label if label_counts[label] == 1 else f"{label}_{label_counts[label]}"
        fields[stored] = value

    if not paragraphs:
        return None

    # Soft-404 / stub page check: require at least one core identity field
    if not (RECORD_MARKERS & set(fields.keys())):
        return None

    fields["ID"] = str(record_id)
    return fields

def fetch_with_retries(
    session: requests.Session,
    url: str,
    timeout: float = 20.0,
    max_retries: int = 3,
) -> tuple[int, str | None, str | None]:
    """
    Returns (status_code, html_or_None, error_tag_or_None).
    error_tag: None on success, "MISSING" on 404, "ERROR:{code}" otherwise.
    """
    delays = [1, 2, 4]
    last_status = 0
    for attempt in range(max_retries + 1):
        try:
            r = session.get(url, timeout=timeout, allow_redirects=True)
            last_status = r.status_code
            if r.status_code == 200:
                return 200, r.text, None
            if r.status_code == 404:
                return 404, None, "MISSING"
        except requests.RequestException:
            last_status = -1
        if attempt < max_retries:
            time.sleep(delays[min(attempt, len(delays) - 1)])
    return last_status, None, f"ERROR:{last_status}"

def fetch_one(
    record_id: int, session: requests.Session | None = None
) -> ScrapeResult:
    """Scrape one record. Public API."""
    sess = session or make_session()
    url = BASE_URL.format(id=record_id)
    status, html, err = fetch_with_retries(sess, url)
    if err == "MISSING":
        return ScrapeResult(record_id, "missing", error_tag="MISSING")
    if err is not None:
        return ScrapeResult(record_id, "error", error_tag=err)
    record = parse_record(html or "", record_id)
    if record is None:
        return ScrapeResult(record_id, "parse_error", error_tag="PARSE_ERROR")
    record["scraped_at"] = datetime.now(timezone.utc).isoformat()
    record["scraper_version"] = SCRAPER_VERSION
    return ScrapeResult(record_id, "ok", record=record)

def autodetect_max_id(
    session: requests.Session | None = None, log: Callable[[str], None] = print
) -> int:
    """
    Find the largest valid ID by exponential + binary search.

    The csol.cz listing page is JS-rendered, so the total record count is
    not in static HTML. Direct probing costs ~30-40 requests and is robust.
    """
    sess = session or make_session()
    log("Auto-detecting max ID by probing...")

    lo, hi = 1, 1
    while hi < HARD_MAX_ID:
        if _probe(sess, hi):
            lo = hi
            hi *= 2
            time.sleep(0.5)
        else:
            break
    if hi >= HARD_MAX_ID and _probe(sess, HARD_MAX_ID):
        log(f"Warning: ID {HARD_MAX_ID} still exists; capping.")
        return HARD_MAX_ID

    hi = min(hi, HARD_MAX_ID)
    while hi - lo > 1:
        mid = (lo + hi) // 2
        any_exists = False
        for offset in (0, 1, -1, 2, -2):
            probe = mid + offset
            if probe <= lo or probe >= hi:
                continue
            if _probe(sess, probe):
                any_exists = True
                break
            time.sleep(0.3)
        if any_exists:
            lo = mid
        else:
            hi = mid
        time.sleep(0.3)

    log(f"Detected max ID: {lo}")
    return lo

class CSVWriter:
    """Append-mode CSV with on-the-fly header expansion."""

    def __init__(self, path: Path):
        self.path = path
        self.fieldnames: list[str] = []
        if path.exists() and path.stat().st_size > 0:
            with path.open("r", encoding="utf-8", newline="") as f:
                reader = csv.reader(f)
                try:
                    self.fieldnames = next(reader)
                except StopIteration:
                    pass

    def existing_ids(self) -> set[str]:
        if not self.path.exists() or "ID" not in self.fieldnames:
            return set()
        ids: set[str] = set()
        with self.path.open("r", encoding="utf-8", newline="") as f:
            for row in csv.DictReader(f):
                rid = row.get("ID")
                if rid:
                    ids.add(rid)
        return ids

    def write(self, record: dict) -> None:
        new_keys = [k for k in record.keys() if k not in self.fieldnames]
        if new_keys or not self.fieldnames:
            existing_rows: list[dict] = []
            if self.path.exists() and self.fieldnames:
                with self.path.open("r", encoding="utf-8", newline="") as f:
                    existing_rows = list(csv.DictReader(f))
            self.fieldnames.extend(new_keys)
            with self.path.open("w", encoding="utf-8", newline="") as f:
                w = csv.DictWriter(f, fieldnames=self.fieldnames, extrasaction="ignore")
                w.writeheader()
                for row in existing_rows:
                    w.writerow(row)
                w.writerow(record)
        else:
            with self.path.open("a", encoding="utf-8", newline="") as f:
                w = csv.DictWriter(f, fieldnames=self.fieldnames, extrasaction="ignore")
                w.writerow(record)

def write_xlsx_from_csv(csv_path: Path, xlsx_path: Path) -> None:
    """
    Materialize XLSX from a streamed CSV. Strips ASCII control characters
    that openpyxl rejects (rare, but legacy WordPress data has them).
    Per-record XLSX appending would require read-modify-write each iteration —
    prohibitive at 133k records — so we stream CSV and convert at the end.
    """
    import openpyxl

    ILLEGAL = re.compile(r"[\x00-\x08\x0B-\x0C\x0E-\x1F]")
    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = "legionnaires"
    with csv_path.open("r", encoding="utf-8", newline="") as f:
        for row in csv.reader(f):
            ws.append([ILLEGAL.sub("", c) if c else c for c in row])
    wb.save(xlsx_path)

            "fields": list(fields) if fields else None,
            "progress": progress,
        }, indent=2))
        log(f"Would scrape IDs {start_id}..{end_id} "
            f"({end_id - start_id + 1} IDs)")
        return ([] if return_records else None), summary

    csv_writer = CSVWriter(csv_path) if fmt in ("csv", "both") else None

    already_done: set[int] = set()
    if resume:
        if csv_writer is not None:
            already_done |= {int(x) for x in csv_writer.existing_ids() if x.isdigit()}
        already_done |= _existing_failed_ids(failed_path)
        summary.skipped_resume = len(already_done)
        log(f"Resume: skipping {len(already_done)} IDs.")

    to_attempt = [i for i in range(start_id, end_id + 1) if i not in already_done]
    log(f"Will attempt {len(to_attempt)} IDs in {start_id}..{end_id}.")

    if progress == "tqdm":
        try:
            from tqdm.auto import tqdm
            iterator = tqdm(to_attempt, desc="scraping", unit="id")
        except ImportError:
            log("tqdm not installed; falling back to 'print' progress.")
            iterator = to_attempt
            progress = "print"
    else:
        iterator = to_attempt

    request_times: list[float] = []
    collected: list[dict] = []
    start_t = time.monotonic()

    try:
        for rid in iterator:
            summary.attempted += 1

            t0 = time.monotonic()
            res = fetch_one(rid, session=sess)
            request_times.append(time.monotonic() - t0)
            if len(request_times) > 500:
                request_times = request_times[-500:]

            if res.status == "missing":
                summary.missing += 1
                _log_failure(failed_path, rid, "MISSING")
            elif res.status == "error":
                summary.errors += 1
                _log_failure(failed_path, rid, res.error_tag or "ERROR")
            elif res.status == "parse_error":
                summary.parse_errors += 1
                raw_errors_dir.mkdir(exist_ok=True)
                # We don't carry HTML on ScrapeResult to keep memory low.
                # If you need HTML dumps on parse errors, use iter_scrape +
                # custom handling, or set --format both / use ad-hoc refetch.
                _log_failure(failed_path, rid, "PARSE_ERROR")
            else:
                rec = res.record or {}
                if whitelist is not None:
                    rec = {k: v for k, v in rec.items() if k in whitelist}
                if csv_writer is not None:
                    csv_writer.write(rec)
                if return_records:
                    collected.append(rec)
                summary.scraped += 1

            time.sleep(max(0.0, delay + random.uniform(-0.3, 0.3)))

            if progress == "print" and summary.attempted % progress_every == 0:
                avg = sum(request_times) / max(len(request_times), 1)
                per = avg + delay
                rem = (len(to_attempt) - summary.attempted) * per
                log(
                    f"[{summary.attempted}/{len(to_attempt)}] "
                    f"ok={summary.scraped} missing={summary.missing} "
                    f"err={summary.errors} parse={summary.parse_errors} "
                    f"eta={_fmt_time(rem)} avg_req={avg:.2f}s"
                )

    except KeyboardInterrupt:
        log("Interrupted; partial progress saved.")

    summary.elapsed_seconds = time.monotonic() - start_t

    if fmt in ("xlsx", "both") and csv_path.exists():
        log(f"Writing XLSX to {xlsx_path}...")
        write_xlsx_from_csv(csv_path, xlsx_path)

    log(
        f"Done. scraped={summary.scraped} missing={summary.missing} "
        f"errors={summary.errors} parse_errors={summary.parse_errors} "
        f"elapsed={_fmt_time(summary.elapsed_seconds)}"
    )

    return (collected if return_records else None), summary

def validate_output(output_dir: str | Path, print_output: bool = True) -> dict:
    """
    Inspect an existing legionari.csv and report record count, ID gaps,
    duplicates, and per-field null rates.

    Returns a dict with the metrics.
    """
    out = Path(output_dir)
    csv_path = out / "legionnaires.csv"
    if not csv_path.exists():
        raise FileNotFoundError(csv_path)

    ids: list[int] = []
    null_counts: defaultdict[str, int] = defaultdict(int)
    total = 0
    fieldnames: list[str] = []

    with csv_path.open("r", encoding="utf-8", newline="") as f:
        reader = csv.DictReader(f)
        fieldnames = reader.fieldnames or []
        for row in reader:
            total += 1
            try:
                ids.append(int(row.get("ID", "")))
            except ValueError:
                pass
            for k in fieldnames:
                if not (row.get(k) or "").strip():
                    null_counts[k] += 1

    ids_sorted = sorted(ids)
    dups = [i for i, c in Counter(ids).items() if c > 1]
    if ids_sorted:
        rng = (ids_sorted[0], ids_sorted[-1])
        gaps = sorted(set(range(rng[0], rng[1] + 1)) - set(ids))
    else:
        rng = (None, None)
        gaps = []

    null_rates = {
        k: {"nulls": null_counts[k], "rate": (null_counts[k] / total if total else 0.0)}
        for k in fieldnames
    }

    report = {
        "total_records": total,
        "unique_ids": len(set(ids)),
        "duplicate_ids": dups,
        "id_range": rng,
        "gaps_count": len(gaps),
        "gaps_first10": gaps[:10],
        "null_rates": null_rates,
    }

    if print_output:
        print(f"Total records: {total}")
        print(f"Unique IDs: {len(set(ids))}")
        print(f"Duplicates: {len(dups)} (first 10: {dups[:10]})")
        if rng[0] is not None:
            print(f"ID range: {rng[0]}..{rng[1]}")
        print(f"Missing in range: {len(gaps)} (first 10: {gaps[:10]})")
        print("\nPer-field null rates:")
        for k in fieldnames:
            info = null_rates[k]
            print(f"  {k:<40} {info['nulls']:>7} ({100 * info['rate']:5.1f}%)")

    return report

    csv_path: str | None = None
    xlsx_path: str | None = None

    def as_dict(self) -> dict:
        return asdict(self)


# ---------------------------------------------------------------------------
# Parsing
# ---------------------------------------------------------------------------


def parse_record(html: str, record_id: int) -> dict | None:
    """
    Extract labelled fields from a csol.cz record page.

    The page structure is:
        <p class="legionar-XXX"><span>Label:</span>Value</p>

    where Label ends with ':' and Value is the text content of the <p>
    following the <span>. The CSS class on <p> (e.g. legionar-jmeno) is a
    stable per-field identifier, but the human-readable Czech label inside
    the <span> is what we store as the column name — same wording as the
    archival source, and easier for the user to read.

    Returns a dict {field_name: value} or None if the page doesn't look like
    a real record (no <p class="legionar-*"> found — used for soft-404s).
    Repeated labels are suffixed _2, _3, ... (e.g. "Rakousko-uherský útvar
    v době zajetí" appears twice on most pages).

    The 'Doporučené odkazy' paragraph uses <strong> instead of <span>;
    we also capture it. Tracking links (legionar-databaziSpravuje) are
    skipped — they have no <span> label and contain site chrome.
    """
    soup = BeautifulSoup(html, "html.parser")

    # Scope to the record's content container if present, otherwise fall back
    # to the whole document. The real page uses <div class="legionar-content">.
    container = (
        soup.find("div", class_="legionar-content")
        or soup.find("div", class_="post-obsah")
        or soup.find("main")
        or soup
    )

    fields: dict[str, str] = {}
    label_counts: Counter[str] = Counter()

    # The <p> elements carrying fields all have a class beginning "legionar-".
    # NOTE: BeautifulSoup's class_= callable is invoked per-class-name (a str),
    # not per-list, so use a regex which it applies correctly.
    paragraphs = container.find_all("p", class_=re.compile(r"^legionar-"))
    for p in paragraphs:
        # Find the label tag: <span> for normal fields, <strong> for the
        # "Doporučené odkazy" block. Skip <p>s with no recognizable label.
        label_tag = p.find(["span", "strong"])
        if label_tag is None:
            continue

        label_text = label_tag.get_text(" ", strip=True)
        # Some labels include a stray </br> inside (e.g. "Zdroj informací:<br/>")
        # — get_text already drops the tag, but trailing whitespace remains.
        if ":" not in label_text:
            continue
        label = label_text.rstrip(": ").strip()
        if not label:
            continue

        # Value = full <p> text with label prefix removed. Using full text
        # rather than next_siblings handles cases where the value is wrapped
        # in nested tags (e.g. <a>) or split across line breaks.
        full = p.get_text(" ", strip=False)
        # Remove the leading label-with-colon (only once, only at start)
        idx = full.find(":")
        if idx == -1:
            continue
        value = full[idx + 1 :]
        value = re.sub(r"\s+", " ", value).strip()
        if not value:
            continue

        label_counts[label] += 1
        stored = label if label_counts[label] == 1 else f"{label}_{label_counts[label]}"
        fields[stored] = value

    if not paragraphs:
        return None

    # Soft-404 / stub page check: require at least one core identity field
    if not (RECORD_MARKERS & set(fields.keys())):
        return None

    fields["ID"] = str(record_id)
    return fields

def make_session(user_agent: str = USER_AGENT) -> requests.Session:
    """Build the HTTP session used for every request.

    A single pooled session keeps the TCP connection alive across the ~133k record
    fetches. Identifies the scraper honestly via *user_agent* and asks for Czech
    first, since csol.cz negotiates the record language on ``Accept-Language``.
    """
    s = requests.Session()
    s.headers.update({"User-Agent": user_agent, "Accept-Language": "cs,en;q=0.7"})
    return s


def fetch_with_retries(
    session: requests.Session,
    url: str,
    timeout: float = 20.0,
    max_retries: int = 3,
) -> tuple[int, str | None, str | None]:
    """
    Returns (status_code, html_or_None, error_tag_or_None).
    error_tag: None on success, "MISSING" on 404, "ERROR:{code}" otherwise.
    """
    delays = [1, 2, 4]
    last_status = 0
    for attempt in range(max_retries + 1):
        try:
            r = session.get(url, timeout=timeout, allow_redirects=True)
            last_status = r.status_code
            if r.status_code == 200:
                return 200, r.text, None
            if r.status_code == 404:
                return 404, None, "MISSING"
        except requests.RequestException:
            last_status = -1
        if attempt < max_retries:
            time.sleep(delays[min(attempt, len(delays) - 1)])
    return last_status, None, f"ERROR:{last_status}"


def fetch_one(
    record_id: int, session: requests.Session | None = None
) -> ScrapeResult:
    """Scrape one record. Public API."""
    sess = session or make_session()
    url = BASE_URL.format(id=record_id)
    status, html, err = fetch_with_retries(sess, url)
    if err == "MISSING":
        return ScrapeResult(record_id, "missing", error_tag="MISSING")
    if err is not None:
        return ScrapeResult(record_id, "error", error_tag=err)
    record = parse_record(html or "", record_id)
    if record is None:
        return ScrapeResult(record_id, "parse_error", error_tag="PARSE_ERROR")
    record["scraped_at"] = datetime.now(timezone.utc).isoformat()
    record["scraper_version"] = SCRAPER_VERSION
    return ScrapeResult(record_id, "ok", record=record)

# Auto-detect end ID

def _probe(session: requests.Session, record_id: int) -> bool:
    url = BASE_URL.format(id=record_id)
    status, html, _ = fetch_with_retries(session, url, max_retries=1)
    return status == 200 and html is not None and parse_record(html, record_id) is not None


def autodetect_max_id(
    session: requests.Session | None = None, log: Callable[[str], None] = print
) -> int:
    """
    Find the largest valid ID by exponential + binary search.

    The csol.cz listing page is JS-rendered, so the total record count is
    not in static HTML. Direct probing costs ~30-40 requests and is robust.
    """
    sess = session or make_session()
    log("Auto-detecting max ID by probing...")

    lo, hi = 1, 1
    while hi < HARD_MAX_ID:
        if _probe(sess, hi):
            lo = hi
            hi *= 2
            time.sleep(0.5)
        else:
            break
    if hi >= HARD_MAX_ID and _probe(sess, HARD_MAX_ID):
        log(f"Warning: ID {HARD_MAX_ID} still exists; capping.")
        return HARD_MAX_ID

    hi = min(hi, HARD_MAX_ID)
    while hi - lo > 1:
        mid = (lo + hi) // 2
        any_exists = False
        for offset in (0, 1, -1, 2, -2):
            probe = mid + offset
            if probe <= lo or probe >= hi:
                continue
            if _probe(sess, probe):
                any_exists = True
                break
            time.sleep(0.3)
        if any_exists:
            lo = mid
        else:
            hi = mid
        time.sleep(0.3)

    log(f"Detected max ID: {lo}")
    return lo

class CSVWriter:
    """Append-mode CSV with on-the-fly header expansion."""

    def __init__(self, path: Path):
        self.path = path
        self.fieldnames: list[str] = []
        if path.exists() and path.stat().st_size > 0:
            with path.open("r", encoding="utf-8", newline="") as f:
                reader = csv.reader(f)
                try:
                    self.fieldnames = next(reader)
                except StopIteration:
                    pass

    def existing_ids(self) -> set[str]:
        if not self.path.exists() or "ID" not in self.fieldnames:
            return set()
        ids: set[str] = set()
        with self.path.open("r", encoding="utf-8", newline="") as f:
            for row in csv.DictReader(f):
                rid = row.get("ID")
                if rid:
                    ids.add(rid)
        return ids

    def write(self, record: dict) -> None:
        new_keys = [k for k in record.keys() if k not in self.fieldnames]
        if new_keys or not self.fieldnames:
            existing_rows: list[dict] = []
            if self.path.exists() and self.fieldnames:
                with self.path.open("r", encoding="utf-8", newline="") as f:
                    existing_rows = list(csv.DictReader(f))
            self.fieldnames.extend(new_keys)
            with self.path.open("w", encoding="utf-8", newline="") as f:
                w = csv.DictWriter(f, fieldnames=self.fieldnames, extrasaction="ignore")
                w.writeheader()
                for row in existing_rows:
                    w.writerow(row)
                w.writerow(record)
        else:
            with self.path.open("a", encoding="utf-8", newline="") as f:
                w = csv.DictWriter(f, fieldnames=self.fieldnames, extrasaction="ignore")
                w.writerow(record)


def write_xlsx_from_csv(csv_path: Path, xlsx_path: Path) -> None:
    """
    Materialize XLSX from a streamed CSV. Strips ASCII control characters
    that openpyxl rejects (rare, but legacy WordPress data has them).
    Per-record XLSX appending would require read-modify-write each iteration —
    prohibitive at 133k records — so we stream CSV and convert at the end.
    """
    import openpyxl

    ILLEGAL = re.compile(r"[\x00-\x08\x0B-\x0C\x0E-\x1F]")
    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = "legionnaires"
    with csv_path.open("r", encoding="utf-8", newline="") as f:
        for row in csv.reader(f):
            ws.append([ILLEGAL.sub("", c) if c else c for c in row])
    wb.save(xlsx_path)


# ---------------------------------------------------------------------------
# Failure log
# ---------------------------------------------------------------------------


def _log_failure(failed_path: Path, record_id: int, tag: str) -> None:
    with failed_path.open("a", encoding="utf-8") as f:
        f.write(f"{record_id}\t{tag}\n")


def _existing_failed_ids(failed_path: Path) -> set[int]:
    if not failed_path.exists():
        return set()
    ids: set[int] = set()
    with failed_path.open("r", encoding="utf-8") as f:
        for line in f:
            s = line.strip()
            if not s:
                continue
            try:
                ids.add(int(s.split("\t", 1)[0]))
            except ValueError:
                pass
    return ids

def iter_scrape(
    start_id: int,
    end_id: int,
    delay: float = 1.0,
    session: requests.Session | None = None,
    skip_ids: Iterable[int] = (),
) -> Iterator[ScrapeResult]:
    """
    Yield ScrapeResult objects one ID at a time. No file output; caller
    handles persistence. Good for ad-hoc Jupyter exploration:

        >>> from scrape_csol import iter_scrape
        >>> results = [r for r in iter_scrape(1, 50) if r.status == "ok"]
        >>> df = pd.DataFrame([r.record for r in results])
    """
    sess = session or make_session()
    skip = set(skip_ids)
    for rid in range(start_id, end_id + 1):
        if rid in skip:
            continue
        yield fetch_one(rid, session=sess)
        time.sleep(max(0.0, delay + random.uniform(-0.3, 0.3)))

def scrape_range(
    start_id: int = 1,
    end_id: int | None = None,
    output_dir: str | Path = "./output",
    fmt: Literal["csv", "xlsx", "both"] = "csv",
    delay: float = 1.0,
    resume: bool = False,
    fields: Sequence[str] | None = None,
    dry_run: bool = False,
    progress: Literal["none", "print", "tqdm"] = "print",
    progress_every: int = 100,
    return_records: bool = False,
    session: requests.Session | None = None,
    log: Callable[[str], None] = print,
) -> tuple[list[dict] | None, ScrapeSummary]:
    """
    Scrape a range of legionnaire IDs.

    Parameters
    ----------
    start_id, end_id
        Inclusive ID range. If end_id is None, auto-detects by probing.
    output_dir
        Folder for legionari.csv, legionari.xlsx, failed_ids.txt, raw_errors/.
    fmt
        "csv" streams during the run. "xlsx" or "both" additionally writes
        legionari.xlsx at the end (converted from the CSV).
    delay
        Base seconds between requests; actual delay is delay ± 0.3s jitter.
    resume
        Skip IDs already in the CSV or in failed_ids.txt.
    fields
        Optional whitelist of field names to retain. ID, scraped_at, and
        scraper_version are always retained.
    dry_run
        Print resolved config and the would-be range, then return early.
    progress
        "tqdm" wraps the loop in a tqdm bar (Jupyter-friendly).
        "print" emits a line every `progress_every` IDs.
        "none" stays silent.
    return_records
        If True, also return a list of all successfully scraped records.
        Default False — at 133k records this would be substantial RAM.

    Returns
    -------
    (records_or_None, ScrapeSummary)
    """
    out = Path(output_dir)
    out.mkdir(parents=True, exist_ok=True)
    csv_path = out / "legionnaires.csv"
    xlsx_path = out / "legionnaires.xlsx"
    failed_path = out / "failed_ids.txt"
    raw_errors_dir = out / "raw_errors"

    sess = session or make_session()

    if end_id is None:
        end_id = autodetect_max_id(sess, log=log)

    summary = ScrapeSummary(
        start_id=start_id,
        end_id=end_id,
        output_dir=str(out),
        csv_path=str(csv_path) if fmt in ("csv", "both") else None,
        xlsx_path=str(xlsx_path) if fmt in ("xlsx", "both") else None,
    )

    if start_id > end_id:
        log(f"start_id ({start_id}) > end_id ({end_id}); nothing to do.")
        return ([] if return_records else None), summary

    whitelist = (set(fields) | {"ID", "scraped_at", "scraper_version"}) if fields else None

    if dry_run:
        log("Resolved config:")
        log(json.dumps({
            "start_id": start_id, "end_id": end_id, "output_dir": str(out),
            "fmt": fmt, "delay": delay, "resume": resume,
            "fields": list(fields) if fields else None,
            "progress": progress,
        }, indent=2))
        log(f"Would scrape IDs {start_id}..{end_id} "
            f"({end_id - start_id + 1} IDs)")
        return ([] if return_records else None), summary

    csv_writer = CSVWriter(csv_path) if fmt in ("csv", "both") else None

    already_done: set[int] = set()
    if resume:
        if csv_writer is not None:
            already_done |= {int(x) for x in csv_writer.existing_ids() if x.isdigit()}
        already_done |= _existing_failed_ids(failed_path)
        summary.skipped_resume = len(already_done)
        log(f"Resume: skipping {len(already_done)} IDs.")

    to_attempt = [i for i in range(start_id, end_id + 1) if i not in already_done]
    log(f"Will attempt {len(to_attempt)} IDs in {start_id}..{end_id}.")

    if progress == "tqdm":
        try:
            from tqdm.auto import tqdm
            iterator = tqdm(to_attempt, desc="scraping", unit="id")
        except ImportError:
            log("tqdm not installed; falling back to 'print' progress.")
            iterator = to_attempt
            progress = "print"
    else:
        iterator = to_attempt

    request_times: list[float] = []
    collected: list[dict] = []
    start_t = time.monotonic()

    try:
        for rid in iterator:
            summary.attempted += 1

            t0 = time.monotonic()
            res = fetch_one(rid, session=sess)
            request_times.append(time.monotonic() - t0)
            if len(request_times) > 500:
                request_times = request_times[-500:]

            if res.status == "missing":
                summary.missing += 1
                _log_failure(failed_path, rid, "MISSING")
            elif res.status == "error":
                summary.errors += 1
                _log_failure(failed_path, rid, res.error_tag or "ERROR")
            elif res.status == "parse_error":
                summary.parse_errors += 1
                raw_errors_dir.mkdir(exist_ok=True)
                # We don't carry HTML on ScrapeResult to keep memory low.
                # If you need HTML dumps on parse errors, use iter_scrape +
                # custom handling, or set --format both / use ad-hoc refetch.
                _log_failure(failed_path, rid, "PARSE_ERROR")
            else:
                rec = res.record or {}
                if whitelist is not None:
                    rec = {k: v for k, v in rec.items() if k in whitelist}
                if csv_writer is not None:
                    csv_writer.write(rec)
                if return_records:
                    collected.append(rec)
                summary.scraped += 1

            time.sleep(max(0.0, delay + random.uniform(-0.3, 0.3)))

            if progress == "print" and summary.attempted % progress_every == 0:
                avg = sum(request_times) / max(len(request_times), 1)
                per = avg + delay
                rem = (len(to_attempt) - summary.attempted) * per
                log(
                    f"[{summary.attempted}/{len(to_attempt)}] "
                    f"ok={summary.scraped} missing={summary.missing} "
                    f"err={summary.errors} parse={summary.parse_errors} "
                    f"eta={_fmt_time(rem)} avg_req={avg:.2f}s"
                )

    except KeyboardInterrupt:
        log("Interrupted; partial progress saved.")

    summary.elapsed_seconds = time.monotonic() - start_t

    if fmt in ("xlsx", "both") and csv_path.exists():
        log(f"Writing XLSX to {xlsx_path}...")
        write_xlsx_from_csv(csv_path, xlsx_path)

    log(
        f"Done. scraped={summary.scraped} missing={summary.missing} "
        f"errors={summary.errors} parse_errors={summary.parse_errors} "
        f"elapsed={_fmt_time(summary.elapsed_seconds)}"
    )

    return (collected if return_records else None), summary

def test_id(
    record_id: int,
    print_output: bool = True,
    debug: bool = False,
    debug_dir: str | Path = "/tmp",
) -> dict | None:
    """
    Scrape a single record and (optionally) pretty-print it.

    Returns the parsed dict, or None on failure.

    If debug=True, the raw HTML is saved to {debug_dir}/csol_{id}.html and the
    path is printed. Use this when a record fails to parse — open the file and
    inspect the actual tag structure to update the parser.
    """
    sess = make_session()
    url = BASE_URL.format(id=record_id)
    status, html, err = fetch_with_retries(sess, url)

    if debug:
        debug_path = Path(debug_dir) / f"csol_{record_id}.html"
        debug_path.parent.mkdir(parents=True, exist_ok=True)
        debug_path.write_text(html or "", encoding="utf-8")
        if print_output:
            print(f"[debug] HTTP {status}, HTML saved to {debug_path} ({len(html or '')} bytes)")

    if err == "MISSING":
        if print_output:
            print(f"ID {record_id}: missing (404)")
        return None
    if err is not None:
        if print_output:
            print(f"ID {record_id}: error ({err})")
        return None

    rec = parse_record(html or "", record_id)
    if rec is None:
        if print_output:
            print(f"ID {record_id}: page loaded but no record fields parsed.")
            if not debug:
                print("    Re-run with debug=True to dump the HTML for inspection.")
        return None

    rec["scraped_at"] = datetime.now(timezone.utc).isoformat()
    rec["scraper_version"] = SCRAPER_VERSION

    if print_output:
        width = max(len(k) for k in rec.keys())
        print(f"Record {record_id}:")
        print("-" * 72)
        for k, v in rec.items():
            print(f"  {k:<{width}}  {v}")
        print("-" * 72)
        print(f"{len(rec)} fields")
    return rec

def validate_output(output_dir: str | Path, print_output: bool = True) -> dict:
    """
    Inspect an existing legionari.csv and report record count, ID gaps,
    duplicates, and per-field null rates.

    Returns a dict with the metrics.
    """
    out = Path(output_dir)
    csv_path = out / "legionnaires.csv"
    if not csv_path.exists():
        raise FileNotFoundError(csv_path)

    ids: list[int] = []
    null_counts: defaultdict[str, int] = defaultdict(int)
    total = 0
    fieldnames: list[str] = []

    with csv_path.open("r", encoding="utf-8", newline="") as f:
        reader = csv.DictReader(f)
        fieldnames = reader.fieldnames or []
        for row in reader:
            total += 1
            try:
                ids.append(int(row.get("ID", "")))
            except ValueError:
                pass
            for k in fieldnames:
                if not (row.get(k) or "").strip():
                    null_counts[k] += 1

    ids_sorted = sorted(ids)
    dups = [i for i, c in Counter(ids).items() if c > 1]
    if ids_sorted:
        rng = (ids_sorted[0], ids_sorted[-1])
        gaps = sorted(set(range(rng[0], rng[1] + 1)) - set(ids))
    else:
        rng = (None, None)
        gaps = []

    null_rates = {
        k: {"nulls": null_counts[k], "rate": (null_counts[k] / total if total else 0.0)}
        for k in fieldnames
    }

    report = {
        "total_records": total,
        "unique_ids": len(set(ids)),
        "duplicate_ids": dups,
        "id_range": rng,
        "gaps_count": len(gaps),
        "gaps_first10": gaps[:10],
        "null_rates": null_rates,
    }

    if print_output:
        print(f"Total records: {total}")
        print(f"Unique IDs: {len(set(ids))}")
        print(f"Duplicates: {len(dups)} (first 10: {dups[:10]})")
        if rng[0] is not None:
            print(f"ID range: {rng[0]}..{rng[1]}")
        print(f"Missing in range: {len(gaps)} (first 10: {gaps[:10]})")
        print("\nPer-field null rates:")
        for k in fieldnames:
            info = null_rates[k]
            print(f"  {k:<40} {info['nulls']:>7} ({100 * info['rate']:5.1f}%)")

    return report

def _fmt_time(seconds: float) -> str:
    if seconds < 0 or seconds != seconds:
        return "?"
    h = int(seconds // 3600)
    m = int((seconds % 3600) // 60)
    s = int(seconds % 60)
    return f"{h}h{m:02d}m" if h else f"{m}m{s:02d}s"

def build_parser() -> argparse.ArgumentParser:
    """Define the command-line interface (the CLI mirror of :func:`scrape_range`).

    Every option maps to a ``scrape_range`` argument, except the three that select a
    different mode entirely: ``--test-id`` (pretty-print one record and exit),
    ``--validate`` (re-check an existing output folder), and ``--dry-run`` (resolve the
    config and print the would-be range without fetching anything).

    Returns:
        The configured parser; :func:`main` parses with it.
    """
    p = argparse.ArgumentParser(description="Scrape legionnaire records from csol.cz.")
    p.add_argument("--output-dir", default="./output")
    p.add_argument("--format", choices=["csv", "xlsx", "both"], default="csv")
    p.add_argument("--start-id", type=int, default=1)
    p.add_argument("--end-id", type=int, default=None)
    p.add_argument("--test-id", type=int, default=None)
    p.add_argument("--delay", type=float, default=1.0)
    p.add_argument("--resume", action="store_true")
    p.add_argument("--dry-run", action="store_true")
    p.add_argument("--fields", default=None, help="Comma-separated field whitelist.")
    p.add_argument("--validate", action="store_true")
    p.add_argument("--progress", choices=["none", "print", "tqdm"], default="print")
    return p


def main(argv: list[str] | None = None) -> None:
    """CLI entry point — dispatch to one of the three modes, else run the scrape.

    Order matters: ``--test-id`` and ``--validate`` short-circuit before any scraping,
    so they are safe to run against a live output folder. Otherwise the arguments are
    forwarded to :func:`scrape_range`, which does the work.

    Args:
        argv: argument list; ``None`` reads ``sys.argv`` (the normal CLI path).
    """
    args = build_parser().parse_args(argv)

    if args.test_id is not None:
        test_id(args.test_id)
        return

    if args.validate:
        validate_output(args.output_dir)
        return

    fields_list = (
        [f.strip() for f in args.fields.split(",")] if args.fields else None
    )

    scrape_range(
        start_id=args.start_id,
        end_id=args.end_id,
        output_dir=args.output_dir,
        fmt=args.format,
        delay=args.delay,
        resume=args.resume,
        fields=fields_list,
        dry_run=args.dry_run,
        progress=args.progress,
    )


if __name__ == "__main__":
    main()
