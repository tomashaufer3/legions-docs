How it works

The Spezialortsrepertorium von Böhmen (Wien 1915) is the Czech volume's twin and prints the same three documents in the same layout — the table, the two apparatus columns under it (Ortsbestandteile im Gerichtsbezirk X: and Ämter, Anstalten usw. im Gerichtsbezirk X:), and the postal and railway legend across the foot. What it lacks is a text layer fit to cut on: the UBW scan's OCR words are too poor to find a heading in (D-67), so every boundary has to come from the pixels and every label from a reading of the heading itself.

This module is the German half of what channel G of run_lexikon_1910_pilot.py does for the Czech volume, built from the same parts rather than beside them:

  • where a section starts is lexikon_1910_image (the geometry measured on four volumes, D-80), through lexikon_1910_notes.image_section_segments with an empty text layer, so every heading the image finds is cut out and read;
  • which section it opens is the German profile's heading matcher (volumes/boehmen_de.toml, VolumeProfile.find_heading) applied to that reading — the same rule as the Czech one, with this volume's words;
  • what the reader is told is the Czech prompt with the German volume's own heading words, abbreviations and legend, and the closed token list of the de pictogram rows (80 symbols, read off the German plate, scan 0012);
  • which court a German heading names is answered by the project's court crosswalk (reference/places/district_cs_curation.csv), so a German entry is keyed by the same court slug as the Czech one and the German gold sheets (which file every section under soudní okres X) — never by page (D-67).

It also reads the column header once a page and checks it against the profile's seventeen labels, which D-80 and D-83 left as the German reader's job: the table geometry is found in the image, and only a reading says the columns are the ones the profile expects.

def build_system_de(abbreviations, section: str = "", atlas: bool = False,
                    think: bool = False, contrast: bool = False,
                    volume: str = "boehmen_de") -> str:
    """The German apparatus prompt: the Czech builder's, with this volume's words.

    Everything after the head — the atlas note (the contrast note when a
    contrast sheet follows the key), the note for a reader that thinks, and the
    closed token list — is ``lexikon_1910_vision``'s own text,
    so the two editions are told the same things about the symbols and differ
    only in what their volume prints. ``abbreviations`` must be the ``de`` rows:
    the token list is read off them, and the German plate has no
    ``[cooperative]``, ``[port]`` or ``[rc-parish-exp]``.
    """
    head = SECTION_HEADS_DE.get(section, _SHARED_HEAD_DE)
    if volume in VOLUMES_DE:                         # Böhmen's text is left byte for byte
        text, coop = VOLUMES_DE[volume]
        head = head.replace(_VOLUME, text).replace(_COOP_LINE_BOEHMEN, coop)
    if contrast and not atlas:
        raise ValueError("a contrast sheet is read against the key: it needs atlas=True")
    if atlas:
        head += (lxv._CONTRAST_NOTE if contrast else lxv._ATLAS_NOTE).format(
            unknown=lxv.UNKNOWN_TOKEN)
    if think:
        head += lxv._THINK_NOTE.format(unknown=lxv.UNKNOWN_TOKEN)
    rows = lxv.pictograms(abbreviations, section=section)
    if not rows:
        raise ValueError("no pictogram rows in the German abbreviation table — load "
                         "it with lang='de'")
    listing = "\n".join(f"    {a.code:<16} {a.english}" for a in rows)
    return head + lxv._ANNOTATE_TAIL.format(
        vocabulary=listing, unknown=lxv.UNKNOWN_TOKEN, example=rows[0].code)

def cut_page(image, stem: str, profile, carry=None, read_heading=None):
    """``(segments, report)``: the page's sections cut on the image, plus its legend.

    Every heading is labelled by ``read_heading(crop) -> str`` through the
    profile's matcher; one it cannot label is reported in
    ``report.headings_image_unread`` and the section before it runs on (D-38).
    The postal legend under the divider is appended as a ``legend`` segment, as
    the pilot's ``--notes-legend`` does. A page whose image shows no divider
    has no apparatus to cut and returns no segments — there is no text layer to
    fall back to.
    """
    segments, report = lxnt.image_section_segments(
        empty_notes(stem), None, image, carry=carry, read_heading=read_heading,
        find=profile.find_heading, full_width_headings=True)
    if report.geometry != "image":
        return [], report
    if report.legend:
        segments.append(lxnt.Segment(kind="legend", district="", page=stem,
                                     region="footer", box=report.legend,
                                     order=900, lines=[]))
    return segments, report

def last_heading_crop(image):
    """The last section heading on a page, as a crop to read, or ``None``.

    What a section carried onto the next page opened under. Reading order is the
    left column then the right, so the last heading of the right column wins,
    and the left column's last only where the right has none.
    """
    from . import lexikon_1910_image as lximg

    geo = lximg.detect(image, full_width_headings=True)
    columns = geo.apparatus()
    for region in ("right", "left"):
        heads = [h for h in geo.headings if h.region == region]
        if heads and region in columns:
            left, _top, right, _foot = columns[region]
            h = max(heads, key=lambda h: h.top)
            pad = lxnt.HEADING_PAD
            return image.crop((max(left, h.left - pad), max(0, h.top - pad),
                               min(right, h.right + pad),
                               min(image.height, h.bottom + pad)))
    return None

def header_check(lines: list[str], profile) -> dict:
    """Which of the profile's column labels the header reading contains.

    A record, never a gate: a label the reader missed is reported by name, so a
    page whose columns are not the profile's (a statutory city's 15, D-80) shows
    as many missing labels rather than as a silently mis-assigned table.
    """
    from rapidfuzz import fuzz

    text = lxnt.normalise(" ".join(str(line) for line in lines))
    missing = []
    for column in profile.columns:
        key = lxnt.normalise(column.label)
        if not key:
            continue
        if key in text or (text and fuzz.partial_ratio(key, text) >= HEADER_MATCH):
            continue
        missing.append(column.name)
    return {"columns": len(profile.columns),
            "found": len(profile.columns) - len(missing),
            "missing": missing}

def name_forms(name: str) -> set[str]:
    """Every spelling a German court name may be matched under: as printed and with its
    abbreviations written out, each folded and plainly normalised."""
    forms = set()
    for variant in {str(name or ""), expand_court_abbreviations(name)}:
        forms |= {fold(variant), lxnt.normalise(variant)}
    return {form for form in forms if form}

def load_court_crosswalk(curation: Path = CURATION,
                         districts: Path | None = DISTRICTS) -> dict[str, str]:
    """``{German name form: Czech court name}`` for every court in the backbone.

    Two sources, and a form both claim for **different** courts is dropped
    rather than given to either — a German name that is two Czech courts is a
    question, not an answer.
    """
    pairs: list[tuple[str, str]] = []
    with Path(curation).open(encoding="utf-8", newline="") as fh:
        for row in csv.DictReader(fh):
            if row.get("level") == "court" and row.get("cs"):
                pairs += [(row.get("key", ""), row["cs"]), (row.get("de", ""), row["cs"])]
    if districts is not None and Path(districts).exists():
        with Path(districts).open(encoding="utf-8", newline="") as fh:
            for row in csv.DictReader(fh):
                if row.get("gerichtsbezirk_cs"):
                    pairs.append((row.get("gerichtsbezirk_de", ""),
                                  row["gerichtsbezirk_cs"]))
    out: dict[str, str] = {}
    clash: set[str] = set()
    for german, czech in pairs:
        for form in name_forms(german):
            if form in out and lxnt.normalise(out[form]) != lxnt.normalise(czech):
                clash.add(form)
            out.setdefault(form, czech)
    for form in clash:
        out.pop(form, None)
    return out

def court_name(german: str, crosswalk: dict[str, str]) -> tuple[str, str]:
    """``(Czech court name, how)`` for a court named in a German heading.

    ``how`` is ``exact``, ``prefix``, ``fuzzy`` or ``unmapped``. ``prefix`` is a
    heading that prints a court's name short (``Gablonz`` for *Gablonz an der
    Neiße*): the read begins exactly one court's name, by at least
    :data:`PREFIX_MIN` letters. An unmapped court returns ``""`` and the caller
    keeps the German name: its entries then share no key with the gold and are
    reported, never guessed onto a neighbour.
    """
    from rapidfuzz import fuzz

    for form in name_forms(german):
        if form in crosswalk:
            return crosswalk[form], "exact"
    key = fold(expand_court_abbreviations(german))
    if len(key) < 3:
        return "", "unmapped"
    if len(key) >= PREFIX_MIN:
        begun = {lxnt.normalise(czech) for form, czech in crosswalk.items()
                 if form.startswith(key)}
        if len(begun) == 1:
            return next(czech for form, czech in crosswalk.items()
                        if form.startswith(key)), "prefix"
    best: dict[str, float] = {}
    for form, czech in crosswalk.items():
        score = fuzz.ratio(key, form)
        best[czech] = max(best.get(czech, 0.0), score)
    ranked = sorted(best.items(), key=lambda kv: -kv[1])
    if ranked and ranked[0][1] >= COURT_FUZZY and (
            len(ranked) == 1 or ranked[0][1] - ranked[1][1] >= COURT_MARGIN):
        return ranked[0][0], "fuzzy"
    return "", "unmapped"

def relabel_sections(sections, crosswalk: dict[str, str]) -> list[dict]:
    """Put each section under its Czech court name; return what was done.

    The decoder keys an entry on ``normalise(section.district)``, and the gold
    sheets key it on the Czech court (``soudní okres Trutnov`` -> ``trutnov``),
    so this is the step that makes a German entry and a gold entry the same
    entry. The German name read off the heading is kept in the returned rows.
    """
    log = []
    for section in sections:
        czech, how = court_name(section.district, crosswalk)
        log.append({"page": section.page, "kind": section.kind,
                    "court_de": section.district, "court_cs": czech, "how": how})
        if czech:
            section.district = czech
    return log

"""The German edition's apparatus, read from the image alone (D-92).

The *Spezialortsrepertorium von Böhmen* (Wien 1915) is the Czech volume's twin
and prints the same three documents in the same layout — the table, the two
apparatus columns under it (``Ortsbestandteile im Gerichtsbezirk X:`` and
``Ämter, Anstalten usw. im Gerichtsbezirk X:``), and the postal and railway
legend across the foot. What it lacks is a text layer fit to cut on: the UBW
scan's OCR words are too poor to find a heading in (D-67), so every boundary has
to come from the pixels and every label from a reading of the heading itself.

This module is the German half of what channel G of
``run_lexikon_1910_pilot.py`` does for the Czech volume, built from the same
parts rather than beside them:

* **where** a section starts is ``lexikon_1910_image`` (the geometry measured on
  four volumes, D-80), through ``lexikon_1910_notes.image_section_segments`` with
  an empty text layer, so every heading the image finds is cut out and read;
* **which** section it opens is the German profile's heading matcher
  (``volumes/boehmen_de.toml``, ``VolumeProfile.find_heading``) applied to that
  reading — the same rule as the Czech one, with this volume's words;
* **what the reader is told** is the Czech prompt with the German volume's own
  heading words, abbreviations and legend, and the closed token list of the
  ``de`` pictogram rows (80 symbols, read off the German plate, scan 0012);
* **which court** a German heading names is answered by the project's court
  crosswalk (``reference/places/district_cs_curation.csv``), so a German entry is
  keyed by the same court slug as the Czech one and the German gold sheets
  (which file every section under ``soudní okres X``) — never by page (D-67).

It also reads the **column header** once a page and checks it against the
profile's seventeen labels, which D-80 and D-83 left as the German reader's job:
the table geometry is found in the image, and only a reading says the columns
are the ones the profile expects.
"""

           "(Special Place Register for Bohemia), printed in Vienna in 1915, in "
           "German. Place names are often printed in German and Czech together "
           '("Altenbuch, Staré Buky").')

_RULES = """\
Rules:
- Transcribe exactly what is printed. Do not translate, expand an abbreviation, \
correct a spelling, or reorder anything.
- Keep the line breaks of the print. Do not join lines and do not reflow the text. \
A word broken across two lines stays broken, with its hyphen.
- If a character is genuinely illegible, write "?" in its place. Do not guess a \

_SHARED_HEAD_DE = f"""\

* section headings — "Ortsbestandteile im Gerichtsbezirk X:" and "Ämter, \

* house-population pairs for the parts of a settlement — "58—386"
* altitudes above sea level — "425 m"
* a numbered postal and railway legend introduced by two symbols and "zum GB. \
X:" — "1) Trautenau. 2) Parschnitz. ..."
* printed pictograms standing for an office, a school, a doctor, and so on

Return every printed line as one string, in printed order, top to bottom.

{_RULES}\
- Keep every number exactly as printed, including the key numbers and their \

- Write the dash between a house count and a population count as an em dash: 58—386.

After the key comes a list separated by commas, mixing:

* ordinary abbreviations — "BG.", "Sta.", "Notar", "2 Adv.", "Gymn.", "Bsch.", \

#: across several header rows and spanned by group labels, so the reply is asked

HEADER_SYSTEM_DE = f"""\

The image shows only the header: the labels printed above the columns, some of \

#: The volumes a German prompt names beyond Böhmen (0.4.0, D-146): Mähren and Schlesien
#: had been read with a prompt that called them Böhmen and their cooperatives bare letters.
#: Both print the cooperative mark before the letters — "stets in Verbindung mit dem
#: vorangestellten Zeichen" — and add "p", polnisch, to the language letters.
_COOP_LINE_BOEHMEN = ('* single capital letters naming registered cooperatives, set in the list among '
                      'the other items — "R", "K", "B", "G", "L", "S", "Sch" — sometimes with a small '
                      '"d" or "b" after them')
_COOP_LINE_MARKED = ('* the registered-cooperative mark followed by capital letters naming the kinds of '
                     'cooperative — "R", "K", "B", "G", "L", "S", "Sch" — each sometimes with a small '
                     '"d", "b" or "p" after it (deutsch, böhmisch, polnisch)')
#: ``{profile slug: (the volume as the prompt names it, the cooperative line)}``.
VOLUMES_DE = {
    "maehren_de": ("the 1910 Austrian census volume *Spezialortsrepertorium von Mähren* "
                   "(Special Place Register for Moravia), printed in Vienna, in German. Place "
                   "names are often printed in German and Czech together.", _COOP_LINE_MARKED),
    "schlesien_de": ("the 1910 Austrian census volume *Spezialortsrepertorium von Schlesien* "
                     "(Special Place Register for Silesia), printed in Vienna, in German. Place "
                     "names are often printed in German, Czech and Polish together.",
                     _COOP_LINE_MARKED),
}


def build_system_de(abbreviations, section: str = "", atlas: bool = False,
                    think: bool = False, contrast: bool = False,
                    volume: str = "boehmen_de") -> str:
    """The German apparatus prompt: the Czech builder's, with this volume's words.

    Everything after the head — the atlas note (the contrast note when a
    contrast sheet follows the key), the note for a reader that thinks, and the
    closed token list — is ``lexikon_1910_vision``'s own text,
    so the two editions are told the same things about the symbols and differ
    only in what their volume prints. ``abbreviations`` must be the ``de`` rows:
    the token list is read off them, and the German plate has no
    ``[cooperative]``, ``[port]`` or ``[rc-parish-exp]``.
    """
    head = SECTION_HEADS_DE.get(section, _SHARED_HEAD_DE)
    if volume in VOLUMES_DE:                         # Böhmen's text is left byte for byte
        text, coop = VOLUMES_DE[volume]
        head = head.replace(_VOLUME, text).replace(_COOP_LINE_BOEHMEN, coop)
    if contrast and not atlas:
        raise ValueError("a contrast sheet is read against the key: it needs atlas=True")
    if atlas:
        head += (lxv._CONTRAST_NOTE if contrast else lxv._ATLAS_NOTE).format(
            unknown=lxv.UNKNOWN_TOKEN)
    if think:
        head += lxv._THINK_NOTE.format(unknown=lxv.UNKNOWN_TOKEN)
    rows = lxv.pictograms(abbreviations, section=section)
    if not rows:
        raise ValueError("no pictogram rows in the German abbreviation table — load "
                         "it with lang='de'")
    listing = "\n".join(f"    {a.code:<16} {a.english}" for a in rows)
    return head + lxv._ANNOTATE_TAIL.format(
        vocabulary=listing, unknown=lxv.UNKNOWN_TOKEN, example=rows[0].code)

def native_width(pdf_path, scan: int) -> int:
    """The width of the scan image embedded in a PDF page, in its own pixels."""
    import fitz

    with fitz.open(str(pdf_path)) as doc:
        return doc[scan - 1].get_images(full=True)[0][2]


def render_native(pdf_path, scan: int):
    """One page at the scan's native resolution (the profile's ``render = native``)."""
    from . import lexikon_1910 as lx

    return lx.render_page(pdf_path, scan, native_width(pdf_path, scan))


def strip_height(czech_height: int, scale: float) -> int:
    """A strip ceiling in this page's pixels holding as much print as
    ``czech_height`` holds on the Czech render.

    ``scale`` is ``lexikon_1910_image``'s: the page's frame over the Czech
    reference frame. The German scans are ~0.75 of the Czech render, so the
    bake-off's 500 px strips become ~375 px here and a strip still holds the
    same number of printed lines.
    """
    return max(120, round(czech_height * scale))


def empty_notes(stem: str) -> lxnt.Notes:
    """A text layer with nothing in it, for a volume that has none to cut on."""
    return lxnt.Notes(page=stem, top=0, bottom=0, split_x=0)


def cut_page(image, stem: str, profile, carry=None, read_heading=None):
    """``(segments, report)``: the page's sections cut on the image, plus its legend.

    Every heading is labelled by ``read_heading(crop) -> str`` through the
    profile's matcher; one it cannot label is reported in
    ``report.headings_image_unread`` and the section before it runs on (D-38).
    The postal legend under the divider is appended as a ``legend`` segment, as
    the pilot's ``--notes-legend`` does. A page whose image shows no divider
    has no apparatus to cut and returns no segments — there is no text layer to
    fall back to.
    """
    segments, report = lxnt.image_section_segments(
        empty_notes(stem), None, image, carry=carry, read_heading=read_heading,
        find=profile.find_heading, full_width_headings=True)
    if report.geometry != "image":
        return [], report
    if report.legend:
        segments.append(lxnt.Segment(kind="legend", district="", page=stem,
                                     region="footer", box=report.legend,
                                     order=900, lines=[]))
    return segments, report


def last_heading_crop(image):
    """The last section heading on a page, as a crop to read, or ``None``.

    What a section carried onto the next page opened under. Reading order is the
    left column then the right, so the last heading of the right column wins,
    and the left column's last only where the right has none.
    """
    from . import lexikon_1910_image as lximg

    geo = lximg.detect(image, full_width_headings=True)
    columns = geo.apparatus()
    for region in ("right", "left"):
        heads = [h for h in geo.headings if h.region == region]
        if heads and region in columns:
            left, _top, right, _foot = columns[region]
            h = max(heads, key=lambda h: h.top)
            pad = lxnt.HEADING_PAD
            return image.crop((max(left, h.left - pad), max(0, h.top - pad),
                               min(right, h.right + pad),
                               min(image.height, h.bottom + pad)))
    return None


def header_crop(image, geo):
    """The table's column header: the frame's width, the table's top to the
    header's foot. ``None`` where the geometry found no header."""
    if geo.frame is None or geo.table_top is None or geo.header_bottom is None:
        return None
    if geo.header_bottom - geo.table_top < 20:
        return None
    return image.crop((geo.frame[0], geo.table_top, geo.frame[1], geo.header_bottom))


#: A printed label counts as found when its letters occur in the reply, or
#: nearly so: a partial ratio at least this high (rapidfuzz, normalised text).
HEADER_MATCH = 85


def header_check(lines: list[str], profile) -> dict:
    """Which of the profile's column labels the header reading contains.

    A record, never a gate: a label the reader missed is reported by name, so a
    page whose columns are not the profile's (a statutory city's 15, D-80) shows
    as many missing labels rather than as a silently mis-assigned table.
    """
    from rapidfuzz import fuzz

    text = lxnt.normalise(" ".join(str(line) for line in lines))
    missing = []
    for column in profile.columns:
        key = lxnt.normalise(column.label)
        if not key:
            continue
        if key in text or (text and fuzz.partial_ratio(key, text) >= HEADER_MATCH):
            continue
        missing.append(column.name)
    return {"columns": len(profile.columns),
            "found": len(profile.columns) - len(missing),
            "missing": missing}


# --------------------------------------------------------------------------- #
# Which court: German heading -> the Czech court slug
# --------------------------------------------------------------------------- #

_UMLAUTS = str.maketrans({"ä": "ae", "ö": "oe", "ü": "ue", "Ä": "Ae", "Ö": "Oe",
                          "Ü": "Ue", "ß": "ss"})


def fold(name: str) -> str:
    """A German name as the curation spells it: umlauts written out, then
    ``lexikon_1910_notes.normalise`` (``Mühlhausen`` -> ``muehlhausen``)."""
    return lxnt.normalise(str(name or "").translate(_UMLAUTS))


#: How the German headings shorten a court name's first word — ``Böhm. Aicha``, read
#: once as ``Bhm. Aicha`` — expanded before the name is looked up (D-120). The 1910
#: court backbone's German names begin with these words in full.
_COURT_ABBREVIATIONS = [
    (re.compile(r"\bB(?:ö|oe|o)?hm?\.\s*", re.I), "Böhmisch "),
    (re.compile(r"\bM(?:ä|ae|a)hr\.\s*", re.I), "Mährisch "),
    (re.compile(r"\bUng\.\s*", re.I), "Ungarisch "),
    (re.compile(r"\bWall\.\s*", re.I), "Wallachisch "),
    (re.compile(r"\b(?:Kgl|Königl|Koenigl)\.\s*", re.I), "Königliche "),
    (re.compile(r"\bSt\.\s*"), "Sankt "),
]


def expand_court_abbreviations(name: str) -> str:
    """``Böhm. Aicha`` -> ``Böhmisch Aicha``; a name without an abbreviation is unchanged."""
    out = str(name or "")
    for pattern, full in _COURT_ABBREVIATIONS:
        out = pattern.sub(full, out)
    return re.sub(r"\s+", " ", out).strip()


def name_forms(name: str) -> set[str]:
    """Every spelling a German court name may be matched under: as printed and with its
    abbreviations written out, each folded and plainly normalised."""
    forms = set()
    for variant in {str(name or ""), expand_court_abbreviations(name)}:
        forms |= {fold(variant), lxnt.normalise(variant)}
    return {form for form in forms if form}


def load_court_crosswalk(curation: Path = CURATION,
                         districts: Path | None = DISTRICTS) -> dict[str, str]:
    """``{German name form: Czech court name}`` for every court in the backbone.

    Two sources, and a form both claim for **different** courts is dropped
    rather than given to either — a German name that is two Czech courts is a
    question, not an answer.
    """
    pairs: list[tuple[str, str]] = []
    with Path(curation).open(encoding="utf-8", newline="") as fh:
        for row in csv.DictReader(fh):
            if row.get("level") == "court" and row.get("cs"):
                pairs += [(row.get("key", ""), row["cs"]), (row.get("de", ""), row["cs"])]
    if districts is not None and Path(districts).exists():
        with Path(districts).open(encoding="utf-8", newline="") as fh:
            for row in csv.DictReader(fh):
                if row.get("gerichtsbezirk_cs"):
                    pairs.append((row.get("gerichtsbezirk_de", ""),
                                  row["gerichtsbezirk_cs"]))
    out: dict[str, str] = {}
    clash: set[str] = set()
    for german, czech in pairs:
        for form in name_forms(german):
            if form in out and lxnt.normalise(out[form]) != lxnt.normalise(czech):
                clash.add(form)
            out.setdefault(form, czech)
    for form in clash:
        out.pop(form, None)
    return out


#: A heading read that is not an exact form of any court is matched only when one
#: court scores at least this (rapidfuzz ratio on the folded name) and the next
#: different court at least :data:`COURT_MARGIN` below it. Otherwise unmapped.
COURT_FUZZY = 90
COURT_MARGIN = 5

#: The fewest letters a shortened court name must keep to be matched by prefix.
PREFIX_MIN = 6


def court_name(german: str, crosswalk: dict[str, str]) -> tuple[str, str]:
    """``(Czech court name, how)`` for a court named in a German heading.

    ``how`` is ``exact``, ``prefix``, ``fuzzy`` or ``unmapped``. ``prefix`` is a
    heading that prints a court's name short (``Gablonz`` for *Gablonz an der
    Neiße*): the read begins exactly one court's name, by at least
    :data:`PREFIX_MIN` letters. An unmapped court returns ``""`` and the caller
    keeps the German name: its entries then share no key with the gold and are
    reported, never guessed onto a neighbour.
    """
    from rapidfuzz import fuzz

    for form in name_forms(german):
        if form in crosswalk:
            return crosswalk[form], "exact"
    key = fold(expand_court_abbreviations(german))
    if len(key) < 3:
        return "", "unmapped"
    if len(key) >= PREFIX_MIN:
        begun = {lxnt.normalise(czech) for form, czech in crosswalk.items()
                 if form.startswith(key)}
        if len(begun) == 1:
            return next(czech for form, czech in crosswalk.items()
                        if form.startswith(key)), "prefix"
    best: dict[str, float] = {}
    for form, czech in crosswalk.items():
        score = fuzz.ratio(key, form)
        best[czech] = max(best.get(czech, 0.0), score)
    ranked = sorted(best.items(), key=lambda kv: -kv[1])
    if ranked and ranked[0][1] >= COURT_FUZZY and (
            len(ranked) == 1 or ranked[0][1] - ranked[1][1] >= COURT_MARGIN):
        return ranked[0][0], "fuzzy"
    return "", "unmapped"


def relabel_sections(sections, crosswalk: dict[str, str]) -> list[dict]:
    """Put each section under its Czech court name; return what was done.

    The decoder keys an entry on ``normalise(section.district)``, and the gold
    sheets key it on the Czech court (``soudní okres Trutnov`` -> ``trutnov``),
    so this is the step that makes a German entry and a gold entry the same
    entry. The German name read off the heading is kept in the returned rows.
    """
    log = []
    for section in sections:
        czech, how = court_name(section.district, crosswalk)
        log.append({"page": section.page, "kind": section.kind,
                    "court_de": section.district, "court_cs": czech, "how": how})
        if czech:
            section.district = czech
    return log
