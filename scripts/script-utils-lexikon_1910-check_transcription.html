How it works

Run it while transcribing, as often as you like. It reports typos against the curated token list, keys that no longer match the register, and how much of the worksheet is filled — and once it is clean it writes inventory_human_freeform.csv, in exactly the shape the model passes produce.

That last part is the point. The human becomes the fourth reader, so compare_symbol_passes.py and the family collapse consume it unchanged, and every figure the pilot reports stops being agreement between readers and becomes accuracy against a person. Nothing in this project has had that before.

The confidence marks are kept, not normalised away:

[school] certain [school]? probably this — recorded as confidence="unsure" [?] a symbol is printed and could not be identified [!] clearly a symbol, and not on the atlas

The trailing ? is what decides an argument the models cannot settle. If they fail where the transcriber was unsure, the limit is the 399 DPI scan and collapsing near-identical symbols into families is the correct resolution of the source; if they fail where the transcriber was confident, the limit is the readers and it is worth attacking. Same for [?]: a high rate says the distinction is not on the paper at the size it was printed.

def split_entries(body: str):
    """``[(name, muni_seq, settlement_seq, symbols), …]`` for one worksheet line.

    The name of an entry is whatever stands between the previous entry's symbols
    and this entry's key. The two are separated on the **last full stop** before
    the key, which is the convention the transcriber used and the one the register
    itself uses — every entry's symbol list is closed with a period before the next
    place is named.

    Where there is no full stop the whole span is taken as the name, not as
    symbols: a missing name costs a join, an invented symbol list corrupts the
    inventory, and between the two failures only one is recoverable later.
    """
    masked = _mask_parens(body)
    keys = list(_KEY_RE.finditer(masked))
    out = []
    carried = body[:keys[0].start()] if keys else ""
    for i, key in enumerate(keys):
        end = keys[i + 1].start() if i + 1 < len(keys) else len(body)
        span = body[key.end():end]
        # The span between this key and the next holds this entry's symbols *and*
        # the next entry's place name. Cut them apart on the last full stop and
        # carry the name forward — assigning the whole span to this entry instead
        # files the next place's topographic symbol as an institution of this one,
        # which is how `[ves]` first appeared as something Humburky *had*.
        if i + 1 < len(keys):
            cut = span.rfind(".")
            symbols, carry = (span[:cut + 1], span[cut + 1:]) if cut >= 0 else (span, "")
        else:
            symbols, carry = span, ""
        out.append((carried.strip(), int(key.group(1)),
                    (key.group(2) or "").strip(" ,—-–") or "0", symbols))
        carried = carry
    return out

def parse(path: Path, allowed: set[str]):
    """``(rows, problems, blank)`` from the worksheet."""
    rows: list[dict] = []
    problems: list[str] = []
    blank = 0
    for number, raw in enumerate(path.read_text(encoding="utf-8").splitlines(), 1):
        if raw.lstrip().startswith("#") or not raw.strip():
            continue
        match = _LINE_RE.match(raw)
        if not match:
            problems.append(f"  line {number}: not a worksheet row: {raw.strip()[:60]!r}")
            continue
        page, district, body = match.groups()
        entries = split_entries(body)
        if not entries:
            blank += 1
            continue
        municipality = ""
        for name_field, muni, settlement, symbols in entries:
            muni_name, settlement_name, topographic = _names(name_field)
            # The municipality is stated once, on the line's first entry; later
            # entries name only their own osada and inherit it.
            if muni_name and not settlement_name and not municipality:
                municipality, settlement_name = muni_name, muni_name
            elif muni_name and settlement_name:
                municipality = muni_name
            elif not settlement_name:
                settlement_name = muni_name
            if not symbols.strip():
                blank += 1
                continue
            position = 0
            for count, token, unsure in _TOKEN_RE.findall(symbols):
                canonical = lxv.canonical(token) if len(token) > 3 else token
                if canonical not in allowed and token not in (UNREADABLE, OFF_ATLAS):
                    problems.append(
                        f"  line {number}: {token!r} is not on the atlas — check "
                        f"the spelling against symbols/manifest.csv, or write "
                        f"{OFF_ATLAS} if the symbol is genuinely not there")
                    continue
                if canonical in TOPOGRAPHIC:
                    # Printed in the symbol list rather than against the name. It
                    # says what the place *is*; it is not an institution and must
                    # not be counted as one.
                    problems.append(
                        f"  line {number}: {canonical} is a topographic symbol in "
                        f"the institution list of {muni}, {settlement} — should it "
                        f"be against the name?")
                    continue
                rows.append({
                    "check": "I13_inventory", "page": page.zfill(4),
                    "district": district, "muni_seq": int(muni),
                    "settlement_seq": settlement, "entry_index": "",
                    "municipality": municipality, "settlement": settlement_name,
                    "topographic": " ".join(topographic),
                    "code": canonical, "count": int(count) if count else 1,
                    "position": position, "source": "human",
                    "confidence": ("unsure" if unsure else
                                   "unreadable" if token == UNREADABLE else
                                   "off-atlas" if token == OFF_ATLAS else "certain"),
                    "english": "", "category": "", "corroborated": "",
                    "count_text": "", "glyph": "",
                    "context": symbols.strip()[:80],
                })
                position += 1
    return rows, problems, blank

def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--sheet", default=str(SHEET))
    ap.add_argument("--write", action="store_true",
                    help="write inventory_human.csv for the comparison tools")
    args = ap.parse_args()

    path = Path(args.sheet)
    if not path.exists():
        print(f"no worksheet at {path}; run make_transcription_worksheet.py first")
        return 1

    abbreviations = lxin.load_abbreviations(ABBREVIATIONS, lang="cs")
    allowed = lxv.vocabulary(abbreviations) | {OFF_ATLAS}
    meaning = {a.code: (a.english, a.category) for a in abbreviations}

    rows, problems, blank = parse(path, allowed)
    # Keyed on the settlement too. Without it every osada of one municipality
    # collapses into a single "entry" and the sheet looks a fifth smaller than it
    # is — 47 municipalities standing in for 55 entries.
    entries = {(r["page"], r["district"], r["muni_seq"], r["settlement_seq"])
               for r in rows}
    places = {(r["page"], r["district"], r["settlement"]) for r in rows
              if r["settlement"]}
    print(f"  {len(rows)} symbols over {len(entries)} entries "
          f"({len(places)} named places); {blank} entries still blank")
    if problems:
        print(f"\n  {len(problems)} problems:")
        for problem in problems[:20]:
            print(problem)
        if len(problems) > 20:
            print(f"  ... {len(problems) - 20} more")

    confidence = Counter(r["confidence"] for r in rows)
    if rows:
        print()
        for kind in ("certain", "unsure", "unreadable", "off-atlas"):
            if confidence.get(kind):
                print(f"    {kind:11s} {confidence[kind]:5d}  "
                      f"{confidence[kind] / len(rows):5.1%}")
        unreadable = confidence.get("unreadable", 0) + confidence.get("unsure", 0)
        if unreadable / len(rows) > 0.25:
            print()
            print("  Over a quarter of the marks could not be read with "
                  "certainty by a person.")
            print("  That is a finding about the source, not about the "
                  "transcription: it says the")
            print("  distinction is not on the paper at this size, and the "
                  "family collapse is the")
            print("  right resolution rather than a compromise.")

        top = Counter(r["code"] for r in rows)
        print()
        print("  most frequent:")
        for code, n in top.most_common(10):
            print(f"    {n:4d}  {code:22s} {meaning.get(code, ('', ''))[0][:30]}")

    if args.write:
        if problems:
            print("\n  not writing: fix the problems above first")
            return 1
        for row in rows:
            row["english"], row["category"] = meaning.get(row["code"], ("", ""))
        OUT.parent.mkdir(parents=True, exist_ok=True)
        with OUT.open("w", newline="", encoding="utf-8") as fh:
            writer = csv.DictWriter(fh, fieldnames=list(rows[0]))
            writer.writeheader()
            writer.writerows(rows)
        print(f"\n  wrote {OUT}")
        print("  now: python utils/lexikon_1910/_archive/compare_symbol_passes.py")
    return 0

Run it while transcribing, as often as you like. It reports typos against the
curated token list, keys that no longer match the register, and how much of the
worksheet is filled — and once it is clean it writes
``inventory_human_freeform.csv``, in exactly the shape the model passes produce.

``compare_symbol_passes.py`` and the family collapse consume it unchanged, and
every figure the pilot reports stops being *agreement between readers* and becomes

The confidence marks are kept, not normalised away:

``[school]``   certain
``[school]?``  probably this — recorded as ``confidence="unsure"``
``[?]``        a symbol is printed and could not be identified
``[!]``        clearly a symbol, and not on the atlas

fail where the transcriber was unsure, the limit is the 399 DPI scan and
collapsing near-identical symbols into families is the correct resolution of the
source; if they fail where the transcriber was confident, the limit is the readers
and it is worth attacking. Same for ``[?]``: a high rate says the distinction is
not on the paper at the size it was printed.

    python utils/lexikon_1910/check_transcription.py --write

from __future__ import annotations

import argparse
import csv
import re
import sys
from collections import Counter
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from utils.lexikon_1910 import lexikon_1910_institutions as lxin    # noqa: E402
from utils.lexikon_1910 import lexikon_1910_vision as lxv           # noqa: E402

SHEET = (ROOT / "reference" / "places" / "transcription_gold"
         / "_superseded" / "freeform_2026-08" / "worksheet.txt")
ABBREVIATIONS = ROOT / "reference" / "places" / "institution_abbreviations.csv"
#: Its own file since 2026-10-05. This is the FIRST reading of the four development
#: pages; the scored gold, ``inventory_human.csv``, is now written from the
#: whole-page sheets by ``read_transcription_md.py --write``. Both used to write
#: that one name, so whichever ran last silently became the gold.
OUT = ROOT / "output" / "lexikon_1910_pilot" / "inventory_human_freeform.csv"

#: ``118  jirkov: <names and keys and symbols>`` — the line's page and district.
#:
#: The transcriber went **past** the sheet that was handed over, and the extra is
#: worth more than the part that was asked for. The generated sheet pre-filled one
#: key per line and nothing else; what came back carries, per line, the bold
#: municipality, then a colon, then each non-bold osada named immediately before
#: the key that belongs to it:
#:
#:     118  jirkov: Albrechtice, Ulbersdorf: Albrechtice, Ulbersdorf 1, 1) …
#:                  Černice, Tschernitz 1, 2) … Eisenberk, Eisenberg 1, 3) …
#:
#: That is the join the pipeline could not make on its own. An entry keyed ``1, 3)``
#: is meaningless without knowing that osada 3 of municipality 1 is Eisenberk, and
#: the OCR of the table's name column is exactly where that is least reliable. The
#: Návod (p. VII) confirms the convention is the source's own: "Nalézá-li se ústav
#: v části místa, která nesdílí jména osady, jest jméno oné části místa výslovně
#: uvedeno."
_LINE_RE = re.compile(r"^\s*(\d{3,4})\s+([^\s:]+)\s*:\s*(.*)$")

#: ``12, 3)`` — a municipality and settlement key.
#:
#: Parenthesised groups are **masked before this runs**. The register prints the
#: cadastral symbol with the osada numbers it covers — ``[cadastral] (3,5)``,
#: ``(1,3,4)`` — and those are digit-comma-digit-paren, which is exactly this
#: pattern. Unmasked, ``(1,3,4)`` yields a phantom entry "municipality 3,
#: settlement 4" that would be filed under a municipality that is not on the page.
_KEY_RE = re.compile(r"(\d{1,3})\s*,\s*([^)]{0,4})\)")

#: A token as written by hand, with its optional count and trailing uncertainty mark.
_TOKEN_RE = re.compile(r"(\d{1,2}\s+)?(\[[^\]]*\])(\?)?")

#: Tokens that describe **what a place is**, not what it has. In the name region
#: they are the source's topographic characteristic (Návod p. VII); they must never
#: reach the institution inventory, where ``[ves]`` would read as a village having
#: been *built* in the village.
TOPOGRAPHIC = {"[obec]", "[osada]", "[cadastral]", "[mesto]", "[mestys]", "[ves]",
               "[city-part]", "[suburb]", "[mestys-part]", "[hamlet]",
               "[manor-farm]", "[cottage-group]", "[house-group]", "[colony]",
               "[scattered]", "[solitary]", "[single-house]"}

#: Printed but unidentifiable, and printed but not on the atlas.
UNREADABLE = "[?]"
OFF_ATLAS = "[!]"


def _mask_parens(text: str) -> str:
    """Blank out ``(...)`` spans, preserving length so offsets still line up."""
    out = list(text)
    depth = 0
    for i, char in enumerate(text):
        if char == "(":
            depth += 1
        if depth:
            out[i] = " "
        if char == ")" and depth:
            depth -= 1
    return "".join(out)


def split_entries(body: str):
    """``[(name, muni_seq, settlement_seq, symbols), …]`` for one worksheet line.

    The name of an entry is whatever stands between the previous entry's symbols
    and this entry's key. The two are separated on the **last full stop** before
    the key, which is the convention the transcriber used and the one the register
    itself uses — every entry's symbol list is closed with a period before the next
    place is named.

    Where there is no full stop the whole span is taken as the name, not as
    symbols: a missing name costs a join, an invented symbol list corrupts the
    inventory, and between the two failures only one is recoverable later.
    """

    keys = list(_KEY_RE.finditer(masked))
    out = []
    carried = body[:keys[0].start()] if keys else ""
    for i, key in enumerate(keys):
        end = keys[i + 1].start() if i + 1 < len(keys) else len(body)
        span = body[key.end():end]
        # The span between this key and the next holds this entry's symbols *and*
        # the next entry's place name. Cut them apart on the last full stop and
        # carry the name forward — assigning the whole span to this entry instead
        # files the next place's topographic symbol as an institution of this one,
        # which is how `[ves]` first appeared as something Humburky *had*.

            cut = span.rfind(".")
            symbols, carry = (span[:cut + 1], span[cut + 1:]) if cut >= 0 else (span, "")
        else:
            symbols, carry = span, ""
        out.append((carried.strip(), int(key.group(1)),
                    (key.group(2) or "").strip(" ,—-–") or "0", symbols))
        carried = carry
    return out


def _names(name_field: str) -> tuple[str, str, list[str]]:
    """``(municipality, settlement, topographic tokens)`` from an entry's name.

    Only the first entry of a line carries both, separated by a colon: the bold
    municipality, then the osada the key belongs to. Later entries name the osada
    alone and inherit the municipality from the line.
    """
    tokens = [lxv.canonical(t) for _c, t, _u in _TOKEN_RE.findall(name_field)]
    text = _TOKEN_RE.sub(" ", name_field)
    text = re.sub(r"\([^)]*\)", " ", text)
    parts = [p.strip(" \t*)(,;.") for p in text.split(":")]
    parts = [p for p in parts if p]
    if len(parts) >= 2:
        return parts[0], parts[-1], tokens
    return (parts[0] if parts else ""), "", tokens

                    continue
                rows.append({
                    "check": "I13_inventory", "page": page.zfill(4),
                    "district": district, "muni_seq": int(muni),
                    "settlement_seq": settlement, "entry_index": "",
                    "municipality": municipality, "settlement": settlement_name,
                    "topographic": " ".join(topographic),
                    "code": canonical, "count": int(count) if count else 1,
                    "position": position, "source": "human",
                    "confidence": ("unsure" if unsure else
                                   "unreadable" if token == UNREADABLE else
                                   "off-atlas" if token == OFF_ATLAS else "certain"),
                    "english": "", "category": "", "corroborated": "",
                    "count_text": "", "glyph": "",
                    "context": symbols.strip()[:80],
                })
                position += 1
    return rows, problems, blank


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--sheet", default=str(SHEET))
    ap.add_argument("--write", action="store_true",
                    help="write inventory_human.csv for the comparison tools")
    args = ap.parse_args()

    path = Path(args.sheet)
    if not path.exists():
        print(f"no worksheet at {path}; run make_transcription_worksheet.py first")
        return 1

    abbreviations = lxin.load_abbreviations(ABBREVIATIONS, lang="cs")
    allowed = lxv.vocabulary(abbreviations) | {OFF_ATLAS}
    meaning = {a.code: (a.english, a.category) for a in abbreviations}

    rows, problems, blank = parse(path, allowed)
    # Keyed on the settlement too. Without it every osada of one municipality
    # collapses into a single "entry" and the sheet looks a fifth smaller than it
    # is — 47 municipalities standing in for 55 entries.
    entries = {(r["page"], r["district"], r["muni_seq"], r["settlement_seq"])
               for r in rows}
    places = {(r["page"], r["district"], r["settlement"]) for r in rows
              if r["settlement"]}
    print(f"  {len(rows)} symbols over {len(entries)} entries "
          f"({len(places)} named places); {blank} entries still blank")
    if problems:
        print(f"\n  {len(problems)} problems:")
        for problem in problems[:20]:
            print(problem)
        if len(problems) > 20:
            print(f"  ... {len(problems) - 20} more")

    confidence = Counter(r["confidence"] for r in rows)
    if rows:
        print()
        for kind in ("certain", "unsure", "unreadable", "off-atlas"):
            if confidence.get(kind):
                print(f"    {kind:11s} {confidence[kind]:5d}  "
                      f"{confidence[kind] / len(rows):5.1%}")
        unreadable = confidence.get("unreadable", 0) + confidence.get("unsure", 0)
        if unreadable / len(rows) > 0.25:
            print()
            print("  Over a quarter of the marks could not be read with "
                  "certainty by a person.")
            print("  That is a finding about the source, not about the "
                  "transcription: it says the")
            print("  distinction is not on the paper at this size, and the "
                  "family collapse is the")
            print("  right resolution rather than a compromise.")

        top = Counter(r["code"] for r in rows)
        print()
        print("  most frequent:")
        for code, n in top.most_common(10):
            print(f"    {n:4d}  {code:22s} {meaning.get(code, ('', ''))[0][:30]}")

    if args.write:
        if problems:
            print("\n  not writing: fix the problems above first")
            return 1
        for row in rows:
            row["english"], row["category"] = meaning.get(row["code"], ("", ""))
        OUT.parent.mkdir(parents=True, exist_ok=True)
        with OUT.open("w", newline="", encoding="utf-8") as fh:
            writer = csv.DictWriter(fh, fieldnames=list(rows[0]))
            writer.writeheader()
            writer.writerows(rows)
        print(f"\n  wrote {OUT}")
        print("  now: python utils/lexikon_1910/_archive/compare_symbol_passes.py")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
