How it works

The register keys a place on (court_key, muni_seq): the 1910 court and the number its section prints. Two faults gave one key to two municipalities, and the register then built one place out of both — the first table row's name and figures, both entries' institutions:

  • A court created after 1910, filed under the court it was cut from (D-122). The Czech volume prints Neustadt u Tafelfichte (7 municipalities, from Frýdlant, 1 July 1913) and Kralupy nad Vltavou (21, from Velvary, 1 March 1912) as courts of their own, each numbering its municipalities from 1. court_aliases.csv files both under their 1910 court, rightly, and so Neustadt's 1–7 shared Frýdlant's 1–7 and Kralupy's 1–21 Velvary's 1–20. renumber_frame and renumber_table give the later court's own number n the key OFFSET + n (1001…): the court stays the 1910 one, the number says the section, and no 1910 court reaches a thousand.
  • A court's opening entries filed under the court before it (the German volumes). Where the cutter missed a court heading inside a column, the entries below it kept the label of the court above; §3 (offices_resplit) finds the heading in the readers' text, but merge moves a row only where every reader's re-split loses it, and a page that continues the new court with no heading keeps the old label outright (Jungbunzlau's 8–23 on Böhmen 0149, labelled Sobotka). carried_entries moves such a row where both hold: a reader's re-split found a heading of court B on the page of court A's label (resplit/splits_&lt;label&gt;.csv, how = court), and the row's number is lower than the highest A had already reached before that page — A's own entries only rise, so a lower one is the next court's. It runs over the page and the pages after it that carry A's label without a break, and leaves a row where B already holds that number off those pages.

Both run on what notebook 17 hands on: merge's last step (apply) on the inventory, types and abbreviations, and register_rechecked.runs (mend_run) on the parts lists, the place attributes and the statistical table, which merge never sees. Nothing in the frozen route changes.

def table_segments(table: pd.DataFrame, courts: pd.DataFrame, parent: str, labels: set[str]) -> dict:
    """``{block_id: section label}`` for the table blocks of 1910 court ``parent`` where it holds a
    post-1910 section: the blocks in printed order, cut where the numbering starts again (a row 0
    numbered 1 or 2 after one numbered higher), each run given the court's sections in the order
    ``courts.csv`` first meets them. Empty where the runs and the sections do not count the same —
    the table's labels cannot be trusted there (Velvary's 14–20 on Czech 0199 read as Kralupy's)."""
    mine = courts[courts["court_id"].astype(str) == parent].copy()
    mine["label"] = mine["district"].map(lxnt.normalise)
    order = list(dict.fromkeys(mine.sort_values("page", kind="stable")["label"]))
    if not set(order) & labels:
        return {}
    pages = set(mine["page"].astype(str).str.zfill(4))
    known = {(str(r.page).zfill(4), r.label): str(r.court_id) for r in mine.itertuples()}
    others = {(str(r.page).zfill(4), lxnt.normalise(r.district)) for r in courts.itertuples()
              if str(r.court_id) != parent}
    def names(frame):
        cs = frame["court_cs"] if "court_cs" in frame else pd.Series(dtype=str)
        return {lxnt.normalise(x) for x in [*frame["district"], *cs] if str(x).strip()}

    own = names(mine) | {lxnt.normalise(l) for l in labels}
    foreign = names(courts[courts["court_id"].astype(str) != parent]) - own

    def theirs(page, label):
        lab = lxnt.normalise(label)
        return (page, lab) not in known and ((page, lab) in others or lab in foreign)

    blocks = [(b, n) for b, page, label, n in _row0_numbers(table)
              if page in pages and not theirs(page, label)]          # not a block of another court
    numbers = [n for _b, n in blocks]
    runs, current, last = [], [], None
    for i, (block_id, n) in enumerate(blocks):
        following = next((m for m in numbers[i + 1:] if m is not None), None)
        # a restart: 1 or 2 after a higher number, and the next number read is n + 1 — a lone misread
        # "1" between Velvary's 17 and 18 (Czech 0199) is none
        if (n is not None and last is not None and n <= 2 and last > n + 1 and following == n + 1
                and current):
            runs.append(current)
            current = []
        current.append(block_id)
        if n is not None:
            last = n
    if current:
        runs.append(current)
    if len(runs) != len(order):
        return {}
    return {b: order[i] for i, run in enumerate(runs) for b in run}

def renumber_table(table: pd.DataFrame, courts: pd.DataFrame, labels: set[str]) -> tuple[pd.DataFrame, dict]:
    """The statistical table with each post-1910 section's municipalities numbered ``OFFSET + n``.
    A block's row 0 opens a municipality; a later row whose number is neither the next part's nor
    lower opens the next one (the register's own reading, ``_add_table_rows``) — both offset; part
    rows keep their numbers. Returns ``(table, report)``."""
    report = {"courts": {}, "rows": 0, "court_rows": []}
    if table is None or table.empty or "channel" not in table or not labels:
        return table, report
    parents = {str(c) for c, d in zip(courts["court_id"], courts["district"])
               if lxnt.normalise(d) in labels and str(c).isdigit()}
    out = table.copy()
    for parent in sorted(parents):
        segments = table_segments(out, courts, parent, labels)
        if not segments:
            report["courts"][parent] = "left: the table's runs do not match the court's sections"
            continue
        n_rows = 0
        known = {(str(p).zfill(4), lxnt.normalise(d)) for p, d in zip(courts["page"], courts["district"])}
        for block_id, label in segments.items():
            page = str(out.loc[out["block_id"] == block_id, "page"].iloc[0]).zfill(4)
            if (page, label) not in known:             # the register files a table row by (page, label)
                known.add((page, label))
                report["court_rows"].append({"page": page, "district": label, "court_id": parent})
            if label not in labels:
                continue
            block = out[(out["block_id"] == block_id) & (out["channel"] == "A")]
            muni, expected = None, 1
            for idx, r in block.iterrows():
                seq = str(r["seq"]).strip()
                if not seq.isdigit():
                    continue
                n = int(seq)
                if str(r["row"]) == "0" or (muni is not None and n != expected and n > muni):
                    muni, expected = n, 1
                    out.at[idx, "seq"] = str(OFFSET + n)
                    n_rows += 1
                elif muni is not None and n == expected:
                    expected += 1
        report["courts"][parent] = f"{n_rows} municipalities renumbered"
        report["rows"] += n_rows
    return out, report

def court_splits(edition: str, recheck: Path) -> list[dict]:
    """``[{page, a, b}]``: every court heading a reader's re-split found inside another court's
    section, both courts as section labels, once per page and pair. A heading of a post-1910 court
    (*Kralup an der Moldau* on Böhmen 0212–0213) counts too: the re-split recorded it as a heading
    of the crop's own court (``how = heading``) only because the crosswalk holds no such 1910 court."""
    from utils.lexikon_1910 import lexikon_1910_de_notes as lxdn
    crosswalk = lxdn.load_court_crosswalk()
    later = post1910_labels(edition)

    def label(german: str) -> str:
        czech = lxdn.court_name(german, crosswalk)[0]
        if czech:
            return lxnt.normalise(czech)
        return lxnt.normalise(german) if lxnt.normalise(german) in later else ""

    found: dict[tuple, set] = {}
    paths = sorted(glob.glob(str(Path(recheck) / edition / "resplit" / "splits_*.csv")))
    for path in paths:
        for r in _read(Path(path)):
            a, b = label(r["court_de"]), label(r["heading_court"])
            if r.get("how") != "court" and b not in later:
                continue
            if a and b and a != b:
                found.setdefault((r["page"].zfill(4), a, b), set()).add(path)
    return [{"page": k[0], "a": k[1], "b": k[2], "readers": len(v), "of": len(paths)}
            for k, v in sorted(found.items())]

def trusted(split: dict, run_pages: list[str], courts: pd.DataFrame) -> bool:
    """A heading most readers found (3 of 4), or one the route itself confirms: court B has a section

def carried_entries(rows: list[dict], splits: list[dict], courts: pd.DataFrame) -> tuple[dict, list[dict]]:
    """``({(page, label, muni): new label}, kept)``: the entries of one table (``rows``: dicts with
    page, district, muni_seq) filed under court A that belong to the court B whose heading a reader
    found on A's page. ``kept``: candidates left where they are, and why."""
    by_label: dict[str, list[tuple[str, int]]] = {}
    for r in rows:
        if str(r["muni_seq"]).isdigit():
            by_label.setdefault(r["district"], []).append((str(r["page"]).zfill(4), int(r["muni_seq"])))
    pages_of = {}
    for page, label in zip(courts["page"].astype(str).str.zfill(4), courts["district"].map(lxnt.normalise)):
        pages_of.setdefault(label, set()).add(page)
    moves, kept = {}, []
    for s in splits:
        a, b, start = s["a"], s["b"], s["page"]
        entries = by_label.get(a, [])
        reached = max((m for p, m in entries if p < start), default=None)
        if reached is None:
            continue                                   # A opens on this page: nothing to compare with
        run_pages, page = [], int(start)
        while f"{page:04d}" in pages_of.get(a, set()):
            run_pages.append(f"{page:04d}")
            page += 1
        if not trusted(s, run_pages, courts):
            kept.append({"page": start, "district": a, "muni_seq": "", "to": b,
                         "why": f"{s['readers']} of {s['of']} readers found {b}'s heading and the route has no {b} there"})
            continue
        b_elsewhere = {m for p, m in by_label.get(b, []) if p not in run_pages}
        for p, m in sorted(set(entries)):
            if p not in run_pages or m > reached - RESTART_MIN:
                continue
            if (p, a, m) in moves:
                continue
            if m in b_elsewhere:
                kept.append({"page": p, "district": a, "muni_seq": m, "to": b,
                             "why": f"{b} holds no. {m} on its own pages"})
                continue
            moves[(p, a, m)] = b
    return moves, kept

def move_carried(t, edition: str, recheck: Path, courts: pd.DataFrame) -> dict:
    """German opening entries filed under the court before them moved to their own (step ``keys``)."""
    report = {"moved_entries": 0, "moved_rows": 0, "kept": []}
    if edition != "cz":

def renumber(t, edition: str) -> dict:
    """Every post-1910 section's own numbers keyed ``OFFSET + n`` under its 1910 court (step ``keys``)."""
    report = {"renumbered_rows": 0}
    later = post1910_courts(edition)
    for table, rows in t.rows.items():
        for r in rows:
            label = lxnt.normalise(r["district"])
            if label in later:
                new = offset_number(r["muni_seq"])
                if new is None:
                    continue
                old = r["muni_seq"]
                if not r.get("court_id"):              # moved here above: the label is no section of the run's
                    r["court_id"], r["court_source"] = later[label], "alias (section_keys, D-208)"
                court = r["court_id"]
                r["muni_seq"] = new
                if "entry" in r:
                    r["entry"] = "/".join((r["page"], court, new, str(r["settlement_seq"])))
                t._log("keys", table, "renumber", (r["page"], r["district"], old, str(r["settlement_seq"])),
                       r["code"], f"a post-1910 court's own number, keyed {new} under its 1910 court (D-208)")
                report["renumbered_rows"] += 1
    return report

def mend_run(run) -> tuple:
    """The parts lists, place attributes and statistical table of a ``RunInput`` with each post-1910
    section numbered ``OFFSET + n`` (its inventory and abbreviations come renumbered from merge).
    Returns ``(run, report)``."""
    later = post1910_courts(run.edition)
    labels = set(later)
    report = {"edition": run.edition, "labels": sorted(labels)}
    courts = _with_table_labels(run.courts, run.table, later)
    parts, report["parts_rows"] = renumber_frame(run.parts, labels)
    attributes, report["attribute_rows"] = renumber_frame(run.attributes, labels)
    table, report["table"] = renumber_table(run.table, courts, labels)
    added = pd.DataFrame(report["table"]["court_rows"])
    if len(added):
        added["court_source"] = "table run (section_keys, D-208)"
        courts = pd.concat([courts, added], ignore_index=True)
    report["courts_added"] = len(courts) - len(run.courts)
    return replace(run, parts=parts, attributes=attributes, table=table, courts=courts), report

"""One number, one municipality: the two ways a court's numbering held two places (D-208).

The register keys a place on ``(court_key, muni_seq)``: the 1910 court and the number its section
prints. Two faults gave one key to two municipalities, and the register then built one place out of
both — the first table row's name and figures, both entries' institutions:

* **A court created after 1910, filed under the court it was cut from** (D-122). The Czech volume
  prints *Neustadt u Tafelfichte* (7 municipalities, from Frýdlant, 1 July 1913) and *Kralupy nad
  Vltavou* (21, from Velvary, 1 March 1912) as courts of their own, each numbering its municipalities
  from 1. ``court_aliases.csv`` files both under their 1910 court, rightly, and so Neustadt's 1–7
  shared Frýdlant's 1–7 and Kralupy's 1–21 Velvary's 1–20. :func:`renumber_frame` and
  :func:`renumber_table` give the later court's own number *n* the key ``OFFSET + n`` (1001…): the
  court stays the 1910 one, the number says the section, and no 1910 court reaches a thousand.
* **A court's opening entries filed under the court before it** (the German volumes). Where the cutter
  missed a court heading inside a column, the entries below it kept the label of the court above;
  §3 (``offices_resplit``) finds the heading in the readers' text, but merge moves a row only where
  *every* reader's re-split loses it, and a page that continues the new court with no heading keeps
  the old label outright (Jungbunzlau's 8–23 on Böhmen 0149, labelled Sobotka). :func:`carried_entries`
  moves such a row where **both** hold: a reader's re-split found a heading of court B on the page
  of court A's label (``resplit/splits_<label>.csv``, ``how = court``), and the row's number is
  lower than the highest A had already reached before that page — A's own entries only rise, so a
  lower one is the next court's. It runs over the page and the pages after it that carry A's label
  without a break, and leaves a row where B already holds that number off those pages.

Both run on what notebook 17 hands on: merge's last step (:func:`apply`) on the inventory, types
and abbreviations, and ``register_rechecked.runs`` (:func:`mend_run`) on the parts lists, the place
attributes and the statistical table, which merge never sees. Nothing in the frozen route changes.
"""
from __future__ import annotations

__version__ = "0.2.0"   # 0.2.0 apply split in move_carried and renumber, so merge can rule between them (D-209); 0.1.0 post-1910 courts numbered OFFSET + n; German opening entries filed under the court whose heading the readers found (D-208)

import csv
import glob
import sys
from dataclasses import replace
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT), str(ROOT / "utils" / "lexikon_1910")]

from utils.lexikon_1910 import lexikon_1910_notes as lxnt             # noqa: E402

OFFSET = 1000          # a post-1910 court's own number n is keyed OFFSET + n under its 1910 court
ALIASES = ROOT / "reference" / "places" / "court_aliases.csv"
RESTART_MIN = 3        # a number counts as a restart only below what the court had reached by this much


def _read(path: Path) -> list[dict]:
    with open(path, encoding="utf-8", newline="") as fh:
        return list(csv.DictReader(fh))


def post1910_courts(edition: str, aliases: Path = ALIASES) -> dict[str, str]:
    """``{section label: 1910 court id}`` for the courts created after 1910 that ``court_aliases.csv``
    files under a 1910 court, for this edition's language (labels as ``lexikon_1910_notes.normalise``)."""
    language = "cz" if edition == "cz" else "de"
    return {lxnt.normalise(r["printed"]): r["court_id"] for r in _read(aliases)
            if r["kind"] == "post-1910" and r["edition"] == language}


def post1910_labels(edition: str, aliases: Path = ALIASES) -> set[str]:
    return set(post1910_courts(edition, aliases))


def offset_number(value) -> str | None:
    """``"7"`` → ``"1007"``; ``None`` for a number that is not one or is already offset."""
    text = str(value).strip()
    return str(OFFSET + int(text)) if text.isdigit() and int(text) < OFFSET else None

def renumber_frame(frame: pd.DataFrame | None, labels: set[str], district: str = "district") -> tuple:
    """``(frame, n)``: every row of a post-1910 section (its ``district`` label in ``labels``) keyed
    ``OFFSET + muni_seq``; ``n`` rows changed. ``None`` passes through."""
    if frame is None or frame.empty or not labels:
        return frame, 0
    out = frame.copy()
    mine = out[district].map(lxnt.normalise).isin(labels)
    n = 0
    for idx in out.index[mine]:
        value = out.at[idx, "muni_seq"]
        new = offset_number(value)
        if new is not None:
            out.at[idx, "muni_seq"] = new if isinstance(value, str) else int(new)   # the frame's own type
            n += 1
    return out, n


def _row0_numbers(table: pd.DataFrame) -> list[tuple]:
    """``[(block_id, page, label, n)]`` per channel-A block in printed order (``n`` the row 0 number
    or ``None``)."""
    a = table[table["channel"] == "A"]
    out = []
    for block_id, g in a.groupby("block_id", sort=False):
        zero = g[g["row"].astype(str) == "0"]
        seq = str(zero["seq"].iloc[0]).strip() if len(zero) else ""
        out.append((block_id, str(g["page"].iloc[0]).zfill(4), str(g["court_label"].iloc[0]),
                    int(seq) if seq.isdigit() else None))
    return out


def table_segments(table: pd.DataFrame, courts: pd.DataFrame, parent: str, labels: set[str]) -> dict:
    """``{block_id: section label}`` for the table blocks of 1910 court ``parent`` where it holds a
    post-1910 section: the blocks in printed order, cut where the numbering starts again (a row 0
    numbered 1 or 2 after one numbered higher), each run given the court's sections in the order
    ``courts.csv`` first meets them. Empty where the runs and the sections do not count the same —
    the table's labels cannot be trusted there (Velvary's 14–20 on Czech 0199 read as Kralupy's)."""
    mine = courts[courts["court_id"].astype(str) == parent].copy()
    mine["label"] = mine["district"].map(lxnt.normalise)
    order = list(dict.fromkeys(mine.sort_values("page", kind="stable")["label"]))
    if not set(order) & labels:
        return {}
    pages = set(mine["page"].astype(str).str.zfill(4))
    known = {(str(r.page).zfill(4), r.label): str(r.court_id) for r in mine.itertuples()}
    others = {(str(r.page).zfill(4), lxnt.normalise(r.district)) for r in courts.itertuples()
              if str(r.court_id) != parent}
    def names(frame):
        cs = frame["court_cs"] if "court_cs" in frame else pd.Series(dtype=str)
        return {lxnt.normalise(x) for x in [*frame["district"], *cs] if str(x).strip()}

    own = names(mine) | {lxnt.normalise(l) for l in labels}
    foreign = names(courts[courts["court_id"].astype(str) != parent]) - own

    def theirs(page, label):
        lab = lxnt.normalise(label)
        return (page, lab) not in known and ((page, lab) in others or lab in foreign)

    blocks = [(b, n) for b, page, label, n in _row0_numbers(table)
              if page in pages and not theirs(page, label)]          # not a block of another court
    numbers = [n for _b, n in blocks]
    runs, current, last = [], [], None
    for i, (block_id, n) in enumerate(blocks):
        following = next((m for m in numbers[i + 1:] if m is not None), None)
        # a restart: 1 or 2 after a higher number, and the next number read is n + 1 — a lone misread
        # "1" between Velvary's 17 and 18 (Czech 0199) is none
        if (n is not None and last is not None and n <= 2 and last > n + 1 and following == n + 1
                and current):
            runs.append(current)
            current = []
        current.append(block_id)
        if n is not None:
            last = n
    if current:
        runs.append(current)
    if len(runs) != len(order):
        return {}
    return {b: order[i] for i, run in enumerate(runs) for b in run}


def renumber_table(table: pd.DataFrame, courts: pd.DataFrame, labels: set[str]) -> tuple[pd.DataFrame, dict]:
    """The statistical table with each post-1910 section's municipalities numbered ``OFFSET + n``.
    A block's row 0 opens a municipality; a later row whose number is neither the next part's nor
    lower opens the next one (the register's own reading, ``_add_table_rows``) — both offset; part
    rows keep their numbers. Returns ``(table, report)``."""
    report = {"courts": {}, "rows": 0, "court_rows": []}
    if table is None or table.empty or "channel" not in table or not labels:
        return table, report
    parents = {str(c) for c, d in zip(courts["court_id"], courts["district"])
               if lxnt.normalise(d) in labels and str(c).isdigit()}
    out = table.copy()
    for parent in sorted(parents):
        segments = table_segments(out, courts, parent, labels)
        if not segments:
            report["courts"][parent] = "left: the table's runs do not match the court's sections"
            continue
        n_rows = 0
        known = {(str(p).zfill(4), lxnt.normalise(d)) for p, d in zip(courts["page"], courts["district"])}
        for block_id, label in segments.items():
            page = str(out.loc[out["block_id"] == block_id, "page"].iloc[0]).zfill(4)
            if (page, label) not in known:             # the register files a table row by (page, label)
                known.add((page, label))
                report["court_rows"].append({"page": page, "district": label, "court_id": parent})
            if label not in labels:
                continue
            block = out[(out["block_id"] == block_id) & (out["channel"] == "A")]
            muni, expected = None, 1
            for idx, r in block.iterrows():
                seq = str(r["seq"]).strip()
                if not seq.isdigit():
                    continue
                n = int(seq)
                if str(r["row"]) == "0" or (muni is not None and n != expected and n > muni):
                    muni, expected = n, 1
                    out.at[idx, "seq"] = str(OFFSET + n)
                    n_rows += 1
                elif muni is not None and n == expected:
                    expected += 1
        report["courts"][parent] = f"{n_rows} municipalities renumbered"
        report["rows"] += n_rows
    return out, report

def court_splits(edition: str, recheck: Path) -> list[dict]:
    """``[{page, a, b}]``: every court heading a reader's re-split found inside another court's
    section, both courts as section labels, once per page and pair. A heading of a post-1910 court
    (*Kralup an der Moldau* on Böhmen 0212–0213) counts too: the re-split recorded it as a heading
    of the crop's own court (``how = heading``) only because the crosswalk holds no such 1910 court."""
    from utils.lexikon_1910 import lexikon_1910_de_notes as lxdn
    crosswalk = lxdn.load_court_crosswalk()
    later = post1910_labels(edition)

    def label(german: str) -> str:
        czech = lxdn.court_name(german, crosswalk)[0]
        if czech:
            return lxnt.normalise(czech)
        return lxnt.normalise(german) if lxnt.normalise(german) in later else ""

    found: dict[tuple, set] = {}
    paths = sorted(glob.glob(str(Path(recheck) / edition / "resplit" / "splits_*.csv")))
    for path in paths:
        for r in _read(Path(path)):
            a, b = label(r["court_de"]), label(r["heading_court"])
            if r.get("how") != "court" and b not in later:
                continue
            if a and b and a != b:
                found.setdefault((r["page"].zfill(4), a, b), set()).add(path)
    return [{"page": k[0], "a": k[1], "b": k[2], "readers": len(v), "of": len(paths)}
            for k, v in sorted(found.items())]


def trusted(split: dict, run_pages: list[str], courts: pd.DataFrame) -> bool:
    """A heading most readers found (3 of 4), or one the route itself confirms: court B has a section
    of its own on a page of the run or the page after it. Two readers' *Odrau* on Silesian 0024 — a
    misread of the page's own *Oderberg* — is neither."""
    if split["readers"] * 4 >= split["of"] * 3:
        return True
    after = f"{int(run_pages[-1]) + 1:04d}" if run_pages else split["page"]
    pages = set(run_pages) | {after}
    return any(p in pages and lxnt.normalise(d) == split["b"]
               for p, d in zip(courts["page"].astype(str).str.zfill(4), courts["district"]))


def carried_entries(rows: list[dict], splits: list[dict], courts: pd.DataFrame) -> tuple[dict, list[dict]]:
    """``({(page, label, muni): new label}, kept)``: the entries of one table (``rows``: dicts with
    page, district, muni_seq) filed under court A that belong to the court B whose heading a reader
    found on A's page. ``kept``: candidates left where they are, and why."""
    by_label: dict[str, list[tuple[str, int]]] = {}
    for r in rows:
        if str(r["muni_seq"]).isdigit():
            by_label.setdefault(r["district"], []).append((str(r["page"]).zfill(4), int(r["muni_seq"])))
    pages_of = {}
    for page, label in zip(courts["page"].astype(str).str.zfill(4), courts["district"].map(lxnt.normalise)):
        pages_of.setdefault(label, set()).add(page)
    moves, kept = {}, []
    for s in splits:
        a, b, start = s["a"], s["b"], s["page"]
        entries = by_label.get(a, [])
        reached = max((m for p, m in entries if p < start), default=None)
        if reached is None:
            continue                                   # A opens on this page: nothing to compare with
        run_pages, page = [], int(start)
        while f"{page:04d}" in pages_of.get(a, set()):
            run_pages.append(f"{page:04d}")
            page += 1
        if not trusted(s, run_pages, courts):
            kept.append({"page": start, "district": a, "muni_seq": "", "to": b,
                         "why": f"{s['readers']} of {s['of']} readers found {b}'s heading and the route has no {b} there"})
            continue
        b_elsewhere = {m for p, m in by_label.get(b, []) if p not in run_pages}
        for p, m in sorted(set(entries)):
            if p not in run_pages or m > reached - RESTART_MIN:
                continue
            if (p, a, m) in moves:
                continue
            if m in b_elsewhere:
                kept.append({"page": p, "district": a, "muni_seq": m, "to": b,
                             "why": f"{b} holds no. {m} on its own pages"})
                continue
            moves[(p, a, m)] = b
    return moves, kept


def apply(t, edition: str, recheck: Path, courts: pd.DataFrame) -> dict:
    """Merge's last step on the ``merge.Tables`` ``t``: German opening entries moved to their court
    (:func:`move_carried`), then every post-1910 section renumbered (:func:`renumber`). Merge runs the
    two apart since 0.8.0, the author's revenue-glyph rulings between them (D-209)."""
    return {**move_carried(t, edition, recheck, courts), **renumber(t, edition)}


def move_carried(t, edition: str, recheck: Path, courts: pd.DataFrame) -> dict:
    """German opening entries filed under the court before them moved to their own (step ``keys``)."""
    report = {"moved_entries": 0, "moved_rows": 0, "kept": []}
    if edition != "cz":
        splits = court_splits(edition, recheck)
        for table in ("inventory", "abbreviations", "types"):
            moves, kept = carried_entries(t.rows[table], splits, courts)
            report["kept"] += [dict(k, table=table) for k in kept]
            report["moved_entries"] += len(moves) if table == "inventory" else 0
            for (page, a, m), b in sorted(moves.items()):
                for r in [r for r in t.rows[table] if (str(r["page"]).zfill(4), r["district"], str(r["muni_seq"]))
                          == (page, a, str(m))]:
                    part = (r["page"], r["district"], str(r["muni_seq"]), str(r["settlement_seq"]))
                    new = (r["page"], b, str(r["muni_seq"]), str(r["settlement_seq"]))
                    if t.move("keys", table, part, new, r["code"],
                              f"moved: a reader found {b}'s heading on the page and no. {m} is below what {a} had reached",
                              f"moved from {a} (the court before it, D-208)"):
                        report["moved_rows"] += 1
    return report


def renumber(t, edition: str) -> dict:
    """Every post-1910 section's own numbers keyed ``OFFSET + n`` under its 1910 court (step ``keys``)."""
    report = {"renumbered_rows": 0}
    later = post1910_courts(edition)
    for table, rows in t.rows.items():
        for r in rows:
            label = lxnt.normalise(r["district"])
            if label in later:
                new = offset_number(r["muni_seq"])
                if new is None:
                    continue
                old = r["muni_seq"]
                if not r.get("court_id"):              # moved here above: the label is no section of the run's
                    r["court_id"], r["court_source"] = later[label], "alias (section_keys, D-208)"
                court = r["court_id"]
                r["muni_seq"] = new
                if "entry" in r:
                    r["entry"] = "/".join((r["page"], court, new, str(r["settlement_seq"])))
                t._log("keys", table, "renumber", (r["page"], r["district"], old, str(r["settlement_seq"])),
                       r["code"], f"a post-1910 court's own number, keyed {new} under its 1910 court (D-208)")
                report["renumbered_rows"] += 1
    return report

# 3. On the run the register builds from

def mend_run(run) -> tuple:
    """The parts lists, place attributes and statistical table of a ``RunInput`` with each post-1910
    section numbered ``OFFSET + n`` (its inventory and abbreviations come renumbered from merge).
    Returns ``(run, report)``."""
    later = post1910_courts(run.edition)
    labels = set(later)
    report = {"edition": run.edition, "labels": sorted(labels)}
    courts = _with_table_labels(run.courts, run.table, later)
    parts, report["parts_rows"] = renumber_frame(run.parts, labels)
    attributes, report["attribute_rows"] = renumber_frame(run.attributes, labels)
    table, report["table"] = renumber_table(run.table, courts, labels)
    added = pd.DataFrame(report["table"]["court_rows"])
    if len(added):
        added["court_source"] = "table run (section_keys, D-208)"
        courts = pd.concat([courts, added], ignore_index=True)
    report["courts_added"] = len(courts) - len(run.courts)
    return replace(run, parts=parts, attributes=attributes, table=table, courts=courts), report


def _with_table_labels(courts: pd.DataFrame, table: pd.DataFrame | None, later: dict) -> pd.DataFrame:
    """``courts`` with a row for each page whose table labels a block with a post-1910 court the page's
    sections never named (*Kralup an der Moldau* on German Böhmen 0212–0213, whose rows the register
    otherwise gave the page's one court, Manětín)."""
    if table is None or table.empty or "court_label" not in table:
        return courts
    known = {(str(p).zfill(4), lxnt.normalise(d)) for p, d in zip(courts["page"], courts["district"])}
    rows = []
    for page, label in {(str(p).zfill(4), lxnt.normalise(l)) for p, l in zip(table["page"], table["court_label"])}:
        if label in later and (page, label) not in known:
            rows.append({"page": page, "district": label, "court_id": later[label],
                         "court_source": "table label (section_keys, D-208)"})
    return pd.concat([courts, pd.DataFrame(rows)], ignore_index=True) if rows else courts
