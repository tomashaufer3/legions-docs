TL;DR

Cisleithanian analogue: utils.parsing.records_geo (birth_okres → okres_id). This maps a soldier's Slovak/Rusyn birth (or home) district string to the 1910 járás unit id (unit_id = census kod) and its parent vármegye (county) of the Upper-Hungary backbone (utils.parsing.districts_hu), so the járás-level census (utils.parsing.census_hu) can be joined on — the Transleithanian counterpart to the okres census key.

How it works

Scope & resolution

Runs only on records the Cisleithanian matcher left unplaced ({prefix}_okres_id is &lt;NA&gt;) — those are the Slovak / Hungarian / foreign births. The soldier records use Slovak names (Nitra, Trenčín, Žilina) while the GIS backbone is Magyar (Nyitra, Trencsén, Zsolna); the join key is therefore the curated Slovak name (name_sk / name_sk_alt / name_sk_modern), plus the Slovak county name for county-level record strings. Match tiers ({prefix}_jaras_match):

jaras a district (or a district's alias settlement) — strongest town a chartered town (rtv./tjv.) census unit megye only the county resolved (no district) → järás left &lt;NA&gt;, county filled fuzzy token-set ratio ≥ threshold (spelling variants: Brezno, Kráľovský Chlumec), and only where the top score is not tied across two different units &lt;NA&gt; no confident match (a non-Slovak birth, a name outside the Felvidék frame, or a cell that names more than one district and so cannot be resolved to either)

Adds {prefix}_jaras_id / {prefix}_jaras_match / {prefix}_jaras_sk / {prefix}_jaras_hu and the parent {prefix}_megye_id / {prefix}_megye_sk / {prefix}_megye_hu.

Járás / town twins

Twenty-six seats name two units, a járás and the chartered town at its centre (Bratislava, Košice, Nitra, Prešov, Žilina …) — the Transleithanian counterpart of the Cisleithanian Statutarstadt pairs. A district string alone cannot tell them apart, so add_jaras_id resolves the bare name to the járás throughout, a trailing rural qualifier (Košice vonkov) is stripped by rule rather than reached by fuzzy accident, and resolve_jaras_twins then promotes to the town unit exactly those records whose municipality is the town itself.

Unlike the Czech twins there is no pooling: the GISta census carries a separate row for every one of the 52 units, so the split is purely an assignment problem and both members keep their own covariates. See DECISIONS D-19, and records_geo.resolve_twin_pairs for the mirror.

Usage

from utils.parsing import records_geo_hu as rgh
legion = rgh.add_jaras_id(legion, backbone_hu, prefix="birth")           # district-seat pass
legion = rgh.add_settlement_jaras(legion, cross_sk, backbone_hu, "birth")  # village pass
legion = rgh.resolve_jaras_twins(legion, backbone_hu, cross_sk, "birth")   # twin members
rgh.jaras_report(legion, prefix="birth")

def add_jaras_id(df: pd.DataFrame, backbone: pd.DataFrame, prefix: str = "birth",
                 fuzzy: bool = True, threshold: int = 90,
                 only_unplaced: bool = True) -> pd.DataFrame:
    """Add ``{prefix}_jaras_*`` and ``{prefix}_megye_*`` from the Slovak district string.

    Resolves the **distinct** ``{prefix}_okres`` values, then broadcasts. With *only_unplaced*
    (default), rows already carrying a Cisleithanian ``{prefix}_okres_id`` are skipped so the two
    frames never collide. Unresolved values get ``<NA>``.

    Resolution order per distinct value: exact unit name → county-only → **rural-suffix strip and
    retry** → token-set fuzzy. The strip sits immediately before fuzzy for the same reason it does
    on the Czech side (:func:`records_geo.add_birth_okres_id`): ``Košice vonkov`` and
    ``Bratislava - venkov`` mean the *járás* around the town, which the backbone names bare, and
    they were reaching that answer only because token-set fuzzy happened to drop the extra token —
    589 records that scored at the ``fuzzy`` tier and were afterwards indistinguishable from a
    bare ``Košice``. They now resolve at ``jaras``. DECISIONS D-19.
    """
    okres_col = f"{prefix}_okres"
    unit_lut, megye_lut, fkeys = _build_lookups(backbone)
    unit_meta = backbone.drop_duplicates("unit_id").set_index("unit_id")
    megye_meta = backbone.drop_duplicates("megye_id").set_index("megye_id")

    distinct = pd.Index(df[okres_col].dropna().unique())
    jid: dict[str, "str | None"] = {}
    tier: dict[str, "str | None"] = {}
    mid: dict[str, "str | None"] = {}
    for raw in distinct:
        k = _norm(raw)
        if not k:
            jid[raw] = tier[raw] = mid[raw] = None
            continue
        hit = unit_lut.get(k) if k not in _AMBIGUOUS_SEATS else None
        if hit is not None:
            jid[raw], tier[raw] = hit
            mid[raw] = unit_meta.at[hit[0], "megye_id"]
            continue
        ck = _county_key(raw)
        if ck in megye_lut:                                  # county-only match
            jid[raw], tier[raw], mid[raw] = None, "megye", megye_lut[ck]
            continue
        bare = _RURAL_SUFFIX.sub("", k).strip()              # "kosice vonkov" -> "kosice"
        if bare and bare != k and bare not in _AMBIGUOUS_SEATS:
            hit = unit_lut.get(bare)
            if hit is not None:
                jid[raw], tier[raw] = hit
                mid[raw] = unit_meta.at[hit[0], "megye_id"]
                continue
        if fuzzy:
            m = process.extractOne(k, fkeys, scorer=fuzz.token_set_ratio)
            if m is not None and m[0] in _AMBIGUOUS_SEATS:
                m = None                                     # "Brody - Halič" is Galicia
            if m is not None and m[1] >= threshold:
                # A token-set ratio scores 100 against every district the cell names, so a cell
                # naming two ("Ilava nebo Zlaté Moravce", "Hlohovec Nitra", "Bratislava-Malacky")
                # ties at the top and the winner is whichever key the lookup happens to hold
                # first — a coin flip dressed as a match. Refuse instead, exactly as the
                # settlement pass refuses `len(units) != 1`. 38 records across 32 such strings,
                # every one a multi-district or truncated cell ("Bystrica" is Považská or Banská).
                best = {unit_lut[x][0] for x in fkeys
                        if fuzz.token_set_ratio(k, x) == m[1]}
                if len(best) == 1:
                    uid = unit_lut[m[0]][0]
                    jid[raw], tier[raw], mid[raw] = uid, "fuzzy", unit_meta.at[uid, "megye_id"]
                    continue
                jid[raw] = tier[raw] = mid[raw] = None
                continue
        jid[raw] = tier[raw] = mid[raw] = None

    out = df.copy()
    gate = out[f"{prefix}_okres_id"].isna() if (only_unplaced and f"{prefix}_okres_id" in out) \
        else pd.Series(True, index=out.index)
    src = out[okres_col].where(gate)

    uid_s = src.map(jid)
    out[f"{prefix}_jaras_id"] = uid_s
    out[f"{prefix}_jaras_match"] = pd.Categorical(src.map(tier),
                                                  categories=["jaras", "town", "megye", "fuzzy"])
    out[f"{prefix}_megye_id"] = src.map(mid)
    out[f"{prefix}_jaras_sk"] = uid_s.map(lambda u: unit_meta.at[u, "name_sk"] if pd.notna(u) else pd.NA)
    out[f"{prefix}_jaras_hu"] = uid_s.map(lambda u: unit_meta.at[u, "name_hu"] if pd.notna(u) else pd.NA)
    mid_s = out[f"{prefix}_megye_id"]
    out[f"{prefix}_megye_sk"] = mid_s.map(lambda m: megye_meta.at[m, "megye_sk"] if pd.notna(m) else pd.NA)
    out[f"{prefix}_megye_hu"] = mid_s.map(lambda m: megye_meta.at[m, "megye_hu"] if pd.notna(m) else pd.NA)
    return out

Settlement pass: resolve still-unplaced Slovak births to a járás via the obec crosswalk.

Cisleithanian counterpart: records_geo.add_birth_settlement_okres. add_jaras_id resolves a birthplace only when the string is a district/county seat; this second pass uses the Majo-2012 obec crosswalk (utils.parsing.lexikon_sk) to place ordinary villages by matching {prefix}_municipality against the obec / časť obce name. Must run after add_jaras_id (it fills only rows still lacking a {prefix}_jaras_id), and it runs only on rows the Cisleithanian frame left unplaced ({prefix}_okres_id is &lt;NA&gt;) — a Czech-placed record must never also acquire a Slovak district.

Assignment rules (refuse-over-guess, mirroring the Czech muni tier):

A district string that contradicts every candidate refuses the match (it is evidence the birth lies elsewhere, e.g. Volhynian Dubno), it no longer falls back to the uncorroborated candidate set;

Matched rows get tier settlement and {prefix}_obec_sk (the matched obec, the key to the crosswalk's obec-level mother-tongue shares). Note: the returned frame has a reset RangeIndex (positional buffers are used internally).

def add_settlement_jaras(df: pd.DataFrame, crosswalk: pd.DataFrame, backbone: pd.DataFrame,
                         prefix: str = "birth", muni_col: str | None = None,
                         exclude_names: set[str] | None = None) -> pd.DataFrame:
    """Settlement pass: resolve still-unplaced Slovak births to a járás via the obec crosswalk.

    Cisleithanian counterpart: :func:`records_geo.add_birth_settlement_okres`. :func:`add_jaras_id`
    resolves a birthplace only when the string is a district/county **seat**; this second pass uses
    the Majo-2012 obec crosswalk (:mod:`utils.parsing.lexikon_sk`) to place ordinary **villages** by
    matching ``{prefix}_municipality`` against the obec / *časť obce* name. Must run **after**
    :func:`add_jaras_id` (it fills only rows still lacking a ``{prefix}_jaras_id``), and it runs
    **only on rows the Cisleithanian frame left unplaced** (``{prefix}_okres_id`` is ``<NA>``) —
    a Czech-placed record must never also acquire a Slovak district.

    Assignment rules (refuse-over-guess, mirroring the Czech muni tier):

    * a municipality must map to a **single** járás ``unit_id`` across all like-named obce;
      cross-county homonyms (*Nová Ves*, *Lehota*) are left ``<NA>``;
    * when the record carries a district string, it must **corroborate** — equal (normalised) to a
      candidate obec's ``okres_sk`` or ``zupa_sk``. A district string that contradicts every
      candidate refuses the match (it is evidence the birth lies elsewhere, e.g. Volhynian
      *Dubno*), it no longer falls back to the uncorroborated candidate set;
    * with **no** district string, the match is accepted only when the record is not positively
      foreign (:data:`_VETO_LANDS` on ``{prefix}_land``) and — unless the land is positively
      "Slovakia" — the name is **not cross-frame ambiguous**: pass *exclude_names* = the Czech
      settlement name set (:func:`records_geo.settlement_name_set` on ``settlements_1910``) so a
      bare name existing in both countries (Petrovice, Čechy, Praha) is refused rather than
      guessed.

    Matched rows get tier ``settlement`` and ``{prefix}_obec_sk`` (the matched obec, the key to
    the crosswalk's obec-level mother-tongue shares). Note: the returned frame has a **reset
    RangeIndex** (positional buffers are used internally).
    """
    if muni_col is None:
        muni_col = f"{prefix}_municipality"
    jid_col, match_col = f"{prefix}_jaras_id", f"{prefix}_jaras_match"
    if jid_col not in df.columns:
        raise ValueError(f"run add_jaras_id(..., prefix='{prefix}') first")

    unit_meta = backbone.drop_duplicates("unit_id").set_index("unit_id")
    megye_meta = backbone.drop_duplicates("megye_id").set_index("megye_id")

    # obec/part name → {unit_id}; plus all corroboration forms of the obec's district and
    # county per (name, unit) — okres_sk/zupa_sk expanded via _district_keys so a record's
    # county-form string ("Nitranská stol.") still corroborates an obec in župa "Nitrianska"
    mapped = crosswalk[crosswalk["unit_id"].notna()]
    name_units: dict[str, set] = {}
    name_okres: dict[tuple, set] = {}
    for _, r in mapped.iterrows():
        k = _norm(r["obec_sk"])
        if not k:
            continue
        name_units.setdefault(k, set()).add(r["unit_id"])
        corro = name_okres.setdefault((k, r["unit_id"]), set())
        corro |= _district_keys(r.get("okres_sk")) | _district_keys(r.get("zupa_sk"))

    out = df.copy().reset_index(drop=True)
    okres_col = f"{prefix}_okres"
    jid = out[jid_col].astype("object").to_numpy()
    if not isinstance(out[match_col].dtype, pd.CategoricalDtype):
        out[match_col] = out[match_col].astype("category")
    if "settlement" not in out[match_col].cat.categories:
        out[match_col] = out[match_col].cat.add_categories(["settlement"])
    tiers = out[match_col].astype("object").to_numpy()
    obec = out.get(f"{prefix}_obec_sk", pd.Series(pd.NA, index=out.index)).astype("object").to_numpy()
    mid = out[f"{prefix}_megye_id"].astype("object").to_numpy()
    lands = (out[f"{prefix}_land"] if f"{prefix}_land" in out.columns
             else pd.Series(pd.NA, index=out.index))

    gate = out[jid_col].isna() & out[muni_col].notna()
    if f"{prefix}_okres_id" in out.columns:              # never touch Czech-placed rows
        gate &= out[f"{prefix}_okres_id"].isna()
    for pos in out.index[gate]:
        k = _norm(out.at[pos, muni_col])
        units = name_units.get(k)
        if not units:
            continue
        ok_keys = _district_keys(out.at[pos, okres_col]) if okres_col in out.columns else set()
        if ok_keys:                                      # district present → must corroborate
            units = {u for u in units if ok_keys & name_okres.get((k, u), set())}
            if not units:
                continue                                 # contradicts every candidate → refuse
        else:                                            # bare name → positive-signal rules
            land = lands.iat[pos]
            if pd.notna(land) and land in _VETO_LANDS:
                continue                                 # positively foreign → refuse
            is_slovak = pd.notna(land) and land == "Slovakia"
            if not is_slovak and exclude_names and k in exclude_names:
                continue                                 # exists in the Czech frame too → refuse
        if len(units) != 1:
            continue                                     # ambiguous → leave <NA>
        u = next(iter(units))
        jid[pos], tiers[pos], obec[pos] = u, "settlement", out.at[pos, muni_col]
        mid[pos] = unit_meta.at[u, "megye_id"]

    out[jid_col] = jid
    out[match_col] = pd.Categorical(tiers, categories=list(out[match_col].cat.categories))
    out[f"{prefix}_obec_sk"] = obec
    out[f"{prefix}_megye_id"] = mid
    jid_s = pd.Series(jid, index=out.index)
    out[f"{prefix}_jaras_sk"] = jid_s.map(lambda u: unit_meta.at[u, "name_sk"] if pd.notna(u) else pd.NA)
    out[f"{prefix}_jaras_hu"] = jid_s.map(lambda u: unit_meta.at[u, "name_hu"] if pd.notna(u) else pd.NA)
    mid_s = pd.Series(mid, index=out.index)
    out[f"{prefix}_megye_sk"] = mid_s.map(lambda m: megye_meta.at[m, "megye_sk"] if pd.notna(m) else pd.NA)
    out[f"{prefix}_megye_hu"] = mid_s.map(lambda m: megye_meta.at[m, "megye_hu"] if pd.notna(m) else pd.NA)
    return out

def build_jaras_twin_pairs(backbone: pd.DataFrame, settlements: pd.DataFrame | None=None) -> dict[str, tuple[str, str, frozenset[str]]]

{unit_id → (jaras_id, town_id, town name keys)} for both members of every twin pair.

Twenty-six 1910 seats name two Upper-Hungary units: the járás, and the chartered town (rtv./tjv.) at its centre, which sat outside the district it was named for. The pairs are derived from the backbone — a Slovak name shared between a járás row and a non-járás row of the same county — rather than listed here, and the caller is expected to assert the count. The county qualifier is what makes it safe: without it, unrelated same-named units in different counties would pair up.

The town name keys are what a record's municipality is tested against, and they come from the pair's own names: every Slovak form the backbone carries for the town plus its Magyar name (a record may write Nyitra or Pozsony), and — when settlements is passed — every obec name the crosswalk places inside the town unit. That last source is exact rather than clever: a chartered town was a single municipality, so the crosswalk holds one obec row per town unit and it is the town.

Cisleithanian mirror: records_geo.build_twin_pairs. See DECISIONS D-19.

def build_jaras_twin_pairs(backbone: pd.DataFrame,
                           settlements: pd.DataFrame | None = None,
                           ) -> dict[str, tuple[str, str, frozenset[str]]]:
    """``{unit_id → (jaras_id, town_id, town name keys)}`` for both members of every twin pair.

    Twenty-six 1910 seats name **two** Upper-Hungary units: the járás, and the chartered town
    (``rtv.``/``tjv.``) at its centre, which sat outside the district it was named for. The pairs
    are **derived** from the backbone — a Slovak name shared between a ``járás`` row and a
    non-``járás`` row **of the same county** — rather than listed here, and the caller is expected
    to assert the count. The county qualifier is what makes it safe: without it, unrelated
    same-named units in different counties would pair up.

    The *town name keys* are what a record's municipality is tested against, and they come from
    the pair's own names: every Slovak form the backbone carries for the town plus its Magyar
    name (a record may write *Nyitra* or *Pozsony*), and — when *settlements* is passed — every
    obec name the crosswalk places **inside** the town unit. That last source is exact rather
    than clever: a chartered town was a single municipality, so the crosswalk holds one obec row
    per town unit and it is the town.

    Cisleithanian mirror: :func:`records_geo.build_twin_pairs`. See DECISIONS D-19.
    """
    jar = backbone[(backbone["unit_type"] == "járás") & backbone["name_sk"].notna()]
    twn = backbone[(backbone["unit_type"] != "járás") & backbone["name_sk"].notna()]
    pairs: dict[str, tuple[str, str, frozenset[str]]] = {}
    for _, t in twn.iterrows():
        hit = jar[(jar["name_sk"] == t["name_sk"]) & (jar["megye_id"] == t["megye_id"])]
        if len(hit) != 1:                       # no járás twin (or an ambiguity we refuse to guess)
            continue
        jaras_id = hit.iloc[0]["unit_id"]
        keys = {_norm(t.get(c)) for c in
                ("name_sk", "name_sk_alt", "name_sk_modern", "name_hu")}
        if settlements is not None and "unit_id" in settlements.columns:
            inside = settlements[settlements["unit_id"] == t["unit_id"]]
            keys |= {_norm(v) for v in inside["obec_sk"].dropna()}
        keys.discard("")
        entry = (jaras_id, t["unit_id"], frozenset(keys))
        pairs[jaras_id] = entry
        pairs[t["unit_id"]] = entry
    return pairs

def resolve_jaras_twins(df: pd.DataFrame, backbone: pd.DataFrame, settlements: pd.DataFrame | None=None, prefix: str='birth') -> pd.DataFrame

Split a twin pair's records between the chartered town and the járás by their municipality.

A district string alone cannot decide a twin: the name belongs to both units, and add_jaras_id therefore sends every bare form to the járás. The municipality can decide, and it decides by definition rather than by inference — the chartered town was one municipality, so a record naming the town belongs to the town unit and a record naming anything else belongs to the district around it. Adds {prefix}_jaras_twin:

city the municipality is the chartered town itself ring the municipality is some other place → the járás undecided no municipality at all; assigned to the járás, flagged so it can be dropped &lt;NA&gt; the record is not in a twin pair

Run after both járás passes: add_settlement_jaras can itself place a record on a town unit (via an obec row), and those must be checked by the same rule rather than exempted. {prefix}_jaras_sk / _hu are re-derived from the resolved id; the county is untouched because both members of a pair sit in the same one. This does not touch {prefix}_jaras_match — the tier records how the district was found, which is unchanged; the twin column records how the member was chosen.

Cisleithanian mirror: records_geo.resolve_twin_pairs. See DECISIONS D-19.

def resolve_jaras_twins(df: pd.DataFrame, backbone: pd.DataFrame,
                        settlements: pd.DataFrame | None = None,
                        prefix: str = "birth") -> pd.DataFrame:
    """Split a twin pair's records between the chartered town and the járás by their municipality.

    A district string alone cannot decide a twin: the name belongs to both units, and
    :func:`add_jaras_id` therefore sends every bare form to the járás. The municipality can decide,
    and it decides *by definition* rather than by inference — the chartered town was one
    municipality, so a record naming the town belongs to the town unit and a record naming
    anything else belongs to the district around it. Adds ``{prefix}_jaras_twin``:

    ``city``       the municipality is the chartered town itself
    ``ring``       the municipality is some other place → the járás
    ``undecided``  no municipality at all; assigned to the járás, flagged so it can be dropped
    ``<NA>``       the record is not in a twin pair

    Run **after** both járás passes: :func:`add_settlement_jaras` can itself place a record on a
    town unit (via an obec row), and those must be checked by the same rule rather than exempted.
    ``{prefix}_jaras_sk`` / ``_hu`` are re-derived from the resolved id; the county is untouched
    because both members of a pair sit in the same one. This does **not** touch
    ``{prefix}_jaras_match`` — the tier records how the *district* was found, which is unchanged;
    the twin column records how the *member* was chosen.

    Cisleithanian mirror: :func:`records_geo.resolve_twin_pairs`. See DECISIONS D-19.
    """
    jid_col, muni_col = f"{prefix}_jaras_id", f"{prefix}_municipality"
    if jid_col not in df.columns:
        raise ValueError(f"run add_jaras_id(..., prefix='{prefix}') first")
    pairs = build_jaras_twin_pairs(backbone, settlements)
    unit_meta = backbone.drop_duplicates("unit_id").set_index("unit_id")

    out = df.copy()
    ids = out[jid_col].astype("object").to_numpy()
    munis = out[muni_col].to_numpy() if muni_col in out.columns else [pd.NA] * len(out)
    twin: list[object] = [pd.NA] * len(out)
    for i, (uid, mu) in enumerate(zip(ids, munis)):
        if pd.isna(uid):
            continue
        entry = pairs.get(uid)
        if entry is None:
            continue
        jaras_id, town_id, town_keys = entry
        k = _norm(mu)
        if not k:
            ids[i], twin[i] = jaras_id, "undecided"
        elif k in town_keys:
            ids[i], twin[i] = town_id, "city"
        else:
            ids[i], twin[i] = jaras_id, "ring"

    out[jid_col] = ids
    out[f"{prefix}_jaras_twin"] = pd.Categorical(twin, categories=["city", "ring", "undecided"])
    uid_s = pd.Series(ids, index=out.index)
    out[f"{prefix}_jaras_sk"] = uid_s.map(
        lambda u: unit_meta.at[u, "name_sk"] if pd.notna(u) else pd.NA)
    out[f"{prefix}_jaras_hu"] = uid_s.map(
        lambda u: unit_meta.at[u, "name_hu"] if pd.notna(u) else pd.NA)
    return out

"""
records_geo_hu.py — attach the 1910 Transleithanian járás/county to Slovak-born records
=======================================================================================
Cisleithanian analogue: :mod:`utils.parsing.records_geo` (birth_okres → okres_id). This maps a
soldier's Slovak/Rusyn birth (or home) district string to the 1910 **járás** unit id
(``unit_id`` = census ``kod``) and its parent **vármegye** (county) of the Upper-Hungary backbone
(:mod:`utils.parsing.districts_hu`), so the járás-level census (:mod:`utils.parsing.census_hu`) can
be joined on — the Transleithanian counterpart to the okres census key.

Scope & resolution
------------------
Runs **only on records the Cisleithanian matcher left unplaced** (``{prefix}_okres_id`` is ``<NA>``)
— those are the Slovak / Hungarian / foreign births. The soldier records use **Slovak** names
(Nitra, Trenčín, Žilina) while the GIS backbone is **Magyar** (Nyitra, Trencsén, Zsolna); the join
key is therefore the curated Slovak name (``name_sk`` / ``name_sk_alt`` / ``name_sk_modern``), plus
the Slovak county name for county-level record strings. Match tiers (``{prefix}_jaras_match``):

``jaras``  a district (or a district's alias settlement) — strongest
``town``   a chartered town (rtv./tjv.) census unit
``megye``  only the county resolved (no district) → järás left ``<NA>``, county filled
``fuzzy``  token-set ratio ≥ *threshold* (spelling variants: *Brezno*, *Kráľovský Chlumec*),
           and only where the top score is not tied across two different units
``<NA>``   no confident match (a non-Slovak birth, a name outside the Felvidék frame, or a cell
           that names more than one district and so cannot be resolved to either)

Adds ``{prefix}_jaras_id`` / ``{prefix}_jaras_match`` / ``{prefix}_jaras_sk`` / ``{prefix}_jaras_hu``
and the parent ``{prefix}_megye_id`` / ``{prefix}_megye_sk`` / ``{prefix}_megye_hu``.

Járás / town twins
------------------
Twenty-six seats name **two** units, a járás and the chartered town at its centre (Bratislava,
Košice, Nitra, Prešov, Žilina …) — the Transleithanian counterpart of the Cisleithanian
*Statutarstadt* pairs. A district string alone cannot tell them apart, so :func:`add_jaras_id`
resolves the bare name to the **járás** throughout, a trailing rural qualifier (``Košice vonkov``)
is stripped by rule rather than reached by fuzzy accident, and :func:`resolve_jaras_twins` then
promotes to the town unit exactly those records whose *municipality* is the town itself.

Unlike the Czech twins there is **no pooling**: the GISta census carries a separate row for every
one of the 52 units, so the split is purely an assignment problem and both members keep their own
covariates. See DECISIONS D-19, and :func:`records_geo.resolve_twin_pairs` for the mirror.

Usage
-----

    legion = rgh.add_settlement_jaras(legion, cross_sk, backbone_hu, "birth")  # village pass

def _build_lookups(backbone: pd.DataFrame):
    """Return ``(unit_lut, megye_lut, fuzzy_keys)`` from the backbone.

    ``unit_lut`` ``norm(slovak name) → (unit_id, tier)``. **The járás always wins a shared name**:
    twenty-six seats name both a district and the chartered town inside it (Nitra, Bratislava,
    Prešov …), the bare string cannot distinguish them, and the district is the larger, safer
    default — the town is reached only through the record's *municipality*
    (:func:`resolve_jaras_twins`, DECISIONS D-19). The ordering is now stated outright rather than
    left to ``sort_values("unit_type")``, which put járás first only because ``'j' < 'r' < 't'``
    in the Magyar unit-type labels — correct, but an accident one relabelling would silently undo.

    ``megye_lut`` ``norm(county) → megye_id``, keyed on the curated Slovak noun **plus** the Magyar
    county name (records write "Bereg" while the curation's Slovak column holds the seat
    "Berehovo") **plus** the adjectival župa forms (:data:`_COUNTY_ADJ` — "Nitrianska",
    "Hontianska stol." …).
    """
    unit_lut: dict[str, tuple[str, str]] = {}
    ordered = backbone.assign(_is_town=(backbone["unit_type"] != "járás").astype(int)) \
                      .sort_values("_is_town", kind="stable")
    for _, r in ordered.iterrows():
        tier = "jaras" if r["unit_type"] == "járás" else "town"
        for nm in (r.get("name_sk"), r.get("name_sk_alt"), r.get("name_sk_modern")):
            k = _norm(nm)
            if k and k not in unit_lut:
                unit_lut[k] = (r["unit_id"], tier)
    megye_lut: dict[str, str] = {}
    for _, r in backbone.drop_duplicates("megye_id").iterrows():
        for nm in (r.get("megye_sk"), r.get("megye_hu")):
            k = _norm(nm)
            if k and k not in megye_lut:
                megye_lut[k] = r["megye_id"]
    for adj, noun in _COUNTY_ADJ.items():                # adjectival forms → the noun's county
        if noun in megye_lut and adj not in megye_lut:
            megye_lut[adj] = megye_lut[noun]
    return unit_lut, megye_lut, list(unit_lut.keys())


def _county_key(raw: str) -> str:
    """Drop a trailing stolica/župa word so the (adjectival) county name reaches the lookup."""
    k = _norm(raw)
    for suf in _STOLICA:
        if k.endswith(" " + suf):
            k = k[: -len(suf) - 1].strip()
            break
    return k


def _district_keys(raw) -> set[str]:
    """All normalised forms a district/county string can corroborate under.

    A record's district text and the crosswalk's ``okres_sk``/``zupa_sk`` may name the
    same county in different grammatical forms ("Nitranská stol." vs "Nitrianska" vs
    "Nitra"), so both sides are expanded to {raw, stolica-stripped, county-noun} before
    intersection.
    """
    k = _norm(raw)
    if not k:
        return set()
    ck = _county_key(raw)
    return {x for x in (k, ck, _COUNTY_ADJ.get(k), _COUNTY_ADJ.get(ck)) if x}


def add_jaras_id(df: pd.DataFrame, backbone: pd.DataFrame, prefix: str = "birth",

                 only_unplaced: bool = True) -> pd.DataFrame:
    """Add ``{prefix}_jaras_*`` and ``{prefix}_megye_*`` from the Slovak district string.

    Resolves the **distinct** ``{prefix}_okres`` values, then broadcasts. With *only_unplaced*
    (default), rows already carrying a Cisleithanian ``{prefix}_okres_id`` are skipped so the two
    frames never collide. Unresolved values get ``<NA>``.

    Resolution order per distinct value: exact unit name → county-only → **rural-suffix strip and
    retry** → token-set fuzzy. The strip sits immediately before fuzzy for the same reason it does
    on the Czech side (:func:`records_geo.add_birth_okres_id`): ``Košice vonkov`` and
    ``Bratislava - venkov`` mean the *járás* around the town, which the backbone names bare, and
    they were reaching that answer only because token-set fuzzy happened to drop the extra token —
    589 records that scored at the ``fuzzy`` tier and were afterwards indistinguishable from a
    bare ``Košice``. They now resolve at ``jaras``. DECISIONS D-19.

    okres_col = f"{prefix}_okres"
    unit_lut, megye_lut, fkeys = _build_lookups(backbone)
    unit_meta = backbone.drop_duplicates("unit_id").set_index("unit_id")
    megye_meta = backbone.drop_duplicates("megye_id").set_index("megye_id")

    distinct = pd.Index(df[okres_col].dropna().unique())
    jid: dict[str, "str | None"] = {}
    tier: dict[str, "str | None"] = {}
    mid: dict[str, "str | None"] = {}
    for raw in distinct:
        k = _norm(raw)
        if not k:
            jid[raw] = tier[raw] = mid[raw] = None
            continue
        hit = unit_lut.get(k) if k not in _AMBIGUOUS_SEATS else None
        if hit is not None:
            jid[raw], tier[raw] = hit
            mid[raw] = unit_meta.at[hit[0], "megye_id"]
            continue
        ck = _county_key(raw)
        if ck in megye_lut:                                  # county-only match
            jid[raw], tier[raw], mid[raw] = None, "megye", megye_lut[ck]
            continue
        bare = _RURAL_SUFFIX.sub("", k).strip()              # "kosice vonkov" -> "kosice"
        if bare and bare != k and bare not in _AMBIGUOUS_SEATS:
            hit = unit_lut.get(bare)
            if hit is not None:
                jid[raw], tier[raw] = hit
                mid[raw] = unit_meta.at[hit[0], "megye_id"]
                continue
        if fuzzy:
            m = process.extractOne(k, fkeys, scorer=fuzz.token_set_ratio)
            if m is not None and m[0] in _AMBIGUOUS_SEATS:
                m = None                                     # "Brody - Halič" is Galicia
            if m is not None and m[1] >= threshold:
                # A token-set ratio scores 100 against every district the cell names, so a cell
                # naming two ("Ilava nebo Zlaté Moravce", "Hlohovec Nitra", "Bratislava-Malacky")
                # ties at the top and the winner is whichever key the lookup happens to hold
                # first — a coin flip dressed as a match. Refuse instead, exactly as the
                # settlement pass refuses `len(units) != 1`. 38 records across 32 such strings,
                # every one a multi-district or truncated cell ("Bystrica" is Považská or Banská).
                best = {unit_lut[x][0] for x in fkeys
                        if fuzz.token_set_ratio(k, x) == m[1]}
                if len(best) == 1:
                    uid = unit_lut[m[0]][0]
                    jid[raw], tier[raw], mid[raw] = uid, "fuzzy", unit_meta.at[uid, "megye_id"]
                    continue
                jid[raw] = tier[raw] = mid[raw] = None
                continue
        jid[raw] = tier[raw] = mid[raw] = None

    out = df.copy()
    gate = out[f"{prefix}_okres_id"].isna() if (only_unplaced and f"{prefix}_okres_id" in out) \
        else pd.Series(True, index=out.index)
    src = out[okres_col].where(gate)

    uid_s = src.map(jid)
    out[f"{prefix}_jaras_id"] = uid_s
    out[f"{prefix}_jaras_match"] = pd.Categorical(src.map(tier),
                                                  categories=["jaras", "town", "megye", "fuzzy"])
    out[f"{prefix}_megye_id"] = src.map(mid)
    out[f"{prefix}_jaras_sk"] = uid_s.map(lambda u: unit_meta.at[u, "name_sk"] if pd.notna(u) else pd.NA)
    out[f"{prefix}_jaras_hu"] = uid_s.map(lambda u: unit_meta.at[u, "name_hu"] if pd.notna(u) else pd.NA)
    mid_s = out[f"{prefix}_megye_id"]
    out[f"{prefix}_megye_sk"] = mid_s.map(lambda m: megye_meta.at[m, "megye_sk"] if pd.notna(m) else pd.NA)
    out[f"{prefix}_megye_hu"] = mid_s.map(lambda m: megye_meta.at[m, "megye_hu"] if pd.notna(m) else pd.NA)
    return out


# Positively-non-Hungarian lands that veto an uncorroborated (no-district) settlement match:
# a Galician/Austrian/Russian birth must not be claimed by a like-named Slovak obec. "Slovakia"
# is the positive accept signal; NA / "Hungary" / "Transleithania" (= somewhere in the Hungarian
# half, precise land unknown) stay eligible — Upper Hungary IS this frame.
_VETO_LANDS = {"Austria", "Galicia", "Russia", "Croatia", "Balkans", "Poland",
               "Germany", "Overseas", "Czech lands"}


def add_settlement_jaras(df: pd.DataFrame, crosswalk: pd.DataFrame, backbone: pd.DataFrame,
                         prefix: str = "birth", muni_col: str | None = None,
                         exclude_names: set[str] | None = None) -> pd.DataFrame:
    """Settlement pass: resolve still-unplaced Slovak births to a járás via the obec crosswalk.

    Cisleithanian counterpart: :func:`records_geo.add_birth_settlement_okres`. :func:`add_jaras_id`
    resolves a birthplace only when the string is a district/county **seat**; this second pass uses
    the Majo-2012 obec crosswalk (:mod:`utils.parsing.lexikon_sk`) to place ordinary **villages** by
    matching ``{prefix}_municipality`` against the obec / *časť obce* name. Must run **after**
    :func:`add_jaras_id` (it fills only rows still lacking a ``{prefix}_jaras_id``), and it runs
    **only on rows the Cisleithanian frame left unplaced** (``{prefix}_okres_id`` is ``<NA>``) —
    a Czech-placed record must never also acquire a Slovak district.

    Assignment rules (refuse-over-guess, mirroring the Czech muni tier):

    * a municipality must map to a **single** járás ``unit_id`` across all like-named obce;
      cross-county homonyms (*Nová Ves*, *Lehota*) are left ``<NA>``;
    * when the record carries a district string, it must **corroborate** — equal (normalised) to a
      candidate obec's ``okres_sk`` or ``zupa_sk``. A district string that contradicts every
      candidate refuses the match (it is evidence the birth lies elsewhere, e.g. Volhynian
      *Dubno*), it no longer falls back to the uncorroborated candidate set;
    * with **no** district string, the match is accepted only when the record is not positively
      foreign (:data:`_VETO_LANDS` on ``{prefix}_land``) and — unless the land is positively
      "Slovakia" — the name is **not cross-frame ambiguous**: pass *exclude_names* = the Czech
      settlement name set (:func:`records_geo.settlement_name_set` on ``settlements_1910``) so a
      bare name existing in both countries (Petrovice, Čechy, Praha) is refused rather than
      guessed.

    Matched rows get tier ``settlement`` and ``{prefix}_obec_sk`` (the matched obec, the key to
    the crosswalk's obec-level mother-tongue shares). Note: the returned frame has a **reset
    RangeIndex** (positional buffers are used internally).
    """
    if muni_col is None:
        muni_col = f"{prefix}_municipality"
    jid_col, match_col = f"{prefix}_jaras_id", f"{prefix}_jaras_match"
    if jid_col not in df.columns:
        raise ValueError(f"run add_jaras_id(..., prefix='{prefix}') first")

    unit_meta = backbone.drop_duplicates("unit_id").set_index("unit_id")
    megye_meta = backbone.drop_duplicates("megye_id").set_index("megye_id")

    # obec/part name → {unit_id}; plus all corroboration forms of the obec's district and
    # county per (name, unit) — okres_sk/zupa_sk expanded via _district_keys so a record's
    # county-form string ("Nitranská stol.") still corroborates an obec in župa "Nitrianska"
    mapped = crosswalk[crosswalk["unit_id"].notna()]
    name_units: dict[str, set] = {}
    name_okres: dict[tuple, set] = {}
    for _, r in mapped.iterrows():
        k = _norm(r["obec_sk"])
        if not k:
            continue
        name_units.setdefault(k, set()).add(r["unit_id"])
        corro = name_okres.setdefault((k, r["unit_id"]), set())
        corro |= _district_keys(r.get("okres_sk")) | _district_keys(r.get("zupa_sk"))

    out = df.copy().reset_index(drop=True)
    okres_col = f"{prefix}_okres"
    jid = out[jid_col].astype("object").to_numpy()
    if not isinstance(out[match_col].dtype, pd.CategoricalDtype):
        out[match_col] = out[match_col].astype("category")
    if "settlement" not in out[match_col].cat.categories:
        out[match_col] = out[match_col].cat.add_categories(["settlement"])
    tiers = out[match_col].astype("object").to_numpy()
    obec = out.get(f"{prefix}_obec_sk", pd.Series(pd.NA, index=out.index)).astype("object").to_numpy()
    mid = out[f"{prefix}_megye_id"].astype("object").to_numpy()
    lands = (out[f"{prefix}_land"] if f"{prefix}_land" in out.columns
             else pd.Series(pd.NA, index=out.index))

    gate = out[jid_col].isna() & out[muni_col].notna()
    if f"{prefix}_okres_id" in out.columns:              # never touch Czech-placed rows
        gate &= out[f"{prefix}_okres_id"].isna()
    for pos in out.index[gate]:
        k = _norm(out.at[pos, muni_col])
        units = name_units.get(k)
        if not units:
            continue
        ok_keys = _district_keys(out.at[pos, okres_col]) if okres_col in out.columns else set()
        if ok_keys:                                      # district present → must corroborate
            units = {u for u in units if ok_keys & name_okres.get((k, u), set())}
            if not units:
                continue                                 # contradicts every candidate → refuse
        else:                                            # bare name → positive-signal rules
            land = lands.iat[pos]
            if pd.notna(land) and land in _VETO_LANDS:
                continue                                 # positively foreign → refuse
            is_slovak = pd.notna(land) and land == "Slovakia"
            if not is_slovak and exclude_names and k in exclude_names:
                continue                                 # exists in the Czech frame too → refuse
        if len(units) != 1:
            continue                                     # ambiguous → leave <NA>
        u = next(iter(units))
        jid[pos], tiers[pos], obec[pos] = u, "settlement", out.at[pos, muni_col]
        mid[pos] = unit_meta.at[u, "megye_id"]

    out[jid_col] = jid
    out[match_col] = pd.Categorical(tiers, categories=list(out[match_col].cat.categories))
    out[f"{prefix}_obec_sk"] = obec
    out[f"{prefix}_megye_id"] = mid
    jid_s = pd.Series(jid, index=out.index)
    out[f"{prefix}_jaras_sk"] = jid_s.map(lambda u: unit_meta.at[u, "name_sk"] if pd.notna(u) else pd.NA)
    out[f"{prefix}_jaras_hu"] = jid_s.map(lambda u: unit_meta.at[u, "name_hu"] if pd.notna(u) else pd.NA)
    mid_s = pd.Series(mid, index=out.index)
    out[f"{prefix}_megye_sk"] = mid_s.map(lambda m: megye_meta.at[m, "megye_sk"] if pd.notna(m) else pd.NA)
    out[f"{prefix}_megye_hu"] = mid_s.map(lambda m: megye_meta.at[m, "megye_hu"] if pd.notna(m) else pd.NA)
    return out


# ── Járás / chartered-town twins ──────────────────────────────────────────────
def build_jaras_twin_pairs(backbone: pd.DataFrame,
                           settlements: pd.DataFrame | None = None,
                           ) -> dict[str, tuple[str, str, frozenset[str]]]:
    """``{unit_id → (jaras_id, town_id, town name keys)}`` for both members of every twin pair.

    Twenty-six 1910 seats name **two** Upper-Hungary units: the járás, and the chartered town
    (``rtv.``/``tjv.``) at its centre, which sat outside the district it was named for. The pairs
    are **derived** from the backbone — a Slovak name shared between a ``járás`` row and a
    non-``járás`` row **of the same county** — rather than listed here, and the caller is expected
    to assert the count. The county qualifier is what makes it safe: without it, unrelated
    same-named units in different counties would pair up.

    The *town name keys* are what a record's municipality is tested against, and they come from
    the pair's own names: every Slovak form the backbone carries for the town plus its Magyar
    name (a record may write *Nyitra* or *Pozsony*), and — when *settlements* is passed — every
    obec name the crosswalk places **inside** the town unit. That last source is exact rather
    than clever: a chartered town was a single municipality, so the crosswalk holds one obec row
    per town unit and it is the town.

    Cisleithanian mirror: :func:`records_geo.build_twin_pairs`. See DECISIONS D-19.
    """
    jar = backbone[(backbone["unit_type"] == "járás") & backbone["name_sk"].notna()]
    twn = backbone[(backbone["unit_type"] != "járás") & backbone["name_sk"].notna()]
    pairs: dict[str, tuple[str, str, frozenset[str]]] = {}
    for _, t in twn.iterrows():
        hit = jar[(jar["name_sk"] == t["name_sk"]) & (jar["megye_id"] == t["megye_id"])]
        if len(hit) != 1:                       # no járás twin (or an ambiguity we refuse to guess)
            continue
        jaras_id = hit.iloc[0]["unit_id"]
        keys = {_norm(t.get(c)) for c in
                ("name_sk", "name_sk_alt", "name_sk_modern", "name_hu")}
        if settlements is not None and "unit_id" in settlements.columns:
            inside = settlements[settlements["unit_id"] == t["unit_id"]]
            keys |= {_norm(v) for v in inside["obec_sk"].dropna()}
        keys.discard("")
        entry = (jaras_id, t["unit_id"], frozenset(keys))
        pairs[jaras_id] = entry
        pairs[t["unit_id"]] = entry
    return pairs


def resolve_jaras_twins(df: pd.DataFrame, backbone: pd.DataFrame,
                        settlements: pd.DataFrame | None = None,
                        prefix: str = "birth") -> pd.DataFrame:
    """Split a twin pair's records between the chartered town and the járás by their municipality.

    A district string alone cannot decide a twin: the name belongs to both units, and
    :func:`add_jaras_id` therefore sends every bare form to the járás. The municipality can decide,
    and it decides *by definition* rather than by inference — the chartered town was one
    municipality, so a record naming the town belongs to the town unit and a record naming
    anything else belongs to the district around it. Adds ``{prefix}_jaras_twin``:

    ``city``       the municipality is the chartered town itself
    ``ring``       the municipality is some other place → the járás
    ``undecided``  no municipality at all; assigned to the járás, flagged so it can be dropped
    ``<NA>``       the record is not in a twin pair

    Run **after** both járás passes: :func:`add_settlement_jaras` can itself place a record on a
    town unit (via an obec row), and those must be checked by the same rule rather than exempted.
    ``{prefix}_jaras_sk`` / ``_hu`` are re-derived from the resolved id; the county is untouched
    because both members of a pair sit in the same one. This does **not** touch
    ``{prefix}_jaras_match`` — the tier records how the *district* was found, which is unchanged;
    the twin column records how the *member* was chosen.

    Cisleithanian mirror: :func:`records_geo.resolve_twin_pairs`. See DECISIONS D-19.
    """
    jid_col, muni_col = f"{prefix}_jaras_id", f"{prefix}_municipality"
    if jid_col not in df.columns:
        raise ValueError(f"run add_jaras_id(..., prefix='{prefix}') first")
    pairs = build_jaras_twin_pairs(backbone, settlements)
    unit_meta = backbone.drop_duplicates("unit_id").set_index("unit_id")

    out = df.copy()
    ids = out[jid_col].astype("object").to_numpy()
    munis = out[muni_col].to_numpy() if muni_col in out.columns else [pd.NA] * len(out)
    twin: list[object] = [pd.NA] * len(out)
    for i, (uid, mu) in enumerate(zip(ids, munis)):
        if pd.isna(uid):
            continue
        entry = pairs.get(uid)
        if entry is None:
            continue
        jaras_id, town_id, town_keys = entry
        k = _norm(mu)
        if not k:
            ids[i], twin[i] = jaras_id, "undecided"
        elif k in town_keys:
            ids[i], twin[i] = town_id, "city"
        else:
            ids[i], twin[i] = jaras_id, "ring"

    out[jid_col] = ids
    out[f"{prefix}_jaras_twin"] = pd.Categorical(twin, categories=["city", "ring", "undecided"])
    uid_s = pd.Series(ids, index=out.index)
    out[f"{prefix}_jaras_sk"] = uid_s.map(
        lambda u: unit_meta.at[u, "name_sk"] if pd.notna(u) else pd.NA)
    out[f"{prefix}_jaras_hu"] = uid_s.map(
        lambda u: unit_meta.at[u, "name_hu"] if pd.notna(u) else pd.NA)
    return out


def set_land_from_jaras(df: pd.DataFrame, prefix: str = "birth") -> pd.DataFrame:
    """Set ``{prefix}_land`` to ``"Slovakia"`` wherever a járás was assigned, then re-sync
    ``{prefix}_cisleithanian``.

    ``records_geo.add_land`` runs *before* the Transleithanian passes, so a record those passes
    place — typically a Slovak village birth with no district text — kept the land it had
    before: ``<NA>`` on 1,041 legion and 2,707 VÚA records, and a wrong foreign label on others
    (audit 2026-09-22 §B6). The land label follows the project's existing convention that
    "Slovakia" names the whole Upper-Hungary frame, Subcarpathian units included.
    Run after :func:`resolve_jaras_twins`.
    """
    jid, land = f"{prefix}_jaras_id", f"{prefix}_land"
    if jid not in df.columns:
        return df
    out = df.copy()
    placed = out[jid].notna()
    if land in out.columns:
        out[land] = out[land].astype("object").where(~placed, "Slovakia")
    else:
        out[land] = pd.Series("Slovakia", index=out.index).where(placed, pd.NA)
    return sync_cisleithanian(out, prefix)


def records_geo_hu_selftest(backbone: pd.DataFrame) -> list[str]:
    """Behaviours that shipped wrong once; needs the backbone, not the person frames."""
    fails: list[str] = []
    probe = pd.DataFrame({"birth_okres": ["Halič", "Brody - Halič", "Nitra"],
                          "birth_okres_id": [pd.NA] * 3})
    got = add_jaras_id(probe, backbone, prefix="birth")["birth_jaras_id"].tolist()
    if pd.notna(got[0]) or pd.notna(got[1]):
        fails.append(f"'Halič' (= Galicia in Czech) resolved by name to a járás: {got[:2]}")
    if pd.isna(got[2]):
        fails.append("'Nitra' no longer resolves to a járás")
    return fails


def jaras_twin_report(df: pd.DataFrame, backbone: pd.DataFrame, prefix: str = "birth") -> None:
    """Print the town/ring split per járás–town twin pair, and the undecided residual."""
    jid_col, twin_col = f"{prefix}_jaras_id", f"{prefix}_jaras_twin"
    name = dict(zip(backbone["unit_id"], backbone["name_sk"]))
    sub = df[df[twin_col].notna()]
    print(f"{prefix}: {len(sub):,} record(s) in a járás/town twin pair "
          f"({len(sub)/max(len(df),1)*100:.1f}% of rows)\n")
    print(f"    {'pair':<26}{'town':>8}{'ring':>8}{'undecided':>11}")
    pairs = build_jaras_twin_pairs(backbone)
    seen: set[str] = set()
    for jaras_id, town_id, _ in pairs.values():
        if jaras_id in seen:
            continue
        seen.add(jaras_id)
        g = sub[sub[jid_col].isin([jaras_id, town_id])][twin_col].value_counts()
        print(f"    {str(name.get(jaras_id, jaras_id)):<26}{g.get('city', 0):>8}"
              f"{g.get('ring', 0):>8}{g.get('undecided', 0):>11}")
    tot = sub[twin_col].value_counts()
    print(f"\n    {'TOTAL':<26}{tot.get('city', 0):>8}{tot.get('ring', 0):>8}"
          f"{tot.get('undecided', 0):>11}")
