How it works

Every municipality's entries in the register's apparatus sit under a heading that names its court district ("Úřady, ústavy a t. d. v soudním okresu Blovice:"), and that name is what ties each institution to a 1910 court, to the census and to the other edition. Three things stand between the printed name and the court backbone (districts_1910.csv, MPIDR), measured on the 1,248 headings of the Czech volume's text layer (2026-10-24):

  • the scan's text layer damaged the name — biovice for Blovice, koilrim for Kouřim, Nová Paka stored as Nová Рака with a Cyrillic Рака;
  • the register prints its own name — Krumlov, Hanšpach, Bastianperk, Teplice-Šanov — where the backbone has Český Krumlov, Lipová, Hora Svatého Šebestiána, Teplice; and two courts it prints were created after the census;
  • the name is shared — Rokytnice is a court in okres Žamberk and in okres Jilemnice; Liberec is the rural court's name and the statutory city's.

A section's court is resolved in steps, and the step that settles it is kept:

  1. backbone — the name against every name form of the edition's land (Czech, modern Czech, German): exact; the one form it ends (≥ 5 letters: Maršov for Horní Maršov); the one it begins (≥ 6); the one it nearly is (ratio ≥ 85 and ≥ 10 above the next court) — D-121's rule. Cyrillic look-alikes are folded first.
  2. alias — the register's own names (reference/places/court_aliases.csv), including the two courts created after the census, each filed under the one 1910 court it was cut from (the German edition's Änderungen C.1–2).
  3. head — the page's running head prints its courts a second time ("Teplá: Teplá. — Teplice-Šanov: Teplice-Šanov."). A name steps 1–2 cannot settle is matched among those few names, and the name it matches is resolved by steps 1–2, else among the courts of the okres the head names.
  4. okres — the name matched among the backbone courts of the okresy the head names (lexikon_1910_names.Authority.match_okres_name: look-alikes folded, ties refused).
  5. readers — the readers' own transcriptions of the heading, which the crop audit keeps (D-85: a reader is never told the label), each resolved by steps 1–4: a court at least READER_MIN readings name, and no other court as often.

What none of them settles stays unresolved: a court is never guessed. Among a page's few names a match needs five letters — the German Tepl (Teplá) begins Teplice-Šanov — and the best of several must lead the next by ten points.

Twin courts. The backbone carries Liberec twice: the rural court (okres Liberec) and the statutory city's (okres Liberec (město)). A bare name is the rural one, as a bare district name is everywhere in the project (D-17); the city's own pages are running text, not the table and apparatus this resolves (D-82).

The resolved court sits beside the section's label, never in its place: the label (the text layer's slug) is how the gold sheets, the ledger and the final inventory key a section, and a sealed sheet is never edited (D-55).

def slug(name: str, german: bool = False) -> str:
    """A printed court name's match key.

    Czech: Cyrillic look-alikes folded (``Nová Рака`` -> ``novapaka``), then accents
    and all but letters dropped (``lexikon_1910_notes.normalise``). German: its
    abbreviations written out and its umlauts as the backbone spells them
    (``Böhm. Aicha`` -> ``boehmischaicha``).
    """
    if german:
        return lxdn.fold(lxdn.expand_court_abbreviations(lxn.fold_lookalikes(name)))
    return lxnt.normalise(lxn.fold_lookalikes(name))

def slug_variants(key: str) -> list[str]:
    """``key`` and, for a German slug that begins with a shortened first word, the slug
    with that word written out (``bohmaicha`` -> ``boehmischaicha``)."""
    out = [key]
    for short, full in _SLUG_ABBREVIATIONS:
        if key.startswith(short) and not key.startswith(full) \
                and not key.startswith(short + "isch") and len(key) > len(short) + 2:
            out.append(full + key[len(short):])

def court_forms(rows: list[dict]) -> tuple[dict, list, list]:
    """``({form: {court_id}}, Czech [(form, court_id)], German [(form, court_id)])``.

    Every court under its Czech, modern Czech and German name, each plainly normalised
    and with umlauts written out (``Görkau`` / ``Goerkau``)."""
    forms: dict = defaultdict(set)
    czech, german = set(), set()
    for r in rows:
        for column, side in (("gerichtsbezirk_cs", czech), ("gerichtsbezirk_cs_modern", czech),
                             ("gerichtsbezirk_de", german)):
            name = r.get(column, "") or ""
            for form in {lxnt.normalise(name), lxdn.fold(name)} - {""}:
                forms[form].add(r["court_id"])
                side.add((form, r["court_id"]))
    return dict(forms), sorted(czech), sorted(german)

def court_id(key: str, forms: dict, names: list) -> str | None:
    """The court a name names against the whole land, else None (D-121): an exact form;
    the one name it *ends* (≥ :data:`TAIL_MIN` letters: *Maršov* for *Horní Maršov*); the
    one it *begins* (≥ :data:`PREFIX_MIN`: ``hluboka``); the one it nearly is
    (``mechanice``, ``cliabarovice``). An ambiguous name (``nova``) is never forced."""
    ids = forms.get(key, set())
    if len(ids) == 1:
        return next(iter(ids))
    for found in ({cid for name, cid in names if len(key) >= TAIL_MIN and name.endswith(key)},
                  {cid for name, cid in names if len(key) >= PREFIX_MIN and name.startswith(key)}):
        if len(found) == 1:
            return next(iter(found))
    from rapidfuzz import fuzz

    best: dict = {}
    for name, cid in names:
        best[cid] = max(best.get(cid, 0.0), fuzz.ratio(key, name))
    ranked = sorted(best.items(), key=lambda kv: -kv[1])
    if ranked and ranked[0][1] >= FUZZY_MIN and (
            len(ranked) == 1 or ranked[0][1] - ranked[1][1] >= FUZZY_MARGIN):
        return ranked[0][0]
    return None

def near(key: str, candidates: list[tuple[str, str]]) -> str | None:
    """The one candidate a name names among a page's few, as ``(form, what)`` pairs:
    the shorter of the two begins or ends the other (≥ :data:`NEAR_LETTERS` letters),
    else the best ratio reaches :data:`NEAR_MIN` and leads the next candidate by
    :data:`NEAR_MARGIN`. Returns the candidate's ``what``; None when it is not one."""
    from rapidfuzz import fuzz

    if not key:
        return None
    hits = {what for form, what in candidates
            if min(len(key), len(form)) >= NEAR_LETTERS
            and (form.startswith(key) or key.startswith(form)
                 or form.endswith(key) or key.endswith(form))}
    if len(hits) == 1:
        return next(iter(hits))
    if hits:
        return None
    best: dict = {}
    for form, what in candidates:
        best[what] = max(best.get(what, 0.0), fuzz.ratio(key, form))
    ranked = sorted(best.items(), key=lambda kv: -kv[1])
    if ranked and ranked[0][1] >= NEAR_MIN and (
            len(ranked) == 1 or ranked[0][1] - ranked[1][1] >= NEAR_MARGIN):
        return ranked[0][0]
    return None

def load_aliases(edition: str = "cz", path: Path = ALIASES) -> dict[str, tuple[str, str]]:
    """``{slug: (court_id, kind)}`` from the curated table, for the edition's land and
    language (``edition`` column ``cz``, ``de`` or ``both``)."""
    if not Path(path).exists():
        return {}
    language = "cz" if edition == "cz" else "de"
    out: dict = {}
    with Path(path).open(encoding="utf-8", newline="") as fh:
        for r in csv.DictReader(fh):
            if r["land"] == LAND[edition] and r["edition"] in (language, "both"):
                out[slug(r["printed"], german=language == "de")] = (r["court_id"], r["kind"])
    return out

class CourtKey:
    """Resolves a section's printed court name to a backbone court of the edition's land.

    ``authority`` (``lexikon_1910_names.build_authority``) reads the okres a running
    head names; without it step 4 is skipped, and step 3 resolves a head's court by
    name only.
    """

    def __init__(self, edition: str = "cz", rows: list[dict] | None = None,
                 aliases: dict | None = None, authority=None):
        self.edition = edition
        self.german = edition != "cz"
        rows = load_backbone(edition) if rows is None else rows
        self.courts = {r["court_id"]: r for r in rows}
        self.forms, czech, german = court_forms(rows)
        self.names = czech + german if self.german else czech
        self.city = {r["court_id"] for r in rows if "(město)" in (r.get("okres_cs") or "")}
        self.okres_courts: dict = defaultdict(list)
        for form, cid in czech + german:
            self.okres_courts[int(cid[:6])].append((form, cid))
        self.aliases = load_aliases(edition) if aliases is None else aliases
        self.authority = authority

    def by_name(self, key: str) -> tuple[str | None, str]:
        """Steps 1–2 on a match key: ``(court_id, "backbone" | "alias")`` or ``(None, "")``."""
        for variant in (slug_variants(key) if self.german else [key]):
            ids = self.forms.get(variant, set())
            if len(ids) == 1:
                return next(iter(ids)), "backbone"
            if len(ids - self.city) == 1:                  # twin: the bare name is the rural court
                return next(iter(ids - self.city)), "backbone"
            if variant in self.aliases:
                return self.aliases[variant][0], "alias"
        for variant in (slug_variants(key) if self.german else [key]):
            found = court_id(variant, {}, self.names)        # exact forms were tried above
            if found:
                return found, "backbone"
        return None, ""

    def head_okresy(self, head: str) -> list[int]:
        """The okresy a running head names, as the backbone numbers them.

        A political name the okres authority does not take is read as the court it
        names, an okres being named after its seat court: the register's *Fridland*
        (scan 0090, second in its head) is court Frýdlant's alias, so okres Frýdlant."""
        if self.authority is None or not head:
            return []
        out = []
        for i, name in enumerate(lxn.head_politicals(head)):
            okres_id = self.authority.match_okres_name(name, weak=(i == 0))[0]
            if okres_id is None:
                seat = self.by_name(slug(name, self.german))[0]
                okres_id = int(seat[:6]) if seat else None
            if okres_id is not None:
                out.append(int(okres_id))
        return out

    def in_okres(self, key: str, okresy: list[int]) -> str | None:
        """The court of the head's okresy a name names (step 4)."""
        candidates = [pair for o in okresy for pair in self.okres_courts.get(o, [])]
        return near(key, candidates) if candidates else None

    def resolve(self, name: str, head: str = "", readings: tuple = ()) -> tuple[str | None, str]:
        """``(court_id, source)`` for a printed court name; ``(None, "unresolved")``."""
        found = self._resolve(name, head)
        if found[0]:
            return found
        votes = Counter(cid for cid, _how in (self._resolve(r, head) for r in readings) if cid)
        ranked = votes.most_common(2)
        if ranked and ranked[0][1] >= READER_MIN and (len(ranked) == 1 or ranked[1][1] < ranked[0][1]):
            return ranked[0][0], f"readers ({ranked[0][1]} of {len(readings)})"
        return None, "unresolved"

    def _resolve(self, name: str, head: str) -> tuple[str | None, str]:
        key = slug(name, self.german)
        found, how = self.by_name(key)
        okresy = self.head_okresy(head)
        if found and okresy and int(found[:6]) not in okresy:
            # A short name the register prints can be another court's whole name: on the
            # Broumov pages *Teplice* is Teplice nad Metují, not Teplice(-Šanov); on the
            # Děčín pages *Benešov* is Benešov nad Ploučnicí; on the Krumlov pages *Planá*
            # is Horní Planá. A court of the head's own okresy that the name names wins.
            local = self.in_okres(key, okresy)
            if local:
                return local, "okres"
        if found:
            return found, how
        printed = [(slug(c, self.german), c) for c in head_courts(head)]
        hit = near(key, printed)
        if hit:
            found = self.by_name(slug(hit, self.german))[0] or self.in_okres(slug(hit, self.german), okresy)
            if found:
                return found, "head"
        found = self.in_okres(key, okresy)
        return (found, "okres") if found else (None, "")

    def check(self, court: str | None, head: str) -> str:
        """``""`` when the court lies in an okres the head names (or the head names none);
        else a note. A guard on every step, the first ones included."""
        okresy = self.head_okresy(head)
        if not court or not okresy or int(court[:6]) in okresy:
            return ""
        return "court outside the okresy the running head names"

    def name_of(self, court: str | None) -> str:
        return (self.courts.get(court) or {}).get("gerichtsbezirk_cs", "") if court else ""

def lexikon_1910_courts_report(rows: list[dict]) -> str:
    """How the sections' courts were settled: a count per source, and every unresolved or
    checked section."""
    by_source = Counter(r["court_source"].split(" (")[0] for r in rows)
    lines = [f"courts: {len(rows)} sections — " + ", ".join(f"{k} {n}" for k, n in by_source.most_common())]
    for r in rows:
        if not r["court_id"] or r.get("court_check"):
            lines.append(f"  {r['page']} {r['district']:24s} -> {r['court_id'] or '—'} "
                         f"{r.get('court_cs', '')} {r.get('court_check', '')}".rstrip())
    return "\n".join(lines)

  Kouřim, *Nová Paka* stored as ``Nová Рака`` with a Cyrillic *Рака*;
* **the register prints its own name** — *Krumlov*, *Hanšpach*, *Bastianperk*,
  *Teplice-Šanov* — where the backbone has *Český Krumlov*, *Lipová*, *Hora Svatého
  Šebestiána*, *Teplice*; and two courts it prints were created after the census;
* **the name is shared** — *Rokytnice* is a court in okres Žamberk and in okres
  Jilemnice; *Liberec* is the rural court's name and the statutory city's.

A section's court is resolved in steps, and the step that settles it is kept:

1. ``backbone`` — the name against every name form of the edition's land (Czech, modern
   Czech, German): exact; the one form it ends (≥ 5 letters: *Maršov* for *Horní
   Maršov*); the one it begins (≥ 6); the one it nearly is (ratio ≥ 85 and ≥ 10 above the
   next court) — D-121's rule. Cyrillic look-alikes are folded first.
2. ``alias`` — the register's own names (``reference/places/court_aliases.csv``),
   including the two courts created after the census, each filed under the one 1910
   court it was cut from (the German edition's *Änderungen* C.1–2).
3. ``head`` — the page's running head prints its courts a second time ("Teplá: Teplá. —
   Teplice-Šanov: Teplice-Šanov."). A name steps 1–2 cannot settle is matched among
   those few names, and the name it matches is resolved by steps 1–2, else among the
   courts of the okres the head names.
4. ``okres`` — the name matched among the backbone courts of the okresy the head names
   (``lexikon_1910_names.Authority.match_okres_name``: look-alikes folded, ties refused).
5. ``readers`` — the readers' own transcriptions of the heading, which the crop audit
   keeps (D-85: a reader is never told the label), each resolved by steps 1–4: a court
   at least :data:`READER_MIN` readings name, and no other court as often.

What none of them settles stays unresolved: a court is never guessed. Among a page's few
names a match needs five letters — the German *Tepl* (Teplá) begins *Teplice-Šanov* —
and the best of several must lead the next by ten points.

**Twin courts.** The backbone carries *Liberec* twice: the rural court (okres Liberec)
and the statutory city's (okres Liberec (město)). A bare name is the rural one, as a
bare district name is everywhere in the project (D-17); the city's own pages are
running text, not the table and apparatus this resolves (D-82).

The resolved court sits **beside** the section's label, never in its place: the label
(the text layer's slug) is how the gold sheets, the ledger and the final inventory key a
section, and a sealed sheet is never edited (D-55).
"""

from __future__ import annotations

__version__ = "0.1.0"

import csv
import re
from collections import Counter, defaultdict
from pathlib import Path

from utils.lexikon_1910 import lexikon_1910_de_notes as lxdn
from utils.lexikon_1910 import lexikon_1910_names as lxn
from utils.lexikon_1910 import lexikon_1910_notes as lxnt

ROOT = Path(__file__).resolve().parents[2]
DISTRICTS = ROOT / "data" / "final" / "geo" / "districts_1910.csv"
ALIASES = ROOT / "reference" / "places" / "court_aliases.csv"

#: Each edition's land in the backbone. A Bohemian register names no Moravian court, and
#: with Moravia left in, *Krumlov* and *Budějovice* each ended two courts (Moravský
#: Krumlov, Moravské Budějovice) and resolved to neither.
LAND = {"cz": "Böhmen", "de-bohemia": "Böhmen", "de-moravia": "Mähren", "de-silesia": "Schlesien"}

#: D-121's rule for a name against the whole land: the one form it ends, the one it
#: begins, the one it nearly is and by how much the next court trails.
TAIL_MIN, PREFIX_MIN = 5, 6
FUZZY_MIN, FUZZY_MARGIN = 85, 10

#: Among the few names one page prints (steps 3–5): the shorter name of a pair one of
#: which begins or ends the other needs this many letters, and a best ratio must reach
#: the floor and lead the next candidate by the margin. Five letters keep the German
#: *Tepl* (Teplá, 4) from taking *Teplice-Šanov*; a floor of 60 takes ``jiralovice`` to
#: *Kralovice* (84) and ``sedlsc`` to *Sedlec* (83) when they are the page's own courts.
NEAR_LETTERS, NEAR_MIN, NEAR_MARGIN = 5, 60, 10

#: Readings of a heading (step 5) that must name one court, none naming another as often.
READER_MIN = 2

#: A German heading shortens a court's first word; the section label keeps it without
#: the full stop (``bohmaicha`` for *Böhm. Aicha*). The slug's start is written out as the
#: backbone spells it — the same words as ``lexikon_1910_de_notes._COURT_ABBREVIATIONS``.
_SLUG_ABBREVIATIONS = (("bohm", "boehmisch"), ("bhm", "boehmisch"), ("mahr", "maehrisch"),
                       ("ung", "ungarisch"))


def slug(name: str, german: bool = False) -> str:
    """A printed court name's match key.

    Czech: Cyrillic look-alikes folded (``Nová Рака`` -> ``novapaka``), then accents
    and all but letters dropped (``lexikon_1910_notes.normalise``). German: its
    abbreviations written out and its umlauts as the backbone spells them
    (``Böhm. Aicha`` -> ``boehmischaicha``).
    """
    if german:
        return lxdn.fold(lxdn.expand_court_abbreviations(lxn.fold_lookalikes(name)))
    return lxnt.normalise(lxn.fold_lookalikes(name))


def slug_variants(key: str) -> list[str]:
    """``key`` and, for a German slug that begins with a shortened first word, the slug
    with that word written out (``bohmaicha`` -> ``boehmischaicha``)."""
    out = [key]
    for short, full in _SLUG_ABBREVIATIONS:
        if key.startswith(short) and not key.startswith(full) \
                and not key.startswith(short + "isch") and len(key) > len(short) + 2:
            out.append(full + key[len(short):])
    return out


def load_backbone(edition: str = "cz", path: Path = DISTRICTS) -> list[dict]:
    """The backbone's courts of the edition's land."""
    with Path(path).open(encoding="utf-8", newline="") as fh:
        return [r for r in csv.DictReader(fh) if r["land"] == LAND[edition]]


def court_forms(rows: list[dict]) -> tuple[dict, list, list]:
    """``({form: {court_id}}, Czech [(form, court_id)], German [(form, court_id)])``.

    Every court under its Czech, modern Czech and German name, each plainly normalised
    and with umlauts written out (``Görkau`` / ``Goerkau``)."""
    forms: dict = defaultdict(set)
    czech, german = set(), set()
    for r in rows:
        for column, side in (("gerichtsbezirk_cs", czech), ("gerichtsbezirk_cs_modern", czech),
                             ("gerichtsbezirk_de", german)):
            name = r.get(column, "") or ""
            for form in {lxnt.normalise(name), lxdn.fold(name)} - {""}:
                forms[form].add(r["court_id"])
                side.add((form, r["court_id"]))
    return dict(forms), sorted(czech), sorted(german)


def court_id(key: str, forms: dict, names: list) -> str | None:
    """The court a name names against the whole land, else None (D-121): an exact form;
    the one name it *ends* (≥ :data:`TAIL_MIN` letters: *Maršov* for *Horní Maršov*); the
    one it *begins* (≥ :data:`PREFIX_MIN`: ``hluboka``); the one it nearly is
    (``mechanice``, ``cliabarovice``). An ambiguous name (``nova``) is never forced."""
    ids = forms.get(key, set())
    if len(ids) == 1:
        return next(iter(ids))
    for found in ({cid for name, cid in names if len(key) >= TAIL_MIN and name.endswith(key)},
                  {cid for name, cid in names if len(key) >= PREFIX_MIN and name.startswith(key)}):
        if len(found) == 1:
            return next(iter(found))
    from rapidfuzz import fuzz

    best: dict = {}
    for name, cid in names:
        best[cid] = max(best.get(cid, 0.0), fuzz.ratio(key, name))
    ranked = sorted(best.items(), key=lambda kv: -kv[1])
    if ranked and ranked[0][1] >= FUZZY_MIN and (
            len(ranked) == 1 or ranked[0][1] - ranked[1][1] >= FUZZY_MARGIN):
        return ranked[0][0]
    return None


def near(key: str, candidates: list[tuple[str, str]]) -> str | None:
    """The one candidate a name names among a page's few, as ``(form, what)`` pairs:
    the shorter of the two begins or ends the other (≥ :data:`NEAR_LETTERS` letters),
    else the best ratio reaches :data:`NEAR_MIN` and leads the next candidate by
    :data:`NEAR_MARGIN`. Returns the candidate's ``what``; None when it is not one."""
    from rapidfuzz import fuzz

    if not key:
        return None
    hits = {what for form, what in candidates
            if min(len(key), len(form)) >= NEAR_LETTERS
            and (form.startswith(key) or key.startswith(form)
                 or form.endswith(key) or key.endswith(form))}
    if len(hits) == 1:
        return next(iter(hits))
    if hits:
        return None
    best: dict = {}
    for form, what in candidates:
        best[what] = max(best.get(what, 0.0), fuzz.ratio(key, form))
    ranked = sorted(best.items(), key=lambda kv: -kv[1])
    if ranked and ranked[0][1] >= NEAR_MIN and (
            len(ranked) == 1 or ranked[0][1] - ranked[1][1] >= NEAR_MARGIN):
        return ranked[0][0]
    return None


#: The running head prints ``political okres: court, court.`` — ``Chomútov: Chomútov,
#: Jirkov.`` — twice on a page where one okres ends and the next begins, the two parts
#: separated by an em dash. The same pattern as ``build_de_page_map._HEAD_RE``.
_HEAD_RE = re.compile(r"([A-Za-zÁ-Žá-žÄÖÜäöüß][\w.\-’'À-ſ ]{1,28}?)\s*:\s*([^.:]{2,60})\.")


def head_courts(head: str) -> list[str]:
    """The court names a running head prints, as written, in order."""
    out: list[str] = []
    for _political, courts in _HEAD_RE.findall(lxn.fold_lookalikes(head or "")):
        out += [c.strip() for c in courts.split(",") if len(slug(c)) >= 3]
    return out


def load_aliases(edition: str = "cz", path: Path = ALIASES) -> dict[str, tuple[str, str]]:
    """``{slug: (court_id, kind)}`` from the curated table, for the edition's land and
    language (``edition`` column ``cz``, ``de`` or ``both``)."""
    if not Path(path).exists():
        return {}
    language = "cz" if edition == "cz" else "de"
    out: dict = {}
    with Path(path).open(encoding="utf-8", newline="") as fh:
        for r in csv.DictReader(fh):
            if r["land"] == LAND[edition] and r["edition"] in (language, "both"):
                out[slug(r["printed"], german=language == "de")] = (r["court_id"], r["kind"])
    return out


class CourtKey:
    """Resolves a section's printed court name to a backbone court of the edition's land.

    ``authority`` (``lexikon_1910_names.build_authority``) reads the okres a running
    head names; without it step 4 is skipped, and step 3 resolves a head's court by
    name only.
    """

    def __init__(self, edition: str = "cz", rows: list[dict] | None = None,
                 aliases: dict | None = None, authority=None):
        self.edition = edition
        self.german = edition != "cz"
        rows = load_backbone(edition) if rows is None else rows
        self.courts = {r["court_id"]: r for r in rows}
        self.forms, czech, german = court_forms(rows)
        self.names = czech + german if self.german else czech
        self.city = {r["court_id"] for r in rows if "(město)" in (r.get("okres_cs") or "")}
        self.okres_courts: dict = defaultdict(list)
        for form, cid in czech + german:
            self.okres_courts[int(cid[:6])].append((form, cid))
        self.aliases = load_aliases(edition) if aliases is None else aliases
        self.authority = authority

    def by_name(self, key: str) -> tuple[str | None, str]:
        """Steps 1–2 on a match key: ``(court_id, "backbone" | "alias")`` or ``(None, "")``."""
        for variant in (slug_variants(key) if self.german else [key]):
            ids = self.forms.get(variant, set())
            if len(ids) == 1:
                return next(iter(ids)), "backbone"
            if len(ids - self.city) == 1:                  # twin: the bare name is the rural court
                return next(iter(ids - self.city)), "backbone"
            if variant in self.aliases:
                return self.aliases[variant][0], "alias"
        for variant in (slug_variants(key) if self.german else [key]):
            found = court_id(variant, {}, self.names)        # exact forms were tried above
            if found:
                return found, "backbone"
        return None, ""

    def head_okresy(self, head: str) -> list[int]:
        """The okresy a running head names, as the backbone numbers them.

        A political name the okres authority does not take is read as the court it
        names, an okres being named after its seat court: the register's *Fridland*
        (scan 0090, second in its head) is court Frýdlant's alias, so okres Frýdlant."""
        if self.authority is None or not head:
            return []
        out = []
        for i, name in enumerate(lxn.head_politicals(head)):
            okres_id = self.authority.match_okres_name(name, weak=(i == 0))[0]
            if okres_id is None:
                seat = self.by_name(slug(name, self.german))[0]
                okres_id = int(seat[:6]) if seat else None
            if okres_id is not None:
                out.append(int(okres_id))
        return out

    def in_okres(self, key: str, okresy: list[int]) -> str | None:
        """The court of the head's okresy a name names (step 4)."""
        candidates = [pair for o in okresy for pair in self.okres_courts.get(o, [])]
        return near(key, candidates) if candidates else None

    def resolve(self, name: str, head: str = "", readings: tuple = ()) -> tuple[str | None, str]:
        """``(court_id, source)`` for a printed court name; ``(None, "unresolved")``."""
        found = self._resolve(name, head)
        if found[0]:
            return found
        votes = Counter(cid for cid, _how in (self._resolve(r, head) for r in readings) if cid)
        ranked = votes.most_common(2)
        if ranked and ranked[0][1] >= READER_MIN and (len(ranked) == 1 or ranked[1][1] < ranked[0][1]):
            return ranked[0][0], f"readers ({ranked[0][1]} of {len(readings)})"
        return None, "unresolved"

    def _resolve(self, name: str, head: str) -> tuple[str | None, str]:
        key = slug(name, self.german)
        found, how = self.by_name(key)
        okresy = self.head_okresy(head)
        if found and okresy and int(found[:6]) not in okresy:
            # A short name the register prints can be another court's whole name: on the
            # Broumov pages *Teplice* is Teplice nad Metují, not Teplice(-Šanov); on the
            # Děčín pages *Benešov* is Benešov nad Ploučnicí; on the Krumlov pages *Planá*
            # is Horní Planá. A court of the head's own okresy that the name names wins.
            local = self.in_okres(key, okresy)
            if local:
                return local, "okres"
        if found:
            return found, how
        printed = [(slug(c, self.german), c) for c in head_courts(head)]
        hit = near(key, printed)
        if hit:
            found = self.by_name(slug(hit, self.german))[0] or self.in_okres(slug(hit, self.german), okresy)
            if found:
                return found, "head"
        found = self.in_okres(key, okresy)
        return (found, "okres") if found else (None, "")

    def check(self, court: str | None, head: str) -> str:
        """``""`` when the court lies in an okres the head names (or the head names none);
        else a note. A guard on every step, the first ones included."""
        okresy = self.head_okresy(head)
        if not court or not okresy or int(court[:6]) in okresy:
            return ""
        return "court outside the okresy the running head names"

    def name_of(self, court: str | None) -> str:
        return (self.courts.get(court) or {}).get("gerichtsbezirk_cs", "") if court else ""


def lexikon_1910_courts_report(rows: list[dict]) -> str:
    """How the sections' courts were settled: a count per source, and every unresolved or
    checked section."""
    by_source = Counter(r["court_source"].split(" (")[0] for r in rows)
    lines = [f"courts: {len(rows)} sections — " + ", ".join(f"{k} {n}" for k, n in by_source.most_common())]
    for r in rows:
        if not r["court_id"] or r.get("court_check"):
            lines.append(f"  {r['page']} {r['district']:24s} -> {r['court_id'] or '—'} "
                         f"{r.get('court_cs', '')} {r.get('court_check', '')}".rstrip())
    return "\n".join(lines)
