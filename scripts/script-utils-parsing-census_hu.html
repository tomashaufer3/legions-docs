TL;DR

Cisleithanian analogue: utils.parsing.census (okres-level Rumpler-Seger). This loads the járás-level 1910 census for the Felvidék backbone (utils.parsing.districts_hu) from the GISta Hungarorum workbooks in data/raw/census_hu/tables/ and joins them by exact unit code (kod / IDJaras1910 == unit_id) — no name matching.

How it works

Tables loaded

  • …_ETN.xlsx — mother tongue + religion (the key ethnic covariate: tot_anyanyelv = Slovak speakers, plus magyar / német / ruthén / …, and the confessional breakdown). Prefixed etn_.
  • …_Agrar_1910.xlsx — agricultural land use, yields and agrarian population. Prefixed agr_.

The sprawling 281-column general table (…_3_1910.xlsx; vital statistics, industry, post/ electricity infrastructure) is not loaded here — low analytic value, messy Magyar headers; add later if needed. Columns are prefixed by family (etn_ / agr_) during the join, then renamed to English by utils.parsing.census_labels as the final step (lang_slovak, rel_jewish …); the Magyar→English crosswalk is reference/census_column_labels.csv.

Derived shares: slovak_share, magyar_share, german_share, ruthen_share (mother tongue) plus religion shares — the Transleithanian analogue of the Czech-speaker share.

Usage

from utils.parsing import census_hu as czh
census = czh.build_census_hu(TABLES_DIR, backbone)     # 192 Felvidék units × covariates
czh.census_hu_report(census)

def build_census_hu(tables_dir, backbone: pd.DataFrame) -> pd.DataFrame:
    """Build the járás-level census panel, restricted to the *backbone* units.

    Joins ETN (mother tongue + religion) and Agrar (agriculture) on ``unit_id`` and adds derived
    language shares. Returns one row per backbone unit (``unit_id`` + ``megye_id`` + covariates);
    units with no census row keep NA covariates.
    """
    tables_dir = Path(tables_dir)
    etn = pd.read_excel(tables_dir / _ETN)
    agr = pd.read_excel(tables_dir / _AGRAR)

    etn["unit_id"] = _clean_key(etn["kod"])
    agr["unit_id"] = _clean_key(agr["IDJaras1910"])

    # prefix source (Magyar) columns per family; drop the redundant key/label columns
    etn_drop = {"kod", "jarasok_varosok", "unit_id"}
    agr_drop = {"IDJaras1910", "IDJaras_varos1910", "nev_eredeti_1912", "Jaras_1910",
                "IDJaras1910.1", "Jaras_1910.1", "IDMegye1910", "Megye_1910", "unit_id"}
    etn_v = etn[["unit_id"] + [c for c in etn.columns if c not in etn_drop]].copy()
    etn_v.columns = ["unit_id"] + [f"etn_{c}" for c in etn_v.columns if c != "unit_id"]
    agr_v = agr[["unit_id"] + [c for c in agr.columns if c not in agr_drop]].copy()
    agr_v.columns = ["unit_id"] + [f"agr_{c}" for c in agr_v.columns if c != "unit_id"]

    keys = backbone[["unit_id", "megye_id", "megye_hu", "name_hu", "name_sk"]].drop_duplicates("unit_id")
    out = (keys.merge(etn_v, on="unit_id", how="left")
                .merge(agr_v.drop_duplicates("unit_id"), on="unit_id", how="left"))

    # derived mother-tongue shares (fraction of present population)
    pop = out.get("etn_jelenlevo_nepesseg")
    if pop is not None:
        for share, col in [("slovak_share", "etn_tot_anyanyelv"),
                           ("magyar_share", "etn_magyar_anyanyelv"),
                           ("german_share", "etn_nemet_anyanyelv"),
                           ("ruthen_share", "etn_ruthen_anyanyelv")]:
            if col in out.columns:
                out[share] = pd.to_numeric(out[col], errors="coerce") / pop.replace(0, pd.NA)
    out = out.reset_index(drop=True)
    from utils.parsing import census_labels as _cl          # English labels + religion shares
    return _cl.to_english_census(out, "hu")

def census_hu_report(census: pd.DataFrame) -> None:
    """Print coverage of the járás census panel."""
    n = len(census)
    popcol = "pop_present" if "pop_present" in census else "etn_jelenlevo_nepesseg"
    matched = census[popcol].notna().sum() if popcol in census else 0
    n_cov = sum(not c.startswith(("unit_", "megye_", "name_")) for c in census.columns)
    print(f"Járás census panel  (units = {n}, covariates = {n_cov})")
    print(f"  population present: {matched}/{n} ({matched/n*100:.1f}%)")
    if "slovak_share" in census:
        s = census["slovak_share"].dropna()
        print(f"  Slovak-speaker share:  mean {s.mean():.1%}, median {s.median():.1%} "
              f"(units ≥50% Slovak: {(s >= 0.5).sum()}/{len(s)})")

============================================================================

from __future__ import annotations

from pathlib import Path

import pandas as pd

__version__ = "1.1.0"

_ETN = "10_MO_JARAS_4_1910_ETN.xlsx"
_AGRAR = "10_MO_JARAS_2_Agrar_1910.xlsx"


def _clean_key(s: pd.Series) -> pd.Series:
    """Unit code as a clean string (strip stray whitespace some cells carry)."""
    return s.astype(str).str.strip()
