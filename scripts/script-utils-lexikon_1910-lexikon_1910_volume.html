How it works

The Spezialortsrepertorium / Podrobný seznam míst was set by one office, on one plan, for every crown land: the same 17-column table, the same two-column apparatus under it, the same postal legend in the footer. What changes from volume to volume is small and enumerable, and this module is where it is written down rather than scattered through the parsers:

  • the source — which PDF, whether it carries an ALTO, a word layer, plain text, or nothing; the scans that are table pages, and which are the statutory cities' prose pages;
  • the language of the headings — Části osad v soudním okresu X: against Ortsbestandteile im Gerichtsbezirk X: — and of the abbreviations;
  • the column schema — Silesia prints a fourth Umgangssprache column (polnisch), so its table has 18 columns and its language identity four parts. A crown-land volume not yet owned (Galicia, Bukovina, the Littoral) will differ in exactly this place.

A profile is a TOML file under reference/places/volumes/&lt;slug&gt;.toml, read with the standard library's tomllib (Python 3.11+).

Nothing in the existing pipeline reads a profile. lexikon_1910, lexikon_1910_notes and the pilot keep their own constants, which are the Czech volume's; czech_drift checks that the Czech profile states exactly those constants, so the two cannot diverge unnoticed, and the image-only geometry (lexikon_1910_image) and its measurement scripts are the code that takes a profile. Moving the module constants themselves onto the profile is a mechanical follow-up that would make every import depend on tomllib; it is left for the author to decide.

    @property
    def n_columns(self) -> int:
        return len(self.columns)

    @property
    def column_names(self) -> list[str]:
        return [c.name for c in self.columns]

    @property
    def table_pages(self) -> list[int]:
        skip = set(self.prose_pages) | set(self.other_pages)
        return [n for n in range(self.table_first, self.table_last + 1) if n not in skip]

    def column_index(self, name: str) -> int:
        return self.column_names.index(name)

    def header_spans(self) -> list[tuple[int, int]]:
        """``(first, last)`` column indices of each header group, for comparing
        with what :mod:`lexikon_1910_image` reads off the rules."""
        return [(self.column_index(g.first), self.column_index(g.last))
                for g in self.header_groups]

    @property
    def tail_re(self) -> re.Pattern:
        if self._tail_re is None:
            self._tail_re = re.compile(self.heading_tail, re.IGNORECASE)
        return self._tail_re

    def find_heading(self, line: str) -> tuple[str, str, int] | None:
        """``lexikon_1910_notes.find_heading`` with this volume's words.

        The same rule — a tail ending in a colon, and exactly one kind's marks in
        the ``mark_window`` letters before it — so on the Czech profile it gives
        the same answer on every line (checked by :func:`czech_drift` and the
        geometry measurement).
        """
        for match in self.tail_re.finditer(line):
            prefix = normalise(line[:match.start()])[-self.mark_window:]
            parts = any(mark in prefix for mark in self.parts_marks)
            offices = any(mark in prefix for mark in self.offices_marks)
            if parts == offices:
                continue
            return ("parts" if parts else "offices",
                    match.group("district").strip(" .,;-—–"), match.end())
        return None

def normalise(name: str) -> str:
    """As ``lexikon_1910_notes.normalise``: unaccented, lowercase, letters only."""
    decomposed = unicodedata.normalize("NFKD", str(name or ""))
    stripped = "".join(c for c in decomposed if not unicodedata.combining(c))
    return re.sub(r"[^a-z]", "", stripped.lower())

def parse_profile(data: dict) -> VolumeProfile:
    """A :class:`VolumeProfile` from a parsed TOML document."""
    src, pages, head, abbr = (data.get(k, {}) for k in
                              ("source", "pages", "headings", "abbreviations"))
    columns = tuple(Column(c["name"], c["kind"], c.get("label", ""))
                    for c in data["columns"])
    names = {c.name for c in columns}
    identities = tuple(Identity(i["key"], i.get("label", ""), tuple(i["parts"]), i["whole"])
                       for i in data.get("identities", []))
    groups = tuple(HeaderGroup(g["label"], g["first"], g["last"])
                   for g in data.get("header_groups", []))
    for ident in identities:
        missing = [c for c in (*ident.parts, ident.whole) if c not in names]
        if missing:
            raise ValueError(f"{data['slug']}: identity {ident.key} names unknown columns {missing}")
    for g in groups:
        if g.first not in names or g.last not in names:
            raise ValueError(f"{data['slug']}: header group {g.label!r} names unknown columns")
    if len(names) != len(columns):
        raise ValueError(f"{data['slug']}: duplicate column names")
    profile = VolumeProfile(
        slug=data["slug"], title=data.get("title", ""),
        crown_land=data.get("crown_land", ""), language=data["language"],
        source_pdf=_path(src["pdf"]), text_layer=src.get("text_layer", "none"),
        render=src.get("render", "native"),
        alto_dir=_path(src.get("alto_dir")), plaintext_dir=_path(src.get("plaintext_dir")),
        grid_volume=src.get("grid_volume", ""),
        table_first=int(pages["table_first"]), table_last=int(pages["table_last"]),
        prose_pages=tuple(pages.get("prose", ())), other_pages=tuple(pages.get("other", ())),
        sealed_pages=tuple(pages.get("sealed", ())),
        columns=columns, identities=identities, header_groups=groups,
        parts_marks=tuple(head.get("parts_marks", ())),
        offices_marks=tuple(head.get("offices_marks", ())),
        heading_tail=head.get("tail_regex", ""), mark_window=int(head.get("mark_window", 40)),
        parts_phrase=head.get("parts_phrase", ""), offices_phrase=head.get("offices_phrase", ""),
        postal_phrase=head.get("postal_phrase", ""),
        abbreviations=_path(abbr.get("file")), abbreviation_lang=abbr.get("lang", ""),
        notes=tuple(data.get("notes", ())),
    )
    if profile.heading_tail:
        profile.tail_re                     # compile now: a bad pattern fails at load
    return profile

def load_profile(slug_or_path: str | Path) -> VolumeProfile:
    """Load ``reference/places/volumes/<slug>.toml`` (or a path to a TOML file)."""
    import tomllib

    path = Path(slug_or_path)
    if path.suffix != ".toml":
        path = PROFILE_DIR / f"{slug_or_path}.toml"
    with path.open("rb") as fh:
        return parse_profile(tomllib.load(fh))

def czech_drift(profile: VolumeProfile | None = None) -> list[str]:
    """Where the Czech profile disagrees with the constants the pipeline runs on.

    Empty means the profile *is* the pipeline's Czech configuration: the column
    schema of ``lexikon_1910.COLUMNS``, its invariants, the heading marks, tail
    and window of ``lexikon_1910_notes``, and the abbreviation file and PDF of
    ``lexikon_1910_pages``.
    """
    from utils.lexikon_1910 import lexikon_1910 as lx
    from utils.lexikon_1910 import lexikon_1910_notes as lxnt
    from utils.lexikon_1910 import lexikon_1910_pages as lxp

    p = profile or load_profile("cechy_cz")
    out = []
    if [(c.name, c.kind) for c in p.columns] != list(lx.COLUMNS):
        out.append("columns differ from lexikon_1910.COLUMNS")
    mine = [(i.key, i.label, i.parts, i.whole) for i in p.identities]
    theirs = [(i.key, i.label, i.parts, i.whole) for i in lx.INVARIANTS]
    if mine != theirs:
        out.append("identities differ from lexikon_1910.INVARIANTS")
    if p.parts_marks != lxnt.PARTS_MARKS:
        out.append("parts marks differ from lexikon_1910_notes.PARTS_MARKS")
    if p.offices_marks != lxnt.OFFICES_MARKS:
        out.append("offices marks differ from lexikon_1910_notes.OFFICES_MARKS")
    if p.heading_tail != lxnt.HEADING_TAIL_RE.pattern or not lxnt.HEADING_TAIL_RE.flags & re.IGNORECASE:
        out.append("heading tail differs from lexikon_1910_notes.HEADING_TAIL_RE")
    if p.mark_window != lxnt._MARK_WINDOW:
        out.append("mark window differs from lexikon_1910_notes._MARK_WINDOW")
    if p.abbreviations != lxp.ABBREVIATIONS:
        out.append("abbreviation file differs from lexikon_1910_pages.ABBREVIATIONS")
    if p.source_pdf != lxp.CZ_PDF:
        out.append("PDF differs from lexikon_1910_pages.CZ_PDF")
    if p.alto_dir != lxp.CZ_PAGES:
        out.append("ALTO folder differs from lexikon_1910_pages.CZ_PAGES")
    return out

def lexikon_1910_volume_report(profile: VolumeProfile) -> str:
    lines = [f"lexikon_1910_volume v{__version__} — {profile.slug}: {profile.title}", "",
             f"  source: {profile.source_pdf.name if profile.source_pdf else '-'} "
             f"(text layer: {profile.text_layer}, render: {profile.render})",
             f"  table scans {profile.table_first}-{profile.table_last}: "
             f"{len(profile.table_pages)} table pages, prose {list(profile.prose_pages)}, "
             f"other {list(profile.other_pages)}",
             f"  {profile.n_columns} columns; identities "
             + ", ".join(f"{i.key} ({len(i.parts)} parts)" for i in profile.identities),
             f"  headings: {profile.parts_phrase!r} / {profile.offices_phrase!r}"]
    lines += [f"  note: {n}" for n in profile.notes]
    return "\n".join(lines)

    def find_heading(self, line: str) -> tuple[str, str, int] | None:
        """``lexikon_1910_notes.find_heading`` with this volume's words.

        The same rule — a tail ending in a colon, and exactly one kind's marks in
        the ``mark_window`` letters before it — so on the Czech profile it gives
        the same answer on every line (checked by :func:`czech_drift` and the
        geometry measurement).
        """
        for match in self.tail_re.finditer(line):
            prefix = normalise(line[:match.start()])[-self.mark_window:]
            parts = any(mark in prefix for mark in self.parts_marks)
            offices = any(mark in prefix for mark in self.offices_marks)
            if parts == offices:
                continue
            return ("parts" if parts else "offices",
                    match.group("district").strip(" .,;-—–"), match.end())
        return None


def normalise(name: str) -> str:
    """As ``lexikon_1910_notes.normalise``: unaccented, lowercase, letters only."""
    decomposed = unicodedata.normalize("NFKD", str(name or ""))
    stripped = "".join(c for c in decomposed if not unicodedata.combining(c))
    return re.sub(r"[^a-z]", "", stripped.lower())


def _path(value: str | None) -> Path | None:
    return (ROOT / value) if value else None


def parse_profile(data: dict) -> VolumeProfile:
    """A :class:`VolumeProfile` from a parsed TOML document."""
    src, pages, head, abbr = (data.get(k, {}) for k in
                              ("source", "pages", "headings", "abbreviations"))
    columns = tuple(Column(c["name"], c["kind"], c.get("label", ""))
                    for c in data["columns"])
    names = {c.name for c in columns}
    identities = tuple(Identity(i["key"], i.get("label", ""), tuple(i["parts"]), i["whole"])
                       for i in data.get("identities", []))
    groups = tuple(HeaderGroup(g["label"], g["first"], g["last"])
                   for g in data.get("header_groups", []))
    for ident in identities:
        missing = [c for c in (*ident.parts, ident.whole) if c not in names]
        if missing:
            raise ValueError(f"{data['slug']}: identity {ident.key} names unknown columns {missing}")
    for g in groups:
        if g.first not in names or g.last not in names:
            raise ValueError(f"{data['slug']}: header group {g.label!r} names unknown columns")
    if len(names) != len(columns):
        raise ValueError(f"{data['slug']}: duplicate column names")
    profile = VolumeProfile(
        slug=data["slug"], title=data.get("title", ""),
        crown_land=data.get("crown_land", ""), language=data["language"],
        source_pdf=_path(src["pdf"]), text_layer=src.get("text_layer", "none"),
        render=src.get("render", "native"),
        alto_dir=_path(src.get("alto_dir")), plaintext_dir=_path(src.get("plaintext_dir")),
        grid_volume=src.get("grid_volume", ""),
        table_first=int(pages["table_first"]), table_last=int(pages["table_last"]),
        prose_pages=tuple(pages.get("prose", ())), other_pages=tuple(pages.get("other", ())),
        sealed_pages=tuple(pages.get("sealed", ())),
        columns=columns, identities=identities, header_groups=groups,
        parts_marks=tuple(head.get("parts_marks", ())),
        offices_marks=tuple(head.get("offices_marks", ())),
        heading_tail=head.get("tail_regex", ""), mark_window=int(head.get("mark_window", 40)),
        parts_phrase=head.get("parts_phrase", ""), offices_phrase=head.get("offices_phrase", ""),
        postal_phrase=head.get("postal_phrase", ""),
        abbreviations=_path(abbr.get("file")), abbreviation_lang=abbr.get("lang", ""),
        notes=tuple(data.get("notes", ())),
    )
    if profile.heading_tail:
        profile.tail_re                     # compile now: a bad pattern fails at load
    return profile


def load_profile(slug_or_path: str | Path) -> VolumeProfile:
    """Load ``reference/places/volumes/<slug>.toml`` (or a path to a TOML file)."""
    import tomllib

    path = Path(slug_or_path)
    if path.suffix != ".toml":
        path = PROFILE_DIR / f"{slug_or_path}.toml"
    with path.open("rb") as fh:
        return parse_profile(tomllib.load(fh))


def profiles() -> list[str]:
    """Slugs of every profile on disk."""
    return sorted(p.stem for p in PROFILE_DIR.glob("*.toml"))


def czech_drift(profile: VolumeProfile | None = None) -> list[str]:
    """Where the Czech profile disagrees with the constants the pipeline runs on.

    Empty means the profile *is* the pipeline's Czech configuration: the column
    schema of ``lexikon_1910.COLUMNS``, its invariants, the heading marks, tail
    and window of ``lexikon_1910_notes``, and the abbreviation file and PDF of
    ``lexikon_1910_pages``.
    """
    from utils.lexikon_1910 import lexikon_1910 as lx
    from utils.lexikon_1910 import lexikon_1910_notes as lxnt
    from utils.lexikon_1910 import lexikon_1910_pages as lxp

    p = profile or load_profile("cechy_cz")
    out = []
    if [(c.name, c.kind) for c in p.columns] != list(lx.COLUMNS):
        out.append("columns differ from lexikon_1910.COLUMNS")
    mine = [(i.key, i.label, i.parts, i.whole) for i in p.identities]
    theirs = [(i.key, i.label, i.parts, i.whole) for i in lx.INVARIANTS]
    if mine != theirs:
        out.append("identities differ from lexikon_1910.INVARIANTS")
    if p.parts_marks != lxnt.PARTS_MARKS:
        out.append("parts marks differ from lexikon_1910_notes.PARTS_MARKS")
    if p.offices_marks != lxnt.OFFICES_MARKS:
        out.append("offices marks differ from lexikon_1910_notes.OFFICES_MARKS")
    if p.heading_tail != lxnt.HEADING_TAIL_RE.pattern or not lxnt.HEADING_TAIL_RE.flags & re.IGNORECASE:
        out.append("heading tail differs from lexikon_1910_notes.HEADING_TAIL_RE")
    if p.mark_window != lxnt._MARK_WINDOW:
        out.append("mark window differs from lexikon_1910_notes._MARK_WINDOW")
    if p.abbreviations != lxp.ABBREVIATIONS:
        out.append("abbreviation file differs from lexikon_1910_pages.ABBREVIATIONS")
    if p.source_pdf != lxp.CZ_PDF:
        out.append("PDF differs from lexikon_1910_pages.CZ_PDF")
    if p.alto_dir != lxp.CZ_PAGES:
        out.append("ALTO folder differs from lexikon_1910_pages.CZ_PAGES")
    return out


def lexikon_1910_volume_report(profile: VolumeProfile) -> str:
    lines = [f"lexikon_1910_volume v{__version__} — {profile.slug}: {profile.title}", "",
             f"  source: {profile.source_pdf.name if profile.source_pdf else '-'} "
             f"(text layer: {profile.text_layer}, render: {profile.render})",
             f"  table scans {profile.table_first}-{profile.table_last}: "
             f"{len(profile.table_pages)} table pages, prose {list(profile.prose_pages)}, "
             f"other {list(profile.other_pages)}",
             f"  {profile.n_columns} columns; identities "
             + ", ".join(f"{i.key} ({len(i.parts)} parts)" for i in profile.identities),
             f"  headings: {profile.parts_phrase!r} / {profile.offices_phrase!r}"]
    lines += [f"  note: {n}" for n in profile.notes]
    return "\n".join(lines)
