def religion_report(df: pd.DataFrame, source_col: str = "religion") -> None:
    """Print coverage of the English gloss and the top unmapped raw values."""
    if source_col not in df.columns:
        print(f"No {source_col} column.")
        return
    s = df[source_col]
    n = int(s.notna().sum())
    en = s.map(translate_religion)
    cov = int(en.notna().sum())
    print(f"religion_en: {cov:,}/{n:,} raw values glossed ({cov/n:.1%})")
    unmapped = s[s.notna() & en.isna()]
    if len(unmapped):
        print("  top unmapped:")
        for val, c in unmapped.value_counts().head(10).items():
            print(f"    {c:>5}  {val!r}")

"""
religion.py — English gloss for the ``religion`` value (legionnaire dataset)
============================================================================
The raw ``Vyznání`` field is a small, abbreviated Czech vocabulary: denominations
(``řím.kat.`` = Roman Catholic, ``pravosl.`` = Orthodox, ``bez vyz.`` = no
confession…) plus a handful of *action* phrases recording a change of confession
(``vystoupil z řím.kat., bez vyz.`` = "left Roman Catholic, no confession"). The
top twelve values already cover **98.8 %** of records and a curated map of ~40
covers **99.4 %**; the long tail is left ``<NA>`` rather than guessed.

This is a **value** translation (the column *name* is already English). It adds a
companion ``religion_en`` and leaves ``religion`` untouched.

Usage
-----
    from utils.parsing.religion import add_religion_en, religion_report
    df = add_religion_en(df)          # adds religion_en
    religion_report(df)
"""

def _norm(s: str) -> str:
    """Lowercase, collapse whitespace — the key form the map is stored in."""
    return re.sub(r"\s+", " ", str(s).strip().lower())

def translate_religion(raw) -> str | None:
    """English gloss for one raw ``Vyznání`` value (``None`` if unmapped)."""
    if raw is None or (not isinstance(raw, str) and pd.isna(raw)):
        return None
    return RELIGION_EN.get(_norm(raw))


def add_religion_en(df: pd.DataFrame, source_col: str = "religion",
                    out: str = "religion_en") -> pd.DataFrame:
    """Append ``religion_en`` — the English gloss of ``source_col`` (not in place)."""
    df = df.copy()
    if source_col in df.columns:
        df[out] = df[source_col].map(translate_religion).astype("string")
    return df


def religion_report(df: pd.DataFrame, source_col: str = "religion") -> None:
