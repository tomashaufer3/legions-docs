How it works

Matching an OCR'd Czech block name against an OCR'd German block name compares two noisy strings and hopes. This module replaces that with two independent comparisons against a clean authority: each block is matched to a known 1910 obec, and two blocks are the same municipality when they matched the same obec. That is the settlements_1910 idiom the rest of the project already uses, and it buys three things a direct fuzzy join cannot:

  • Cross-volume pairing that survives bad OCR on both sides. Neither block has to resemble the other, only the authority.
  • Over- and under-segmentation become measurable. A block matching no obec in its okres is a settlement row wrongly promoted; an obec with no block is a municipality the parser lost. Neither is visible to any arithmetic identity.
  • A canonical identity for the output — okres_id plus the modern Czech name — which the final dataset needs regardless.

The German names are the hard part. settlements_1910 carries de_name for only 18% of Bohemian obce and name_1910 for 33%. Two further sources close most of the gap:

  • settlement_aliases_1910.csv — 10,028 "viz" cross-references from the ČSÚ historical lexicon, most of them German forms pointing at a Czech target.
  • The Czech volume itself. It prints every municipality bilingually — "Bílek, Bilek", "Štěpánov, Stěpanow", "Bobkov, také Doubkov, Dobkau" — so once a Czech block is anchored, the German half of its own name cell becomes a variant of that obec. The source we are parsing is the best-covered crosswalk available for it, which is why anchoring runs in two passes: Czech first, harvest, then German.

def head_politicals(head: str) -> list[str]:
    """The political okres names a running head prints, in order.

    The head prints ``Politický okres: Soudní okres.``, and on a page where one
    okres ends and the next begins, two of them: ``Ney dek: Neydek. — Nová Рака:
    Hořice.`` Each name is the text before a colon, after the previous court list
    has ended (a full stop or a dash). Reading stops at the column header, whose
    labels follow the head in the text layer. The page number and any header words
    in front of the first name are left in; :meth:`Authority.match_okres_name`
    tolerates a short lead and the header words are cut here.
    """
    chunks = str(head or "").split(":")
    names: list[str] = []
    for index, chunk in enumerate(chunks[:-1]):
        if index > 0:
            if _HEADER_START_RE.search(chunk):
                break
            chunk = re.split(r"[.—]", chunk)[-1]
        name = _HEADER_START_RE.split(chunk)[-1].strip()
        if len(name) > _MAX_HEAD_NAME:      # the colon is in the table, not the head
            break
        if _STATUTORY_RE.search(name):
            # Scans 0016-0018: the city after the colon is the okres.
            city = re.split(r"[.,;—]", chunks[index + 1])[0].strip()
            if city:
                names.append(city)
            continue
        names.append(name)
    return names

def normalise(name: str) -> str:
    """Match key: unaccented, lowercase, letters and digits only."""
    decomposed = unicodedata.normalize("NFKD", str(name or ""))
    stripped = "".join(c for c in decomposed if not unicodedata.combining(c))
    return re.sub(r"[^a-z0-9]", "", stripped.lower())

def parse_name_cell(text: str) -> list[str]:
    """Split a printed name cell into its candidate name forms.

    The register prints ``Czech, German``, sometimes with an alternative Czech form
    between them (``Bobkov, také Doubkov, Dobkau``) and usually with symbols and
    bracketed settlement numbers trailing. Every separated part is returned as a
    candidate rather than assuming which position holds which language — the volume
    is not consistent about it, and the authority match will pick.

    The **unsplit** cell is returned as a candidate too. Splitting on the full stop
    is what makes ``Rosice. Rositz`` matchable, but it would also cut a name that
    legitimately contains one (``Sv. Jan``) into fragments too short to survive the
    length filter, silently losing a name that used to match. Offering both forms
    costs one extra comparison and cannot lose.

    A leading ``Y`` is also offered as ``V``. Czech has essentially no native word
    beginning with Y, so an initial Y in a printed Czech name is a misread V —
    ``Yepříkov``, ``Ylčnov``, ``Yýžice``, ``Yejvanovice`` were all unmatched for
    this one reason. The substitution is offered as an *additional* candidate, never
    a replacement, so the tie discipline below still has to accept it.
    """
    cleaned = _NOISE_RE.sub(",", str(text or ""))
    parts = [_TRAILING_RE.sub("", p).strip() for p in _SEPARATOR_RE.split(cleaned)]
    parts.append(_TRAILING_RE.sub("", _SEPARATOR_RE.sub(" ", cleaned)).strip())

    forms = [p for p in parts if len(normalise(p)) >= 3]
    forms += [f"V{form[1:]}" for form in forms if form[:1] in ("Y", "y")]
    # Order-preserving dedup: the same form arrives twice whenever a cell has no
    # separator at all and the unsplit variant repeats the only part.
    return list(dict.fromkeys(forms))

class Authority:
    """Known municipalities, grouped by 1910 okres."""

    by_okres: dict[int, list[Obec]] = field(default_factory=dict)
    okres_name_to_id: dict[str, int] = field(default_factory=dict)
    #: The okres name as written, by its :func:`normalise` key — where the words
    #: begin, which the folded match needs (:meth:`match_okres_name`).
    okres_display: dict[str, str] = field(default_factory=dict)

    def okres(self, okres_id: int) -> list[Obec]:
        return self.by_okres.get(okres_id, [])

    def match_okres_name(self, name: str,
                         min_similarity: int = FOLD_MIN_SIMILARITY,
                         tie_margin: int = FOLD_TIE_MARGIN,
                         max_lead: int = FOLD_MAX_LEAD,
                         weak: bool = False) -> tuple[int | None, str, float, float]:
        """Match one printed political-okres name, OCR look-alikes folded.

        Returns ``(okres_id, how, best, runner_up)``; ``how`` is ``exact``,
        ``contains``, ``fuzzy``, or why nothing was accepted (``empty``,
        ``below``, ``tie``). The name is folded (:func:`fold_lookalikes`),
        normalised with digits dropped, then

        1. looked up exactly;
        2. searched for an okres name of four letters or more it contains,
           longest first (``vvcaslav`` holds ``caslav``); a shorter name — ``Aš`` —
           is accepted only exactly, since ``as`` sits inside any number of words;
        3. scored against every okres name of four letters or more on the shape keys
           (:func:`shape_key`), allowing up to ``max_lead`` leading characters of
           junk. A score is the lower of the whole-name ratio and the ratio of the
           name's **first word** against the same stretch of the head, so two names
           that share a long tail (``Kralupy nad Vltavou`` / ``Týn nad Vltavou``)
           cannot match on the tail alone. The best must reach ``min_similarity``
           and beat the runner-up by ``tie_margin``; a near-tie is refused, never
           broken on order (D-22).
        """
        from rapidfuzz import fuzz

        key = re.sub(r"\d", "", normalise(fold_lookalikes(name)))
        if not key:
            return None, "empty", 0.0, 0.0
        if key in self.okres_name_to_id:
            return self.okres_name_to_id[key], "exact", 100.0, 0.0
        for okres_key, okres_id in sorted(self.okres_name_to_id.items(),
                                          key=lambda kv: -len(kv[0])):
            if len(okres_key) >= 4 and okres_key in key:
                return okres_id, "contains", 100.0, 0.0

        head_shape = shape_key(key)
        scored: list[tuple[float, int]] = []
        for okres_key, okres_id in self.okres_name_to_id.items():
            if len(okres_key) < 4:
                continue
            words = self.okres_display.get(okres_key, okres_key).split()
            whole = shape_key(okres_key)
            first = shape_key(normalise(words[0])) if words else whole
            best = 0.0
            for lead in range(0, min(max_lead, max(len(head_shape) - 2, 0)) + 1):
                rest = head_shape[lead:]
                best = max(best, min(fuzz.ratio(whole, rest),
                                     fuzz.ratio(first, rest[:len(first)])))
            scored.append((best, okres_id))
        if not scored:
            return None, "below", 0.0, 0.0
        scored.sort(reverse=True)
        best_score, best_id = scored[0]
        runner_up = scored[1][0] if len(scored) > 1 else 0.0
        if best_score < min_similarity:
            if (weak and best_score >= FOLD_WEAK_SIMILARITY
                    and best_score - runner_up >= FOLD_WEAK_MARGIN):
                return best_id, "weak", best_score, runner_up
            return None, "below", best_score, runner_up
        if best_score - runner_up < tie_margin:
            return None, "tie", best_score, runner_up
        return best_id, "fuzzy", best_score, runner_up

    def resolve_okres(self, text: str, min_similarity: int = 86,
                      tie_margin: int = 4, fold: bool = False) -> int | None:
        """Find the okres named in a running head.

        The head prints ``Politický okres: Soudní okres``, so the part before the
        colon is tried first. Exact containment is tried before fuzzy matching, and
        longest-first: an okres whose name is a substring of another's ("Ledeč"
        inside "Ledeč nad Sázavou") would otherwise win on iteration order.

        Fuzzy matching is needed because the OCR mangles the head itself — ``Dééín``
        for Děčín, ``Chotěbor`` for Chotěboř — but it is held to the same tie
        discipline as everything else here: a near-tie resolves to nothing.

        ``fold=True`` (the default since 0.3.0; 0.2.0 added it opt-in) reads only the **first political name** the
        head prints (:func:`head_politicals`) and matches it with
        :meth:`match_okres_name`. It never falls back to the whole head: that
        fallback is where today's path finds an okres name inside the column
        header or the table (``vlastním`` holds ``as``, so the statutory-city pages
        and three Nová Paka pages read as Aš). ``min_similarity`` and
        ``tie_margin`` apply to the default path only.
        """
        if fold:
            names = head_politicals(text)
            return self.match_okres_name(names[0])[0] if names else None

        from rapidfuzz import fuzz

        head = str(text or "")
        candidates = [head.split(":")[0], head] if ":" in head else [head]
        for candidate in candidates:
            key = normalise(candidate)
            if not key:
                continue
            for name, okres_id in sorted(self.okres_name_to_id.items(),
                                         key=lambda kv: -len(kv[0])):
                if name and name in key:
                    return okres_id

        key = normalise(candidates[0])
        if len(key) < 4:
            return None
        scored = sorted(((fuzz.partial_ratio(name, key), okres_id)
                         for name, okres_id in self.okres_name_to_id.items()
                         if len(name) >= 4), reverse=True)
        if not scored:
            return None
        best_score, best_id = scored[0]
        runner_up = scored[1][0] if len(scored) > 1 else 0
        if best_score < min_similarity or best_score - runner_up < tie_margin:
            return None
        return best_id

    ``page_heads`` is ``(page_stem, head_text)`` in printed order. The register runs
    one okres across consecutive pages, so a page whose head is unreadable sits in
    the same district as the page before it — a structural fact about the source,
    not a guess about the text. It is still recorded as ``"carried"`` rather than
    ``"head"``, because an inherited assignment is weaker evidence than a read one
    and the ledger should say which it had.

    ``fold=True`` (the default since 0.3.0; 0.2.0 added it opt-in) matches each head with
    :meth:`Authority.match_okres_name` — look-alikes folded, shape-fuzzy, ties
    refused — and records a fuzzy read as ``"head_fuzzy"``, weaker than an exact
    one. It also carries the **last** okres the previous page names rather than its
    first: a page that prints ``Neydek: Neydek. — Nová Paka: Hořice.`` hands Nová
    Paka, not Nejdek, to an unreadable page after it. The page itself is still
    assigned the first okres it names.
    """
    if fold:
        return _assign_okresy_folded(page_heads, authority)
    assignment: dict[str, tuple[int | None, str]] = {}
    previous: int | None = None
    for stem, head in page_heads:
        okres_id = authority.resolve_okres(head)
        if okres_id is not None:
            previous = okres_id
            assignment[stem] = (okres_id, "head")
        elif previous is not None:
            assignment[stem] = (previous, "carried")
        else:
            assignment[stem] = (None, "unresolved")
    return assignment

def build_authority(final_dir: str | Path, land: str = "Böhmen") -> Authority:
    """Assemble the per-okres name authority from the project's own crosswalks."""
    import pandas as pd

    final_dir = Path(final_dir)
    settlements = pd.read_parquet(final_dir / "geo" / "settlements_1910.parquet")
    settlements = settlements[(settlements["land"] == land)
                              & (settlements["status_1910"] == "obec")]

    authority = Authority()
    index: dict[tuple[int, str], Obec] = {}
    for row in settlements.itertuples():
        if pd.isna(row.okres_id):
            continue
        okres_id = int(row.okres_id)
        obec = Obec(cs_name=str(row.cs_name), okres_id=okres_id,
                    okres_1910=str(row.okres_1910))
        for value in (row.cs_name, row.name_1910, row.de_name):
            if isinstance(value, str) and value.strip():
                obec.variants.add(value.strip())
        authority.by_okres.setdefault(okres_id, []).append(obec)
        index[(okres_id, normalise(row.cs_name))] = obec
        authority.okres_name_to_id[normalise(row.okres_1910)] = okres_id
    # One written form per okres key, the commonest: a stray variant such as
    # "Kralovi ce" must not decide where the name's first word ends.
    spellings = settlements.dropna(subset=["okres_id"])["okres_1910"].astype(str)
    for spelling in spellings.value_counts().index[::-1]:
        authority.okres_display[normalise(spelling)] = spelling

    # The ČSÚ "viz" cross-references are mostly German forms pointing at a Czech
    # target. They carry a *modern* okres, not the 1910 one, so the alias is added
    # to every 1910 obec whose Czech name it points at — an alias attaching to two
    # same-named obce in different okresy is harmless, because matching is always
    # restricted to one okres.
    alias_path = final_dir / "geo" / "settlement_aliases_1910.csv"
    if alias_path.exists():
        aliases = pd.read_csv(alias_path)
        by_target: dict[str, list[str]] = {}
        for row in aliases.itertuples():
            target = normalise(getattr(row, "viz_target", ""))
            alias = str(getattr(row, "cs_name", "") or "").strip()
            if target and alias:
                by_target.setdefault(target, []).append(alias)
        for (_okres_id, key), obec in index.items():
            for alias in by_target.get(key, []):
                obec.variants.add(alias)
    add_register_okresy(authority, REGISTER_OKRES_CURATION)
    return authority

def anchor_block(block, candidates: list[Obec], used: set[int] | None = None,
                 min_similarity: int = MIN_SIMILARITY,
                 tie_margin: int = TIE_MARGIN) -> Anchor:
    """Match one block's printed name against the obce of its okres.

    Every comma-separated form printed in the cell is tried against every known
    variant of every candidate, and the obec's best form wins. A match is accepted
    only if it clears ``min_similarity`` *and* beats the second-best obec by
    ``tie_margin``; otherwise the block is left unanchored, which is a reportable
    outcome rather than a silent wrong answer.
    """
    from rapidfuzz import fuzz

    used = used or set()
    forms = [normalise(f) for f in parse_name_cell(block.municipality.get("name"))]
    forms = [f for f in forms if len(f) >= 3]
    if not forms or not candidates:
        return Anchor(block.block_id, block.name, None, 0, 0, "unmatched")

    scored: list[tuple[int, int]] = []
    for i, obec in enumerate(candidates):
        if i in used:
            continue
        keys = obec.keys
        if not keys:
            continue
        best = max(fuzz.ratio(form, key) for form in forms for key in keys)
        scored.append((best, i))
    if not scored:
        return Anchor(block.block_id, block.name, None, 0, 0, "unmatched")

    scored.sort(reverse=True)
    best_score, best_index = scored[0]
    runner_up = scored[1][0] if len(scored) > 1 else 0
    if best_score < min_similarity:
        return Anchor(block.block_id, block.name, None, best_score, runner_up,
                      "unmatched")
    if best_score - runner_up < tie_margin:
        return Anchor(block.block_id, block.name, None, best_score, runner_up,
                      "ambiguous")
    return Anchor(block.block_id, block.name, candidates[best_index], best_score,
                  runner_up, "matched")

                  label: str = "blocks", full_okres_coverage: bool = False) -> str:
    """Coverage summary, including what the arithmetic checks cannot see.

    ``expected`` is the number of obce the covered okresy are known to contain.
    The *deficit* against it only means anything when every page of those okresy is
    in scope — on a 20-page sample most of the "missing" municipalities are simply
    on pages nobody read — so it is reported only under ``full_okres_coverage``.
    The surplus and the unmatched count are meaningful either way.
    """

    counts = Counter(a.verdict for a in anchors)
    total = len(anchors)
    if not total:

    lines = [
        f"  {label}: {total} anchored against the authority",
        f"    matched   {counts['matched']:5d} ({counts['matched'] / total:.1%})",
        f"    ambiguous {counts['ambiguous']:5d}  (best beat runner-up by "
        f"< {TIE_MARGIN} — refused, not guessed)",

        f"{MIN_SIMILARITY}; a block that matches nothing is a candidate "
        f"spurious block)",

                     f"{len(matched_obce)} distinct obce found")

"""Anchor 1910 register blocks to known municipalities, and let the volume name itself.

Matching an OCR'd Czech block name against an OCR'd German block name compares two
noisy strings and hopes. This module replaces that with two independent
comparisons against a *clean* authority: each block is matched to a known 1910
obec, and two blocks are the same municipality when they matched the same obec.
That is the ``settlements_1910`` idiom the rest of the project already uses, and it
buys three things a direct fuzzy join cannot:

* **Cross-volume pairing that survives bad OCR on both sides.** Neither block has to
  resemble the other, only the authority.
* **Over- and under-segmentation become measurable.** A block matching no obec in
  its okres is a settlement row wrongly promoted; an obec with no block is a
  municipality the parser lost. Neither is visible to any arithmetic identity.
* **A canonical identity for the output** — ``okres_id`` plus the modern Czech name
  — which the final dataset needs regardless.

The German names are the hard part. ``settlements_1910`` carries ``de_name`` for
only 18% of Bohemian obce and ``name_1910`` for 33%. Two further sources close most
of the gap:

* ``settlement_aliases_1910.csv`` — 10,028 "viz" cross-references from the ČSÚ
  historical lexicon, most of them German forms pointing at a Czech target.
* **The Czech volume itself.** It prints every municipality bilingually — "Bílek,
  Bilek", "Štěpánov, Stěpanow", "Bobkov, také Doubkov, Dobkau" — so once a Czech
  block is anchored, the German half of its own name cell becomes a variant of that
  obec. The source we are parsing is the best-covered crosswalk available for it,
  which is why anchoring runs in two passes: Czech first, harvest, then German.

FOLD_WEAK_SIMILARITY = 75
FOLD_WEAK_MARGIN = 10

#: The column header's first words. The running head runs straight into them, so a
#: colon after one belongs to the table, not to the head.
_HEADER_START_RE = re.compile(r"Po[řr][áa]d|Politick|[čc][íi]s[l1i][oap]", re.IGNORECASE)


#: Longest text accepted as one head name. The longest okres name the register
#: prints is "Dvůr Králové nad Labem" (22); the rest is room for the page number
#: and stray marks. Scan 0015 has no head of its own and its first colon is deep in
#: the table text.
_MAX_HEAD_NAME = 40

#: The statutory cities' pages print ``Města s vlastním řádem obecním: Praha.``
#: (``Städte mit eigenem Statut: Prag.``): the city is the okres, and it stands
#: AFTER the colon, where an ordinary head prints its court districts.
_STATUTORY_RE = re.compile(r"vlastn\w*\s+[řr][áa]d\w*\s+obecn|eigenem\s+Statut",
                           re.IGNORECASE)

def fold_lookalikes(text: str) -> str:
    """``text`` with each Cyrillic look-alike replaced by its Latin twin."""
    return str(text or "").translate(_LOOKALIKE_TABLE)


def shape_key(key: str) -> str:
    """A :func:`normalise` key reduced to the OCR's glyph-shape classes."""
    for pair, single in _SHAPE_PAIRS:
        key = key.replace(pair, single)
    return key.translate(_SHAPE_LETTERS)


def head_politicals(head: str) -> list[str]:
    """The political okres names a running head prints, in order.

    The head prints ``Politický okres: Soudní okres.``, and on a page where one
    okres ends and the next begins, two of them: ``Ney dek: Neydek. — Nová Рака:
    Hořice.`` Each name is the text before a colon, after the previous court list
    has ended (a full stop or a dash). Reading stops at the column header, whose
    labels follow the head in the text layer. The page number and any header words
    in front of the first name are left in; :meth:`Authority.match_okres_name`
    tolerates a short lead and the header words are cut here.
    """
    chunks = str(head or "").split(":")
    names: list[str] = []
    for index, chunk in enumerate(chunks[:-1]):
        if index > 0:
            if _HEADER_START_RE.search(chunk):
                break
            chunk = re.split(r"[.—]", chunk)[-1]
        name = _HEADER_START_RE.split(chunk)[-1].strip()
        if len(name) > _MAX_HEAD_NAME:      # the colon is in the table, not the head
            break
        if _STATUTORY_RE.search(name):
            # Scans 0016-0018: the city after the colon is the okres.
            city = re.split(r"[.,;—]", chunks[index + 1])[0].strip()
            if city:
                names.append(city)
            continue
        names.append(name)
    return names



def normalise(name: str) -> str:
    """Match key: unaccented, lowercase, letters and digits only."""
    decomposed = unicodedata.normalize("NFKD", str(name or ""))
    stripped = "".join(c for c in decomposed if not unicodedata.combining(c))
    return re.sub(r"[^a-z0-9]", "", stripped.lower())


#: The separator between the Czech and German forms is a comma in print, but the
#: OCR renders it as a full stop often enough to cost real matches — ``Rosice.
#: Rositz``, ``Skála. Skala``, ``Vápenný Podol. Kalkpodol..`` all failed to anchor
#: because the cell was read as one long unsplittable string.
_SEPARATOR_RE = re.compile(r"[,.]")


def parse_name_cell(text: str) -> list[str]:
    """Split a printed name cell into its candidate name forms.

    The register prints ``Czech, German``, sometimes with an alternative Czech form
    between them (``Bobkov, také Doubkov, Dobkau``) and usually with symbols and
    bracketed settlement numbers trailing. Every separated part is returned as a
    candidate rather than assuming which position holds which language — the volume
    is not consistent about it, and the authority match will pick.

    The **unsplit** cell is returned as a candidate too. Splitting on the full stop
    is what makes ``Rosice. Rositz`` matchable, but it would also cut a name that
    legitimately contains one (``Sv. Jan``) into fragments too short to survive the
    length filter, silently losing a name that used to match. Offering both forms
    costs one extra comparison and cannot lose.

    A leading ``Y`` is also offered as ``V``. Czech has essentially no native word
    beginning with Y, so an initial Y in a printed Czech name is a misread V —
    ``Yepříkov``, ``Ylčnov``, ``Yýžice``, ``Yejvanovice`` were all unmatched for
    this one reason. The substitution is offered as an *additional* candidate, never
    a replacement, so the tie discipline below still has to accept it.
    """
    cleaned = _NOISE_RE.sub(",", str(text or ""))
    parts = [_TRAILING_RE.sub("", p).strip() for p in _SEPARATOR_RE.split(cleaned)]
    parts.append(_TRAILING_RE.sub("", _SEPARATOR_RE.sub(" ", cleaned)).strip())

    forms = [p for p in parts if len(normalise(p)) >= 3]
    forms += [f"V{form[1:]}" for form in forms if form[:1] in ("Y", "y")]
    # Order-preserving dedup: the same form arrives twice whenever a cell has no
    # separator at all and the unsplit variant repeats the only part.
    return list(dict.fromkeys(forms))

@dataclass
class Obec:
    """One known 1910 municipality and every name form it is known under."""

    cs_name: str
    okres_id: int
    okres_1910: str
    variants: set[str] = field(default_factory=set)
    printed_names: set[str] = field(default_factory=set)   # harvested from the volume

    @property
    def keys(self) -> set[str]:
        return {normalise(v) for v in self.variants | self.printed_names
                if len(normalise(v)) >= 3}


@dataclass
class Authority:
    """Known municipalities, grouped by 1910 okres."""

    by_okres: dict[int, list[Obec]] = field(default_factory=dict)
    okres_name_to_id: dict[str, int] = field(default_factory=dict)
    #: The okres name as written, by its :func:`normalise` key — where the words
    #: begin, which the folded match needs (:meth:`match_okres_name`).
    okres_display: dict[str, str] = field(default_factory=dict)

    def okres(self, okres_id: int) -> list[Obec]:
        return self.by_okres.get(okres_id, [])

    def match_okres_name(self, name: str,
                         min_similarity: int = FOLD_MIN_SIMILARITY,
                         tie_margin: int = FOLD_TIE_MARGIN,
                         max_lead: int = FOLD_MAX_LEAD,
                         weak: bool = False) -> tuple[int | None, str, float, float]:
        """Match one printed political-okres name, OCR look-alikes folded.

        Returns ``(okres_id, how, best, runner_up)``; ``how`` is ``exact``,
        ``contains``, ``fuzzy``, or why nothing was accepted (``empty``,
        ``below``, ``tie``). The name is folded (:func:`fold_lookalikes`),
        normalised with digits dropped, then

        1. looked up exactly;
        2. searched for an okres name of four letters or more it contains,
           longest first (``vvcaslav`` holds ``caslav``); a shorter name — ``Aš`` —
           is accepted only exactly, since ``as`` sits inside any number of words;
        3. scored against every okres name of four letters or more on the shape keys
           (:func:`shape_key`), allowing up to ``max_lead`` leading characters of
           junk. A score is the lower of the whole-name ratio and the ratio of the
           name's **first word** against the same stretch of the head, so two names
           that share a long tail (``Kralupy nad Vltavou`` / ``Týn nad Vltavou``)
           cannot match on the tail alone. The best must reach ``min_similarity``
           and beat the runner-up by ``tie_margin``; a near-tie is refused, never
           broken on order (D-22).
        """
        from rapidfuzz import fuzz

        key = re.sub(r"\d", "", normalise(fold_lookalikes(name)))
        if not key:
            return None, "empty", 0.0, 0.0
        if key in self.okres_name_to_id:
            return self.okres_name_to_id[key], "exact", 100.0, 0.0
        for okres_key, okres_id in sorted(self.okres_name_to_id.items(),
                                          key=lambda kv: -len(kv[0])):
            if len(okres_key) >= 4 and okres_key in key:
                return okres_id, "contains", 100.0, 0.0

        head_shape = shape_key(key)
        scored: list[tuple[float, int]] = []
        for okres_key, okres_id in self.okres_name_to_id.items():
            if len(okres_key) < 4:
                continue
            words = self.okres_display.get(okres_key, okres_key).split()
            whole = shape_key(okres_key)
            first = shape_key(normalise(words[0])) if words else whole
            best = 0.0
            for lead in range(0, min(max_lead, max(len(head_shape) - 2, 0)) + 1):
                rest = head_shape[lead:]
                best = max(best, min(fuzz.ratio(whole, rest),
                                     fuzz.ratio(first, rest[:len(first)])))
            scored.append((best, okres_id))
        if not scored:
            return None, "below", 0.0, 0.0
        scored.sort(reverse=True)
        best_score, best_id = scored[0]
        runner_up = scored[1][0] if len(scored) > 1 else 0.0
        if best_score < min_similarity:
            if (weak and best_score >= FOLD_WEAK_SIMILARITY
                    and best_score - runner_up >= FOLD_WEAK_MARGIN):
                return best_id, "weak", best_score, runner_up
            return None, "below", best_score, runner_up
        if best_score - runner_up < tie_margin:
            return None, "tie", best_score, runner_up
        return best_id, "fuzzy", best_score, runner_up

    def resolve_okres(self, text: str, min_similarity: int = 86,
                      tie_margin: int = 4, fold: bool = False) -> int | None:
        """Find the okres named in a running head.

        The head prints ``Politický okres: Soudní okres``, so the part before the
        colon is tried first. Exact containment is tried before fuzzy matching, and
        longest-first: an okres whose name is a substring of another's ("Ledeč"
        inside "Ledeč nad Sázavou") would otherwise win on iteration order.

        Fuzzy matching is needed because the OCR mangles the head itself — ``Dééín``
        for Děčín, ``Chotěbor`` for Chotěboř — but it is held to the same tie
        discipline as everything else here: a near-tie resolves to nothing.

        ``fold=True`` (the default since 0.3.0; 0.2.0 added it opt-in) reads only the **first political name** the
        head prints (:func:`head_politicals`) and matches it with
        :meth:`match_okres_name`. It never falls back to the whole head: that
        fallback is where today's path finds an okres name inside the column
        header or the table (``vlastním`` holds ``as``, so the statutory-city pages
        and three Nová Paka pages read as Aš). ``min_similarity`` and
        ``tie_margin`` apply to the default path only.
        """
        if fold:
            names = head_politicals(text)
            return self.match_okres_name(names[0])[0] if names else None

        from rapidfuzz import fuzz

        head = str(text or "")
        candidates = [head.split(":")[0], head] if ":" in head else [head]
        for candidate in candidates:
            key = normalise(candidate)
            if not key:
                continue
            for name, okres_id in sorted(self.okres_name_to_id.items(),
                                         key=lambda kv: -len(kv[0])):
                if name and name in key:
                    return okres_id

        key = normalise(candidates[0])
        if len(key) < 4:
            return None
        scored = sorted(((fuzz.partial_ratio(name, key), okres_id)
                         for name, okres_id in self.okres_name_to_id.items()
                         if len(name) >= 4), reverse=True)
        if not scored:
            return None
        best_score, best_id = scored[0]
        runner_up = scored[1][0] if len(scored) > 1 else 0
        if best_score < min_similarity or best_score - runner_up < tie_margin:
            return None
        return best_id


def assign_okresy(page_heads: list[tuple[str, str]],
                  authority: "Authority",
                  fold: bool = True) -> dict[str, tuple[int | None, str]]:
    """Assign each page an okres, carrying the previous one forward over gaps.

    ``page_heads`` is ``(page_stem, head_text)`` in printed order. The register runs
    one okres across consecutive pages, so a page whose head is unreadable sits in
    the same district as the page before it — a structural fact about the source,
    not a guess about the text. It is still recorded as ``"carried"`` rather than
    ``"head"``, because an inherited assignment is weaker evidence than a read one
    and the ledger should say which it had.

    ``fold=True`` (the default since 0.3.0; 0.2.0 added it opt-in) matches each head with
    :meth:`Authority.match_okres_name` — look-alikes folded, shape-fuzzy, ties
    refused — and records a fuzzy read as ``"head_fuzzy"``, weaker than an exact
    one. It also carries the **last** okres the previous page names rather than its
    first: a page that prints ``Neydek: Neydek. — Nová Paka: Hořice.`` hands Nová
    Paka, not Nejdek, to an unreadable page after it. The page itself is still
    assigned the first okres it names.
    """
    if fold:
        return _assign_okresy_folded(page_heads, authority)
    assignment: dict[str, tuple[int | None, str]] = {}
    previous: int | None = None
    for stem, head in page_heads:
        okres_id = authority.resolve_okres(head)
        if okres_id is not None:
            previous = okres_id
            assignment[stem] = (okres_id, "head")
        elif previous is not None:
            assignment[stem] = (previous, "carried")
        else:
            assignment[stem] = (None, "unresolved")
    return assignment


def _assign_okresy_folded(page_heads: list[tuple[str, str]],
                          authority: "Authority") -> dict[str, tuple[int | None, str]]:
    """:func:`assign_okresy` with ``fold=True``."""
    assignment: dict[str, tuple[int | None, str]] = {}
    previous: int | None = None
    for stem, head in page_heads:
        names = head_politicals(head)
        matches = [authority.match_okres_name(name, weak=(i == 0))
                   for i, name in enumerate(names)]
        first_id, how = (matches[0][0], matches[0][1]) if matches else (None, "")
        if first_id is not None:
            assignment[stem] = (first_id, {"fuzzy": "head_fuzzy",
                                           "weak": "head_weak"}.get(how, "head"))
        elif names and any(c.isalpha() for c in names[0]):
            # The head NAMES an okres and it matches none: say so. Carrying the
            # previous page's okres here was a confident wrong answer on every
            # such page but two (2026-10-14: Loket and Kralupy nad Vltavou are not
            # in the authority at all, and became Litomyšl and Královské
            # Vinohrady). A head with no name in it still carries, below.
            assignment[stem] = (None, "named_unmatched")
        elif previous is not None:
            assignment[stem] = (previous, "carried")
        else:
            assignment[stem] = (None, "unresolved")
        named = [okres_id for okres_id, *_ in matches if okres_id is not None]
        if named:
            previous = named[-1]
    return assignment


def build_authority(final_dir: str | Path, land: str = "Böhmen") -> Authority:
    """Assemble the per-okres name authority from the project's own crosswalks."""
    import pandas as pd

    final_dir = Path(final_dir)
    settlements = pd.read_parquet(final_dir / "geo" / "settlements_1910.parquet")
    settlements = settlements[(settlements["land"] == land)
                              & (settlements["status_1910"] == "obec")]

    authority = Authority()
    index: dict[tuple[int, str], Obec] = {}
    for row in settlements.itertuples():
        if pd.isna(row.okres_id):
            continue
        okres_id = int(row.okres_id)
        obec = Obec(cs_name=str(row.cs_name), okres_id=okres_id,
                    okres_1910=str(row.okres_1910))
        for value in (row.cs_name, row.name_1910, row.de_name):
            if isinstance(value, str) and value.strip():
                obec.variants.add(value.strip())
        authority.by_okres.setdefault(okres_id, []).append(obec)
        index[(okres_id, normalise(row.cs_name))] = obec
        authority.okres_name_to_id[normalise(row.okres_1910)] = okres_id
    # One written form per okres key, the commonest: a stray variant such as
    # "Kralovi ce" must not decide where the name's first word ends.
    spellings = settlements.dropna(subset=["okres_id"])["okres_1910"].astype(str)
    for spelling in spellings.value_counts().index[::-1]:
        authority.okres_display[normalise(spelling)] = spelling

    # The ČSÚ "viz" cross-references are mostly German forms pointing at a Czech
    # target. They carry a *modern* okres, not the 1910 one, so the alias is added
    # to every 1910 obec whose Czech name it points at — an alias attaching to two
    # same-named obce in different okresy is harmless, because matching is always
    # restricted to one okres.
    alias_path = final_dir / "geo" / "settlement_aliases_1910.csv"
    if alias_path.exists():
        aliases = pd.read_csv(alias_path)
        by_target: dict[str, list[str]] = {}
        for row in aliases.itertuples():
            target = normalise(getattr(row, "viz_target", ""))
            alias = str(getattr(row, "cs_name", "") or "").strip()
            if target and alias:
                by_target.setdefault(target, []).append(alias)
        for (_okres_id, key), obec in index.items():
            for alias in by_target.get(key, []):
                obec.variants.add(alias)
    add_register_okresy(authority, REGISTER_OKRES_CURATION)
    return authority


#: Political okresy the 1910 register prints that the district backbone does not
#: carry (Loket, Kralupy nad Vltavou). The register is authoritative for the name
#: (author, 2026-10-15); each is mapped to the backbone okres that contains its
#: courts, so anchoring to obce and the census check pool it with that okres.
REGISTER_OKRES_CURATION = (Path(__file__).resolve().parents[2] / "reference"
                           / "places" / "register_okres_curation.csv")


def add_register_okresy(authority: "Authority", path: Path) -> int:
    """Make each register okres name resolve to its backbone okres. Returns how many."""
    import csv

    if not path.exists():
        return 0
    added = 0
    with path.open(encoding="utf-8", newline="") as fh:
        for row in csv.DictReader(fh):
            # `authority_okres` is the backbone okres as the authority spells it
            # (the obce's okres_1910): "Falknov", where the backbone prints
            # "Falknov nad Ohří".
            parent = authority.okres_name_to_id.get(normalise(row["authority_okres"]))
            if parent is None:
                raise ValueError(f"{path.name}: {row['authority_okres']!r} is not an "
                                 f"okres of the authority")
            key = normalise(row["register_okres"])
            authority.okres_name_to_id[key] = parent
            authority.okres_display[key] = row["register_okres"]
            added += 1
    return added

@dataclass
class Anchor:
    """The result of matching one block to the authority."""

    block_id: str
    printed: str
    obec: Obec | None
    score: int
    runner_up: int
    verdict: str        # "matched" | "ambiguous" | "unmatched"


def anchor_block(block, candidates: list[Obec], used: set[int] | None = None,
                 min_similarity: int = MIN_SIMILARITY,
                 tie_margin: int = TIE_MARGIN) -> Anchor:
    """Match one block's printed name against the obce of its okres.

    Every comma-separated form printed in the cell is tried against every known
    variant of every candidate, and the obec's best form wins. A match is accepted
    only if it clears ``min_similarity`` *and* beats the second-best obec by
    ``tie_margin``; otherwise the block is left unanchored, which is a reportable
    outcome rather than a silent wrong answer.
    """
    from rapidfuzz import fuzz

    used = used or set()
    forms = [normalise(f) for f in parse_name_cell(block.municipality.get("name"))]
    forms = [f for f in forms if len(f) >= 3]
    if not forms or not candidates:
        return Anchor(block.block_id, block.name, None, 0, 0, "unmatched")

    scored: list[tuple[int, int]] = []
    for i, obec in enumerate(candidates):
        if i in used:
            continue
        keys = obec.keys
        if not keys:
            continue
        best = max(fuzz.ratio(form, key) for form in forms for key in keys)
        scored.append((best, i))
    if not scored:
        return Anchor(block.block_id, block.name, None, 0, 0, "unmatched")

    scored.sort(reverse=True)
    best_score, best_index = scored[0]
    runner_up = scored[1][0] if len(scored) > 1 else 0
    if best_score < min_similarity:
        return Anchor(block.block_id, block.name, None, best_score, runner_up,
                      "unmatched")
    if best_score - runner_up < tie_margin:
        return Anchor(block.block_id, block.name, None, best_score, runner_up,
                      "ambiguous")
    return Anchor(block.block_id, block.name, candidates[best_index], best_score,
                  runner_up, "matched")


def anchor_blocks(blocks, candidates: list[Obec], one_to_one: bool = True,
                  **kwargs) -> list[Anchor]:
    """Anchor a whole okres's blocks, best-scoring first.

    Anchoring in descending score order and consuming each obec once means a
    confident match is never displaced by a marginal one that happened to be
    processed first. ``one_to_one`` can be turned off for the German volume when a
    municipality legitimately appears on two pages.
    """
    provisional = [(anchor_block(b, candidates, **kwargs), b) for b in blocks]
    provisional.sort(key=lambda pair: -pair[0].score)

    anchors: list[Anchor] = []
    used: set[int] = set()
    position = {id(o): i for i, o in enumerate(candidates)}
    for anchor, block in provisional:
        if anchor.verdict != "matched" or not one_to_one:
            anchors.append(anchor)
            continue
        index = position[id(anchor.obec)]
        if index in used:
            anchors.append(anchor_block(block, candidates, used=used, **kwargs))
        else:
            used.add(index)
            anchors.append(anchor)
    return anchors


def harvest_printed_names(anchors: list[Anchor], blocks) -> int:
    """Feed the volume's own bilingual name cells back into the authority.

    The Czech register prints each municipality under both names, so an anchored
    Czech block donates every form in its cell — including the German one — to that
    obec. This is what lifts German coverage above the 18% ``de_name`` carries, and
    it is why anchoring the Czech volume must happen before the German one.
    """
    by_id = {b.block_id: b for b in blocks}
    added = 0
    for anchor in anchors:
        if anchor.verdict != "matched" or anchor.obec is None:
            continue
        block = by_id.get(anchor.block_id)
        if block is None:
            continue
        for form in parse_name_cell(block.municipality.get("name")):
            key = normalise(form)
            if key and key not in {normalise(v) for v in anchor.obec.variants}:
                anchor.obec.printed_names.add(form)
                added += 1
    return added


def pair_by_obec(czech_anchors: list[Anchor], german_anchors: list[Anchor]) -> list[tuple]:
    """Pair Czech and German blocks that anchored to the same obec."""
    german = {id(a.obec): a for a in german_anchors if a.verdict == "matched"}
    return [(a.obec, a, german[id(a.obec)])
            for a in czech_anchors
            if a.verdict == "matched" and id(a.obec) in german]


def anchor_report(anchors: list[Anchor], expected: int | None = None,
                  label: str = "blocks", full_okres_coverage: bool = False) -> str:
    """Coverage summary, including what the arithmetic checks cannot see.

    ``expected`` is the number of obce the covered okresy are known to contain.
    The *deficit* against it only means anything when every page of those okresy is
    in scope — on a 20-page sample most of the "missing" municipalities are simply
    on pages nobody read — so it is reported only under ``full_okres_coverage``.
    The surplus and the unmatched count are meaningful either way.
    """
    from collections import Counter

    counts = Counter(a.verdict for a in anchors)
    total = len(anchors)
    if not total:
        return f"  {label}: nothing to anchor"
    lines = [
        f"  {label}: {total} anchored against the authority",
        f"    matched   {counts['matched']:5d} ({counts['matched'] / total:.1%})",
        f"    ambiguous {counts['ambiguous']:5d}  (best beat runner-up by "
        f"< {TIE_MARGIN} — refused, not guessed)",
        f"    unmatched {counts['unmatched']:5d}  (no obec in this okres above "
        f"{MIN_SIMILARITY}; a block that matches nothing is a candidate "
        f"spurious block)",
    ]
    if expected is not None:
        matched_obce = {id(a.obec) for a in anchors if a.verdict == "matched"}
        lines.append(f"    known obce in the covered okresy: {expected}; "
                     f"{len(matched_obce)} distinct obce found")
        if total > expected:
            lines.append(f"    surplus blocks over known obce: {total - expected} "
                         f"(over-segmentation)")
        if full_okres_coverage:
            lines.append(f"    obce with no block: {expected - len(matched_obce)} "
                         f"(under-segmentation)")
        else:
            lines.append("    deficit not reported — these pages do not cover the "
                         "whole of every okres")
    return "\n".join(lines)
