How it works

E3's first half (locate_glyphs) puts each claimed mark on a cluster of ink. It is trained, with no hand label, on the Czech gold; on the German gold run 28 % of the reader of record's claims are never placed (the Czech: 8 %), and DINOv3 can only judge a claim that was placed. This is the first step toward a German bank: train the same locator on the German gold sections, each page placed by a locator that never saw it (locate_all's leave-page-out), and count how many true gold claims it places, beside the Czech-trained locator on the same sections.

What it can say: the share of gold claims placed. What it cannot: whether a placed claim sits on the right glyph (no German glyph was ever clicked); that is DINOv3's agreement with the claimed token, measured on Heavyhorse6. A placement share that rises while the agreement does not is a locator that forces a box.

Read-only against output/frozen/de-bohemia_gold/sections and the German gold sheets (calibration and purposive, never certification); writes nothing. It lives outside utils/lexikon_1910 so as not to change the route's code digests.

def german_sections(sections_dir: Path = SECTIONS):
    """One ``locate_glyphs.Section`` per (page, court, kind) of the German gold run, its images
    from the run's saved crops and its claims from the gold sheet, in printed order."""
    import locate_glyphs as lg
    from build_token_review import _sheet_sections
    from read_transcription_md import _key
    from utils.lexikon_1910 import lexikon_1910_notes as lxnt

    with (sections_dir / "index.csv").open(encoding="utf-8", newline="") as fh:
        index = list(csv.DictReader(fh))
    images = {}
    for row in index:
        images.setdefault((row["page"], row["district"], row["kind"]), []).append(row)
    out, unmatched = [], []
    for role in ROLES:
        for path in sorted((GOLD_DE / role).glob("[0-9][0-9][0-9][0-9].md")):
            text = path.read_text(encoding="utf-8")
            headings = [line for line in text.splitlines() if re.match(r"^##\s*2[ab]\s*—", line)]
            for n, block in enumerate(_sheet_sections(text)):
                found = COURT_RE.search(headings[n]) if n < len(headings) else None
                district = lxnt.normalise(found.group(1)) if found else ""
                kind = "parts" if block["section"] == "2a" else "offices"
                rows = images.get((path.stem, district, kind))
                if not rows:
                    unmatched.append((path.stem, district, kind))
                    continue
                section = lg.Section(scan=path.stem, role=role, kind=kind,
                                     images=[r["image"] for r in sorted(rows, key=lambda r: int(r["order"]))],
                                     sheet_dir=sections_dir, district=district)
                for _lineno, line in block["lines"]:
                    parsed = _key(line)
                    if not parsed or parsed[0] == "__" or not parsed[-1]:
                        continue
                    muni, seq, *_rest, answer = parsed
                    if int(muni) not in section.printed:
                        section.printed.append(int(muni))
                    section.keys.append((int(muni), str(seq)))
                    for token in lg.TOKEN_RE.findall(answer):
                        if token not in lg.NOT_A_GLYPH:
                            section.claims.append(lg.Claim(token, int(muni), seq, len(section.claims)))
                out.append(section)
    return out, unmatched

def safe_plate_templates(symbols: Path) -> dict:
    """``locate_glyphs.plate_templates`` one glyph at a time: a glyph file with no usable ink makes
    the whole call fail (the German plate has some), so each manifest row gets a call of its own."""
    import tempfile

    import locate_glyphs as lg

    with (symbols / "manifest.csv").open(encoding="utf-8", newline="") as fh:
        rows = list(csv.DictReader(fh))
    out, failed = {}, []
    with tempfile.TemporaryDirectory() as tmp:
        for row in rows:
            one = Path(tmp)
            with (one / "manifest.csv").open("w", encoding="utf-8", newline="") as fh:
                writer = csv.DictWriter(fh, fieldnames=list(row))
                writer.writeheader()
                writer.writerow({**row, "file": str(symbols / (row.get("file") or ""))})
            try:
                out.update(lg.plate_templates(one))
            except ValueError:
                failed.append(row["token"])
    print(f"plate seeds from {symbols.name}: {len(out)} tokens; no usable ink in {failed or 'none'}")
    return out

def placed_share(placed: dict) -> dict:
    """``{kind: (placed, claims)}`` over ``{key: (section, cands, placed)}``."""
    count = Counter()
    for section, _cands, where in placed.values():
        count[section.kind, "claims"] += len(section.claims)
        count[section.kind, "placed"] += sum(1 for i in where if i is not None)
    return {kind: (count[kind, "placed"], count[kind, "claims"]) for kind in ("offices", "parts")}

def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--rounds", type=int, default=4)
    parser.add_argument("--symbols", type=Path, default=ROOT / "reference" / "places" / "symbols" / "de",
                        help="the German plate's glyph folder, the locator's first templates")
    args = parser.parse_args(argv)

    import locate_glyphs as lg

    sections, unmatched = german_sections()
    print(f"German gold sections: {len(sections)} ({sum(len(s.claims) for s in sections)} claims, "
          f"pages {sorted({s.scan for s in sections})}); unmatched sheet blocks: {unmatched or 'none'}")
    data = [(s, lg.section_candidates(s)) for s in sections]
    print(f"clusters found: {sum(len(c) for _s, c in data)}")

    print("\nCzech-trained locator (every Czech calibration page) on the German sections:")
    czech = lg.load_sections()
    czech_locator = lg.train_locator(czech, {s.scan for s, _ in czech if s.role == "calibration"},
                                     args.rounds)
    czech_placed = {}
    for section, cands in data:
        scores = czech_locator.glyph.score(cands) if cands else []
        for c, v in zip(cands, scores):
            c.glyphness = float(v)
        czech_placed[section.key] = (section, cands, czech_locator.place(cands, section.claims))
    for kind, (p, n) in placed_share(czech_placed).items():
        print(f"  {kind:8s} placed {p} of {n}  ({p / n:.3f})" if n else f"  {kind}: none")

    calibration = sorted({s.scan for s, _ in data if s.role == "calibration"})
    for label, seeds in (("German plate seeds", safe_plate_templates(args.symbols)),
                         ("Czech plate seeds", lg.plate_templates())):
        print(f"\nGerman-trained locator, each page placed by the others ({label}; "
              f"{len(calibration)} calibration pages):")
        trained, placed = {}, {}
        for section, cands in data:
            key = tuple(sorted(set(calibration) - {section.scan}))
            if key not in trained:
                trained[key] = lg.train_locator(data, set(key), args.rounds, seeds)
            locator = trained[key]
            scores = locator.glyph.score(cands) if cands else []
            for c, v in zip(cands, scores):
                c.glyphness = float(v)
            placed[section.key] = (section, cands, locator.place(cands, section.claims))
        for kind, (p, n) in placed_share(placed).items():
            print(f"  {kind:8s} placed {p} of {n}  ({p / n:.3f})" if n else f"  {kind}: none")
    return 0
