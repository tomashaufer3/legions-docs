How it works

The Token Review page asks for every fill-pair glyph an entry prints to be clicked (build_token_review.py --boxes), on the entry's crop or on the whole section. This script takes those clicks back to the section image they fell on — through the crop pieces saved with the decision — and cuts the glyph there, two ways:

tight the ink around the click — the connected components that touch its centre, joined with any close enough to be the same mark — padded a little. What an encoder would see from a good box detector. window a square of fixed size centred on the click, 1.6 line heights wide. What it would see from a rough one.

Every glyph is written at the gold crops' own scale (about twice native), from calibration and purposive sheets only. contact_&lt;token&gt;.png puts each token's cuts side by side, so a bad cut is seen before the test reads it.

def tight_box(ink, x: int, y: int, lh: int) -> tuple[int, int, int, int] | None:
    """The bounding box of the mark under the click, or ``None`` if no ink is near."""
    import numpy as np
    from scipy import ndimage

    half = int(WINDOW_LINES * lh)
    x0, y0 = max(0, x - half), max(0, y - half)
    sub = ink[y0:y + half, x0:x + half]
    labels, n = ndimage.label(sub, structure=np.ones((3, 3)))
    if not n:
        return None
    cx, cy = x - x0, y - y0
    r = max(3, int(SEED_LINES * lh))
    disc = labels[max(0, cy - r):cy + r + 1, max(0, cx - r):cx + r + 1]
    keep = set(np.unique(disc[disc > 0]).tolist())
    boxes = {i + 1: s for i, s in enumerate(ndimage.find_objects(labels))}
    if not keep:
        ys, xs = np.nonzero(labels)
        nearest = int(np.argmin((ys - cy) ** 2 + (xs - cx) ** 2))
        keep = {int(labels[ys[nearest], xs[nearest]])}

    def union(ids):
        top = min(boxes[i][0].start for i in ids)
        bottom = max(boxes[i][0].stop for i in ids)
        left = min(boxes[i][1].start for i in ids)
        right = max(boxes[i][1].stop for i in ids)
        return top, bottom, left, right

    gap = max(2, int(JOIN_LINES * lh))
    grew = True
    while grew:
        grew = False
        top, bottom, left, right = union(keep)
        for i, (ys, xs) in boxes.items():
            if i in keep:
                continue
            near = (ys.start <= bottom + gap and ys.stop >= top - gap
                    and xs.start <= right + gap and xs.stop >= left - gap)
            if not near:
                continue
            t, b, l, rr = union(keep | {i})
            if b - t <= MAX_TIGHT[0] * lh and rr - l <= MAX_TIGHT[1] * lh:
                keep.add(i)
                grew = True
    top, bottom, left, right = union(keep)
    return (x0 + left - PAD, y0 + top - PAD, x0 + right + PAD, y0 + bottom + PAD)

def cut(decisions: list[dict], queue: dict) -> list[dict]:
    import numpy as np
    from PIL import Image

    items = {i["id"]: i for i in queue["items"]}
    cache: dict[str, tuple] = {}
    rows = []
    for d in decisions:
        item = items.get(d["id"])
        if not item or item["kind"] != "box" or d.get("decision") != "boxed":
            continue
        sheet_dir = (ROOT / item["sheet"]).parent
        # The crop the reviewer actually clicked on: saved with the decision,
        # since a later build may cut the entry differently.
        clicked_on = {**item, "pieces": d.get("pieces") or item["pieces"]}
        for k, click in enumerate(d.get("clicks", [])):
            if len(click) == 3:                     # on the whole section image
                x, y, name = int(click[0]), int(click[1]), click[2]
            else:
                name, x, y = to_source(clicked_on, *click)
            if name not in cache:
                image = Image.open(sheet_dir / name).convert("L")
                lh = line_height(image) or scan_line_height(sheet_dir, item["scan"])
                cache[name] = (image, np.asarray(image) < 128, lh)
            image, ink, lh = cache[name]
            half = int(WINDOW_LINES * lh / 2)
            window = (x - half, y - half, x + half, y + half)
            box = tight_box(ink, x, y, lh) or window
            token = item["token"].strip("[]")
            glyph_id = f"{item['id'].replace('-box-' + token, '')}-{token}-{k + 1}"
            paths = {}
            for variant, bbox in (("tight", box), ("window", window)):
                out = BANK / variant / token / f"{glyph_id}.png"
                out.parent.mkdir(parents=True, exist_ok=True)
                image.crop(bbox).save(out)
                paths[variant] = str(out.relative_to(BANK)).replace("\\", "/")
            rows.append({"glyph_id": glyph_id, "token": token, "scan": item["scan"],
                         "role": item["source"].split(":")[-1], "section": item["section"],
                         "muni": item["muni"], "image": name, "x": x, "y": y,
                         "line_height": lh, "tight": paths["tight"],
                         "window": paths["window"], "tight_box": ",".join(map(str, box)),
                         "expected": item["expected"], "clicked": len(d["clicks"])})
    return rows

def contact_sheets(rows: list[dict]) -> None:
    from PIL import Image, ImageDraw

    by: dict[str, list] = {}
    for r in rows:
        by.setdefault(r["token"], []).append(r)
    for token, group in by.items():
        cells = [Image.open(BANK / r["tight"]) for r in group]
        side = 120
        cols = 10
        sheet = Image.new("L", (cols * side, -(-len(cells) // cols) * (side + 14)), 255)
        draw = ImageDraw.Draw(sheet)
        for n, (cell, r) in enumerate(zip(cells, group)):
            cell.thumbnail((side - 6, side - 6))
            x, y = (n % cols) * side, (n // cols) * (side + 14)
            sheet.paste(cell, (x + 3, y + 3))
            draw.text((x + 3, y + side), r["scan"], fill=0)
        sheet.save(BANK / f"contact_{token}.png")

"""Cut the boxed glyphs out of the gold crops: the bank test A reads.

The *Token Review* page asks for every fill-pair glyph an entry prints to be
clicked (``build_token_review.py --boxes``), on the entry's crop or on the whole
section. This script takes those clicks back to the section image they fell on
— through the crop pieces saved with the decision — and cuts the glyph there,
two ways:

``tight``   the ink around the click — the connected components that touch its
            centre, joined with any close enough to be the same mark — padded a
            little. What an encoder would see from a good box detector.
``window``  a square of fixed size centred on the click, 1.6 line heights wide.
            What it would see from a rough one.

Every glyph is written at the gold crops' own scale (about twice native), from
calibration and purposive sheets only. ``contact_<token>.png`` puts each
token's cuts side by side, so a bad cut is seen before the test reads it.

    python utils/lexikon_1910/build_glyph_bank.py decisions.json

from __future__ import annotations

import argparse
import csv
import json
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]
sys.path.insert(0, str(ROOT))
sys.path.insert(0, str(HERE))

from build_token_review import OUT_DIR as REVIEW_DIR, text_lines    # noqa: E402

BANK = ROOT / "output" / "glyph_bank"
#: The fixed window's side, in line heights.
WINDOW_LINES = 1.6
#: Ink within this share of a line height of the click seeds the tight cut.
SEED_LINES = 0.3
#: Components this close to the cut (in line heights) join it.
JOIN_LINES = 0.08
#: A tight cut never grows past this, in line heights (height, width).
MAX_TIGHT = (1.3, 2.5)
PAD = 3


#: A printed line at the gold crops' scale is ~65-70 px; outside this range the
#: estimate has merged lines (a short image of a few touching lines) or caught specks.
PLAUSIBLE_LINE = (35, 110)


def line_height(image) -> int | None:
    heights = sorted(b - t for t, b, _x in text_lines(image))
    lh = heights[len(heights) // 2] if heights else None
    return lh if lh and PLAUSIBLE_LINE[0] <= lh <= PLAUSIBLE_LINE[1] else None
