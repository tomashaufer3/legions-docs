How it works

The route's German decoder opens an entry only at a key written 35, —) or 2, 1). Where a reader wrote the key with a full stop — 35. —) — the line opened no entry. At the head of a section the whole section was dropped: both readers did so over Nové Město na Moravě 0104, nos. 35–48, and the register holds nothing for 15 municipalities. After an entry the line was filed under that entry: Moravská Ostrava 10 holds the claims of nos. 11–14. Found by the blind sample's no-pictogram control (D-199), 13 of whose 26 municipalities print a pictogram.

Nothing is read again. Each reader's stored lines (notes_&lt;label&gt;.csv, as the route read them) are decoded twice by the route's decoder, with §3's heading splits (offices_resplit): as the route decoded them, and with every full-stop key mended (offices_resplit.mend_key). The entries the mending changes for any reader are settled across the run's four readers as §3 and §4 settle claims:

  • adds — on a changed entry every reader's mended decode reaches, a claim all four make is agreed; one only some make is disputed, asked of Opus as a token question on the entry (stage STAGE, with the route's part-0 rule, D-197); on a changed entry only some readers reach, the reader of record's claim stands alone;
  • removes — a claim the route holds on a changed entry that a reader made there before the mending and none makes after it: it belonged to the entry whose key was read with a full stop.

German volumes only: the Czech readers write the comma (3 full stops in 23,446 lines).

def settle(base: dict, mended: dict, record: str, route: set) -> tuple[list[dict], list[dict]]:
    """``base`` and ``mended`` — ``{label: {(page, district, muni, part, code)}}`` — settled on the
    entries the mending changes for any reader. Returns ``(adds, removes)``: the mended claims with
    their status and whether the route holds them; the route's claims no mended decode makes."""
    touched = set().union(*({c[:4] for c in base[label] ^ mended[label]} for label in base))
    every = touched & set.intersection(*({c[:4] for c in cs} for cs in mended.values()))
    shared = {label: {c for c in cs if c[:4] in every} for label, cs in mended.items()}
    union, agreed = set().union(*shared.values()), set.intersection(*shared.values())
    adds = [dict(zip(CLAIM, c), status="agreed" if c in agreed else "disputed",
                 support=sum(c in s for s in shared.values()), in_route=c in route) for c in sorted(union)]
    alone = {c for c in mended[record] if c[:4] in touched - every}
    adds += [dict(zip(CLAIM, c), status="reader of record", support=1, in_route=c in route)
             for c in sorted(alone)]
    made, was = set().union(*mended.values()), set().union(*base.values())
    removes = [dict(zip(CLAIM, c)) for c in sorted(route) if c[:4] in touched and c in was and c not in made]
    return adds, removes

def run(edition: str, pages: str = PAGES, out: Path = OUT) -> dict:
    """Every reader of one German volume; writes under ``<out>/<edition>/keys/``: each reader's
    mended inventory (``inventory_<label>.csv``, whose context carries the cooperatives' language
    letter for the merge), ``consensus.csv`` (the adds), ``removes.csv``, ``mended_lines.csv`` and
    ``summary.json``."""
    if edition not in GERMAN:
        raise ValueError(f"{edition}: the Czech readers write the key with a comma; nothing to mend")
    folder = Path(out) / edition / SUB
    folder.mkdir(parents=True, exist_ok=True)
    labels = rs.readers(edition, pages)
    base, mended, lines = {}, {}, []
    for label in labels:
        b, m, mended_lines = reader_decodes(edition, label)
        base[label] = {rs._key(r) for r in b if r.get("code")}
        mended[label] = {rs._key(r) for r in m if r.get("code")}
        rs._write(folder / f"inventory_{label}.csv", m)
        lines += [dict(x, reader=label) for x in mended_lines]
    adds, removes = settle(base, mended, labels[0], route_claims(edition, pages))
    rs._write(folder / "consensus.csv", adds)
    rs._write(folder / "removes.csv", removes)
    rs._write(folder / "mended_lines.csv", lines)
    new = [a for a in adds if not a["in_route"]]
    summary = {"module": __version__, "edition": edition, "pages": pages,
               "run": time.strftime("%Y-%m-%d %H:%M:%S"),
               "lines_mended": dict(Counter(x["reader"] for x in lines)),
               "entries_changed": len({tuple(a[k] for k in CLAIM[:4]) for a in adds}
                                      | {tuple(r[k] for k in CLAIM[:4]) for r in removes}),
               "adds_new_to_route": dict(Counter(a["status"] for a in new)),
               "adds_already_in_route": sum(a["in_route"] for a in adds),
               "removes": len(removes),
               "pages_changed": len({a["page"] for a in new} | {r["page"] for r in removes})}
    (folder / "summary.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return summary

def items(edition: str, out: Path = OUT) -> list[dict]:
    """One Opus call per part with a disputed add the route lacks: *is X printed for this part?*,
    shown the page's offices sections of the court (its parts sections where it has none), with the
    route's part-0 rule on (D-197: notebook 17's context had it off)."""
    from utils.recheck import opus_pass as op

    path = Path(out) / edition / SUB / "consensus.csv"
    by_part = defaultdict(set)
    for c in (_read(path) if path.exists() else []):
        if c["status"] == "disputed" and c["in_route"] != "True":
            by_part[tuple(c[k] for k in CLAIM[:4])].add(c["code"])
    if not by_part:
        return []
    ctx = dict(op._context(edition), part0_named=True)
    sections = rf.out_dir(edition, PAGES) / "sections"
    calls = []
    for part, tokens in sorted(by_part.items()):
        images = (sorted(sections.glob(f"{part[0]}_{part[1]}_offices_*.png"))
                  or sorted(sections.glob(f"{part[0]}_{part[1]}_parts_*.png")))
        calls.append(op.build_call("k-" + tb.unit_id(part), part, [], sorted(tokens), ctx,
                                   [(p.name, "0", "") for p in images]))
    return calls

    entries the mending changes for any reader. Returns ``(adds, removes)``: the mended claims with
    their status and whether the route holds them; the route's claims no mended decode makes."""
    touched = set().union(*({c[:4] for c in base[label] ^ mended[label]} for label in base))
    every = touched & set.intersection(*({c[:4] for c in cs} for cs in mended.values()))
    shared = {label: {c for c in cs if c[:4] in every} for label, cs in mended.items()}
    union, agreed = set().union(*shared.values()), set.intersection(*shared.values())
    adds = [dict(zip(CLAIM, c), status="agreed" if c in agreed else "disputed",
                 support=sum(c in s for s in shared.values()), in_route=c in route) for c in sorted(union)]
    alone = {c for c in mended[record] if c[:4] in touched - every}
    adds += [dict(zip(CLAIM, c), status="reader of record", support=1, in_route=c in route)
             for c in sorted(alone)]
    made, was = set().union(*mended.values()), set().union(*base.values())
    removes = [dict(zip(CLAIM, c)) for c in sorted(route) if c[:4] in touched and c in was and c not in made]
    return adds, removes


def route_claims(edition: str, pages: str = PAGES) -> set:
    run = rf.out_dir(edition, pages)
    return {rs._key(r) for name in ("final_inventory.csv", "final_abbreviations.csv")
            if (run / name).exists() for r in _read(run / name)}


def run(edition: str, pages: str = PAGES, out: Path = OUT) -> dict:
    """Every reader of one German volume; writes under ``<out>/<edition>/keys/``: each reader's
    mended inventory (``inventory_<label>.csv``, whose context carries the cooperatives' language
    letter for the merge), ``consensus.csv`` (the adds), ``removes.csv``, ``mended_lines.csv`` and
    ``summary.json``."""
    if edition not in GERMAN:
        raise ValueError(f"{edition}: the Czech readers write the key with a comma; nothing to mend")
    folder = Path(out) / edition / SUB
    folder.mkdir(parents=True, exist_ok=True)
    labels = rs.readers(edition, pages)
    base, mended, lines = {}, {}, []
    for label in labels:
        b, m, mended_lines = reader_decodes(edition, label)
        base[label] = {rs._key(r) for r in b if r.get("code")}
        mended[label] = {rs._key(r) for r in m if r.get("code")}
        rs._write(folder / f"inventory_{label}.csv", m)
        lines += [dict(x, reader=label) for x in mended_lines]
    adds, removes = settle(base, mended, labels[0], route_claims(edition, pages))
    rs._write(folder / "consensus.csv", adds)
    rs._write(folder / "removes.csv", removes)
    rs._write(folder / "mended_lines.csv", lines)
    new = [a for a in adds if not a["in_route"]]
    summary = {"module": __version__, "edition": edition, "pages": pages,
               "run": time.strftime("%Y-%m-%d %H:%M:%S"),
               "lines_mended": dict(Counter(x["reader"] for x in lines)),
               "entries_changed": len({tuple(a[k] for k in CLAIM[:4]) for a in adds}
                                      | {tuple(r[k] for k in CLAIM[:4]) for r in removes}),
               "adds_new_to_route": dict(Counter(a["status"] for a in new)),
               "adds_already_in_route": sum(a["in_route"] for a in adds),
               "removes": len(removes),
               "pages_changed": len({a["page"] for a in new} | {r["page"] for r in removes})}
    (folder / "summary.json").write_text(json.dumps(summary, indent=1), encoding="utf-8")
    return summary


def items(edition: str, out: Path = OUT) -> list[dict]:
    """One Opus call per part with a disputed add the route lacks: *is X printed for this part?*,
    shown the page's offices sections of the court (its parts sections where it has none), with the
    route's part-0 rule on (D-197: notebook 17's context had it off)."""
    from utils.recheck import opus_pass as op

    path = Path(out) / edition / SUB / "consensus.csv"
    by_part = defaultdict(set)
    for c in (_read(path) if path.exists() else []):
        if c["status"] == "disputed" and c["in_route"] != "True":
            by_part[tuple(c[k] for k in CLAIM[:4])].add(c["code"])
    if not by_part:
        return []
    ctx = dict(op._context(edition), part0_named=True)
    sections = rf.out_dir(edition, PAGES) / "sections"
    calls = []
    for part, tokens in sorted(by_part.items()):
        images = (sorted(sections.glob(f"{part[0]}_{part[1]}_offices_*.png"))
                  or sorted(sections.glob(f"{part[0]}_{part[1]}_parts_*.png")))
        calls.append(op.build_call("k-" + tb.unit_id(part), part, [], sorted(tokens), ctx,
                                   [(p.name, "0", "") for p in images]))
    return calls


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--edition", default="de-moravia", choices=GERMAN)
    ap.add_argument("--out", default=str(OUT))
    args = ap.parse_args(argv)
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    print(run(args.edition, out=Path(args.out)))
    return 0


if __name__ == "__main__":
