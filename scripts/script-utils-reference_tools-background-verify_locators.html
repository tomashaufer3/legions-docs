"""Check every ledger entry's locator against its slice (scratch tool for the reading pass).

usage: python verify_locators.py <evidence dir> <slices dir> [slice key ...]
A locator passes if it occurs in the slice on the page the entry cites (whitespace-
normalised); 'elsewhere' if it occurs on another page; 'absent' if nowhere.
"""
import re
import sys
from collections import Counter
from pathlib import Path

ev, sl = Path(sys.argv[1]), Path(sys.argv[2])
keys = sys.argv[3:]


def norm(s):
    return re.sub(r"\s+", " ", s).strip()


def pages(slice_text):
    parts = re.split(r"^===== (.+?) =====\s*$", slice_text, flags=re.M)
    return {parts[i].split("|")[0].strip(): norm(parts[i + 1]) for i in range(1, len(parts) - 1, 2)}


files = sorted(list(ev.glob("parts/*.md")) + list(ev.glob("*.md")))
total = Counter()
for key in sorted({f.name.split(".")[0] for f in files if f.name != "READER_BRIEF.md"}):
    if keys and key not in keys:
        continue
    src = sl / f"{key}.txt"
    if not src.exists():
        continue
    pg = pages(src.read_text(encoding="utf-8"))
    whole = " ".join(pg.values())
    text = "\n".join(f.read_text(encoding="utf-8") for f in files if f.name.split(".")[0] == key)
    c = Counter()
    bad = []
    for block in re.split(r"^### ", text, flags=re.M)[1:]:
        loc = re.search(r"^- locator:\s*(.+)$", block, flags=re.M)
        page = re.search(r"^- page:\s*(.+)$", block, flags=re.M)
        if not loc:
            c["no locator"] += 1
            continue
        L = norm(loc.group(1)).strip("\"'`“”„")
        marker = re.search(r"(PAGE \d+|PDF PAGE \d+|SECTION \d+)", page.group(1) if page else "")
        on_page = pg.get(marker.group(1), "") if marker else ""
        if len(L.split()) > 9:
            c["locator too long"] += 1
        if L and L in on_page:
            c["on page"] += 1
        elif L and L in whole:
            c["elsewhere"] += 1
        else:
            c["absent"] += 1
            bad.append((marker.group(1) if marker else "?", L))
    total += c
    print(f"{key:14s} {dict(c)}")
    for m, L in bad[:6]:
        print(f"      absent: [{m}] {L}")
print("TOTAL", dict(total))
